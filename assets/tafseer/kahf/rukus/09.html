پچھلا رکوع اس اعلان پر ختم ہوا تھا کہ بستیاں اپنے مقررہ وقت پر ہلاک ہوئیں، اور اب سورت ایک سفر کی طرف مڑتی ہے جو سورۂ کہف کا مرکز ہے۔ وَاِذْ قَالَ مُوْسٰى لِفَتٰىهُ لَاۤ اَبْرَحُ حَتّٰۤى اَبْلُغَ مَجْمَعَ الْبَحْرَيْنِ اَوْ اَمْضِيَ حُقُبًا۔ "اور جب موسیٰ نے اپنے نوجوان رفیق سے کہا: میں چلتا رہوں گا یہاں تک کہ دو سمندروں کے ملنے کی جگہ تک پہنچ جاؤں، یا زمانوں تک چلتا رہوں۔" حدیث کے مفہوم کے مطابق اس سفر کا سبب یہ تھا کہ حضرت موسیٰ علیہ السلام سے پوچھا گیا کہ لوگوں میں سب سے بڑا عالم کون ہے، اور حق تعالیٰ نے چاہا کہ علم کی نسبت اسی کی طرف لوٹائی جائے؛ چنانچہ انہیں اپنے ایک بندے کا پتا دیا گیا جو دو سمندروں کے سنگم پر تھا۔ یہ کسی کمی کی سرزنش نہیں بلکہ ایک اولوالعزم رسول کی تربیت تھی، اور سارا قصہ اسی ادب پر ختم ہوگا کہ فعل اور علم کی نسبت کس کی طرف کی جائے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں دو سمندر علمِ ظاہر اور علمِ باطن ہیں، شریعت اور حقیقت، اور ان کا سنگم وہ برزخ ہے جہاں دونوں ملتے ہیں مگر ایک دوسرے پر چڑھ نہیں آتے، جیسا کہ دوسری جگہ فرمایا: مَرَجَ الْبَحْرَيْنِ يَلْتَقِيٰنِ ۙ بَيْنَهُمَا بَرْزَخٌ لَّا يَبْغِيٰنِ۔ اہلِ معرفت کے نزدیک اس برزخ کی کامل صورت انسانِ کامل ہے، جس میں نہ ظاہر باطن کو باطل کرتا ہے نہ باطن ظاہر کو۔ "لا ابرح" میں طالب کا عزم ہے اور "حقبا" میں یہ اقرار کہ مطلوب کی خاطر زمانے خرچ ہو جائیں تو بھی سودا سستا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ سفر حقیقی تھا اور سمندر بھی حقیقی۔

پھر سفر کا وہ لمحہ آتا ہے جس پر سب کچھ ٹکا ہوا تھا: فَلَمَّا بَلَغَا مَجْمَعَ بَيْنِهِمَا نَسِيَا حُوْتَهُمَا فَاتَّخَذَ سَبِيْلَهٗ فِي الْبَحْرِ سَرَبًا۔ "پھر جب وہ دونوں ان کے ملنے کی جگہ پہنچے تو اپنی مچھلی بھول گئے، اور اس نے سمندر میں سرنگ کی طرح اپنی راہ بنا لی۔" روایات کے مطابق یہی مچھلی نشانی ٹھہرائی گئی تھی کہ جہاں یہ گم ہو وہیں مطلوب بندہ ملے گا۔ ٹوکری میں رکھی بے جان مچھلی دو سمندروں کے سنگم پر زندہ ہو کر پانی میں اتر گئی۔ اہلِ تاویل کہتے ہیں کہ جہاں ظاہر اور باطن ملتے ہیں وہاں مردہ دل زندہ ہو اٹھتا ہے اور اپنی اصل کی طرف، جو سمندر ہے، لوٹ جاتا ہے۔ شیخِ اکبر کے نزدیک کائنات کی ہر شے میں حیات ہے اور ہر شے اپنی اصل کی طرف رواں ہے؛ مچھلی کی راہ "سربا" تھی، ایک پوشیدہ سرنگ، کیونکہ حیات کا یہ سفر آنکھ سے اوجھل رہتا ہے۔ اس کے بعد فرمایا: فَلَمَّا جَاوَزَا قَالَ لِفَتٰىهُ اٰتِنَا غَدَآءَنَا لَقَدْ لَقِيْنَا مِنْ سَفَرِنَا هٰذَا نَصَبًا۔ "جب وہ آگے نکل گئے تو موسیٰ نے اپنے رفیق سے کہا: ہمارا کھانا لاؤ، اس سفر میں ہمیں بڑی تھکن پہنچی ہے۔" مفسرین نے یہ لطیف نکتہ لکھا ہے کہ تھکن منزل سے آگے بڑھ جانے کے بعد ہی محسوس ہوئی؛ جب تک قدم مطلوب کی سمت تھے سفر ہلکا تھا۔ اہلِ ذوق کہتے ہیں کہ سالک پر بوجھ اسی وقت اترتا ہے جب وہ اپنے مقام سے آگے نکل جائے۔ یوں "نصب" بھی ایک رہنما ہے: تھکن بتاتی ہے کہ کچھ پیچھے رہ گیا ہے، اور بھوک اسی جگہ کی یاد دلاتی ہے جسے بندہ بے خبری میں چھوڑ آیا۔

رفیق کو یاد آیا اور اس نے عرض کیا: قَالَ اَرَءَيْتَ اِذْ اَوَيْنَاۤ اِلَى الصَّخْرَةِ فَاِنِّيْ نَسِيْتُ الْحُوْتَ وَمَاۤ اَنْسٰىنِيْهُ اِلَّا الشَّيْطٰنُ اَنْ اَذْكُرَهٗ ۚ وَاتَّخَذَ سَبِيْلَهٗ فِي الْبَحْرِ ۖ عَجَبًا۔ "اس نے کہا: کیا آپ نے دیکھا، جب ہم اس چٹان کے پاس ٹھہرے تھے تو میں مچھلی کو بھول گیا، اور اس کا ذکر کرنا مجھے شیطان ہی نے بھلایا، اور اس نے عجیب طرح سمندر میں اپنی راہ لی۔" یہ رفیق حضرت یوشع بن نون علیہ السلام تھے۔ ان کے جملے میں ادبِ نسبت کی پہلی جھلک ہے جو آگے چل کر اس قصے کا تاج بنے گی: بھول کو نہ حق کی طرف منسوب کیا نہ اپنے استاد کی طرف، بلکہ شیطان کی طرف۔ شیطان کا بس اتنا ہی ہے کہ ایک لمحے کی غفلت ڈال دے؛ سورۂ ابراہیم میں اس کا اپنا اقرار گزر چکا کہ اسے کسی پر زور حاصل نہیں۔ "اوینا" وہی فعل ہے جو اصحابِ کہف کے لیے آیا تھا: غار کی پناہ اور چٹان کی پناہ ایک ہی دروازے کے دو نام ہیں۔ حضرت موسیٰ علیہ السلام نے سنا تو فرمایا: قَالَ ذٰلِكَ مَا كُنَّا نَبْغِ ۖ فَارْتَدَّا عَلٰۤى اٰثَارِهِمَا قَصَصًا۔ "کہا: یہی تو وہ ہے جس کی ہمیں تلاش تھی؛ پس وہ دونوں اپنے قدموں کے نشان دیکھتے ہوئے واپس لوٹے۔" اہلِ معرفت کے نزدیک جو شے گم ہوئی وہی نشانی تھی؛ مطلوب اسی جگہ ہے جہاں بندہ کچھ کھو بیٹھا۔ اسی لیے یہ خبر حضرت موسیٰ علیہ السلام کے لیے رنج نہیں، بشارت تھی۔ "قصص" کے معنی نشانِ قدم کا پیچھا کرنا ہیں، اور یہی لفظ قصہ گوئی کا بھی ہے؛ گویا ہر سچا قصہ ایک واپسی ہے۔ سالک کی توبہ بھی یہی ہے کہ اپنے قدموں کے نشان پر لوٹے اور وہاں پہنچے جہاں دل غافل ہوا تھا۔

اور وہاں وہ ملے جن کی تلاش تھی: فَوَجَدَا عَبْدًا مِّنْ عِبَادِنَاۤ اٰتَيْنٰهُ رَحْمَةً مِّنْ عِنْدِنَا وَعَلَّمْنٰهُ مِنْ لَّدُنَّا عِلْمًا۔ "تو انہوں نے ہمارے بندوں میں سے ایک بندے کو پایا جسے ہم نے اپنے پاس سے رحمت عطا کی تھی اور اپنی خاص جناب سے ایک علم سکھایا تھا۔" غور کیجیے، ان کا پہلا وصف "عبد" ہے؛ نام نہیں لیا گیا، بندگی بتائی گئی۔ سورۂ بنی اسرائیل میں اسریٰ کا ذکر بھی "بعبدہٖ" سے ہوا تھا، اور اسی سورت کا آغاز بھی کتاب کے اپنے عبد پر اترنے سے ہوا۔ اس مکتب میں عبدیت ہر علم کا دروازہ ہے؛ جو بندہ نہیں وہ عطا کا ظرف نہیں۔ پھر رحمت کو علم سے پہلے رکھا: جو علم رحمت سے نہ نکلے وہ خشک رہتا ہے، اور جو رحمت کے برتن میں اترے وہی شفا بنتا ہے۔ "من عندنا" اور "من لدنا" میں بھی فرق ہے؛ اہلِ معرفت کے نزدیک "لدن" زیادہ قرب اور بے واسطگی کا لفظ ہے۔ اسی سے علمِ لدنی کی اصطلاح بنی: وہ علم جو پڑھنے، سوچنے اور استدلال سے حاصل نہیں ہوتا بلکہ براہِ راست حق کی جناب سے دل پر وارد ہوتا ہے۔ یاد رہے کہ اصحابِ کہف نے بھی یہی مانگا تھا: رَبَّنَاۤ اٰتِنَا مِنْ لَّدُنْكَ رَحْمَةً۔ غار کی دعا اور سنگم کی عطا ایک ہی جناب سے ہیں۔ حدیث کے مفہوم کے مطابق انہیں خضر اس لیے کہا گیا کہ وہ خشک زمین پر بیٹھے تو وہ سرسبز ہو اٹھی؛ اہلِ ذوق کہتے ہیں کہ جس دل پر "لدن" کا علم اترے وہ جہاں بیٹھے زندگی بانٹتا ہے۔

اب اس رکوع کا سب سے روشن منظر آتا ہے، ایک عظیم رسول کا طالب بن کر سوال کرنا: قَالَ لَهٗ مُوْسٰى هَلْ اَتَّبِعُكَ عَلٰۤى اَنْ تُعَلِّمَنِ مِمَّا عُلِّمْتَ رُشْدًا۔ "موسیٰ نے ان سے کہا: کیا میں اس شرط پر آپ کے ساتھ رہوں کہ جو ہدایت کا علم آپ کو سکھایا گیا ہے اس میں سے کچھ مجھے بھی سکھا دیں؟" حکم نہیں، سوال؛ "اتبعک" میں پیروی کی عاجزی؛ "مما" میں یہ اقرار کہ سب نہیں، کچھ؛ اور سب سے لطیف "عُلِّمْتَ" کا صیغہ: یہ نہیں کہا کہ جو آپ جانتے ہیں، بلکہ جو آپ کو سکھایا گیا۔ یعنی جس ادبِ نسبت کے لیے یہ سفر ہوا تھا وہ پہلے ہی جملے میں ادا ہو گیا۔ "رشدا" وہی لفظ ہے جو غار والوں کی دعا اور "ان شاء اللہ" کی آیت میں آیا تھا۔ یہاں ادب کا ایک اصول صاف رہنا چاہیے: حضرت موسیٰ علیہ السلام کلیم اللہ، اولوالعزم رسول اور صاحبِ شریعت ہیں، اور ان کا مرتبہ حضرت خضر علیہ السلام سے بلند ہے۔ حضرت خضر علیہ السلام کا علم ایک دوسری نوع کا علم تھا، اونچا درجہ نہیں۔ حدیث کے مفہوم کے مطابق خود انہوں نے فرمایا کہ آپ کے پاس اللہ کا دیا ہوا ایک علم ہے جو میں نہیں جانتا، اور میرے پاس ایک علم ہے جو آپ نہیں جانتے۔ جمہور کے نزدیک حضرت خضر علیہ السلام نبی ہیں، اور بعض نے انہیں ولی کہا ہے؛ دونوں صورتوں میں یہ طے ہے کہ کوئی ولی کسی نبی کے مرتبے کو نہیں پہنچتا۔ شیخِ اکبر کے ہاں جہاں ولایت کی فضیلت کا ذکر آتا ہے وہاں مراد نبی کی اپنی ولایت ہے، نہ یہ کہ کوئی ولی نبی سے بڑھ جائے۔ فصوص الحکم میں حضرت موسیٰ علیہ السلام کا باب حکمتِ علویہ کے نام سے ہے، اور شیخِ اکبر نے فتوحاتِ مکیہ میں حضرت خضر علیہ السلام سے اپنی ملاقات کا ذکر بھی کیا ہے۔ بڑے کا ایک خاص پہلو میں چھوٹے سے سیکھنا اس کی پستی نہیں، کمالِ ادب ہے۔

جواب میں حضرت خضر علیہ السلام نے ایک حقیقت بیان کی: قَالَ اِنَّكَ لَنْ تَسْتَطِيْعَ مَعِيَ صَبْرًا، وَكَيْفَ تَصْبِرُ عَلٰى مَا لَمْ تُحِطْ بِهٖ خُبْرًا۔ "کہا: آپ میرے ساتھ ہرگز صبر نہ کر سکیں گے، اور آپ اس بات پر صبر کر بھی کیسے سکتے ہیں جس کی حقیقت آپ کے احاطۂ علم میں نہیں۔" یہاں صبر کو علم سے باندھ دیا گیا: جو شے سمجھ میں نہ آئے اس پر دل ٹھہر نہیں سکتا۔ "خُبر" محض خبر نہیں، اندر سے چکھی ہوئی آگاہی ہے، جسے اہلِ تصوف ذوق کہتے ہیں۔ یہ جملہ حضرت موسیٰ علیہ السلام کے حلم پر کوئی حرف نہیں؛ ان کے اعتراضات بے صبری کی خو نہیں بلکہ اپنے علم سے وفاداری تھے۔ شریعت ظاہری ظلم پر خاموش نہیں رہ سکتی، اور یہی اس کی شان ہے۔ شیخِ اکبر کے نزدیک ہر بندے کا ایک ربِّ خاص ہے، یعنی وہ اسمِ الٰہی جس کے سائے میں وہ پلتا ہے، اور ہر اسم کا اپنا حکم ہے۔ جو شریعت کے اسم کا امین ہو وہ دوسرے اسم کے حکم پر اسی کی زبان سے بول نہیں سکتا۔ "معی" کا لفظ بتاتا ہے کہ بات صحبت کی ہے: میری راہ میں میرے ساتھ چلنا آپ کے منصب کی راہ نہیں۔ اہلِ معرفت کہتے ہیں کہ یہاں کمی کسی ایک طرف نہیں، دو سمندر ہیں؛ اسی لیے سنگم پر ملاقات ہوئی، اور اسی لیے یہ ملاقات عارضی رہی۔ یہ حیرت کا مقام ہے، جہاں بندہ جان لیتا ہے کہ حق کا علم کسی ایک ظرف میں سما نہیں سکتا۔

حضرت موسیٰ علیہ السلام نے ادب سے وعدہ کیا: قَالَ سَتَجِدُنِيْۤ اِنْ شَآءَ اللّٰهُ صَابِرًا وَّلَاۤ اَعْصِيْ لَكَ اَمْرًا۔ "کہا: ان شاء اللہ آپ مجھے صابر پائیں گے، اور میں کسی بات میں آپ کی نافرمانی نہ کروں گا۔" چند رکوع پہلے حضورِ اکرم صلی اللہ علیہ وسلم کو سکھایا گیا تھا کہ کل کے کسی کام کو "ان شاء اللہ" کے بغیر نہ کہیں؛ یہاں ایک جلیل القدر رسول اسی ادب پر عمل کرتے دکھائی دیتے ہیں۔ بندہ کل کا مالک نہیں، اور صبر بھی عطا ہے۔ اہلِ معرفت کہتے ہیں کہ انہوں نے صبر کو مشیتِ الٰہی سے مشروط کیا، سو جو آگے ہوا، جدائی سمیت، اسی مشیت کے اندر ہوا۔ حدیث کے مفہوم کے مطابق حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ اللہ موسیٰ پر رحم فرمائے، ہم چاہتے تھے کہ وہ کچھ اور صبر کرتے تاکہ ہمیں ان دونوں کی مزید باتیں سنائی جاتیں۔ اس کے بعد حضرت خضر علیہ السلام نے شرط رکھی: قَالَ فَاِنِ اتَّبَعْتَنِيْ فَلَا تَسْـَٔلْنِيْ عَنْ شَيْءٍ حَتّٰۤى اُحْدِثَ لَكَ مِنْهُ ذِكْرًا۔ "کہا: اگر آپ میرے ساتھ چلتے ہیں تو مجھ سے کسی چیز کے بارے میں نہ پوچھیں جب تک میں خود آپ سے اس کا ذکر نہ کروں۔" یہ مرید کا ادب ہے: اسرار کا علم کھینچا نہیں جاتا، اپنے وقت پر دیا جاتا ہے۔ "احدث" کا لفظ نئی بات پیدا کرنے کا ہے؛ شیخِ اکبر کے نزدیک ہر معرفت ایک نئی تجلی ہے، تجددِ امثال کی طرح ہر لمحہ تازہ، کسی گودام میں رکھی پرانی چیز نہیں۔ البتہ یہ شرط اسرار کی راہ کی ہے؛ احکامِ شریعت میں سوال کا دروازہ ہمیشہ کھلا ہے، اور وہاں پوچھنا ہی ادب ہے۔
