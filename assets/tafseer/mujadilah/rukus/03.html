اس رکوع کی ابتدا ایک ایسے گروہ کی تصویر سے ہوتی ہے جو دو کشتیوں میں پاؤں رکھے ہوئے ہے: اَلَمْ تَرَ اِلَى الَّذِيْنَ تَوَلَّوْا قَوْمًا غَضِبَ اللّٰهُ عَلَيْهِمْ ۭ مَا هُمْ مِّنْكُمْ وَلَا مِنْهُمْ ۙ وَيَحْلِفُوْنَ عَلَى الْكَذِبِ وَهُمْ يَعْلَمُوْنَ۔ کیا آپ نے ان لوگوں کو نہیں دیکھا جنہوں نے ایسی قوم کو دوست بنایا جس پر اللہ کا غضب ہے؟ وہ نہ تم میں سے ہیں نہ ان میں سے، اور جانتے بوجھتے جھوٹ پر قسمیں کھاتے ہیں۔ ظاہر میں یہ مدینہ کے منافقین کا بیان ہے جو مسلمانوں کے ساتھ بیٹھتے اور ان کے دشمنوں سے خفیہ دوستی رکھتے تھے، اور پچھلے رکوع کی سرگوشیوں کا سلسلہ یہیں آ کر کھلتا ہے۔ اہلِ معرفت "مَا هُمْ مِّنْكُمْ وَلَا مِنْهُمْ" پر رکتے ہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں برزخ وہ ہے جو دو چیزوں کے بیچ ہو اور دونوں کا حق ادا کرے؛ انسانِ کامل حق اور خلق کے بیچ ایسا ہی برزخ ہے کہ دونوں جانب اسی سے جڑتی ہیں۔ منافق کا بیچ میں ہونا اس کا الٹ ہے: وہ دونوں سے جڑا نہیں، دونوں سے کٹا ہوا ہے۔ ایک برزخ جامع ہے، دوسرا خالی دراڑ۔ اور "وَهُمْ يَعْلَمُوْنَ" اس سارے رکوع کی کنجی رکھ دیتا ہے: ان کا جھوٹ بے خبری کا نہیں، جان کر اختیار کیا ہوا رخ ہے۔ جس گمراہی کا ذکر آگے آئے گا، اس کی بنیاد یہیں بندے کے اپنے علم اور اپنے انتخاب پر رکھ دی گئی؛ حق نے کسی کو اس رخ پر دھکیلا نہیں۔

پھر انجام کا اعلان ہے: اَعَدَّ اللّٰهُ لَهُمْ عَذَابًا شَدِيْدًا ۭ اِنَّهُمْ سَآءَ مَا كَانُوْا يَعْمَلُوْنَ۔ اللہ نے ان کے لیے سخت عذاب تیار کر رکھا ہے؛ بے شک بہت برا ہے جو وہ کرتے تھے۔ اِتَّخَذُوْۤا اَيْمَانَهُمْ جُنَّةً فَصَدُّوْا عَنْ سَبِيْلِ اللّٰهِ فَلَهُمْ عَذَابٌ مُّهِيْنٌ۔ انہوں نے اپنی قسموں کو ڈھال بنا لیا، پھر اللہ کی راہ سے روکا، سو ان کے لیے رسوا کن عذاب ہے۔ غور کیجیے کہ عذاب کے ذکر کے فوراً بعد سبب بتا دیا گیا: "سَآءَ مَا كَانُوْا يَعْمَلُوْنَ"۔ سزا باہر سے آ کر نہیں چپکی، ان کے اپنے عمل سے اٹھی۔ اس مکتب میں اعمال آخرت میں صورتیں اختیار کرتے ہیں؛ دنیا میں جو عمل معنی تھا وہاں پیکر بن کر سامنے آتا ہے، اور ہر شخص اپنے ہی کیے کی صورت سے ملتا ہے۔ "جُنَّةً" ڈھال ہے، اور اہلِ لغت بتاتے ہیں کہ اس مادے میں چھپانے اور ڈھانپنے کا معنی ہے؛ جنت گھنے باغ کی چھاؤں ہے اور جنین ماں کے پیٹ میں چھپا ہوا بچہ۔ اہلِ ذوق کہتے ہیں کہ منافق نے زبان کو ڈھال بنایا تاکہ دل چھپا رہے، مگر جو ڈھال حق سے بچنے کے لیے اٹھائی جائے وہ خود سب سے بڑا حجاب بن جاتی ہے۔ مومن کی ڈھال اس کے برعکس ہے: حدیث کا مفہوم ہے کہ روزہ ڈھال ہے؛ وہ ڈھال نفس سے بچاتی ہے، حق سے نہیں۔ اور جس نے اپنی قسم کو پردہ بنایا وہ دوسروں کا راستہ بھی کاٹنے لگا: "فَصَدُّوْا"۔ پہلے اپنا رخ پھیرا، پھر اوروں کو روکا۔ اسی لیے عذاب "مُّهِيْنٌ" ہے: جس نے حق کی عزت کو ٹھکرا کر اپنی جھوٹی عزت بچائی، اس کا بدلہ رسوائی ہے۔

اب ان سہاروں کا انجام سنیے جن پر وہ بھروسا کیے بیٹھے تھے: لَنْ تُغْنِيَ عَنْهُمْ اَمْوَالُهُمْ وَلَاۤ اَوْلَادُهُمْ مِّنَ اللّٰهِ شَيْئًا ۭ اُولٰۤئِكَ اَصْحٰبُ النَّارِ ۭ هُمْ فِيْهَا خٰلِدُوْنَ۔ اللہ کے مقابلے میں نہ ان کے مال ان کے کچھ کام آئیں گے نہ ان کی اولاد؛ یہی دوزخ والے ہیں، اس میں ہمیشہ رہیں گے۔ يَوْمَ يَبْعَثُهُمُ اللّٰهُ جَمِيْعًا فَيَحْلِفُوْنَ لَهٗ كَمَا يَحْلِفُوْنَ لَكُمْ وَيَحْسَبُوْنَ اَنَّهُمْ عَلٰى شَيْءٍ ۭ اَلَاۤ اِنَّهُمْ هُمُ الْكٰذِبُوْنَ۔ جس دن اللہ ان سب کو اٹھائے گا تو وہ اس کے سامنے بھی ویسی ہی قسمیں کھائیں گے جیسی تمہارے سامنے کھاتے ہیں، اور سمجھیں گے کہ وہ کسی بنیاد پر ہیں؛ خبردار، وہی جھوٹے ہیں۔ شیخِ اکبر کے نزدیک اسباب حق کے پردے ہیں، اور ان میں جو اثر دکھائی دیتا ہے وہ حق ہی کا ہے؛ جو سبب کو حق کے مقابل کھڑا کر دے، اس کا سبب اس کے ہاتھ میں خالی رہ جاتا ہے۔ اسی لیے "مِنَ اللّٰهِ" کی قید آئی: مال اور اولاد نعمت ہیں، مگر اللہ کے مقابلے میں کچھ نہیں۔ پھر ایک حیرت انگیز منظر ہے: حشر میں بھی وہی قسمیں۔ حدیث کا مفہوم ہے کہ ہر بندہ اسی حال پر اٹھایا جائے گا جس پر وہ مرا؛ جھوٹ ان کی عادت نہیں رہا، ان کی صورت بن گیا۔ اور "وَيَحْسَبُوْنَ اَنَّهُمْ عَلٰى شَيْءٍ" پر اہلِ ذوق ٹھہرتے ہیں۔ ظاہر معنی یہ ہے کہ وہ سمجھیں گے کہ ان کے پاس کوئی بنیاد ہے؛ اور اشارہ یہ کہ منافق کی اصل بیماری یہ گمان ہے کہ وہ خود کچھ ہے۔ اس مکتب کا معروف مفہوم ہے کہ اعیانِ ثابتہ نے اپنی ذات میں وجود کی خوشبو تک نہیں سونگھی؛ ہر شے کا وجود حق سے مستعار ہے۔ عارف اپنے آپ کو کچھ نہ جان کر فقر میں سچا ہوتا ہے، اور منافق اپنے آپ کو کچھ سمجھ کر جھوٹا۔ اسی لیے فیصلہ یہ ہے: "اَلَاۤ اِنَّهُمْ هُمُ الْكٰذِبُوْنَ"؛ ان کا سب سے بڑا جھوٹ وہ تھا جو انہوں نے اپنی ہستی کے بارے میں خود سے بولا۔

اب وہ آیت آتی ہے جو اس رکوع کا، بلکہ ہدایت و گمراہی کے پورے باب کا اصول ہے: اِسْتَحْوَذَ عَلَيْهِمُ الشَّيْطٰنُ فَاَنْسٰىهُمْ ذِكْرَ اللّٰهِ۔ شیطان ان پر قابو پا گیا اور اس نے انہیں اللہ کی یاد بھلا دی۔ اُولٰۤئِكَ حِزْبُ الشَّيْطٰنِ ۭ اَلَاۤ اِنَّ حِزْبَ الشَّيْطٰنِ هُمُ الْخٰسِرُوْنَ۔ یہی شیطان کی جماعت ہیں؛ خبردار، شیطان کی جماعت ہی خسارہ اٹھانے والی ہے۔ "اِسْتَحْوَذَ" کے مادے میں اونٹوں کو گھیر کر تیزی سے ہانکنے کا معنی ہے؛ یعنی شیطان نے انہیں ہر طرف سے گھیرا اور جدھر چاہا ہانک لیا۔ مگر سوال یہ ہے کہ اسے یہ گھیرا ملا کیسے؟ آیت خود جواب کی طرف اشارہ کرتی ہے: اس کا ہتھیار تلوار نہیں، فراموشی ہے۔ اس نے انہیں مارا نہیں، بھلایا۔ اور بھلایا اسی کو جا سکتا ہے جو یاد سے پہلے ہی منہ موڑ چکا ہو۔ پچھلی آیات پر نظر ڈالیے: انہوں نے خود غضب والی قوم کو دوست بنایا، خود جان بوجھ کر جھوٹی قسمیں کھائیں، خود قسموں کو ڈھال بنایا۔ ہر قدم ان کا اپنا تھا۔ پس شیطان کا غلبہ ابتدا نہیں، نتیجہ ہے۔ اہلِ معرفت کہتے ہیں کہ ذکرِ الٰہی کو بھولنا ہی شیطان کی سلطنت کا دروازہ ہے؛ جب تک یہ دروازہ بند ہے، وہ باہر کھڑا وسوسے ڈالتا رہتا ہے مگر اندر حکم نہیں چلا سکتا۔ اور یہ دروازہ کوئی باہر سے نہیں کھولتا؛ بندے کا اپنا منہ پھیرنا اس کی کنڈی کھولتا ہے۔ اسی لیے یہ آیت محض ایک گروہ کا حال نہیں، ایک قاعدہ ہے جو ہر دل پر یکساں جاری ہے: جہاں یاد ہے وہاں شیطان مہمان بھی نہیں، اور جہاں یاد گئی وہاں وہ مالک بن بیٹھتا ہے۔

یہ اصول قرآن نے دوسری جگہوں پر بھی کھول کر بیان کیا ہے۔ سورۂ الزخرف میں ہم پڑھ چکے: وَمَنْ يَّعْشُ عَنْ ذِكْرِ الرَّحْمٰنِ نُقَيِّضْ لَهٗ شَيْطٰنًا فَهُوَ لَهٗ قَرِيْنٌ۔ ترتیب دیکھیے: پہلے بندہ رحمٰن کے ذکر سے آنکھ چراتا ہے، پھر شیطان اس کا ساتھی بنتا ہے۔ اور سورۂ ابراہیم میں شیطان نے خود اقرار کیا کہ اس کا کوئی زور نہ تھا اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ۔ دعوت اس کی تھی، قبول کرنا بندے کا۔ ان آیتوں کو ملا کر یہ حقیقت سامنے آتی ہے کہ حق کسی کو زبردستی گمراہ نہیں کرتا، نہ شیطان کو کسی پر جبر کا اختیار دیتا ہے؛ اور جو اس کی یاد طلب کرے اسے وہ کبھی فراموشی کے حوالے نہیں کرتا۔ جو ذکر سے منہ موڑے، اس کا منہ موڑنا ہی اس کی گمراہی ہے، اور حق اس رخ کو وجود دے دیتا ہے جو بندے نے خود چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے؛ حق ہر شے پر وہی حکم جاری فرماتا ہے جو اس شے کی اپنی حقیقت اور اپنے رخ کا تقاضا ہو۔ اگر وہ چاہتا تو سب کو مجبور کر کے ہدایت پر لے آتا؛ اس نے ایسا نہیں چاہا، اور یہی انسان کی عزت ہے کہ اسے اختیار کی امانت سونپی گئی۔ "فَاَنْسٰىهُمْ" میں بھلانے کی نسبت شیطان کی طرف ہے، اور شیطان کی رسائی صرف اس دل تک ہے جس نے اپنا دروازہ خود کھولا۔ یوں فراموشی ایک دائرہ بن جاتی ہے: پہلی غفلت بندے کی، پھر شیطان کا گھیرا، پھر اور گہری غفلت۔ اور اس دائرے سے نکلنے کا راستہ بھی وہی ہے جس سے داخلہ ہوا تھا: یاد کی طرف واپسی، جس کے لیے دروازہ کبھی بند نہیں ہوتا۔

ذکر کیا ہے اور اس کا بھولنا کیا؟ اہلِ لغت میں یہ قول مشہور ہے کہ انسان کو انسان اس لیے کہا گیا کہ وہ بھولتا ہے؛ نسیان اس کی سرشت میں ہے اور ذکر اس کا علاج۔ اہلِ تصوف کہتے ہیں کہ ذکر دراصل یاد دہانی ہے، کوئی نئی بات سیکھنا نہیں: روح نے ازل میں اپنے رب کو پہچانا تھا، دنیا میں آ کر وہ پہچان پردوں میں چھپ گئی، اور ذکر انہی پردوں کو سرکاتا ہے۔ شیخِ اکبر کے مکتب میں ذکر کے مراتب ہیں: زبان کا ذکر، جس میں لفظ ہے؛ دل کا ذکر، جس میں حضوری ہے؛ اور سرّ کا ذکر، جس میں ذاکر اپنے ذکر سے بھی غائب ہو کر مذکور میں گم ہو جاتا ہے۔ اس مکتب میں یہ بھی کہا جاتا ہے کہ بندے کا ذکر حق کے ذکر کا سایہ ہے؛ حدیثِ قدسی کا مفہوم ہے کہ بندہ مجھے اپنے دل میں یاد کرے تو میں اسے اپنے ہاں یاد کرتا ہوں۔ جس دل میں ذکر بستا ہے وہاں شیطان کے لیے جگہ نہیں رہتی؛ حضرت ابنِ عباس رضی اللہ عنہما سے خناس کی تفسیر میں منقول ہے، جس کا مفہوم یہ ہے کہ شیطان دل پر جما رہتا ہے، بندہ غافل ہو تو وسوسہ ڈالتا ہے اور اللہ کو یاد کرے تو پیچھے ہٹ جاتا ہے۔ اسی سورت کی ابتدا میں فرمایا گیا تھا: اَحْصٰىهُ اللّٰهُ وَنَسُوْهُ۔ یہاں اس کی جڑ کھلتی ہے: جنہوں نے اللہ کو بھلایا وہ اپنے اعمال کو بھی بھول گئے، مگر وہ ذات کچھ نہیں بھولتی۔ پس شیطان کی ساری حکومت ایک خلا پر قائم ہے، اور ایک سچی یاد اس خلا کو بھر دیتی ہے۔ گمراہی کا آغاز کوئی بڑا گناہ نہیں، ایک چھوٹی فراموشی ہے؛ اور ہدایت کا آغاز کوئی بڑا کارنامہ نہیں، ایک سچی یاد ہے۔

اب مخالفت کا انجام اور حق کا وعدہ: اِنَّ الَّذِيْنَ يُحَآدُّوْنَ اللّٰهَ وَرَسُوْلَهٗۤ اُولٰۤئِكَ فِي الْاَذَلِّيْنَ۔ بے شک جو لوگ اللہ اور اس کے رسول کی مخالفت پر کمر بستہ ہیں، وہی سب سے زیادہ ذلیلوں میں ہیں۔ كَتَبَ اللّٰهُ لَاَغْلِبَنَّ اَنَا وَرُسُلِيْ ۭ اِنَّ اللّٰهَ قَوِيٌّ عَزِيْزٌ۔ اللہ نے لکھ دیا ہے کہ میں اور میرے رسول ضرور غالب رہیں گے؛ بے شک اللہ قوت والا، غالب ہے۔ "يُحَآدُّوْنَ" کی اصل حد ہے: محادّت کا مطلب ہے دوسرے کنارے پر کھڑا ہونا، مقابل حد میں ڈٹ جانا۔ اسی سورت کے پہلے رکوع میں کفارۂ ظہار کے بعد فرمایا گیا تھا کہ یہ اللہ کی حدیں ہیں؛ جو ان حدوں کے مقابل اپنی حد کھینچ لے وہ خود کو اس دائرے سے باہر کر لیتا ہے جہاں عزت ہے۔ عزت کا سرچشمہ "عَزِيْزٌ" ہے؛ جو اس سے کٹا وہ ذلیلوں میں جا ملا، ذلت کسی نے اس پر تھوپی نہیں۔ پھر "كَتَبَ اللّٰهُ" کا لفظ آتا ہے، اور اہلِ ذوق اس پر چونکتے ہیں کہ یہی فعل اگلی آیت میں مومنوں کے دلوں کے بارے میں آئے گا۔ ایک تحریر تقدیر کی تختی پر ہے کہ حق غالب رہے گا، دوسری دلوں کی تختی پر کہ ایمان ثبت ہوا۔ شیخِ اکبر کے نزدیک رسولوں کا غلبہ دراصل اسمائے الٰہیہ کا غلبہ ہے: رسول اسمِ ہادی اور اسمِ قوی کے مظاہر ہیں، اور ان کی فتح حق ہی کی فتح ہے؛ اسی لیے "اَنَا وَرُسُلِيْ" کو ایک جملے میں جمع کیا گیا۔ یہ غلبہ پہلے دلیل اور حجت کا ہے، پھر انجام کا؛ ظاہر میں کبھی دیر ہوتی ہے، مگر لکھی ہوئی بات بدلتی نہیں۔ اور سالک کے لیے اس میں بشارت ہے: جو دل اللہ کے ساتھ ہو جائے وہ اپنی کمزوری کے باوجود "قَوِيٌّ عَزِيْزٌ" کی پناہ میں ہے۔

اب سورت اپنی چوٹی پر پہنچتی ہے: لَا تَجِدُ قَوْمًا يُّؤْمِنُوْنَ بِاللّٰهِ وَالْيَوْمِ الْاٰخِرِ يُوَآدُّوْنَ مَنْ حَآدَّ اللّٰهَ وَرَسُوْلَهٗ۔ آپ ایسی کوئی قوم نہ پائیں گے جو اللہ اور یومِ آخرت پر ایمان رکھتی ہو اور ان لوگوں سے دلی دوستی رکھے جو اللہ اور اس کے رسول سے دشمنی پر اترے ہوئے ہیں، وَلَوْ كَانُوْۤا اٰبَآءَهُمْ اَوْ اَبْنَآءَهُمْ اَوْ اِخْوَانَهُمْ اَوْ عَشِيْرَتَهُمْ۔ خواہ وہ ان کے باپ ہوں یا بیٹے یا بھائی یا کنبے والے۔ یہاں ادب کا ایک ضروری نکتہ واضح رہنا چاہیے۔ آیت کا لفظ "حَآدَّ" ہے: وہ جو اللہ اور رسول کے مقابل حد کھینچ کر عملی دشمنی میں سرگرم ہو، حق سے لڑ رہا ہو اور حق والوں کو مٹانا چاہتا ہو۔ مفسرین نے اس آیت کو ان صحابہ رضی اللہ عنہم کے حال سے جوڑا ہے جن کے قریبی رشتہ دار میدانِ جنگ میں ان کے مقابل کھڑے تھے۔ اس سے ہرگز یہ نہیں نکلتا کہ پُرامن غیر مسلموں سے حسنِ سلوک ممنوع ہے۔ قرآن نے خود یہ فرق کھول دیا: لَا يَنْهٰىكُمُ اللّٰهُ عَنِ الَّذِيْنَ لَمْ يُقَاتِلُوْكُمْ فِي الدِّيْنِ وَلَمْ يُخْرِجُوْكُمْ مِّنْ دِيَارِكُمْ اَنْ تَبَرُّوْهُمْ وَتُقْسِطُوْۤا اِلَيْهِمْ۔ اللہ تمہیں ان لوگوں کے ساتھ نیکی اور انصاف کرنے سے نہیں روکتا جنہوں نے دین کے معاملے میں تم سے جنگ نہیں کی اور تمہیں تمہارے گھروں سے نہیں نکالا۔ صحیح روایت کا مفہوم ہے کہ حضرت اسماء بنت ابی بکر رضی اللہ عنہما کی والدہ، جو ابھی اسلام نہیں لائی تھیں، ان سے ملنے آئیں تو حضورِ اکرم صلی اللہ علیہ وسلم نے انہیں ماں کے ساتھ صلہ رحمی کا حکم دیا۔ پس منع وہ مودّت ہے جو حق سے دشمنی کی طرف داری بن جائے، نہ کہ وہ نیکی، عدل اور رحم جو ہر انسان کا حق ہے۔ اہلِ معرفت کہتے ہیں کہ دل ایک ہے اور اس کی سب سے گہری وفاداری ایک ہی سمت جا سکتی ہے؛ جس کی محبت کا قبلہ حق ہو وہ حق کی دشمنی کو اپنی وفاداری میں شریک نہیں کرتا، مگر ہر مخلوق پر رحمت کا سایہ ضرور ڈالتا ہے، کیونکہ حق کی رحمت ہر شے پر وسیع ہے۔

پھر ان لوگوں کا انعام بیان ہوتا ہے: اُولٰۤئِكَ كَتَبَ فِيْ قُلُوْبِهِمُ الْاِيْمَانَ۔ یہی ہیں جن کے دلوں میں اللہ نے ایمان لکھ دیا ہے۔ منافقوں کا ایمان زبان کی قسموں میں تھا؛ ان کا ایمان دل کی تختی پر لکھا ہوا ہے۔ قسم بولی جاتی ہے اور ہوا میں بکھر جاتی ہے، تحریر ثبت ہوتی ہے اور باقی رہتی ہے۔ اہلِ معرفت کہتے ہیں کہ دل ایک لوح ہے؛ جس دل پر خواہش لکھتی ہے وہاں خواہش کی عبارت پڑھی جاتی ہے، اور جس پر حق لکھے وہاں ایمان کی۔ لکھنے والا یہاں خود اللہ ہے، اس لیے یہ ایمان عطا کی مہر ہے؛ مگر عطا خالی تختی پر اترتی ہے۔ ان بندوں نے حق کے مقابلے میں خون کے رشتے کو ترجیح نہ دی، اور یوں اپنی تختی غیر کی تحریر سے صاف رکھی؛ حق نے اس پر ایمان ثبت کر دیا۔ شیخِ اکبر کے مکتب میں قلب کی اصل تقلب ہے؛ دل ہر لمحہ پلٹتا ہے، حالات میں، تجلیات میں، کیفیتوں میں۔ یہ تلوین ہے۔ مگر جس دل میں ایمان لکھ دیا گیا اس کی پلٹیں ایک مرکز کے گرد ہوتی ہیں؛ وہ ہر تجلی کے ساتھ پھرتا ہے اور پھر بھی اپنی جگہ سے نہیں ہٹتا۔ یہی تمکین ہے: حرکت میں ثبات۔ اور پچھلی آیت کا "كَتَبَ اللّٰهُ" یہاں اپنا دوسرا رخ دکھاتا ہے: جو تحریر عالم میں رسولوں کے غلبے کی ہے، وہی تحریر مومن کے دل میں ایمان کی ہے؛ اہلِ ذوق کہتے ہیں کہ باہر کا غلبہ اسی اندر کی تحریر کا ظہور ہے۔ جس دل میں ایمان لکھا گیا اس پر شیطان کا گھیرا نہیں چلتا؛ گھیرا اسی پر پڑتا ہے جس نے اپنی تختی غفلت کے حوالے کر دی۔

اور تحریر کے ساتھ تائید بھی: وَاَيَّدَهُمْ بِرُوْحٍ مِّنْهُ۔ اور اپنی طرف سے ایک روح کے ذریعے ان کی تائید فرمائی۔ وَيُدْخِلُهُمْ جَنّٰتٍ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِيْنَ فِيْهَا۔ اور انہیں ایسے باغوں میں داخل کرے گا جن کے نیچے نہریں بہتی ہیں، ہمیشہ ان میں رہیں گے۔ مفسرین نے یہاں روح کے کئی معنی بیان کیے ہیں: نورِ ایمان، قرآن، نصرتِ الٰہی، اور حضرت جبریل علیہ السلام کے ذریعے تائید۔ صحیح حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے حضرت حسان بن ثابت رضی اللہ عنہ کے لیے دعا فرمائی کہ روح القدس سے ان کی تائید ہو۔ اہلِ معرفت کہتے ہیں کہ جس نے حق کے مقابلے میں خون کا رشتہ نہ چنا، اسے ایک ایسی نسبت دی گئی جو خون سے بڑھ کر ہے: "مِّنْهُ" کی نسبت۔ حضرت آدم علیہ السلام کے پیکر میں روح پھونکی گئی تو انسان زندہ ہوا؛ یہاں مومن کے دل کو ایک اور روح سے تقویت ملی تو اس کا ایمان زندہ ہوا۔ شیخِ اکبر کے نزدیک ہر شے کی ایک روح ہے جس سے اس کی زندگی ہے؛ جسم کی روح جان ہے، علم کی روح معرفت، اور ایمان کی روح وہ تائید جو بندے کو اپنی کوشش سے آگے حق کے سہارے کھڑا کر دیتی ہے۔ "مِّنْهُ" بتاتا ہے کہ یہ روح بندے کی کمائی نہیں، عطا ہے؛ مگر عطا اسی برتن میں اترتی ہے جو خالی کیا گیا ہو۔ یہ استعداد کا راز ہے: فیض ہر وقت جاری ہے، قبول ظرف کے مطابق ہے۔ پھر جنت کا ذکر ہے، اور اہلِ ذوق کہتے ہیں کہ جس دل میں ایمان لکھا گیا اور روح سے تائید ہوئی، اس کی جنت یہیں سے شروع ہو جاتی ہے؛ معرفت کی نہریں اس کے باطن میں بہنے لگتی ہیں۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: آخرت کے باغ برحق ہیں، اور یہ باطنی بہار انہی کی پیشگی خوشبو ہے۔

اور اب وہ جملہ جس سے بلند کوئی انعام نہیں: رَضِيَ اللّٰهُ عَنْهُمْ وَرَضُوْا عَنْهُ۔ اللہ ان سے راضی ہوا اور وہ اس سے راضی ہوئے۔ یہ اس سورت کی چوٹی ہے، اور اہلِ معرفت کے نزدیک سلوک کی بھی۔ ترتیب پر غور کیجیے: اللہ کی رضا پہلے، بندوں کی رضا بعد میں۔ یہ سبقتِ رحمت ہے؛ ہر خیر کی ابتدا اسی کی طرف سے ہے، اور بندے کی رضا اس رضا کا عکس ہے جو پہلے اس پر اتری۔ رضا یہ ہے کہ بندہ اپنے رب کے فیصلوں پر مطمئن ہو، اس کی چاہت حق کی چاہت سے ہم آہنگ ہو جائے، اور اگر شکوہ ہو تو صرف اللہ کے حضور، جیسا اس سورت کی پہلی آیت میں حضرت خولہ رضی اللہ عنہا نے کیا تھا؛ وہ شکوہ بے رضائی نہیں تھا، رضا ہی کی ایک صورت تھی کہ فریاد کا رخ صرف اسی کی طرف رہا۔ شیخِ اکبر کے مکتب میں رضا وہ مقام ہے جہاں بندہ قضا کو اپنے عینِ ثابت کے تقاضے کے مطابق پہچان لیتا ہے: جو کچھ اس پر جاری ہوا، اس کی اپنی حقیقت کے مطابق جاری ہوا؛ پس اعتراض کا محل نہیں رہتا اور دل ہر حکم میں حکمت دیکھتا ہے۔ صحیح حدیث کا مفہوم ہے کہ اللہ اہلِ جنت سے پوچھے گا: کیا تم راضی ہو؟ وہ عرض کریں گے: ہم کیوں راضی نہ ہوں؟ تو فرمائے گا: میں تم پر اپنی رضا اتارتا ہوں، اس کے بعد کبھی تم سے ناراض نہ ہوں گا۔ گویا جنت کی سب نعمتوں کے اوپر ایک نعمت ہے، اور وہ رضا ہے۔ اہلِ ذوق کہتے ہیں کہ دنیا میں جو بندہ حق سے راضی ہو گیا، اس نے آخرت کی اس نعمت کا ذائقہ یہیں چکھ لیا۔

آخر میں اعلان ہوتا ہے: اُولٰۤئِكَ حِزْبُ اللّٰهِ ۭ اَلَاۤ اِنَّ حِزْبَ اللّٰهِ هُمُ الْمُفْلِحُوْنَ۔ یہی اللہ کی جماعت ہیں؛ خبردار، اللہ کی جماعت ہی فلاح پانے والی ہے۔ رکوع دو جماعتوں پر ختم ہوا، اور دونوں کا ذکر ایک ہی سانچے میں ہے: "اُولٰۤئِكَ حِزْبُ الشَّيْطٰنِ" اور "اُولٰۤئِكَ حِزْبُ اللّٰهِ"۔ فرق کہاں سے پڑا؟ ایک کو ذکر بھلا دیا گیا، دوسرے کے دل میں ایمان لکھ دیا گیا۔ ایک نے خود فراموشی کا دروازہ کھولا، دوسرے نے اپنا دل حق کی تحریر کے آگے رکھ دیا۔ کوئی زبردستی کسی جماعت میں بھرتی نہیں کیا جاتا؛ ہر شخص اپنی یاد اور اپنی وفاداری کے رخ سے اپنی جماعت خود چنتا ہے، اور حق اس انتخاب کو حکم بنا دیتا ہے۔ اہلِ لغت بتاتے ہیں کہ فلاح کی اصل زمین کو چیرنا ہے؛ کاشت کار کو فلاح اسی لیے کہتے ہیں کہ وہ زمین پھاڑ کر بیج بوتا ہے۔ اہلِ ذوق کہتے ہیں کہ مفلح وہ ہے جس کا دل ذکر کے ہل سے نرم ہوا اور جس میں ایمان کا بیج لکھ دیا گیا؛ اس کھیتی کا پھل رضا ہے۔ یہ ایک مشہور نکتہ ہے کہ اس سورت کی ہر آیت میں اللہ کا نام آیا ہے، اور اس آخری آیت میں یہ نام بار بار جلوہ گر ہے۔ سورت ایک خاتون کی فریاد سے شروع ہوئی جو سات آسمانوں کے اوپر سنی گئی، اور ایک جماعت کی فلاح پر ختم ہوئی؛ بیچ میں وہی ایک راز ہے: یاد۔ جس نے یاد رکھا وہ سنا گیا، لکھا گیا، تائید پا گیا اور راضی کیا گیا؛ اور جس نے بھلایا، وہ اپنے ہی بھلانے میں کھو گیا۔
