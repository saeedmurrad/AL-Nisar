یہ رکوع پھر غزوۂ تبوک کی فضا میں لوٹتا ہے۔ وہ سفر سخت گرمی میں تھا، فاصلہ لمبا تھا، کھجوروں کے پکنے کا موسم تھا اور سایہ دل کو کھینچتا تھا؛ اسی لیے اہلِ سیرت اس لشکر کو تنگی کا لشکر کہتے ہیں۔ اس پس منظر میں فرمایا: فَرِحَ الْمُخَلَّفُوْنَ بِمَقْعَدِهِمْ خِلٰفَ رَسُوْلِ اللّٰهِ وَكَرِهُوْۤا اَنْ يُّجَاهِدُوْا بِاَمْوَالِهِمْ وَاَنْفُسِهِمْ فِيْ سَبِيْلِ اللّٰهِ وَقَالُوْا لَا تَنْفِرُوْا فِي الْحَرِّ ۗ قُلْ نَارُ جَهَنَّمَ اَشَدُّ حَرًّا ۗ لَوْ كَانُوْا يَفْقَهُوْنَ۔ “پیچھے چھوڑ دیے جانے والے اللہ کے رسول صلی اللہ علیہ وسلم کے پیچھے اپنے بیٹھ رہنے پر خوش ہوئے، اور انہیں ناگوار ہوا کہ اپنے مالوں اور جانوں سے اللہ کی راہ میں جہاد کریں، اور انہوں نے کہا: گرمی میں نہ نکلو۔ کہہ دیجیے: جہنم کی آگ کہیں زیادہ گرم ہے، کاش وہ سمجھتے۔” لفظ فَرِحَ الْمُخَلَّفُوْنَ پر غور کیجیے: یہ مفعول کا صیغہ ہے، یعنی “پیچھے چھوڑ دیے گئے”، نہ کہ صرف “پیچھے رہنے والے”۔ اہلِ تاویل کہتے ہیں کہ وہ اپنے خیال میں خود رکے تھے اور خوش تھے کہ مشقت سے بچ گئے، مگر حقیقت میں قافلہ انہیں چھوڑ کر آگے بڑھ گیا؛ ان کا اپنا رکنا ہی ان کا چھوڑ دیا جانا بن گیا۔ اور بِمَقْعَدِهِمْ خِلٰفَ رَسُوْلِ اللّٰهِ کے دو معنی بیان ہوئے ہیں: آپ صلی اللہ علیہ وسلم کے تشریف لے جانے کے بعد پیچھے، اور آپ صلی اللہ علیہ وسلم کی مرضی کے خلاف۔ دونوں معنی جڑے ہوئے ہیں: جو رسول صلی اللہ علیہ وسلم کے پیچھے رہ جائے وہ رفتہ رفتہ آپ صلی اللہ علیہ وسلم کے راستے کے خلاف چلنے لگتا ہے۔ یاد رہے کہ یہ ایک خاص تاریخی گروہ کا ذکر ہے جس کی خبر وحی نے دی؛ آج کسی فرد یا جماعت کو اس آیت کا مصداق ٹھہرانا ہمارا کام نہیں۔

ان کا جملہ تھا لَا تَنْفِرُوْا فِي الْحَرِّ، اور جواب ایک ہی جملے میں آیا: قُلْ نَارُ جَهَنَّمَ اَشَدُّ حَرًّا۔ صحیح بخاری اور صحیح مسلم میں حضرت ابو ہریرہ رضی اللہ عنہ سے روایت ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ تمہاری یہ آگ جہنم کی آگ کے ستر حصوں میں سے ایک حصہ ہے۔ جو ایک گرمی سے بھاگا وہ اس سے کہیں سخت گرمی کی طرف بھاگا۔ آیت کا آخری ٹکڑا لَوْ كَانُوْا يَفْقَهُوْنَ بتاتا ہے کہ کمی معلومات کی نہیں، فقہ کی تھی۔ قرآن کی زبان میں فقہ دل کی گہری سمجھ ہے، صرف جان لینا نہیں؛ وہ جانتے تھے کہ جہنم ہے، مگر اس جاننے نے ان کے قدم نہ اٹھائے۔ اہلِ معرفت کہتے ہیں کہ سالک کی راہ میں بھی ایک گرمی آتی ہے: مجاہدے کی تپش، خواہش کے خلاف چلنے کی جلن، رات کو اٹھنے کی مشقت۔ نفس ہمیشہ یہی کہتا ہے کہ اس گرمی میں نہ نکلو؛ اور جواب یہی ہے کہ حجاب کی آگ، یعنی حق سے دوری کی جلن، اس سے کہیں زیادہ سخت ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کہا جاتا ہے کہ تجلی کا ایک رخ جلال ہے اور ایک جمال، اور جلال کے پردے میں اکثر جمال چھپا ہوتا ہے؛ مجاہدے کی تپش جلال کا لباس ہے، اور جو اس کی پہلی تپش سے بھاگ جائے وہ اس جمال تک نہیں پہنچتا جو اس کے پیچھے ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: جہنم کی آگ حق ہے، اور دوری کی یہ باطنی جلن اسی حقیقت کا ایک سایہ ہے جو اسی دنیا میں محسوس ہو جاتا ہے۔

پھر ایک فیصلہ کن جملہ آتا ہے: فَلْيَضْحَكُوْا قَلِيْلًا وَّلْيَبْكُوْا كَثِيْرًا ۚ جَزَآءًۢ بِمَا كَانُوْا يَكْسِبُوْنَ۔ “پس وہ تھوڑا ہنسیں اور بہت روئیں، اس کے بدلے میں جو وہ کماتے تھے۔” مفسرین کہتے ہیں کہ یہ حکم کی صورت میں خبر ہے: دنیا کی مختصر زندگی میں ان کی ہنسی تھوڑی ہے، اور جو انجام انہوں نے اپنے لیے کمایا اس میں ان کا رونا طویل ہے۔ اور آیت کا آخری ٹکڑا پھر وہی اصول دہراتا ہے: بِمَا كَانُوْا يَكْسِبُوْنَ، یعنی یہ ان کی اپنی کمائی ہے، کوئی بیرونی جبر نہیں۔ صحیح بخاری اور صحیح مسلم میں حضرت انس رضی اللہ عنہ سے روایت ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ جو میں جانتا ہوں اگر تم جانتے تو تھوڑا ہنستے اور بہت روتے؛ اور روایت میں ہے کہ یہ سن کر صحابۂ کرام رضی اللہ عنہم نے اپنے چہرے ڈھانپ لیے اور ان کی سسکیاں سنائی دینے لگیں۔ اس سے یہ نہ سمجھا جائے کہ ہنسی بذاتِ خود مذموم ہے؛ حضورِ اکرم صلی اللہ علیہ وسلم کا تبسم مشہور ہے۔ مذمت اس ہنسی کی ہے جو غفلت سے پھوٹے، جیسے ان لوگوں کی ہنسی جو پیچھے رہ جانے پر خوش تھے۔ اہلِ ذوق کہتے ہیں کہ عارف کا رونا شوق کا رونا ہے، اور یہ آنسو دل کو دھوتے ہیں؛ قبض کی حالت میں یہی آنسو سالک کا سرمایہ ہیں، اور بسط کی حالت میں اس کی مسکراہٹ بھی حضوری کی مسکراہٹ ہوتی ہے، غفلت کی نہیں۔

اب ان کا انجام دنیا ہی میں بیان ہوتا ہے: فَاِنْ رَّجَعَكَ اللّٰهُ اِلٰى طَآئِفَةٍ مِّنْهُمْ فَاسْتَاْذَنُوْكَ لِلْخُرُوْجِ فَقُلْ لَّنْ تَخْرُجُوْا مَعِيَ اَبَدًا وَّلَنْ تُقَاتِلُوْا مَعِيَ عَدُوًّا ۗ اِنَّكُمْ رَضِيْتُمْ بِالْقُعُوْدِ اَوَّلَ مَرَّةٍ فَاقْعُدُوْا مَعَ الْخٰلِفِيْنَ۔ “پھر اگر اللہ آپ صلی اللہ علیہ وسلم کو ان میں سے کسی گروہ کی طرف واپس لے آئے اور وہ آپ صلی اللہ علیہ وسلم سے نکلنے کی اجازت مانگیں تو کہہ دیجیے: تم میرے ساتھ کبھی نہیں نکلو گے اور میرے ساتھ ہو کر کسی دشمن سے نہیں لڑو گے؛ تم پہلی بار بیٹھ رہنے پر راضی ہوئے تھے، تو اب پیچھے رہنے والوں کے ساتھ بیٹھے رہو۔” مفسرین نکتہ بیان کرتے ہیں کہ آیت نے “ان میں سے ایک گروہ” کہا، سب نہیں، کیونکہ پیچھے رہنے والوں میں کچھ ایسے بھی تھے جنہوں نے بعد میں سچی توبہ کی۔ اور فیصلے کا سبب آیت نے خود بتا دیا: اِنَّكُمْ رَضِيْتُمْ بِالْقُعُوْدِ اَوَّلَ مَرَّةٍ۔ پہلی بار ان کی اپنی رضا تھی، اب حکم اسی رضا کے مطابق ہے۔ یہ سرِّ قدر کی صاف مثال ہے: حکم محکوم کے مطابق۔ حضورِ اکرم صلی اللہ علیہ وسلم کی معیت ایک عزت ہے، اور عزت اس پر زبردستی نہیں لادی جاتی جس نے پہلے اسے ٹھکرا دیا ہو۔ اہلِ معرفت ایک باریک نکتہ یہ بھی بتاتے ہیں کہ رضا راہِ سلوک کا بلند مقام ہے، اور ان لوگوں نے اسی مقام کو الٹا برت لیا: اللہ سے راضی ہونے کی جگہ وہ بیٹھ رہنے سے راضی ہو گئے۔ دل کی رضا جس چیز پر ٹھہرتی ہے، بندہ اسی کی صف میں لکھا جاتا ہے۔

اب وہ آیت جس کے ساتھ اس سورت کا ایک نہایت نازک واقعہ جڑا ہوا ہے: وَلَا تُصَلِّ عَلٰۤى اَحَدٍ مِّنْهُمْ مَّاتَ اَبَدًا وَّلَا تَقُمْ عَلٰى قَبْرِهٖ ۗ اِنَّهُمْ كَفَرُوْا بِاللّٰهِ وَرَسُوْلِهٖ وَمَاتُوْا وَهُمْ فٰسِقُوْنَ۔ “اور ان میں سے جو کوئی مر جائے اس پر کبھی نماز نہ پڑھیے اور نہ اس کی قبر پر کھڑے ہوں؛ انہوں نے اللہ اور اس کے رسول صلی اللہ علیہ وسلم کا انکار کیا اور اس حال میں مرے کہ وہ نافرمان تھے۔” صحیح بخاری میں حضرت عبد اللہ بن عمر رضی اللہ عنہما سے روایت ہے کہ جب منافقوں کے سردار عبد اللہ بن ابی کا انتقال ہوا تو اس کے بیٹے حضرت عبد اللہ رضی اللہ عنہ، جو سچے اور مخلص صحابی تھے، حضورِ اکرم صلی اللہ علیہ وسلم کی خدمت میں حاضر ہوئے اور عرض کیا کہ اپنی قمیص عطا فرمائیے کہ میں اس میں اپنے والد کو کفن دوں، اور اس کی نمازِ جنازہ پڑھا دیجیے اور اس کے لیے مغفرت کی دعا فرمائیے۔ آپ صلی اللہ علیہ وسلم نے اپنی قمیص عطا فرما دی۔ پھر جب آپ صلی اللہ علیہ وسلم نماز کے لیے کھڑے ہونے لگے تو حضرت عمر بن خطاب رضی اللہ عنہ نے آپ صلی اللہ علیہ وسلم کا دامن تھام لیا اور عرض کیا: یا رسول اللہ صلی اللہ علیہ وسلم! کیا آپ صلی اللہ علیہ وسلم اس پر نماز پڑھیں گے جبکہ اللہ نے ان کے لیے استغفار سے منع فرمایا ہے؟ آپ صلی اللہ علیہ وسلم نے فرمایا کہ اللہ نے مجھے اختیار دیا ہے، اور پچھلے رکوع کی ستر بار والی آیت کی تلاوت فرمائی، اور فرمایا کہ میں ستر سے زیادہ بار مانگ لوں گا۔ پھر آپ صلی اللہ علیہ وسلم نے نماز پڑھائی، اور اس کے بعد یہ آیت نازل ہوئی۔

حضرت عمر رضی اللہ عنہ کی اپنی روایت، جو صحیح بخاری ہی میں ہے، اس واقعے میں ادب کا ایک اور رخ کھولتی ہے۔ وہ فرماتے ہیں کہ میں نے بار بار عرض کیا تو حضورِ اکرم صلی اللہ علیہ وسلم مسکرائے اور فرمایا: عمر! پیچھے ہٹ جاؤ، مجھے اختیار دیا گیا اور میں نے اختیار کر لیا؛ اگر مجھے معلوم ہوتا کہ ستر سے زیادہ بار مانگنے سے اس کی بخشش ہو جائے گی تو میں زیادہ مانگتا۔ پھر آپ صلی اللہ علیہ وسلم نے نماز پڑھائی اور واپس تشریف لائے، اور تھوڑی ہی دیر گزری تھی کہ یہ آیت اتری۔ حضرت عمر رضی اللہ عنہ فرماتے ہیں کہ بعد میں مجھے رسول اللہ صلی اللہ علیہ وسلم کے حضور اپنی اس دن کی جرأت پر تعجب ہوا۔ ادب سے دیکھیے تو اس واقعے میں ہر کردار اپنی جگہ روشن ہے۔ حضرت عمر رضی اللہ عنہ کا اعتراض دین کی غیرت سے تھا، اور وہ خود بعد میں اپنی جرأت پر حیران ہوئے؛ اہلِ علم اس واقعے کو ان مواقع میں شمار کرتے ہیں جن میں ان کی رائے بعد میں اترنے والی وحی کے موافق نکلی۔ اور حضورِ اکرم صلی اللہ علیہ وسلم نے جو کیا وہ اللہ کے دیے ہوئے اختیار کے اندر تھا، کسی حکم کی خلاف ورزی نہیں تھی؛ آپ صلی اللہ علیہ وسلم نے رحمت والا پہلو چنا، ایک مخلص صحابی بیٹے کے دل کی دلداری فرمائی، اور ایک پرانا احسان چکایا۔ صحیح بخاری میں حضرت جابر رضی اللہ عنہ کی روایت کے ساتھ یہ ذکر آیا ہے کہ بدر کے دن حضرت عباس رضی اللہ عنہ کو قمیص کی ضرورت پڑی تو اسی شخص نے اپنی قمیص دی تھی، اور اہلِ علم نے کہا کہ آپ صلی اللہ علیہ وسلم نے اس احسان کا بدلہ چکانا پسند فرمایا۔ اور مفسرین لکھتے ہیں کہ اس آیت کے بعد آپ صلی اللہ علیہ وسلم نے کسی منافق کی نمازِ جنازہ نہیں پڑھی۔

اس واقعے کی گہرائی میں حضورِ اکرم صلی اللہ علیہ وسلم کی رحمت کا وہ وصف جھلکتا ہے جسے قرآن نے یوں بیان کیا: وَمَاۤ اَرْسَلْنٰكَ اِلَّا رَحْمَةً لِّلْعٰلَمِيْنَ۔ شیخِ اکبر حکمتِ فردیہ میں جو بات بیان کرتے ہیں اس کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم اس نوعِ انسانی میں سب سے کامل وجود ہیں؛ اور اس مکتب میں انسانِ کامل وہ ہے جس میں رحمت اور حکم دونوں اپنی اپنی جگہ پر ہوں۔ آپ صلی اللہ علیہ وسلم نے اختیار کی گنجائش میں رحمت کو چنا، اور جب حکم آیا تو اسی رحمت نے حکم کے سامنے سر جھکا دیا؛ یہ دونوں حالتیں ایک ہی کمال کی دو تجلیاں ہیں، اور دونوں میں عبدیتِ کاملہ ہے۔ پھر آیت نے ممانعت کا سبب بھی خود بیان فرمایا: وَمَاتُوْا وَهُمْ فٰسِقُوْنَ۔ رحمت کا دروازہ بند نہیں ہوا تھا؛ لینے والے نے عمر بھر اپنا ظرف الٹا رکھا اور اسی حال پر دنیا سے گیا۔ قاری کے لیے آئینہ یہ ہے کہ اعمال کا دار و مدار خاتمے پر ہے، جیسا کہ صحیح بخاری میں حضرت سہل بن سعد رضی اللہ عنہ کی روایت کا مفہوم ہے؛ اس لیے بندہ اپنے لیے حسنِ خاتمہ کی دعا کرتا رہے۔ اور یہ بھی یاد رہے کہ یہ حکم وحی کے ذریعے متعین لوگوں کے بارے میں تھا؛ ہمارا حکم ظاہر پر ہے، جو کلمہ پڑھتا ہو اس کی نمازِ جنازہ پڑھی جاتی ہے، اور باطن کا علم صرف اللہ کے پاس ہے۔

اگلی آیت ایک ایسی چیز سے خبردار کرتی ہے جو آنکھ کو آسانی سے بہکا دیتی ہے: وَلَا تُعْجِبْكَ اَمْوَالُهُمْ وَاَوْلَادُهُمْ ۗ اِنَّمَا يُرِيْدُ اللّٰهُ اَنْ يُّعَذِّبَهُمْ بِهَا فِي الدُّنْيَا وَتَزْهَقَ اَنْفُسُهُمْ وَهُمْ كٰفِرُوْنَ۔ “اور ان کے مال اور ان کی اولاد آپ صلی اللہ علیہ وسلم کو تعجب میں نہ ڈالیں؛ اللہ تو بس یہ چاہتا ہے کہ انہی کے ذریعے انہیں دنیا میں عذاب دے اور ان کی جانیں اس حال میں نکلیں کہ وہ کافر ہوں۔” یہی مضمون اسی سورت میں پہلے بھی تھوڑے فرق کے ساتھ آ چکا ہے، اور مفسرین کہتے ہیں کہ اس کی تکرار تاکید کے لیے ہے، کیونکہ دل بار بار مال کی چمک کی طرف کھنچتا ہے؛ اور خطاب اگرچہ حضورِ اکرم صلی اللہ علیہ وسلم سے ہے مگر مراد پوری امت کی تعلیم ہے۔ مال اور اولاد عذاب کیسے بنتے ہیں؟ جمع کرنے کی فکر، کھو دینے کا خوف، اور خرچ کرتے وقت ایسی تنگی جیسے کوئی تاوان بھر رہا ہو۔ اور وَتَزْهَقَ اَنْفُسُهُمْ کا منظر یہ ہے کہ جان اس حال میں نکلے کہ دل انہی چیزوں سے چمٹا ہوا ہو؛ پھر جدائی چیرنے کی طرح ہوتی ہے۔ شیخِ اکبر کے نزدیک ہر شے اسماءِ الٰہیہ کا مظہر ہے؛ وہی مال عارف کے لیے آئینہ ہے جس میں وہ دینے والے کو دیکھتا ہے، اور غافل کے لیے پردہ ہے جو دینے والے کو چھپا دیتا ہے۔ چیز ایک ہے، فرق دیکھنے والے دل کے رخ کا ہے؛ اور یہاں بھی حکم اسی رخ کے مطابق ہے جو بندے نے خود اختیار کیا۔

پھر ان لوگوں کی ایک اور عادت سامنے آتی ہے: وَاِذَاۤ اُنْزِلَتْ سُوْرَةٌ اَنْ اٰمِنُوْا بِاللّٰهِ وَجَاهِدُوْا مَعَ رَسُوْلِهِ اسْتَاْذَنَكَ اُولُوا الطَّوْلِ مِنْهُمْ وَقَالُوْا ذَرْنَا نَكُنْ مَّعَ الْقٰعِدِيْنَ۔ “اور جب کوئی سورت اترتی ہے کہ اللہ پر ایمان لاؤ اور اس کے رسول صلی اللہ علیہ وسلم کے ساتھ مل کر جہاد کرو، تو ان میں سے مقدور والے آپ صلی اللہ علیہ وسلم سے اجازت مانگتے ہیں اور کہتے ہیں: ہمیں رہنے دیجیے کہ ہم بیٹھنے والوں کے ساتھ رہیں۔” رَضُوْا بِاَنْ يَّكُوْنُوْا مَعَ الْخَوَالِفِ وَطُبِعَ عَلٰى قُلُوْبِهِمْ فَهُمْ لَا يَفْقَهُوْنَ۔ “وہ اس پر راضی ہو گئے کہ پیچھے رہ جانے والوں کے ساتھ رہیں، اور ان کے دلوں پر مہر لگ گئی، سو وہ نہیں سمجھتے۔” حکم کی ترتیب دیکھیے: پہلے ایمان، پھر جہاد؛ عمل کی جڑ ایمان ہے۔ اور اجازت مانگنے والے کون تھے؟ اُولُوا الطَّوْلِ مِنْهُمْ، یعنی وسعت اور طاقت والے، جن کے پاس کوئی عذر نہ تھا۔ مفسرین نے مَعَ الْخَوَالِفِ کی تشریح میں لکھا کہ اس سے مراد وہ ہیں جو گھروں میں پیچھے رہتے ہیں: عورتیں، بچے، بوڑھے اور بیمار۔ یہاں ادب کا ایک نکتہ لازم ہے: اس میں ان کی کوئی تحقیر نہیں جو حق اور عذر کی بنا پر گھروں میں رہے؛ اگلے ہی رکوع میں قرآن خود کمزوروں، بیماروں اور خرچ نہ پانے والوں سے ہر ملامت اٹھا دیتا ہے۔ ملامت صرف ان طاقت ور مردوں پر ہے جنہوں نے پکار سن کر بھی بیٹھ رہنا پسند کیا۔

اس آیت کی ترتیب خود اس سورت کے بنیادی اصول کی گواہ ہے: پہلے رَضُوْا بِاَنْ يَّكُوْنُوْا مَعَ الْخَوَالِفِ، پھر وَطُبِعَ عَلٰى قُلُوْبِهِمْ۔ پہلے ان کی رضا، پھر مہر۔ مہر ان کی پسند پر لگی، ان کی پسند کے خلاف نہیں۔ جیسے خط لکھنے والا خط مکمل کر لے تو اس پر مہر لگا دی جاتی ہے؛ مہر نے خط کا مضمون نہیں لکھا، صرف لکھے ہوئے کو پختہ کر دیا۔ سورۂ المطففین میں یہی بات یوں آئی: رَانَ عَلٰى قُلُوْبِهِمْ مَّا كَانُوْا يَكْسِبُوْنَ؛ دلوں پر زنگ ان کی اپنی کمائی کا تھا۔ اہلِ معرفت ایک اور نکتہ دکھاتے ہیں: آیت ۸۱ کے آخر میں حسرت تھی، کاش وہ سمجھتے؛ اور آیت ۸۷ کے آخر میں خبر ہے: وہ نہیں سمجھتے۔ درمیان میں ان کی اپنی رضا اور ان کا اپنا اصرار تھا، جس نے امکان کو حالت بنا دیا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا؛ وہ چاہتا تو سب کو ہدایت پر جمع کر دیتا، مگر اس نے مجبور کرنا نہیں چاہا اور انسان کو اختیار کی عزت دی۔ اللہ کسی کو زبردستی گمراہ نہیں کرتا اور ہدایت کے طالب کو کبھی گمراہ نہیں کرتا؛ آگے اسی سورت میں یہ اصول صاف لفظوں میں آئے گا کہ اللہ کسی قوم کو ہدایت دینے کے بعد گمراہ نہیں ٹھہراتا جب تک ان پر واضح نہ کر دے کہ انہیں کس چیز سے بچنا ہے۔ جو آج بھی پکار سن کر اٹھ کھڑا ہو، اس کے دل کی مہر ٹوٹ جاتی ہے، کیونکہ طالب ہمیشہ جواب پاتا ہے۔

پھر رکوع کا رخ بدلتا ہے، اور ایک چھوٹا سا حرف ساری تصویر پلٹ دیتا ہے: لٰكِنِ الرَّسُوْلُ وَالَّذِيْنَ اٰمَنُوْا مَعَهٗ جٰهَدُوْا بِاَمْوَالِهِمْ وَاَنْفُسِهِمْ ۗ وَاُولٰۤئِكَ لَهُمُ الْخَيْرٰتُ ۚ وَاُولٰۤئِكَ هُمُ الْمُفْلِحُوْنَ۔ “لیکن رسول صلی اللہ علیہ وسلم اور وہ لوگ جو ان کے ساتھ ایمان لائے، انہوں نے اپنے مالوں اور اپنی جانوں سے جہاد کیا؛ اور انہی کے لیے بھلائیاں ہیں، اور یہی کامیاب ہیں۔” تبوک کی اسی تنگی میں صحابۂ کرام رضی اللہ عنہم کی قربانیاں تاریخ کا روشن باب ہیں: مروی ہے کہ حضرت ابو بکر صدیق رضی اللہ عنہ اپنا سارا مال لے آئے، اور حضرت عثمان رضی اللہ عنہ نے لشکر کے بڑے حصے کا سامان مہیا کیا، یہاں تک کہ حضورِ اکرم صلی اللہ علیہ وسلم نے ان کے حق میں بشارت کے کلمات فرمائے۔ آیت کا لفظ وَالَّذِيْنَ اٰمَنُوْا مَعَهٗ قابلِ غور ہے: ان کا ایمان بھی رسول صلی اللہ علیہ وسلم کی معیت میں تھا اور ان کا جہاد بھی۔ اہلِ معرفت کہتے ہیں کہ سالک کے لیے یہ معیت آج بھی کھلی ہے: سنت کی پیروی، آپ صلی اللہ علیہ وسلم کی محبت اور درود کی کثرت۔ جو جس کے ساتھ ہوتا ہے اسی کا رنگ پکڑتا ہے۔ اور لَهُمُ الْخَيْرٰتُ جمع اور معرفہ ہے، یعنی دنیا اور آخرت کی ساری بھلائیاں۔ فلاح کا مادہ زمین کو چیرنے کے معنی رکھتا ہے، جیسے کسان ہل سے زمین چیرتا ہے؛ کامیاب وہ ہے جو رکاوٹوں کو چیر کر آگے نکل جائے، نہ کہ وہ جو سائے میں بیٹھ کر خوش ہو۔

اور رکوع اس اعلان پر ختم ہوتا ہے: اَعَدَّ اللّٰهُ لَهُمْ جَنّٰتٍ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِيْنَ فِيْهَا ۗ ذٰلِكَ الْفَوْزُ الْعَظِيْمُ۔ “اللہ نے ان کے لیے ایسے باغ تیار کر رکھے ہیں جن کے نیچے نہریں بہتی ہیں، وہ ان میں ہمیشہ رہیں گے؛ یہی بڑی کامیابی ہے۔” لفظ اَعَدَّ اللّٰهُ لَهُمْ بتاتا ہے کہ انعام پہلے سے تیار تھا، عمل بعد میں ہوا؛ یہ سبقتِ رحمت ہے کہ دینے والے نے عطا کو بندے کے قدم اٹھانے سے پہلے سجا رکھا تھا۔ شیخِ اکبر فتوحاتِ مکیہ میں جنت کی جو تقسیم بیان کرتے ہیں اس کا مفہوم یہ ہے کہ جنتیں تین طرح کی ہیں: ایک خالص الٰہی عطا اور اختصاص کی جنت، ایک میراث کی جنت، اور ایک اعمال کی جنت؛ اور اہلِ ذوق کہتے ہیں کہ اعمال کی جنت بھی حقیقت میں فضل ہی ہے، کیونکہ عمل کی توفیق بھی اسی کی دین ہے۔ اب رکوع کے آغاز اور انجام کو ساتھ رکھ کر دیکھیے: ایک طرف وہ خوشی جو پیچھے رہ جانے پر منائی گئی، دوسری طرف یہ بڑی کامیابی۔ ایک طرف وہ رضا جو بیٹھ رہنے پر ٹھہری، دوسری طرف وہ رضا جس کے بارے میں سورۂ البینہ نے فرمایا: رَضِيَ اللّٰهُ عَنْهُمْ وَرَضُوْا عَنْهُ۔ ہر قاری اپنے دل سے پوچھے کہ اس کی رضا کہاں ٹھہری ہے؛ اور اگر وہ پیچھے رہ گیا ہو تو اسی سورت نے بتا دیا ہے کہ توبہ کا دروازہ کھلا ہے۔
