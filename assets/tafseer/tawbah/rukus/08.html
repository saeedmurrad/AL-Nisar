پچھلے رکوع کے آخر میں ان لوگوں کا ذکر تھا جو صدقات کی تقسیم پر حضورِ اکرم صلی اللہ علیہ وسلم پر حرف رکھتے تھے: کچھ مل جائے تو راضی، نہ ملے تو ناراض۔ اس کے جواب میں حق تعالیٰ نے تقسیم کا ضابطہ کسی کی رائے پر نہیں چھوڑا، خود اپنے کلام میں طے فرما دیا: اِنَّمَا الصَّدَقٰتُ لِلْفُقَرَآءِ وَالْمَسٰكِيْنِ وَالْعٰمِلِيْنَ عَلَيْهَا وَالْمُؤَلَّفَةِ قُلُوْبُهُمْ وَفِي الرِّقَابِ وَالْغٰرِمِيْنَ وَفِيْ سَبِيْلِ اللّٰهِ وَابْنِ السَّبِيْلِ ؕ فَرِيْضَةً مِّنَ اللّٰهِ ؕ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ “صدقات تو صرف فقیروں کے لیے ہیں، اور مسکینوں کے لیے، اور ان کارکنوں کے لیے جو ان کی وصولی پر مقرر ہوں، اور ان کے لیے جن کے دلوں کو مانوس کرنا مقصود ہو، اور گردنیں چھڑانے میں، اور قرض داروں کے لیے، اور اللہ کی راہ میں، اور مسافر کے لیے۔ یہ اللہ کی طرف سے مقرر کیا ہوا فریضہ ہے، اور اللہ جاننے والا، حکمت والا ہے۔” جمہور علما کے نزدیک یہاں صدقات سے مراد فرض زکوٰۃ ہے، اور “اِنَّمَا” حصر کے لیے ہے: اس کے مصارف یہی آٹھ ہیں۔

ان آٹھ مصارف کا مختصر بیان یہ ہے۔ فقیر اور مسکین دونوں حاجت مند ہیں، اور فقہا میں اختلاف ہے کہ کون زیادہ تنگ حال ہے؛ احناف کے نزدیک مسکین، شوافع کے نزدیک فقیر۔ عاملین وہ ہیں جنہیں حکومت زکوٰۃ کی وصولی اور حفاظت پر مقرر کرے، اور انہیں ان کے کام کا معاوضہ اسی سے ملتا ہے۔ مؤلفۃ القلوب وہ ہیں جن کے دلوں کو اسلام کی طرف مائل کرنا یا اس پر جمانا مقصود ہو؛ احناف کے نزدیک یہ مد حضرت ابو بکر صدیق رضی اللہ عنہ کے دورِ خلافت میں حضرت عمر فاروق رضی اللہ عنہ کی رائے پر، اسلام کی قوت کے سبب، صحابہ رضی اللہ عنہم کے اجماع سے موقوف ہوئی، اور دوسرے ائمہ کے نزدیک ضرورت ہو تو باقی ہے۔ رقاب سے مراد غلاموں کو آزاد کرانا ہے، اور بعض علما نے قیدیوں کو چھڑانا بھی اس میں شامل کیا۔ غارمین وہ قرض دار ہیں جو کسی جائز ضرورت میں مقروض ہوئے اور ادا نہیں کر سکتے۔ فی سبیل اللہ سے جمہور کے نزدیک وہ مراد ہیں جو اللہ کی راہ میں نکلے ہوں اور ان کا کوئی مقرر وظیفہ نہ ہو۔ ابن السبیل وہ مسافر ہے جو سفر میں اپنے مال سے کٹ گیا ہو، چاہے گھر میں مالدار ہو۔ ان مصارف میں تقسیم کی تفصیل اور ترتیب فقہ کی کتابوں میں ہے؛ یہاں ظاہر کا یہ اجمالی بیان کافی ہے، اور یہی بنیاد ہے۔

اب اس ظاہر کے پیچھے ایک گہری بات سنیے۔ زکوٰۃ کے لفظ میں پاکیزگی بھی ہے اور بڑھوتری بھی؛ مال جو اپنا حق ادا کر دے، پاک بھی ہوتا ہے اور اس میں برکت بھی آتی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ نے فتوحاتِ مکیہ میں زکوٰۃ کے اسرار پر تفصیل سے کلام کیا ہے، اور ان کے مکتب میں یہ بات بنیادی ہے کہ مال کا حقیقی مالک حق ہے، بندہ اس میں امین اور واسطہ ہے۔ آیت کا آغاز اِنَّمَا الصَّدَقٰتُ لِلْفُقَرَآءِ سے ہوا، اور اہلِ معرفت اس ترتیب میں اشارہ دیکھتے ہیں کہ فقر سب سے پہلا عنوان ہے۔ قرآن نے دوسری جگہ سب انسانوں کو اسی نام سے پکارا: يٰۤاَيُّهَا النَّاسُ اَنْتُمُ الْفُقَرَآءُ اِلَى اللّٰهِ، جیسا کہ سورۂ فاطر میں گزرا۔ پس دینے والا بھی فقیر ہے اور لینے والا بھی؛ دینے والے کا ہاتھ صرف وہ نالی ہے جس سے غنی کا فیض فقیر تک پہنچتا ہے۔ جو اس راز کو پا لے وہ زکوٰۃ دے کر احسان نہیں جتاتا، شکر کرتا ہے کہ اسے واسطہ بنایا گیا۔ اور مسکین کا لفظ سکون سے ہے؛ اہلِ ذوق کہتے ہیں کہ مسکین وہ ہے جسے حاجت نے ساکن کر دیا، جو حق کے دروازے پر ٹھہر گیا اور کہیں اور نہ گیا۔ شیخِ اکبر کے نزدیک فقر عبد کی ذاتی صفت ہے اور غنا صرف حق کی؛ بندہ جتنا اپنے فقر کو پہچانتا ہے، اتنا ہی اپنی اصل پر آتا ہے۔

اہلِ تاویل باقی مصارف میں بھی سالک کے باطن کی ایک تصویر دیکھتے ہیں، اور یہ تصویر ظاہر کی جگہ نہیں لیتی، اس کے ساتھ چلتی ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے، اور زکوٰۃ کا شرعی حکم انہی آٹھ ظاہری مصارف پر ہے۔ اس اشارے میں عاملین وہ قوتیں ہیں جو دل کی خدمت میں کام کرتی ہیں: آنکھ، کان، زبان اور ہاتھ، جن کا حق یہ ہے کہ انہیں بھی ان کی محنت کا نور ملے۔ مؤلفۃ القلوب وہ دل ہے جو ابھی راہ سے پوری طرح مانوس نہیں ہوا؛ اسے سختی سے نہیں، نرمی اور عطا سے قریب کیا جاتا ہے۔ وَفِي الرِّقَابِ وہ گردن ہے جو عادتوں اور خواہشوں کی غلام ہے، اور جس کی آزادی سب سے بڑا صدقہ ہے۔ غارم وہ بندہ ہے جو اپنے رب کے حقوق کا قرض دار ہے اور اپنی کوتاہی کو پہچانتا ہے۔ اور وَابْنِ السَّبِيْلِ خود سالک ہے؛ صحیح بخاری میں حضرت عبد اللہ بن عمر رضی اللہ عنہما سے مروی ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے ان کا کندھا پکڑ کر فرمایا کہ دنیا میں ایسے رہو گویا تم پردیسی ہو یا راہ گزرتا مسافر۔ آیت کا خاتمہ فَرِيْضَةً مِّنَ اللّٰهِ ؕ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ پر ہوا: یہ تقسیم علم سے ہے، کیونکہ وہی ہر حاجت مند کا حال جانتا ہے، اور حکمت سے ہے، کیونکہ وہی ہر شے کو اس کے مقام پر رکھتا ہے۔ اعتراض کرنے والوں کو جواب یہ ملا کہ جس تقسیم پر تم ناراض ہو وہ رسول صلی اللہ علیہ وسلم کی اپنی پسند نہیں، علیم و حکیم کا فریضہ ہے۔

اب رکوع ان لوگوں کی طرف مڑتا ہے جن کی زبان حضورِ اکرم صلی اللہ علیہ وسلم کی نرمی کو کمزوری سمجھتی تھی: وَمِنْهُمُ الَّذِيْنَ يُؤْذُوْنَ النَّبِيَّ وَيَقُوْلُوْنَ هُوَ اُذُنٌ ؕ قُلْ اُذُنُ خَيْرٍ لَّكُمْ يُؤْمِنُ بِاللّٰهِ وَيُؤْمِنُ لِلْمُؤْمِنِيْنَ وَرَحْمَةٌ لِّلَّذِيْنَ اٰمَنُوْا مِنْكُمْ ؕ وَالَّذِيْنَ يُؤْذُوْنَ رَسُوْلَ اللّٰهِ لَهُمْ عَذَابٌ اَلِيْمٌ۔ “اور ان میں سے کچھ وہ ہیں جو نبی صلی اللہ علیہ وسلم کو ایذا دیتے ہیں اور کہتے ہیں کہ یہ تو نرے کان ہیں۔ کہہ دیجیے: وہ تمہارے لیے بھلائی کے کان ہیں؛ وہ اللہ پر ایمان رکھتے ہیں، اور مومنوں کی بات کا یقین کرتے ہیں، اور تم میں سے جو ایمان لائے ان کے لیے رحمت ہیں۔ اور جو لوگ اللہ کے رسول صلی اللہ علیہ وسلم کو ایذا دیتے ہیں ان کے لیے دردناک عذاب ہے۔” مفسرین نے نقل کیا ہے کہ بعض منافق آپس میں حضورِ اکرم صلی اللہ علیہ وسلم کے بارے میں ناروا باتیں کرتے، اور جب کوئی انہیں ٹوکتا کہ یہ بات آپ صلی اللہ علیہ وسلم تک پہنچ جائے گی، تو کہتے کہ وہ تو کان ہیں؛ ہم جا کر قسم کھا لیں گے اور وہ ہماری بات مان لیں گے۔ ان کے نزدیک یہ ایک طعنہ تھا: ایسا سننے والا جو ہر بات پر یقین کر لے۔ مگر ان کے اس طعنے میں خود ان کا اقرار چھپا تھا: وہ جانتے تھے کہ حضورِ اکرم صلی اللہ علیہ وسلم ان کی قسم کو قبول فرما لیں گے اور ان کا پردہ نہیں چاک کریں گے۔ جس نرمی پر وہ ہنستے تھے، اسی نرمی کی چھاؤں میں وہ محفوظ بیٹھے تھے۔

جواب کا اسلوب دیکھیے۔ قرآن نے “کان” کے لفظ کو رد نہیں کیا، اسے لے کر پلٹ دیا: قُلْ اُذُنُ خَيْرٍ لَّكُمْ۔ ہاں، وہ کان ہیں، مگر بھلائی کے کان؛ ان کا سننا تمہارے لیے خیر ہے، شر نہیں۔ وہ سب کی سنتے ہیں، ظاہر کو قبول فرماتے ہیں، اور باطن کو اللہ کے سپرد کرتے ہیں۔ صحیح بخاری اور صحیح مسلم میں حضرت ابو سعید خدری رضی اللہ عنہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: مجھے یہ حکم نہیں دیا گیا کہ لوگوں کے دل کریدوں اور ان کے پیٹ چاک کروں۔ یہ سادگی نہ تھی، حلم اور حیا تھا۔ آپ صلی اللہ علیہ وسلم بہت کچھ جانتے تھے اور بہت کچھ سے چشم پوشی فرماتے تھے، جیسا کہ قرآن نے ایک اور موقع پر آپ صلی اللہ علیہ وسلم کا یہی انداز بیان فرمایا: عَرَّفَ بَعْضَهٗ وَاَعْرَضَ عَنْ بَعْضٍ۔ اہلِ معرفت کہتے ہیں کہ کریم وہ ہے جو جانتا ہو اور پھر بھی پردہ رکھے؛ جو نہ جانتا ہو اس کا پردہ رکھنا کرم نہیں، بے خبری ہے۔ منافق نے حلم کو غفلت سمجھا، اور یہی اس کی سب سے بڑی غفلت تھی۔ شیخِ اکبر کے نزدیک حلم ان اسماء کا مظہر ہے جو گرفت پر قادر ہو کر اسے مؤخر رکھتے ہیں؛ اور حضورِ اکرم صلی اللہ علیہ وسلم میں حق کے یہ اوصاف اپنے کامل ترین انسانی رنگ میں ظاہر ہوئے۔

پھر آیت نے اس کان کی تین صفات بیان کیں، اور ہر صفت ایک دروازہ ہے: يُؤْمِنُ بِاللّٰهِ وَيُؤْمِنُ لِلْمُؤْمِنِيْنَ۔ اہلِ علم نے اس باریکی پر توجہ دلائی ہے کہ اللہ کے ساتھ “بِ” آیا اور مومنوں کے ساتھ “لِ”۔ اللہ پر ایمان تصدیق اور یقین کا ایمان ہے، جس میں کوئی شبہ نہیں؛ اور مومنوں کے لیے ایمان یہ ہے کہ آپ صلی اللہ علیہ وسلم ان کی بات کا اعتبار فرماتے ہیں، ان کی گواہی کو مانتے ہیں، ان پر بھروسا رکھتے ہیں۔ پہلا ایمان عقیدہ ہے اور دوسرا اعتماد؛ پہلا حق کی طرف ہے اور دوسرا مخلوق کی طرف، اور دونوں کی ترتیب بتاتی ہے کہ جو اللہ پر سچا ایمان رکھے وہی مخلوق پر صحیح اعتماد کر سکتا ہے۔ گویا جواب یہ ہوا کہ آپ صلی اللہ علیہ وسلم ہر سنی بات پر یقین نہیں فرماتے؛ آپ صلی اللہ علیہ وسلم کا یقین اللہ پر ہے، اور اعتماد اہلِ ایمان پر۔ منافق کی قسم کو قبول کرنا پردہ پوشی تھی، تصدیق نہیں۔ شیخِ اکبر کے مکتب میں انسانِ کامل کا سمع پہلے حق سے سنتا ہے، پھر خلق سے؛ جو کان وحی کا ظرف ہو، وہ خلق کی بات بھی اسی نور میں تولتا ہے۔ حدیثِ قربِ نوافل کا مفہوم یہ ہے کہ بندہ نوافل سے قریب ہوتا جائے تو حق اس کا وہ کان بن جاتا ہے جس سے وہ سنتا ہے؛ اور یہ قرب نہ حلول ہے نہ اتحاد، بلکہ یہ کہ بندے کا سننا حق کی رضا کے سوا کسی اور کا تابع نہ رہے۔ جب ایک ولی کا سمع یہ ہو، تو سید الاولین والآخرین صلی اللہ علیہ وسلم کے سمع کا کیا عالم ہو گا۔

تیسری صفت سب سے روشن ہے: وَرَحْمَةٌ لِّلَّذِيْنَ اٰمَنُوْا مِنْكُمْ۔ آیت نے یہ نہیں کہا کہ وہ رحمت کرنے والے ہیں، کہا کہ وہ خود رحمت ہیں؛ آپ صلی اللہ علیہ وسلم کا سننا، آپ صلی اللہ علیہ وسلم کی خاموشی، آپ صلی اللہ علیہ وسلم کی چشم پوشی، سب رحمت کی صورتیں ہیں۔ اہلِ معرفت کہتے ہیں کہ یہاں رحمت کو اہلِ ایمان کے ساتھ خاص کیا گیا، کیونکہ اسی رحمت کا ایک عام رخ بھی ہے جس کے سائے میں منافق بھی محفوظ بیٹھا تھا؛ مگر اس رحمت کو اپنے اندر قبول کرنا ایمان کا کام ہے۔ سورج سب پر چمکتا ہے، مگر روشنی وہی پاتا ہے جو کھڑکی کھولے۔ سورۂ الحاقہ میں ایسے کان کو اُذُنٌ وَّاعِيَةٌ کہا گیا تھا، یعنی وہ کان جو سن کر محفوظ رکھے؛ اور یہاں وہ کان دکھایا گیا جو سن کر رحم کرے۔ آیت کا آخری جملہ وَالَّذِيْنَ يُؤْذُوْنَ رَسُوْلَ اللّٰهِ لَهُمْ عَذَابٌ اَلِيْمٌ اسی رحمت کی غیرت ہے: جس ذات کو حق نے رحمت بنا کر بھیجا، اسے ایذا دینا رحمت کے دروازے پر پتھر مارنا ہے؛ اور جو اس دروازے کو خود بند کرے وہ اپنے لیے درد کا انتخاب کرتا ہے۔ یہاں بھی حکم محکوم کے مطابق ہے: عذاب ایذا کا نتیجہ ہے، ابتدا نہیں۔

اس آیت کو اپنے اندر کا آئینہ بنائیے۔ اہلِ ذوق کہتے ہیں کہ ہر دل کے ساتھ ایک کان لگا ہے، اور سوال یہ ہے کہ وہ کس کا کان ہے: خیر کا یا شر کا۔ کچھ کان غیبت کے لیے کھلے رہتے ہیں اور نصیحت پر بند ہو جاتے ہیں؛ کچھ کان ہر بات میں عیب ڈھونڈتے ہیں؛ اور کچھ کان وہ ہیں جو ہر آواز میں حق کی کوئی بات سن لیتے ہیں۔ “اذنِ خیر” بننے کا مطلب یہ ہے کہ بندہ بات کو اس کے بہترین محمل پر رکھے، بھائی کا عذر قبول کرے، اور جو سنے اسے آگے پھیلانے سے پہلے تولے۔ اور اس آیت میں منافق کے رویے کا ایک باریک آئینہ بھی ہے جو ہمیں اپنی طرف دیکھنے کو کہتا ہے: حق تعالیٰ کا حلم بھی بندے کو مہلت دیتا ہے، اور غافل نفس اس مہلت کو اجازت سمجھ لیتا ہے، جیسے منافق نے حضورِ اکرم صلی اللہ علیہ وسلم کی نرمی کو اپنی چالاکی کی کامیابی سمجھا۔ یہ آئینہ کسی اور کے چہرے کے لیے نہیں، اپنے چہرے کے لیے ہے؛ آج کسی شخص یا گروہ پر یہ لفظ چسپاں کرنا اس رکوع کا منشا نہیں۔ اس کا منشا یہ ہے کہ ہم اپنے دل سے پوچھیں کہ جب کوئی ہمارا عیب جان کر بھی پردہ رکھے تو ہم اسے کرم سمجھتے ہیں یا غفلت۔

اس کے بعد ان کی قسموں کا ذکر آیا: يَحْلِفُوْنَ بِاللّٰهِ لَكُمْ لِيُرْضُوْكُمْ ۚ وَاللّٰهُ وَرَسُوْلُهٗۤ اَحَقُّ اَنْ يُّرْضُوْهُ اِنْ كَانُوْا مُؤْمِنِيْنَ۔ “وہ تمہارے سامنے اللہ کی قسمیں کھاتے ہیں تاکہ تمہیں راضی کر لیں، حالانکہ اللہ اور اس کا رسول صلی اللہ علیہ وسلم اس کے زیادہ حق دار ہیں کہ وہ انہیں راضی کریں، اگر وہ مومن ہیں۔” قسم اللہ کی، مقصود انسان کی رضا؛ یہی نفاق کی جڑ ہے کہ حق کا نام مخلوق کو خوش کرنے کا آلہ بن جائے۔ اہلِ علم نے یہ لطیف بات بیان کی ہے کہ اللہ اور رسول صلی اللہ علیہ وسلم دونوں کا ذکر ہوا مگر ضمیر واحد آئی: اَحَقُّ اَنْ يُّرْضُوْهُ، نہ کہ دونوں کے لیے الگ ضمیر؛ کیونکہ دونوں کی رضا ایک ہے، رسول صلی اللہ علیہ وسلم کو راضی کرنا اللہ ہی کو راضی کرنا ہے، جیسا کہ فرمایا: مَنْ يُّطِعِ الرَّسُوْلَ فَقَدْ اَطَاعَ اللّٰهَ۔ شیخِ اکبر کے مکتب میں یہ وحدتِ رضا اس حقیقت کا ظہور ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم حق کے کامل ترین مظہر اور اس کی مرضی کے سب سے سچے ترجمان ہیں؛ مگر اس کے ساتھ تنزیہ پوری طرح قائم ہے: آپ صلی اللہ علیہ وسلم اللہ کے بندے اور اس کے رسول ہیں، اور یہاں رضا کی یکجائی ہے، ذات کی نہیں۔ جملے کا آخری ٹکڑا اِنْ كَانُوْا مُؤْمِنِيْنَ ایک کسوٹی ہے: ایمان کی پہچان یہ ہے کہ بندہ کس کی رضا کو مقدم رکھتا ہے۔

پھر انجام کی خبر آئی: اَلَمْ يَعْلَمُوْۤا اَنَّهٗ مَنْ يُّحَادِدِ اللّٰهَ وَرَسُوْلَهٗ فَاَنَّ لَهٗ نَارَ جَهَنَّمَ خٰلِدًا فِيْهَا ؕ ذٰلِكَ الْخِزْيُ الْعَظِيْمُ۔ “کیا وہ نہیں جانتے کہ جو اللہ اور اس کے رسول صلی اللہ علیہ وسلم کی مخالفت کرے، اس کے لیے جہنم کی آگ ہے جس میں وہ ہمیشہ رہے گا؟ یہی بڑی رسوائی ہے۔” “محادّت” حد سے ہے: ایک کنارے سے دوسرے کنارے پر جا کھڑا ہونا۔ اہلِ معرفت کہتے ہیں کہ مخالف وہ ہے جس نے اپنا کنارہ خود چنا؛ حق نے اسے اس سرحد پر نہیں دھکیلا، وہ خود اس پار گیا، اور آگ اسی کنارے کا نام ہے۔ یہی سرِّ قدر ہے: حکم بندے کے اپنے رخ کے مطابق۔ اور یہاں ایک نفسیاتی گرہ بھی کھلتی ہے: جو لوگوں کو قسموں سے راضی کرنے میں لگا رہا، اس کا انجام رسوائی بتایا گیا، یعنی جس عزت کے لیے اس نے حق کو چھوڑا تھا وہی اس سے چھن گئی۔ جامع ترمذی میں ام المومنین حضرت عائشہ صدیقہ رضی اللہ عنہا سے مروی حدیث کا مفہوم ہے کہ جو لوگوں کی ناراضی کے باوجود اللہ کی رضا ڈھونڈے، اللہ اسے لوگوں کے بوجھ سے کافی ہو جاتا ہے، اور جو اللہ کی ناراضی کے ساتھ لوگوں کی رضا ڈھونڈے، اللہ اسے لوگوں کے سپرد کر دیتا ہے۔ اہلِ ذوق اس میں اپنے لیے یہ سبق لیتے ہیں کہ ہر بندے کے اندر ایک چہرہ لوگوں کے لیے بنتا ہے، اور اصل امتحان یہ ہے کہ وہ چہرہ دل کے سچ سے کتنا قریب ہے۔

اگلی آیت اس خوف کو بے نقاب کرتی ہے جو ہر دوہرے دل میں چھپا رہتا ہے: يَحْذَرُ الْمُنٰفِقُوْنَ اَنْ تُنَزَّلَ عَلَيْهِمْ سُوْرَةٌ تُنَبِّئُهُمْ بِمَا فِيْ قُلُوْبِهِمْ ؕ قُلِ اسْتَهْزِءُوْا ۚ اِنَّ اللّٰهَ مُخْرِجٌ مَّا تَحْذَرُوْنَ۔ “منافق اس بات سے ڈرتے ہیں کہ ان کے بارے میں کوئی ایسی سورت نازل ہو جائے جو انہیں ان کے دلوں کی بات بتا دے۔ کہہ دیجیے: مذاق کرتے رہو، اللہ یقیناً وہ ظاہر کر کے رہے گا جس سے تم ڈرتے ہو۔” صحیح بخاری میں حضرت سعید بن جبیر رحمۃ اللہ علیہ سے مروی ہے کہ انہوں نے حضرت عبد اللہ بن عباس رضی اللہ عنہما سے سورۂ توبہ کے بارے میں پوچھا تو انہوں نے فرمایا کہ یہ الفاضحہ، یعنی رسوا کرنے والی سورت ہے؛ یہ “اور ان میں سے، اور ان میں سے” کہتی چلی گئی یہاں تک کہ انہیں گمان ہوا کہ کوئی باقی نہ بچے گا جس کا ذکر نہ آئے۔ مگر یہاں ایک عجیب کرم دیکھیے: سورت نے صفات کھول دیں، نام نہیں کھولے۔ دلوں کا حال بتا دیا اور پردہ پھر بھی رکھا، تاکہ ہر سننے والا اپنی طرف دیکھے اور جس کے دل میں وہ صفت ہو وہ توبہ کر لے۔ یہی اس سورت کا ادب ہے، اور یہی اس کے پڑھنے والے کا ادب ہونا چاہیے: صفت کو اپنے اندر ڈھونڈنا، کسی دوسرے کا نام نہ رکھنا۔

اِنَّ اللّٰهَ مُخْرِجٌ مَّا تَحْذَرُوْنَ، اس جملے میں حق کے دو نام ایک ساتھ جلوہ گر ہیں: الباطن، جو دل کی تہہ کو جانتا ہے، اور الظاہر، جو اسے باہر لے آتا ہے۔ شیخِ اکبر کے نزدیک ہر باطن کا ایک ظہور ہے، اور جو کچھ دل میں جمع کیا جاتا ہے وہ کسی نہ کسی عالم میں اپنی صورت پا کر رہتا ہے؛ دنیا میں نہیں تو برزخ میں، اور برزخ میں نہیں تو اس دن جس کے بارے میں سورۂ الطارق میں فرمایا گیا: يَوْمَ تُبْلَى السَّرَآئِرُ۔ پس چھپانا ممکن ہی نہیں، صرف ظہور کی جگہ اور وقت کا انتخاب بندے کے ہاتھ میں ہے۔ اہلِ معرفت کہتے ہیں کہ عقل مند وہ ہے جو اپنے باطن کو خود اپنے رب کے سامنے نکال کر رکھ دے، اس سے پہلے کہ وہ اسے سب کے سامنے نکالے؛ اسی کا نام توبہ اور محاسبہ ہے۔ منافق اس سورت سے ڈرتا تھا جو اس کا دل کھول دے، اور مومن خود اپنا دل کھول کر قرآن کے سامنے رکھ دیتا ہے کہ وہ اس کا علاج کرے۔ ڈر ایک ہی ہے، رخ مختلف: ایک بھاگتا ہے، دوسرا پاس آتا ہے۔

اب وہ واقعہ جس میں مذاق کی حقیقت کھولی گئی: وَلَئِنْ سَاَلْتَهُمْ لَيَقُوْلُنَّ اِنَّمَا كُنَّا نَخُوْضُ وَنَلْعَبُ ؕ قُلْ اَبِاللّٰهِ وَاٰيٰتِهٖ وَرَسُوْلِهٖ كُنْتُمْ تَسْتَهْزِءُوْنَ۔ “اور اگر آپ صلی اللہ علیہ وسلم ان سے پوچھیں تو ضرور کہیں گے کہ ہم تو یوں ہی باتوں میں لگے تھے اور دل لگی کر رہے تھے۔ کہہ دیجیے: کیا تم اللہ، اس کی آیتوں اور اس کے رسول صلی اللہ علیہ وسلم کے ساتھ ہنسی کرتے تھے؟” مفسرین نے غزوۂ تبوک کے سفر کا ایک واقعہ روایت کیا ہے کہ ایک مجلس میں کسی نے حضورِ اکرم صلی اللہ علیہ وسلم کے قاری صحابہ رضی اللہ عنہم پر طعن کیا کہ ان سے بڑھ کر پیٹ کا لالچی، زبان کا جھوٹا اور دشمن کے سامنے بزدل ہم نے کسی کو نہیں دیکھا۔ ایک صحابی رضی اللہ عنہ یہ خبر لے کر حاضر ہوئے تو وحی پہلے پہنچ چکی تھی۔ وہ شخص آیا تو حضورِ اکرم صلی اللہ علیہ وسلم اونٹنی پر سوار ہو چکے تھے؛ روایت ہے کہ وہ اونٹنی کی رسی تھامے ساتھ چلتا اور کہتا جاتا کہ ہم تو یوں ہی باتیں اور دل لگی کر رہے تھے، اور آپ صلی اللہ علیہ وسلم اس کی طرف التفات کیے بغیر صرف یہی آیت تلاوت فرماتے رہے۔ آپ صلی اللہ علیہ وسلم نے اپنی طرف سے کچھ نہ کہا؛ جواب قرآن کا تھا، اور یہی آپ صلی اللہ علیہ وسلم کا ادب تھا۔

پھر فیصلہ آیا: لَا تَعْتَذِرُوْا قَدْ كَفَرْتُمْ بَعْدَ اِيْمَانِكُمْ ؕ اِنْ نَّعْفُ عَنْ طَآئِفَةٍ مِّنْكُمْ نُعَذِّبْ طَآئِفَةًۢ بِاَنَّهُمْ كَانُوْا مُجْرِمِيْنَ۔ “بہانے نہ بناؤ؛ تم اپنے ایمان کے بعد کفر کر چکے ہو۔ اگر ہم تم میں سے ایک گروہ کو معاف کر دیں تو دوسرے گروہ کو عذاب دیں گے، اس لیے کہ وہ مجرم تھے۔” یہاں ادب کی ایک بنیادی بات یاد رہے: یہ حکم وحی نے مخصوص لوگوں کے بارے میں دیا جن کے دلوں کا حال حق تعالیٰ جانتا تھا؛ کسی فرد یا جماعت کو یہ حق نہیں کہ اس آیت کو لے کر آج کسی پر کفر کا فتویٰ لگائے۔ مفسرین نے یہ بھی نقل کیا ہے کہ اس مجلس کا ایک شخص سچے دل سے تائب ہوا اور معافی پا گیا؛ اِنْ نَّعْفُ عَنْ طَآئِفَةٍ مِّنْكُمْ میں عفو کا دروازہ اسی سخت گھڑی میں بھی کھلا رکھا گیا۔ اور جن پر عذاب کا ذکر آیا، ان کا سبب خود آیت نے بتا دیا: بِاَنَّهُمْ كَانُوْا مُجْرِمِيْنَ۔ عذاب کسی جبر کا نتیجہ نہیں، ان کے اپنے جرم پر اصرار کا نتیجہ ہے؛ جس نے لوٹنا چاہا اسے معافی ملی، اور جس نے اپنی روش نہ چھوڑی اسے اسی روش کا پھل ملا۔ یہی اس سورت کا اصول ہے: حق کسی طالب کو رد نہیں کرتا، اور اعراض ہی وہ پردہ ہے جو بندہ خود اپنے اوپر ڈالتا ہے۔

اس واقعے کا باطنی سبق ہر زبان کے لیے ہے۔ صحیح بخاری میں حضرت ابوہریرہ رضی اللہ عنہ سے مروی حدیث کا مفہوم ہے کہ بندہ اللہ کی ناراضی کی کوئی بات بے پروائی سے کہہ دیتا ہے اور اسی کے سبب دور جا گرتا ہے۔ “ہم تو بس دل لگی کر رہے تھے” وہ جملہ ہے جو ہر دور کا نفس اپنے بچاؤ کے لیے بولتا ہے؛ مگر قرآن نے بتا دیا کہ مقدس چیز سے کھیلنا کھیل نہیں رہتا۔ شیخِ اکبر کے نزدیک کلمہ نفَسِ رحمانی سے ظاہر ہوتا ہے، اور ہر بولا ہوا لفظ ایک وجود رکھتا ہے جو اپنا اثر چھوڑتا ہے؛ جو زبان اللہ، اس کی آیات اور اس کے رسول صلی اللہ علیہ وسلم کے ذکر میں کھیل شامل کرے، وہ مراتب کو گڈمڈ کرتی ہے، اور یہ بے ادبی دل کو اندر سے کھوکھلا کر دیتی ہے۔ اسی لیے عذر کو بھی رد کیا گیا: لَا تَعْتَذِرُوْا۔ اہلِ معرفت کہتے ہیں کہ عذر اور توبہ میں فرق ہے: عذر اپنے آپ کو بچاتا ہے اور توبہ اپنے آپ کو حوالے کرتی ہے؛ پہلا کہتا ہے کہ میں نے کچھ غلط نہیں کیا، دوسرا کہتا ہے کہ میں نے غلط کیا، مجھے معاف کر دیجیے۔ اسی لیے جس نے توبہ کی وہ معاف ہوا، اور جس نے عذر کیا وہ اپنے عذر میں قید رہ گیا۔ یہ آئینہ ہمارے سامنے ہے: ہم میں سے کوئی جب ہنسی میں حد پار کرے تو بہانے کے بجائے لوٹ آئے، کیونکہ عفو کا وعدہ لوٹنے والوں سے ہے۔
