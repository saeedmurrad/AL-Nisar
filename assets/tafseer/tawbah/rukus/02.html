یہ رکوع پچھلے اعلان کی وجوہ کھولتا ہے، اور اس کا آغاز پھر ایک استثنا سے ہوتا ہے: كَيْفَ يَكُوْنُ لِلْمُشْرِكِيْنَ عَهْدٌ عِنْدَ اللّٰهِ وَعِنْدَ رَسُوْلِهٖۤ اِلَّا الَّذِيْنَ عٰهَدْتُّمْ عِنْدَ الْمَسْجِدِ الْحَرَامِ ۚ فَمَا اسْتَقَامُوْا لَكُمْ فَاسْتَقِيْمُوْا لَهُمْ ؕ اِنَّ اللّٰهَ يُحِبُّ الْمُتَّقِيْنَ۔ “ان مشرکوں کا اللہ اور اس کے رسول صلی اللہ علیہ وسلم کے نزدیک کوئی عہد کیسے رہ سکتا ہے، سوائے ان لوگوں کے جن سے تم نے مسجدِ حرام کے پاس معاہدہ کیا؛ تو جب تک وہ تمہارے ساتھ سیدھے رہیں تم بھی ان کے ساتھ سیدھے رہو، بے شک اللہ پرہیزگاروں سے محبت فرماتا ہے۔” مفسرین کے نزدیک مسجدِ حرام کے پاس کا معاہدہ صلحِ حدیبیہ کی طرف اشارہ ہے، اور اس سے وہ قبائل مراد ہیں جو اس معاہدے پر قائم رہے۔ “کیسے” کا سوال انہی کے بارے میں ہے جن کی عہد شکنی کا حال آگے کی آیات میں بیان ہو رہا ہے۔ اور حکم دو ٹوک ہے: جب تک وہ سیدھے رہیں تم بھی سیدھے رہو۔ وفا کا معیار فریقِ ثانی کا مذہب نہیں، اس کا رویہ ہے؛ اور جو وفا کرے اس کے ساتھ وفا کرنا تقویٰ ہے اور اللہ کی محبت کا سبب۔ پچھلے رکوع کی آیت چار کے بعد یہ دوسری بار ہے کہ عہد نبھانے والوں کا استثنا اسی جملے پر ختم ہوا کہ اِنَّ اللّٰهَ يُحِبُّ الْمُتَّقِيْنَ؛ گویا قرآن چاہتا ہے کہ قاری یہ استثنا کبھی نہ بھولے۔

اہلِ معرفت فَمَا اسْتَقَامُوْا لَكُمْ فَاسْتَقِيْمُوْا لَهُمْ میں ایک بڑا قانون دیکھتے ہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کہا جاتا ہے کہ عالم ایک آئینہ ہے، اور حق بندے کے ساتھ اسی صورت میں معاملہ فرماتا ہے جس صورت میں بندہ اس کی طرف رخ کرتا ہے؛ یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہوتا ہے۔ اسی کا ایک بیان یہ آیت ہے: فَاذْكُرُوْنِيْۤ اَذْكُرْكُمْ۔ اہلِ ذوق کہتے ہیں کہ مخلوق کے ساتھ استقامت کا یہ حکم دراصل اس بڑی استقامت کی تربیت ہے جو بندے کو حق کے ساتھ رکھنی ہے۔ استقامت کا معنی ہے سیدھا کھڑا رہنا، نہ جھکنا نہ ٹیڑھا ہونا؛ اور شیخِ اکبر کے نزدیک ہر شے اپنے ربِّ خاص کے ساتھ ایک طرح کی استقامت پر ہے، مگر انسان کا کمال یہ ہے کہ وہ اسے شعور اور اختیار کے ساتھ اپنائے۔ جو بندہ عہد کا پاس رکھتا ہے وہ گویا اپنے وجود کو ایک سیدھی لکیر پر قائم کرتا ہے جس پر اعتماد کیا جا سکے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر میں حکم معاہدوں کی پاسداری کا ہے، اور یہ حکم آج بھی ہر مسلمان کے کردار کی بنیاد ہے کہ وہ کسی سے کیے ہوئے عہد میں خیانت نہ کرے۔

اب ان لوگوں کا حال بیان ہوا جن سے عہد ختم کیا گیا: كَيْفَ وَاِنْ يَّظْهَرُوْا عَلَيْكُمْ لَا يَرْقُبُوْا فِيْكُمْ اِلًّا وَّلَا ذِمَّةً ؕ يُرْضُوْنَكُمْ بِاَفْوَاهِهِمْ وَتَاْبٰى قُلُوْبُهُمْ ۚ وَاَكْثَرُهُمْ فٰسِقُوْنَ۔ “کیسے، جبکہ حال یہ ہے کہ اگر وہ تم پر غالب آ جائیں تو تمہارے بارے میں نہ کسی قرابت کا لحاظ کریں نہ کسی عہد کا؛ وہ تمہیں اپنے منہ سے راضی کرتے ہیں اور ان کے دل انکار کرتے ہیں، اور ان میں سے اکثر نافرمان ہیں۔” الّ کا معنی مفسرین نے قرابت بھی بیان کیا اور حلف و عہد بھی، اور ذمہ کا معنی عہد و پیمان۔ یہ ان مخصوص قبائل کی تصویر ہے جنہوں نے بار بار ثابت کیا کہ ان کے معاہدے صرف کمزوری کے وقت کی ڈھال ہیں۔ اور قرآن کا انصاف دیکھیے کہ دشمن کے بیان میں بھی “اکثر” کہا، “سب” نہیں کہا؛ اسی لیے ان ہی میں سے بہت سے لوگ بعد میں ایمان لائے۔ اہلِ معرفت اس آیت کو آئینے کی طرح اپنے سامنے رکھتے ہیں۔ يُرْضُوْنَكُمْ بِاَفْوَاهِهِمْ وَتَاْبٰى قُلُوْبُهُمْ ہر اس حال کی تصویر ہے جس میں زبان کچھ کہے اور دل کچھ اور چاہے۔ سالک اپنے آپ سے پوچھتا ہے کہ کیا میری زبان کا ذکر میرے دل کی رضا بھی ہے؟ شیخِ اکبر کے نزدیک قلب کا نام ہی تقلب سے ہے، اور جس دل کا رخ زبان کے خلاف ہو وہ دراصل کسی اور قبلے کی طرف پلٹا ہوا ہے۔ اور فسق کا معنی ہے حد سے نکل جانا؛ یہ نکلنا ان کا اپنا تھا، کسی نے انہیں دھکیل کر باہر نہیں کیا۔

پھر ان کی اصل بیماری کا ذکر ہوا: اِشْتَرَوْا بِاٰيٰتِ اللّٰهِ ثَمَنًا قَلِيْلًا فَصَدُّوْا عَنْ سَبِيْلِهٖ ؕ اِنَّهُمْ سَآءَ مَا كَانُوْا يَعْمَلُوْنَ۔ “انہوں نے اللہ کی آیتوں کے بدلے تھوڑی سی قیمت لے لی، پھر اس کے راستے سے روکا؛ بے شک بہت برا ہے جو وہ کرتے رہے ہیں۔” اور اس کے ساتھ: لَا يَرْقُبُوْنَ فِيْ مُؤْمِنٍ اِلًّا وَّلَا ذِمَّةً ؕ وَاُولٰۤئِكَ هُمُ الْمُعْتَدُوْنَ۔ “وہ کسی مومن کے بارے میں نہ قرابت کا لحاظ کرتے ہیں نہ عہد کا، اور یہی لوگ زیادتی کرنے والے ہیں۔” مفسرین بیان کرتے ہیں کہ ان لوگوں نے دنیا کے معمولی فائدے، سرداری اور مال کی خاطر حق کو ٹھکرایا، اور صرف ٹھکرایا نہیں بلکہ دوسروں کو بھی اس راہ سے روکا۔ آیت دس کا آخری لفظ پورے حکم کی بنیاد واضح کر دیتا ہے: المعتدون، یعنی زیادتی میں پہل کرنے والے۔ ان آیات کے سیاق میں قرآن نے اس حکم کی بنیاد ان کی زیادتی اور عہد شکنی کو بتایا ہے۔ اہلِ معرفت کہتے ہیں کہ آیات صرف مصحف کے صفحات میں نہیں، آفاق اور انفس میں بھی بکھری ہیں، اور ہر وہ شخص جو کسی فانی چیز کو حق کی کسی نشانی پر ترجیح دیتا ہے، اپنے درجے میں یہی سودا کرتا ہے۔ شیخِ اکبر کے نزدیک حق کے سوا جو کچھ ہے وہ اس کے مقابلے میں ثمنِ قلیل ہے، کیونکہ فانی کو باقی کے بدلے لینا ہمیشہ گھاٹے کا سودا ہے۔ یہ اشارہ قاری کے اپنے دل کے لیے ہے، کسی اور پر فتویٰ کے لیے نہیں۔

اور پھر وہی دروازہ جو اس سورت میں کبھی بند نہیں ہوتا: فَاِنْ تَابُوْا وَاَقَامُوا الصَّلٰوةَ وَاٰتَوُا الزَّكٰوةَ فَاِخْوَانُكُمْ فِي الدِّيْنِ ؕ وَنُفَصِّلُ الْاٰيٰتِ لِقَوْمٍ يَّعْلَمُوْنَ۔ “پھر اگر وہ توبہ کر لیں، نماز قائم کریں اور زکوٰۃ دیں تو وہ دین میں تمہارے بھائی ہیں؛ اور ہم آیتیں جاننے والوں کے لیے کھول کھول کر بیان کرتے ہیں۔” کل تک جو قرابت اور عہد کا لحاظ نہ کرتے تھے، توبہ کے ایک لمحے میں وہ بھائی بن گئے؛ نہ کوئی درجہ بندی، نہ پرانی دشمنی کا طعنہ۔ تاریخ اس کی گواہ ہے کہ حضرت خالد بن ولید رضی اللہ عنہ اور حضرت عکرمہ بن ابی جہل رضی اللہ عنہ جیسے لوگ، جو کبھی مقابل صف میں تھے، اسلام لانے کے بعد اسی امت کے بھائی اور اسی دین کے محافظ بنے۔ ایک حدیث کا مفہوم یہ ہے کہ گناہ سے توبہ کرنے والا ایسا ہے جیسے اس نے گناہ کیا ہی نہ ہو۔ شیخِ اکبر کے نزدیک توبہ کا معنی رجوع ہے، اور رجوع اسی کی طرف ہوتا ہے جس سے بندہ نکلا تھا؛ ہر عین اپنی اصل میں حق کے علم کا حصہ ہے، اور توبہ اس اصل کی یاد ہے۔ اسی لیے توبہ کے بعد کی اخوت کوئی رسمی معاہدہ نہیں بلکہ ایک ہی چشمے سے نکلنے والی نہروں کا دوبارہ مل جانا ہے۔ اور لِقَوْمٍ يَّعْلَمُوْنَ پچھلے رکوع کے قَوْمٌ لَّا يَعْلَمُوْنَ کے مقابل ہے: جو نہیں جانتا اسے سنایا جاتا ہے، اور جو جانتا ہے اس کے لیے آیات کھولی جاتی ہیں۔

اب وہ صورت جس میں جنگ کا حکم دوبارہ آتا ہے، اور اس کی شرطیں غور سے پڑھنے کی ہیں: وَاِنْ نَّكَثُوْۤا اَيْمَانَهُمْ مِّنْ بَعْدِ عَهْدِهِمْ وَطَعَنُوْا فِيْ دِيْنِكُمْ فَقَاتِلُوْۤا اَئِمَّةَ الْكُفْرِ ۙ اِنَّهُمْ لَاۤ اَيْمَانَ لَهُمْ لَعَلَّهُمْ يَنْتَهُوْنَ۔ “اور اگر وہ اپنے عہد کے بعد اپنی قسمیں توڑ دیں اور تمہارے دین پر حملے کریں تو کفر کے ان سرغنوں سے لڑو، ان کی قسموں کا کوئی اعتبار نہیں، تاکہ وہ باز آ جائیں۔” حکم کی بنیاد پہلے عہد شکنی ہے اور پھر دین کے خلاف جارحانہ مہم؛ اور ہدف عام لوگ نہیں بلکہ ائمۂ کفر، یعنی وہ سردار جو عہد شکنی اور دشمنی کی قیادت کر رہے تھے۔ اور غایت بھی بیان ہو گئی: لَعَلَّهُمْ يَنْتَهُوْنَ، تاکہ وہ رک جائیں؛ مقصد زیادتی کا رک جانا ہے، انتقام نہیں۔ یہاں ایک بار پھر صراحت ضروری ہے: یہ احکام حضورِ اکرم صلی اللہ علیہ وسلم کے زمانے میں جزیرۂ عرب کے ان مخصوص عہد شکن قبائل کے بارے میں تھے، اور جنگ کا اعلان اور اس کا انتظام صرف جائز ریاست اور اس کے حکمران کا اختیار ہے۔ کوئی فرد یا گروہ ان آیات کو اپنے فیصلے کی سند نہیں بنا سکتا، اور نہ ان کا اطلاق آج کسی قوم یا کسی مذہب کے ماننے والوں پر جائز ہے۔ اہلِ تاویل باطن میں ائمۂ کفر سے نفس کی وہ بڑی خواہشیں مراد لیتے ہیں جو بار بار توبہ کا عہد توڑ کر دل پر حملہ کرتی ہیں؛ مگر یہ ایک اشاری معنی ہے جو ظاہری معنی کے ساتھ چلتا ہے، اس کی جگہ نہیں لیتا۔

پھر تین الزامات ایک ساتھ بیان ہوئے: اَلَا تُقَاتِلُوْنَ قَوْمًا نَّكَثُوْۤا اَيْمَانَهُمْ وَهَمُّوْا بِاِخْرَاجِ الرَّسُوْلِ وَهُمْ بَدَءُوْكُمْ اَوَّلَ مَرَّةٍ۔ “کیا تم ایسے لوگوں سے نہیں لڑو گے جنہوں نے اپنی قسمیں توڑ دیں، اور رسول صلی اللہ علیہ وسلم کو نکالنے کا ارادہ کیا، اور انہی نے تم سے پہلی بار ابتدا کی؟” بہت سے مفسرین بیان کرتے ہیں کہ قسمیں توڑنے سے مراد قریش کا صلحِ حدیبیہ توڑنا ہے، جب انہوں نے اپنے حلیف بنو بکر کی مدد کی اور حضورِ اکرم صلی اللہ علیہ وسلم کے حلیف قبیلے خزاعہ پر حملہ ہوا؛ رسول صلی اللہ علیہ وسلم کو نکالنے کے ارادے سے مراد ہجرت سے پہلے مکہ مکرمہ میں آپ صلی اللہ علیہ وسلم کے خلاف ان کی سازش ہے؛ اور پہلی بار ابتدا سے مراد یہ ہے کہ زیادتی کی پہل ہمیشہ انہی کی طرف سے ہوئی۔ تینوں باتیں ان کی اپنی زیادتی ہیں، اور ان کے مقابلے میں یہ حکم دفاعی ہے: عہد شکنی، جلاوطنی کی سازش، اور جارحیت میں پہل۔ اس آیت نے جنگ کا جواز بیان کرتے ہوئے ان کے اعمال گنوائے ہیں۔ اہلِ معرفت یہاں ایک لطیف نکتہ دیکھتے ہیں: حضورِ اکرم صلی اللہ علیہ وسلم کو مکہ مکرمہ سے نکالنے کا ارادہ دراصل اس نور کو اس کے مرکز سے ہٹانے کی کوشش تھی، مگر جس رات آپ صلی اللہ علیہ وسلم ہجرت کے لیے تشریف لے گئے اسی رات سے ایک نئی ریاست اور ایک نئی تاریخ کا آغاز ہوا۔ شیخِ اکبر کے نزدیک حضورِ اکرم صلی اللہ علیہ وسلم انسانِ کامل ہیں، اسمِ جامع کے کامل مظہر؛ اور جامع کو کسی ایک مکان سے نکال دینا اسے محدود نہیں کرتا، اس کے ظہور کو وسیع کر دیتا ہے۔

اور اسی آیت کا دوسرا حصہ اہلِ ایمان کے دل سے بات کرتا ہے: اَتَخْشَوْنَهُمْ ۚ فَاللّٰهُ اَحَقُّ اَنْ تَخْشَوْهُ اِنْ كُنْتُمْ مُّؤْمِنِيْنَ۔ “کیا تم ان سے ڈرتے ہو؟ تو اللہ زیادہ حق دار ہے کہ تم اس سے ڈرو، اگر تم ایمان والے ہو۔” یہ سوال ملامت سے زیادہ تربیت ہے۔ مسلمانوں کے سامنے ایک طاقتور اور تجربہ کار دشمن تھا، اور فطری طور پر دل میں اندیشہ اٹھ سکتا تھا؛ قرآن نے اس اندیشے کو ایک بڑے خوف سے بدل دیا۔ اہلِ معرفت کہتے ہیں کہ مخلوق کا خوف دل پر حجاب ہے، اور یہ حجاب اسی وقت اٹھتا ہے جب دل میں حق کی خشیت جگہ پا لے؛ پھر سارے چھوٹے خوف خود بخود اس ایک خشیت میں گم ہو جاتے ہیں۔ شیخِ اکبر کے مکتب میں خشیت کو اس خوف سے الگ کیا جاتا ہے جو کسی نقصان سے ہو؛ خشیت وہ ہیبت ہے جو جلالِ حق کی معرفت سے پیدا ہوتی ہے، اور اسی لیے اس کے ساتھ انس بھی جمع رہتا ہے۔ جو اللہ سے ڈرتا ہے وہ کسی اور سے نہیں ڈرتا، اور جو اللہ سے نہیں ڈرتا وہ ہر چیز سے ڈرتا ہے۔ اور اِنْ كُنْتُمْ مُّؤْمِنِيْنَ میں یہ اشارہ ہے کہ ایمان کی کسوٹی یہی ہے کہ دل میں کس کا خوف سب سے بڑا ہے۔ یہ سبق ہر مومن کے باطن کے لیے ہے، اور اس کا میدان اپنا دل ہے۔

پھر اس معرکے کے انجام کی خبر دی گئی: قَاتِلُوْهُمْ يُعَذِّبْهُمُ اللّٰهُ بِاَيْدِيْكُمْ وَيُخْزِهِمْ وَيَنْصُرْكُمْ عَلَيْهِمْ وَيَشْفِ صُدُوْرَ قَوْمٍ مُّؤْمِنِيْنَ۔ “ان سے لڑو، اللہ تمہارے ہاتھوں انہیں سزا دے گا، انہیں رسوا کرے گا، تمہیں ان پر غلبہ دے گا اور مومنوں کے ایک گروہ کے سینوں کو شفا دے گا۔” اور: وَيُذْهِبْ غَيْظَ قُلُوْبِهِمْ۔ “اور ان کے دلوں کی جلن دور کر دے گا۔” مفسرین میں سے بعض ائمۂ تفسیر نے ان سینوں سے قبیلہ خزاعہ کے لوگ مراد لیے، جن پر عہد شکنی کر کے حملہ کیا گیا تھا اور جن کے دل اس ظلم پر زخمی تھے۔ اہلِ معرفت يُعَذِّبْهُمُ اللّٰهُ بِاَيْدِيْكُمْ پر ٹھہرتے ہیں: فعل اللہ کا ہے اور ہاتھ بندوں کے۔ شیخِ اکبر کے نزدیک یہ تجلیِ افعالی کا مقام ہے، جہاں بندہ دیکھ لیتا ہے کہ حقیقی فاعل حق ہے اور بندے کا ہاتھ اس کے فعل کا مظہر؛ اور یہ دیکھنا بندے سے ذمہ داری نہیں اٹھاتا بلکہ اس سے غرور اٹھا لیتا ہے۔ اور سینوں کی شفا اور دلوں کی جلن کا دور ہونا بتاتا ہے کہ مظلوم کے دل کا زخم بھی حق کی نظر میں ہے؛ مگر یہ شفا آخرکار انتقام سے نہیں بلکہ عدل کے قیام اور سکینت کے نزول سے آئی۔ صدر اور قلب کا یہ فرق بھی قابلِ غور ہے: شفا صدر کو ملی جو ظاہر کے قریب ہے، اور غیظ قلب سے دور ہوا جو باطن کا مرکز ہے۔

اور اسی آیت میں وہ جملہ ہے جس نے پورا منظر بدل دیا: وَيَتُوْبُ اللّٰهُ عَلٰى مَنْ يَّشَآءُ ؕ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ “اور اللہ جس کی چاہے توبہ قبول فرماتا ہے، اور اللہ جاننے والا، حکمت والا ہے۔” یہ جملہ جنگ کے حکم کے بیچ آیا، اور فتحِ مکہ کا منظر اس کی روشن تفسیر ہے۔ آٹھ ہجری میں جب مکہ مکرمہ فتح ہوا تو انہی قریش میں سے بہت بڑی تعداد اسلام لے آئی، اور سیرت کی روایات میں مشہور ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے اس دن عام معافی کا اعلان فرمایا اور حضرت یوسف علیہ السلام کے الفاظ دہرائے: لَا تَثْرِيْبَ عَلَيْكُمُ الْيَوْمَ۔ جن سے لڑنے کا حکم تھا انہی کی توبہ قبول ہوئی، اور سینوں کی اصل شفا یہی تھی۔ یہاں اس سورت کا بنیادی اصول بھی سامنے آتا ہے۔ “جس کی چاہے” کا مطلب یہ نہیں کہ حق کسی طالب کو لوٹا دیتا ہے یا کسی پر زبردستی محرومی ڈالتا ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے کہ اس کی مشیت بندے کی حقیقت اور اس کے رخ کے مطابق ہے؛ جو اس کی طرف پلٹا اس نے اسے اپنی طرف پلٹتا پایا، اور جس نے منہ پھیرا اس کا منہ پھیرنا اس کا اپنا تھا۔ اور حق کی توبہ، یعنی اس کا بندے کی طرف رجوع، ہمیشہ بندے کی توبہ سے پہلے ہوتی ہے؛ وہی پہلے متوجہ ہوتا ہے تو بندے کو لوٹنے کی توفیق ملتی ہے۔ اور آیت عَلِيْمٌ حَكِيْمٌ پر ختم ہوئی: وہ جانتا ہے کہ کس دل میں لوٹنے کی طلب چھپی ہے، اور اپنی حکمت سے اس طلب کو اس کے وقت پر ظاہر کرتا ہے۔

رکوع کا اختتام اہلِ ایمان کی آزمائش پر ہوا: اَمْ حَسِبْتُمْ اَنْ تُتْرَكُوْا وَلَمَّا يَعْلَمِ اللّٰهُ الَّذِيْنَ جٰهَدُوْا مِنْكُمْ وَلَمْ يَتَّخِذُوْا مِنْ دُوْنِ اللّٰهِ وَلَا رَسُوْلِهٖ وَلَا الْمُؤْمِنِيْنَ وَلِيْجَةً ؕ وَاللّٰهُ خَبِيْرٌۢ بِمَا تَعْمَلُوْنَ۔ “کیا تم نے یہ سمجھ رکھا ہے کہ تم یونہی چھوڑ دیے جاؤ گے، حالانکہ ابھی اللہ نے ان لوگوں کو ظاہر نہیں کیا جنہوں نے تم میں سے جہاد کیا اور اللہ اور اس کے رسول صلی اللہ علیہ وسلم اور مومنوں کے سوا کسی کو راز دار دوست نہیں بنایا؛ اور اللہ تمہارے کاموں سے پوری طرح باخبر ہے۔” مفسرین لکھتے ہیں کہ اللہ تعالیٰ ہر چیز کو ازل سے جانتا ہے؛ یہاں جاننے سے مراد وہ علم ہے جو عمل کے ظہور کے ساتھ متعلق ہوتا ہے، یعنی جب تک بندے کا اندرونی حال عمل میں ظاہر نہ ہو جائے۔ ولیجہ وہ اندر کا راز دار ہے جس پر دل کا سارا بھروسا ہو۔ شیخِ اکبر کے نزدیک آزمائش حق کے علم میں کچھ نیا نہیں ڈالتی بلکہ بندے کے عینِ ثابتہ میں جو پہلے سے تھا اسے شہود کی دنیا میں لے آتی ہے، تاکہ بندہ خود اپنے آپ کو پہچان لے۔ یہی بات ایک اور جگہ یوں فرمائی گئی: اَحَسِبَ النَّاسُ اَنْ يُّتْرَكُوْۤا اَنْ يَّقُوْلُوْۤا اٰمَنَّا وَهُمْ لَا يُفْتَنُوْنَ۔ اہلِ معرفت اس آیت سے نفس کے ساتھ جہاد کا اشارہ بھی لیتے ہیں، کہ دل کا ولیجہ حق کے سوا کوئی نہ ہو، اور اسی راہ پر وعدہ ہے: وَالَّذِيْنَ جٰهَدُوْا فِيْنَا لَنَهْدِيَنَّهُمْ سُبُلَنَا۔ مگر یہ اشاری معنی ظاہری تاریخی معنی کی جگہ نہیں لیتا، اس کی گہرائی ہے؛ اور ظاہر میں یہ رکوع حضورِ اکرم صلی اللہ علیہ وسلم کے زمانے کے عرب کی تاریخ ہے، جس کے احکام کا اختیار صرف جائز ریاست کو تھا، اور جس سے آج کسی قوم یا مذہب کے بارے میں کوئی عمومی حکم اخذ کرنا درست نہیں۔ قاری کے لیے اس کا حاصل وفا، استقامت، حق کی خشیت، اور توبہ کرنے والے کو بھائی بنا لینا ہے۔
