وَجَآءَ الْمُعَذِّرُوْنَ مِنَ الْاَعْرَابِ لِيُؤْذَنَ لَهُمْ وَقَعَدَ الَّذِيْنَ كَذَبُوا اللّٰهَ وَرَسُوْلَهٗ ؕ سَيُصِيْبُ الَّذِيْنَ كَفَرُوْا مِنْهُمْ عَذَابٌ اَلِيْمٌ۔ “اور دیہاتیوں میں سے عذر پیش کرنے والے آئے تاکہ انہیں پیچھے رہنے کی اجازت دی جائے، اور وہ لوگ بیٹھے رہ گئے جنہوں نے اللہ اور اس کے رسول صلی اللہ علیہ وسلم سے جھوٹ بولا تھا؛ ان میں سے جنہوں نے کفر کیا انہیں عنقریب دردناک عذاب پہنچے گا۔” پچھلے رکوع میں مدینہ کے ان لوگوں کا ذکر تھا جو غزوۂ تبوک کے وقت پیچھے بیٹھ رہے؛ اب بات مدینہ کے گرد بسنے والے دیہاتی قبیلوں کی طرف مڑتی ہے۔ لفظ مُعَذِّرُوْن کے بارے میں مفسرین کی دو راہیں ہیں: ایک یہ کہ اس سے مراد وہ لوگ ہیں جو بے اصل عذر گھڑتے تھے، اور دوسری یہ کہ اس میں سچے عذر والے بھی شامل ہیں۔ اسی لیے آیت نے وعید کو سب پر عام نہیں کیا بلکہ فرمایا “ان میں سے جنہوں نے کفر کیا”۔ یہ قرآنِ مجید کا عدل ہے کہ ایک جماعت کا ذکر کرتے ہوئے بھی ہر فرد کو اس کے اپنے حال پر رکھتا ہے۔ آیت میں دو گروہ ہیں: ایک وہ جو اجازت مانگنے آئے، اور دوسرا وہ جو اجازت مانگنے کی زحمت بھی کیے بغیر بیٹھ رہا۔ دوسرے گروہ کے بارے میں فرمایا کہ انہوں نے اللہ اور اس کے رسول صلی اللہ علیہ وسلم سے جھوٹ بولا، یعنی ایمان کا جو دعویٰ زبان پر تھا وہ عمل کے وقت جھوٹا نکلا۔

شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس آیت کو ایک باطنی آئینہ بھی بنایا جاتا ہے، اور یہ کسی دوسرے پر انگلی اٹھانے کے لیے نہیں بلکہ اپنے اندر جھانکنے کے لیے ہے۔ اہلِ معرفت کہتے ہیں کہ ہر انسان کے اندر ایک “عذر تراش” قوت ہے جو ہر کوتاہی کے لیے خوبصورت دلیل گھڑ لیتی ہے۔ نفس جب کسی مشقت سے بچنا چاہتا ہے تو سیدھا انکار نہیں کرتا، اجازت مانگتا ہے، اور اجازت مانگنے کے لیے عذر لاتا ہے۔ عذر بذاتِ خود برا نہیں؛ شریعت نے سچے عذر کو قبول کیا ہے، جیسا کہ اگلی آیت بتائے گی۔ برائی اس میں ہے کہ عذر حقیقت کی جگہ لے لے اور بندہ اپنی کمزوری کا اعتراف کرنے کے بجائے اسے دلیل کے لباس میں چھپا دے۔ شیخِ اکبر کے نزدیک بندے کا اصل حال فقر ہے، اور جو اپنے فقر کو پہچان لے اسے عذر تراشنے کی حاجت نہیں رہتی؛ وہ سیدھا کہتا ہے کہ میں عاجز ہوں، مجھ پر رحم فرما۔ عذر حجاب بنتا ہے اور اعتراف دروازہ؛ اسی لیے اس سورت کا اگلا رکوع ان لوگوں کا ذکر کرے گا جنہوں نے عذر کے بجائے اپنے گناہوں کا اقرار کیا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: تاریخ میں یہ واقعہ تبوک کے سفر سے جڑا ہے، اور اسی واقعے میں ہر دور کے سالک کے لیے سبق ہے۔

لَيْسَ عَلَى الضُّعَفَآءِ وَلَا عَلَى الْمَرْضٰى وَلَا عَلَى الَّذِيْنَ لَا يَجِدُوْنَ مَا يُنْفِقُوْنَ حَرَجٌ اِذَا نَصَحُوْا لِلّٰهِ وَرَسُوْلِهٖ ؕ مَا عَلَى الْمُحْسِنِيْنَ مِنْ سَبِيْلٍ ؕ وَاللّٰهُ غَفُوْرٌ رَّحِيْمٌ۔ “کمزوروں پر کوئی حرج نہیں، نہ بیماروں پر، اور نہ ان پر جو خرچ کرنے کو کچھ نہیں پاتے، جبکہ وہ اللہ اور اس کے رسول صلی اللہ علیہ وسلم کے سچے خیرخواہ ہوں؛ نیکوکاروں پر الزام کی کوئی راہ نہیں، اور اللہ بخشنے والا، مہربان ہے۔” ظاہری حکم واضح ہے: جہاد کے لیے نکلنے کی ذمہ داری ان تین طبقوں سے اٹھا لی گئی جو جسمانی کمزوری، بیماری یا تنگ دستی کی وجہ سے نکل نہیں سکتے تھے۔ مفسرین نے کمزوروں میں بوڑھوں اور ان لوگوں کو شمار کیا ہے جن کے بدن سفر کی مشقت نہ اٹھا سکیں۔ مگر یہ رخصت ایک شرط کے ساتھ ہے: اِذَا نَصَحُوْا، یعنی جب وہ اللہ اور اس کے رسول صلی اللہ علیہ وسلم کے خیرخواہ ہوں۔ حضورِ اکرم صلی اللہ علیہ وسلم کا ارشاد، جو صحیح مسلم میں حضرت تمیم داری رضی اللہ عنہ سے مروی ہے، اس کا مفہوم یہ ہے کہ دین خیرخواہی کا نام ہے: اللہ کے لیے، اس کی کتاب کے لیے، اس کے رسول صلی اللہ علیہ وسلم کے لیے، مسلمانوں کے ائمہ کے لیے اور عام مسلمانوں کے لیے۔ گویا بدن کو رخصت ملی، دل کو نہیں۔ جو پیچھے رہا وہ دعا سے، گھر والوں کی نگہبانی سے، افواہوں سے بچنے اور بچانے سے شریکِ سفر رہا۔ مفسرین نے لکھا کہ جو گھر بیٹھ کر مجاہدین کے بارے میں بری خبریں پھیلائے وہ اس رخصت کا اہل نہیں، کیونکہ اس نے خیرخواہی کی شرط توڑ دی۔

اب اس رکوع کا مرکزی جملہ آتا ہے: مَا عَلَى الْمُحْسِنِيْنَ مِنْ سَبِيْلٍ، “نیکوکاروں پر کوئی راہ نہیں۔” یہاں تعجب کی بات یہ ہے کہ جو لوگ میدان میں نہیں گئے، انہی کو قرآنِ مجید نے مُحْسِنِيْن کا لقب دیا۔ گویا احسان کا تعلق قدموں کی مسافت سے پہلے دل کی سمت سے ہے۔ عربی قاعدے کے مطابق نفی کے ساتھ مِنْ کا آنا مکمل نفی کا فائدہ دیتا ہے: کوئی بھی راہ، چھوٹی یا بڑی، ان پر الزام کے لیے کھلی نہیں۔ فقہائے کرام رحمہم اللہ نے اسی جملے سے ایک عام اصول اخذ کیا کہ جو شخص بھلائی کی نیت سے کسی کا کام کرے یا امانت سنبھالے اور زیادتی نہ کرے، اس پر نقصان کا تاوان نہیں۔ یوں یہ آیت جہاد کے ایک موقع سے نکل کر معاملات کی پوری دنیا میں امانت داروں کی ڈھال بن گئی۔ اہلِ ذوق کہتے ہیں کہ سبیل کا لفظ خود بتاتا ہے کہ ملامت ایک راستے سے آتی ہے؛ وہ کسی خلا کو ڈھونڈتی ہے جہاں نیت کمزور ہو۔ جس کی نیت پوری ہو، اس کے گرد کوئی ایسا خلا نہیں رہتا۔ نہ مخلوق کی ملامت کو راہ ملتی ہے، نہ نفس کی اپنی ملامت کو، نہ وسوسے کو۔

احسان کی تعریف حدیثِ جبریل میں آئی ہے، جس کا مفہوم یہ ہے کہ تم اللہ کی عبادت اس طرح کرو گویا اسے دیکھ رہے ہو، اور اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے۔ شیخِ اکبر کے نزدیک احسان شہود کا مقام ہے: بندہ ہر حال میں حق کی نگاہ کے سامنے رہتا ہے۔ اس مکتب میں حدیث کے دوسرے جملے کی ایک ذوقی قراءت بھی مشہور ہے کہ جب تم نہ رہو، یعنی اپنی ہستی کے دعوے سے فانی ہو جاؤ، تو تم اسے دیکھو گے۔ یہ اشارہ ظاہری معنی کی جگہ نہیں لیتا، اس کے اندر ایک دریچہ کھولتا ہے۔ اب ان کمزوروں اور بیماروں کو دیکھیے: ان کی کمزوری نے ان سے اپنی طاقت کا دعویٰ چھین لیا تھا۔ وہ عجز کے مقام پر کھڑے تھے، اور عجز فنا کا دروازہ ہے۔ جو اپنی قوت سے خالی ہو اور دل حق کی طرف متوجہ رکھے، وہ احسان کی اسی سیڑھی پر ہے۔ اسی لیے آیت کا خاتمہ وَاللّٰهُ غَفُوْرٌ رَّحِيْمٌ پر ہوا: اگر ان سے کوئی کمی رہ بھی گئی تو مغفرت اسے ڈھانپ لے گی، اور رحمت ان کی نیت کو عمل کا درجہ دے دے گی۔ سبقتِ رحمت کا یہی مطلب ہے کہ حق کی رحمت بندے کی کوتاہی سے پہلے پہنچ جاتی ہے۔

وَّلَا عَلَى الَّذِيْنَ اِذَا مَاۤ اَتَوْكَ لِتَحْمِلَهُمْ قُلْتَ لَاۤ اَجِدُ مَاۤ اَحْمِلُكُمْ عَلَيْهِ تَوَلَّوْا وَّاَعْيُنُهُمْ تَفِيْضُ مِنَ الدَّمْعِ حَزَنًا اَلَّا يَجِدُوْا مَا يُنْفِقُوْنَ۔ “اور نہ ان لوگوں پر کوئی الزام ہے جو آپ صلی اللہ علیہ وسلم کے پاس آئے کہ آپ صلی اللہ علیہ وسلم انہیں سواری عطا فرمائیں، آپ صلی اللہ علیہ وسلم نے فرمایا کہ میرے پاس کوئی سواری نہیں جس پر تمہیں سوار کروں، تو وہ اس حال میں لوٹے کہ ان کی آنکھیں اس غم میں آنسو بہا رہی تھیں کہ ان کے پاس خرچ کرنے کو کچھ نہیں۔” یہ وہ صحابۂ کرام رضی اللہ عنہم ہیں جنہیں سیرت کی کتابوں میں بکائین، یعنی رونے والے، کہا گیا۔ امام ابنِ اسحاق رحمۃ اللہ علیہ کی روایت میں ان کے نام آئے ہیں، جن میں حضرت سالم بن عمیر، حضرت علبہ بن زید اور حضرت ابو لیلیٰ عبد الرحمٰن بن کعب رضی اللہ عنہم کا ذکر ملتا ہے۔ تبوک کا سفر سخت گرمی میں، طویل صحرا کے پار تھا، اور سواری کے بغیر اس کا تصور مشکل تھا۔ وہ حضورِ اکرم صلی اللہ علیہ وسلم کی خدمت میں آئے، اور آپ صلی اللہ علیہ وسلم کے پاس دینے کو کچھ نہ تھا۔ قرآنِ مجید نے ان کی واپسی کا منظر یوں کھینچا کہ ان کی آنکھیں آنسوؤں سے بہہ رہی تھیں۔ عربی بلاغت میں یہ مبالغے کا اسلوب ہے: آنسو آنکھ سے نہیں ٹپکے، گویا آنکھ خود آنسو بن کر بہہ نکلی۔ اور ان کا غم اپنی محرومی پر نہیں تھا، اس پر تھا کہ وہ اللہ کی راہ میں کچھ دے نہ سکے۔ روایات میں آتا ہے کہ ان میں سے بعض کو بعد میں حضرت عثمان رضی اللہ عنہ اور دوسرے صحابہ رضی اللہ عنہم نے سواریاں دیں اور وہ لشکر میں شامل ہو گئے۔

یہاں ایک لطیف منظر ہے جس پر اہلِ معرفت ٹھہرتے ہیں۔ قُلْتَ لَاۤ اَجِدُ مَاۤ اَحْمِلُكُمْ عَلَيْهِ، “آپ صلی اللہ علیہ وسلم نے فرمایا کہ میرے پاس کچھ نہیں۔” حضورِ اکرم صلی اللہ علیہ وسلم، جن کے ہاتھ سے بے شمار خزانے تقسیم ہوئے، اس دن اپنے فقر کا اظہار فرما رہے ہیں؛ اور سامنے کھڑے صحابہ رضی اللہ عنہم بھی فقیر ہیں۔ دو فقر آمنے سامنے آئے، اور اس ملاقات سے آنسو پیدا ہوئے جو قیامت تک تلاوت کیے جائیں گے۔ صحیح بخاری میں حضرت انس رضی اللہ عنہ سے روایت ہے، جس کا مفہوم یہ ہے کہ تبوک سے واپسی پر حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: مدینہ میں کچھ لوگ ہیں کہ تم نے جو سفر بھی کیا اور جو وادی بھی پار کی، وہ تمہارے ساتھ تھے۔ صحابہ رضی اللہ عنہم نے عرض کیا: حالانکہ وہ مدینہ میں ہیں؟ آپ صلی اللہ علیہ وسلم نے فرمایا: ہاں، وہ مدینہ میں ہیں، انہیں عذر نے روک رکھا ہے۔ یہ حدیث اس آیت کی شرح ہے۔ بدن مدینہ میں تھے اور دل لشکر کے ساتھ صحرا میں۔ شیخِ اکبر کے نزدیک نیت عمل کی روح ہے، اور عالمِ معانی میں جس طرف دل نے سچا رخ کیا وہی سفر شمار ہوتا ہے۔ حدیثِ نیت، کہ اعمال کا دار و مدار نیتوں پر ہے، اسی حقیقت کا عام قانون ہے؛ اور یہ آیت اس قانون کی سب سے روشن مثال۔

اہلِ ذوق کہتے ہیں کہ ان صحابہ رضی اللہ عنہم کے پاس خرچ کرنے کو مال نہیں تھا، تو انہوں نے آنسو خرچ کیے؛ اور حق تعالیٰ نے ان آنسوؤں کو قبول کر کے اپنے کلام میں محفوظ کر دیا۔ فقیر کا سرمایہ اس کی تڑپ ہے۔ جس کے ہاتھ خالی ہوں مگر دل بھرا ہو، اس کا خالی ہاتھ بھی ایک قسم کا انفاق ہے۔ ترمذی میں حضرت ابنِ عباس رضی اللہ عنہما سے مروی حدیث کا مفہوم یہ ہے کہ دو آنکھوں کو آگ نہیں چھوئے گی: وہ آنکھ جو اللہ کے خوف سے روئی، اور وہ آنکھ جس نے اللہ کی راہ میں پہرہ دیتے رات گزاری۔ ان صحابہ رضی اللہ عنہم کے آنسو اسی قبیل کے تھے: وہ اللہ کی راہ میں شریک نہ ہو سکنے کے غم میں بہے، اور یہ غم خود اللہ کی محبت کی گواہی تھا۔ سیرت کی روایات میں حضرت علبہ بن زید رضی اللہ عنہ کا واقعہ آتا ہے کہ انہوں نے رات کو نماز پڑھ کر عرض کیا کہ اے اللہ! میرے پاس دینے کو کچھ نہیں، میں ہر اس شخص کو معاف کرتا ہوں جس نے میری عزت پر زیادتی کی ہو، اور یہ معافی صدقہ کے طور پر پیش کرتا ہوں؛ اور صبح حضورِ اکرم صلی اللہ علیہ وسلم نے انہیں خوشخبری دی کہ ان کا صدقہ قبول ہو گیا۔ اس مکتب میں فقر و تجرید کا یہی ذوق ہے: جب سب کچھ چھن جائے تو بندہ اپنی انا دیتا ہے، اور یہ سب سے گراں قیمت صدقہ ہے۔

اِنَّمَا السَّبِيْلُ عَلَى الَّذِيْنَ يَسْتَاْذِنُوْنَكَ وَهُمْ اَغْنِيَآءُ ۚ رَضُوْا بِاَنْ يَّكُوْنُوْا مَعَ الْخَوَالِفِ ۙ وَطَبَعَ اللّٰهُ عَلٰى قُلُوْبِهِمْ فَهُمْ لَا يَعْلَمُوْنَ۔ “الزام کی راہ تو صرف ان پر ہے جو مالدار ہوتے ہوئے آپ صلی اللہ علیہ وسلم سے اجازت مانگتے ہیں؛ وہ اس پر راضی ہو گئے کہ پیچھے رہ جانے والوں کے ساتھ رہیں، اور اللہ نے ان کے دلوں پر مہر لگا دی، سو وہ نہیں جانتے۔” دیکھیے، سبیل کا لفظ پلٹ کر آیا ہے۔ محسنین پر کوئی سبیل نہیں تھی، اور یہاں فرمایا کہ سبیل تو انہی پر ہے۔ ظاہری عمل دونوں جگہ ایک تھا: اجازت مانگنا۔ فرق اس دل میں تھا جس سے اجازت مانگی گئی۔ ایک طرف آنسو تھے، دوسری طرف دولت کے باوجود سستی پر رضامندی۔ یہ آیت سورت کے اصولی مقامات میں سے ہے، اور اسے اسی ترتیب سے پڑھنا لازم ہے جس ترتیب سے قرآنِ مجید نے رکھا: پہلے رَضُوْا بِاَنْ يَّكُوْنُوْا مَعَ الْخَوَالِفِ، یعنی ان کا اپنا انتخاب، اور اس کے بعد مہر۔ حق تعالیٰ کسی طالبِ ہدایت کو زبردستی گمراہ نہیں کرتا۔ جب بندہ کمتر حال پر راضی ہو جاتا ہے اور اس پر جم جاتا ہے، تو حق اس کے چنے ہوئے رخ کو وجود دے دیتا ہے؛ یہی مہر ہے۔ شیخِ اکبر کے نزدیک سرِّ قدر کا خلاصہ یہی ہے کہ حکم محکوم کے مطابق ہوتا ہے: فیصلہ اسی کے مطابق آتا ہے جو چیز خود ہے اور جس طرف وہ مڑی۔ اور آخری جملہ “سو وہ نہیں جانتے” بتاتا ہے کہ اس رضامندی کی سب سے بڑی سزا علم کا چھن جانا ہے: جو دل پستی پر راضی ہو جائے، بلندی کی خبر اس تک نہیں پہنچتی۔

يَعْتَذِرُوْنَ اِلَيْكُمْ اِذَا رَجَعْتُمْ اِلَيْهِمْ ؕ قُلْ لَّا تَعْتَذِرُوْا لَنْ نُّؤْمِنَ لَكُمْ قَدْ نَبَّاَنَا اللّٰهُ مِنْ اَخْبَارِكُمْ ؕ وَسَيَرَى اللّٰهُ عَمَلَكُمْ وَرَسُوْلُهٗ ثُمَّ تُرَدُّوْنَ اِلٰى عٰلِمِ الْغَيْبِ وَالشَّهَادَةِ فَيُنَبِّئُكُمْ بِمَا كُنْتُمْ تَعْمَلُوْنَ۔ “جب تم ان کی طرف لوٹو گے تو وہ تمہارے سامنے عذر پیش کریں گے۔ کہہ دیجیے: عذر نہ کرو، ہم ہرگز تمہاری بات کا یقین نہیں کریں گے، اللہ نے ہمیں تمہارے حالات بتا دیے ہیں؛ اور آگے اللہ اور اس کا رسول صلی اللہ علیہ وسلم تمہارا عمل دیکھیں گے، پھر تم غیب اور ظاہر کے جاننے والے کی طرف لوٹائے جاؤ گے، تو وہ تمہیں بتا دے گا جو کچھ تم کرتے تھے۔” یہ آیت تبوک سے واپسی سے پہلے نازل ہوئی اور پیش گوئی کی کہ واپسی پر عذروں کی قطار لگ جائے گی، اور ایسا ہی ہوا۔ مگر غور کیجیے کہ دروازہ مکمل بند نہیں کیا گیا: وَسَيَرَى اللّٰهُ عَمَلَكُمْ وَرَسُوْلُهٗ۔ مفسرین کہتے ہیں کہ اس میں اشارہ ہے کہ اب باتوں کا وقت گیا، آگے کا عمل بتائے گا کہ تم توبہ کرتے ہو یا اپنی روش پر قائم رہتے ہو۔ عذر قبول نہیں، مگر عمل کی مہلت باقی ہے۔ سورۂ القیامہ میں انسان کی یہی کیفیت بیان ہوئی: بَلِ الْاِنْسَانُ عَلٰى نَفْسِهٖ بَصِيْرَةٌ وَّلَوْ اَلْقٰى مَعَاذِيْرَهٗ، “بلکہ انسان اپنے آپ کو خوب دیکھنے والا ہے، چاہے وہ کتنے ہی عذر پیش کرے۔” اہلِ معرفت کے نزدیک عٰلِمِ الْغَيْبِ وَالشَّهَادَةِ کا ذکر یہاں پوری بات کھول دیتا ہے: عذر صرف ظاہر کو ڈھانپ سکتا ہے، اور جو ذات باطن اور ظاہر دونوں کو یکساں جانتی ہے، اس کے سامنے عذر کا پردہ کچھ نہیں چھپاتا۔ اسی لیے سالک عذر تراشنا چھوڑ کر اعتراف سیکھتا ہے۔

سَيَحْلِفُوْنَ بِاللّٰهِ لَكُمْ اِذَا انْقَلَبْتُمْ اِلَيْهِمْ لِتُعْرِضُوْا عَنْهُمْ ؕ فَاَعْرِضُوْا عَنْهُمْ ؕ اِنَّهُمْ رِجْسٌ ۚ وَّمَاْوٰىهُمْ جَهَنَّمُ ۚ جَزَآءًۢ بِمَا كَانُوْا يَكْسِبُوْنَ۔ “جب تم ان کی طرف پلٹو گے تو وہ تمہارے سامنے اللہ کی قسمیں کھائیں گے تاکہ تم ان سے درگزر کرو؛ سو ان سے رخ پھیر لو، وہ ناپاک ہیں، اور ان کا ٹھکانا جہنم ہے، اس کے بدلے جو وہ کماتے تھے۔” يَحْلِفُوْنَ لَكُمْ لِتَرْضَوْا عَنْهُمْ ۚ فَاِنْ تَرْضَوْا عَنْهُمْ فَاِنَّ اللّٰهَ لَا يَرْضٰى عَنِ الْقَوْمِ الْفٰسِقِيْنَ۔ “وہ تمہارے سامنے قسمیں کھاتے ہیں تاکہ تم ان سے راضی ہو جاؤ؛ پس اگر تم ان سے راضی ہو بھی جاؤ تو اللہ نافرمان لوگوں سے راضی نہیں ہوتا۔” ان کی قسموں کا مقصد یہ تھا کہ ملامت سے بچ جائیں؛ جواب ملا کہ ان سے رخ پھیر لو۔ مفسرین کہتے ہیں کہ یہ قبولیت کا اعراض نہیں بلکہ ترکِ التفات کا اعراض ہے: ان کی باتوں میں نہ الجھو۔ اور رِجْس سے مراد ان کے باطن اور اعمال کی ناپاکی ہے، بدن کی نہیں۔ ان دو آیتوں میں نفس کی ایک پرانی بیماری کا نقشہ ہے: مخلوق کی رضا کی طلب۔ وہ لوگ اللہ کی قسم کھا رہے تھے مگر اللہ کی رضا نہیں، لوگوں کی رضا چاہ رہے تھے۔ ترمذی میں ام المؤمنین حضرت عائشہ رضی اللہ عنہا سے مروی حدیث کا مفہوم یہ ہے کہ جو لوگوں کو ناراض کر کے اللہ کی رضا ڈھونڈے، اللہ اس کے لیے لوگوں کی طرف سے کافی ہو جاتا ہے؛ اور جو اللہ کو ناراض کر کے لوگوں کی رضا ڈھونڈے، اللہ اسے لوگوں کے حوالے کر دیتا ہے۔ اور فِسق کا ذکر بتاتا ہے کہ اللہ کی ناراضی ان کی اپنی نافرمانی کا نتیجہ ہے، کوئی اچانک فیصلہ نہیں۔

اہلِ ذوق کہتے ہیں کہ ہم میں سے ہر ایک کے اندر ایک چھوٹی سی عدالت لگتی ہے جہاں نفس لوگوں کی نگاہ کو قاضی بنا لیتا ہے۔ جب ہم کوئی کوتاہی کرتے ہیں تو سب سے پہلے یہ سوچتے ہیں کہ لوگ کیا کہیں گے، اور اسی کے لیے دلیلیں اور قسمیں جمع کرتے ہیں۔ یہ آیت ہمیں اس عدالت سے اٹھا کر اصل بارگاہ میں کھڑا کرتی ہے: اگر سب لوگ راضی ہو جائیں اور حق راضی نہ ہو تو اس رضا کی کوئی قیمت نہیں۔ شیخِ اکبر کے نزدیک بندے کی نگاہ جب مخلوق سے ہٹ کر خالق پر جم جائے تو وہ ہر چیز میں اسی کی تجلی دیکھتا ہے، اور پھر مخلوق کی رضا اسے اپنی طرف نہیں کھینچتی۔ اس کے برعکس جو مخلوق کے آئینے میں اپنی صورت سنوارتا رہے، وہ ہر روز ایک نئی قسم کا محتاج رہتا ہے۔ یہ آیات ان تاریخی لوگوں کی بات کرتی ہیں جن کا حال وحی نے کھولا؛ آج کسی شخص یا گروہ پر انہیں چسپاں کرنا ہمارا منصب نہیں۔ ہمارا منصب یہ ہے کہ اپنے دل سے پوچھیں کہ ہماری قسمیں اور وضاحتیں کس کی رضا کے لیے ہیں۔ اگلے رکوع میں جب رَضِيَ اللّٰهُ عَنْهُمْ وَرَضُوْا عَنْهُ کی آیت آئے گی تو یہ فرق اور روشن ہو جائے گا۔

اَلْاَعْرَابُ اَشَدُّ كُفْرًا وَّنِفَاقًا وَّاَجْدَرُ اَلَّا يَعْلَمُوْا حُدُوْدَ مَاۤ اَنْزَلَ اللّٰهُ عَلٰى رَسُوْلِهٖ ؕ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ “دیہاتی کفر اور نفاق میں زیادہ سخت ہیں، اور اس کے زیادہ لائق کہ ان حدوں سے ناواقف رہیں جو اللہ نے اپنے رسول صلی اللہ علیہ وسلم پر اتاری ہیں؛ اور اللہ جاننے والا، حکمت والا ہے۔” وَمِنَ الْاَعْرَابِ مَنْ يَّتَّخِذُ مَا يُنْفِقُ مَغْرَمًا وَّيَتَرَبَّصُ بِكُمُ الدَّوَآئِرَ ؕ عَلَيْهِمْ دَآئِرَةُ السَّوْءِ ؕ وَاللّٰهُ سَمِيْعٌ عَلِيْمٌ۔ “اور دیہاتیوں میں کوئی ایسا ہے جو اپنے خرچ کو تاوان سمجھتا ہے اور تم پر زمانے کی گردشوں کا انتظار کرتا ہے؛ بری گردش انہی پر ہے، اور اللہ سننے والا، جاننے والا ہے۔” یہاں ادب سے سمجھنا ضروری ہے کہ اَعْرَاب سے مراد اس زمانے کے وہ صحرا نشین قبیلے ہیں جو مدینہ منورہ کے گرد آباد تھے؛ یہ کسی قوم یا نسل کا نام نہیں، اور نہ یہ آیت آج کے کسی بھی گروہ پر چسپاں کرنے کے لیے ہے۔ مفسرین نے وضاحت کی کہ ان میں سے جو کفر یا نفاق پر تھے وہ شہر والوں سے زیادہ سخت تھے، اور اس کا سبب حضورِ اکرم صلی اللہ علیہ وسلم کی صحبت سے دوری، علم کی مجلسوں سے فاصلہ اور زندگی کی سختی تھی؛ نسب نہیں۔ اور خود اگلی دو آیتیں ان میں تقسیم کر دیتی ہیں، اور دوسری آیت ان کے مؤمنوں کی کھلی تعریف کرتی ہے۔ صحابۂ کرام رضی اللہ عنہم میں بہت سے جلیل القدر حضرات انہی قبیلوں سے تھے۔ اشاری طور پر اہلِ معرفت کہتے ہیں کہ ہر دل کے اندر ایک صحرا ہے: جو دل ذکر کی مجلس اور علم کی صحبت سے دور رہے، وہ سخت ہوتا جاتا ہے، اور جو خرچ کرے اسے مَغْرَم، یعنی تاوان، سمجھتا ہے۔

وَمِنَ الْاَعْرَابِ مَنْ يُّؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْاٰخِرِ وَيَتَّخِذُ مَا يُنْفِقُ قُرُبٰتٍ عِنْدَ اللّٰهِ وَصَلَوٰتِ الرَّسُوْلِ ؕ اَلَاۤ اِنَّهَا قُرْبَةٌ لَّهُمْ ؕ سَيُدْخِلُهُمُ اللّٰهُ فِيْ رَحْمَتِهٖ ؕ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِيْمٌ۔ “اور دیہاتیوں میں کوئی ایسا ہے جو اللہ اور آخرت کے دن پر ایمان رکھتا ہے، اور جو خرچ کرتا ہے اسے اللہ کے ہاں قرب کا اور رسول صلی اللہ علیہ وسلم کی دعاؤں کا ذریعہ سمجھتا ہے؛ سن لو! وہ یقیناً ان کے لیے قرب کا ذریعہ ہے؛ اللہ انہیں جلد اپنی رحمت میں داخل فرمائے گا؛ بے شک اللہ بخشنے والا، مہربان ہے۔” بعض مفسرین نے روایت کیا ہے کہ یہ آیت قبیلۂ مزینہ کے بنو مقرن کے بارے میں نازل ہوئی۔ ان مؤمنوں کی تعریف حق تعالیٰ نے خود اپنی زبان سے کی اور “سن لو!” کہہ کر اس پر مہر تصدیق ثبت کی۔ اب پچھلی آیت سے ملا کر دیکھیے: عمل ایک ہے، یعنی خرچ کرنا۔ ایک کے لیے وہ تاوان ہے اور دوسرے کے لیے قربت۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کا ایک روشن نمونہ ہے: عمل کی صورت ایک ہوتی ہے، مگر اس کی حقیقت بندے کے رخ کے مطابق ڈھلتی ہے۔ ایک ہی سکہ ایک ہاتھ سے بوجھ بن کر گرتا ہے اور دوسرے ہاتھ سے سیڑھی بن کر اوپر چڑھتا ہے۔ اور وَصَلَوٰتِ الرَّسُوْلِ کا لفظ بتاتا ہے کہ وہ مؤمن اپنے صدقے کے بدلے حضورِ اکرم صلی اللہ علیہ وسلم کی دعا کے طلب گار تھے؛ ان کے نزدیک یہ دعا مال سے زیادہ قیمتی تھی۔ یہی بات اگلے رکوع کی آیت میں حکم بن کر آئے گی کہ ان کے لیے دعا فرمائیے، آپ صلی اللہ علیہ وسلم کی دعا ان کے لیے سکون ہے۔ اور رحمت میں داخل کرنے کا اسلوب بتاتا ہے کہ رحمت ایک گھر ہے جس میں بندہ داخل ہوتا ہے اور جس کی چھت اسے ہر طرف سے ڈھانپ لیتی ہے۔
