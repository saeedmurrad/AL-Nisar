اس رکوع میں سورت نشانیوں کی ایک لڑی پروتی ہے جس کے چھ دانے ومن آیاتہٖ سے کھلتے ہیں، گویا کائنات ایک کھلی کتاب ہے اور اس کا ہر منظر ایک آیت۔ پہلا دانہ خود انسان ہے: وَمِنْ اٰيٰتِهٖۤ اَنْ خَلَقَكُمْ مِّنْ تُرَابٍ ثُمَّ اِذَآ اَنْتُمْ بَشَرٌ تَنْتَشِرُوْنَ۔ “اور اس کی نشانیوں میں سے یہ ہے کہ اس نے تمہیں مٹی سے پیدا کیا، پھر یکایک تم انسان ہو کہ زمین میں پھیلتے چلے جا رہے ہو۔” ظاہر میں یہ حضرت آدم علیہ السلام کی تخلیق اور نسلِ انسانی کے پھیلاؤ کا بیان ہے، اور اشاری معنی اس ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ معرفت کہتے ہیں کہ تراب وجود کا سب سے نچلا اور سب سے زیادہ قبول کرنے والا مرتبہ ہے؛ مٹی ہر بیج کو اپنے اندر جگہ دیتی ہے، ہر قدم سہتی ہے اور اپنا کوئی دعویٰ نہیں رکھتی۔ انسان کی اصل اسی عاجزی میں رکھی گئی تاکہ وہ جانے کہ اس کا کمال دعوے میں نہیں، استعداد میں ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک ایک اصل کا بے شمار صورتوں میں پھیل جانا وحدت کے کثرت میں ظاہر ہونے کی تمثیل ہے: ایک نفَسِ رحمانی ہے جو حروف و کلمات بن کر پھیلتا ہے، اور ایک آدمیت ہے جو قوموں، قبیلوں اور زمانوں میں منتشر ہوتی ہے۔ بشر کا لفظ بشرہ یعنی ظاہری جلد سے ہے؛ یہ ظاہر ہونے والا وجود ہے، اور اس کا ظہور خود ایک آیت ہے کہ جو مٹی میں پوشیدہ تھا وہ نمایاں ہو گیا۔ سالک کے لیے سبق یہ ہے کہ جتنا پھیلے، اپنی مٹی نہ بھولے؛ انتشار اصل سے کٹ جائے تو بکھراؤ ہے، اور اصل سے جڑا رہے تو حق کے اسماء کا ظہور۔

دوسرا دانہ گھر کے اندر رکھا گیا: وَمِنْ اٰيٰتِهٖۤ اَنْ خَلَقَ لَكُمْ مِّنْ اَنْفُسِكُمْ اَزْوَاجًا لِّتَسْكُنُوْۤا اِلَيْهَا وَجَعَلَ بَيْنَكُمْ مَّوَدَّةً وَّرَحْمَةً ۚ اِنَّ فِيْ ذٰلِكَ لَاٰيٰتٍ لِّقَوْمٍ يَّتَفَكَّرُوْنَ۔ “اور اس کی نشانیوں میں سے یہ ہے کہ اس نے تمہارے لیے تمہاری ہی جانوں میں سے جوڑے بنائے تاکہ تم ان کے پاس سکون پاؤ، اور تمہارے درمیان محبت اور رحمت رکھ دی؛ بے شک اس میں غور کرنے والوں کے لیے نشانیاں ہیں۔” لفظوں کا ادب دیکھیے: جوڑے تمہاری اپنی جانوں سے ہیں، کسی اجنبی جنس سے نہیں؛ پس شوہر اور بیوی ایک ہی اصل کے دو رخ ہیں، اور ایک اصل کے دو رخوں میں نہ کوئی بالا ہے نہ کوئی پست۔ زوج کہتے ہی اسے ہیں جو دوسرے کے ساتھ مل کر پورا ہو؛ ہر ایک دوسرے کا جوڑا ہے اور ہر ایک دوسرے کے بغیر ادھورا۔ مودت اور رحمت کو کسی ایک کی طرف سے دوسرے کی طرف نہیں، بلکہ دونوں کے درمیان رکھا گیا، یعنی یہ دو طرفہ امانت ہے۔ پھر فعل پر غور کیجیے: سکون پانا بندوں کا کام بتایا، مگر محبت اور رحمت کو حق نے خود رکھا؛ یہ دونوں انسان کی کمائی سے پہلے حق کی عطا ہیں۔ شیخِ اکبر کے مکتب میں زوجین کی باہمی محبت اس محبت کا عکس ہے جو حق کو اپنی مخلوق سے ہے۔ اہلِ تصوف میں مشہور اثر کنتُ کنزاً مخفیاً کا مضمون یہی ہے کہ عالم کی بنیاد ہی محبت ہے، اور جب ایک جان دوسری جان کے لیے نرم پڑتی ہے تو اسی اصلی محبت کی ایک کرن اس رشتے میں اترتی ہے۔ مودت اسمِ الودود کا مظہر ہے اور رحمت اسمِ الرحیم کا، اور یہ دونوں اسماء دونوں جانوں میں یکساں ظاہر ہوتے ہیں۔

بعض مفسرین نے کہا کہ مودت توانائی کے دنوں کی چاہت ہے اور رحمت کمزوری اور بڑھاپے کی نگہداشت۔ اہلِ ذوق اس میں یہ نکتہ پاتے ہیں کہ ازدواج کا رشتہ دونوں حالتوں میں حق کے اسماء کا آئینہ ہے: جب دونوں توانا ہوں تو ایک دوسرے کو چاہتے ہیں، اور جب کوئی کمزور پڑے تو دوسرا اس پر رحم کرتا ہے، اور یہ کمزوری کبھی ایک کے حصے میں آتی ہے کبھی دوسرے کے۔ اس مکتب میں ہر ملاپ میں دینے اور لینے کی ایک نسبت ہوتی ہے، مگر یہ نسبت ٹھہری ہوئی نہیں، گردش کرتی رہتی ہے؛ کبھی ایک دیتا ہے اور دوسرا قبول کرتا ہے، کبھی معاملہ الٹ جاتا ہے۔ اسی لیے کسی ایک کو ہمیشہ کا دینے والا اور دوسرے کو ہمیشہ کا محتاج ٹھہرانا اس آیت کے ذوق کے خلاف ہے۔ جن کلاسیکی تاویلات میں زوجین کو روح اور نفس جیسی باطنی قوتوں کی تمثیل بنایا گیا، وہ محض تمثیل ہیں؛ ان میں نہ کسی جنس کی برتری کا دعویٰ ہے نہ کسی کی تحقیر۔ حضورِ اکرم صلی اللہ علیہ وسلم نے اپنے گھر والوں کے ساتھ بھلائی کو بہترین لوگوں کی پہچان فرمایا، جس کا مفہوم یہ ہے کہ آدمی کی خوبی سب سے پہلے اس کے گھر میں پرکھی جاتی ہے۔ اور آیت کا اختتام تفکر پر ہے: گھر کی یہ خاموش محبت بھی ایک آیت ہے جو سوچنے والے کو اس محبت تک پہنچا دیتی ہے جس سے سارا عالم وجود میں آیا۔ گویا یہی سکون معرفت کی پہلی درسگاہ ہے، اور جو اپنے جوڑے میں حق کی رحمت نہ دیکھ سکا وہ آفاق میں اسے کم ہی دیکھ پاتا ہے۔

تیسرا دانہ آفاق اور انسانوں کے تنوع کو ایک ہی سانس میں جمع کرتا ہے: وَمِنْ اٰيٰتِهٖ خَلْقُ السَّمٰوٰتِ وَالْاَرْضِ وَاخْتِلَافُ اَلْسِنَتِكُمْ وَاَلْوَانِكُمْ ۚ اِنَّ فِيْ ذٰلِكَ لَاٰيٰتٍ لِّلْعٰلِمِيْنَ۔ “اور اس کی نشانیوں میں سے آسمانوں اور زمین کی پیدائش ہے، اور تمہاری زبانوں اور تمہارے رنگوں کا مختلف ہونا؛ بے شک اس میں علم والوں کے لیے نشانیاں ہیں۔” زبانوں اور رنگوں کے اختلاف کو آسمان و زمین کی تخلیق کے برابر رکھا گیا، گویا انسانوں کا تنوع بھی کائنات جتنی بڑی نشانی ہے۔ یہاں اختلاف کو آیت کہا گیا ہے، درجہ بندی نہیں؛ نہ کوئی زبان دوسری سے شریف تر ٹھہرائی گئی نہ کوئی رنگ دوسرے سے برتر۔ حضورِ اکرم صلی اللہ علیہ وسلم نے خطبۂ حجۃ الوداع میں اسی کو کھول دیا، جس کا مفہوم یہ ہے کہ نہ عربی کو عجمی پر فضیلت ہے نہ گورے کو کالے پر، فضیلت صرف تقویٰ سے ہے۔ شیخِ اکبر کے نزدیک تجلی میں تکرار نہیں: حق ایک ہی صورت میں دو بار ظاہر نہیں ہوتا، اسی لیے کوئی دو چہرے یکساں ہیں نہ کوئی دو لہجے۔ ہر رنگ کسی اسم کا مظہر ہے اور ہر زبان کسی اسم کی تسبیح؛ جو کسی رنگ یا زبان کو حقیر جانتا ہے وہ دراصل ایک الٰہی اسم کے ظہور سے منہ موڑتا ہے۔ اہلِ تصوف میں مشہور قول ہے کہ پانی کا رنگ اس کے برتن کا رنگ ہے؛ اس مکتب میں اس کا مفہوم یہ ہے کہ نورِ وجود ایک ہے اور رنگ ہر قابل کی استعداد سے پیدا ہوتے ہیں۔ سورۂ ابراہیم میں لسانِ قوم کا جو نکتہ گزرا تھا، یہاں وہ پھیل کر سارے انسانوں پر محیط ہو گیا۔ اور یہ نشانی علم والوں کے لیے کہی گئی، کیونکہ کثرت میں وحدت کو پہچاننا علم کا کام ہے، اور کثرت کو مٹائے بغیر اسے ایک کی طرف لوٹا دینا معرفت کا۔

چوتھا دانہ رات اور دن کی گردش میں چھپا ہے: وَمِنْ اٰيٰتِهٖ مَنَامُكُمْ بِالَّيْلِ وَالنَّهَارِ وَابْتِغَآؤُكُمْ مِّنْ فَضْلِهٖ ۚ اِنَّ فِيْ ذٰلِكَ لَاٰيٰتٍ لِّقَوْمٍ يَّسْمَعُوْنَ۔ “اور اس کی نشانیوں میں سے رات اور دن میں تمہارا سونا اور اس کا فضل تلاش کرنا ہے؛ بے شک اس میں سننے والوں کے لیے نشانیاں ہیں۔” نیند چھوٹی موت ہے اور ہر بیداری ایک چھوٹا حشر۔ شیخِ اکبر کے نزدیک نیند میں روح عالمِ خیال کے برزخ میں داخل ہوتی ہے جہاں معانی صورتیں پہنتے ہیں، اسی لیے سچے خواب کو حدیث میں نبوت کے اجزا میں شمار کیا گیا۔ اہلِ تصوف میں مشہور اثر ہے کہ لوگ سوئے ہوئے ہیں، جب مریں گے تو جاگیں گے؛ اس مکتب میں اس کا مفہوم یہ ہے کہ یہ سارا عالم بھی ایک خواب ہے جو تعبیر مانگتا ہے۔ روزی کی دوڑ کو یہاں فضل کی تلاش کہا گیا: محنت بندے کی ہے مگر ملتا جو کچھ ہے، عطا ہے۔ اہلِ لطائف کہتے ہیں کہ نیند میں سب سے آخر میں سونے والی اور سب سے پہلے جاگنے والی حس سماعت ہے، اسی لیے اس آیت کا خاتمہ سننے والوں پر ہوا۔ پانچواں دانہ آسمان کی چمک ہے: وَمِنْ اٰيٰتِهٖ يُرِيْكُمُ الْبَرْقَ خَوْفًا وَّطَمَعًا وَّيُنَزِّلُ مِنَ السَّمَآءِ مَآءً فَيُحْيٖ بِهِ الْاَرْضَ بَعْدَ مَوْتِهَا ۚ اِنَّ فِيْ ذٰلِكَ لَاٰيٰتٍ لِّقَوْمٍ يَّعْقِلُوْنَ۔ “اور اس کی نشانیوں میں سے یہ ہے کہ وہ تمہیں بجلی دکھاتا ہے، ڈراتی بھی اور امید بھی دلاتی ہے، اور آسمان سے پانی اتارتا ہے پھر اس سے زمین کو اس کے مرنے کے بعد زندہ کر دیتا ہے؛ بے شک اس میں عقل والوں کے لیے نشانیاں ہیں۔” سورۂ رعد میں بھی یہی فرمایا گیا تھا: هُوَ الَّذِيْ يُرِيْكُمُ الْبَرْقَ خَوْفًا وَّطَمَعًا۔

ایک ہی چمک میں خوف بھی ہے اور امید بھی؛ مسافر ڈرتا ہے کہ کہیں گرے نہ، اور کسان آس لگاتا ہے کہ اب بارش آئے گی۔ شیخِ اکبر کے مکتب میں یہ جلال و جمال کے جمع کی تمثیل ہے: تجلی ایک ہے، مگر قابل کے حال کے مطابق کسی پر ہیبت بن کر اترتی ہے اور کسی پر انس بن کر۔ اہلِ معرفت کہتے ہیں کہ سالک کا حال دو پروں والے پرندے کا سا ہے، ایک پر خوف کا اور ایک رجا کا؛ ایک پر کٹ جائے تو اڑان ٹوٹ جاتی ہے۔ پھر بارش اور مردہ زمین: اہلِ تاویل کہتے ہیں کہ زمین دل ہے اور آسمانی پانی علم، ذکر اور وحی کا فیض؛ جو دل غفلت سے مر چکا ہو، معرفت کی ایک بوند اسے پھر ہرا کر دیتی ہے، اور یہ پچھلے رکوع کے زندہ کو مردہ سے نکالنے والے مضمون کی تکمیل ہے۔ آیت کا خاتمہ عقل پر ہوا؛ عقل کا لفظ عقال یعنی اونٹ کی رسی سے ہے، اور شیخِ اکبر کے نزدیک عقل کا کام باندھنا اور ضبط میں لانا ہے۔ یہ نشانی ایسی ہے کہ باندھ کر سمجھنے والی عقل بھی اس کا سبق پا لیتی ہے: بجلی کا ایک لمحہ ڈر اور آس دونوں کو ایک ساتھ اٹھاتا ہے، اور جو اسے سمجھ لے وہ جان لیتا ہے کہ حق کی ہر تجلی میں بندے کے لیے دونوں حصے رکھے گئے ہیں، اور قلب اسی دوہرے حال میں پلٹتا رہتا ہے۔

چھٹا دانہ عالم کے کھڑے رہنے کا راز ہے: وَمِنْ اٰيٰتِهٖۤ اَنْ تَقُوْمَ السَّمَآءُ وَالْاَرْضُ بِاَمْرِهٖ ۚ ثُمَّ اِذَا دَعَاكُمْ دَعْوَةً مِّنَ الْاَرْضِ اِذَآ اَنْتُمْ تَخْرُجُوْنَ۔ “اور اس کی نشانیوں میں سے یہ ہے کہ آسمان اور زمین اس کے حکم سے قائم ہیں، پھر جب وہ تمہیں زمین سے ایک بار پکارے گا تو یکایک تم نکل آؤ گے۔” آسمان اور زمین اپنے بل پر نہیں کھڑے، امرِ الٰہی کے سہارے کھڑے ہیں۔ شیخِ اکبر کے نزدیک یہ سہارا ایک بار کا نہیں: عالم ہر آن اپنی ذات میں معدوم ہے اور ہر آن نئے سرے سے وجود پاتا ہے، اور یہی تجددِ امثال ہے؛ چونکہ مثل کے بعد مثل آتا ہے اس لیے آنکھ تسلسل دیکھتی ہے اور تبدیلی نہیں دیکھتی۔ جو امر انہیں ہر لمحہ قائم رکھتا ہے وہی ایک پکار بن کر قبروں سے نکالے گا؛ ایک ہی دعوت کافی ہے، کیونکہ کلمۂ کن کو دہرانے کی حاجت نہیں۔ پھر فرمایا: وَلَهٗ مَنْ فِي السَّمٰوٰتِ وَالْاَرْضِ ۚ كُلٌّ لَّهٗ قٰنِتُوْنَ۔ “اور آسمانوں اور زمین میں جو کوئی ہے سب اسی کا ہے، سب اسی کے فرمانبردار ہیں۔” اس مکتب میں امر کی دو قسمیں ہیں: امرِ تکوینی، جس سے کوئی ذرہ سرتابی نہیں کر سکتا، اور امرِ تکلیفی، جس میں بندے کے اختیار کو عزت دی گئی ہے۔ وجود کے اعتبار سے منکر کا دل بھی، اس کی سانس بھی، قانت ہے؛ مگر رخ کا انتخاب اس کا اپنا ہے۔ حق کسی کو زبردستی گمراہ نہیں کرتا؛ بندہ جس سمت مڑتا ہے، حق اس سمت کو وجود دے دیتا ہے، اور یہی سرِّ قدر ہے کہ حکم محکوم کے اپنے حال کے مطابق ہو۔

اور لڑی کا آخری دانہ سارے سلسلے کو سمیٹ لیتا ہے: وَهُوَ الَّذِيْ يَبْدَؤُا الْخَلْقَ ثُمَّ يُعِيْدُهٗ وَهُوَ اَهْوَنُ عَلَيْهِ ۚ وَلَهُ الْمَثَلُ الْاَعْلٰى فِي السَّمٰوٰتِ وَالْاَرْضِ ۚ وَهُوَ الْعَزِيْزُ الْحَكِيْمُ۔ “اور وہی ہے جو خلق کی ابتدا کرتا ہے پھر اسے دوبارہ پیدا کرے گا، اور یہ اس پر زیادہ آسان ہے؛ اور آسمانوں اور زمین میں سب سے بلند شان اسی کی ہے، اور وہی غالب، حکمت والا ہے۔” مفسرین کہتے ہیں کہ زیادہ آسان ہونا تمہارے پیمانے سے فرمایا گیا، ورنہ اس کے لیے کچھ مشکل نہیں؛ تمہاری عقل مانتی ہے کہ دوبارہ بنانا پہلی بار بنانے سے سہل ہے، تو جس نے پہلی بار بنایا اس پر دوبارہ بنانا کیوں دشوار ہو۔ اہلِ معرفت اس میں ایک اور نکتہ پاتے ہیں: ابتدا اور اعادہ صرف قیامت کا معاملہ نہیں، ہر آن جاری ہے؛ ہر لمحہ خلق کی ابتدا ہوتی ہے اور ہر لمحہ اعادہ، اور یہی خلقِ جدید ہے جس سے غافل انسان دوبارہ اٹھنے کو بعید سمجھتا ہے، حالانکہ وہ ہر سانس میں نیا اٹھایا جا رہا ہے۔ اس کے بعد وہ جملہ آیا جو اس پوری لڑی کی کنجی ہے: سب سے بلند مثال اسی کی ہے۔ یہاں مثل نہیں فرمایا، مثل یعنی ہم پلہ اور برابر کی نفی تو دوسری جگہ کر دی گئی؛ یہاں مَثَل فرمایا، یعنی وہ شان اور وصف جس سے وہ پہچانا جائے۔ اور یہ مثال آسمانوں اور زمین میں ہے، یعنی اس رکوع کی ہر نشانی، مٹی، جوڑا، زبان، رنگ، نیند، بجلی اور بارش، اسی بلند شان کا ایک پرتو ہے۔

یہی تنزیہ و تشبیہ کے جمع کا مقام ہے۔ سورۂ شوریٰ میں فرمایا: لَيْسَ كَمِثْلِهٖ شَيْءٌ ۚ وَهُوَ السَّمِيْعُ الْبَصِيْرُ۔ “اس کی مانند کوئی چیز نہیں، اور وہی سننے والا دیکھنے والا ہے۔” آیت کا پہلا حصہ تنزیہ ہے اور دوسرا حصہ تشبیہ، اور دونوں ایک ہی سانس میں ہیں۔ حکمتِ سبوحیہ میں شیخِ اکبر کا مفہوم یہ ہے کہ جو صرف تنزیہ کرے وہ حق کو مقید کرتا ہے، اور جو صرف تشبیہ کرے وہ بھی اسے محدود کرتا ہے؛ کامل معرفت ان دونوں کو جمع کرنا ہے۔ یہاں بھی یہی ترتیب ہے: مثال آسمانوں اور زمین میں ہے، یہ تشبیہ کا رخ ہے کہ وہ اپنی نشانیوں میں پہچانا جاتا ہے؛ اور وہ مثال سب سے بلند ہے، یہ تنزیہ کا رخ ہے کہ کوئی نشانی اسے گھیر نہیں سکتی۔ اہلِ ذوق کہتے ہیں کہ انسانِ کامل اس بلند مثال کا سب سے جامع آئینہ ہے، کیونکہ اسی میں سارے اسماء جمع ہیں؛ مگر آئینہ آئینہ ہی ہے، نہ صاحبِ صورت بنتا ہے نہ اس کا ہم پلہ۔ اور آیت کا خاتمہ دو اسماء پر ہوا: العزیز، جس تک کوئی رسائی نہیں، یہ تنزیہ ہے؛ اور الحکیم، جو ہر شے کو اس کی جگہ رکھتا ہے اور عالم کے نظم میں جلوہ گر ہے، یہ تشبیہ کا رخ ہے۔ جو آنکھ نشانیوں میں اسے دیکھے اور پھر بھی جانے کہ وہ ان سب سے بلند ہے، وہ اس حیرت تک پہنچتی ہے جو اہلِ معرفت کے نزدیک علم کی انتہا ہے، اور یہی وہ تفکر ہے جس کی دعوت پورا رکوع دیتا رہا۔
