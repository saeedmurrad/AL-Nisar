پچھلا رکوع زمین کے مرنے اور جینے پر ختم ہوا تھا؛ یہ رکوع انسان کی اپنی زندگی کو اسی دائرے میں رکھ کر دکھاتا ہے: اَللّٰهُ الَّذِيْ خَلَقَكُمْ مِّنْ ضَعْفٍ ثُمَّ جَعَلَ مِنْۢ بَعْدِ ضَعْفٍ قُوَّةً ثُمَّ جَعَلَ مِنْۢ بَعْدِ قُوَّةٍ ضَعْفًا وَّشَيْبَةً۔ اللہ ہی ہے جس نے تمہیں کمزوری سے پیدا کیا، پھر کمزوری کے بعد قوت دی، پھر قوت کے بعد کمزوری اور بڑھاپا دے دیا۔ ظاہر میں یہ عمر کی تین منزلیں ہیں: بچپن کی ناتوانی، جوانی کا زور، اور بڑھاپے کی سفیدی؛ اور یہی معنی اصل ہے، اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس آیت کی ترکیب پر نظر ڈالی جاتی ہے۔ پیدائش ضعف سے ہوئی، گویا کمزوری انسان کا مادہ ہے، اس کی اصل؛ قوت کے لیے خَلَقَ نہیں فرمایا، جَعَلَ فرمایا، یعنی رکھ دی گئی، اوپر سے عطا ہوئی۔ اصل ضعف ہے اور قوت عاریت۔ پھر عاریت واپس لی گئی اور بندہ اپنی اصل کی طرف لوٹ آیا۔ اہلِ معرفت کہتے ہیں کہ یہ دائرہ فقر سے شروع ہو کر فقر پر بند ہوتا ہے، اور بیچ کی قوت ہی امتحان کی گھڑی ہے، کیونکہ اسی میں بندہ بھول جاتا ہے کہ یہ زور کس کا ہے۔ قوت دراصل اسمِ القوی کا ظہور تھی جو بندے کے مظہر میں چمکا؛ جب وہ نور سمٹا تو معلوم ہوا کہ مالک کون تھا۔ اور شَيْبَة یعنی بالوں کی سفیدی کو کمزوری کے ساتھ جوڑا گیا؛ ایک مشہور حدیث کا مفہوم ہے کہ جس کے بال اسلام میں سفید ہوئے وہ سفیدی قیامت کے دن اس کے لیے نور ہوگی۔ عارف کے لیے بڑھاپا زوال نہیں، اپنی اصل کا اقرار ہے۔

اسی آیت کا دوسرا حصہ اس دائرے کا راز کھولتا ہے: يَخْلُقُ مَا يَشَآءُ ۚ وَهُوَ الْعَلِيْمُ الْقَدِيْرُ۔ وہ جو چاہتا ہے پیدا کرتا ہے، اور وہی سب کچھ جاننے والا، ہر چیز پر قدرت رکھنے والا ہے۔ یہاں علم کو قدرت سے پہلے رکھا گیا، اور شیخِ اکبر کے نزدیک اس ترتیب میں سرِّ قدر چھپا ہے: حق تعالیٰ ہر عینِ ثابتہ کو اس کی حقیقت کے مطابق جانتا ہے، اور قدرت اسی علم کے مطابق ظاہر کرتی ہے؛ علم معلوم کے تابع ہے اور قدرت علم کے۔ جو جس استعداد پر ہے، اسے اسی کے مطابق وجود ملتا ہے، کسی پر کوئی ایسی چیز نہیں ڈالی جاتی جو اس کی حقیقت سے اجنبی ہو۔ اس دائرے کی سب سے بڑی نصیحت یہ ہے کہ بندہ قوت کی منزل میں بھی ضعف کو یاد رکھے۔ قرآن نے سب انسانوں سے فرمایا: يٰۤاَيُّهَا النَّاسُ اَنْتُمُ الْفُقَرَآءُ اِلَى اللّٰهِ۔ اہلِ ذوق کہتے ہیں کہ فقر بڑھاپے میں آنے والی چیز نہیں، ہر حال میں بندے کی ذاتی صفت ہے؛ فرق صرف یہ ہے کہ بڑھاپا اسے آنکھ کے سامنے لے آتا ہے۔ سورۂ القصص میں حضرت موسیٰ علیہ السلام کا حال یاد کیجیے: پوری جوانی اور زورِ بازو کے ساتھ، اجنبی بستی میں پانی پلا کر درخت کے سائے میں بیٹھے تو اپنے رب کے حضور اپنی محتاجی کا اقرار کیا۔ یہی کمال ہے کہ قوت کے عین بیچ فقر پہچان لیا جائے۔ جس نے جوانی میں یہ پہچان لیا، اس کا بڑھاپا اس کے لیے کوئی نیا صدمہ نہیں لاتا، صرف ایک پرانی بات کی تصدیق کرتا ہے۔

عمر کا دائرہ بند ہوا تو بات اس گھڑی تک پہنچی جس پر سب دائرے بند ہوتے ہیں: وَيَوْمَ تَقُوْمُ السَّاعَةُ يُقْسِمُ الْمُجْرِمُوْنَ ۙ مَا لَبِثُوْا غَيْرَ سَاعَةٍ ۭ كَذٰلِكَ كَانُوْا يُؤْفَكُوْنَ۔ اور جس دن قیامت قائم ہوگی مجرم قسم کھائیں گے کہ وہ ایک گھڑی سے زیادہ نہیں ٹھہرے؛ اسی طرح وہ الٹے پھیرے جاتے تھے۔ ایک ہی آیت میں السَّاعَة دو بار آیا: ایک وہ بڑی گھڑی، ایک یہ چھوٹی گھڑی جس میں پوری زندگی سمٹ گئی۔ اہلِ معرفت کے نزدیک زمانہ ایک نسبت ہے؛ جو عمر دنیا میں لمبی دکھائی دیتی تھی، آخرت کی نظر سے ایک لمحہ نکلی۔ اور كَذٰلِكَ کا اشارہ بتاتا ہے کہ ان کی غلطی نئی نہیں: جیسے آج اندازہ غلط ہے ویسے ہی دنیا میں ہر اندازہ غلط تھا۔ حدیث کا مفہوم ہے کہ ہر بندہ اسی حال پر اٹھایا جائے گا جس پر مرا؛ الٹا دیکھنے کی عادت قبر کے پار بھی ساتھ گئی۔ اس کے مقابل اہلِ علم کی آواز آئی: وَقَالَ الَّذِيْنَ اُوْتُوا الْعِلْمَ وَالْاِيْمَانَ لَقَدْ لَبِثْتُمْ فِيْ كِتٰبِ اللّٰهِ اِلٰى يَوْمِ الْبَعْثِ فَهٰذَا يَوْمُ الْبَعْثِ وَلٰكِنَّكُمْ كُنْتُمْ لَا تَعْلَمُوْنَ۔ اور جنہیں علم اور ایمان دیا گیا تھا وہ کہیں گے: تم اللہ کے لکھے کے مطابق اٹھائے جانے کے دن تک ٹھہرے رہے؛ تو یہی اٹھائے جانے کا دن ہے، لیکن تم جانتے نہ تھے۔ علم اور ایمان ایک ساتھ دیے گئے؛ اس مکتب میں ایک کے بغیر دوسرا ادھورا ہے۔ اور فِيْ كِتٰبِ اللّٰهِ میں لطف یہ ہے کہ ٹھہرنا بھی کتاب کے اندر تھا؛ زمان اور برزخ سب اسی لکھے ہوئے کے پیمانے میں ہیں۔ آخری جملہ پورے رکوع کا محور ہے: ان کی اصل بیماری نہ جاننا تھی، اور یہی لفظ آگے مہر کی آیت میں لوٹے گا۔

پھر وہ دن جس میں عذر کا دروازہ بند ہو چکا: فَيَوْمَئِذٍ لَّا يَنْفَعُ الَّذِيْنَ ظَلَمُوْا مَعْذِرَتُهُمْ وَلَا هُمْ يُسْتَعْتَبُوْنَ۔ تو اس دن ظالموں کو ان کی معذرت کچھ فائدہ نہ دے گی، اور نہ ان سے راضی کرنے کو کہا جائے گا۔ اِسْتِعْتَاب کا مطلب ہے کسی کو منانے اور روٹھے کو راضی کرنے کا موقع دینا۔ اہلِ ذوق کہتے ہیں کہ یہ آیت دراصل آج کا دروازہ دکھاتی ہے: اس دن استعتاب نہیں، اس لیے آج ہے۔ آج محبوب کا عتاب بھی رحمت ہے، کیونکہ عتاب وہیں ہوتا ہے جہاں تعلق باقی ہو؛ جو آج اپنی کوتاہی پر رو لے اس کی معذرت قبول کے دن میں پیش ہو رہی ہے۔ پھر قرآن نے اپنی وسعت کا ذکر کیا: وَلَقَدْ ضَرَبْنَا لِلنَّاسِ فِيْ هٰذَا الْقُرْاٰنِ مِنْ كُلِّ مَثَلٍ ۭ وَلَئِنْ جِئْتَهُمْ بِاٰيَةٍ لَّيَقُوْلَنَّ الَّذِيْنَ كَفَرُوْۤا اِنْ اَنْتُمْ اِلَّا مُبْطِلُوْنَ۔ اور ہم نے اس قرآن میں لوگوں کے لیے ہر طرح کی مثال بیان کر دی ہے، اور اگر آپ ان کے پاس کوئی بھی نشانی لے آئیں تو کافر یہی کہیں گے کہ تم سب جھوٹ پر ہو۔ شیخِ اکبر کے نزدیک مثال وہ پل ہے جس پر معنی صورت میں اتر کر قریب آتا ہے، اور ہر استعداد کے لیے الگ مثال ہے؛ مِنْ كُلِّ مَثَلٍ کا مطلب یہ کہ کوئی دل ایسا نہیں چھوڑا گیا جس کے ناپ کی مثال قرآن میں نہ ہو۔ حجت پوری ہو چکی۔ اس کے بعد بھی انکار ہو تو مسئلہ مثال میں نہیں، سننے والے کے رخ میں ہے؛ اور منکر نے اپنا باطل اہلِ حق پر ڈال دیا، جو اعراض کی پرانی پہچان ہے۔

اب وہ آیت جسے اصول کی روشنی میں پڑھنا لازم ہے، اور جو خود اپنی شرط ساتھ لاتی ہے: كَذٰلِكَ يَطْبَعُ اللّٰهُ عَلٰى قُلُوْبِ الَّذِيْنَ لَا يَعْلَمُوْنَ۔ اسی طرح اللہ ان لوگوں کے دلوں پر مہر لگا دیتا ہے جو علم نہیں رکھتے۔ اس آیت کا پہلا لفظ كَذٰلِكَ پوری کنجی ہے: یعنی اسی طریقے سے جو ابھی بیان ہوا۔ ہر مثال دی گئی، ہر نشانی آئی، اور انہوں نے ہر نشانی کو جھوٹ کہا؛ مہر اس پورے عمل کا آخری نتیجہ ہے، اس کا آغاز نہیں۔ اور جن پر مہر ہے ان کی پہچان بھی بتا دی: لَا يَعْلَمُوْنَ، وہ جو نہیں جانتے؛ یہ وہ نادانی نہیں جو عاجزی سے ہو، بلکہ وہ جو جاننے سے انکار سے پیدا ہوئی، کیونکہ علم کا ہر دروازہ ان کے سامنے کھولا گیا تھا۔ اللہ کسی کو زبردستی گمراہ نہیں کرتا اور جو ہدایت کا طالب ہو اسے کبھی گمراہ نہیں کرتا؛ جو طلب کرے پاتا ہے، اور جو منہ موڑے اس کا منہ موڑنا ہی اس کی گمراہی ہے۔ مہر دراصل اس دروازے پر تصدیق کی مہر ہے جسے بندے نے اپنے ہاتھ سے بند کیا تھا؛ حق نے اسی رخ کو وجود دیا جو اس نے چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ اگر وہ مجبور کرنا چاہتا تو کر سکتا تھا، جیسا فرمایا وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ، مگر اس نے بندے کو اختیار کی عزت دی۔ اور جب تک سانس باقی ہے، مہر والا دل بھی پلٹ سکتا ہے، کیونکہ پچھلے رکوع نے بتایا تھا کہ وہ مردہ زمین کو زندہ کرتا ہے۔

سورت کا آخری کلمہ حضورِ اکرم صلی اللہ علیہ وسلم کے قلبِ اطہر سے خطاب ہے: فَاصْبِرْ اِنَّ وَعْدَ اللّٰهِ حَقٌّ وَّلَا يَسْتَخِفَّنَّكَ الَّذِيْنَ لَا يُوْقِنُوْنَ۔ پس آپ صبر کیجیے، بےشک اللہ کا وعدہ سچا ہے، اور جو لوگ یقین نہیں رکھتے وہ آپ کو ہلکا نہ کر پائیں۔ سورت رومیوں کی شکست اور فتح کے وعدے سے شروع ہوئی تھی، اور وعدے ہی پر ختم ہوئی؛ دائرہ پورا ہوا، اور اس دائرے کی سچائی تاریخ نے ظاہر میں دکھا دی۔ صبر یہاں بےبسی کا نام نہیں، یقین کا پھل ہے: جو وعدے کو حق جان لے وہ انتظار میں بھی ٹھہرا رہتا ہے۔ اور يَسْتَخِفَّنَّكَ میں خفت یعنی ہلکے پن کا معنی ہے؛ بےیقینی آدمی کو ہلکا کر دیتی ہے، ہر جھونکا اسے اڑا لے جاتا ہے، اور ہلکا آدمی دوسروں کو بھی ہلکا کرنا چاہتا ہے۔ یقین وزن ہے، وقار ہے۔ اہلِ معرفت کہتے ہیں کہ یہی تلوین اور تمکین کا فرق ہے: صاحبِ تلوین حالات کے ساتھ رنگ بدلتا ہے، صاحبِ تمکین پہاڑ کی طرح جما رہتا ہے۔ یقین کے بھی درجے ہیں: علم الیقین سننے سے، عین الیقین دیکھنے سے، اور حق الیقین اس میں فنا ہو کر؛ اور سورت کا سفر اسی ترتیب میں ہے، پہلے خبر، پھر نشانیاں، پھر وہ یقین جسے کوئی منکر ہلکا نہیں کر سکتا۔
