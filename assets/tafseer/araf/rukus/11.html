رسولوں کے سلسلے کی یہ پانچویں کڑی ہے، اور آغاز اسی جانے پہچانے کلمے سے ہوتا ہے: وَاِلٰى مَدْيَنَ اَخَاهُمْ شُعَيْبًا ؕ قَالَ يٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَيْرُهٗ ؕ قَدْ جَآءَتْكُمْ بَيِّنَةٌ مِّنْ رَّبِّكُمْ۔ “اور مدین کی طرف ان کے بھائی شعیب علیہ السلام کو بھیجا۔ انہوں نے کہا: اے میری قوم، اللہ کی عبادت کرو، اس کے سوا تمہارا کوئی معبود نہیں۔ تمہارے پاس تمہارے رب کی طرف سے روشن دلیل آ چکی ہے۔” مفسرین بیان کرتے ہیں کہ مدین ایک قبیلہ تھا جو حضرت ابراہیم علیہ السلام کے فرزند مدین کی نسل سے تھا اور خلیجِ عقبہ کے قریب حجاز اور شام کی تجارتی شاہراہ پر آباد تھا؛ ان کی زندگی بازار اور قافلوں سے جڑی تھی۔ حضرت شعیب علیہ السلام کو ان کی شیریں اور مدلل گفتگو کی بنا پر بعض مفسرین نے خطیب الانبیاء کے لقب سے یاد کیا ہے۔ ترتیب پر غور کیجیے: حضرت نوح، حضرت ہود اور حضرت صالح علیہم السلام کی طرح یہاں بھی پہلی بات توحید ہے، ناپ تول کا ذکر اس کے بعد آتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہ ترتیب ایک اصول سکھاتی ہے: جو دل حق کو اس کا حق نہیں دیتا وہ مخلوق کو بھی اس کا حق پورا نہیں دے سکتا۔ بازار کا ترازو دل کے ترازو کی شاخ ہے؛ جب اندر کا میزان سیدھا ہو تو باہر کا میزان خود سیدھا ہونے لگتا ہے۔ “اَخَاهُمْ” کا لفظ بتاتا ہے کہ رسول اجنبی بن کر نہیں آتا، اپنوں میں سے، اپنوں کی زبان اور اپنوں کے دکھ کے ساتھ آتا ہے۔ اور “بَيِّنَةٌ” کی تفصیل یہاں نہیں دی گئی؛ مفسرین کہتے ہیں کہ کوئی معجزہ تھا جس کا ذکر قرآن نے نہیں کیا، یا خود ان کی دعوت ہی اپنی سچائی کی دلیل تھی۔

اب وہ احکام جو اس قوم کی خاص بیماری کا علاج تھے: فَاَوْفُوا الْكَيْلَ وَالْمِيْزَانَ وَلَا تَبْخَسُوا النَّاسَ اَشْيَآءَهُمْ وَلَا تُفْسِدُوْا فِي الْاَرْضِ بَعْدَ اِصْلَاحِهَا ؕ ذٰلِكُمْ خَيْرٌ لَّكُمْ اِنْ كُنْتُمْ مُّؤْمِنِيْنَ۔ “پس ناپ اور تول پورا کرو، لوگوں کو ان کی چیزیں گھٹا کر نہ دو، اور زمین میں اس کی اصلاح کے بعد فساد نہ پھیلاؤ۔ یہی تمہارے لیے بہتر ہے اگر تم ایمان والے ہو۔” ظاہر بالکل واضح ہے: یہ لوگ ناپ تول میں کمی کرتے تھے، خریدتے وقت زیادہ لیتے اور بیچتے وقت کم دیتے۔ مگر “وَلَا تَبْخَسُوا النَّاسَ اَشْيَآءَهُمْ” کا دائرہ ترازو سے بڑا ہے؛ بخس ہر وہ کمی ہے جو کسی کے حق میں کی جائے، چاہے مال میں ہو، مزدوری میں ہو، یا کسی کی خوبی اور قدر کو گھٹا کر بیان کرنے میں۔ “بَعْدَ اِصْلَاحِهَا” سے مفسرین یہ سمجھتے ہیں کہ رسولوں اور شریعت کے ذریعے جو زمین سنواری گئی، اسے پھر بگاڑ دینا دوہرا جرم ہے۔ اہلِ معرفت کہتے ہیں کہ “اَشْيَآءَهُمْ” میں ایک لطیف بات ہے: ہر شے کا ایک حق ہے، اور شیخِ اکبر کے نزدیک ہر موجود کی ایک استعداد ہے جو اس کے عینِ ثابتہ میں لکھی ہے؛ حق تعالیٰ ہر شے کو اس کی استعداد کے مطابق پورا عطا فرماتا ہے، کسی کا حصہ نہیں گھٹاتا۔ جو بندہ اس صفتِ عدل سے متخلق ہو، وہ ہر چیز کو اس کا پورا حق دیتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ پہلے بازار کا ترازو سیدھا ہو، تب دل کے ترازو کی بات معتبر ہے۔ اور آخری جملہ “اِنْ كُنْتُمْ مُّؤْمِنِيْنَ” بتاتا ہے کہ دیانت ایمان کا پھل ہے، الگ کوئی ضابطہ نہیں۔

پھر ایک اور جرم کی طرف اشارہ: وَلَا تَقْعُدُوْا بِكُلِّ صِرَاطٍ تُوْعِدُوْنَ وَتَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ مَنْ اٰمَنَ بِهٖ وَتَبْغُوْنَهَا عِوَجًا ۚ وَاذْكُرُوْۤا اِذْ كُنْتُمْ قَلِيْلًا فَكَثَّرَكُمْ ۚ وَانْظُرُوْا كَيْفَ كَانَ عَاقِبَةُ الْمُفْسِدِيْنَ۔ “اور ہر راستے پر دھمکیاں دیتے ہوئے نہ بیٹھو، اور اللہ کی راہ سے اس کو نہ روکو جو اس پر ایمان لائے، اور اس راہ میں ٹیڑھ نہ ڈھونڈو۔ اور یاد کرو جب تم تھوڑے تھے تو اس نے تمہیں بڑھا دیا، اور دیکھو فساد کرنے والوں کا انجام کیسا ہوا۔” مفسرین بتاتے ہیں کہ یہ لوگ راستوں پر بیٹھ کر مسافروں کو ڈراتے، ان سے محصول اینٹھتے، اور جو کوئی حضرت شعیب علیہ السلام کی طرف آنا چاہتا اسے دھمکا کر لوٹا دیتے تھے۔ اب اسی سورت کے دوسرے رکوع کی طرف لوٹیے: ابلیس نے کہا تھا فَبِمَاۤ اَغْوَيْتَنِيْ لَاَقْعُدَنَّ لَهُمْ صِرَاطَكَ الْمُسْتَقِيْمَ؛ وہاں بھی بیٹھنا ہے اور صراط بھی، یہاں بھی۔ اہلِ تاویل کہتے ہیں کہ جو دوسروں کی راہ روکنے بیٹھتا ہے وہ اسی پہلے روکنے والے کی نشست سنبھال لیتا ہے۔ اور یاد رہے کہ ابلیس نے اپنی گمراہی کا الزام اپنے رب پر ڈالا تھا، اور وہ قول اسی کا دعویٰ ہے، اصول نہیں؛ اصول وہ ہے جو حضرت آدم علیہ السلام کی توبہ میں ظاہر ہوا کہ بندہ اپنے نفس کو قصوروار مانے۔ “وَتَبْغُوْنَهَا عِوَجًا” دکھاتا ہے کہ روکنے والا راہ کو ٹیڑھا ثابت کرنے کی دلیلیں بھی گھڑتا ہے۔ پھر شکر کی یاد دہانی: تم تھوڑے تھے، اس نے بڑھایا۔ اہلِ معرفت کہتے ہیں کہ کثرت نعمت بھی ہے اور امتحان بھی؛ جو کثرت کو شکر میں نہ بدلے، اس کے لیے وہ فساد کا ایندھن بن جاتی ہے۔

حضرت شعیب علیہ السلام کی تحمل بھری بات دیکھیے: وَاِنْ كَانَ طَآئِفَةٌ مِّنْكُمْ اٰمَنُوْا بِالَّذِيْۤ اُرْسِلْتُ بِهٖ وَطَآئِفَةٌ لَّمْ يُؤْمِنُوْا فَاصْبِرُوْا حَتّٰى يَحْكُمَ اللّٰهُ بَيْنَنَا ۚ وَهُوَ خَيْرُ الْحٰكِمِيْنَ۔ “اور اگر تم میں سے ایک گروہ اس پیغام پر ایمان لے آیا ہے جس کے ساتھ میں بھیجا گیا ہوں اور ایک گروہ ایمان نہیں لایا، تو صبر کرو یہاں تک کہ اللہ ہمارے درمیان فیصلہ فرما دے، اور وہی سب سے بہتر فیصلہ کرنے والا ہے۔” یہ خطاب دونوں گروہوں سے ہے: ایمان والوں کو تسلی کہ ظلم پر صبر کریں، اور انکار کرنے والوں کو مہلت کہ زبردستی نہ کریں، فیصلہ اللہ پر چھوڑ دیں۔ اہلِ معرفت کہتے ہیں کہ “حَتّٰى يَحْكُمَ اللّٰهُ” میں صبر کی حد بھی ہے اور اس کا سہارا بھی: صبر بے انتہا انتظار نہیں، اس حکم کا انتظار ہے جو “خَيْرُ الْحٰكِمِيْنَ” کی طرف سے آئے گا۔

مگر سرداروں کا جواب صبر نہیں، دھمکی تھی: قَالَ الْمَلَاُ الَّذِيْنَ اسْتَكْبَرُوْا مِنْ قَوْمِهٖ لَنُخْرِجَنَّكَ يٰشُعَيْبُ وَالَّذِيْنَ اٰمَنُوْا مَعَكَ مِنْ قَرْيَتِنَاۤ اَوْ لَتَعُوْدُنَّ فِيْ مِلَّتِنَا ؕ قَالَ اَوَلَوْ كُنَّا كٰرِهِيْنَ۔ “ان کی قوم کے متکبر سرداروں نے کہا: اے شعیب علیہ السلام، ہم تمہیں اور تمہارے ساتھ ایمان لانے والوں کو اپنی بستی سے ضرور نکال دیں گے، ورنہ تمہیں ہمارے دین میں لوٹنا ہوگا۔ انہوں نے فرمایا: کیا اس صورت میں بھی کہ ہم اسے ناپسند کرتے ہوں؟” یہ مختصر جواب ایک پوری حکمت ہے: عقیدہ دل کا معاملہ ہے، اور جبر سے دل نہیں بدلتا؛ ناپسندیدگی کے ساتھ اختیار کیا ہوا دین دین ہی نہیں۔ قرآن نے یہ اصول دوسری جگہ کھول کر فرمایا: لَاۤ اِكْرَاهَ فِي الدِّيْنِ۔ اہلِ معرفت کہتے ہیں کہ حق سبحانہ چاہتا تو سب کو مجبور کر دیتا، مگر اس نے انسان کو اختیار کی عزت دی؛ رسول بھی دعوت دیتا ہے، دباتا نہیں۔ جبر کا راستہ ہمیشہ متکبر سرداروں نے اپنایا، اور یہ ان کے کمزور موقف کی علامت تھی: جس کے پاس دلیل نہ ہو وہ دھمکی اٹھاتا ہے۔

اب وہ آیت جو اس رکوع کا دل ہے: قَدِ افْتَرَيْنَا عَلَى اللّٰهِ كَذِبًا اِنْ عُدْنَا فِيْ مِلَّتِكُمْ بَعْدَ اِذْ نَجّٰىنَا اللّٰهُ مِنْهَا ؕ۔ “اگر ہم تمہارے دین میں لوٹ آئیں جبکہ اللہ ہمیں اس سے نجات دے چکا ہے تو ہم نے اللہ پر جھوٹ باندھا۔” یہاں ایک ادب کی بات پہلے طے کر لیجیے: انبیاء علیہم السلام معصوم ہیں، اور حضرت شعیب علیہ السلام کبھی ان کے شرک والے دین میں نہیں تھے۔ مفسرین نے “لوٹنے” کے لفظ کی دو توجیہیں کی ہیں: ایک یہ کہ آپ علیہ السلام اپنے ساتھ کے ان اہلِ ایمان کی طرف سے بھی بات فرما رہے تھے جو پہلے اسی دین پر تھے، اور جماعت کی زبان میں سب کو شامل کر لیا گیا؛ دوسری یہ کہ عربی میں عَادَ کا لفظ کبھی “ہو جانے” کے معنی میں بھی آتا ہے، اور یہ سرداروں ہی کے لفظ کا جواب ہے۔ اب جملے کی گہرائی دیکھیے: آپ علیہ السلام نے لوٹنے کو صرف غلطی نہیں کہا، اللہ پر جھوٹ باندھنا کہا۔ کیوں؟ اس لیے کہ جو حق کو پہچان کر باطل کی طرف پلٹے، وہ اپنے عمل سے یہ گواہی دیتا ہے کہ جو اللہ نے دکھایا وہ سچ نہیں تھا۔ اہلِ معرفت کہتے ہیں کہ پہچان ایک امانت ہے؛ جس دل پر حق کا نور اتر چکا، اس کا پلٹنا عام غفلت نہیں، اپنے ہی مشاہدے کی تکذیب ہے۔ اور “نَجّٰىنَا اللّٰهُ” میں نجات کی نسبت اللہ کی طرف ہے: ہدایت پر شکر ہے، اپنے کمال کا دعویٰ نہیں۔ شیخِ اکبر کے نزدیک عارف کی پہچان یہی ہے کہ وہ ہر خیر کو حق کی عطا دیکھتا ہے اور اپنے حصے میں صرف قبول کی توفیق کا شکر رکھتا ہے۔

اور پھر وہ استثنا جس میں نبوت کا ادب جھلکتا ہے: وَمَا يَكُوْنُ لَنَاۤ اَنْ نَّعُوْدَ فِيْهَاۤ اِلَّاۤ اَنْ يَّشَآءَ اللّٰهُ رَبُّنَا ؕ۔ “اور ہمارے لیے ممکن نہیں کہ ہم اس میں لوٹیں، مگر یہ کہ اللہ ہمارا رب چاہے۔” سرسری نظر سے یہ بات عجیب لگ سکتی ہے، مگر اس میں شک کا کوئی شائبہ نہیں؛ نبی معصوم ہے اور اس کا لوٹنا ممکن ہی نہیں۔ یہ استثنا ادب کا ہے: حضرت شعیب علیہ السلام اپنی ثابت قدمی کو اپنی طاقت کی طرف منسوب نہیں فرماتے، بلکہ مشیتِ الٰہی کے سپرد کرتے ہیں۔ متکبر سردار کہتے تھے “ہم نکال دیں گے”، “تمہیں لوٹنا ہوگا”؛ اور رسول فرماتے ہیں: نہ ہمارا ٹھہرنا ہمارے بس میں، نہ کسی کا ہٹانا تمہارے بس میں؛ سب کچھ اللہ کی مشیت میں ہے۔ یہاں ہدایت اور گمراہی کا وہ اصول بھی سامنے رہے جو اس سلسلے میں طے ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور جو اس کی طلب میں ہو اسے کبھی گمراہ نہیں کرتا۔ یہ جملہ اس اندیشے کا اظہار نہیں کہ اللہ ایک سچے طالب کو گمراہی کی طرف پھیر دے گا؛ یہ بندے کے فقر کا اقرار ہے کہ ہدایت پر قائم رہنا بھی اسی کی توفیق سے ہے۔ اہلِ ذوق کہتے ہیں کہ “رَبُّنَا” کا اضافہ دیکھیے: آپ علیہ السلام نے مشیت کو ربوبیت کے نام کے ساتھ جوڑا، یعنی وہ جو ہماری پرورش کرتا آیا ہے؛ جو رب پالتا ہے وہ اپنے طالب کو ضائع نہیں کرتا۔ یہ استثنا خوف اور اعتماد دونوں کو ایک ساتھ تھامے ہوئے ہے، اور یہی سالک کا صحیح حال ہے۔

اس کے بعد وسعتِ علم کا اعلان اور توکل کا عہد: وَسِعَ رَبُّنَا كُلَّ شَيْءٍ عِلْمًا ؕ عَلَى اللّٰهِ تَوَكَّلْنَا ؕ۔ “ہمارے رب کا علم ہر چیز پر چھایا ہوا ہے۔ ہم نے اللہ ہی پر بھروسا کیا۔” یہ دونوں جملے پچھلے استثنا کی شرح ہیں: ہم اپنا انجام نہیں جانتے، ہمارا رب جانتا ہے؛ اس لیے ہم اپنی پونجی پر نہیں، اسی پر بھروسا رکھتے ہیں۔ عربی ترکیب میں “عَلَى اللّٰهِ” کو پہلے لانا حصر کا فائدہ دیتا ہے: صرف اللہ پر، کسی اور پر نہیں؛ نہ اپنی جماعت پر، نہ اپنی دلیل پر، نہ اپنی بستی کی زمین پر جس سے نکالے جانے کی دھمکی دی جا رہی تھی۔ یہی بات حضرت شعیب علیہ السلام کی زبان سے سورۂ ہود میں یوں آئی: وَمَا تَوْفِيْقِيْۤ اِلَّا بِاللّٰهِ ؕ عَلَيْهِ تَوَكَّلْتُ وَاِلَيْهِ اُنِيْبُ۔ شیخِ اکبر کے مکتب میں توکل کا راز علم سے کھلتا ہے: جو جان لے کہ حق کا علم ہر شے کو گھیرے ہوئے ہے اور ہر شے کا حال اس کے علم کے مطابق ظاہر ہوتا ہے، اس کے دل سے مخلوق کا ڈر نکل جاتا ہے۔ اور یہاں سرِّ قدر کی ایک جھلک بھی ہے: حق کا علم چیزوں کو ویسا ہی جانتا ہے جیسی وہ ہیں، اور حکم اسی کے مطابق ہوتا ہے؛ بندے کا کام اپنے رخ کو درست رکھنا ہے، باقی سب اسی علیم کے سپرد۔ اہلِ معرفت کہتے ہیں کہ توکل اسباب کا انکار نہیں، اسباب سے دل کی آزادی ہے؛ حضرت شعیب علیہ السلام نے دعوت کا سبب پورا اختیار کیا، مگر دل کو اس کے نتیجے سے نہیں باندھا۔

اور اب وہ دعا جو اس رکوع کا عنوان ہے: رَبَّنَا افْتَحْ بَيْنَنَا وَبَيْنَ قَوْمِنَا بِالْحَقِّ وَاَنْتَ خَيْرُ الْفٰتِحِيْنَ۔ “اے ہمارے رب، ہمارے اور ہماری قوم کے درمیان حق کے ساتھ فیصلہ فرما دے، اور تو سب سے بہتر فیصلہ فرمانے والا ہے۔” حضرت عبد اللہ بن عباس رضی اللہ عنہما سے منقول ہے کہ یہاں “افْتَحْ” کے معنی “فیصلہ فرما” کے ہیں، اور یہ معنی عرب کی بعض زبانوں میں معروف تھا؛ اسی سے قاضی کو فتّاح کہا جاتا ہے۔ مگر لفظ میں کھولنے کا مفہوم بھی باقی ہے: فیصلہ وہ ہے جو بند معاملے کو کھول دے اور حق و باطل کو الگ کر دے۔ دعا کا ادب دیکھیے: آپ علیہ السلام نے قوم کی ہلاکت نہیں مانگی، “بِالْحَقِّ” فیصلہ مانگا؛ اپنی جیت نہیں مانگی، حق کی جیت مانگی۔ اور فیصلہ “بَيْنَنَا وَبَيْنَ قَوْمِنَا” ہے، یعنی آپ علیہ السلام نے خود کو بھی فیصلے کے ترازو میں رکھا؛ یہ وہی میزان ہے جس کی دعوت دی تھی۔ اہلِ معرفت کے نزدیک “فتح” سلوک کی بھی اصطلاح ہے: وہ گھڑی جب حق دل پر کوئی دروازہ کھول دے اور جو پردے میں تھا آشکار ہو جائے۔ شیخِ اکبر کے مکتب میں فتح مجاہدے کا انعام بھی ہے اور محض عطا بھی؛ اور یہ کہ “خَيْرُ الْفٰتِحِيْنَ” اسی کا نام ہے، بتاتا ہے کہ کھولنا اسی کا کام ہے۔ بندہ دروازے پر کھڑا رہتا ہے، دستک دیتا ہے، اور کھولنے والا وہ ہے۔ جو دل اپنے اور اپنی خواہشوں کے درمیان فیصلہ مانگے، اس کے لیے یہ دعا ہر روز کی دعا ہے: اے رب، میرے اور میرے نفس کے درمیان حق کے ساتھ فیصلہ فرما۔

فصوص الحکم میں حضرت شعیب علیہ السلام کا باب حکمتِ قلبیہ کے نام سے ہے، اور اس کا مفہوم یہ ہے کہ عارف کا دل ایسی وسعت رکھتا ہے کہ حق کی تجلیات کو قبول کرے، اور قلب کا نام ہی تقلب سے ہے: وہ ہر لمحہ نئی تجلی کے ساتھ نیا رنگ قبول کرتا ہے اور ہر تجلی کو اپنی استعداد کے مطابق پاتا ہے۔ اسی باب میں شیخِ اکبر خلقِ جدید کی بات کرتے ہیں: عالم ہر سانس میں نیا ہوتا ہے، اور اکثر لوگ اس سے غافل ہیں۔ اہلِ تصوف میں ایک مشہور اثر ہے کہ زمین و آسمان مجھے نہ سما سکے اور میرے مومن بندے کے دل نے مجھے سما لیا؛ اس کا مطلب حلول نہیں، بلکہ یہ کہ مومن کا دل حق کی معرفت اور اس کی تجلیات کا ظرف بنتا ہے، اور حق سبحانہ ہر ظرف سے پاک اور بلند ہے۔ اس رکوع پر یہ حکمت کیسے اترتی ہے؟ مدین کے سرداروں کا دل اپنی پرانی ملت میں بند تھا؛ انہوں نے باپ دادا کے طریقے کو دائرہ بنا لیا اور اس سے باہر کی ہر بات کو ٹیڑھ کہا۔ اور حضرت شعیب علیہ السلام کا دل وسیع تھا: اس میں دھمکی کے سامنے ثابت قدمی بھی تھی، قوم کے لیے خیر خواہی بھی، اور اپنی ثابت قدمی کو بھی اللہ کی مشیت کے سپرد کرنے کا ادب بھی۔ اہلِ ذوق کہتے ہیں کہ ترازو اور دل میں ایک مناسبت ہے: دونوں کا کام توازن ہے، دونوں جھکتے ہیں، اور دونوں کی سچائی اس میں ہے کہ وہ جھکیں تو حق کی طرف جھکیں۔ یہ اشارہ ہے، ظاہر کا بدل نہیں: ظاہر میں ناپ تول کا حکم شریعت کا قطعی حکم ہے، اور دل کی وسعت کا دعویٰ وہ کرے جس کا ترازو پہلے سیدھا ہو۔

ایک اور نکتہ اسی حکمت سے جڑا ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ بنی آدم کے دل رحمٰن کی انگلیوں میں سے دو انگلیوں کے درمیان ہیں، وہ جیسے چاہتا ہے انہیں پھیرتا ہے؛ یہ صحیح مسلم کی حدیث کا مفہوم ہے، اور اہلِ سنت اسے بلا کیف اور بغیر تشبیہ مانتے ہیں۔ اسی لیے آپ صلی اللہ علیہ وسلم کثرت سے یہ دعا فرمایا کرتے تھے کہ اے دلوں کے پھیرنے والے، میرے دل کو اپنے دین پر ثابت رکھ۔ حضرت شعیب علیہ السلام کا “اِلَّاۤ اَنْ يَّشَآءَ اللّٰهُ رَبُّنَا” اسی ادب کا اظہار ہے۔ مگر یہاں وہ اصول پھر یاد رکھیے جو اس سلسلے کی بنیاد ہے: دلوں کا پھیرنا اندھا جبر نہیں۔ شیخِ اکبر کے نزدیک سرِّ قدر یہ ہے کہ حکم محکوم کے حال کے مطابق ہوتا ہے؛ جو دل حق کی طرف رخ کرے، اسے حق کی طرف پھیرا جاتا ہے، اور جو منہ موڑے، اس کا منہ موڑنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ اسی لیے قلب کا تقلب عارف کے لیے خوف کی بات بھی ہے اور امید کی بھی: خوف اس لیے کہ دل ایک حال پر نہیں ٹھہرتا، اور امید اس لیے کہ پھیرنے والا وہ ہے جو طالب کو خالی نہیں لوٹاتا۔ یہی وجہ ہے کہ اہلِ معرفت ثابت قدمی کو اپنی کمائی نہیں سمجھتے، اور ہر صبح اسی دعا سے دل کو اس کے مالک کے سپرد کرتے ہیں۔

اب انجام کی طرف کہانی تیزی سے بڑھتی ہے: وَقَالَ الْمَلَاُ الَّذِيْنَ كَفَرُوْا مِنْ قَوْمِهٖ لَئِنِ اتَّبَعْتُمْ شُعَيْبًا اِنَّكُمْ اِذًا لَّخٰسِرُوْنَ۔ “اور ان کی قوم کے کافر سرداروں نے کہا: اگر تم نے شعیب علیہ السلام کی پیروی کی تو تم یقیناً گھاٹے میں رہو گے۔” ان کا گھاٹا ترازو والا گھاٹا تھا: پورا تولو گے تو منافع کم ہوگا، راستوں سے محصول نہ لو گے تو آمدنی گھٹے گی۔ پھر: فَاَخَذَتْهُمُ الرَّجْفَةُ فَاَصْبَحُوْا فِيْ دَارِهِمْ جٰثِمِيْنَ۔ “پس انہیں زلزلے نے آ پکڑا، تو وہ اپنے گھروں میں اوندھے منہ پڑے رہ گئے۔” سورۂ ہود میں ان کے عذاب کو چیخ کہا گیا اور سورۂ الشعراء میں اصحابِ ایکہ پر سائبان کے دن کا عذاب آیا؛ جو مفسرین اصحابِ ایکہ کو اہلِ مدین ہی مانتے ہیں وہ کہتے ہیں کہ یہ ایک ہی عذاب کے مختلف پہلو ہیں: اوپر سے ہولناک آواز، نیچے سے زمین کی لرزش۔ الَّذِيْنَ كَذَّبُوْا شُعَيْبًا كَاَنْ لَّمْ يَغْنَوْا فِيْهَا ۚ اَلَّذِيْنَ كَذَّبُوْا شُعَيْبًا كَانُوْا هُمُ الْخٰسِرِيْنَ۔ “جنہوں نے شعیب علیہ السلام کو جھٹلایا، گویا وہ کبھی ان گھروں میں بسے ہی نہ تھے؛ جنہوں نے شعیب علیہ السلام کو جھٹلایا وہی گھاٹے میں رہے۔” قرآن نے ان کا اپنا لفظ ان پر لوٹا دیا: جو دوسروں کو گھاٹے سے ڈراتے تھے، خود اصل گھاٹے والے نکلے۔ یہ تاریخ کی ایک قوم کا واقعہ ہے جو قرآن نے سنجیدگی سے بیان کیا، اور اس کا اطلاق آج کی کسی قوم پر کرنا اس کا مقصد نہیں؛ اس کا رخ پڑھنے والے کے اپنے دل کی طرف ہے۔ اہلِ معرفت کہتے ہیں کہ جس زمین پر ترازو لرزتا تھا، آخرکار وہی زمین لرز اٹھی؛ اور شیخِ اکبر کے نزدیک ہر جزا عمل کی ہم شکل ہوتی ہے، کیونکہ حکم عمل ہی سے پھوٹتا ہے۔

اور آخر میں ایک رسول کی وداعی: فَتَوَلّٰى عَنْهُمْ وَقَالَ يٰقَوْمِ لَقَدْ اَبْلَغْتُكُمْ رِسٰلٰتِ رَبِّيْ وَنَصَحْتُ لَكُمْ ۚ فَكَيْفَ اٰسٰى عَلٰى قَوْمٍ كٰفِرِيْنَ۔ “تب شعیب علیہ السلام ان سے منہ پھیر کر چلے اور فرمایا: اے میری قوم، میں نے تمہیں اپنے رب کے پیغامات پہنچا دیے اور تمہاری خیر خواہی کی؛ تو اب میں انکار کرنے والی قوم پر کیسے غم کروں؟” یہ جملہ حضرت صالح علیہ السلام کی وداعی سے ملتا جلتا ہے جو پچھلے رکوع میں گزری، اور دونوں میں ایک ہی سبق ہے۔ “يٰقَوْمِ” کا خطاب آخر تک باقی ہے: ہلاکت کے بعد بھی آپ علیہ السلام انہیں “میری قوم” فرماتے ہیں؛ رشتہ ٹوٹا نہیں، فیصلہ ہو گیا۔ “وَنَصَحْتُ لَكُمْ” بتاتا ہے کہ دعوت کی روح خیر خواہی ہے، جیت نہیں۔ اور “فَكَيْفَ اٰسٰى” میں غم بھی ہے اور غم کا تھم جانا بھی: دل دکھا، مگر حق کے فیصلے کے سامنے دل نے سر جھکا دیا۔ اہلِ معرفت کہتے ہیں کہ یہ انبیاء علیہم السلام کی شفقت اور رضا کا جمع ہے؛ وہ آخری لمحے تک قوم کے لیے تڑپتے ہیں، مگر جب حق کا حکم ظاہر ہو جائے تو اپنی تڑپ کو اس کے حکم پر مقدم نہیں کرتے۔ شیخِ اکبر کے مکتب میں یہی دل کی وسعت کا آخری درجہ ہے: ایک ہی دل میں رحمت بھی سمائے اور حق کے عدل پر رضا بھی۔ سالک کے لیے سبق یہ ہے کہ جس کی خیر خواہی کرو، پوری کرو؛ پھر نتیجہ اللہ پر چھوڑو، اور جو حق کو رد کرے اس کے انجام کو حق کے سپرد کر کے اپنا دل اسی کے فیصلے پر راضی رکھو۔
