اعراف والوں کی دوسری پکار اب دوزخ کی طرف ہے، اور اس میں ملامت بھی ہے اور عبرت بھی: وَنَادٰۤى اَصْحٰبُ الْاَعْرَافِ رِجَالًا يَّعْرِفُوْنَهُمْ بِسِيْمٰىهُمْ قَالُوْا مَاۤ اَغْنٰى عَنْكُمْ جَمْعُكُمْ وَمَا كُنْتُمْ تَسْتَكْبِرُوْنَ۔ “اور اعراف والے کچھ ایسے لوگوں کو، جنہیں وہ ان کی علامتوں سے پہچانتے ہوں گے، پکار کر کہیں گے: نہ تمہاری جمعیت تمہارے کچھ کام آئی اور نہ وہ تکبر جو تم کیا کرتے تھے۔” اَهٰۤؤُلَآءِ الَّذِيْنَ اَقْسَمْتُمْ لَا يَنَالُهُمُ اللّٰهُ بِرَحْمَةٍ ؕ۔ “کیا یہ وہی لوگ ہیں جن کے بارے میں تم قسمیں کھاتے تھے کہ اللہ انہیں اپنی رحمت میں سے کچھ نہ دے گا؟” مفسرین کہتے ہیں کہ یہ اشارہ ان کمزور اور نادار اہلِ ایمان کی طرف ہے جنہیں دنیا میں حقیر سمجھا گیا؛ مکہ کے سرداروں کے ذکر میں روایات حضرت بلال، حضرت صہیب، حضرت خباب اور حضرت سلمان رضی اللہ عنہم جیسے اصحاب کے نام لیتی ہیں۔ جمع کا لفظ مال کے ڈھیر پر بھی بولا جاتا ہے اور جتھے کی کثرت پر بھی، اور اس دن دونوں کچھ نہ دے سکے۔ اہلِ معرفت کہتے ہیں کہ پچھلے رکوع کا اونٹ یہیں پہچانا جاتا ہے: جمع کیا ہوا مال، جمع کیا ہوا جتھا، اور اوپر سے تکبر کا کوہان؛ یہی بوجھ تھا جو سوئی کے ناکے سے نہ گزر سکا۔ اور ان کی قسم دیکھیے: انہوں نے رحمت پر حد باندھنے کی قسم کھائی تھی، گویا رحمت کا دروازہ ان کی مٹھی میں ہو۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں رحمت حق کی وہ صفت ہے جس نے ہر شے کو وجود بخشا؛ جو اسے دوسروں کے لیے تنگ کرنا چاہے وہ دراصل اپنے ہی دل میں رحمت کی جگہ تنگ کرتا ہے۔ جنہیں وہ رحمت سے محروم سمجھتے تھے وہی رحمت کے گھر میں تھے، اور فیصلہ قسموں سے نہیں، دلوں سے ہوا۔

اور پھر وہ کلمہ آتا ہے جس پر اس پورے منظر کا پردہ گرتا ہے: اُدْخُلُوا الْجَنَّةَ لَا خَوْفٌ عَلَيْكُمْ وَلَاۤ اَنْتُمْ تَحْزَنُوْنَ۔ “جنت میں داخل ہو جاؤ، تم پر نہ کوئی خوف ہے اور نہ تم غمگین ہو گے۔” یہ خطاب کس سے ہے؟ اس میں مفسرین کے کئی اقوال ہیں۔ ایک یہ کہ یہ انہی کمزور اہلِ ایمان کا ذکر ہے جن کے بارے میں قسمیں کھائی گئی تھیں: جنہیں تم رحمت سے محروم کہتے تھے ان سے تو کہہ دیا گیا کہ جنت میں داخل ہو جاؤ۔ دوسرا یہ کہ یہ اللہ تعالیٰ کی طرف سے خود اعراف والوں کو خطاب ہے: انتظار ختم ہوا، اب داخل ہو جاؤ۔ سلف سے اس مفہوم کے اقوال منقول ہیں کہ اعراف والوں کا آخری انجام جنت ہے، اور اکثر مفسرین کا رجحان بھی یہی ہے۔ دونوں صورتوں میں یہ جملہ رحمت کا آخری لفظ ہے۔ خوف اور غم کو الگ الگ ذکر کیا گیا: خوف آنے والی بات کا ہوتا ہے اور غم گزری ہوئی بات کا۔ جنت والے سے دونوں اٹھا لیے گئے؛ نہ آگے کا اندیشہ رہا، نہ پیچھے کا پچھتاوا۔ اعراف والے اپنی برائیوں کی وجہ سے رکے تھے؛ ان کے حق میں غم کے اٹھ جانے کا مطلب یہ ہے کہ وہ گزری کوتاہیاں بھی اب دل پر بوجھ نہیں رہیں، گویا ان کا حساب ہی دھل گیا۔ اہلِ ذوق کہتے ہیں کہ یہ کرم کی انتہا ہے کہ جن کے ترازو کے دونوں پلڑے برابر تھے، ان کے فیصلے میں رحمت کا پلڑا جھکا دیا گیا۔ یہ ظاہر ہے، اور ظاہر ہی اصل ہے؛ اشاری بات اسی بنیاد پر کھڑی ہوتی ہے، اسے ہٹا کر نہیں۔

شیخِ اکبر کے مکتب میں اعراف کو برزخ کی صورت میں پڑھا گیا ہے، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ برزخ وہ حد ہے جو دو چیزوں کے بیچ ہو اور جس کا ایک رخ ہر ایک کی طرف ہو؛ وہ دونوں کو دیکھتا ہے مگر کسی ایک میں جذب نہیں ہوتا۔ اعراف والوں کا حال یہی ہے: ایک طرف جنت پر نظر کر کے سلام کہتے ہیں، دوسری طرف دوزخ کی جانب نظر پھیری جاتی ہے تو پناہ مانگتے ہیں۔ اسی لیے انہیں رجال کہا گیا اور ان کی پہچان کا وصف بیان ہوا: ان کا علم دونوں طرف کا ہے۔ اہلِ معرفت کہتے ہیں کہ اس مقام کی تربیت انتظار ہے، اور انتظار محرومی نہیں۔ جو برابر کے ترازو پر رکا، وہ رک کر یہ دیکھتا ہے کہ نہ اس کی نیکیاں اسے جنت تک لے جانے کو کافی تھیں اور نہ اس کی برائیاں اتنی بھاری کہ دوزخ لازم ہو؛ فیصلہ صرف رحمت کر سکتی تھی۔ اسی دید سے وہ فقر پیدا ہوتا ہے جو داخلے کی کنجی ہے، اور اسی لیے ان کی امید کو طمع کہا گیا: اپنی پونجی سے مایوس اور رب کے فضل کا طالب۔ اس مکتب میں سبقتِ رحمت کا اصول یہاں ایک ٹھوس منظر میں دکھائی دیتا ہے: جہاں عدل کا پلڑا برابر ہو وہاں رحمت آگے بڑھتی ہے، کیونکہ اس کی رحمت اس کے غضب سے آگے ہے۔ اعراف کی کہانی کا آخری لفظ داخلے کا حکم ہے، واپسی کا نہیں؛ اور سلام سے شروع ہونے والا انتظار سلامتی ہی پر ختم ہوتا ہے۔

اہلِ ذوق اس منظر کو سالک کے سفر پر بھی پھیلاتے ہیں۔ دنیا میں ہر طالب کی زندگی میں ایک اعراف آتا ہے: وہ مرحلہ جب وہ غفلت کی وادی سے نکل آیا مگر ابھی قرب کے باغ میں داخل نہیں ہوا؛ پیچھے دیکھتا ہے تو پرانی غفلت سے ڈرتا ہے، آگے دیکھتا ہے تو اپنی کمی سے شرماتا ہے۔ یہ قبض اور بسط کا، خوف اور رجا کا درمیانی مقام ہے۔ اس مکتب میں کہا جاتا ہے کہ اس مرحلے کی دو آفتیں ہیں: ایک یہ کہ آدمی پیچھے دیکھتے دیکھتے مایوس ہو جائے، دوسری یہ کہ آگے دیکھ کر اپنے آپ کو پہنچا ہوا سمجھ لے۔ اعراف والوں کا ادب دونوں سے بچاتا ہے: وہ جنت والوں کو سلام کہتے ہیں مگر خود کو ان میں شمار نہیں کرتے، اور دوزخ والوں سے پناہ مانگتے ہیں مگر خود کو ان سے بے خوف نہیں سمجھتے۔ اہلِ تصوف صوفی کو ابن الوقت کہتے ہیں، یعنی وہ جو گزرے ہوئے کے غم اور آنے والے کے خوف میں نہیں جیتا بلکہ اسی لمحے میں حق کے سامنے حاضر رہتا ہے؛ آیت کا یہ وعدہ کہ نہ خوف ہے نہ غم، اسی حال کا کامل اور دائمی ظہور ہے۔ یہ اشارہ ہے، ظاہر کا بدل نہیں: ظاہر یہی ہے کہ اعراف آخرت کا ایک حقیقی مقام ہے، اور کسی کو اس امید پر عمل میں ڈھیل نہیں دینی چاہیے کہ پلڑا برابر رہ جائے گا۔ عقل مند وہ ہے جو اس انتظار تک پہنچنے سے پہلے نیکی کے پلڑے کو بھاری کر لے، اور پھر بھی بھروسا اپنے پلڑے پر نہیں، رب کی رحمت پر رکھے۔

اب منظر دوزخ کی طرف سے آنے والی ایک فریاد کا ہے: وَنَادٰۤى اَصْحٰبُ النَّارِ اَصْحٰبَ الْجَنَّةِ اَنْ اَفِيْضُوْا عَلَيْنَا مِنَ الْمَآءِ اَوْ مِمَّا رَزَقَكُمُ اللّٰهُ ؕ قَالُوْۤا اِنَّ اللّٰهَ حَرَّمَهُمَا عَلَى الْكٰفِرِيْنَ۔ “اور دوزخ والے جنت والوں کو پکاریں گے کہ ہم پر کچھ پانی بہا دو، یا اس میں سے کچھ جو اللہ نے تمہیں رزق دیا ہے۔ وہ کہیں گے: اللہ نے یہ دونوں چیزیں کافروں پر روک دی ہیں،” الَّذِيْنَ اتَّخَذُوْا دِيْنَهُمْ لَهْوًا وَّلَعِبًا وَّغَرَّتْهُمُ الْحَيٰوةُ الدُّنْيَا ۚ۔ “جنہوں نے اپنے دین کو کھیل تماشا بنا لیا تھا اور جنہیں دنیا کی زندگی نے دھوکے میں ڈال دیا تھا۔” پیاس کی شدت دیکھیے کہ پہلی مانگ پانی ہے، اور "اَفِيْضُوْا" میں بہا دینے کا مفہوم ہے، گویا ایک چھینٹا بھی غنیمت ہو۔ یہاں تحریم شرعی حکم کے معنی میں نہیں، روک دینے اور محروم رکھنے کے معنی میں ہے۔ جواب میں کوئی طعنہ نہیں، صرف ایک خبر ہے، اور پھر وجہ بتا دی گئی: دین کو کھیل بنا لینا اور دنیا کی زندگی کا دھوکا کھانا۔ اہلِ تاویل کہتے ہیں کہ پانی حیات اور علم کی تمثیل ہے؛ جس نے دنیا میں وحی کے پانی کو کھیل سمجھا، اس کے پاس آج اپنی پیاس کے سوا کچھ نہیں، اور جنت والوں کا پانی اس تک نہیں پہنچ سکتا، کیونکہ بیچ میں وہ حجاب ہے جس کا ذکر اعراف کے ساتھ ہوا۔ شیخِ اکبر کے نزدیک ہر جان وہی پاتی ہے جو اس کے ظرف میں سما سکے، اور آخرت میں ظرف وہ ہے جو بندے نے دنیا میں اپنے رخ سے بنایا؛ ایک کا رزق دوسرے کے ظرف میں نہیں اترتا۔ لہو وہ ہے جو دل کو اصل کام سے پھیر دے، اور لعب وہ جس میں کوئی سنجیدہ غایت نہ ہو؛ جس نے دین کو یہ دونوں بنایا، اس نے اپنی زندگی کے سب سے سنجیدہ معاملے کو تفریح سمجھا۔

پھر وہ آیت جس پر قلب ٹھٹک جاتا ہے: فَالْيَوْمَ نَنْسٰىهُمْ كَمَا نَسُوْا لِقَآءَ يَوْمِهِمْ هٰذَا ۙ وَمَا كَانُوْا بِاٰيٰتِنَا يَجْحَدُوْنَ۔ “تو آج ہم انہیں بھلا دیں گے جیسے وہ اپنے اس دن کی ملاقات کو بھولے رہے، اور جیسے وہ ہماری آیتوں کا جان بوجھ کر انکار کرتے رہے۔” اللہ تعالیٰ بھول سے پاک ہے؛ قرآن خود کہتا ہے: لَا يَضِلُّ رَبِّيْ وَلَا يَنْسٰى۔ اسی لیے حضرت عبد اللہ بن عباس رضی اللہ عنہما اور جمہور مفسرین نے یہاں بھلانے کے معنی چھوڑ دینے کے کیے ہیں: جیسے انہوں نے اس دن کی تیاری چھوڑ دی، ویسے ہی آج انہیں رحمت سے محروم چھوڑ دیا جائے گا۔ عربی بلاغت میں جزا کو عمل ہی کے لفظ سے بیان کرنا معروف ہے، تاکہ معلوم ہو کہ یہ اسی عمل کا بدلہ ہے۔ یہ آیت ہدایت اور گمراہی کے اصول کی شرح بھی ہے: پہل ان کی تھی۔ انہوں نے بھلایا تو انہیں چھوڑا گیا؛ حکم ان کے اپنے رخ کے مطابق ہوا، کسی جبر سے نہیں۔ سورۂ طٰہٰ کے صفحے پر یہی ترتیب گزری: كَذٰلِكَ اَتَتْكَ اٰيٰتُنَا فَنَسِيْتَهَا ۚ وَكَذٰلِكَ الْيَوْمَ تُنْسٰى، اور سورۂ الحشر میں اس کی جڑ یوں دکھائی گئی: نَسُوا اللّٰهَ فَاَنْسٰىهُمْ اَنْفُسَهُمْ۔ اہلِ معرفت کہتے ہیں کہ ذکر حضور ہے اور نسیان غیبت؛ جو دنیا میں حق کی یاد سے غائب رہا، وہ آخرت میں عنایت کی نظر سے اوجھل رہا، اور یہ اوجھل ہونا اس کی اپنی غیبت کا عکس ہے، کوئی نیا ظلم نہیں۔ شیخِ اکبر کے نزدیک بندہ جس حال میں دنیا کی زندگی گزارتا ہے، آخرت میں اسی کا حکم اس پر ظاہر ہوتا ہے؛ جس نے ملاقات کو بھلایا، اس کے لیے ملاقات کا دن ملاقات کا دن نہ رہا۔

اب حجت تمام کرنے والی بات: وَلَقَدْ جِئْنٰهُمْ بِكِتٰبٍ فَصَّلْنٰهُ عَلٰى عِلْمٍ هُدًى وَّرَحْمَةً لِّقَوْمٍ يُّؤْمِنُوْنَ۔ “اور ہم ان کے پاس ایک ایسی کتاب لے آئے جسے ہم نے علم کی بنیاد پر کھول کھول کر بیان کیا، ایمان لانے والوں کے لیے ہدایت اور رحمت۔” یہ جملہ پچھلی ساری گفتگو پر مہر لگاتا ہے: انہوں نے بے خبری میں نہیں بھلایا، کتاب آ چکی تھی، اور مبہم نہیں بلکہ مفصل۔ "عَلٰى عِلْمٍ" کے دو معنی مفسرین نے ذکر کیے ہیں: ایک یہ کہ ہم نے اسے اپنے کامل علم کی بنیاد پر مفصل کیا، دوسرا یہ کہ ایسی تفصیل جو علم عطا کرتی ہے۔ شیخِ اکبر کے مکتب میں قرآن جمع بھی ہے اور تفصیل بھی؛ وہ ایک ہی کلام ہے جو حقیقت کو اجمال سے تفصیل میں لاتا ہے، اور یہ تفصیل اس علم پر قائم ہے جو ہر دل کی استعداد اور ہر زمانے کے حال کو جانتا ہے۔ اسی لیے ایک ہی آیت ایک دل کو تسلی دیتی ہے، دوسرے کو بیدار کرتی ہے اور تیسرے کو حیرت میں ڈال دیتی ہے۔ پھر فرمایا کہ وہ ہدایت اور رحمت ہے، مگر ایمان لانے والوں کے لیے۔ کتاب سب کے لیے اتری اور دعوت سب تک پہنچی، لیکن ہدایت اور رحمت کا پھل وہی پاتا ہے جو قبول کا ظرف کھولے؛ جیسے بارش ایک ہے اور ہر زمین اپنی قابلیت کے مطابق اگاتی ہے، جس کی تمثیل اسی سورت کے اگلے رکوع کے آخر میں آنے والی ہے۔ یہاں بھی اصول وہی ہے: عطا عام ہے، قبولیت بندے کے رخ پر ہے۔

اور پھر ایک سوال جو منکروں کے انتظار کی حقیقت کھول دیتا ہے: هَلْ يَنْظُرُوْنَ اِلَّا تَاْوِيْلَهٗ ؕ يَوْمَ يَاْتِيْ تَاْوِيْلُهٗ يَقُوْلُ الَّذِيْنَ نَسُوْهُ مِنْ قَبْلُ قَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ۚ۔ “کیا یہ لوگ اب بس اس کے انجام کے منتظر ہیں؟ جس دن اس کا انجام آ پہنچے گا تو وہی لوگ جو پہلے اسے بھولے ہوئے تھے کہیں گے: یقیناً ہمارے رب کے رسول حق لے کر آئے تھے۔” یہاں تاویل کے معنی اس حقیقت کے ہیں جس کی طرف بات لوٹتی ہے، یعنی وعدے اور وعید کا واقع ہو جانا؛ لفظ کا مادہ لوٹنے کا مفہوم رکھتا ہے۔ ہر خبر کی ایک تاویل ہے، اور قیامت کا دن قرآن کی خبروں کی تاویل کا دن ہے۔ اب ایک عجیب ہم آہنگی دیکھیے: یہی جملہ کہ ہمارے رب کے رسول علیہم السلام حق لے کر آئے، پچھلے رکوع میں جنت والوں کی زبان پر تھا، اور یہاں حسرت زدہ منکروں کی زبان پر ہے۔ لفظ ایک ہیں، حال دو: وہاں شکر کا ترانہ، یہاں پچھتاوے کی فریاد۔ اہلِ معرفت کہتے ہیں کہ فرق اس میں نہیں کہ کس نے کیا مانا، فرق اس میں ہے کہ کب مانا۔ جس نے تاویل کے آنے سے پہلے مانا، اس کا ماننا ایمان بالغیب تھا اور وہی کام آیا؛ جس نے تاویل کو آنکھ سے دیکھ کر مانا، اس کا ماننا مجبوری کی گواہی ہے۔ شیخِ اکبر کے مکتب میں عارف وہ ہے جس کے دل میں تاویل پہلے اتر آتی ہے: وہ وعدے کو یوں دیکھتا ہے گویا وہ پورا ہو چکا، اور یہی حدیثِ احسان کا ذوق ہے کہ اللہ کی عبادت یوں کرو گویا اسے دیکھ رہے ہو۔ جو آج کتاب کی تاویل کو دل سے پا لے، کل کی تاویل اس کے لیے اجنبی نہ ہوگی۔

آخر میں ان کی دو تمنائیں اور ان کا انجام: فَهَلْ لَّنَا مِنْ شُفَعَآءَ فَيَشْفَعُوْا لَنَاۤ اَوْ نُرَدُّ فَنَعْمَلَ غَيْرَ الَّذِيْ كُنَّا نَعْمَلُ ؕ قَدْ خَسِرُوْۤا اَنْفُسَهُمْ وَضَلَّ عَنْهُمْ مَّا كَانُوْا يَفْتَرُوْنَ۔ “تو کیا ہمارے کوئی سفارشی ہیں جو ہماری سفارش کریں، یا ہمیں واپس بھیج دیا جائے تو ہم اس کے سوا عمل کریں جو کیا کرتے تھے؟ انہوں نے اپنے آپ کو گھاٹے میں ڈال دیا، اور جو کچھ وہ گھڑا کرتے تھے سب ان سے گم ہو گیا۔” دو ہی تمنائیں رہ گئیں: کوئی سفارشی، یا دنیا میں واپسی؛ اور دونوں کا وقت گزر چکا۔ قرآن نے ان کے نقصان کو مال یا مرتبے کا نقصان نہیں کہا، "خَسِرُوْۤا اَنْفُسَهُمْ" فرمایا: انہوں نے اپنی ذات ہی ہار دی۔ اہلِ معرفت یہاں ایک نازک فرق دکھاتے ہیں۔ عارف بھی اپنے آپ کو کھوتا ہے، جسے فنا کہتے ہیں، مگر وہ اپنی انانیت کو حق کی یاد میں کھوتا ہے اور اسی میں بقا پاتا ہے؛ یہ گھاٹا نہیں، سب سے بڑا نفع ہے۔ یہ لوگ اپنی ذات کو حق سے غافل ہو کر ہارتے ہیں، تو ہاتھ کچھ نہیں آتا۔ ایک کا کھونا پانا ہے، دوسرے کا کھونا خالی ہاتھ رہ جانا۔ پھر فرمایا کہ جو کچھ وہ گھڑتے تھے گم ہو گیا: جھوٹے سہارے، خود ساختہ معبود، دل کی بنائی ہوئی امیدیں؛ ان کا حقیقت میں کوئی وجود تھا ہی نہیں، اس لیے حقیقت کے دن وہ غائب ہو گئے۔ سورۂ النور کے صفحے پر سراب کی تمثیل اسی کی شرح ہے: كَسَرَابٍ بِقِيْعَةٍ يَّحْسَبُهُ الظَّمْاٰنُ مَآءً؛ پیاسا پانی سمجھ کر بڑھا اور پہنچا تو کچھ نہ تھا۔ دوزخ والوں نے بھی پانی ہی مانگا تھا۔ شیخِ اکبر کے نزدیک جو حق کے ساتھ قائم نہیں وہ آخرکار محض نمود ثابت ہوتا ہے، اور جو حق کے ساتھ قائم ہے وہی باقی ہے۔
