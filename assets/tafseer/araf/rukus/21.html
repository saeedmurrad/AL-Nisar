پچھلے رکوع کے آخر میں ان لوگوں کا ذکر تھا جنہوں نے کہی ہوئی بات کو بدل دیا؛ اب اسی سلسلے کی ایک اور گواہی سامنے لائی جاتی ہے، اور خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے: وَسْئَلْهُمْ عَنِ الْقَرْيَةِ الَّتِيْ كَانَتْ حَاضِرَةَ الْبَحْرِ ۘ اِذْ يَعْدُوْنَ فِي السَّبْتِ اِذْ تَاْتِيْهِمْ حِيْتَانُهُمْ يَوْمَ سَبْتِهِمْ شُرَّعًا وَّيَوْمَ لَا يَسْبِتُوْنَ ۙ لَا تَاْتِيْهِمْ ۚ كَذٰلِكَ ۚ نَبْلُوْهُمْ بِمَا كَانُوْا يَفْسُقُوْنَ۔ “اور ان سے اس بستی کا حال پوچھیے جو سمندر کے کنارے آباد تھی، جب وہ سبت کے معاملے میں حد سے نکل جاتے تھے، جب ان کی مچھلیاں ان کے سبت کے دن سر اٹھائے سامنے آتیں اور جس دن سبت نہ ہوتا وہ نہ آتیں؛ اسی طرح ہم انہیں آزماتے تھے، اس لیے کہ وہ نافرمانی کرتے تھے۔” مفسرین کہتے ہیں کہ یہ سوال معلومات لینے کے لیے نہیں، یاد دلانے کے لیے تھا: جو واقعہ ان کے اپنے علم میں محفوظ تھا، وہی ایک امی نبی صلی اللہ علیہ وسلم کی زبان سے سنا کر انہیں ان کا اپنا آئینہ دکھایا گیا۔ بعض مفسرین نے اس بستی کا نام بھی لیا ہے، مگر قرآن نے نام نہیں لیا، اور ہمارے لیے قرآن کا اجمال کافی ہے؛ نام اور تفصیل کی کرید سے زیادہ ضروری یہ ہے کہ بات کی روح پکڑی جائے۔ اور روح آیت کے آخری ٹکڑے میں ہے: كَذٰلِكَ نَبْلُوْهُمْ۔ آزمائش ان کی پہلے سے جاری نافرمانی کے مطابق آئی۔ یہی وہ اصول ہے جسے شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں سرِّ قدر کہا جاتا ہے: حکم محکوم کے مطابق ہوتا ہے۔ حق تعالیٰ نے کسی کو زبردستی نافرمان نہیں بنایا؛ جو رخ وہ پہلے اختیار کر چکے تھے، آزمائش نے اسی رخ کو ظاہر کر دیا۔ امتحان دل میں کوئی نئی چیز نہیں ڈالتا، جو اندر پہلے سے موجود ہو اسے باہر لے آتا ہے۔

اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ واقعہ اپنی جگہ تاریخ ہے، اور اہلِ معرفت اسی میں اپنے باطن کا نقشہ بھی دیکھتے ہیں۔ ان کے نزدیک سمندر کنارے کی بستی انسان کا نفس ہے جو دنیا کے سمندر کے کنارے آباد ہے، اور اس کی لہریں ہر وقت دروازے پر دستک دیتی ہیں۔ سبت وہ حد ہے جو حق تعالیٰ نے بندے کے لیے کھینچی، وہ دن جس میں ہاتھ روک لینا ہی عبادت ہے۔ اور حِيْتَانُهُمْ وہ خواہشیں ہیں جو عام دنوں میں گہرائی میں چھپی رہتی ہیں، مگر جیسے ہی کوئی حد لگتی ہے، شُرَّعًا یعنی سر اٹھائے سطح پر آ جاتی ہیں۔ یہ نفس کی پرانی عادت ہے کہ جو چیز منع ہو، وہی سب سے زیادہ دلکش دکھائی دیتی ہے۔ شیخِ اکبر کے مکتب میں حد کی حکمت یہ بیان ہوتی ہے کہ حد بندے کو اس کی عبدیت یاد دلاتی ہے؛ جہاں کوئی حد نہ ہو وہاں بندہ اپنے آپ کو مالک سمجھنے لگتا ہے۔ اسی لیے آزمائش ٹھیک حد کے دن آئی: جس دن مچھلیاں آتی ہی نہ تھیں اس دن ہاتھ روکنا کوئی کمال نہ تھا، کمال یہ تھا کہ جب سامنے آئیں تب ہاتھ رکے۔ یہاں یہ بات صاف رہے کہ یہ قصہ کسی قوم پر انگلی اٹھانے کے لیے نہیں سنایا گیا، نہ آج کے کسی گروہ یا کسی مذہب کے ماننے والوں پر اسے چسپاں کرنا روا ہے؛ یہ ایک گزری ہوئی بستی کا حال ہے، اور ہمارے لیے آئینہ ہے۔ ہر بندے کی اپنی ایک بستی ہے، اپنا ایک سبت ہے، اور اپنی مچھلیاں ہیں جو ٹھیک اسی وقت سطح پر آتی ہیں جب اسے رکنا ہوتا ہے۔

اب بستی تین حصوں میں بٹ جاتی ہے، اور یہ تقسیم قرآن نے خود بیان فرمائی: وَاِذْ قَالَتْ اُمَّةٌ مِّنْهُمْ لِمَ تَعِظُوْنَ قَوْمَا ۨاللّٰهُ مُهْلِكُهُمْ اَوْ مُعَذِّبُهُمْ عَذَابًا شَدِيْدًا ۗ قَالُوْا مَعْذِرَةً اِلٰى رَبِّكُمْ وَلَعَلَّهُمْ يَتَّقُوْنَ۔ “اور جب ان میں سے ایک گروہ نے کہا: تم ایسے لوگوں کو کیوں نصیحت کرتے ہو جنہیں اللہ ہلاک کرنے والا ہے یا سخت عذاب دینے والا ہے؟ انہوں نے کہا: تمہارے رب کے حضور عذر پیش کرنے کے لیے، اور اس امید پر کہ شاید وہ بچ جائیں۔” ایک گروہ وہ تھا جو حد سے نکلا؛ دوسرا وہ جو انہیں روکتا رہا؛ اور تیسرا وہ جو خود تو حد کے اندر رہا مگر نصیحت سے تھک کر خاموش ہو گیا اور نصیحت کرنے والوں سے کہنے لگا کہ اب اس کا فائدہ کیا ہے۔ یہ تیسرا گروہ بدکار نہ تھا، مگر اس کی بات میں ایک باریک مایوسی تھی: اس نے اللہ کے فیصلے کا اندازہ اپنی طرف سے لگا لیا اور اسی اندازے کو نصیحت چھوڑ دینے کا جواز بنا لیا۔ اہلِ معرفت کہتے ہیں کہ ایسی خاموشی بظاہر دانائی کا لباس پہنے ہوتی ہے، مگر اس کے اندر دو چیزیں چھپی ہوتی ہیں: دوسروں کی اصلاح سے نا امیدی، اور اپنی ذمہ داری سے سبکدوش ہو جانے کی خواہش۔ نصیحت کرنے والوں کا جواب دو حصوں میں تھا اور دونوں میں دعوت کا پورا ادب سمٹ آیا: پہلا حصہ اپنے رب کی طرف دیکھتا ہے اور دوسرا اپنے بھائیوں کی طرف۔ یعنی ہم یہ کام نتیجے کی ضمانت پر نہیں، اپنے فرض کی ادائیگی کے لیے کرتے ہیں؛ اور اس کے باوجود امید ہمارا ساتھ نہیں چھوڑتی۔

مَعْذِرَةً کا لفظ غور طلب ہے۔ معذرت یہاں کسی غلطی کی معافی نہیں، بلکہ وہ عذر ہے جو بندہ کل اپنے رب کے سامنے پیش کر سکے: کہ میں نے دیکھا، اور جتنا میرے بس میں تھا، کہہ دیا۔ نصیحت کرنے والے کی نگاہ لوگوں کے ردِّ عمل پر نہیں، اپنے رب کی رضا پر ہوتی ہے؛ اس لیے لوگوں کا انکار اسے مایوس نہیں کرتا، کیونکہ اس کا حساب نتیجے پر نہیں، اخلاص پر ہے۔ صحیح مسلم میں حضرت ابو سعید خدری رضی اللہ عنہ سے حضورِ اکرم صلی اللہ علیہ وسلم کا جو ارشاد منقول ہے، اس کا مفہوم یہ ہے کہ تم میں سے جو کوئی برائی دیکھے اسے ہاتھ سے بدل دے، اس کی طاقت نہ ہو تو زبان سے، اور اس کی بھی طاقت نہ ہو تو دل سے، اور یہ ایمان کا سب سے کمزور درجہ ہے۔ دیکھیے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے کسی درجے میں بھی بے پروائی کی گنجائش نہیں چھوڑی؛ آخری درجہ بھی دل کا انکار ہے، یعنی برائی کو برائی جاننا اور اس سے مانوس نہ ہونا۔ سورۂ مائدہ میں ایک گزری ہوئی جماعت کا یہی عیب بیان ہوا: كَانُوْا لَا يَتَنَاهَوْنَ عَنْ مُّنْكَرٍ فَعَلُوْهُ ؕ لَبِئْسَ مَا كَانُوْا يَفْعَلُوْنَ۔ “وہ ایک دوسرے کو اس برائی سے نہیں روکتے تھے جو وہ کرتے تھے؛ بہت برا تھا جو وہ کرتے تھے۔” گویا جماعتوں کا زوال صرف گناہ سے نہیں آتا، گناہ پر خاموش ہو جانے سے آتا ہے۔ شیخِ اکبر کے نزدیک بھلائی کا حکم دینے والا اسمِ ہادی کا مظہر بنتا ہے؛ ہدایت دینے والا حق ہے، بندہ صرف وہ زبان ہے جس سے یہ اسم ظاہر ہوتا ہے۔ اسی لیے نتیجہ بھی اسی کے سپرد ہے، اور داعی کے ذمے صرف یہ ہے کہ وہ زبان صاف رہے۔

جواب کا دوسرا حصہ وَلَعَلَّهُمْ يَتَّقُوْنَ ہے، اور یہ امید کا وہ دروازہ ہے جسے نصیحت کرنے والا کبھی بند نہیں کرتا۔ خاموش گروہ نے لوگوں کو ان کے گزرے ہوئے کل پر پرکھا تھا: یہ تو ایسے ہی ہیں، یہ نہیں بدلیں گے۔ نصیحت کرنے والوں نے انہیں ان کے آنے والے کل کی امید میں دیکھا۔ شیخِ اکبر کے مکتب میں تجددِ امثال یا خلقِ جدید کی تعلیم یہ ہے کہ کائنات ہر آن نئے سرے سے وجود پاتی ہے اور ہر لمحہ حق کا فیض تازہ اترتا ہے؛ جو شخص کل نافرمان تھا، آج وہ ایک نئے لمحے میں کھڑا ہے، اور اس لمحے میں توبہ کا دروازہ کھلا ہے۔ اس لیے کسی کے بارے میں قطعی فیصلہ کر لینا کہ یہ کبھی نہیں بدلے گا، دراصل اپنی آنکھ کو حق کے تازہ فیض سے بند کر لینا ہے۔ عارف کسی گنہگار کو حقیر نہیں جانتا، کیونکہ وہ نہیں جانتا کہ اگلی گھڑی اس دل پر کون سا اسم تجلی کرے گا۔ اور لَعَلَّ کا لفظ بھی ادب سکھاتا ہے: نصیحت کرنے والوں نے یہ نہیں کہا کہ وہ ضرور سدھر جائیں گے، کہ یہ دعویٰ ہوتا؛ نہ یہ کہا کہ وہ کبھی نہیں سدھریں گے، کہ یہ مایوسی ہوتی۔ انہوں نے امید رکھی اور انجام رب کے سپرد کیا۔ یہی داعی کا مقام ہے: کوشش اپنی، امید پوری، اور فیصلہ اس کا جو دلوں کو پھیرنے والا ہے۔ داعی کا کام یاد دلانا ہے، زبردستی منوانا نہیں؛ اور جو یاد دلانے سے تھک جائے وہ گویا یہ بھول گیا کہ اس کے اپنے دل کو بھی کسی نے بار بار یاد دلایا تھا۔

اس جواب میں ایک اور باریک نکتہ چھپا ہے: نصیحت کرنے والوں نے یہ نہیں کہا کہ ہمارے رب کے حضور، بلکہ کہا تمہارے رب کے حضور۔ وہ خاموش رہنے والوں کو بھی یاد دلا رہے تھے کہ جس رب کے سامنے ہمیں جواب دینا ہے وہ تمہارا بھی رب ہے، اور یہ ذمہ داری صرف ہماری نہیں؛ گویا نصیحت کرنے والوں نے خاموش رہنے والوں کو بھی نصیحت کر دی، اور وہ بھی ایسی نرمی سے کہ الزام کا ایک لفظ نہ کہا۔ یہیں سے نصیحت کا ادب نکلتا ہے۔ حضرت موسیٰ اور حضرت ہارون علیہما السلام کو فرعون جیسے سرکش کے پاس بھیجا گیا تو نرم بات کہنے کا حکم ملا؛ تو عام مسلمان کو اپنے بھائی سے سختی کا حق کہاں سے مل گیا؟ امام شافعی رحمۃ اللہ علیہ سے منقول ہے، جس کا مفہوم یہ ہے کہ جس نے اپنے بھائی کو تنہائی میں نصیحت کی اس نے اس کی خیر خواہی کی، اور جس نے سب کے سامنے کی اس نے اسے رسوا کیا۔ اہلِ معرفت ایک اور شرط بڑھاتے ہیں: نصیحت پہلے اپنے کانوں تک پہنچے۔ جو بھلائی کا حکم دیتے ہوئے خود کو اونچا دیکھے، وہ ایک برائی سے روکتے ہوئے دوسری برائی میں جا پڑا، یعنی خود پسندی میں۔ شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ جس عیب کو تم دوسرے میں دیکھتے ہو وہ کسی نہ کسی صورت میں تمہارے اندر بھی موجود ہے، ورنہ تمہاری آنکھ اسے اتنی جلدی نہ پہچانتی؛ اس لیے سچی نصیحت عاجزی سے نکلتی ہے، برتری سے نہیں۔

اہلِ تاویل کہتے ہیں کہ یہ تینوں گروہ ایک ہی انسان کے اندر بھی آباد ہیں۔ ایک حصہ وہ ہے جو حد کو دیکھتے ہی اسے پھلانگنے کی تدبیر سوچنے لگتا ہے؛ یہ نفسِ امارہ ہے۔ دوسرا حصہ وہ ہے جو ہر بار ٹوکتا ہے، ملامت کرتا ہے اور روکتا ہے؛ یہ نفسِ لوامہ اور ضمیر کی وہ آواز ہے جسے حق تعالیٰ نے بندے کے اندر رکھ دیا۔ اور تیسرا حصہ وہ تھکا ہوا دل ہے جو بار بار کی ہار کے بعد کہنے لگتا ہے کہ اب کوئی فائدہ نہیں، یہ عادت نہیں چھوٹے گی۔ نجات اس حصے کو ملتی ہے جو روکنا نہیں چھوڑتا، چاہے ہزار بار ہارے۔ اندر کا یہ خاموش گروہ بہت خطرناک ہے، کیونکہ وہ گناہ نہیں کرتا، صرف گناہ کے مقابلے سے دست بردار ہو جاتا ہے۔ اس مکتب میں مجاہدے کا راز یہی بتایا جاتا ہے کہ بندہ اپنے نفس سے کبھی مایوس نہ ہو، اور اپنے اندر کے ناصح کو خاموش نہ ہونے دے۔ گھر میں، خاندان میں اور اپنے دائرے میں بھی یہی ترتیب ہے: پہلے اپنے اندر کا ناصح زندہ ہو، پھر باہر کی نصیحت میں اثر آتا ہے۔ اور یہ یاد رہے کہ یہ اشاری تقسیم ظاہر کی جگہ نہیں لیتی؛ ظاہر میں وہ تین تاریخی گروہ تھے، اور باطن میں ان کا آئینہ ہمارے سینوں میں ہے۔

پھر انجام سنایا جاتا ہے: فَلَمَّا نَسُوْا مَا ذُكِّرُوْا بِهٖۤ اَنْجَيْنَا الَّذِيْنَ يَنْهَوْنَ عَنِ السُّوْٓءِ وَاَخَذْنَا الَّذِيْنَ ظَلَمُوْا بِعَذَابٍ بَئِيْسٍ بِمَا كَانُوْا يَفْسُقُوْنَ۔ “پھر جب وہ اس نصیحت کو بھلا بیٹھے جو انہیں کی گئی تھی تو ہم نے ان لوگوں کو بچا لیا جو برائی سے روکتے تھے، اور ظلم کرنے والوں کو ان کی نافرمانی کے سبب سخت عذاب میں پکڑ لیا۔” یہ بھولنا غفلت کی بے خبری نہیں تھا، کیونکہ انہیں بار بار یاد دلایا گیا تھا؛ مفسرین کہتے ہیں کہ یہاں نسیان کا مطلب جان بوجھ کر چھوڑ دینا ہے۔ یہی وہ اعراض ہے جو ہر گمراہی کی جڑ ہے: یاد دلانے والا موجود تھا، سننے والے نے منہ پھیر لیا۔ اور پکڑ کا سبب بھی آیت نے خود بتا دیا: بِمَا كَانُوْا يَفْسُقُوْنَ، یعنی ان کے اپنے عمل کی وجہ سے۔ نجات پانے والوں کا وصف دیکھیے: الَّذِيْنَ يَنْهَوْنَ، وہ جو روکتے رہتے تھے؛ نجات کو اسی وصف سے جوڑا گیا۔ رہا تیسرا گروہ، تو قرآن نے اس کے انجام پر خاموشی اختیار کی۔ مفسرین نے نقل کیا ہے کہ حضرت عبداللہ بن عباس رضی اللہ عنہما نے پہلے اس بارے میں توقف کیا، پھر ان کے شاگرد حضرت عکرمہ رحمۃ اللہ علیہ نے عرض کیا کہ وہ لوگ اس فعل کو ناپسند کرتے تھے، تو آپ نے ان کی بات قبول کی۔ ہمارے لیے ادب یہ ہے کہ جہاں قرآن خاموش ہو وہاں ہم فیصلہ نہ سنائیں؛ مگر سبق یہ ضرور لیں کہ خاموشی کوئی محفوظ پناہ گاہ نہیں، نجات کا وصف صراحت سے روکنے والوں کے لیے بیان ہوا ہے۔

اس کے بعد وہ آیت آتی ہے جس میں ادب کا تقاضا ہے کہ صرف قرآن کے الفاظ پر رکا جائے: فَلَمَّا عَتَوْا عَنْ مَّا نُهُوْا عَنْهُ قُلْنَا لَهُمْ كُوْنُوْا قِرَدَةً خٰسِئِيْنَ۔ “پھر جب وہ اس کام میں سرکشی کرتے چلے گئے جس سے انہیں روکا گیا تھا تو ہم نے ان سے کہا: ذلیل بندر بن جاؤ۔” عَتَوْا سرکشی کی وہ انتہا ہے جس میں بار بار کی تنبیہ کے بعد بھی بندہ اپنی ضد پر قائم رہے؛ سزا پہلے قدم پر نہیں، اس انتہا پر آئی۔ جمہور مفسرین کے نزدیک یہ حقیقی مسخ تھا کہ ان کی صورتیں بدل دی گئیں، اور امام مجاہد رحمۃ اللہ علیہ سے یہ قول بھی منقول ہے کہ ان کے دل مسخ کیے گئے؛ جمہور کا قول پہلا ہے۔ اس سے آگے کی کوئی تفصیل قرآن نے نہیں دی، اور ہم بھی کہانیوں کا اضافہ نہیں کرتے۔ ایک بات البتہ صحیح حدیث سے ثابت ہے اور اسے ہر پڑھنے والے کو یاد رہنا چاہیے: صحیح مسلم میں حضرت عبداللہ بن مسعود رضی اللہ عنہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ اللہ نے کسی مسخ کی گئی قوم کی نسل نہیں چلائی، اور بندر اور سور اس سے پہلے بھی موجود تھے۔ اس لیے آج کے کسی انسان، کسی قوم یا کسی مذہب کے ماننے والوں کو اس لفظ سے یاد کرنا نہ صرف بد زبانی ہے بلکہ قرآن و سنت کی تعلیم کے خلاف ہے؛ قرآن نے تو ایک دوسرے کو برے القاب سے پکارنے سے بھی منع فرمایا ہے۔

اہلِ تاویل نے اس آیت میں ایک اشارہ بھی دیکھا ہے، اور وہ ظاہر کی نفی نہیں بلکہ اس کی گہرائی ہے۔ حکم كُوْنُوْا کے لفظ سے آیا، یعنی ہو جاؤ؛ یہ وہی کلمۂ کن کا انداز ہے جس سے ہر شے وجود پاتی ہے۔ شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ حق تعالیٰ کا حکم شے کی اپنی حقیقت کے مطابق ہوتا ہے، اور یہاں بھی صورت اسی حال کے تابع ہوئی جو انہوں نے اپنے باطن میں اختیار کر رکھا تھا۔ قرآن نے ان کی سرکشی کی تفصیل نہیں بتائی، اس لیے ہم واقعے پر کوئی کہانی نہیں جوڑتے؛ اہلِ ذوق اتنا کہتے ہیں کہ جو باطن حق کی حد کا احترام کھو دے، اس کا ظاہر بھی آخرکار اسی باطن کی گواہی دینے لگتا ہے۔ یہ سرِّ قدر کا ایک ہولناک منظر ہے کہ حکم محکوم کے مطابق ہوتا ہے۔ اور آئینہ یہاں بھی ہمارے سامنے ہے: ہر وہ حیلہ جس میں شریعت کا ظاہر بچا لیا جائے اور اس کی روح ذبح کر دی جائے، اسی بستی کی راہ ہے۔ جو شخص نماز کی صورت ادا کرے مگر دل کو حاضر نہ کرے، جو معاملے میں قانون کا لفظ پکڑے اور انصاف کا مقصد چھوڑ دے، وہ اپنے باطن کو اسی سمت لے جا رہا ہے۔ خٰسِئِيْنَ کا لفظ بتاتا ہے کہ جس نے حق کی حد کو حقیر جانا، وہ خود حقیر ہو گیا؛ عزت حد کے احترام میں تھی۔

پھر بات اس بستی سے آگے بڑھ کر تاریخ کے ایک قانون تک جاتی ہے: وَاِذْ تَاَذَّنَ رَبُّكَ لَيَبْعَثَنَّ عَلَيْهِمْ اِلٰى يَوْمِ الْقِيٰمَةِ مَنْ يَّسُوْمُهُمْ سُوْٓءَ الْعَذَابِ ۗ اِنَّ رَبَّكَ لَسَرِيْعُ الْعِقَابِ ۚ وَاِنَّهٗ لَغَفُوْرٌ رَّحِيْمٌ۔ “اور جب آپ کے رب نے اعلان فرما دیا کہ وہ قیامت کے دن تک ان پر ایسے لوگ ضرور مسلط کرتا رہے گا جو انہیں بری تکلیف دیں؛ بے شک آپ کا رب جلد سزا دینے والا ہے، اور بے شک وہ بڑا بخشنے والا، نہایت مہربان ہے۔” مفسرین کے نزدیک یہ اعلان ان لوگوں کے بارے میں ہے جو عہد توڑنے اور حد سے نکلنے پر قائم رہے؛ یعنی یہ ایک قانون ہے جو عمل سے بندھا ہے، نسل سے نہیں۔ اور یہاں ایک بات پوری وضاحت سے کہنی ضروری ہے: یہ آیت ایک خبر ہے، کسی کو یہ حق نہیں دیتی کہ وہ کسی پر ظلم کرے اور اسے اس آیت کا مصداق بنا لے؛ ظلم ہر حال میں ظلم ہے، اور تاریخ کے فیصلے حق تعالیٰ کے ہیں، ہمارے نہیں۔ پھر آیت کا اختتام دیکھیے: سزا کی جلدی کے فوراً بعد مغفرت اور رحمت کے دو نام آئے، اور تاکید کے ساتھ آئے۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا اظہار ہے: جلال کی ہر خبر کے پیچھے جمال کا دروازہ کھلا رکھا گیا، تاکہ کوئی بندہ، اور کوئی جماعت، کبھی یہ نہ سمجھے کہ لوٹنے کا راستہ بند ہو چکا ہے۔

اگلی آیت خود اس بات کی ضمانت دیتی ہے کہ کسی پوری جماعت کو ایک لاٹھی سے نہ ہانکا جائے: وَقَطَّعْنٰهُمْ فِي الْاَرْضِ اُمَمًا ۚ مِنْهُمُ الصّٰلِحُوْنَ وَمِنْهُمْ دُوْنَ ذٰلِكَ ۚ وَبَلَوْنٰهُمْ بِالْحَسَنٰتِ وَالسَّيِّاٰتِ لَعَلَّهُمْ يَرْجِعُوْنَ۔ “اور ہم نے انہیں زمین میں گروہ گروہ کر کے بکھیر دیا؛ ان میں کچھ نیک تھے اور کچھ اس سے کم تر، اور ہم نے انہیں اچھے حالات اور برے حالات دونوں سے آزمایا تاکہ وہ لوٹ آئیں۔” دیکھیے کہ قرآن نے سب کو ایک حکم میں نہیں سمیٹا، بلکہ صاف فرمایا: مِنْهُمُ الصّٰلِحُوْنَ، ان میں نیک لوگ بھی تھے۔ جب خود قرآن تعمیم سے بچتا ہے تو ہمیں اس کا کیا حق ہے کہ کسی قوم یا مذہب کے سب ماننے والوں پر ایک فیصلہ سنا دیں؟ یہی قرآن کا عدل ہے کہ وہ نیکی کو جہاں بھی ہو، پہچانتا اور سراہتا ہے۔ پھر بکھیرنا بھی صرف سزا نہ تھا؛ آیت کہتی ہے کہ اس کے ساتھ آزمائش بھی تھی، اور آزمائش کا مقصد بھی بتا دیا: لَعَلَّهُمْ يَرْجِعُوْنَ، تاکہ وہ لوٹ آئیں۔ ہر آزمائش کی غایت رجوع ہے۔ حق تعالیٰ کسی کو اس لیے نہیں آزماتا کہ اسے گرا دے، بلکہ اس لیے کہ وہ گرا ہوا ہو تو اٹھ کھڑا ہو، اور بھٹکا ہوا ہو تو گھر کی راہ پکڑ لے۔

بِالْحَسَنٰتِ وَالسَّيِّاٰتِ کے جوڑ پر اہلِ معرفت بہت ٹھہرتے ہیں۔ عام آدمی سمجھتا ہے کہ آزمائش صرف تکلیف میں ہے، مگر قرآن کہتا ہے کہ آسائش بھی آزمائش ہے، بلکہ بعض اوقات زیادہ کڑی۔ تکلیف میں بندہ رب کو یاد کرتا ہے، آسائش میں اکثر بھول جاتا ہے۔ شیخِ اکبر کے مکتب میں اسے قبض و بسط اور جلال و جمال کی زبان میں بیان کیا جاتا ہے: کبھی دل پر تنگی آتی ہے اور کبھی کشادگی، کبھی جلال کی تجلی ہوتی ہے اور کبھی جمال کی؛ اور دونوں ایک ہی دروازے کی طرف بلانے والے قاصد ہیں۔ عارف نہ کشادگی میں مغرور ہوتا ہے نہ تنگی میں مایوس، کیونکہ وہ ہر حال کو ایک پیغام سمجھتا ہے جس کا خلاصہ ایک ہی لفظ ہے: لوٹ آؤ۔ اور قَطَّعْنٰهُمْ میں بھی ایک باطنی اشارہ ہے: اہلِ ذوق کہتے ہیں کہ جو دل حق سے رخ پھیر لے وہ اندر سے بھی بکھر جاتا ہے، اس کی توجہ ہزار وادیوں میں بٹ جاتی ہے؛ اسے تفرقہ کہتے ہیں۔ اور رجوع کا مطلب ہے تفرقہ سے جمع کی طرف لوٹنا، بکھری ہوئی توجہ کو سمیٹ کر ایک رخ پر لانا۔ یہی وجہ ہے کہ اس رکوع کا ہر حصہ کسی نہ کسی طرح لوٹنے کی دعوت پر ختم ہوتا ہے۔

اب ان کے بعد آنے والی نسل کا ذکر ہوتا ہے: فَخَلَفَ مِنْۢ بَعْدِهِمْ خَلْفٌ وَّرِثُوا الْكِتٰبَ يَاْخُذُوْنَ عَرَضَ هٰذَا الْاَدْنٰى وَيَقُوْلُوْنَ سَيُغْفَرُ لَنَا ۚ وَاِنْ يَّاْتِهِمْ عَرَضٌ مِّثْلُهٗ يَاْخُذُوْهُ ۗ اَلَمْ يُؤْخَذْ عَلَيْهِمْ مِّيْثَاقُ الْكِتٰبِ اَنْ لَّا يَقُوْلُوْا عَلَى اللّٰهِ اِلَّا الْحَقَّ وَدَرَسُوْا مَا فِيْهِ ۗ وَالدَّارُ الْاٰخِرَةُ خَيْرٌ لِّلَّذِيْنَ يَتَّقُوْنَ ۗ اَفَلَا تَعْقِلُوْنَ۔ “پھر ان کے بعد ایسے ناخلف جانشین آئے جو کتاب کے وارث بنے؛ وہ اس حقیر دنیا کا سامان لیتے اور کہتے کہ ہمیں بخش دیا جائے گا، اور اگر ویسا ہی سامان پھر ان کے سامنے آئے تو اسے بھی لے لیں۔ کیا ان سے کتاب کا یہ عہد نہیں لیا گیا تھا کہ اللہ کے بارے میں حق کے سوا کچھ نہ کہیں، اور جو کچھ اس میں ہے وہ پڑھ چکے ہیں؟ اور آخرت کا گھر پرہیزگاروں کے لیے بہتر ہے؛ کیا تم سمجھتے نہیں؟” خَلْفٌ لام کے سکون کے ساتھ اہلِ لغت کے ہاں برے جانشینوں کے لیے آتا ہے۔ عَرَضَ وہ ہے جو آئے اور گزر جائے، اور الْاَدْنٰى کے دو معنی ہیں: سب سے قریب اور سب سے نیچا۔ دنیا کا نام ہی دنو، یعنی قربت سے ہے؛ وہ ہاتھ کے قریب ہے اس لیے آسانی سے پکڑی جاتی ہے، اور نیچی ہے اس لیے پکڑنے والے کو بھی نیچے کھینچ لیتی ہے۔ انہوں نے دور کی پائیدار نعمت کے بدلے قریب کی گزر جانے والی چیز چن لی۔

اس آیت کا سب سے چبھتا ہوا جملہ سَيُغْفَرُ لَنَا ہے، اور اسے دوسروں کی طرف اشارہ بنانے کے بجائے اپنے سامنے آئینہ بنا کر رکھنا چاہیے۔ یہ وہ امید ہے جو بدلنے کا ارادہ ساتھ نہیں رکھتی۔ اہلِ دل رجا اور تمنا میں فرق کرتے ہیں: رجا وہ امید ہے جو عمل کے ساتھ چلے، جیسے کسان بیج بو کر بارش کی امید رکھتا ہے؛ اور تمنا وہ ہے جو بیج بوئے بغیر فصل کا انتظار کرے۔ ترمذی کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: سمجھ دار وہ ہے جو اپنے نفس کا محاسبہ کرے اور موت کے بعد کے لیے عمل کرے، اور عاجز وہ ہے جو اپنے نفس کو خواہشوں کے پیچھے لگا دے اور پھر اللہ سے آرزوئیں باندھے۔ حضرت حسن بصری رحمۃ اللہ علیہ سے منقول ہے، جس کا مفہوم یہ ہے کہ ایمان آرزو اور ظاہری آرائش کا نام نہیں، بلکہ وہ ہے جو دل میں جم جائے اور عمل اس کی تصدیق کرے۔ اور قرآن نے یہ قاعدہ دوسری جگہ کھول کر بیان کر دیا: اِنَّ اللّٰهَ لَا يُغَيِّرُ مَا بِقَوْمٍ حَتّٰى يُغَيِّرُوْا مَا بِاَنْفُسِهِمْ۔ “اللہ کسی قوم کی حالت نہیں بدلتا جب تک وہ اپنے اندر کی حالت نہ بدلیں۔” شیخِ اکبر کے نزدیک اسمِ غفور سچا اور بہت وسیع ہے؛ مگر جو شخص گناہ کو تھامے رکھ کر اس نام کو پکارے، وہ نام کو ڈھال بنا رہا ہے، مسمیٰ کی طرف رخ نہیں کر رہا۔ مغفرت کی طرف رخ کرنا ہی توبہ ہے، اور توبہ کے معنی لوٹنا ہیں؛ جو اپنی جگہ سے ہلا ہی نہیں، وہ لوٹا کہاں؟

آیت کا اگلا حصہ اس دعوے کو اور گہرا کر دیتا ہے۔ ان سے عہد لیا گیا تھا کہ اللہ کے بارے میں حق کے سوا کچھ نہ کہیں؛ اور سَيُغْفَرُ لَنَا کہنا خود اللہ کے بارے میں ایک ایسا یقینی دعویٰ تھا جس کا انہیں کوئی علم نہ تھا۔ یعنی جس امید کو وہ عاجزی سمجھ رہے تھے، وہ دراصل اللہ پر ایک بات گھڑ لینا تھی۔ مومن مغفرت مانگتا ہے، اس کا دعویٰ نہیں کرتا۔ پھر وَدَرَسُوْا مَا فِيْهِ، یعنی وہ کتاب پڑھ چکے تھے؛ ان کی محرومی جہالت کی نہیں، ایسے علم کی تھی جو عمل میں نہ ڈھلا۔ اہلِ معرفت کہتے ہیں کہ پڑھ لینا اور ہو جانا دو الگ چیزیں ہیں؛ کتاب کا وارث وہ نہیں جس کے پاس کتاب ہو، بلکہ وہ ہے جس کے اندر کتاب اتر جائے۔ اور آیت ایک سوال پر ختم ہوتی ہے: اَفَلَا تَعْقِلُوْنَ۔ عقل کا مادہ عقال سے ہے، وہ رسی جس سے اونٹ کا پاؤں باندھا جاتا ہے تاکہ وہ بھٹک نہ جائے۔ سچی عقل وہ ہے جو نفس کے پاؤں باندھ دے؛ جو عقل قریب کے سامان پر دور کی بہتری کو قربان کر دے، وہ ہوشیاری تو ہو سکتی ہے، عقل نہیں۔ یہ سوال صرف ان سے نہیں، ہر اس دل سے ہے جو علم رکھتا ہو اور پھر بھی قریب کی طرف جھک جائے۔

اور اسی مقام پر قرآن پھر ان کا ذکر کرتا ہے جو اس عام بگاڑ سے الگ رہے: وَالَّذِيْنَ يُمَسِّكُوْنَ بِالْكِتٰبِ وَاَقَامُوا الصَّلٰوةَ ۗ اِنَّا لَا نُضِيْعُ اَجْرَ الْمُصْلِحِيْنَ۔ “اور جو لوگ کتاب کو مضبوطی سے تھامے رہتے ہیں اور نماز قائم کرتے ہیں، تو ہم اصلاح کرنے والوں کا اجر ضائع نہیں کرتے۔” یہاں پھر قرآن کا انصاف دیکھیے کہ برے جانشینوں کے ساتھ ہی ان نیک لوگوں کا ذکر کر دیا، تاکہ کوئی پوری جماعت کو ایک رنگ میں نہ دیکھے؛ مفسرین نے اہلِ کتاب کے ان لوگوں کو بھی اس میں شمار کیا ہے جنہوں نے کتاب کو تھاما اور پھر حضورِ اکرم صلی اللہ علیہ وسلم پر ایمان لائے، جیسے حضرت عبداللہ بن سلام رضی اللہ عنہ۔ يُمَسِّكُوْنَ کا صیغہ شدت اور تکرار کا ہے: ایک بار پکڑ لینا نہیں، بار بار اور مضبوطی سے تھامے رہنا، جیسے ڈوبنے والا رسی کو تھامتا ہے۔ اور کتاب کے ساتھ نماز کا ذکر ہوا؛ اہلِ معرفت کہتے ہیں کہ کتاب حق کا کلام ہے جو بندے کی طرف اترتا ہے، اور نماز بندے کی مناجات ہے جو حق کی طرف چڑھتی ہے؛ ایک اترنا اور ایک چڑھنا، اور ان دونوں کے بیچ بندے کی پوری زندگی۔ پھر ان کا نام مُصْلِحِيْنَ رکھا گیا: جو کتاب کو تھامے اور نماز قائم کرے، وہ خود بخود اصلاح کرنے والا بن جاتا ہے؛ اصلاح کا آغاز اپنے اندر سے ہوتا ہے، اور جو اندر سے درست ہو وہ باہر بھی درستی پھیلاتا ہے۔

رکوع اس منظر پر ختم ہوتا ہے جس میں کتاب کا عہد لیا گیا تھا: وَاِذْ نَتَقْنَا الْجَبَلَ فَوْقَهُمْ كَاَنَّهٗ ظُلَّةٌ وَّظَنُّوْۤا اَنَّهٗ وَاقِعٌۢ بِهِمْ ۚ خُذُوْا مَاۤ اٰتَيْنٰكُمْ بِقُوَّةٍ وَّاذْكُرُوْا مَا فِيْهِ لَعَلَّكُمْ تَتَّقُوْنَ۔ “اور جب ہم نے پہاڑ کو اکھاڑ کر ان کے اوپر یوں اٹھا دیا گویا وہ سائبان ہو، اور انہوں نے سمجھا کہ وہ ان پر گرنے والا ہے؛ جو کچھ ہم نے تمہیں دیا ہے اسے مضبوطی سے پکڑو اور جو اس میں ہے اسے یاد رکھو، تاکہ تم بچ جاؤ۔” مفسرین کہتے ہیں کہ یہ منظر عہد کی سنجیدگی دکھانے کے لیے تھا؛ مگر قبول کرنا اور نبھانا پھر بھی ان کا اپنا کام رہا، اور اسی لیے بعد میں کچھ نے نبھایا اور کچھ نے توڑا۔ بِقُوَّةٍ کا مطلب جسم کی طاقت نہیں، ارادے کی پختگی ہے۔ اہلِ معرفت کہتے ہیں کہ سروں پر اٹھا ہوا پہاڑ اس بوجھ کی تصویر ہے جو کلامِ الٰہی دل پر رکھتا ہے؛ اسی امانت کا ذکر سورۂ احزاب میں ہے: اِنَّا عَرَضْنَا الْاَمَانَةَ عَلَى السَّمٰوٰتِ وَالْاَرْضِ وَالْجِبَالِ فَاَبَيْنَ اَنْ يَّحْمِلْنَهَا وَاَشْفَقْنَ مِنْهَا وَحَمَلَهَا الْاِنْسَانُ۔ “ہم نے امانت آسمانوں، زمین اور پہاڑوں پر پیش کی تو انہوں نے اسے اٹھانے سے انکار کیا اور اس سے ڈر گئے، اور انسان نے اسے اٹھا لیا۔” جس امانت سے پہاڑ ڈر گئے، وہی انسان کے سپرد ہوئی؛ اسی لیے پہاڑ کا ان کے سروں پر آنا گویا یہ کہنا تھا کہ جو تم اٹھا رہے ہو اس کا وزن پہچانو۔ اور آخری حکم وَاذْكُرُوْا مَا فِيْهِ ہے، یعنی یاد رکھو۔ پورا رکوع بھولنے اور یاد رکھنے کی کشمکش تھا: سبت والوں نے نصیحت بھلائی، جانشینوں نے عہد بھلایا، اور یہاں حکم ہے کہ یاد رکھو۔ اگلا رکوع اسی یاد کو اس کی سب سے پہلی جڑ تک لے جاتا ہے، اس دن تک جب ہر روح سے پوچھا گیا تھا کہ کیا میں تمہارا رب نہیں ہوں۔
