پچھلا رکوع اس منظر پر ختم ہوا تھا کہ دوزخ میں داخل ہونے والی ہر جماعت اپنی پیش رو جماعت کو کوسے گی؛ اب قرآن اس کشمکش کی جڑ کو ایک ایسی تصویر میں کھولتا ہے جو ہمیشہ کے لیے ذہن پر نقش ہو جاتی ہے: اِنَّ الَّذِيْنَ كَذَّبُوْا بِاٰيٰتِنَا وَاسْتَكْبَرُوْا عَنْهَا لَا تُفَتَّحُ لَهُمْ اَبْوَابُ السَّمَآءِ وَلَا يَدْخُلُوْنَ الْجَنَّةَ حَتّٰى يَلِجَ الْجَمَلُ فِيْ سَمِّ الْخِيَاطِ ؕ وَكَذٰلِكَ نَجْزِي الْمُجْرِمِيْنَ۔ “بے شک جن لوگوں نے ہماری آیتوں کو جھٹلایا اور ان کے مقابلے میں تکبر کیا، ان کے لیے آسمان کے دروازے نہیں کھولے جائیں گے، اور وہ جنت میں داخل نہ ہوں گے یہاں تک کہ اونٹ سوئی کے ناکے میں سے گزر جائے؛ اور ہم مجرموں کو ایسا ہی بدلہ دیتے ہیں۔” مفسرین کہتے ہیں کہ آسمان کے دروازے نہ ان کے اعمال کے لیے کھلتے ہیں، نہ ان کی دعاؤں کے لیے، اور نہ موت کے وقت ان کی روحوں کے لیے۔ حضرت براء بن عازب رضی اللہ عنہ کی طویل حدیث میں، جسے امام احمد وغیرہ نے روایت کیا، اس کا مفہوم یہ ہے کہ منکر کی روح کو فرشتے آسمان کی طرف لے جاتے ہیں اور دروازہ کھلوانا چاہتے ہیں تو نہیں کھولا جاتا، اور اس موقع پر حضورِ اکرم صلی اللہ علیہ وسلم نے یہی آیت تلاوت فرمائی۔ اونٹ اور سوئی کا ناکہ عرب کے محاورے میں ناممکن کی تمثیل ہے: اتنا بھاری جسم اور اتنا تنگ راستہ۔ بعض مفسرین نے ایک قراءت کی بنا پر جمل کو جہاز باندھنے کی موٹی رسی بھی کہا ہے، اور دونوں معنوں میں تمثیل ایک ہی رہتی ہے۔ قابلِ غور یہ ہے کہ آیت نے سبب خود بیان کر دیا: جھٹلانا اور تکبر۔ دروازہ پہلے انہوں نے اپنی طرف سے بند کیا تھا؛ آسمان کا بند رہنا اسی بندش کا جواب ہے۔

اہلِ معرفت کے نزدیک اس تمثیل میں ایک لطیف اشارہ ہے، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ سوئی کے ناکے سے صرف دھاگا گزرتا ہے؛ اونٹ نہیں گزرتا، کیونکہ اس کے پاس کوہان ہے، بوجھ ہے، پھیلاؤ ہے۔ تکبر نفس کا کوہان ہے، یعنی انانیت کا وہ ابھار جو ہر دروازے پر اٹک جاتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں بندگی کی اصل یہ ہے کہ بندہ اپنے فقر کو پہچانے اور عبدیت کی اس حد پر ٹھہرے جو اس کی اصل حد ہے؛ جو اپنی حد پہچان لیتا ہے وہ دھاگے کی طرح باریک ہو جاتا ہے، اور باریک چیز کے لیے تنگ ناکہ بھی کشادہ راستہ ہے۔ اسی لیے اس مکتب میں کہا جاتا ہے کہ قرب کا دروازہ تنگ نہیں، آنے والا بھاری ہے۔ اسی سورت میں ابلیس کا پہلا قدم یہی تکبر تھا جب اس نے اپنے آپ کو حضرت آدم علیہ السلام سے بہتر سمجھا، اور یہاں وہی تکبر آسمان کے دروازوں کے سامنے دیوار بن کر کھڑا ہے۔ ایک اور نکتہ یہ ہے کہ آیت نے "لَا تُفَتَّحُ" فرمایا، یعنی دروازے موجود ہیں اور بہت ہیں، صرف ان کے لیے نہیں کھلتے؛ آسمان کسی سے بخل نہیں کرتا۔ اہلِ ذوق کہتے ہیں کہ دل کا بھی ایک آسمان ہے جس پر الہام اور معرفت کے دروازے ہیں؛ جو آیات کو جھٹلائے اور ان کے مقابلے میں بڑا بنے، اس کے دل کا آسمان بھی بند رہتا ہے، اس لیے نہیں کہ عطا روک لی گئی، بلکہ اس لیے کہ اس نے لینے والا ہاتھ خود بند کر لیا۔ اور "نَجْزِي" کا لفظ بتاتا ہے کہ یہ جزا ہے، یعنی عمل ہی کی وہ صورت جو آخرت میں ظاہر ہوتی ہے۔

اب منکروں کے ٹھکانے کی ایک جھلک: لَهُمْ مِّنْ جَهَنَّمَ مِهَادٌ وَّمِنْ فَوْقِهِمْ غَوَاشٍ ؕ وَكَذٰلِكَ نَجْزِي الظّٰلِمِيْنَ۔ “ان کے لیے جہنم ہی کا بچھونا ہوگا اور ان کے اوپر اسی کے اوڑھنے؛ اور ہم ظالموں کو ایسا ہی بدلہ دیتے ہیں۔” مہاد وہ بستر ہے جو آرام کے لیے بچھایا جائے، اور غواش وہ پردے جو اوپر سے ڈھانپ لیں۔ اہلِ تاویل کہتے ہیں کہ جس نے دنیا میں غفلت کو بستر بنایا اور خواہش کو اوڑھ لیا، آخرت میں اس کا بستر اور اوڑھنا اسی کی حقیقت بن کر ظاہر ہوا؛ نیچے سے بھی گھیراؤ اور اوپر سے بھی، کہ کوئی رخ خالی نہ رہے۔ پھر فوراً اہلِ ایمان کا ذکر آتا ہے: وَالَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ لَا نُكَلِّفُ نَفْسًا اِلَّا وُسْعَهَاۤ اُولٰۤئِكَ اَصْحٰبُ الْجَنَّةِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ۔ “اور جو لوگ ایمان لائے اور نیک عمل کیے، ہم کسی جان پر اس کی وسعت سے بڑھ کر بوجھ نہیں ڈالتے، یہی لوگ جنت والے ہیں، وہ اس میں ہمیشہ رہیں گے۔” مفسرین نے لکھا ہے کہ ایمان والوں کے ذکر اور ان کے انجام کے بیچ یہ جملہ اس لیے رکھا گیا کہ کوئی یہ نہ سمجھے کہ نیک عمل کوئی ناقابلِ رسائی بلندی ہے؛ جو مانگا گیا وہ طاقت کے اندر ہے، اس لیے جنت پہنچ میں ہے۔ یہی اصول سورۂ البقرہ کے آخر میں آیا: لَا يُكَلِّفُ اللّٰهُ نَفْسًا اِلَّا وُسْعَهَا۔ شیخِ اکبر کے نزدیک وسعت وہی استعداد ہے جو ہر جان کو اس کے عینِ ثابت کے مطابق ملی؛ حق تعالیٰ ہر جان سے اتنا ہی مانگتا ہے جتنا اس کے ظرف میں رکھا، اور ظرف بھی اسی کا عطیہ ہے۔ خطاب کی یہ رعایت خود رحمت ہے: تکلیف بندے کو توڑنے کے لیے نہیں، اس کی چھپی ہوئی وسعت کو ظاہر کرنے کے لیے ہے، جیسے بیج کو پانی اس لیے دیا جاتا ہے کہ اس کے اندر کا درخت باہر آئے۔

جنت کے دروازے پر پہلا منظر کسی محل یا باغ کا نہیں، دل کی صفائی کا ہے: وَنَزَعْنَا مَا فِيْ صُدُوْرِهِمْ مِّنْ غِلٍّ تَجْرِيْ مِنْ تَحْتِهِمُ الْاَنْهٰرُ ۚ۔ “اور ان کے سینوں میں جو کدورت ہوگی ہم اسے کھینچ نکالیں گے، ان کے نیچے نہریں بہتی ہوں گی۔” غل وہ چھپی ہوئی کدورت، رنجش اور کینہ ہے جو دل کے کسی گوشے میں بیٹھ جائے۔ صحیح بخاری میں حضرت ابو سعید خدری رضی اللہ عنہ کی روایت کا مفہوم یہ ہے کہ اہلِ ایمان جب دوزخ سے نجات پا لیں گے تو جنت اور دوزخ کے درمیان ایک پل پر روکے جائیں گے، اور دنیا میں ان کے آپس میں جو زیادتیاں ہوئی تھیں ان کا بدلہ ایک دوسرے سے دلایا جائے گا، یہاں تک کہ جب وہ پاک صاف ہو جائیں گے تو انہیں جنت میں داخلے کی اجازت ملے گی۔ مفسرین نے یہ بھی نقل کیا ہے کہ حضرت علی رضی اللہ عنہ نے فرمایا، جس کا مفہوم یہ ہے کہ مجھے امید ہے میں، حضرت عثمان، حضرت طلحہ اور حضرت زبیر رضی اللہ عنہم انہی لوگوں میں ہوں گے جن کے سینوں سے کدورت نکال دی جائے گی۔ سورۂ الحجر کے صفحے پر یہی بات آگے بڑھ کر آئی تھی: وَنَزَعْنَا مَا فِيْ صُدُوْرِهِمْ مِّنْ غِلٍّ اِخْوَانًا عَلٰى سُرُرٍ مُّتَقٰبِلِيْنَ، یعنی کدورت نکلی تو بھائی بن کر آمنے سامنے تختوں پر۔ اہلِ معرفت یہاں دیکھتے ہیں کہ صدور کا لفظ آیا ہے، قلوب کا نہیں۔ صدر دل کا بیرونی صحن ہے جہاں وسوسے اور رنجشیں آتی جاتی ہیں، اور قلب وہ اندرونی گھر ہے جو ایمان کو سنبھالے رکھتا ہے۔ مومن کی رنجش صحن تک تھی، گھر تک نہ پہنچی تھی؛ اسی لیے اسے کھینچ کر نکال دیا گیا، کہ جڑ اندر نہ تھی۔ اور کدورت نکلتے ہی نہروں کا ذکر ہے: اہلِ ذوق کہتے ہیں کہ صاف سینے ہی میں علم اور سکینت کی نہریں بہتی ہیں۔

پھر جنت والوں کی پہلی بات سنیے: وَقَالُوا الْحَمْدُ لِلّٰهِ الَّذِيْ هَدٰىنَا لِهٰذَا ۗ وَمَا كُنَّا لِنَهْتَدِيَ لَوْلَاۤ اَنْ هَدٰىنَا اللّٰهُ ۚ۔ “اور وہ کہیں گے: سب تعریف اللہ کے لیے ہے جس نے ہمیں اس تک پہنچایا، اور اگر اللہ ہمیں راہ نہ دکھاتا تو ہم کبھی راہ نہ پاتے۔” یہ اس سورت کے ان مقامات میں سے ہے جہاں ہدایت کا اصول خود قرآن کی زبان سے بیان ہوا، اس لیے اسے اصول کی حیثیت سے پڑھنا چاہیے۔ جنت والوں نے حمد میں اپنے لیے کچھ نہ رکھا۔ یہ وہی راستہ ہے جو اس سورت کی ابتدا میں حضرت آدم علیہ السلام اور حضرت حوا علیہا السلام نے دکھایا: رَبَّنَا ظَلَمْنَاۤ اَنْفُسَنَا، کہ کوتاہی اپنی طرف منسوب کی؛ اور اس کے مقابل ابلیس کا راستہ: فَبِمَاۤ اَغْوَيْتَنِيْ، کہ اپنی گمراہی کا الزام اپنے رب پر رکھا۔ جنت والوں نے بات مکمل کر دی: بھلائی اس کی عطا، کوتاہی ہماری اپنی۔ اس جملے کو اس معنی میں لینا درست نہیں کہ جو نہ پہنچا اسے روک دیا گیا۔ اسی آیت کا اگلا ٹکڑا گواہ ہے کہ رب کے رسول علیہم السلام حق لے کر آئے، یعنی دعوت سب تک پہنچی۔ اللہ کسی کو زبردستی گمراہ نہیں کرتا اور کسی ہدایت کے طالب کو گمراہ نہیں کرتا؛ جو مانگتا ہے پاتا ہے، اور جو منہ موڑتا ہے اس کا منہ موڑنا ہی اس کی گمراہی ہے۔ شیخِ اکبر کے مکتب میں سرِّ قدر یہی ہے کہ حکم محکوم کے مطابق ہوتا ہے؛ مگر عارف جب اپنے پہنچنے کو دیکھتا ہے تو اسے اپنی طلب بھی عطا دکھائی دیتی ہے، کیونکہ طلب کی چنگاری بھی اسی نے دل میں رکھی تھی۔ یوں اس حمد میں ادب بھی ہے اور حقیقت بھی۔

اہلِ ذوق "لِهٰذَا" کے اشارے پر ٹھہرتے ہیں۔ یہ قریب کا اشارہ ہے: جو منزل سفر کے دوران دور اور غیب تھی، اب سامنے ہے اور اسے "یہ" کہا جا سکتا ہے۔ دنیا میں ایمان علم الیقین تھا، اب وہ عین الیقین بن گیا؛ سورۂ التکاثر کے صفحے پر یہی دو مرتبے گزر چکے ہیں۔ پھر "وَمَا كُنَّا لِنَهْتَدِيَ" میں نفی کا وہ زور ہے جو عربی میں لامِ جحود سے پیدا ہوتا ہے، یعنی ہماری ذات میں یہ بات تھی ہی نہیں کہ ہم خود سے راہ پا لیتے۔ شیخِ اکبر کے نزدیک ہر موجود اپنی ذات میں فقیر ہے، وجود اسے حق کی طرف سے عطا ہوا ہے؛ اور جو اپنا وجود بھی اپنا نہیں رکھتا وہ اپنی ہدایت کا مالک کیسے ہو؟ یہ اقرار جنت میں آ کر نہیں سیکھا گیا؛ یہ وہی حال ہے جسے سالک دنیا میں فقر کہتا ہے، جنت میں صرف اس پر سے پردہ اٹھا۔ اس مکتب میں حمد کو ثنا کا وہ مرتبہ سمجھا جاتا ہے جس میں بندہ ہر کمال کو اس کے اصل مالک کی طرف لوٹا دیتا ہے؛ اسی لیے قرآن کا آغاز بھی حمد سے ہے اور جنت والوں کی بات کا انجام بھی حمد پر ہے، جیسا کہ سورۂ یونس میں فرمایا: وَاٰخِرُ دَعْوٰىهُمْ اَنِ الْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِيْنَ۔ ابتدا بھی وہی، انتہا بھی وہی؛ بیچ کا سفر بندے کا تھا، مگر قدم اٹھانے کی توفیق اسی کی تھی۔ اسی لیے عارف اپنی کسی نیکی پر نہیں اکڑتا۔ وہ جانتا ہے کہ وہ اونٹ نہیں، دھاگا ہے، اور جو دروازہ اس کے لیے کھلا وہ کھولا گیا تھا۔

اور آیت کا آخری حصہ: لَقَدْ جَآءَتْ رُسُلُ رَبِّنَا بِالْحَقِّ ؕ وَنُوْدُوْۤا اَنْ تِلْكُمُ الْجَنَّةُ اُوْرِثْتُمُوْهَا بِمَا كُنْتُمْ تَعْمَلُوْنَ۔ “یقیناً ہمارے رب کے رسول حق لے کر آئے تھے۔ اور انہیں پکارا جائے گا کہ یہ وہ جنت ہے جس کے تم اپنے اعمال کے بدلے وارث بنائے گئے۔” یہاں ایک بظاہر الجھن ہے جسے علما نے سلجھایا ہے۔ صحیح بخاری اور صحیح مسلم میں حضورِ اکرم صلی اللہ علیہ وسلم کا ارشاد ہے جس کا مفہوم یہ ہے کہ تم میں سے کسی کو اس کا عمل جنت میں داخل نہیں کرے گا؛ صحابہ رضی اللہ عنہم نے عرض کیا کہ آپ صلی اللہ علیہ وسلم کو بھی نہیں؟ آپ صلی اللہ علیہ وسلم نے فرمایا کہ مجھے بھی نہیں، مگر یہ کہ اللہ مجھے اپنے فضل و رحمت میں ڈھانپ لے۔ اور یہاں فرمایا کہ تمہارے اعمال کے سبب۔ علما کہتے ہیں کہ داخلہ رحمت سے ہے اور درجات عمل کے مطابق؛ اور عمل کی توفیق خود رحمت ہے، تو عمل کے سبب ملنا بھی دراصل رحمت ہی سے ملنا ہے۔ وراثت کا لفظ اسی طرف اشارہ کرتا ہے: وارث کو میراث مزدوری کے طور پر نہیں، تعلق کی بنا پر ملتی ہے۔ بعض روایات کے مفہوم میں، جنہیں مفسرین نے ذکر کیا، آیا ہے کہ ہر انسان کے لیے جنت میں بھی ایک ٹھکانا ہے اور دوزخ میں بھی، اور ایمان والا اس کے ٹھکانے کا وارث بنتا ہے جس نے اپنا ٹھکانا ٹھکرا دیا۔ شیخِ اکبر کے مکتب میں یہاں حق کے کرم کا ایک عجیب پہلو دیکھا جاتا ہے: اعمال کا خالق وہی ہے اور بندہ کسب کرنے والا؛ پھر بھی وہ عمل کو بندے کی طرف منسوب کرتا ہے اور اس پر جزا دیتا ہے، گویا تحفہ بھی وہی دیتا ہے اور تحفہ قبول کرنے کا انعام بھی۔ اور جنت والوں نے رسولوں علیہم السلام کی سچائی کی جو گواہی دی، وہ بتاتی ہے کہ جو بات دنیا میں خبر تھی وہ اب مشاہدہ ہے۔

اب جنت اور دوزخ کے بیچ ایک مکالمہ: وَنَادٰۤى اَصْحٰبُ الْجَنَّةِ اَصْحٰبَ النَّارِ اَنْ قَدْ وَجَدْنَا مَا وَعَدَنَا رَبُّنَا حَقًّا فَهَلْ وَجَدْتُّمْ مَّا وَعَدَ رَبُّكُمْ حَقًّا ؕ قَالُوْا نَعَمْ ۚ فَاَذَّنَ مُؤَذِّنٌۢ بَيْنَهُمْ اَنْ لَّعْنَةُ اللّٰهِ عَلَى الظّٰلِمِيْنَ۔ “اور جنت والے دوزخ والوں کو پکار کر کہیں گے کہ ہم نے تو اس وعدے کو سچا پایا جو ہمارے رب نے ہم سے کیا تھا، کیا تم نے بھی اس وعدے کو سچا پایا جو تمہارے رب نے کیا تھا؟ وہ کہیں گے: ہاں۔ پھر ان کے درمیان ایک پکارنے والا پکارے گا کہ ظالموں پر اللہ کی لعنت ہو،” الَّذِيْنَ يَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ وَيَبْغُوْنَهَا عِوَجًا ۚ وَهُمْ بِالْاٰخِرَةِ كٰفِرُوْنَ۔ “جو اللہ کی راہ سے روکتے تھے اور اس میں کجی ڈھونڈتے تھے، اور وہ آخرت کے منکر تھے۔” اہلِ لطائف نے یہاں ایک نکتہ نکالا ہے: جنت والوں نے اپنے لیے کہا کہ جو ہمارے رب نے ہم سے وعدہ کیا، اور دوزخ والوں سے پوچھتے وقت صرف یہ کہا کہ جو تمہارے رب نے وعدہ کیا، تم سے کا لفظ نہیں جوڑا، گویا وعید کو براہِ راست مخاطب پر رکھنے سے ادب برتا گیا۔ دوزخ والوں کا جواب ایک لفظ ہے: "نَعَمْ"۔ ساری زندگی کا انکار ایک ہاں پر ختم ہوا، مگر یہ ہاں آنکھوں دیکھی ہے، ایمان بالغیب کی نہیں، اس لیے کام نہیں آتی۔ اعلان کرنے والا، جسے مفسرین فرشتہ کہتے ہیں، لعنت کا حکم سناتا ہے، اور لعنت کے معنی رحمت سے دوری کے ہیں۔ پھر آیت نے خود ظالموں کی پہچان کرا دی: راہ سے روکنا، راہ میں ٹیڑھ ڈھونڈنا، آخرت کا انکار۔ اہلِ معرفت کہتے ہیں کہ یہ دوری باہر سے لگائی ہوئی دیوار نہیں؛ جس نے دوسروں کو راہ سے دور رکھا، اس کا اپنا فاصلہ اسی کا عکس بن گیا، اور جس نے سیدھی راہ کو ٹیڑھا دکھانا چاہا، اس کی اپنی نظر ٹیڑھی ہو گئی۔ شیخِ اکبر کے نزدیک آخرت کی ہر جزا عمل ہی کی وہ صورت ہے جو دنیا میں چھپی تھی اور وہاں ظاہر ہوئی۔

اسی مقام کے نام پر یہ سورت الاعراف کہلاتی ہے: وَبَيْنَهُمَا حِجَابٌ ۚ وَعَلَى الْاَعْرَافِ رِجَالٌ يَّعْرِفُوْنَ كُلًّۢا بِسِيْمٰىهُمْ ۚ۔ “اور ان دونوں گروہوں کے درمیان ایک پردہ ہوگا، اور اعراف پر کچھ لوگ ہوں گے جو ہر ایک کو اس کی علامت سے پہچانیں گے۔” عرف بلند جگہ کو کہتے ہیں، جیسے گھوڑے کی ایال یا مرغ کی کلغی؛ اعراف اس حجاب کی بلندیاں ہیں جو جنت اور دوزخ کے درمیان حائل ہے۔ اعراف والے کون ہیں؟ اس میں علما کے کئی اقوال ہیں۔ اکثر مفسرین نے، جن میں حضرت حذیفہ، حضرت عبد اللہ بن عباس اور حضرت عبد اللہ بن مسعود رضی اللہ عنہم سے منقول اقوال شامل ہیں، یہی بیان کیا ہے کہ یہ وہ لوگ ہیں جن کی نیکیاں اور برائیاں برابر ہو گئیں؛ نیکیوں نے انہیں دوزخ سے بچا لیا اور برائیوں نے جنت سے روک لیا، یہاں تک کہ اللہ اپنی رحمت سے ان کا فیصلہ فرمائے۔ بعض نے دوسرے اقوال بھی ذکر کیے، مثلاً یہ کہ وہ اہلِ علم و فضل ہیں یا فرشتے؛ اور ان اقوال کو جمع کرنے والے مفسرین نے پہلے قول ہی کو سب سے زیادہ معروف بتایا ہے۔ سیما وہ نشان ہے جو چہرے پر ظاہر ہو؛ سورۂ الفتح کے صفحے پر سجدے کے نشان کا ذکر گزر چکا ہے۔ آخرت میں باطن چہروں پر لکھا ہوگا: ایک کا نشان نور، دوسرے کا سیاہی۔ اہلِ معرفت کہتے ہیں کہ اعراف کے لفظ میں معرفت بھی چھپی ہے، کیونکہ عرف اور معرفت ایک ہی مادے سے ہیں؛ بلندی پر وہی کھڑا ہے جو دونوں طرف پہچانتا ہے، اور آیت میں پہچاننے کا فعل اسی وصف کو نمایاں کرتا ہے۔ یہ اشارہ ظاہر پر اضافہ ہے، ظاہر کا بدل نہیں۔

اعراف والوں کی پہلی پکار جنت کی طرف ہے: وَنَادَوْا اَصْحٰبَ الْجَنَّةِ اَنْ سَلٰمٌ عَلَيْكُمْ لَمْ يَدْخُلُوْهَا وَهُمْ يَطْمَعُوْنَ۔ “اور وہ جنت والوں کو پکار کر کہیں گے: تم پر سلامتی ہو۔ وہ ابھی اس میں داخل نہیں ہوئے، مگر اس کی امید رکھتے ہیں۔” وَاِذَا صُرِفَتْ اَبْصَارُهُمْ تِلْقَآءَ اَصْحٰبِ النَّارِ ۙ قَالُوْا رَبَّنَا لَا تَجْعَلْنَا مَعَ الْقَوْمِ الظّٰلِمِيْنَ۔ “اور جب ان کی نگاہیں دوزخ والوں کی طرف پھیری جائیں گی تو کہیں گے: اے ہمارے رب! ہمیں ان ظالم لوگوں کے ساتھ نہ کرنا۔” ان کی زبان پر پہلا لفظ سلام ہے؛ وہ اپنے انتظار میں بھی دوسروں کی سلامتی کی دعا کرتے ہیں۔ طمع یہاں امید کے معنی میں ہے، اور حضرت حسن بصری رحمۃ اللہ علیہ سے منقول ہے، جس کا مفہوم یہ ہے کہ اللہ نے یہ امید ان کے دلوں میں اسی لیے ڈالی کہ وہ ان کے ساتھ اکرام کا ارادہ رکھتا تھا۔ اور غور کیجیے کہ "صُرِفَتْ" مجہول ہے: ان کی نگاہیں دوزخ کی طرف خود نہیں اٹھتیں، پھیری جاتی ہیں۔ اہلِ ذوق کہتے ہیں کہ ادب والا دل ہولناک منظر کو شوق سے نہیں دیکھتا؛ اس کی آنکھ کا فطری رخ رحمت کی طرف ہے، اور جب دوسری طرف دیکھنا پڑے تو فوراً پناہ کی دعا نکلتی ہے۔ شیخِ اکبر کے مکتب میں اعراف کو برزخ کا مقام سمجھا گیا ہے، یعنی دو حکموں کے بیچ کی وہ حد جو دونوں کو دیکھتی ہے اور کسی ایک میں گم نہیں ہوتی۔ سالک کے لیے اس میں سبق یہ ہے کہ دنیا میں بھی وہ امید اور خوف کے بیچ کھڑا رہے: اپنی کوتاہی دیکھ کر ظالموں کی صف سے پناہ مانگے، اور رحمت دیکھ کر سلام کہتا رہے۔ یہ امید جھوٹی تمنا نہیں؛ یہ اس رب کی رحمت پر بھروسا ہے جس کی رحمت اس کے غضب پر سبقت رکھتی ہے، جیسا کہ صحیح بخاری کی حدیثِ قدسی کا مفہوم ہے۔
