پچھلا رکوع خوش حالوں کے اس دعوے پر ختم ہوا تھا کہ ہم مال اور اولاد میں زیادہ ہیں۔ اب اس دعوے کی جڑ کاٹی جاتی ہے: وَمَاۤ اَمْوَالُكُمْ وَلَاۤ اَوْلَادُكُمْ بِالَّتِيْ تُقَرِّبُكُمْ عِنْدَنَا زُلْفٰۤى اِلَّا مَنْ اٰمَنَ وَعَمِلَ صَالِحًا فَاُولٰۤئِكَ لَهُمْ جَزَآءُ الضِّعْفِ بِمَا عَمِلُوْا وَهُمْ فِي الْغُرُفٰتِ اٰمِنُوْنَ۔ اور تمہارے مال اور تمہاری اولاد وہ چیز نہیں جو تمہیں ہمارے ہاں قرب میں نزدیک کر دے، مگر جو ایمان لایا اور نیک عمل کیا، تو ایسوں کے لیے ان کے عمل کا دگنا بدلہ ہے، اور وہ بالاخانوں میں امن سے رہیں گے۔ ظاہر میں یہ اعلان ہے کہ قرب کا سکہ مال نہیں، ایمان اور عمل ہے؛ اور یہی معنی بنیاد ہے، اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہاں "ہونے" اور "رکھنے" کا فرق دیکھا گیا۔ مال اور اولاد وہ ہیں جو بندہ رکھتا ہے؛ ایمان اور عمل وہ ہیں جو بندہ ہوتا ہے۔ قرب رکھنے سے نہیں، ہونے سے ملتا ہے؛ کیونکہ جو چیز ہاتھ میں ہے وہ بندے سے باہر ہے، اور حق بندے کے باطن سے قریب ہوتا ہے، اس کی ملکیت سے نہیں۔ پھر استثنا کی ترکیب پر غور کیجیے: آیت نے یہ نہیں کہا کہ مال بالکل بے قیمت ہے؛ جو ایمان لایا اور عمل کیا، اس کے ہاتھ کا مال بھی عمل کا ذریعہ بن کر قرب کا سبب ہو جاتا ہے۔ "جَزَآءُ الضِّعْفِ" میں اہلِ ذوق سبقتِ رحمت دیکھتے ہیں، کہ عطا عمل سے بڑھ کر آتی ہے؛ اور "الْغُرُفٰتِ" کو دل کے بلند مقامات کہتے ہیں، جہاں پہنچ کر بندہ اس خوف سے امن پاتا ہے جو ترف والوں کے دل میں چھپا رہتا تھا۔

اس کے مقابل دوسرا گروہ: وَالَّذِيْنَ يَسْعَوْنَ فِيْۤ اٰيٰتِنَا مُعٰجِزِيْنَ اُولٰۤئِكَ فِي الْعَذَابِ مُحْضَرُوْنَ۔ اور جو لوگ ہماری آیتوں کو ہرانے کی کوشش میں دوڑ دھوپ کرتے ہیں، وہ عذاب میں حاضر کیے جائیں گے۔ پھر رزق کا قانون دوبارہ آتا ہے، مگر اس بار ایک نئے اضافے کے ساتھ: قُلْ اِنَّ رَبِّيْ يَبْسُطُ الرِّزْقَ لِمَنْ يَّشَآءُ مِنْ عِبَادِهٖ وَيَقْدِرُ لَهٗ ۭ وَمَاۤ اَنْفَقْتُمْ مِّنْ شَيْءٍ فَهُوَ يُخْلِفُهٗ ۚ وَهُوَ خَيْرُ الرّٰزِقِيْنَ۔ کہہ دیجیے کہ میرا رب اپنے بندوں میں سے جس کے لیے چاہتا ہے رزق کشادہ کرتا ہے اور جس کے لیے چاہتا ہے تنگ کرتا ہے؛ اور تم جو کچھ بھی خرچ کرتے ہو وہ اس کی جگہ اور دیتا ہے، اور وہ سب سے بہتر رزق دینے والا ہے۔ "مُعٰجِزِيْنَ" کا معنی ہے حق کو عاجز کرنے کی کوشش کرنے والے؛ اہلِ معرفت کہتے ہیں کہ جو حق کو ہرانا چاہے وہ دراصل اپنی ہی استعداد سے لڑتا ہے، اور "مُحْضَرُوْنَ" اس کا انجام ہے کہ جس حضور سے بھاگتا تھا وہاں حاضر کیا جائے۔ پچھلے رکوع میں یہی رزق والی آیت خوش حال سرداروں (مترفین) کے جواب میں تھی؛ یہاں "مِنْ عِبَادِهٖ" کے اضافے کے ساتھ مومنوں سے خطاب ہے، اور اس کا ثمر "فَهُوَ يُخْلِفُهٗ" ہے۔ اس مکتب میں دینے والا اپنے خزانے سے نہیں دیتا، حق کے خزانے سے دیتا ہے؛ بندہ محض گزرگاہ ہے۔ اور جو خزانہ اس کا نہیں، اس کے خرچ سے وہ کم کیسے ہوگا؟ پھر خلقِ جدید کا راز یہاں بھی جاری ہے: عالم ہر آن نیا پیدا ہوتا ہے، اور جو کچھ بندے کے ہاتھ سے نکلا، اگلی آن حق اس کی جگہ نیا بھر دیتا ہے۔ "خَيْرُ الرّٰزِقِيْنَ" فرما کر یہ بھی سکھایا کہ دوسرے رزق دینے والے مجازاً ہیں، حقیقت میں رازق ایک ہے۔

اب قیامت کا ایک اور منظر کھلتا ہے: وَيَوْمَ يَحْشُرُهُمْ جَمِيْعًا ثُمَّ يَقُوْلُ لِلْمَلٰۤئِكَةِ اَهٰۤؤُلَآءِ اِيَّاكُمْ كَانُوْا يَعْبُدُوْنَ۔ اور جس دن وہ ان سب کو جمع کرے گا، پھر فرشتوں سے فرمائے گا: کیا یہ لوگ تمہاری عبادت کیا کرتے تھے؟ قَالُوْا سُبْحٰنَكَ اَنْتَ وَلِيُّنَا مِنْ دُوْنِهِمْ ۚ بَلْ كَانُوْا يَعْبُدُوْنَ الْجِنَّ ۚ اَكْثَرُهُمْ بِهِمْ مُّؤْمِنُوْنَ۔ وہ کہیں گے: تو پاک ہے، تو ہی ہمارا کارساز ہے، نہ کہ یہ؛ بلکہ یہ جنوں کی عبادت کرتے تھے، ان میں سے اکثر انہی پر ایمان رکھتے تھے۔ فرشتوں کے جواب میں ادب کی ترتیب دیکھیے۔ پہلے "سُبْحٰنَكَ"، یعنی تنزیہ؛ پھر "اَنْتَ وَلِيُّنَا"، یعنی نسبت اور تعلق کا اقرار؛ پھر "مِنْ دُوْنِهِمْ"، یعنی غیر سے برأت۔ اہلِ معرفت کہتے ہیں کہ جواب دینے سے پہلے حق کو ہر نسبتِ ناروا سے پاک کرنا ملائکہ کا ادب ہے، اور یہی ہر سوال کے سامنے عارف کا ادب ہونا چاہیے۔ پھر وہ ایک گہری بات کھولتے ہیں: یہ لوگ ہمیں پوجتے نہیں تھے، جنوں کو پوجتے تھے۔ ظاہر میں مراد یہ ہے کہ شیاطینِ جن نے انہیں اس پرستش پر ابھارا اور یہ انہی کے کہنے پر چلے۔ اشارے میں اہلِ تاویل کہتے ہیں کہ جو کسی غیرِ حق کو پوجتا ہے وہ دراصل اس صورت کو پوجتا ہے جو اس کے اپنے خیال اور خواہش نے بنائی، اور جن اسی خیال کو سجاتے ہیں؛ فرشتہ اس صورت سے بری ہے۔ یوں ہر باطل عبادت آخرکار اپنی ہی تراشی ہوئی تصویر کی عبادت ہے۔

جب پرستش کی حقیقت کھل گئی تو ملکیت کا وہم بھی ٹوٹ جاتا ہے: فَالْيَوْمَ لَا يَمْلِكُ بَعْضُكُمْ لِبَعْضٍ نَّفْعًا وَّلَا ضَرًّا ۭ وَنَقُوْلُ لِلَّذِيْنَ ظَلَمُوْا ذُوْقُوْا عَذَابَ النَّارِ الَّتِيْ كُنْتُمْ بِهَا تُكَذِّبُوْنَ۔ پس آج تم میں سے کوئی کسی کے لیے نہ نفع کا اختیار رکھتا ہے نہ نقصان کا؛ اور ہم ظالموں سے کہیں گے: اس آگ کا عذاب چکھو جسے تم جھٹلاتے تھے۔ شیخِ اکبر کے مکتب میں "فَالْيَوْمَ" کا مطلب یہ نہیں کہ آج ملکیت چھن گئی؛ ملکیت کسی کے پاس پہلے بھی نہ تھی، آج صرف پردہ اٹھا ہے۔ قیامت حقائق کو بدلتی نہیں، ظاہر کرتی ہے۔ پھر دنیا میں ان کا رویہ بیان ہوتا ہے: وَاِذَا تُتْلٰى عَلَيْهِمْ اٰيٰتُنَا بَيِّنٰتٍ قَالُوْا مَا هٰذَاۤ اِلَّا رَجُلٌ يُّرِيْدُ اَنْ يَّصُدَّكُمْ عَمَّا كَانَ يَعْبُدُ اٰبَآؤُكُمْ۔ اور جب ان پر ہماری کھلی آیتیں پڑھی جاتی ہیں تو کہتے ہیں: یہ تو بس ایک آدمی ہے جو چاہتا ہے کہ تمہیں ان سے روک دے جن کی تمہارے باپ دادا عبادت کرتے تھے۔ اور اسی آیت میں آگے: وَقَالُوْا مَا هٰذَاۤ اِلَّاۤ اِفْكٌ مُّفْتَرًى ۭ وَقَالَ الَّذِيْنَ كَفَرُوْا لِلْحَقِّ لَمَّا جَآءَهُمْ ۙ اِنْ هٰذَاۤ اِلَّا سِحْرٌ مُّبِيْنٌ۔ اور کہا: یہ تو گھڑا ہوا جھوٹ ہے؛ اور کافروں نے حق کے بارے میں، جب وہ ان کے پاس آیا، کہا: یہ تو کھلا جادو ہے۔ اہلِ معرفت کہتے ہیں کہ تین الزام تین پردوں کی خبر دیتے ہیں: پہلا عادت کا پردہ، یعنی باپ دادا کی رسم؛ دوسرا عقل کی تنگی کا پردہ، جو ہر نئی بات کو گھڑنت کہتی ہے؛ اور تیسرا اثر کو دیکھ کر انکار کا پردہ، کہ کلام دل کو کھینچتا تھا تو اسے جادو کہہ دیا۔ تیسرے الزام میں لاشعوری اقرار ہے کہ اس کلام کی تاثیر انکار سے بڑی تھی۔

اب ان کی محرومی کے ساتھ ان کی خوش نصیبی بھی یاد دلائی جاتی ہے: وَمَاۤ اٰتَيْنٰهُمْ مِّنْ كُتُبٍ يَّدْرُسُوْنَهَا وَمَاۤ اَرْسَلْنَاۤ اِلَيْهِمْ قَبْلَكَ مِنْ نَّذِيْرٍ۔ اور ہم نے انہیں نہ کوئی کتابیں دی تھیں جنہیں یہ پڑھتے ہوں، اور نہ آپ سے پہلے ان کی طرف کوئی ڈرانے والا بھیجا۔ وَكَذَّبَ الَّذِيْنَ مِنْ قَبْلِهِمْ ۙ وَمَا بَلَغُوْا مِعْشَارَ مَاۤ اٰتَيْنٰهُمْ فَكَذَّبُوْا رُسُلِيْ فَكَيْفَ كَانَ نَكِيْرِ۔ اور ان سے پہلے والوں نے بھی جھٹلایا تھا، اور یہ اس کے دسویں حصے کو بھی نہیں پہنچے جو ہم نے انہیں دیا تھا؛ پس انہوں نے میرے رسولوں کو جھٹلایا، تو دیکھو میرا انکار کیسا تھا۔ ظاہر میں آیت ۴۴ بتاتی ہے کہ ان کے شرک کی کوئی آسمانی سند نہ تھی؛ نہ کتاب نہ رسول۔ اہلِ ذوق اس میں ایک اور پہلو دیکھتے ہیں: جس قوم کے پاس پہلے کوئی کتاب نہ ہو اس کا ظرف خالی ہے، اور خالی ظرف بھرنے کے لیے سب سے زیادہ تیار ہوتا ہے۔ پہلی بار آنے والی وحی ایک عطا تھی جو کسی پرانی تحریف سے ملی ہوئی نہ تھی؛ مگر انہوں نے اسی خالی پن کو انکار سے بھر لیا۔ آیت ۴۵ میں "مِعْشَارَ" کا لفظ پیمانے کی خبر دیتا ہے: پچھلی قوموں کو قوت اور مال کا جو حصہ ملا تھا، یہ اس کا دسواں حصہ بھی نہیں رکھتے؛ وہ نہ بچے تو یہ کس بنیاد پر بچنے کا گمان رکھتے ہیں۔ اور "نَكِيْرِ" پر غور کیجیے: انہوں نے رسولوں کا انکار کیا تو حق کا انکار ان پر آیا۔ اس مکتب میں یہ بدلہ عمل ہی کا آئینہ ہے؛ جس نے حق کو جھٹلایا، اس کا جھٹلانا ہی پلٹ کر اس پر انکار بن کر آیا۔ یہ زبردستی نہیں، بندے کے اپنے رخ کا عکس ہے۔

اب وہ آیت آتی ہے جس میں اس سورت کی ساری نصیحت، بلکہ پورے طریق کی روش، ایک جملے میں سمٹ آئی ہے: قُلْ اِنَّمَاۤ اَعِظُكُمْ بِوَاحِدَةٍ ۚ اَنْ تَقُوْمُوْا لِلّٰهِ مَثْنٰى وَفُرَادٰى ثُمَّ تَتَفَكَّرُوْا۔ کہہ دیجیے کہ میں تمہیں صرف ایک بات کی نصیحت کرتا ہوں: یہ کہ تم اللہ کے لیے دو دو اور اکیلے اکیلے کھڑے ہو جاؤ، پھر غور کرو۔ ظاہر میں یہ مشرکینِ مکہ سے فرمایا گیا: ضد اور ہجوم کے شور سے الگ ہو کر، خالص نیت سے، حضورِ اکرم صلی اللہ علیہ وسلم کے حال پر غور کرو۔ پہلا لفظ "بِوَاحِدَةٍ" ہے: ایک ہی بات۔ اہلِ معرفت کہتے ہیں کہ راستہ اپنی حقیقت میں ایک ہے، کیونکہ منزل ایک ہے؛ کثرت منزلوں کی ہے، راہ کی نہیں۔ دوسرا کلمہ "اَنْ تَقُوْمُوْا" ہے۔ اہلِ سلوک کے نزدیک پہلی منزل یقظہ ہے، یعنی غفلت کی نیند سے جاگنا؛ اور قیام اسی جاگنے کی صورت ہے۔ جو لیٹا ہے وہ کچھ نہیں دیکھتا، جو اٹھ کھڑا ہوا اس کے سامنے افق کھل گیا۔ اور سب سے نازک کلمہ "لِلّٰهِ" ہے۔ قیام اپنی برتری کے لیے نہیں، بحث جیتنے کے لیے نہیں، کسی گروہ کی حمایت کے لیے نہیں؛ صرف اللہ کے لیے۔ شیخِ اکبر کے مکتب میں اخلاص یہی ہے کہ عمل کا رخ ایک ہو جائے؛ جس کا رخ ایک ہو گیا، وہ ہر طرف اسی ایک کو دیکھتا ہے۔ اسی "لِلّٰهِ" نے بعد کی ساری ترتیب کو بامعنی بنایا؛ اس کے بغیر صحبت محض مجلس، خلوت محض تنہائی اور فکر محض الجھن رہ جاتی۔

"مَثْنٰى" یعنی دو دو۔ ظاہر میں حکمت یہ ہے کہ دو آدمی آپس میں بغیر شور کے، بغیر دکھاوے کے، انصاف سے بات کر سکتے ہیں؛ ہجوم میں عقل گم ہو جاتی ہے اور ہر فرد ہجوم کی آواز بول اٹھتا ہے، جیسا کہ پچھلے رکوع میں مستضعفین نے اپنے سرداروں کی آواز کو اپنی آواز بنا لیا تھا۔ اہلِ تصوف نے اس میں صحبت کا اصول دیکھا۔ سالک اکیلا اپنا عیب نہیں دیکھ سکتا، کیونکہ آنکھ اپنے آپ کو نہیں دیکھتی؛ اسے ایک ساتھی چاہیے جو آئینہ ہو۔ حدیث کا مفہوم ہے کہ مومن مومن کا آئینہ ہے۔ شیخِ اکبر کے مکتب میں آئینے کا یہ تصور بہت گہرا ہے: حق نے چاہا کہ اپنے اسماء کو ایک جامع آئینے میں دیکھے، اور انسانِ کامل وہ آئینہ بنا؛ اسی طرح ہر بندہ اپنے صادق ساتھی کے آئینے میں اپنی حقیقت کا کچھ حصہ دیکھتا ہے جو اکیلے میں اس سے چھپا رہتا۔ اسی لیے طریق میں شیخ اور مرید، اور ساتھی اور ساتھی، کی صحبت کو اتنی اہمیت ملی۔ مگر شرط وہی "لِلّٰهِ" ہے؛ جو صحبت اللہ کے لیے نہ ہو وہ مترفین کی مجلس بن جاتی ہے جہاں ایک دوسرے کی تائید سے باطل مضبوط ہوتا ہے۔ اور "مَثْنٰى" کی تعداد بھی اشارہ رکھتی ہے: دو ہیں، دو سو نہیں۔ صحبت کی حقیقت قلت میں محفوظ رہتی ہے؛ جہاں ساتھی بہت ہو جائیں وہاں دل اکثر ہجوم کی طرف کھنچ جاتا ہے۔

"وَفُرَادٰى" یعنی اکیلے اکیلے۔ صحبت کے بعد خلوت؛ اور یہ ترتیب اتفاقی نہیں۔ ساتھی نے آئینہ دکھایا، اب بندہ اکیلا اپنے رب کے سامنے کھڑا ہو کر اس عکس کو تولتا ہے۔ یہی لفظ ایک اور آیت میں قیامت کے دن کے لیے آیا ہے: وَلَقَدْ جِئْتُمُوْنَا فُرَادٰى كَمَا خَلَقْنٰكُمْ اَوَّلَ مَرَّةٍ۔ اہلِ معرفت کہتے ہیں کہ جو آج اپنے اختیار سے اکیلا ہو کر رب کے سامنے کھڑا ہو گیا، اس کے لیے کل کی تنہائی اجنبی نہ ہوگی؛ وہ اپنی موت سے پہلے اس حضور کا ذائقہ چکھ چکا۔ خلوت میں بندے کے سارے سہارے، جن پر پچھلے رکوع کے متکبر اور مستضعف ٹکے ہوئے تھے، ایک ایک کر کے گر جاتے ہیں؛ نہ سردار ہے نہ تابع، نہ مال ہے نہ اولاد۔ یہی تجرید ہے، اور فتوحاتِ مکیہ میں شیخِ اکبر نے خلوت کے آداب پر مفصل کلام کیا ہے؛ اور اس مکتب میں خلوت کی روح یہ سمجھی گئی کہ وہ جگہ کی تنہائی نہیں، دل کا غیر سے خالی ہونا ہے۔ اور اہلِ ذوق ایک لطیف بات کہتے ہیں: یہ نصیحت اس ہستی کی زبان سے ادا ہوئی جس کی حکمت کو فصوص میں حکمتِ فردیہ کہا گیا؛ فرد ہی فرادیٰ کی دعوت دیتا ہے۔ مگر خلوت کو صحبت سے کاٹ دیا جائے تو وہ نفس کے فریب کا گوشہ بن سکتی ہے، اسی لیے آیت نے دونوں کو ایک ساتھ رکھا: "مَثْنٰى وَفُرَادٰى"۔ صحبت خلوت کو فریب سے بچاتی ہے، اور خلوت صحبت کو ہجوم بننے سے۔

پھر تیسرا قدم: "ثُمَّ تَتَفَكَّرُوْا"۔ "ثُمَّ" ترتیب اور مہلت کا حرف ہے؛ فکر قیام کے بعد، صحبت اور خلوت کے بعد آتی ہے۔ جو اٹھا نہیں، جس نے نیت خالص نہیں کی، جس نے نہ آئینہ دیکھا نہ تنہائی میں رب کے سامنے کھڑا ہوا، اس کی فکر محض خیالات کا ہجوم ہے۔ شیخِ اکبر کے مکتب میں فکر کی ایک حد ہے: عقل مخلوقات میں غور کر کے خالق کی نشانیوں تک پہنچتی ہے، مگر ذاتِ حق اس کی گرفت سے باہر ہے؛ اسی لیے حدیث کا مفہوم ہے کہ اللہ کی مخلوق میں غور کرو، اس کی ذات میں نہیں۔ اور یہاں آیت نے فکر کا موضوع خود متعین کر دیا: مَا بِصَاحِبِكُمْ مِّنْ جِنَّةٍ ۭ اِنْ هُوَ اِلَّا نَذِيْرٌ لَّكُمْ بَيْنَ يَدَيْ عَذَابٍ شَدِيْدٍ۔ تمہارے ساتھی کو کوئی جنون نہیں؛ وہ تو بس تمہیں ایک سخت عذاب سے پہلے خبردار کرنے والا ہے۔ غور کا پہلا موضوع حضورِ اکرم صلی اللہ علیہ وسلم کی ذاتِ گرامی ہے، اور انہیں "صَاحِبِكُمْ" کہا: تمہارا ساتھی، جسے تم بچپن سے جانتے ہو، جس کی امانت اور صدق کے تم خود گواہ ہو۔ اہلِ معرفت کہتے ہیں کہ یہی طریق کی کنجی ہے: انسانِ کامل کے حال میں غور، وہ آئینہ ہے جس سے حق کی پہچان کا دروازہ کھلتا ہے۔ یوں ایک آیت میں پورا سلوک آ گیا: یقظہ اور اخلاص کا قیام، صحبت، خلوت، فکر، اور اس فکر کا ثمر یعنی رسول کی معرفت؛ اور رسول کی معرفت رب کی معرفت کا راستہ ہے۔

نصیحت کے بعد بے غرضی کا اعلان: قُلْ مَا سَاَلْتُكُمْ مِّنْ اَجْرٍ فَهُوَ لَكُمْ ۭ اِنْ اَجْرِيَ اِلَّا عَلَى اللّٰهِ ۚ وَهُوَ عَلٰى كُلِّ شَيْءٍ شَهِيْدٌ۔ کہہ دیجیے کہ میں نے تم سے جو اجر مانگا ہو وہ تمہارا ہی ہے؛ میرا اجر تو صرف اللہ کے ذمے ہے، اور وہ ہر چیز پر گواہ ہے۔ پھر حق کے نزول کی کیفیت: قُلْ اِنَّ رَبِّيْ يَقْذِفُ بِالْحَقِّ ۚ عَلَّامُ الْغُيُوْبِ۔ کہہ دیجیے کہ میرا رب حق کو پھینکتا ہے، وہ سب غیبوں کا خوب جاننے والا ہے۔ "فَهُوَ لَكُمْ" میں ایک لطیف نکتہ ہے: اگر میں کچھ مانگتا بھی تو وہ تمہارے ہی فائدے کے لیے ہوتا؛ نبی کی ہر طلب امت کی بھلائی ہے۔ اہلِ معرفت کہتے ہیں کہ یہی وارثِ نبوت کی پہچان ہے، کہ وہ لوگوں سے کچھ نہیں چاہتا؛ جو رہنما مریدوں سے اپنا حصہ مانگے، وہ مترفین کے طریق پر ہے، انبیاء کے طریق پر نہیں۔ اور "يَقْذِفُ" کا لفظ اچانک اور زور سے پھینکنے کے لیے ہے۔ سورۂ الانبیاء میں یہی فعل آیا تھا: بَلْ نَقْذِفُ بِالْحَقِّ عَلَى الْبَاطِلِ فَيَدْمَغُهٗ فَاِذَا هُوَ زَاهِقٌ۔ شیخِ اکبر کے مکتب میں حق کا دل میں اترنا اکثر یوں ہی ہوتا ہے: سوچ کی سیڑھی سے نہیں، اچانک القا سے؛ اسی کو فتح اور کشف کہا گیا۔ فکر برتن کو صاف کرتی ہے اور قذف اسے بھرتا ہے۔ اور "عَلَّامُ الْغُيُوْبِ" بتاتا ہے کہ وہ ہر دل کے غیب کو جانتا ہے، اس لیے وہی جانتا ہے کہ کس دل میں کب اور کتنا حق ڈالنا ہے۔

پھر اعلانِ فتح: قُلْ جَآءَ الْحَقُّ وَمَا يُبْدِئُ الْبَاطِلُ وَمَا يُعِيْدُ۔ کہہ دیجیے کہ حق آ گیا، اور باطل نہ کچھ شروع کرتا ہے نہ دہراتا ہے۔ شیخِ اکبر کے مکتب میں یہ آیت خلقِ جدید کی کنجی ہے۔ ابتدا اور اعادہ صرف اسی کا کام ہے جس کا وجود ہو؛ اور وجود حقیقت میں حق ہی کا ہے۔ عالم ہر آن نیا پیدا ہوتا ہے اور اپنی مثل میں لوٹتا ہے، اور یہ پیدا کرنا اور لوٹانا صرف حق کرتا ہے؛ باطل عدم ہے، اور عدم نہ کسی چیز کو شروع کر سکتا ہے نہ لوٹا سکتا ہے۔ اسی لیے باطل کو جو قوت نظر آتی ہے وہ اس کی اپنی نہیں، بندوں کی توجہ کی ادھار لی ہوئی ہے؛ توجہ ہٹی تو باطل گر گیا۔ اب وہ آیت جو اس سورت میں ہدایت و گمراہی کا اصول ادب کی زبان میں بیان کرتی ہے، اور اسے اصول ہی کے طور پر پڑھنا چاہیے: قُلْ اِنْ ضَلَلْتُ فَاِنَّمَاۤ اَضِلُّ عَلٰى نَفْسِيْ ۚ وَاِنِ اهْتَدَيْتُ فَبِمَا يُوْحِيْۤ اِلَيَّ رَبِّيْ ۭ اِنَّهٗ سَمِيْعٌ قَرِيْبٌ۔ کہہ دیجیے کہ اگر میں بھٹکوں تو اپنے ہی نقصان پر بھٹکوں گا، اور اگر میں ہدایت پاؤں تو اس وحی کے سبب جو میرا رب میری طرف بھیجتا ہے؛ بے شک وہ سننے والا، قریب ہے۔ گمراہی کو نفس کی طرف منسوب کیا، ہدایت کو رب کی طرف۔ یہ وہی ادبِ نسبت ہے جو حضرت ابراہیم علیہ السلام نے برتا تھا: وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ۔ بیماری اپنی طرف، شفا رب کی طرف۔ اہلِ معرفت کہتے ہیں کہ یہ محض ادب نہیں، حقیقت بھی ہے: ہدایت وجود ہے اور وجود حق کی طرف سے ہے؛ گمراہی بندے کا اپنا اعراض ہے، جو حق کی طرف سے مسلط نہیں۔ حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت طلب کرے اسے کبھی گمراہ نہیں کرتا، کیونکہ وہ "سَمِيْعٌ قَرِيْبٌ" ہے: طلب سنتا ہے اور طالب سے قریب ہے۔

سورت کا اختتام ایک منظر پر ہوتا ہے جو اسی قرب کے مقابل دوری کی تصویر ہے: وَلَوْ تَرٰۤى اِذْ فَزِعُوْا فَلَا فَوْتَ وَاُخِذُوْا مِنْ مَّكَانٍ قَرِيْبٍ۔ اور کاش تم دیکھو جب وہ گھبرائیں گے تو بچ نکلنے کی کوئی راہ نہ ہوگی، اور قریب کی جگہ سے پکڑ لیے جائیں گے۔ وَّقَالُوْۤا اٰمَنَّا بِهٖ ۚ وَاَنّٰى لَهُمُ التَّنَاوُشُ مِنْ مَّكَانٍ بَعِيْدٍ۔ اور کہیں گے کہ ہم اس پر ایمان لے آئے؛ اور اتنی دور کی جگہ سے اسے پا لینا ان کے لیے کہاں ممکن ہے۔ وَقَدْ كَفَرُوْا بِهٖ مِنْ قَبْلُ ۚ وَيَقْذِفُوْنَ بِالْغَيْبِ مِنْ مَّكَانٍ بَعِيْدٍ۔ حالانکہ اس سے پہلے وہ اس کا انکار کر چکے تھے، اور دور کی جگہ سے بن دیکھے تیر چلاتے رہے تھے۔ وَحِيْلَ بَيْنَهُمْ وَبَيْنَ مَا يَشْتَهُوْنَ كَمَا فُعِلَ بِاَشْيَاعِهِمْ مِّنْ قَبْلُ ۭ اِنَّهُمْ كَانُوْا فِيْ شَكٍّ مُّرِيْبٍ۔ اور ان کے اور ان کی چاہت کے درمیان رکاوٹ ڈال دی جائے گی، جیسا کہ ان سے پہلے ان کے ہم مشربوں کے ساتھ کیا گیا؛ بے شک وہ الجھا دینے والے شک میں تھے۔ قریب اور بعید کا تقابل دیکھیے۔ حق "قَرِيْبٌ" ہے، اور پکڑ بھی "مِنْ مَّكَانٍ قَرِيْبٍ" ہے؛ مگر ایمان کا ہاتھ "مِنْ مَّكَانٍ بَعِيْدٍ" سے بڑھا۔ اہلِ معرفت کہتے ہیں کہ دوری حق کی طرف سے نہیں تھی، ان کی اپنی تھی؛ "وَقَدْ كَفَرُوْا بِهٖ مِنْ قَبْلُ" بتاتا ہے کہ یہ دوری ان کے پہلے اعراض نے پیدا کی۔ دیکھنے کے بعد کا ایمان غیب پر ایمان نہیں رہتا، اسی لیے وہ قبول نہ ہوا۔ اور "حِيْلَ بَيْنَهُمْ" میں سرِّ قدر پھر ظاہر ہے: جو ساری عمر چاہت اور حقیقت کے درمیان شک کی دیوار کھڑی کرتا رہا، اس کی وہی دیوار آخر میں اس کے اور اس کی چاہت کے درمیان کھڑی ہو گئی۔ سورت علم سے شروع ہوئی تھی کہ کوئی ذرہ اس سے اوجھل نہیں، اور شک کی خبر پر ختم ہوئی؛ درمیان کا راستہ وہی ایک نصیحت ہے: اللہ کے لیے دو دو اور اکیلے اکیلے کھڑے ہو جاؤ، پھر غور کرو۔
