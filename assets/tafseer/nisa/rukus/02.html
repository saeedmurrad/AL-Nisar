پچھلے رکوع نے اعلان کیا تھا کہ عورت اور مرد دونوں کا حصہ مقرر ہے؛ اب وہ حصے خود حق تعالیٰ بیان فرماتا ہے۔ روایت ہے (ترمذی، ابو داؤد) کہ حضرت سعد بن ربیع رضی اللہ عنہ احد میں شہید ہوئے تو ان کے بھائی نے سارا مال لے لیا، اور ان کی بیوہ اپنی دو بیٹیوں کو لے کر حضورِ اکرم صلی اللہ علیہ وسلم کی خدمت میں حاضر ہوئیں؛ آپ صلی اللہ علیہ وسلم نے فرمایا کہ اللہ اس بارے میں فیصلہ فرمائے گا، اور یہ آیات نازل ہوئیں: يُوْصِيْكُمُ اللّٰهُ فِيْۤ اَوْلَادِكُمْ ۗ لِلذَّكَرِ مِثْلُ حَظِّ الْاُنْثَيَيْنِ ۚ فَاِنْ كُنَّ نِسَآءً فَوْقَ اثْنَتَيْنِ فَلَهُنَّ ثُلُثَا مَا تَرَكَ ۚ وَاِنْ كَانَتْ وَاحِدَةً فَلَهَا النِّصْفُ۔ “اللہ تمہیں تمہاری اولاد کے بارے میں وصیت کرتا ہے: بیٹے کا حصہ دو بیٹیوں کے حصے کے برابر ہے؛ پھر اگر صرف بیٹیاں ہوں، دو سے زیادہ، تو ان کے لیے ترکے کا دو تہائی ہے؛ اور اگر ایک ہی بیٹی ہو تو اس کے لیے آدھا ہے۔” دو بیٹیوں کا حصہ بھی امت کے جمہور کے نزدیک دو تہائی ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم نے حضرت سعد رضی اللہ عنہ کی دونوں بیٹیوں کو دو تہائی اور ان کی والدہ کو آٹھواں حصہ دلوایا، باقی چچا کو۔ آیت کا پہلا لفظ “يُوْصِيْكُمُ” ہے، گویا ایک شفیق کی تاکید؛ حق تعالیٰ نے یہ نہیں فرمایا کہ اپنی اولاد کے بارے میں خود فیصلہ کرو، بلکہ یہ کہ وہ تمہیں تمہاری اولاد کے بارے میں وصیت کرتا ہے، یعنی وہ تمہارے بچوں پر تم سے زیادہ مہربان ہے؛ جیسا کہ حدیث میں ہے، جس کا مفہوم یہ ہے کہ اللہ اپنے بندوں پر اس ماں سے زیادہ رحیم ہے جو اپنے بچے کو سینے سے لگاتی ہے (بخاری، مسلم)۔ قابلِ غور یہ بھی ہے کہ پیمانہ “حَظِّ الْاُنْثَيَيْنِ” کو بنایا گیا، یعنی بیٹی کا حصہ وہ اکائی ہے جس سے بیٹے کا حصہ ناپا گیا؛ عورت یہاں حساب کی بنیاد ہے، حاشیہ نہیں۔

پھر والدین کا حصہ: وَلِاَبَوَيْهِ لِكُلِّ وَاحِدٍ مِّنْهُمَا السُّدُسُ مِمَّا تَرَكَ اِنْ كَانَ لَهٗ وَلَدٌ ۚ فَاِنْ لَّمْ يَكُنْ لَّهٗ وَلَدٌ وَّوَرِثَهٗۤ اَبَوٰهُ فَلِاُمِّهِ الثُّلُثُ ۚ فَاِنْ كَانَ لَهٗۤ اِخْوَةٌ فَلِاُمِّهِ السُّدُسُ مِنْۢ بَعْدِ وَصِيَّةٍ يُّوْصِيْ بِهَاۤ اَوْ دَيْنٍ۔ “اور اگر میت کی اولاد ہو تو اس کے ماں باپ میں سے ہر ایک کے لیے ترکے کا چھٹا حصہ ہے؛ اور اگر اولاد نہ ہو اور ماں باپ ہی وارث ہوں تو ماں کا تہائی ہے؛ اور اگر اس کے بھائی بہن ہوں تو ماں کا چھٹا حصہ ہے؛ یہ سب اس وصیت کے بعد جو وہ کر جائے، یا قرض کے بعد۔” دیکھیے کہ اولاد کی موجودگی میں ماں اور باپ کا حصہ برابر ہے؛ یعنی مرد کا دگنا ہونا کوئی عام قاعدہ نہیں جو ہر جگہ چلے، وہ ذمہ داری کے ساتھ بدلتا ہے۔ ماں باپ ہی وارث ہوں تو ماں کو تہائی اور باقی باپ کو ملتا ہے، اور اگر میت کے دو یا زیادہ بھائی بہن ہوں تو ماں کا حصہ چھٹا رہ جاتا ہے۔ آیت میں وصیت کا ذکر پہلے ہے، مگر حضرت علی رضی اللہ عنہ روایت کرتے ہیں کہ حضورِ اکرم صلی اللہ علیہ وسلم نے قرض کو وصیت سے پہلے ادا کرنے کا فیصلہ فرمایا (ترمذی)؛ اور آپ صلی اللہ علیہ وسلم نے وصیت کو تہائی تک محدود فرمایا (بخاری، مسلم) اور یہ بھی فرمایا کہ وارث کے لیے وصیت نہیں (ابو داؤد، ترمذی)۔ مفسرین کہتے ہیں کہ لفظ میں وصیت کو پہلے لانا اس کی اہمیت جتانے کے لیے ہے، کیونکہ وارث اسے نظر انداز کرنے پر مائل ہوتے ہیں، جبکہ قرض کا حق دار خود مانگ لیتا ہے۔ اہلِ ذوق اس میں یہ ادب دیکھتے ہیں کہ کمزور حق کو لفظ میں آگے رکھ کر اس کی حفاظت کی گئی؛ اور حق تعالیٰ کا کلام ہمیشہ اسی کی طرف جھکتا ہے جس کا کوئی پوچھنے والا نہ ہو۔

اور اب وہ جملہ جو ساری تقسیم کی کنجی ہے: اٰبَآؤُكُمْ وَاَبْنَآؤُكُمْ لَا تَدْرُوْنَ اَيُّهُمْ اَقْرَبُ لَكُمْ نَفْعًا ۚ فَرِيْضَةً مِّنَ اللّٰهِ ۗ اِنَّ اللّٰهَ كَانَ عَلِيْمًا حَكِيْمًا۔ “تمہارے باپ اور تمہارے بیٹے، تم نہیں جانتے کہ ان میں سے کون نفع میں تم سے زیادہ قریب ہے؛ یہ اللہ کی طرف سے مقرر کیا ہوا فریضہ ہے؛ بے شک اللہ سب کچھ جاننے والا، حکمت والا ہے۔” انسان اپنی پسند سے بانٹتا تو کبھی بیٹے کو، کبھی باپ کو، کبھی کسی ایک بچے کو ترجیح دیتا اور گھر کے رشتے ٹوٹ جاتے؛ اس لیے تقسیم اس کے ہاتھ سے لے کر علیم و حکیم کے حکم میں رکھی گئی۔ بیٹے کے دگنے حصے کی حکمت علماء نے یہ بیان کی کہ شریعت نے مہر، بیوی بچوں کا نفقہ، اور بہت سی صورتوں میں ماں، بہن اور دوسرے رشتوں کی کفالت مرد کے ذمے رکھی ہے، جبکہ عورت کا حصہ پورا اس کا اپنا ہے اور اس پر کسی کا نفقہ واجب نہیں؛ پس یہ فرق ذمہ داری کا ہے، قدر و قیمت کا نہیں۔ اسی لیے کئی صورتوں میں عورت کا حصہ مرد کے برابر ہوتا ہے، اور بعض صورتوں میں زیادہ بھی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک حکمت ہر چیز کو اس کے مقام پر رکھنے کا نام ہے، اور عدل یہ ہے کہ ہر صاحبِ حق کو وہ دیا جائے جو اس کے حال کے مطابق ہو؛ ہر جگہ یکساں حصہ عدل نہیں ہوتا۔ سورۂ طٰہٰ میں یہی اصول یوں آیا: اَعْطٰى كُلَّ شَيْءٍ خَلْقَهٗ ثُمَّ هَدٰى۔ جس نے ہر شے کو اس کی ساخت دی، اسی نے ہر وارث کو اس کا حصہ دیا؛ اور “لَا تَدْرُوْنَ” بندے کو حیرت کے اس ادب پر کھڑا کرتا ہے جہاں وہ حکم کو مان لیتا ہے، خواہ اس کی پوری حکمت اس پر نہ کھلے۔

اب ازدواجی رشتے کا حصہ، جس میں دونوں طرف کا ذکر برابر تفصیل سے ہے: وَلَكُمْ نِصْفُ مَا تَرَكَ اَزْوَاجُكُمْ اِنْ لَّمْ يَكُنْ لَّهُنَّ وَلَدٌ ۚ فَاِنْ كَانَ لَهُنَّ وَلَدٌ فَلَكُمُ الرُّبُعُ مِمَّا تَرَكْنَ مِنْۢ بَعْدِ وَصِيَّةٍ يُّوْصِيْنَ بِهَاۤ اَوْ دَيْنٍ ۚ وَلَهُنَّ الرُّبُعُ مِمَّا تَرَكْتُمْ اِنْ لَّمْ يَكُنْ لَّكُمْ وَلَدٌ ۚ فَاِنْ كَانَ لَكُمْ وَلَدٌ فَلَهُنَّ الثُّمُنُ مِمَّا تَرَكْتُمْ مِّنْۢ بَعْدِ وَصِيَّةٍ تُوْصُوْنَ بِهَاۤ اَوْ دَيْنٍ۔ “اور تمہاری بیویاں جو کچھ چھوڑ جائیں اس کا آدھا تمہارا ہے اگر ان کی اولاد نہ ہو؛ اور اگر ان کی اولاد ہو تو ان کے ترکے کا چوتھائی تمہارا ہے، اس وصیت کے بعد جو وہ کر جائیں یا قرض کے بعد؛ اور تم جو کچھ چھوڑ جاؤ اس کا چوتھائی ان کا ہے اگر تمہاری اولاد نہ ہو؛ اور اگر تمہاری اولاد ہو تو ان کا آٹھواں حصہ ہے، اس وصیت کے بعد جو تم کر جاؤ یا قرض کے بعد۔” بیویاں ایک سے زیادہ ہوں تو یہی چوتھائی یا آٹھواں حصہ ان سب میں برابر بٹتا ہے۔ یہاں ایک بات خاص طور پر قابلِ توجہ ہے: “يُوْصِيْنَ بِهَاۤ” میں عورت خود وصیت کرنے والی ہے، اور “مِمَّا تَرَكْنَ” میں وہ خود ترکہ چھوڑنے والی؛ یعنی قرآن نے عورت کو جائیداد کی مستقل مالک اور اپنے مال میں تصرف کا حق رکھنے والی تسلیم کیا، اس زمانے میں جب بہت سی جگہوں پر وہ خود ترکے میں شمار ہوتی تھی۔ اہلِ ذوق کہتے ہیں کہ میاں بیوی کا ایک دوسرے کا وارث ہونا اس بات کی علامت ہے کہ نکاح محض لین دین کا معاہدہ نہیں، ایک ایسا رشتہ ہے جو موت کے بعد بھی اپنا اثر چھوڑتا ہے؛ وہی جوڑا جس کا ذکر سورت کی پہلی آیت میں “وَّخَلَقَ مِنْهَا زَوْجَهَا” کہہ کر ہوا تھا، یہاں آخری حساب میں بھی ایک دوسرے کے ساتھ ہے۔

آیت کا آخری حصہ کلالہ کے بارے میں ہے: وَاِنْ كَانَ رَجُلٌ يُّوْرَثُ كَلٰلَةً اَوِ امْرَاَةٌ وَّلَهٗۤ اَخٌ اَوْ اُخْتٌ فَلِكُلِّ وَاحِدٍ مِّنْهُمَا السُّدُسُ ۚ فَاِنْ كَانُوْۤا اَكْثَرَ مِنْ ذٰلِكَ فَهُمْ شُرَكَآءُ فِي الثُّلُثِ مِنْۢ بَعْدِ وَصِيَّةٍ يُّوْصٰى بِهَاۤ اَوْ دَيْنٍ ۙ غَيْرَ مُضَآرٍّ ۚ وَصِيَّةً مِّنَ اللّٰهِ ۗ وَاللّٰهُ عَلِيْمٌ حَلِيْمٌ۔ “اور اگر کوئی مرد یا عورت ایسا ہو جس کی میراث بٹ رہی ہو اور اس کے نہ باپ دادا ہوں نہ اولاد، اور اس کا ایک بھائی یا ایک بہن ہو تو ان میں سے ہر ایک کا چھٹا حصہ ہے؛ اور اگر وہ اس سے زیادہ ہوں تو سب ایک تہائی میں شریک ہیں؛ اس وصیت کے بعد جو کی گئی ہو، یا قرض کے بعد، بشرطیکہ کسی کو نقصان پہنچانا مقصود نہ ہو؛ یہ اللہ کی طرف سے وصیت ہے، اور اللہ جاننے والا، بردبار ہے۔” کلالہ وہ ہے جس کے پیچھے نہ باپ دادا ہوں نہ اولاد، اور اہلِ علم کا اتفاق ہے کہ یہاں ماں کی طرف سے بھائی بہن مراد ہیں؛ سگے اور باپ شریک بھائی بہنوں کا حکم اسی سورت کی آخری آیت میں آئے گا۔ ان اخیافی بھائی بہنوں میں بھائی اور بہن کا حصہ بالکل برابر ہے؛ یہ اس بات کی ایک اور دلیل ہے کہ دگنا حصہ جنس کا اصول نہیں، ذمہ داری کا حساب ہے۔ “غَيْرَ مُضَآرٍّ” نے مرنے والے کے ہاتھ بھی باندھ دیے: وہ وصیت یا قرض کے جھوٹے اقرار سے کسی وارث کو نقصان نہ پہنچائے۔ اور پچھلی آیت “عَلِيْمًا حَكِيْمًا” پر ختم ہوئی تھی، یہ “عَلِيْمٌ حَلِيْمٌ” پر؛ اہلِ معرفت کہتے ہیں کہ یہاں حلم کا ذکر اس لیے ہے کہ حد توڑنے والے کو فوراً نہیں پکڑا جاتا، اور اس مہلت کو چھوٹ نہ سمجھا جائے؛ حلم دراصل لوٹ آنے کی دعوت ہے۔

پوری تقسیم کے بعد ایک ہی لفظ میں اس کا مرتبہ بتا دیا گیا: تِلْكَ حُدُوْدُ اللّٰهِ ۗ وَمَنْ يُّطِعِ اللّٰهَ وَرَسُوْلَهٗ يُدْخِلْهُ جَنّٰتٍ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِيْنَ فِيْهَا ۗ وَذٰلِكَ الْفَوْزُ الْعَظِيْمُ۔ “یہ اللہ کی حدیں ہیں؛ اور جو اللہ اور اس کے رسول صلی اللہ علیہ وسلم کی اطاعت کرے گا، اللہ اسے ایسے باغوں میں داخل کرے گا جن کے نیچے نہریں بہتی ہیں، وہ ان میں ہمیشہ رہیں گے؛ اور یہی بڑی کامیابی ہے۔” حصوں کو حدیں کہا، کیونکہ حد وہ لکیر ہے جو ایک کے حق کو دوسرے کے حق سے جدا کرتی ہے۔ اللہ کی اطاعت کے ساتھ رسولِ اکرم صلی اللہ علیہ وسلم کی اطاعت کا ذکر اس لیے ہے کہ ان حصوں کی تفصیل، جیسے قرض کی تقدیم اور وصیت کی حد، آپ صلی اللہ علیہ وسلم ہی کے فیصلوں سے مکمل ہوئی۔ شیخِ اکبر کے مکتب میں ہر شے کی ایک حد ہے جس سے وہ پہچانی جاتی ہے، اور بندے کی سب سے بڑی حد اس کی عبدیت ہے؛ جو اپنی حد پہچان لیتا ہے وہ اپنے رب کے ادب کو پہچان لیتا ہے۔ وارث کا اپنے حصے پر راضی ہو جانا گویا اپنی عبدیت پر راضی ہو جانا ہے، اور جو یہ رضا پا لے اس کے لیے “الْفَوْزُ الْعَظِيْمُ” کا دروازہ یہیں سے کھلنے لگتا ہے۔ ایک لطیف نکتہ یہ بھی ہے کہ جنت والوں کے لیے “خٰلِدِيْنَ” جمع کا صیغہ آیا: وہ ایک دوسرے کی رفاقت میں ہوں گے، جیسے دنیا میں انہوں نے ایک دوسرے کا حق پہچانا تھا۔ سورۂ البقرہ میں بھی گھریلو احکام کے بعد یہی تنبیہ آئی تھی: تِلْكَ حُدُوْدُ اللّٰهِ فَلَا تَعْتَدُوْهَا؛ کیونکہ گھر کے حق سب سے پہلے ٹوٹنے والے حق ہیں۔

اور اس کے مقابل: وَمَنْ يَّعْصِ اللّٰهَ وَرَسُوْلَهٗ وَيَتَعَدَّ حُدُوْدَهٗ يُدْخِلْهُ نَارًا خَالِدًا فِيْهَا ۚ وَلَهٗ عَذَابٌ مُّهِيْنٌ۔ “اور جو اللہ اور اس کے رسول صلی اللہ علیہ وسلم کی نافرمانی کرے گا اور اس کی حدوں سے آگے بڑھے گا، اللہ اسے آگ میں داخل کرے گا جس میں وہ ہمیشہ رہے گا، اور اس کے لیے رسوا کرنے والا عذاب ہے۔” اہلِ سنت کے نزدیک ہمیشگی کی یہ وعید اس کے لیے ہے جو ان حدوں کا انکار کرے، اور گناہ گار مومن کے لیے یہ طویل سزا کی سخت تنبیہ ہے۔ یہاں “خَالِدًا” واحد آیا؛ اہلِ تفسیر نے اس میں یہ اشارہ دیکھا کہ جس نے دوسروں کا حق کاٹ کر سب کچھ اپنے لیے سمیٹا وہ وہاں بھی اکیلا ہے؛ اس نے رشتے توڑے تھے، اب کوئی رشتہ اس کے ساتھ نہیں۔ اور عذاب “مُّهِيْنٌ” ہے، یعنی ذلت والا، کیونکہ اس کا جرم اپنی حد سے بڑھ کر بڑا بننا تھا۔ یہاں وہ اصول یاد رکھنا ضروری ہے جو اس سلسلے کی بنیاد ہے: حق تعالیٰ کسی کو زبردستی اس انجام تک نہیں لے جاتا۔ “يَّعْصِ” اور “وَيَتَعَدَّ” دونوں فعل بندے کے ہیں؛ حدیں سامنے تھیں، وارثوں کے حق واضح تھے، اور جس نے انہیں پھلانگا اس نے یہ رخ خود چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہوتا ہے: حق اسی رخ کو وجود دیتا ہے جو بندے نے اپنے لیے چن لیا، اور اس کے انتخاب کی عزت رکھتا ہے۔ پھر بھی حدیثِ قدسی کے مطابق اس کی رحمت اس کے غضب پر سبقت رکھتی ہے، اور توبہ کا دروازہ آخری سانس تک کھلا ہے؛ جو آج کسی وارث کا کاٹا ہوا حق لوٹا دے، وہ اسی لمحے حد کے اندر واپس آ جاتا ہے۔
