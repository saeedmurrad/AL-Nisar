یہ سورت عورتوں کے نام سے موسوم ہے، مگر اس کا آغاز نہ عورت سے ہوتا ہے نہ مرد سے، انسان سے ہوتا ہے؛ گویا پہلے وہ بنیاد رکھی گئی جس پر آگے کے سارے حقوق کھڑے ہوں گے: يٰۤاَيُّهَا النَّاسُ اتَّقُوْا رَبَّكُمُ الَّذِيْ خَلَقَكُمْ مِّنْ نَّفْسٍ وَّاحِدَةٍ وَّخَلَقَ مِنْهَا زَوْجَهَا وَبَثَّ مِنْهُمَا رِجَالًا كَثِيْرًا وَّنِسَآءً۔ “اے لوگو! اپنے رب سے ڈرو جس نے تمہیں ایک جان سے پیدا کیا، اور اسی سے اس کا جوڑا پیدا کیا، اور ان دونوں سے بہت سے مرد اور عورتیں پھیلا دیں۔” خطاب “النَّاسُ” سے ہے، کسی قوم، نسل یا جنس سے نہیں؛ اور تقویٰ کا پہلا حوالہ “رَبَّكُمُ” ہے، یعنی پرورش کرنے والا رب، تاکہ معلوم ہو کہ آگے جو حدیں بیان ہوں گی وہ پرورش کی حدیں ہیں، جبر کی نہیں۔ ظاہر معنی یہ ہے کہ سب انسان حضرت آدم علیہ السلام کی اولاد ہیں، اس لیے کسی کو پیدائش کی بنیاد پر دوسرے پر فخر کا حق نہیں؛ یہی بات سورۂ الحجرات میں اِنَّا خَلَقْنٰكُمْ مِّنْ ذَكَرٍ وَّاُنْثٰى کہہ کر دہرائی گئی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں “نَفْسٍ وَّاحِدَةٍ” صرف نسب کی خبر نہیں، حقیقت کی خبر ہے: تمام افراد ایک ہی حقیقتِ انسانیہ کے مظاہر ہیں، جیسے ایک چہرہ بہت سے آئینوں میں۔ حکمتِ الٰہیہ میں، جس کا مفہوم یہ ہے کہ حضرت آدم علیہ السلام عالم کے آئینے کی جلا اور اس صورت کی روح ہیں، اسی جامعیت کی طرف اشارہ ہے۔ پس جو کسی انسان کی تحقیر کرتا ہے وہ دراصل اسی آئینے کو دھندلا کرتا ہے جس میں اس کا اپنا چہرہ بھی ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

“وَّخَلَقَ مِنْهَا زَوْجَهَا” میں ایک نازک راز ہے جس پر اہلِ معرفت ٹھہرتے ہیں۔ زوج کا معنی جوڑا ہے، یعنی وہ جو دوسرے کے بغیر ادھورا ہو؛ اسی لیے یہ لفظ مرد اور عورت دونوں کے لیے ایک ہی ہے، اور کسی ایک کو کم تر نہیں ٹھہراتا۔ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ دنیا میں سے آپ کو عورتیں اور خوشبو محبوب کی گئیں، اور آپ کی آنکھوں کی ٹھنڈک نماز میں رکھی گئی (نسائی)۔ شیخِ اکبر حکمتِ فردیہ میں اسی حدیث پر کلام کرتے ہیں، جس کا مفہوم یہ ہے کہ عورت مرد سے نکالی گئی، اس لیے مرد کا اس کی طرف شوق ایسا ہے جیسے کوئی شے اپنی ہی ذات کی طرف مشتاق ہو، اور عورت کا شوق ایسا ہے جیسے مسافر اپنے وطن کی طرف؛ اور یہ کہ عورت میں حق کا مشاہدہ کامل ترین مشاہدوں میں سے ہے۔ اس مکتب میں ایک اور لطیف تقسیم بھی ملتی ہے: حضرت آدم علیہ السلام نہ ماں سے ہوئے نہ باپ سے، حضرت حوا علیہا السلام مرد سے ہوئیں، اور حضرت عیسیٰ علیہ السلام صرف ایک پاکیزہ خاتون سے؛ یوں پیدائش کے دروازے میں عورت کو وہ مقام بھی ملا جو مرد کو نہیں ملا۔ پس جس سورت میں عورت کے حقوق آنے ہیں، اس کی پہلی ہی آیت نے بتا دیا کہ وہ مرد کی ملکیت نہیں، اس کا جوڑا ہے؛ اور جوڑے کا حق انصاف اور محبت ہے، غلبہ نہیں۔

پھر آیت تقویٰ کو دوسری بار دہراتی ہے، اور اب رب کی جگہ اسمِ جامع آتا ہے: وَاتَّقُوا اللّٰهَ الَّذِيْ تَسَآءَلُوْنَ بِهٖ وَالْاَرْحَامَ ۗ اِنَّ اللّٰهَ كَانَ عَلَيْكُمْ رَقِيْبًا۔ “اور اللہ سے ڈرو جس کا واسطہ دے کر تم ایک دوسرے سے مانگتے ہو، اور رشتوں کا بھی لحاظ رکھو؛ بے شک اللہ تم پر نگہبان ہے۔” عرب ایک دوسرے سے کہتے کہ میں تجھ سے اللہ کا اور رشتے کا واسطہ دے کر مانگتا ہوں؛ قرآن نے اسی عادت کو پکڑ کر بتایا کہ جس نام کا واسطہ دیتے ہو اسی کی خاطر رشتوں کو بھی نبھاؤ۔ اللہ کے نام کے ساتھ “الْاَرْحَامَ” کا جڑنا معمولی بات نہیں۔ حدیثِ قدسی میں آیا ہے، جس کا مفہوم یہ ہے کہ اللہ تعالیٰ فرماتا ہے: میں رحمٰن ہوں، میں نے رحم کو پیدا کیا اور اس کا نام اپنے نام سے نکالا؛ جو اسے جوڑے گا میں اسے جوڑوں گا، اور جو اسے کاٹے گا میں اسے کاٹ دوں گا (ابو داؤد، ترمذی)۔ شیخِ اکبر کے نزدیک ہر وجود نفَسِ رحمانی کی ایک سانس ہے، اور رشتہ داری اسی رحمت کی وہ صورت ہے جو انسان کو اپنے گھر کے اندر ملتی ہے؛ اس لیے رشتہ کاٹنا رحمت کے بہاؤ میں گرہ ڈالنا ہے۔ آخر میں “رَقِيْبًا” آیا، یعنی وہ جو ہر لمحہ دیکھ رہا ہے۔ اہلِ ذوق کہتے ہیں کہ مراقبہ اسی نام کا عکس ہے؛ جب بندہ جان لیتا ہے کہ وہ دیکھا جا رہا ہے تو اس کے اندر بھی ایک نگاہ جاگ اٹھتی ہے، اور کمزور کے حق میں اس کی پہلی احتیاط اسی نگاہ کے شعور سے پیدا ہوتی ہے۔

نسب کی وحدت کا پہلا عملی نتیجہ یتیم ہے، کیونکہ وہی سب سے پہلے بے سہارا ہوتا ہے: وَاٰتُوا الْيَتٰمٰۤى اَمْوَالَهُمْ وَلَا تَتَبَدَّلُوا الْخَبِيْثَ بِالطَّيِّبِ ۖ وَلَا تَاْكُلُوْۤا اَمْوَالَهُمْ اِلٰۤى اَمْوَالِكُمْ ۗ اِنَّهٗ كَانَ حُوْبًا كَبِيْرًا۔ “اور یتیموں کو ان کے مال دے دو، اور پاکیزہ کے بدلے ناپاک نہ لو، اور ان کے مال اپنے مالوں کے ساتھ ملا کر نہ کھاؤ؛ یقیناً یہ بہت بڑا گناہ ہے۔” بعض سرپرست یتیم کا عمدہ مال لے کر اپنا گھٹیا مال اس کی جگہ رکھ دیتے اور سمجھتے کہ گنتی پوری ہے؛ قرآن نے اسے “الْخَبِيْثَ بِالطَّيِّبِ” کا سودا کہا، کیونکہ حلال میں حرام کی ملاوٹ پورے مال کو بدل دیتی ہے۔ “اِلٰۤى اَمْوَالِكُمْ” کا اشارہ یہ ہے کہ اکثر خیانت لوٹ مار سے نہیں، ملا دینے سے ہوتی ہے: حساب الگ نہ رکھا تو یتیم کا مال آہستہ آہستہ اپنا محسوس ہونے لگا۔ حضورِ اکرم صلی اللہ علیہ وسلم خود یتیم تھے، اور سورۂ الضحیٰ میں حق تعالیٰ نے آپ سے فرمایا: اَلَمْ يَجِدْكَ يَتِيْمًا فَاٰوٰى؛ اور آپ صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ میں اور یتیم کی کفالت کرنے والا جنت میں ان دو انگلیوں کی طرح ہوں گے (بخاری)۔ اہلِ معرفت کہتے ہیں کہ یتیم ہمارے اپنے فقر کا آئینہ ہے: ہر مخلوق حقیقت میں بے سہارا ہے اور اس کا اصل والی صرف حق ہے۔ جو یتیم کا مال اسے لوٹاتا ہے وہ اسی ولایت کا نائب بنتا ہے، اور جو اسے کھا جاتا ہے وہ اپنے ہی فقر کو بھول جاتا ہے۔

اسی سیاق میں وہ آیت آتی ہے جسے اکثر سیاق سے کاٹ کر پڑھا جاتا ہے: وَاِنْ خِفْتُمْ اَلَّا تُقْسِطُوْا فِي الْيَتٰمٰى فَانْكِحُوْا مَا طَابَ لَكُمْ مِّنَ النِّسَآءِ مَثْنٰى وَثُلٰثَ وَرُبٰعَ۔ “اور اگر تمہیں اندیشہ ہو کہ یتیم لڑکیوں کے بارے میں انصاف نہ کر سکو گے تو جو عورتیں تمہیں پسند ہوں ان میں سے دو دو، تین تین، چار چار سے نکاح کر لو۔” ام المؤمنین حضرت عائشہ رضی اللہ عنہا نے اس کا پس منظر بیان فرمایا (بخاری): ایک یتیم لڑکی کسی سرپرست کی پرورش میں ہوتی، اس کے مال اور حسن کی وجہ سے وہ اس سے نکاح کرنا چاہتا مگر اسے اس جیسی عورتوں کے برابر مہر نہ دیتا؛ انہیں روکا گیا کہ یا تو اس کے ساتھ پورا انصاف کرو، ورنہ دوسری عورتوں سے نکاح کرو۔ پس آیت کا مرکز یتیم بچی کی حفاظت ہے، تعداد کا شوق نہیں۔ اس دور میں بے حد و حساب نکاح رائج تھے؛ قرآن نے حد مقرر کی، اسے چار پر روکا، اور ساتھ ہی ایک کڑی شرط لگا دی جو اگلے ٹکڑے میں ہے۔ “مَا طَابَ لَكُمْ” میں بھی ایک ادب ہے: نکاح پاکیزہ رغبت اور دونوں کی رضا سے ہو، دباؤ اور سودے سے نہیں۔ شیخِ اکبر کے مکتب میں عدد کا ہر مرتبہ وحدت ہی کی تکرار ہے؛ دو، تین اور چار، سب ایک ہی کے ظہور ہیں۔ اہلِ ذوق اسی سے یہ نکتہ لیتے ہیں کہ کثرت جائز ہو تب بھی اس کا میزان وحدت ہے، یعنی ہر ایک کے ساتھ ویسا ہی انصاف جیسے وہ اکیلی ہو؛ اور جو یہ میزان نہ سنبھال سکے اس کے لیے کثرت کی اجازت باقی نہیں رہتی۔

اب وہ شرط جس پر اجازت کھڑی ہے: فَاِنْ خِفْتُمْ اَلَّا تَعْدِلُوْا فَوَاحِدَةً اَوْ مَا مَلَكَتْ اَيْمَانُكُمْ ۗ ذٰلِكَ اَدْنٰۤى اَلَّا تَعُوْلُوْا۔ “پھر اگر اندیشہ ہو کہ عدل نہ کر سکو گے تو ایک ہی پر بس کرو، یا جو تمہاری ملکیت میں ہوں؛ یہ اس بات کے زیادہ قریب ہے کہ تم ناانصافی کی طرف نہ جھکو۔” فقہاء نے یہاں عدل کا مطلب ان امور میں برابری بتایا ہے جو انسان کے اختیار میں ہیں: نان نفقہ، رہائش، باری اور برتاؤ۔ دل کا میلان اختیار میں نہیں، اسی لیے اسی سورت میں آگے فرمایا: وَلَنْ تَسْتَطِيْعُوْۤا اَنْ تَعْدِلُوْا بَيْنَ النِّسَآءِ وَلَوْ حَرَصْتُمْ؛ اور ساتھ ہی حکم دیا کہ ایک طرف پوری طرح نہ جھک جاؤ۔ حضورِ اکرم صلی اللہ علیہ وسلم اپنی ازواجِ مطہرات میں باری برابر رکھتے اور دعا فرماتے، جس کا مفہوم یہ ہے کہ اے اللہ! یہ میری تقسیم ہے اس میں جس کا میں مالک ہوں، پس مجھے اس پر ملامت نہ فرما جس کا تو مالک ہے اور میں نہیں (ابو داؤد)۔ اور آپ صلی اللہ علیہ وسلم نے خبر دی کہ جو دو بیویوں میں سے ایک کی طرف جھک جائے وہ قیامت کے دن اس حال میں آئے گا کہ اس کا ایک پہلو جھکا ہوا ہو (ابو داؤد، ترمذی)۔ “مَا مَلَكَتْ اَيْمَانُكُمْ” اس زمانے کے ایک تاریخی ادارے کی طرف اشارہ ہے جسے شریعت نے آزادی کی بے شمار ترغیبات اور کفاروں کے ذریعے بتدریج سمیٹا۔ اہلِ معرفت کے نزدیک “تَعُوْلُوْا” کا جھکاؤ صرف گھر کا ظلم نہیں، دل کا جھکاؤ بھی ہے؛ جو ایک کے ساتھ عدل نہ کر سکے وہ بہتوں کے ساتھ کیسے کرے گا، اور عدل کی پہلی درسگاہ اپنا ہی گھر ہے۔

اب عورت کا وہ حق جو نکاح کے ساتھ ہی اس کا ہو جاتا ہے: وَاٰتُوا النِّسَآءَ صَدُقٰتِهِنَّ نِحْلَةً ۗ فَاِنْ طِبْنَ لَكُمْ عَنْ شَيْءٍ مِّنْهُ نَفْسًا فَكُلُوْهُ هَنِيْٓئًا مَّرِيْٓئًا۔ “اور عورتوں کو ان کے مہر خوش دلی سے دو؛ پھر اگر وہ اپنی خوشی سے اس میں سے کچھ تمہیں چھوڑ دیں تو اسے مزے سے، خوشگواری کے ساتھ کھاؤ۔” مہر کا نام یہاں “صَدُقٰتِهِنَّ” رکھا گیا، جو صدق سے ہے؛ گویا مہر مرد کے سچے ارادے کی گواہی ہے، عورت کی قیمت نہیں۔ اور “نِحْلَةً” کے معنی ہیں ایسا عطیہ جو بدلے کی نیت کے بغیر، دل کی خوشی سے دیا جائے۔ جاہلیت میں بعض ولی مہر خود رکھ لیتے تھے؛ قرآن نے ضمیر عورتوں کی طرف لوٹا کر بتا دیا کہ یہ صرف اس کا حق ہے، نہ باپ کا نہ شوہر کا۔ اس میں سے کچھ واپس لینا بھی صرف اس کی آزاد رضا سے جائز ہے، اور “فَاِنْ طِبْنَ لَكُمْ” کہہ کر اس رضا کو دل کی پوری خوشی سے مشروط کیا گیا؛ شرم، دباؤ یا ڈر سے لی ہوئی رضا رضا نہیں۔ شیخِ اکبر کے نزدیک حق تعالیٰ کی ساری عطا اسی قسم کی ہے: وجود ہمیں کسی بدلے میں نہیں ملا، محض فضل سے ملا، اور فیضِ اقدس نے کسی استحقاق کے بغیر اعیان کو ان کی استعداد عطا کی۔ جو بندہ اس عطا کو پہچان لیتا ہے وہ اپنی دی ہوئی چیز پر احسان نہیں جتاتا؛ اسے معلوم ہے کہ وہ صرف واسطہ ہے، اور دینے والا اصل میں کوئی اور ہے۔

پھر مال کی حفاظت کا دوسرا رخ: وَلَا تُؤْتُوا السُّفَهَآءَ اَمْوَالَكُمُ الَّتِيْ جَعَلَ اللّٰهُ لَكُمْ قِيٰمًا وَّارْزُقُوْهُمْ فِيْهَا وَاكْسُوْهُمْ وَقُوْلُوْا لَهُمْ قَوْلًا مَّعْرُوْفًا۔ “اور ناسمجھوں کو اپنے وہ مال نہ دو جنہیں اللہ نے تمہارے لیے گزران کا سہارا بنایا ہے، ہاں ان میں سے انہیں کھلاؤ پہناؤ اور ان سے بھلی بات کہو۔” سفیہ وہ ہے جو مال سنبھالنے کی سمجھ نہ رکھتا ہو، خواہ کم سن ہو یا بڑا، مرد ہو یا عورت؛ یہ کسی جنس کا وصف نہیں، ایک حال کا نام ہے۔ مال کو “قِيٰمًا” کہا، یعنی جس سے زندگی کھڑی رہتی ہے؛ اسے ضائع ہونے دینا اس ستون کو گرانا ہے۔ مگر حفاظت کے ساتھ عزت بھی باقی رکھی: کھلاؤ، پہناؤ، اور بات نرمی سے کرو۔ پھر یتیموں کی تربیت کا طریقہ: وَابْتَلُوا الْيَتٰمٰى حَتّٰۤى اِذَا بَلَغُوا النِّكَاحَ ۚ فَاِنْ اٰنَسْتُمْ مِّنْهُمْ رُشْدًا فَادْفَعُوْۤا اِلَيْهِمْ اَمْوَالَهُمْ۔ “اور یتیموں کو آزماتے رہو یہاں تک کہ وہ نکاح کی عمر کو پہنچ جائیں؛ پھر اگر ان میں سمجھ داری دیکھو تو ان کے مال ان کے حوالے کر دو۔” “اٰنَسْتُمْ” وہی لفظ ہے جو سورۂ القصص میں حضرت موسیٰ علیہ السلام کے لیے آیا: اٰنَسَ مِنْ جَانِبِ الطُّوْرِ نَارًا، یعنی دور سے کسی روشنی کو پہچان لینا اور اس سے مانوس ہو جانا۔ اہلِ تاویل کہتے ہیں کہ ہر سالک کا مربی اسی طرح انتظار کرتا ہے کہ کب اس کے اندر رشد کی روشنی دکھائی دے، تب امانت اس کے سپرد کرتا ہے؛ پہلے دے تو امانت ضائع ہو، بعد تک روکے تو ظلم ہو۔

آیت اسی امانت داری کی باریکیوں تک جاتی ہے: وَلَا تَاْكُلُوْهَاۤ اِسْرَافًا وَّبِدَارًا اَنْ يَّكْبَرُوْا ۗ وَمَنْ كَانَ غَنِيًّا فَلْيَسْتَعْفِفْ ۚ وَمَنْ كَانَ فَقِيْرًا فَلْيَاْكُلْ بِالْمَعْرُوْفِ ۚ فَاِذَا دَفَعْتُمْ اِلَيْهِمْ اَمْوَالَهُمْ فَاَشْهِدُوْا عَلَيْهِمْ ۗ وَكَفٰى بِاللّٰهِ حَسِيْبًا۔ “اور اس ڈر سے کہ وہ بڑے ہو جائیں گے، ان کا مال فضول خرچی اور جلد بازی سے نہ کھا جاؤ؛ اور جو سرپرست مالدار ہو وہ بچا رہے، اور جو محتاج ہو وہ دستور کے مطابق کھائے؛ پھر جب ان کے مال ان کے حوالے کرو تو ان پر گواہ بنا لو، اور اللہ حساب لینے کو کافی ہے۔” “بِدَارًا اَنْ يَّكْبَرُوْا” نفس کی ایک چھپی چال کو بے نقاب کرتا ہے: جب امانت لوٹانے کا وقت قریب آتا ہے تو اسے جلدی جلدی خرچ کر ڈالنے کی خواہش جاگتی ہے۔ مالدار سرپرست سے “فَلْيَسْتَعْفِفْ” کہا، یعنی عفت اختیار کرے؛ یہ صرف قانونی حد نہیں، کردار کی بلندی ہے۔ محتاج کو اجازت دی مگر “بِالْمَعْرُوْفِ” کی قید کے ساتھ، یعنی محنت کے مطابق، بے جا نہیں۔ گواہوں کا حکم ظاہری انتظام ہے، مگر آخری جملہ “وَكَفٰى بِاللّٰهِ حَسِيْبًا” بتاتا ہے کہ اصل گواہ وہ ہے جس سے کوئی کھاتہ چھپا نہیں۔ شیخِ اکبر کے نزدیک ہمارے پاس جو کچھ ہے سب امانت ہے، حتیٰ کہ ہمارا وجود بھی؛ وجود حق کا ہے اور ہم اس کے مظاہر ہیں، یوں حقیقت میں ہم سب حق کے مال کے سرپرست ہیں۔ جو اس شعور کے ساتھ جیتا ہے وہ ہر دن اپنا حساب خود کرتا ہے، اور اس کے لیے “حَسِيْبًا” ڈراوا نہیں، سکون ہے۔

اب وہ اعلان جس نے جاہلیت کا ایک پورا دستور الٹ دیا: لِلرِّجَالِ نَصِيْبٌ مِّمَّا تَرَكَ الْوَالِدٰنِ وَالْاَقْرَبُوْنَ ۚ وَلِلنِّسَآءِ نَصِيْبٌ مِّمَّا تَرَكَ الْوَالِدٰنِ وَالْاَقْرَبُوْنَ مِمَّا قَلَّ مِنْهُ اَوْ كَثُرَ ۚ نَصِيْبًا مَّفْرُوْضًا۔ “مردوں کے لیے اس میں حصہ ہے جو ماں باپ اور قریبی رشتہ دار چھوڑ جائیں، اور عورتوں کے لیے بھی اس میں حصہ ہے جو ماں باپ اور قریبی رشتہ دار چھوڑ جائیں؛ تھوڑا ہو یا بہت، یہ مقرر کیا ہوا حصہ ہے۔” عرب میں میراث عموماً انہی مردوں کو ملتی جو تلوار اٹھا سکتے تھے؛ عورتیں اور کم سن بچے محروم رہتے۔ قرآن چاہتا تو ایک جملے میں سب کو جمع کر دیتا، مگر اس نے عورتوں کا ذکر الگ اور پورے الفاظ کے ساتھ دہرایا؛ گویا ان کا حق کسی کے طفیل نہیں، اپنی جگہ مستقل ہے۔ “مِمَّا قَلَّ مِنْهُ اَوْ كَثُرَ” نے یہ دروازہ بھی بند کیا کہ تھوڑی چیز کو حقیر جان کر عورت کو ٹال دیا جائے؛ اور “مَّفْرُوْضًا” کہہ کر واضح کیا کہ یہ کسی کا احسان نہیں، اللہ کا مقرر کیا ہوا حق ہے۔ پھر تقسیم کے موقع کا ادب: وَاِذَا حَضَرَ الْقِسْمَةَ اُولُوا الْقُرْبٰى وَالْيَتٰمٰى وَالْمَسٰكِيْنُ فَارْزُقُوْهُمْ مِّنْهُ وَقُوْلُوْا لَهُمْ قَوْلًا مَّعْرُوْفًا۔ “اور جب تقسیم کے وقت وہ رشتہ دار جو وارث نہیں، اور یتیم اور مسکین آ جائیں تو انہیں بھی اس میں سے کچھ دو اور ان سے بھلی بات کہو۔” اہلِ معرفت کہتے ہیں کہ جو آنکھ کسی محروم کو سامنے دیکھ کر بھی صرف حساب دیکھے، وہ رزق کو اپنا سمجھ بیٹھی ہے؛ اور جو ہاتھ کچھ دے دے، وہ پہچانتا ہے کہ اس محروم کی حاضری بھی رزاق کی طرف سے ایک پیغام ہے، اور قول معروف اسی پیغام کا ادب۔

پھر ایک ایسا تصور جو دل کو دوسرے کی جگہ کھڑا کر دیتا ہے: وَلْيَخْشَ الَّذِيْنَ لَوْ تَرَكُوْا مِنْ خَلْفِهِمْ ذُرِّيَّةً ضِعٰفًا خَافُوْا عَلَيْهِمْ ۖ فَلْيَتَّقُوا اللّٰهَ وَلْيَقُوْلُوْا قَوْلًا سَدِيْدًا۔ “اور ان لوگوں کو ڈرنا چاہیے جو اگر اپنے پیچھے کمزور اولاد چھوڑ جاتے تو انہیں ان کی فکر ہوتی؛ پس وہ اللہ سے ڈریں اور سیدھی بات کہیں۔” یتیم کے ساتھ وہی کرو جو تم چاہتے ہو کہ تمہارے بعد تمہارے بچوں کے ساتھ کیا جائے۔ اہلِ ذوق کہتے ہیں کہ یہ انصاف کی سب سے گہری تعلیم ہے، یعنی اپنے آپ کو دوسرے کی صورت میں دیکھنا؛ اور یہ وہی نفسِ واحدہ ہے جس سے رکوع شروع ہوا تھا۔ “قَوْلًا سَدِيْدًا” وہ بات ہے جو نشانے پر بیٹھے، نہ ٹیڑھی ہو نہ بے جا نرم۔ پھر رکوع ایک ہلا دینے والی حقیقت پر ختم ہوتا ہے: اِنَّ الَّذِيْنَ يَاْكُلُوْنَ اَمْوَالَ الْيَتٰمٰى ظُلْمًا اِنَّمَا يَاْكُلُوْنَ فِيْ بُطُوْنِهِمْ نَارًا ۗ وَسَيَصْلَوْنَ سَعِيْرًا۔ “بے شک جو لوگ ظلم سے یتیموں کا مال کھاتے ہیں وہ اپنے پیٹوں میں آگ ہی بھرتے ہیں، اور وہ عنقریب بھڑکتی آگ میں جائیں گے۔” آیت یہ نہیں کہتی کہ وہ کل آگ کھائیں گے؛ کہتی ہے کہ وہ ابھی آگ کھا رہے ہیں۔ شیخِ اکبر کے نزدیک آخرت کی صورتیں اعمال کی حقیقتیں ہیں جو یہاں پردے میں ہیں اور وہاں ظاہر ہوں گی؛ لقمہ یہاں میٹھا دکھائی دیتا ہے، اس کی اصل صورت آگ ہے۔ جس دن پردہ اٹھے گا، کوئی نئی سزا نہیں گھڑی جائے گی؛ وہی سامنے آئے گا جو اس نے خود چنا اور کھایا تھا۔ اور یہی آیت امید کا دروازہ بھی ہے: جو آج لوٹا دے، وہ آج ہی اس آگ کو اپنے اندر بجھا دیتا ہے۔
