اَلَمْ تَرَ اِلَى الَّذِيْنَ قِيْلَ لَهُمْ كُفُّوْۤا اَيْدِيَكُمْ وَاَقِيْمُوا الصَّلٰوةَ وَاٰتُوا الزَّكٰوةَ ۚ فَلَمَّا كُتِبَ عَلَيْهِمُ الْقِتَالُ اِذَا فَرِيْقٌ مِّنْهُمْ يَخْشَوْنَ النَّاسَ كَخَشْيَةِ اللّٰهِ اَوْ اَشَدَّ خَشْيَةً۔ “کیا تم نے انہیں نہیں دیکھا جن سے کہا گیا تھا کہ اپنے ہاتھ روکے رکھو، نماز قائم کرو اور زکوٰۃ دو؛ پھر جب ان پر قتال فرض ہوا تو ان میں سے ایک گروہ لوگوں سے ایسے ڈرنے لگا جیسے اللہ سے ڈرا جاتا ہے، یا اس سے بھی زیادہ۔” مکہ مکرمہ کے برس صبر کے برس تھے۔ امام نسائی رحمۃ اللہ علیہ کی روایت کا مفہوم ہے کہ حضرت عبد الرحمٰن بن عوف رضی اللہ عنہ اور ان کے کچھ ساتھیوں نے عرض کیا کہ ایمان کے بعد ہم کمزور اور بے عزت ہو گئے ہیں، تو حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ مجھے درگزر کا حکم ملا ہے، اس لیے لڑائی نہ کرو۔ پھر مدینہ منورہ میں جب ایک باقاعدہ جماعت اور نظم قائم ہو گیا تب دفاع کا قتال فرض ہوا۔ اس ترتیب میں خود ایک سبق ہے: قتال جذبات کا ردِ عمل نہیں، اللہ کے حکم اور نظمِ جماعت کا معاملہ ہے۔ جس گروہ کے خوف کا ذکر ہے، اس کے بارے میں مفسرین کے اقوال ہیں: بعض نے اسے منافقین پر محمول کیا، اور جنہوں نے بعض اہلِ ایمان کی طرف نسبت کی انہوں نے اسے وہ طبعی خوف کہا جو ایمان کے منافی نہیں۔ اہلِ معرفت اسے اپنے اندر دیکھتے ہیں: مخلوق کا خوف جب خالق کے خوف کے برابر ہو جائے تو یہ وہ پوشیدہ شرک ہے جس سے کوئی دل اپنی کوشش سے پوری طرح خالی نہیں ہوتا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں سالک کی پہلی منزل یہی ہاتھ روکنا ہے، یعنی نفس کے ردِ عمل کو تھام لینا، اور اس کے بعد ہی بڑے مجاہدے کا حکم آتا ہے؛ جس نے ہاتھ روکنا نہیں سیکھا اس کا ہاتھ اٹھانا بھی خواہش کا ہوتا ہے، حکم کا نہیں۔

آیت آگے ان کی بات نقل کرتی اور جواب دیتی ہے: وَقَالُوْا رَبَّنَا لِمَ كَتَبْتَ عَلَيْنَا الْقِتَالَ ۚ لَوْلَاۤ اَخَّرْتَنَاۤ اِلٰۤى اَجَلٍ قَرِيْبٍ ۗ قُلْ مَتَاعُ الدُّنْيَا قَلِيْلٌ ۚ وَالْاٰخِرَةُ خَيْرٌ لِّمَنِ اتَّقٰى ۚ وَلَا تُظْلَمُوْنَ فَتِيْلًا۔ “اور کہنے لگے: اے ہمارے رب! تو نے ہم پر قتال کیوں فرض کر دیا، تھوڑی سی مدت اور ہمیں مہلت کیوں نہ دی؟ کہہ دو: دنیا کا سامان تھوڑا ہے، اور آخرت اس کے لیے بہتر ہے جو تقویٰ اختیار کرے، اور تم پر دھاگے برابر بھی ظلم نہ ہو گا۔” پھر وہ جملہ جو ہر مہلت مانگنے والے کو جگا دیتا ہے: اَيْنَمَا تَكُوْنُوْا يُدْرِكْكُّمُ الْمَوْتُ وَلَوْ كُنْتُمْ فِيْ بُرُوْجٍ مُّشَيَّدَةٍ۔ “تم جہاں کہیں بھی ہو موت تمہیں آ پکڑے گی، خواہ تم مضبوط اور بلند قلعوں میں ہو۔” فتیل کھجور کی گٹھلی کے شگاف کا باریک دھاگا ہے؛ یعنی جو کچھ تم نے اللہ کے لیے دیا اس کا ایک تار بھی ضائع نہ ہو گا۔ مہلت مانگنے والا دراصل موت سے نہیں، ملاقات سے گھبرا رہا ہوتا ہے، اور جواب نے بتا دیا کہ مہلت اسے موت سے بچا نہیں سکتی۔ اہلِ معرفت کہتے ہیں کہ بلند قلعے صرف پتھر کے نہیں ہوتے؛ نفس بھی اپنے گرد مال، منصب، دلیل اور مصروفیت کے قلعے بناتا ہے تاکہ حقیقت اس تک نہ پہنچے، مگر يُدْرِكْكُّمُ کا لفظ بتاتا ہے کہ موت پیچھا کرتی ہے اور جا لیتی ہے۔ شیخِ اکبر کے نزدیک کائنات ہر لمحہ مٹتی اور اپنی جیسی نئی صورت میں پیدا ہوتی ہے، جسے تجددِ امثال کہتے ہیں؛ گویا فنا کا ایک لطیف ذائقہ ہر سانس میں ہے، اور بڑی موت اسی سلسلے کا آخری اور کھلا پردہ ہے۔ اس لیے عارف اختیاری موت کا راستہ چنتا ہے، یعنی نفس کے دعوے کا مٹ جانا، تاکہ طبعی موت جب آئے تو اسے ملاقات کی صورت میں ملے۔

آیت کا اگلا حصہ وہ الجھن بیان کرتا ہے جو انسان کو ہمیشہ سے گھیرتی رہی: وَاِنْ تُصِبْهُمْ حَسَنَةٌ يَّقُوْلُوْا هٰذِهٖ مِنْ عِنْدِ اللّٰهِ ۚ وَاِنْ تُصِبْهُمْ سَيِّئَةٌ يَّقُوْلُوْا هٰذِهٖ مِنْ عِنْدِكَ ۗ قُلْ كُلٌّ مِّنْ عِنْدِ اللّٰهِ ۗ فَمَالِ هٰۤؤُلَآءِ الْقَوْمِ لَا يَكَادُوْنَ يَفْقَهُوْنَ حَدِيْثًا۔ “اور اگر انہیں کوئی بھلائی پہنچے تو کہتے ہیں: یہ اللہ کی طرف سے ہے، اور اگر کوئی برائی پہنچے تو کہتے ہیں: یہ تمہاری وجہ سے ہے۔ کہہ دو: سب کچھ اللہ کی طرف سے ہے۔ ان لوگوں کو کیا ہو گیا کہ کوئی بات سمجھنے کے قریب بھی نہیں آتے۔” مدینہ منورہ میں منافقین اور بعض مخالفین کا یہ طریقہ تھا کہ فصل اچھی ہو یا خوشحالی آئے تو اسے اللہ کی عطا کہتے، اور قحط یا جنگ کا نقصان ہو تو اسے حضورِ اکرم صلی اللہ علیہ وسلم کی تشریف آوری سے جوڑ کر بدشگونی ٹھہراتے۔ یہ پرانی بیماری ہے؛ فرعون کی قوم نے یہی حضرت موسیٰ علیہ السلام کے ساتھ کیا تھا: وَاِنْ تُصِبْهُمْ سَيِّئَةٌ يَّطَّيَّرُوْا بِمُوْسٰى وَمَنْ مَّعَهٗ، “اور اگر انہیں کوئی برائی پہنچتی تو موسیٰ علیہ السلام اور ان کے ساتھیوں کو منحوس ٹھہراتے۔” جواب ایک جملے میں آیا کہ سب اللہ کی طرف سے ہے: پیدا کرنے اور مقدر کرنے کے اعتبار سے نہ کوئی بھلائی اس کے دائرے سے باہر ہے نہ کوئی تکلیف، اور بدشگونی کا خیال ہی باطل ہے؛ نبی صلی اللہ علیہ وسلم کی ذات رحمت ہے، نحوست نہیں۔ آیت کا آخری جملہ اس ساری الجھن کی جڑ بتاتا ہے: یہ لوگ بات سمجھنے کے قریب بھی نہیں آتے۔ اہلِ معرفت کہتے ہیں کہ جو ہر واقعے کو صرف اپنی خوشی اور ناخوشی کے پیمانے سے دیکھے، واقعہ اس سے بات کرتا ہے مگر وہ سنتا نہیں۔ شیخِ اکبر کے نزدیک موجودات کلماتِ الٰہیہ ہیں، ہر حادثہ حق کی طرف سے ایک خطاب ہے، اور فقہ یعنی سمجھ اسی خطاب کو سننے کا نام ہے؛ جو انہیں صرف نفع و نقصان کی زبان میں پڑھے، اس نے حروف دیکھے، معنی نہ پایا۔

اب وہ آیت جو اس رکوع کی کنجی ہے، اور جس میں اس سلسلے کا بنیادی اصول قرآن کی اپنی زبان میں بیان ہوا ہے: مَاۤ اَصَابَكَ مِنْ حَسَنَةٍ فَمِنَ اللّٰهِ ۖ وَمَاۤ اَصَابَكَ مِنْ سَيِّئَةٍ فَمِنْ نَّفْسِكَ۔ “تمہیں جو بھلائی پہنچتی ہے وہ اللہ کی طرف سے ہے، اور جو برائی پہنچتی ہے وہ تمہارے اپنے نفس کی طرف سے ہے۔” پہلی نظر میں لگتا ہے کہ ایک آیت کہتی ہے سب اللہ کی طرف سے ہے اور اگلی کہتی ہے برائی نفس سے ہے۔ علمائے اہلِ سنت نے اس کا حل یوں بیان کیا: پچھلی آیت پیدا کرنے اور تقدیر کی بات کرتی ہے، اس اعتبار سے ہر چیز کا خالق اللہ ہے؛ اور یہ آیت سبب اور کسب کی بات کرتی ہے، اس اعتبار سے بھلائی اللہ کا فضل ہے جو بندے کے کسی استحقاق کے بغیر ملتی ہے، اور برائی بندے کی اپنی کمائی کا نتیجہ ہے۔ قرآن نے دوسری جگہ یہی بات کھول دی: وَمَاۤ اَصَابَكُمْ مِّنْ مُّصِيْبَةٍ فَبِمَا كَسَبَتْ اَيْدِيْكُمْ وَيَعْفُوْا عَنْ كَثِيْرٍ، “اور تمہیں جو مصیبت پہنچتی ہے وہ تمہارے اپنے ہاتھوں کی کمائی سے ہے، اور وہ بہت کچھ معاف بھی کر دیتا ہے۔” اور صحیح مسلم میں حضرت ابو ذر رضی اللہ عنہ سے مروی حدیثِ قدسی کا مفہوم یہ ہے: اے میرے بندو! یہ تمہارے ہی اعمال ہیں جنہیں میں تمہارے لیے گن رکھتا ہوں، پھر تمہیں پورا پورا لوٹا دیتا ہوں؛ تو جو بھلائی پائے وہ اللہ کی حمد کرے، اور جو کچھ اور پائے وہ اپنے سوا کسی کو ملامت نہ کرے۔ اہلِ معرفت کہتے ہیں کہ دونوں آیتوں کو ساتھ پڑھنے سے بندہ دو لغزشوں سے بچ جاتا ہے: ایک یہ کہ وہ اپنے آپ کو خالق کے مقابل ایک مستقل فاعل سمجھ لے، اور دوسری یہ کہ اپنی کوتاہی کا بوجھ تقدیر پر ڈال کر خود کو بری کر لے۔ شیخِ اکبر کے مکتب میں یہی جمع کا مقام ہے: وجود اس کا، ذمہ داری اپنی۔

شیخِ اکبر کے مکتب میں اس آیت کی گہرائی وجود کے راز سے کھلتی ہے۔ ان کے نزدیک وجود خالص خیر ہے، اور جو کچھ بھی موجود ہے وہ حق کی طرف سے فیض ہے۔ شر کی اپنی کوئی مستقل ہستی نہیں؛ وہ کسی کمال کا نہ ہونا، کسی رخ کا ٹیڑھا ہونا، یا قبول کرنے والے برتن کی تنگی ہے۔ اس مکتب میں اس کی مثال یوں دی جاتی ہے کہ سورج کی روشنی ایک ہے اور بے رنگ ہے، مگر جب رنگین شیشوں سے گزرتی ہے تو ہر شیشہ اسے اپنے رنگ میں دکھاتا ہے؛ روشنی سورج کی ہے، رنگ شیشے کا۔ پچھلی آیت کا یہ فرمان کہ سب کچھ اللہ کی طرف سے ہے، وجود کی طرف اشارہ ہے؛ اور اس آیت کا یہ فرمان کہ برائی تمہارے نفس سے ہے، قبول کرنے والے کی طرف۔ پھر سرِّ قدر کا دروازہ کھلتا ہے۔ شیخِ اکبر کے نزدیک علم معلوم کے تابع ہے اور حکم علم کے تابع۔ ہر شے کی ایک ثابت حقیقت حق کے علم میں ہے جسے عینِ ثابتہ کہتے ہیں؛ فیضِ اقدس سے اسے اس کی استعداد ملی، اور فیضِ مقدس سے وہ اپنی استعداد کے مطابق وجود میں ظاہر ہوئی۔ حق اسے وہی دیتا ہے جو وہ اپنے حال کی زبان سے مانگتی ہے۔ اس لیے حکم محکوم کے مطابق ہے، باہر سے مسلط کیا ہوا نہیں۔ اس مکتب میں کہا جاتا ہے کہ جو اس راز کو جان لے اس کی شکایت ختم ہو جاتی ہے؛ وہ کسی پر الزام نہیں دھرتا، نہ تقدیر پر، نہ کسی نبی پر، نہ کسی دوسرے انسان پر، اور ہر خوبی پر حمد اور ہر کمی پر توبہ اس کا شیوہ بن جاتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر یہی رہتا ہے کہ بندہ اپنے عمل کا ذمہ دار ہے۔

یہ اس سلسلے کا وہ اصول ہے جس پر کوئی سمجھوتہ نہیں، اور یہاں قرآن نے اسے خود بیان کر دیا ہے، اس لیے اسے اصول کے طور پر نشان زد کیا جاتا ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور کسی ہدایت کے طالب کو گمراہ نہیں کرتا۔ گمراہی دل پر آنے والی سب سے بڑی برائی ہے، اور اس آیت کے مطابق اس برائی کا سرچشمہ بندے کا اپنا نفس ہے۔ جو طلب کرتا ہے ہدایت پاتا ہے؛ اور جو منہ موڑتا ہے، اس کا منہ موڑنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دے دیتا ہے جو بندے نے خود چنا۔ اسی سورت میں آگے یہ اصول اور کھل کر آئے گا: نُوَلِّهٖ مَا تَوَلّٰى، “ہم اسے اسی طرف پھیر دیتے ہیں جدھر وہ خود پھرا۔” اور جو قدرت کا سوال اٹھائے، اس کے لیے قرآن کا جواب ہے: وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ، “اور اگر وہ چاہتا تو تم سب کو ہدایت دے دیتا۔” یعنی وہ مجبور کر سکتا تھا اور اس نے مجبور نہ کرنا چنا، تاکہ انسان کا انتخاب انسان کا رہے اور اس کی بندگی ایک باوقار بندگی ہو۔ اسی لیے يُضِلُّ مَنْ يَّشَآءُ جیسے جملے اس سلسلے میں ہمیشہ سرِّ قدر کی روشنی میں پڑھے جاتے ہیں: مشیت بندے کے اپنے حال اور اس کے پہلے سے اختیار کیے ہوئے اعراض کے مطابق ہے، اس پر کوئی جبر نہیں۔ اگلے رکوع میں منافقین کے بارے میں آئے گا کہ اللہ نے انہیں اوندھا کر دیا، مگر ساتھ ہی فرمایا کہ ان کے اپنے کمائے کے سبب۔ گمراہی کو اس کی طرف سے مسلط کردہ کہنا اس آیت کے صریح بیان کے خلاف ہے؛ اہلِ معرفت کہتے ہیں کہ حق کی طرف سے صرف وجود آتا ہے، اور وجود کا رخ بندے کا اپنا ہے۔

اس آیت سے ایک ادب بھی نکلتا ہے جو انبیاء علیہم السلام کی زبان پر رہا۔ سورۂ الشعراء میں حضرت ابراہیم علیہ السلام نے فرمایا: وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ، “اور جب میں بیمار ہوتا ہوں تو وہی مجھے شفا دیتا ہے۔” بیماری کو اپنی طرف منسوب کیا، شفا کو رب کی طرف۔ اور سورۂ الکہف میں حضرت خضر علیہ السلام نے کشتی کے عیب کے بارے میں کہا: فَاَرَدْتُّ اَنْ اَعِيْبَهَا، “تو میں نے چاہا کہ اسے عیب دار کر دوں”، اور یتیموں کے خزانے کی بھلائی کے بارے میں کہا: فَاَرَادَ رَبُّكَ اَنْ يَّبْلُغَاۤ اَشُدَّهُمَا، “تو تمہارے رب نے چاہا کہ وہ دونوں اپنی جوانی کو پہنچیں۔” اہلِ معرفت کہتے ہیں کہ یہ صرف زبان کا ادب نہیں، حقیقت کا بیان بھی ہے، کیونکہ خیر وجود کی جہت سے آتی ہے اور کمی قبول کرنے والے کی جہت سے۔ شیخِ اکبر کے نزدیک عارف کا کمال یہ ہے کہ وہ دونوں نسبتوں کو ان کے مقام پر رکھے: توحید کی آنکھ سے دیکھے تو ہر شے میں حق کا فعل پائے، اور ادب کی زبان سے بولے تو ہر عیب اپنے سر لے۔ جو صرف پہلی آنکھ رکھے وہ بے ادب ہو کر اپنا گناہ تقدیر کے سر ڈالنے لگتا ہے؛ جو صرف دوسری رکھے وہ اپنے آپ کو خالق کے مقابل ایک مستقل قدرت سمجھنے لگتا ہے۔ کامل وہ ہے جو دونوں کو جمع کرے؛ حمد اس کی، استغفار اپنا۔

آیت کا آخری حصہ بدشگونی کے الزام کا جواب ہے: وَاَرْسَلْنٰكَ لِلنَّاسِ رَسُوْلًا ۚ وَكَفٰى بِاللّٰهِ شَهِيْدًا۔ “اور ہم نے تمہیں سب لوگوں کے لیے رسول بنا کر بھیجا ہے، اور اللہ کی گواہی کافی ہے۔” جسے وہ لوگ مصیبت کا سبب کہتے تھے، وہ ساری انسانیت کے لیے پیغامِ رحمت ہے۔ پھر فرمایا: مَنْ يُّطِعِ الرَّسُوْلَ فَقَدْ اَطَاعَ اللّٰهَ ۚ وَمَنْ تَوَلّٰى فَمَاۤ اَرْسَلْنٰكَ عَلَيْهِمْ حَفِيْظًا۔ “جس نے رسول کی اطاعت کی اس نے اللہ کی اطاعت کی، اور جو منہ پھیرے تو ہم نے تمہیں ان پر نگہبان بنا کر نہیں بھیجا۔” حضورِ اکرم صلی اللہ علیہ وسلم کا حکم اللہ کے حکم کے مقابل کوئی دوسرا حکم نہیں، بلکہ وہی ایک حکم ہے جو آپ صلی اللہ علیہ وسلم کے واسطے سے مخلوق تک پہنچا؛ اسی لیے آپ صلی اللہ علیہ وسلم کی اطاعت عین اللہ کی اطاعت ہے۔ شیخِ اکبر نے فصوص الحکم میں آپ صلی اللہ علیہ وسلم کے باب کو حکمتِ فردیہ کا نام دیا، جس کا مفہوم یہ ہے کہ آپ صلی اللہ علیہ وسلم نوعِ انسانی کے سب سے کامل فرد ہیں اور اسی لیے اپنے رب پر سب سے روشن دلیل۔ مگر یہ کمالِ مظہریت ہے، حلول یا اتحاد نہیں؛ آپ صلی اللہ علیہ وسلم ہمیشہ عبد ہیں، اور عبدیت ہی آپ صلی اللہ علیہ وسلم کا سب سے بلند تاج ہے۔ اور آیت کا دوسرا جملہ انتخاب کے وقار کی حفاظت کرتا ہے، جیسے سورۂ الغاشیہ میں فرمایا: لَسْتَ عَلَيْهِمْ بِمُصَيْطِرٍ، “تم ان پر داروغہ نہیں ہو۔” رسول کا کام پہنچانا ہے، زبردستی منوانا نہیں؛ اور لفظ تَوَلّٰى وہی ہے جو آگے اس اصول میں آئے گا کہ حق اسے اسی طرف پھیر دیتا ہے جدھر وہ خود پھرا۔

پھر منافقت کی ایک تصویر، جسے اس مکتب میں ہمیشہ اپنے اندر کی طرف موڑ کر پڑھا جاتا ہے: وَيَقُوْلُوْنَ طَاعَةٌ ۚ فَاِذَا بَرَزُوْا مِنْ عِنْدِكَ بَيَّتَ طَآئِفَةٌ مِّنْهُمْ غَيْرَ الَّذِيْ تَقُوْلُ ۗ وَاللّٰهُ يَكْتُبُ مَا يُبَيِّتُوْنَ ۚ فَاَعْرِضْ عَنْهُمْ وَتَوَكَّلْ عَلَى اللّٰهِ ۗ وَكَفٰى بِاللّٰهِ وَكِيْلًا۔ “اور کہتے ہیں: ہم نے مان لیا؛ پھر جب تمہارے پاس سے نکلتے ہیں تو ان میں سے ایک گروہ رات کو اس کے خلاف مشورے کرتا ہے جو تم کہتے ہو، اور اللہ لکھ رہا ہے جو وہ رات کو مشورے کرتے ہیں؛ تو ان سے درگزر کرو اور اللہ پر بھروسہ رکھو، اور اللہ کارساز کافی ہے۔” بَيَّتَ کا تعلق رات سے ہے: وہ منصوبے جو اندھیرے میں پکتے ہیں۔ اہلِ معرفت کہتے ہیں کہ ہر سالک کے اندر یہ منظر دہرایا جاتا ہے: ذکر کی مجلس میں زبان کہتی ہے کہ ہم نے مان لیا، اور مجلس سے نکلتے ہی نفس رات کی تنہائی میں کوئی اور منصوبہ بنانے لگتا ہے۔ شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ بندے کا ہر خیال اس کے اپنے باطن کی تختی پر نقش ہوتا جاتا ہے، اور یہی نقش ایک دن اس کے سامنے کھلی کتاب بن کر آئے گا؛ اس لیے مراقبہ یہ ہے کہ رات کی تدبیروں کو بھی دن کی بیعت کے مطابق رکھا جائے۔ اور فَاَعْرِضْ عَنْهُمْ میں حضورِ اکرم صلی اللہ علیہ وسلم کا وہ حلم جھلکتا ہے کہ آپ صلی اللہ علیہ وسلم نے منافقین کو جانتے ہوئے بھی سرِ عام رسوا نہیں کیا۔ اسی سے یہ ادب نکلتا ہے کہ ہم آج کسی کو منافق نہ کہیں؛ ہمارا معاملہ اپنے نفس سے ہے۔

اسی الجھن کا علاج بھی رکوع نے خود بتا دیا: اَفَلَا يَتَدَبَّرُوْنَ الْقُرْاٰنَ ۚ وَلَوْ كَانَ مِنْ عِنْدِ غَيْرِ اللّٰهِ لَوَجَدُوْا فِيْهِ اخْتِلَافًا كَثِيْرًا۔ “کیا یہ قرآن میں غور نہیں کرتے؟ اور اگر یہ اللہ کے سوا کسی اور کی طرف سے ہوتا تو وہ اس میں بہت اختلاف پاتے۔” تدبر کا مادہ دُبُر ہے، یعنی کسی چیز کا پچھلا حصہ؛ تدبر بات کے انجام تک پہنچنا ہے، سطح پر رک جانا نہیں۔ یہ آیت بالکل صحیح جگہ آئی ہے: ابھی دو آیتیں گزریں جو بظاہر ایک دوسرے کے خلاف لگتی تھیں، سب کچھ اللہ کی طرف سے، اور برائی نفس کی طرف سے؛ مگر تدبر کرنے والے نے دیکھا کہ ان میں کوئی تضاد نہیں، دونوں ایک ہی حقیقت کے دو رخ ہیں۔ تئیس برس میں، جنگ اور امن، غم اور خوشی کے حالات میں اترنے والا کلام اگر انسانی ہوتا تو اس کے رنگ بدل جاتے۔ شیخِ اکبر کے نزدیک قرآن کا نام ہی جمع سے ہے: وہ ان سب پہلوؤں کو جمع کرتا ہے جنہیں عقل الگ الگ دیکھتی ہے، جیسے تنزیہ اور تشبیہ، حق کی فاعلیت اور بندے کی ذمہ داری۔ اور فتوحاتِ مکیہ میں ان کا یہ موقف معروف ہے، جس کا مفہوم یہ ہے کہ عربی الفاظ جن معانی کا احتمال رکھتے ہوں اور جو شریعت سے نہ ٹکرائیں، وہ سب اس بندے کے حق میں اللہ کی مراد ہیں جس پر وہ کھولے گئے۔ اسی لیے تدبر کرنے والے پر ہر تلاوت ایک نیا دروازہ کھولتی ہے، اور ظاہر کی حد اس دروازے کی چوکھٹ ہے۔

پھر جنگ کے ماحول کی ایک بڑی آفت کا ذکر: وَاِذَا جَآءَهُمْ اَمْرٌ مِّنَ الْاَمْنِ اَوِ الْخَوْفِ اَذَاعُوْا بِهٖ ۗ وَلَوْ رَدُّوْهُ اِلَى الرَّسُوْلِ وَاِلٰۤى اُولِي الْاَمْرِ مِنْهُمْ لَعَلِمَهُ الَّذِيْنَ يَسْتَنْۢبِطُوْنَهٗ مِنْهُمْ ۗ وَلَوْلَا فَضْلُ اللّٰهِ عَلَيْكُمْ وَرَحْمَتُهٗ لَاتَّبَعْتُمُ الشَّيْطٰنَ اِلَّا قَلِيْلًا۔ “اور جب ان کے پاس امن یا خوف کی کوئی خبر آتی ہے تو اسے پھیلا دیتے ہیں؛ اور اگر وہ اسے رسول کی طرف اور اپنے میں سے صاحبانِ امر کی طرف لوٹا دیتے تو ان میں سے جو اس کی تہہ تک پہنچنے والے ہیں وہ اس کی حقیقت جان لیتے۔ اور اگر تم پر اللہ کا فضل اور اس کی رحمت نہ ہوتی تو چند کے سوا تم سب شیطان کے پیچھے لگ جاتے۔” صحیح مسلم کی حدیث کا مفہوم ہے کہ آدمی کے جھوٹا ہونے کے لیے اتنا کافی ہے کہ ہر سنی سنائی بات آگے بیان کر دے۔ استنباط کا مادہ نَبَط ہے، وہ پانی جو گہرا کنواں کھود کر نکالا جائے؛ یعنی خبر کو پھیلانا آسان ہے، اس کی تہہ سے حقیقت نکالنا اہلِ علم کا کام ہے۔ اہلِ معرفت اس آیت کو باطن پر بھی پڑھتے ہیں: سالک پر کبھی قبض کا خوف اترتا ہے، کبھی بسط کا امن؛ کبھی کوئی وارد یا کشف۔ ہر کیفیت کو فوراً بیان کر دینا اسرار کی بے ادبی ہے۔ شیخِ اکبر کے نزدیک ہر وارد کو شریعت کی میزان پر لوٹانا لازم ہے؛ جو اس پر پورا اترے وہ قبول، جو نہ اترے وہ رد، چاہے کتنا ہی روشن کیوں نہ لگے۔ اور آخری جملہ عاجزی سکھاتا ہے کہ اس حفاظت کا سہرا بھی بندے کی ذہانت کے سر نہیں، اللہ کے فضل اور رحمت کے سر ہے۔

فَقَاتِلْ فِيْ سَبِيْلِ اللّٰهِ ۚ لَا تُكَلَّفُ اِلَّا نَفْسَكَ وَحَرِّضِ الْمُؤْمِنِيْنَ ۚ عَسَى اللّٰهُ اَنْ يَّكُفَّ بَاْسَ الَّذِيْنَ كَفَرُوْا ۗ وَاللّٰهُ اَشَدُّ بَاْسًا وَّاَشَدُّ تَنْكِيْلًا۔ “تو اللہ کی راہ میں لڑو، تم پر اپنی ذات کے سوا کسی کی ذمہ داری نہیں، اور ایمان والوں کو ابھارو؛ قریب ہے کہ اللہ کافروں کا زور روک دے، اور اللہ زور میں سب سے سخت اور سزا دینے میں سب سے سخت ہے۔” یہ خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے بحیثیت امام و قائد ہے۔ مفسرین نے اس کا پس منظر احد کے بعد بدرِ صغریٰ کا وعدہ لکھا ہے، جب کچھ لوگ نکلنے سے ہچکچائے تو آپ صلی اللہ علیہ وسلم کو حکم ملا کہ آپ صلی اللہ علیہ وسلم اپنی ذات کے ذمہ دار ہیں، چاہے اکیلے نکلنا پڑے۔ مفسرین کہتے ہیں کہ اللہ کی طرف سے عَسٰى وعدے کے معنی میں ہے؛ اور واقعی دشمن مقابلے پر نہ آیا۔ اور حکم ابھارنے کا ہے، مجبور کرنے کا نہیں؛ ایمان والوں کے دل میں شوق جگانا ہے، ان پر جبر نہیں۔ اہلِ معرفت کے نزدیک لَا تُكَلَّفُ اِلَّا نَفْسَكَ ہر سالک کے لیے اصلاح کا نقطۂ آغاز ہے: دنیا کی درستی کا بوجھ اٹھانے سے پہلے اس ایک امانت کو سنبھالو جو تمہارے سپرد ہے، یعنی تمہارا اپنا نفس۔ اہلِ تصوف میں مشہور اثر ہے کہ جس نے اپنے نفس کو پہچانا اس نے اپنے رب کو پہچانا۔ شیخِ اکبر کے مکتب میں اپنی پہچان ہی رب کی پہچان کا دروازہ ہے، کیونکہ انسان اسمائے الٰہیہ کا جامع آئینہ ہے؛ جو اپنا آئینہ صاف کر لے، دوسروں کے لیے بھی اس کا وجود ابھارنے کا ذریعہ بن جاتا ہے۔

مَنْ يَّشْفَعْ شَفَاعَةً حَسَنَةً يَّكُنْ لَّهٗ نَصِيْبٌ مِّنْهَا ۚ وَمَنْ يَّشْفَعْ شَفَاعَةً سَيِّئَةً يَّكُنْ لَّهٗ كِفْلٌ مِّنْهَا ۗ وَكَانَ اللّٰهُ عَلٰى كُلِّ شَيْءٍ مُّقِيْتًا۔ “جو اچھی سفارش کرے اسے اس میں سے حصہ ملے گا، اور جو بری سفارش کرے اس پر اس کا بوجھ آئے گا، اور اللہ ہر چیز پر نگہبان اور قدرت والا ہے۔” شفاعت کا مادہ شَفْع ہے، یعنی جفت؛ سفارش کرنے والا اپنے آپ کو کسی ضرورت مند کے ساتھ ملا کر اس کا جوڑا بن جاتا ہے۔ صحیح بخاری کی حدیث کا مفہوم ہے کہ سفارش کرو، تمہیں اجر ملے گا۔ آیت نے اچھی سفارش کو نصیب اور بری کو کفل کہا: بھلائی کی طرف ملانے والا اجر میں شریک، برائی کی طرف ملانے والا بوجھ میں۔ آخرت میں سب سے بڑی شفاعت حضورِ اکرم صلی اللہ علیہ وسلم کی ہے؛ اور صحیح مسلم میں حضرت ابو سعید خدری رضی اللہ عنہ کی حدیث کا مفہوم ہے کہ فرشتے سفارش کریں گے، انبیاء علیہم السلام سفارش کریں گے، ایمان والے سفارش کریں گے، اور پھر ارحم الراحمین باقی رہ جائے گا۔ شیخِ اکبر کے مکتب میں اس حدیث سے سبقتِ رحمت کا راز کھولا جاتا ہے: آخری فیصلہ اسمِ ارحم الراحمین کے ہاتھ میں ہے، اور رحمت غضب پر سبقت رکھتی ہے اور ہر چیز پر وسیع ہے۔ مُقِيْت کا مادہ قُوت ہے: وہ جو ہر ایک کو اس کی روزی اور حصہ اس کے اندازے کے مطابق پہنچاتا ہے؛ کسی کی سفارش کا کوئی ذرہ اس کی نگاہ سے باہر نہیں۔

سفارش کے بعد سلام کا ذکر آیا: وَاِذَا حُيِّيْتُمْ بِتَحِيَّةٍ فَحَيُّوْا بِاَحْسَنَ مِنْهَاۤ اَوْ رُدُّوْهَا ۗ اِنَّ اللّٰهَ كَانَ عَلٰى كُلِّ شَيْءٍ حَسِيْبًا۔ “اور جب تمہیں کوئی دعائے خیر کے ساتھ سلام کرے تو تم اس سے بہتر جواب دو یا وہی لوٹا دو؛ بے شک اللہ ہر چیز کا حساب رکھنے والا ہے۔” جنگ کی آیات کے بیچ سلام کا یہ حکم بتاتا ہے کہ سختی کے ماحول میں بھی مومن کا پہلا رویہ سلامتی ہے۔ سنن ابی داود اور ترمذی کی روایت کا مفہوم ہے کہ ایک صاحب نے السلام علیکم کہا تو حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا دس نیکیاں، دوسرے نے رحمۃ اللہ بڑھایا تو بیس، تیسرے نے برکاتہ بھی کہا تو تیس۔ اور صحیح مسلم کی حدیث کا مفہوم ہے کہ تم جنت میں داخل نہ ہو گے جب تک ایمان نہ لاؤ، اور ایمان والے نہ ہو گے جب تک آپس میں محبت نہ کرو؛ کیا میں تمہیں وہ چیز نہ بتاؤں جس سے محبت پیدا ہو؟ آپس میں سلام کو عام کرو۔ تحیت کا مادہ حیات ہے: سلام دراصل زندگی کی دعا ہے۔ شیخِ اکبر کے مکتب میں السلام اسمائے حسنیٰ میں سے ہے؛ سلام کرنے والا اپنے بھائی کو اللہ کا ایک نام ہدیہ کرتا ہے، اور بہتر جواب دینا تخلق باسماء اللہ ہے، یعنی اس رب کے اخلاق میں رنگ جانا جو ہمیشہ اپنی عطا سے بڑھ کر دیتا ہے۔ اور حَسِيْب بتاتا ہے کہ ایک سلام کا جواب بھی اس کے حساب میں درج ہے۔

رکوع توحید اور جمع پر ختم ہوتا ہے: اَللّٰهُ لَاۤ اِلٰهَ اِلَّا هُوَ ۗ لَيَجْمَعَنَّكُمْ اِلٰى يَوْمِ الْقِيٰمَةِ لَا رَيْبَ فِيْهِ ۗ وَمَنْ اَصْدَقُ مِنَ اللّٰهِ حَدِيْثًا۔ “اللہ، اس کے سوا کوئی معبود نہیں؛ وہ تم سب کو ضرور قیامت کے دن جمع کرے گا جس میں کوئی شک نہیں؛ اور اللہ سے بڑھ کر بات کا سچا کون ہے؟” اس رکوع کی ایک خوبصورتی یہ ہے کہ اسے لفظ حدیث نے گھیر رکھا ہے: آیت ۷۸ میں وہ لوگ تھے جو بات سمجھنے کے قریب بھی نہ آتے تھے، بیچ میں آیت ۸۲ کا وہ کلام جس میں کوئی اختلاف نہیں، اور اب آخر میں یہ اعلان کہ اللہ سے سچی بات کسی کی نہیں۔ اہلِ معرفت کہتے ہیں کہ جو اس سچی بات کو سن لے، اس کے لیے بھلائی اور برائی، موت اور قلعہ، سفارش اور سلام سب ایک ہی مرکز کے گرد گھومنے لگتے ہیں۔ شیخِ اکبر کے نزدیک اللہ وہ اسمِ جامع ہے جس میں سب اسماء جمع ہیں، اور قیامت کا دن جمع کا دن ہے جب کثرت کے پردے اٹھ جائیں گے اور ہر شخص وہ دیکھ لے گا جو ہمیشہ سے حقیقت تھی۔ وہاں ہر ایک اپنا ہی کمایا ہوا پائے گا، اور یوں آیت ۷۹ کا اصول اپنی آخری صورت میں ظاہر ہو گا: خیر اس کی عطا، کوتاہی اپنی کمائی۔
