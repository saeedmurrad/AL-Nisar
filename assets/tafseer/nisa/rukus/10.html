پچھلا رکوع ان لوگوں کے ذکر پر ختم ہوا تھا جن پر اللہ نے انعام فرمایا: انبیاء علیہم السلام، صدیقین، شہداء اور صالحین۔ اب یہ رکوع اس جماعت کو، جو انہی کی رفاقت کی امیدوار ہے، اپنی حفاظت اور اپنے کمزوروں کی خبر گیری کا حکم دیتا ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا خُذُوْا حِذْرَكُمْ فَانْفِرُوْا ثُبَاتٍ اَوِ انْفِرُوْا جَمِيْعًا۔ “اے ایمان والو! اپنی احتیاط کا سامان لے لو، پھر دستہ دستہ نکلو یا سب اکٹھے نکلو۔” یہ آیات مدینہ منورہ کے اس دور میں اتریں جب ایک چھوٹی سی بستی چاروں طرف سے دشمنوں میں گھری تھی اور احد کے زخم ابھی تازہ تھے۔ مفسرین کے نزدیک یہاں قتال کا حکم ایک تاریخی پس منظر رکھتا ہے: جماعت کا دفاع، حضورِ اکرم صلی اللہ علیہ وسلم کی قیادت میں، اور جائز امام کے حکم سے۔ یہ آیات کسی فرد یا ٹولی کو اپنی مرضی سے ہتھیار اٹھانے کی اجازت نہیں دیتیں؛ فقہاء نے ہمیشہ قتال کو امام اور نظمِ جماعت سے وابستہ رکھا ہے۔ اور حِذر کا حکم بتاتا ہے کہ تدبیر توکل کے خلاف نہیں؛ حدیث میں آیا ہے جس کا مفہوم یہ ہے کہ اونٹ کا گھٹنا باندھو، پھر بھروسہ کرو۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اسباب کو حق کا پردہ کہا جاتا ہے: عارف اس پردے کو پھاڑتا نہیں، اس کا ادب کرتا ہے، اور جانتا ہے کہ پردے کے پیچھے فاعل وہی ہے۔ اہلِ ذوق اس آیت کے دو راستوں میں سالک کی دو راہیں دیکھتے ہیں: ثُبَاتٍ یعنی ٹکڑی ٹکڑی، منزل بہ منزل چلنا، جو سلوک ہے؛ اور جَمِيْعًا یعنی سب کا ایک ساتھ اٹھ کھڑا ہونا، جو جذب ہے، جب حق کی ایک کشش دل کی ساری قوتوں کو یکبارگی سمیٹ لیتی ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

اسی صف میں ایک اور کردار بھی تھا جسے قرآن نے بے نام رکھ کر آئینہ بنا دیا: وَاِنَّ مِنْكُمْ لَمَنْ لَّيُبَطِّئَنَّ ۚ فَاِنْ اَصَابَتْكُمْ مُّصِيْبَةٌ قَالَ قَدْ اَنْعَمَ اللّٰهُ عَلَيَّ اِذْ لَمْ اَكُنْ مَّعَهُمْ شَهِيْدًا۔ “اور تم میں کوئی ایسا بھی ہے جو ضرور پیچھے رہ جاتا ہے؛ پھر اگر تم پر کوئی مصیبت آ پڑے تو کہتا ہے: اللہ نے مجھ پر انعام کیا کہ میں ان کے ساتھ موجود نہ تھا۔” وَلَئِنْ اَصَابَكُمْ فَضْلٌ مِّنَ اللّٰهِ لَيَقُوْلَنَّ كَاَنْ لَّمْ تَكُنْۢ بَيْنَكُمْ وَبَيْنَهٗ مَوَدَّةٌ يّٰلَيْتَنِيْ كُنْتُ مَعَهُمْ فَاَفُوْزَ فَوْزًا عَظِيْمًا۔ “اور اگر تمہیں اللہ کا فضل ملے تو، گویا تمہارے اور اس کے درمیان کبھی دوستی تھی ہی نہیں، ضرور کہتا ہے: کاش میں بھی ان کے ساتھ ہوتا تو بڑی کامیابی پا لیتا۔” مفسرین نے اسے منافقین کا حال لکھا ہے؛ بعض نے اس کا مصداق ان لوگوں کو بتایا جو احد کے دن راستے سے لوٹ گئے تھے۔ مگر اس مکتب کا ادب یہ ہے کہ منافق کی ہر صفت کو اپنے اندر تلاش کیا جائے، آج کے کسی انسان پر چسپاں نہ کیا جائے۔ ہم میں سے ہر ایک کے اندر یہ پیچھے رہ جانے والا نفس موجود ہے جو دین کو نفع و نقصان کے ترازو میں تولتا ہے: آزمائش سے بچ جائے تو اسے نعمت کہتا ہے، اور دوسروں کو فضل ملے تو حسرت کرتا ہے۔ اہلِ معرفت کہتے ہیں کہ اس آیت کا سب سے باریک لفظ شَهِيْدًا ہے، یعنی حاضر۔ یہ شخص غیر حاضری پر شکر ادا کرتا ہے۔ شیخِ اکبر کے نزدیک نعمت کی حقیقت وہ ہے جو بندے کو حضوری کی طرف لے جائے؛ جو غفلت کو سلامتی سمجھ کر اس پر شکر کرے، اس نے نعمت کا نام غلط جگہ رکھ دیا۔ اور مودت والا جملہ بتاتا ہے کہ مفاد کی محبت فائدے کے ساتھ آتی ہے اور فائدے کے ساتھ ہی رخصت ہو جاتی ہے؛ سالک کی محبت وہ ہے جو حق کی خاطر ہو، اس لیے نہ مصیبت میں ٹوٹتی ہے نہ فضل پر حسد بنتی ہے۔

فَلْيُقَاتِلْ فِيْ سَبِيْلِ اللّٰهِ الَّذِيْنَ يَشْرُوْنَ الْحَيٰوةَ الدُّنْيَا بِالْاٰخِرَةِ ۚ وَمَنْ يُّقَاتِلْ فِيْ سَبِيْلِ اللّٰهِ فَيُقْتَلْ اَوْ يَغْلِبْ فَسَوْفَ نُؤْتِيْهِ اَجْرًا عَظِيْمًا۔ “تو اللہ کی راہ میں وہ لوگ لڑیں جو دنیا کی زندگی آخرت کے بدلے بیچ دیتے ہیں؛ اور جو اللہ کی راہ میں لڑے، پھر مارا جائے یا غالب آئے، ہم اسے بڑا اجر دیں گے۔” اکثر مفسرین کے نزدیک یہاں يَشْرُوْنَ کا معنی بیچنا ہے: سودا یہ ہے کہ فانی دے کر باقی لیا جائے۔ آیت نے قتال کی شرط فِيْ سَبِيْلِ اللّٰهِ رکھی، اور صحیح حدیث کا مفہوم ہے کہ اللہ کی راہ میں وہی ہے جو اس لیے لڑے کہ اللہ کا کلمہ بلند ہو؛ نہ غنیمت کے لیے، نہ شہرت کے لیے، نہ قبیلے کی حمیت کے لیے۔ پھر دو انجام رکھے گئے اور دونوں کو ایک ہی اجر میں جمع کر دیا گیا، جیسے دوسری جگہ فرمایا: قُلْ هَلْ تَرَبَّصُوْنَ بِنَاۤ اِلَّاۤ اِحْدَى الْحُسْنَيَيْنِ، “کہہ دو: تم ہمارے بارے میں دو بھلائیوں میں سے ایک ہی کا تو انتظار کرتے ہو۔” شیخِ اکبر کے مکتب میں اس کا اشاری پہلو یہ ہے کہ باطن کے میدان میں بھی دو ہی انجام ہیں: ایک یہ کہ بندے کا دعویٰ حق کے سامنے مٹ جائے، جسے فنا کہتے ہیں، اور یہ اس کا مارا جانا ہے؛ دوسرا یہ کہ وہ حق کے ساتھ باقی ہو کر اپنے نفس پر غالب آ جائے، جسے بقا کہتے ہیں، اور یہ اس کا غالب آنا ہے۔ دونوں حالتوں کا اجر عظیم ہے، کیونکہ دونوں میں بندہ اپنے سے نکل کر اس کا ہو گیا۔ اہلِ ذوق کہتے ہیں کہ ہارنے کا کوئی تیسرا خانہ اس آیت میں ہے ہی نہیں؛ جو اللہ کے لیے نکلا وہ ہر حال میں جیتا۔ یہ اشارہ ظاہر کو منسوخ نہیں کرتا، ظاہر کی تہہ کھولتا ہے۔

پھر سوال کی صورت میں ایک پکار آتی ہے جو صدیوں سے دلوں کو جھنجھوڑ رہی ہے: وَمَا لَكُمْ لَا تُقَاتِلُوْنَ فِيْ سَبِيْلِ اللّٰهِ وَالْمُسْتَضْعَفِيْنَ مِنَ الرِّجَالِ وَالنِّسَآءِ وَالْوِلْدَانِ۔ “اور تمہیں کیا ہوا کہ تم اللہ کی راہ میں نہیں لڑتے، اور ان بے بس مردوں، عورتوں اور بچوں کی خاطر نہیں لڑتے” الَّذِيْنَ يَقُوْلُوْنَ رَبَّنَاۤ اَخْرِجْنَا مِنْ هٰذِهِ الْقَرْيَةِ الظَّالِمِ اَهْلُهَا ۚ وَاجْعَلْ لَّنَا مِنْ لَّدُنْكَ وَلِيًّا ۚ وَّاجْعَلْ لَّنَا مِنْ لَّدُنْكَ نَصِيْرًا۔ “جو کہتے ہیں: اے ہمارے رب! ہمیں اس بستی سے نکال جس کے رہنے والے ظالم ہیں، اور اپنی طرف سے ہمارا کوئی سرپرست بنا دے، اور اپنی طرف سے ہمارا کوئی مددگار بنا دے۔” یہ وہ مسلمان تھے جو مکہ مکرمہ میں رہ گئے تھے، ہجرت کی طاقت نہ رکھتے تھے، اور ایمان کی پاداش میں ستائے جا رہے تھے۔ صحیح بخاری میں حضرت عبد اللہ بن عباس رضی اللہ عنہما کا قول ہے جس کا مفہوم یہ ہے کہ میں اور میری والدہ انہی بے بسوں میں سے تھے۔ آیت نے اس قتال کا مقصد خود بیان کر دیا: کسی کی زمین چھیننا نہیں، مظلوم کو ظلم سے چھڑانا۔ اور قرآن نے مردوں کے ساتھ عورتوں اور بچوں کو الگ الگ نام لے کر پکارا، کہ ہر ایک کا حق مستقل ہے اور کوئی کسی کے ضمن میں گم نہیں۔ یہ بے بس خاموش بھی نہ تھے؛ ان کی دعا وحی کا حصہ بن گئی۔ بعض مفسرین نے لکھا ہے کہ یہ دعا فتحِ مکہ کی صورت میں قبول ہوئی، جب حضورِ اکرم صلی اللہ علیہ وسلم ان کے سرپرست بنے اور آپ صلی اللہ علیہ وسلم کے مقرر کردہ والی نے کمزور کو طاقتور سے اس کا حق دلایا۔ اہلِ معرفت کہتے ہیں کہ مستضعف کی دعا کی قوت اس کی بے بسی ہی سے آتی ہے؛ شیخِ اکبر کے مکتب میں فقر وہ برتن ہے جس میں عطا اترتی ہے، اور جس کے ہاتھ میں کچھ نہ رہا اس کا دامن عطا کے لیے سب سے کشادہ ہے۔ اس ظاہر کی روح آج بھی زندہ ہے: مظلوم کی پکار سننا امت کا فریضہ ہے، ہر شخص کے اپنے دائرۂ اختیار میں اور جائز ذرائع سے۔

اب اس آیت کا اشاری رخ، جو ظاہر کی نفی نہیں بلکہ اس کی گہرائی ہے۔ اہلِ تاویل کہتے ہیں کہ ہر انسان کے اندر بھی ایک بستی ہے جس کے مکین کبھی ظالم ہو جاتے ہیں: عادتیں، خواہشیں اور غفلتیں جب دل کے تخت پر قابض ہوں تو اندر کی فطرت، حق کی طلب اور سچائی کی پیاس اسی بستی میں بے بس ہو کر رہ جاتی ہیں۔ یہاں ایک وضاحت ضروری ہے: بعض اہلِ تاویل باطنی قوتوں کو مردوں، عورتوں اور بچوں سے تمثیل دیتے ہیں؛ یہ محض تمثیل ہے، اس میں نہ کسی کی برتری کا دعویٰ ہے نہ کسی کی تحقیر، اور ہم یہاں تینوں کو ایک ساتھ ہر اس کمزوری کا نام سمجھتے ہیں جو نفس کے تسلط میں دب گئی ہو۔ ان دبے ہوئے جوہروں کی دعا بھی وہی ہے: مِنْ لَّدُنْكَ، اپنی خاص بارگاہ سے۔ لفظِ لدن، لفظِ عند سے زیادہ قرب کا لفظ ہے، اور اسی سے علمِ لدنی کی اصطلاح بنی جس کا ذکر سورۂ الکہف میں گزر چکا۔ شیخِ اکبر کے نزدیک ہر بندے کا ایک ربِّ خاص ہے، یعنی وہ اسمِ الٰہی جو اس کی استعداد کی پرورش کرتا ہے۔ دعا میں ولی اور نصیر دو الگ اسم ہیں: الولی قرب اور سرپرستی کا اسم ہے، النصیر مدد اور غلبے کا۔ اس مکتب میں ترتیب یوں سمجھی جاتی ہے کہ پہلے حق اپنے اسم الولی سے دل کو اپنا بناتا ہے، پھر اسم النصیر سے اسے نفس کے قبضے سے چھڑاتا ہے۔ اور یہ بھی کہا گیا کہ حق کی ولایت اکثر کسی انسانی رہنما کے ہاتھ سے ظاہر ہوتی ہے، جیسے ظاہر میں یہ دعا حضورِ اکرم صلی اللہ علیہ وسلم کے ذریعے قبول ہوئی؛ رہنما وسیلہ ہے، عطا حق کی ہے۔

اَلَّذِيْنَ اٰمَنُوْا يُقَاتِلُوْنَ فِيْ سَبِيْلِ اللّٰهِ ۚ وَالَّذِيْنَ كَفَرُوْا يُقَاتِلُوْنَ فِيْ سَبِيْلِ الطَّاغُوْتِ فَقَاتِلُوْۤا اَوْلِيَآءَ الشَّيْطٰنِ ۚ اِنَّ كَيْدَ الشَّيْطٰنِ كَانَ ضَعِيْفًا۔ “جو ایمان لائے وہ اللہ کی راہ میں لڑتے ہیں، اور جنہوں نے کفر کیا وہ طاغوت کی راہ میں لڑتے ہیں؛ تو شیطان کے دوستوں سے لڑو، بے شک شیطان کی چال کمزور ہے۔” آیت نے انسانوں کو کوشش کرنے والوں اور نہ کرنے والوں میں نہیں بانٹا، کیونکہ ہر ایک کسی نہ کسی راہ میں جدوجہد کر رہا ہے؛ فرق صرف راہ کا ہے۔ طاغوت ہر وہ چیز ہے جو حد سے بڑھ کر بندگی کا دعویٰ کرے، باہر کا سرکش ہو یا اندر کی خواہش۔ پھر رکوع اس جملے پر ختم ہوا جو سالک کے دل کو حوصلہ دیتا ہے: شیطان کی چال کمزور ہے۔ کیوں کمزور ہے؟ سورۂ ابراہیم میں شیطان نے خود اقرار کیا: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ، “اور مجھے تم پر کوئی زور نہ تھا، سوائے اس کے کہ میں نے تمہیں بلایا اور تم نے میری بات مان لی۔” اس کا کام دعوت ہے، اور دعوت پر زور نہیں چلتا۔ شیخِ اکبر کے نزدیک وجود سراسر خیر ہے اور شر کی اپنی کوئی مستقل ہستی نہیں؛ شیطان کی چال میں جو طاقت دکھائی دیتی ہے وہ اس کی اپنی نہیں، ہماری قبولیت سے مستعار ہے۔ جو بندہ استجابت روک لے، اس چال کے پاس کھڑا ہونے کو زمین نہیں رہتی۔ اور یہیں وہ اصول بھی جھلکتا ہے جس پر یہ سلسلہ قائم ہے: حق کسی کو زبردستی گمراہ نہیں کرتا، اور نہ کسی ہدایت کے طالب کو گمراہ کرتا ہے؛ گمراہی بندے کا اپنا رخ پھیر لینا ہے، اور حق اسی رخ کو وجود دے دیتا ہے جو بندے نے خود چنا۔ لفظ كَانَ بتاتا ہے کہ یہ کمزوری ہمیشہ سے ہے، کسی خاص زمانے کی نہیں؛ اور جو اللہ کی راہ میں ہے، اس کے مقابل یہ چال مکڑی کے جالے سے زیادہ کچھ نہیں۔
