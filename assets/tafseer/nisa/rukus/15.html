پچھلا رکوع ہجرت کے وعدے پر ختم ہوا تھا کہ جو اللہ اور اس کے رسول صلی اللہ علیہ وسلم کی طرف ہجرت کرتا ہوا گھر سے نکلے اس کا اجر اللہ کے ذمے ہے۔ اب مسافر کے لیے سب سے پہلے جس رعایت کا ذکر ہوا وہ نماز کی ہے: وَاِذَا ضَرَبْتُمْ فِي الْاَرْضِ فَلَيْسَ عَلَيْكُمْ جُنَاحٌ اَنْ تَقْصُرُوْا مِنَ الصَّلٰوةِ ۖ اِنْ خِفْتُمْ اَنْ يَّفْتِنَكُمُ الَّذِيْنَ كَفَرُوْا ۭ اِنَّ الْكٰفِرِيْنَ كَانُوْا لَكُمْ عَدُوًّا مُّبِيْنًا۔ “اور جب تم زمین میں سفر کرو تو تم پر کوئی گناہ نہیں کہ نماز میں کمی کر دو، اگر تمہیں اندیشہ ہو کہ کافر تمہیں ستائیں گے؛ بے شک کافر تمہارے کھلے دشمن ہیں۔” زمین میں ضرب عربی محاورے میں سفر کو کہتے ہیں، گویا مسافر قدم مار مار کر راستہ طے کرتا ہے۔ قصر یہ ہے کہ چار رکعت والے فرض، یعنی ظہر، عصر اور عشا، دو رکعت پڑھے جائیں؛ فجر اور مغرب میں قصر نہیں۔ صحیح مسلم کی روایت کا مفہوم ہے کہ حضرت یعلیٰ بن امیہ رضی اللہ عنہ نے حضرت عمر رضی اللہ عنہ سے پوچھا کہ آیت میں تو خوف کی قید ہے اور اب لوگ امن میں ہیں؛ آپ رضی اللہ عنہ نے فرمایا کہ مجھے بھی یہی تعجب ہوا تھا، میں نے حضورِ اکرم صلی اللہ علیہ وسلم سے پوچھا تو آپ صلی اللہ علیہ وسلم نے فرمایا: یہ صدقہ ہے جو اللہ نے تم پر کیا ہے، سو اس کا صدقہ قبول کرو۔ یوں معلوم ہوا کہ خوف کی قید اس دور کے عام حال کا بیان تھی، شرط نہیں۔ احناف کے نزدیک مسافر پر قصر واجب ہے اور امام شافعی رحمۃ اللہ علیہ کے نزدیک رخصت؛ مسافت کی تحدید فقہ کی کتابوں میں ہے۔ اہلِ معرفت اس حدیث کے لفظ صدقہ پر ٹھہرتے ہیں۔ صدقہ لینے والا فقیر ہوتا ہے، اور جو فقیر اپنے رب کا صدقہ ٹھکرا دے وہ گویا خود کو غنی سمجھتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں فقر بندے کی اصل حقیقت ہے؛ اس لیے رخصت کو قبول کرنا بھی عبدیت کا ادب ہے، اور حدیث کا مفہوم ہے کہ اللہ کو اپنی رخصتوں پر عمل اسی طرح پسند ہے جیسے اپنے تاکیدی احکام پر۔

اب اس آیت کا وہ رخ جو اہلِ ذوق پر کھلتا ہے، اس شرط کے ساتھ کہ ظاہر کا حکم اپنی جگہ قائم رہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ شیخِ اکبر کے نزدیک سارا وجود سفر میں ہے: کوئی شے دو لمحے ایک حال پر نہیں ٹھہرتی، اور انسان ماں کے پیٹ سے قبر تک اور قبر سے آخرت تک ایک منزل سے دوسری منزل کی طرف رواں ہے۔ اہلِ تاویل کہتے ہیں کہ زمین میں سفر کا ایک باطنی رخ نفس کی زمین میں سفر ہے، جہاں سالک اپنی عادتوں کی بستی چھوڑ کر رب کی طرف نکلتا ہے۔ اس سفر میں بھی حق کی نرمی ساتھ چلتی ہے: صورت ہلکی کر دی جاتی ہے تاکہ حضوری بوجھل نہ ہو۔ ظاہری قصر میں رکعتیں گھٹتی ہیں، حضوری نہیں گھٹتی؛ بلکہ مسافر کی دو رکعتیں اکثر مقیم کی چار سے زیادہ حاضر ہوتی ہیں، کیونکہ غربت میں دل اپنے رب کو زیادہ پکارتا ہے اور سہارے کم ہوں تو سہارا دینے والا زیادہ یاد آتا ہے۔ پھر آیت کا آخری جملہ کہ کافر تمہارے کھلے دشمن ہیں، ظاہر میں اس دور کے ان لشکروں کا بیان ہے جو راستوں میں مسلمانوں کو گھیرتے اور ستاتے تھے۔ اہلِ معرفت اسے اپنے اندر کی طرف بھی موڑتے ہیں کہ سفرِ سلوک میں بھی ایک دشمن گھات میں رہتا ہے جو راہی کو راستے میں آزمائش میں ڈالنا چاہتا ہے۔ یہ کسی گروہ پر حکم نہیں، اپنے نفس کی نگہبانی کی تعلیم ہے۔

اب وہ نماز جو تلواروں کے سائے میں بھی ترک نہیں ہوئی: وَاِذَا كُنْتَ فِيْهِمْ فَاَقَمْتَ لَهُمُ الصَّلٰوةَ فَلْتَقُمْ طَآئِفَةٌ مِّنْهُمْ مَّعَكَ وَلْيَاْخُذُوْۤا اَسْلِحَتَهُمْ ۚ فَاِذَا سَجَدُوْا فَلْيَكُوْنُوْا مِنْ وَّرَآئِكُمْ ۖ وَلْتَاْتِ طَآئِفَةٌ اُخْرٰى لَمْ يُصَلُّوْا فَلْيُصَلُّوْا مَعَكَ وَلْيَاْخُذُوْا حِذْرَهُمْ وَاَسْلِحَتَهُمْ۔ “اور جب آپ صلی اللہ علیہ وسلم ان میں موجود ہوں اور ان کے لیے نماز قائم کریں تو ان میں سے ایک گروہ آپ صلی اللہ علیہ وسلم کے ساتھ کھڑا ہو اور اپنے ہتھیار لیے رہے؛ پھر جب وہ سجدہ کر چکیں تو تمہارے پیچھے ہو جائیں، اور دوسرا گروہ جس نے ابھی نماز نہیں پڑھی آئے اور آپ صلی اللہ علیہ وسلم کے ساتھ نماز پڑھے، اور وہ بھی اپنی احتیاط اور اپنے ہتھیار لیے رہیں۔” یہ صلوٰۃ الخوف ہے۔ سنن ابی داؤد اور سنن نسائی میں حضرت ابو عیاش زرقی رضی اللہ عنہ کی روایت کا مفہوم ہے کہ عسفان کے مقام پر مشرکین نے مسلمانوں کو ظہر پڑھتے دیکھا تو کہا کہ اب ان کی ایک ایسی نماز آنے والی ہے جو انہیں اپنی اولاد اور جانوں سے زیادہ عزیز ہے، اسی میں ان پر حملہ کریں گے؛ تب ظہر اور عصر کے درمیان یہ آیت اتری۔ حضورِ اکرم صلی اللہ علیہ وسلم سے اس نماز کی کئی صورتیں منقول ہیں، جن کا خلاصہ یہ ہے کہ لشکر دو حصوں میں بٹ جائے: ایک حصہ امام کے ساتھ نماز میں ہو اور دوسرا دشمن کے مقابل پہرے پر، پھر دونوں جگہ بدل لیں۔ امام ابو یوسف رحمۃ اللہ علیہ نے اس خطاب سے یہ رائے لی کہ یہ نماز حضورِ اکرم صلی اللہ علیہ وسلم کی موجودگی سے خاص تھی، مگر جمہور کے نزدیک حکم باقی ہے، کیونکہ صحابہ کرام رضی اللہ عنہم نے بعد میں بھی اسے ادا کیا۔ اور یہ سب باقاعدہ لشکر، جائز امارت اور دفاع کے میدان کا حکم ہے۔ اہلِ معرفت کہتے ہیں کہ نماز کی عظمت پر اس سے بڑی گواہی کیا ہوگی کہ جان کا خطرہ بھی جماعت کو نہ توڑ سکا، اور ہر گروہ نے حضورِ اکرم صلی اللہ علیہ وسلم کی اقتدا کا حصہ پایا۔ شیخِ اکبر کے مکتب میں نماز مناجات ہے، اور خوف کی گھڑی وہ گھڑی ہے جب بندہ اپنے فقر کو سب سے زیادہ پہچانتا ہے؛ اس لیے مناجات کا دروازہ اسی وقت سب سے زیادہ کھلا رکھا گیا۔

پھر دشمن کی نیت کھول دی گئی، اور اس کے ساتھ مجبوری کی رعایت بھی: وَدَّ الَّذِيْنَ كَفَرُوْا لَوْ تَغْفُلُوْنَ عَنْ اَسْلِحَتِكُمْ وَاَمْتِعَتِكُمْ فَيَمِيْلُوْنَ عَلَيْكُمْ مَّيْلَةً وَّاحِدَةً ۭ وَلَا جُنَاحَ عَلَيْكُمْ اِنْ كَانَ بِكُمْ اَذًى مِّنْ مَّطَرٍ اَوْ كُنْتُمْ مَّرْضٰۤى اَنْ تَضَعُوْۤا اَسْلِحَتَكُمْ ۚ وَخُذُوْا حِذْرَكُمْ ۭ اِنَّ اللّٰهَ اَعَدَّ لِلْكٰفِرِيْنَ عَذَابًا مُّهِيْنًا۔ “کافر چاہتے ہیں کہ تم اپنے ہتھیاروں اور اپنے سامان سے غافل ہو جاؤ تو وہ تم پر یکبارگی ٹوٹ پڑیں۔ اور تم پر کوئی گناہ نہیں کہ اگر بارش سے تکلیف ہو یا تم بیمار ہو تو اپنے ہتھیار اتار رکھو، مگر اپنی احتیاط کیے رہو؛ بے شک اللہ نے کافروں کے لیے رسوا کن عذاب تیار کر رکھا ہے۔” حذر چوکسی اور احتیاط کو کہتے ہیں۔ آیت نے ہتھیاروں کے ساتھ حذر کا ذکر کیا اور پھر ہتھیار اتارنے کی اجازت دی مگر حذر اتارنے کی نہیں: بوجھ ہلکا ہو سکتا ہے، بیداری ہلکی نہیں ہو سکتی۔ اہلِ ذوق اس میں باطن کا نقشہ دیکھتے ہیں۔ دشمن کی آرزو یہ ہے کہ تم غافل ہو اور وہ ایک ہی جھپٹ میں سب لے اڑے؛ یہی اس وسوسہ ڈالنے والے کا طریقہ ہے جس سے سورۂ الناس نے پناہ سکھائی: مِنْ شَرِّ الْوَسْوَاسِ الْخَنَّاسِ؛ وہ ذکر کے وقت پیچھے ہٹ جاتا ہے اور غفلت کے لمحے میں لوٹ آتا ہے۔ شیخِ اکبر کے نزدیک خلق ہر آن نئی ہے، تجددِ امثال کا یہی مطلب ہے؛ اس لیے غفلت بھی ایک لمحے کی ہوتی ہے اور بیداری بھی ایک لمحے کی، اور سالک کا کام ہر نئے لمحے کو نئی حضوری سے بھرنا ہے۔ ایک لمحے کی غفلت برسوں کی کمائی پر بھاری پڑ سکتی ہے۔ پھر بھی بارش اور بیماری کی رعایت بتاتی ہے کہ مشقت کے عین درمیان بھی رحمت بندے کی کمزوری کو ملحوظ رکھتی ہے۔

نماز ختم ہوئی تو ذکر ختم نہیں ہوا؛ یہی اس رکوع کی کنجی ہے: فَاِذَا قَضَيْتُمُ الصَّلٰوةَ فَاذْكُرُوا اللّٰهَ قِيٰمًا وَّقُعُوْدًا وَّعَلٰى جُنُوْبِكُمْ۔ “پھر جب تم نماز ادا کر چکو تو اللہ کو یاد کرو، کھڑے بھی، بیٹھے بھی اور اپنی کروٹوں پر لیٹے بھی۔” ظاہر میں یہ صلوٰۃ الخوف کے بعد کا حکم ہے: میدان میں نماز کی صورت مختصر ہوئی تو ذکر سے اس کمی کو بھر دو۔ مفسرین حضرت عبداللہ بن عباس رضی اللہ عنہما سے ایک قول نقل کرتے ہیں جس کا مفہوم یہ ہے کہ اللہ نے ہر فرض کی ایک حد رکھی اور عذر کی حالت میں اس میں رعایت دی، سوائے ذکر کے؛ اس کی کوئی حد نہیں اور کسی کو اسے چھوڑنے کا عذر نہیں: رات اور دن میں، خشکی اور سمندر میں، سفر اور حضر میں، تنگی اور فراخی میں، بیماری اور صحت میں۔ کھڑا ہونا، بیٹھنا اور لیٹنا انسانی جسم کی تمام حالتوں کو گھیر لیتے ہیں؛ گویا کہا گیا کہ تمہاری کوئی حالت ذکر سے خالی نہ رہے۔ اہلِ دل کی یہی صفت سورۂ آلِ عمران میں بھی آئی: الَّذِيْنَ يَذْكُرُوْنَ اللّٰهَ قِيٰمًا وَّقُعُوْدًا وَّعَلٰى جُنُوْبِهِمْ۔ اہلِ معرفت ذکر کے مراتب بیان کرتے ہیں: زبان کا ذکر، دل کا ذکر، اور سرّ کا ذکر، جہاں ذاکر اپنے ذکر کو بھی بھول کر مذکور میں گم ہو جاتا ہے۔ شیخِ اکبر کے مکتب میں ذکر کی اصل یہ ہے کہ بندے کا یاد کرنا حق کے یاد کرنے کا جواب ہے، جیسا کہ فرمایا: فَاذْكُرُوْنِيْۤ اَذْكُرْكُمْ۔ جو ہر حال میں یاد کرتا ہے وہ ہر حال میں یاد کیا جاتا ہے؛ اور اس کا کھڑا ہونا، بیٹھنا اور سونا سب حضوری بن جاتا ہے۔

پھر اطمینان کا وقت آیا تو نماز اپنی پوری صورت میں لوٹ آئی: فَاِذَا اطْمَاْنَنْتُمْ فَاَقِيْمُوا الصَّلٰوةَ ۚ اِنَّ الصَّلٰوةَ كَانَتْ عَلَى الْمُؤْمِنِيْنَ كِتٰبًا مَّوْقُوْتًا۔ “پھر جب تمہیں اطمینان ہو جائے تو نماز کو پوری طرح قائم کرو؛ بے شک نماز مومنوں پر مقررہ اوقات کے ساتھ فرض ہے۔” ظاہر میں مطلب یہ ہے کہ خوف اور سفر ختم ہو جائے تو تمام ارکان اور پوری رکعتوں کے ساتھ نماز ادا کرو۔ اور موقوت کا معنی وقت کا پابند فرض ہے؛ سنن ابی داؤد اور جامع ترمذی میں حضرت جبریل علیہ السلام کی امامت کی روایت کا مفہوم ہے کہ انہوں نے دو دن حضورِ اکرم صلی اللہ علیہ وسلم کو اول اور آخر وقت میں نمازیں پڑھا کر ہر نماز کا وقت متعین کیا۔ اور صحیحین میں حضرت عبداللہ بن مسعود رضی اللہ عنہ کی روایت کا مفہوم ہے کہ اللہ کو سب سے محبوب عمل وقت پر نماز ہے۔ شیخِ اکبر نے فتوحاتِ مکیہ میں نماز کے اسرار اور اس کے اوقات پر طویل کلام کیا ہے، اور اس مکتب کا ایک لطیف نکتہ یہ ہے کہ ذکر مطلق رکھا گیا اور نماز وقت سے مقید؛ بندہ اس ذات کی عبادت کرتا ہے جو ہر قید سے پاک ہے، مگر وہ عبادت دن رات کی مقررہ گھڑیوں میں ادا ہوتی ہے۔ یہی تنزیہ اور تشبیہ کا جمع ہے: اطلاق کی طرف رخ، اور قید میں ادائیگی۔ اور اطمینان کا لفظ اہلِ ذوق کو اس نفس کی یاد دلاتا ہے جسے سورۂ الفجر میں يٰۤاَيَّتُهَا النَّفْسُ الْمُطْمَئِنَّةُ کہہ کر پکارا گیا؛ جب دل کو قرار ملتا ہے تو نماز بھی قائم ہو جاتی ہے، یعنی سیدھی کھڑی ہو جاتی ہے۔ اور سورۂ المعارج نے جن نمازیوں کو اپنی نماز پر دائم کہا، اہلِ معرفت کے نزدیک ان کا دوام یہی ہے کہ وقت پر نماز اور ہر وقت ذکر۔

رکوع کا آخری حکم میدان کی ہمت سے متعلق ہے: وَلَا تَهِنُوْا فِي ابْتِغَآءِ الْقَوْمِ ۭ اِنْ تَكُوْنُوْا تَاْلَمُوْنَ فَاِنَّهُمْ يَاْلَمُوْنَ كَمَا تَاْلَمُوْنَ ۚ وَتَرْجُوْنَ مِنَ اللّٰهِ مَا لَا يَرْجُوْنَ ۭ وَكَانَ اللّٰهُ عَلِيْمًا حَكِيْمًا۔ “اور اس دشمن قوم کا پیچھا کرنے میں ہمت نہ ہارو؛ اگر تمہیں تکلیف پہنچتی ہے تو انہیں بھی ویسی ہی تکلیف پہنچتی ہے جیسی تمہیں پہنچتی ہے، اور تم اللہ سے اس چیز کی امید رکھتے ہو جس کی وہ امید نہیں رکھتے؛ اور اللہ سب کچھ جاننے والا، حکمت والا ہے۔” بعض مفسرین نے اسے احد کے بعد کے حالات سے جوڑا ہے، جب حضورِ اکرم صلی اللہ علیہ وسلم نے زخمی صحابہ کرام رضی اللہ عنہم کو دشمن کے تعاقب میں حمراء الاسد تک نکلنے کا حکم دیا۔ یہ ایک جاری جنگ کا حکم تھا جو حضورِ اکرم صلی اللہ علیہ وسلم کی قیادت میں دفاع کے لیے لڑی گئی؛ اسے آج کسی قوم پر چسپاں کرنا آیت کے ساتھ ناانصافی ہے۔ اس آیت کا گہرا نکتہ یہ ہے کہ درد دونوں طرف برابر ہے؛ فرق درد میں نہیں، امید میں ہے۔ اہلِ معرفت کہتے ہیں کہ مومن کا درد ایک کھڑکی رکھتا ہے جس سے اسمِ حکیم کی روشنی آتی ہے، اور غافل کا درد ایک بند کمرہ ہے۔ شیخِ اکبر کے مکتب میں رجا سبقتِ رحمت کی پہچان ہے؛ اور حدیثِ قدسی کا مفہوم ہے کہ میں اپنے بندے کے گمان کے ساتھ ہوں جو وہ مجھ سے رکھتا ہے۔ اہلِ تاویل اس آیت کو سالک کی طرف بھی موڑتے ہیں کہ مجاہدے میں تھکن آئے تو ہمت نہ ہارو؛ نفس بھی تھکتا ہے، مگر تمہارے پاس ایک امید ہے جو اس کے پاس نہیں۔ یوں یہ رکوع سفر سے شروع ہو کر خوف، ذکر، اوقات اور امید پر ختم ہوا؛ ہر منزل پر ظاہر کا حکم اپنی جگہ اور باطن کا اشارہ اس کی گہرائی۔
