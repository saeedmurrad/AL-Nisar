اس رکوع میں قرآن ایک چوری کے مقدمے میں عدالت کے کٹہرے تک اترتا ہے، اور ایک بے گناہ کی صفائی میں آیات نازل ہوتی ہیں: اِنَّاۤ اَنْزَلْنَاۤ اِلَيْكَ الْكِتٰبَ بِالْحَقِّ لِتَحْكُمَ بَيْنَ النَّاسِ بِمَاۤ اَرٰىكَ اللّٰهُ ۭ وَلَا تَكُنْ لِّلْخَآئِنِيْنَ خَصِيْمًا۔ “بے شک ہم نے آپ صلی اللہ علیہ وسلم کی طرف یہ کتاب حق کے ساتھ اتاری ہے تاکہ آپ صلی اللہ علیہ وسلم لوگوں کے درمیان اس کے مطابق فیصلہ فرمائیں جو اللہ نے آپ صلی اللہ علیہ وسلم کو دکھایا ہے، اور آپ صلی اللہ علیہ وسلم خیانت کرنے والوں کی طرف سے جھگڑنے والے نہ بنیں۔” جامع ترمذی میں حضرت قتادہ بن نعمان رضی اللہ عنہ سے، اور امام ابنِ جریر طبری رحمۃ اللہ علیہ نے کئی طرق سے، جو واقعہ نقل کیا ہے اس کا خلاصہ یہ ہے کہ مدینہ میں بنو ابیرق کے ایک شخص نے، جس کا نام بعض روایات میں طعمہ بن ابیرق اور بعض میں بشیر آیا ہے، حضرت رفاعہ بن زید رضی اللہ عنہ کے گھر سے سامان اور ہتھیار چرائے۔ جب بات کھلنے لگی تو الزام ایک بے قصور شخص پر ڈال دیا گیا؛ مشہور روایات میں وہ ایک یہودی تھا جس کے گھر چرائی ہوئی زرہ رکھ دی گئی، اور بعض روایات میں ایک مسلمان کا ذکر ہے۔ چور کے قبیلے والوں نے حضورِ اکرم صلی اللہ علیہ وسلم کی خدمت میں آ کر اس کی صفائی دی اور ظاہر کی گواہیاں اسی کے حق میں جمع کر دیں؛ تب یہ آیات اتریں اور حقیقت کھول دی۔ اس واقعے کا سبق صاف ہے: قرآن نے اپنی جماعت کے ایک فرد کے مقابلے میں ایک ایسے بے گناہ کی صفائی دی جو اس جماعت سے باہر تھا۔ عدل قبیلہ اور مذہب نہیں دیکھتا، حق دیکھتا ہے۔ اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں بِمَاۤ اَرٰىكَ اللّٰهُ پر خاص توجہ دی گئی: فیصلے کی بنیاد اپنی رائے نہیں، وہ دکھانا ہے جو حق کی طرف سے ہو۔ انسانِ کامل وہ ہے جس کی نگاہ کو حق نے اپنے نور سے بینائی دی ہو؛ اور حدیثِ قربِ نوافل کا مفہوم یہی ہے کہ حق بندے کی آنکھ بن جاتا ہے جس سے وہ دیکھتا ہے، مگر حلول اور اتحاد کے بغیر۔

پھر حکم ہوا: وَّاسْتَغْفِرِ اللّٰهَ ۭ اِنَّ اللّٰهَ كَانَ غَفُوْرًا رَّحِيْمًا۔ “اور اللہ سے مغفرت طلب کیجیے؛ بے شک اللہ بہت بخشنے والا، نہایت رحم والا ہے۔” علما نے یہاں پورے ادب سے وضاحت کی ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم سے کوئی گناہ سرزد نہیں ہوا تھا۔ ظاہر کی گواہی پر فیصلے کی طرف رجحان شریعت کا اپنا طریقہ ہے، اور صحیحین میں حضرت ام سلمہ رضی اللہ عنہا کی روایت کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم نے فرمایا: میں ایک بشر ہوں، تم اپنے جھگڑے میرے پاس لاتے ہو، ہو سکتا ہے کوئی اپنی دلیل میں دوسرے سے زیادہ چرب زبان ہو اور میں جو سنوں اس کے مطابق فیصلہ کر دوں؛ تو جسے میں اس کے بھائی کے حق میں سے کچھ دے دوں وہ اسے نہ لے، کیونکہ میں اسے آگ کا ایک ٹکڑا دے رہا ہوں۔ اس لیے یہاں استغفار کا حکم درجات کی بلندی اور امت کی تعلیم کے لیے ہے، کہ قاضی اپنے ہر فیصلے کے بعد اپنے رب کی طرف لوٹے۔ صحیح بخاری میں ہے کہ آپ صلی اللہ علیہ وسلم دن میں ستر سے زیادہ بار استغفار فرماتے تھے۔ اہلِ معرفت کہتے ہیں کہ اکابر کا استغفار گناہ سے نہیں، ترقی سے ہوتا ہے: ہر لمحے حق کا نیا شہود پچھلے شہود کو کم دکھاتا ہے، اور بندہ اپنے کل کے حال سے بھی مغفرت مانگتا ہے۔ شیخِ اکبر کے مکتب میں غفر کا اصل معنی ڈھانپ لینا ہے؛ انبیاء علیہم السلام کے حق میں مغفرت وہ ڈھال ہے جو خطا کو ان تک پہنچنے ہی نہیں دیتی، اور یہی عصمت ہے۔

اب ان کی بات جو دوسروں کا مال چرا کر دراصل اپنی ہی جان لوٹتے ہیں: وَلَا تُجَادِلْ عَنِ الَّذِيْنَ يَخْتَانُوْنَ اَنْفُسَهُمْ ۭ اِنَّ اللّٰهَ لَا يُحِبُّ مَنْ كَانَ خَوَّانًا اَثِيْمًا۔ “اور ان لوگوں کی طرف سے نہ جھگڑیے جو اپنے آپ سے خیانت کرتے ہیں؛ بے شک اللہ کسی ایسے کو پسند نہیں کرتا جو بڑا خائن، بڑا گناہ گار ہو۔” غور کیجیے: چوری رفاعہ رضی اللہ عنہ کے گھر ہوئی تھی، مگر قرآن نے کہا کہ یہ لوگ اپنی جانوں سے خیانت کرتے ہیں۔ اہلِ معرفت کہتے ہیں کہ ہر خیانت کا پہلا شکار خائن خود ہے؛ مال لٹنے والے کا گیا، مگر امانت کا جوہر خائن کے اپنے اندر سے گیا۔ شیخِ اکبر کے نزدیک انسان کا نفس اس کے پاس سب سے پہلی امانت ہے؛ جو اسے جھوٹ اور چوری کی راہ پر ڈالتا ہے وہ اس امانت میں خیانت کرتا ہے جو حق نے اسے سونپی تھی۔ پھر خوّان اور اثیم دونوں مبالغے کے صیغے ہیں: یعنی وہ جو بار بار خیانت کرے اور اس پر اڑا رہے۔ اللہ کی ناپسندیدگی ایک لغزش پر نہیں، اس عادت پر ہے جو توبہ کا دروازہ خود بند کر دے۔ اس آیت سے فقہا نے یہ اصول بھی نکالا کہ جان بوجھ کر کسی مجرم کی وکالت کرنا اور اسے سچا ثابت کرنے کے لیے دلیلیں گھڑنا حرام ہے؛ وکیل، گواہ، رشتہ دار اور قبیلہ سب اس خطاب میں شامل ہیں۔

پھر ان کے دلوں کا حال کھولا گیا: يَّسْتَخْفُوْنَ مِنَ النَّاسِ وَلَا يَسْتَخْفُوْنَ مِنَ اللّٰهِ وَهُوَ مَعَهُمْ اِذْ يُبَيِّتُوْنَ مَا لَا يَرْضٰى مِنَ الْقَوْلِ ۭ وَكَانَ اللّٰهُ بِمَا يَعْمَلُوْنَ مُحِيْطًا۔ “وہ لوگوں سے چھپتے ہیں مگر اللہ سے نہیں چھپتے، حالانکہ وہ ان کے ساتھ ہوتا ہے جب وہ راتوں کو ایسی باتوں کے مشورے کرتے ہیں جو اسے پسند نہیں؛ اور اللہ ان کے سب کاموں کو گھیرے ہوئے ہے۔” استخفا چھپنا بھی ہے اور شرم کرنا بھی؛ یعنی لوگوں سے حیا کی، اللہ سے حیا نہ کی۔ یبیّتون رات کے مشوروں کو کہتے ہیں: بنو ابیرق نے راتوں کو بیٹھ کر یہ تدبیر سوچی تھی کہ الزام کس پر ڈالا جائے۔ جامع ترمذی کی روایت کا مفہوم ہے کہ اللہ سے ویسی حیا کرو جیسا اس سے حیا کا حق ہے۔ اور حدیثِ احسان کا دوسرا درجہ یہی ہے کہ اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے؛ ان لوگوں سے یہی درجہ چھن گیا تھا۔ وَهُوَ مَعَهُمْ پر شیخِ اکبر کا مکتب پوری تنزیہ کے ساتھ ٹھہرتا ہے: یہ معیت علم، قدرت اور احاطے کی ہے، حلول کی نہیں؛ حق ہر شے کے ساتھ ہے مگر کسی شے میں سمایا ہوا نہیں۔ یہی مضمون سورۂ المجادلہ میں آیا: مَا يَكُوْنُ مِنْ نَّجْوٰى ثَلٰثَةٍ اِلَّا هُوَ رَابِعُهُمْ۔ اور محیط کا معنی یہ ہے کہ کوئی رات، کوئی کونا اور کوئی سرگوشی اس کے علم کے دائرے سے باہر نہیں۔ اہلِ معرفت کہتے ہیں کہ مراقبہ اسی ایک آیت کا عملی نام ہے: جو جان لے کہ وہ ساتھ ہے، اس کی تنہائی بھی حاضری بن جاتی ہے۔

اب خطاب ان حمایتیوں کی طرف مڑا جو چور کی صفائی دینے آئے تھے: هٰۤاَنْتُمْ هٰۤؤُلَآءِ جٰدَلْتُمْ عَنْهُمْ فِي الْحَيٰوةِ الدُّنْيَا ۚ فَمَنْ يُّجَادِلُ اللّٰهَ عَنْهُمْ يَوْمَ الْقِيٰمَةِ اَمْ مَّنْ يَّكُوْنُ عَلَيْهِمْ وَكِيْلًا۔ “سنو! تم وہ لوگ ہو جنہوں نے دنیا کی زندگی میں ان کی طرف سے جھگڑا کیا؛ پھر قیامت کے دن ان کی طرف سے اللہ سے کون جھگڑے گا، یا کون ان کا کارساز ہوگا؟” دنیا کی عدالت میں چرب زبانی کام آ جاتی ہے، گواہیاں بنائی جا سکتی ہیں، قبیلے کا رعب چل جاتا ہے؛ مگر وہ دن آنے والا ہے جسے سورۂ الطارق نے يَوْمَ تُبْلَى السَّرَآئِرُ کہا، جب چھپی باتیں جانچی جائیں گی اور کوئی وکیل کسی کا بوجھ نہ اٹھا سکے گا۔ اہلِ معرفت کہتے ہیں کہ دنیا پردوں کا گھر ہے اور آخرت پردے اٹھنے کا؛ یہاں کی ہر دلیل جو حقیقت کو ڈھانپنے کے لیے دی گئی، وہاں خود گواہ بن جائے گی۔ پھر وکیل کے لفظ پر ٹھہریے۔ ان لوگوں نے اپنے قبیلے کو اپنا کارساز بنایا تھا، اور قرآن پوچھتا ہے کہ کل یہ کارساز کہاں ہوگا۔ شیخِ اکبر کے مکتب میں سرِّ قدر کا یہی خلاصہ ہے کہ بندہ جسے اپنا سہارا چنتا ہے، اسی کے سپرد ہوتا ہے؛ اور یہ بات اگلے رکوع کی آخری آیت میں قرآن خود ایک اصول کے طور پر بیان کرنے والا ہے۔ جس نے اللہ کو وکیل بنایا اس کا سہارا کبھی نہیں ٹوٹتا، اور جس نے مخلوق کو بنایا اس کا سہارا وہیں ٹوٹ جاتا ہے جہاں سب سے زیادہ ضرورت ہو۔

سخت تنبیہ کے بعد فوراً دروازہ کھولا گیا، اور یہ اس سورت کی سب سے امید افزا آیتوں میں سے ہے: وَمَنْ يَّعْمَلْ سُوْۤءًا اَوْ يَظْلِمْ نَفْسَهٗ ثُمَّ يَسْتَغْفِرِ اللّٰهَ يَجِدِ اللّٰهَ غَفُوْرًا رَّحِيْمًا۔ “اور جو کوئی برائی کرے یا اپنی جان پر ظلم کرے، پھر اللہ سے مغفرت مانگے، وہ اللہ کو بہت بخشنے والا، نہایت رحم والا پائے گا۔” مفسرین حضرت عبداللہ بن مسعود رضی اللہ عنہ سے نقل کرتے ہیں کہ سورۂ نساء کی چند آیتیں انہیں دنیا و مافیہا سے زیادہ محبوب تھیں، اور ان میں یہ آیت بھی شمار کی گئی ہے۔ برائی اور اپنی جان پر ظلم کے فرق میں مفسرین کہتے ہیں کہ پہلی وہ جس کا اثر دوسروں تک پہنچے اور دوسری وہ جو بندے اور اس کے رب کے درمیان رہے؛ دونوں کا علاج ایک ہے۔ اور یہ آیت اسی قصے کے بیچ آئی ہے: چور کے لیے بھی، اس کے حمایتیوں کے لیے بھی، اور ہر اس شخص کے لیے جو آج اس آیت کو پڑھ رہا ہے۔ پھر لفظ يَجِدِ اللّٰهَ پر غور کیجیے: یہ نہیں کہا کہ اللہ اسے بخش دے گا، بلکہ کہا کہ وہ اللہ کو بخشنے والا پائے گا۔ اللہ تو ہمیشہ سے غفور و رحیم ہے؛ بدلتا بندے کا پانا ہے۔ اہلِ معرفت کہتے ہیں کہ وجد اور وجود ایک ہی جڑ سے ہیں؛ جو استغفار کے دروازے پر کھڑا ہوا اس نے حق کو اسی اسم میں پایا جس کی طرف اس نے رخ کیا۔

شیخِ اکبر کے مکتب میں یہ آیت ایک بڑے اصول کا رحمت والا رخ ہے: اسماءِ الٰہیہ بندے کے رخ کے مطابق اپنا ظہور کرتے ہیں۔ جو گناہ میں ڈوبا رہے وہ اسمِ منتقم کا سامنا کرتا ہے، اور جو گناہ کے بعد پلٹ آئے اس کے سامنے اسمِ غفور ظاہر ہوتا ہے؛ حق ایک ہے، مگر بندہ جس دروازے پر دستک دیتا ہے وہی دروازہ کھلتا ہے۔ یہاں کوئی جبر نہیں، کوئی شرط اتنی بھاری نہیں کہ بندہ اسے اٹھا نہ سکے؛ صرف رخ بدلنا ہے۔ اور لفظ ثمّ بتاتا ہے کہ گناہ اور استغفار کے درمیان کتنا ہی فاصلہ ہو، جب تک سانس باقی ہے دروازہ بند نہیں۔ جامع ترمذی میں حدیثِ قدسی کا مفہوم ہے کہ اے ابنِ آدم! جب تک تو مجھ سے دعا کرتا رہے گا اور امید رکھے گا، میں تجھے بخشتا رہوں گا جو کچھ بھی تجھ سے ہوا، اور مجھے پروا نہیں۔ اہلِ ذوق کہتے ہیں کہ جو سچا استغفار کرتا ہے وہ تجددِ امثال کے راز سے ایک نئی خلق بن کر اٹھتا ہے؛ اس کا کل اس کے آج پر حاکم نہیں رہتا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں حقوق العباد کی ادائیگی اور ندامت شرط ہے، اور باطن میں یہ یقین کہ پلٹنے کی توفیق بھی اسی کی پکار کا جواب ہے۔

پھر گناہ کی حقیقت ایک جملے میں کھول دی گئی: وَمَنْ يَّكْسِبْ اِثْمًا فَاِنَّمَا يَكْسِبُهٗ عَلٰى نَفْسِهٖ ۭ وَكَانَ اللّٰهُ عَلِيْمًا حَكِيْمًا۔ “اور جو کوئی گناہ کماتا ہے تو وہ اسے اپنے ہی خلاف کماتا ہے؛ اور اللہ سب کچھ جاننے والا، حکمت والا ہے۔” کسب کا لفظ قرآن کی ایک بنیادی اصطلاح ہے: پیدا کرنا اللہ کا کام ہے، اور اختیار کر کے کمانا بندے کا۔ اسی سورت میں یہ اصول پہلے بیان ہو چکا: وَمَاۤ اَصَابَكَ مِنْ سَيِّئَةٍ فَمِنْ نَّفْسِكَ۔ گناہ سے اللہ کا کچھ نہیں بگڑتا، نہ کسی دوسرے کا بوجھ بڑھتا ہے؛ ہر گناہ لوٹ کر اپنے کمانے والے کے سینے میں جا بیٹھتا ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے، اور ہر عمل کا پھل اسی شاخ پر لگتا ہے جس نے اسے اگایا۔ اور آیت کا اختتام علیم و حکیم پر ہوا: علیم وہ جو ہر کمائی کو جانتا ہے، اور حکیم وہ جو ہر نتیجے کو اس کی ٹھیک جگہ رکھتا ہے۔ اہلِ معرفت کہتے ہیں کہ حکمت ہر شے کو اس کے مقام پر رکھنا ہے؛ اس لیے گناہ کا بوجھ کبھی بے گناہ کے کندھے پر نہیں پڑتا، خواہ دنیا کی عدالت دھوکا کھا جائے۔

اور اب وہ آیت جو اس واقعے کا عین فیصلہ ہے: وَمَنْ يَّكْسِبْ خَطِيْۤئَةً اَوْ اِثْمًا ثُمَّ يَرْمِ بِهٖ بَرِيْۤئًا فَقَدِ احْتَمَلَ بُهْتَانًا وَّاِثْمًا مُّبِيْنًا۔ “اور جو کوئی خطا یا گناہ کمائے پھر اسے کسی بے گناہ کے سر تھوپ دے، تو اس نے بہتان اور کھلے گناہ کا بوجھ اپنے اوپر لاد لیا۔” مفسرین نے خطیئہ اور اثم میں یہ فرق کیا کہ ایک چھوٹی یا بے ارادہ لغزش ہے اور دوسرا جان بوجھ کر کیا گیا گناہ؛ مگر جو بھی ہو، اسے کسی بے قصور پر ڈالنا اسے دوہرا کر دیتا ہے: اصل گناہ بھی، اور ایک بے گناہ کی عزت پر بہتان بھی۔ اہلِ تاویل اس آیت کو اپنے اندر کی طرف موڑتے ہیں۔ نفس کی پرانی عادت یہ ہے کہ اپنی خطا کسی اور کے سر ڈالے: حالات کے، لوگوں کے، یہاں تک کہ تقدیر کے۔ حضرت آدم علیہ السلام نے لغزش کے بعد کہا: رَبَّنَا ظَلَمْنَاۤ اَنْفُسَنَا، اور ابلیس نے کہا: فَبِمَاۤ اَغْوَيْتَنِيْ۔ ایک نے بوجھ اپنے کندھے پر لیا اور چن لیے گئے، دوسرے نے اپنا بوجھ اپنے رب پر ڈالا اور دور ہو گیا۔ یہی ادب حضرت ابراہیم علیہ السلام کی زبان پر تھا: وَاِذَا مَرِضْتُ فَهُوَ يَشْفِيْنِ؛ بیماری اپنی طرف، شفا رب کی طرف۔ شیخِ اکبر کے مکتب میں اہلِ ادب برائی کو اپنی طرف منسوب کرتے ہیں، اگرچہ خلق سب کی سب حق کی ہے؛ کیونکہ تقدیر کو اپنی خطا کا ذمہ دار ٹھہرانا بھی ایک بے گناہ پر تہمت ہے۔
