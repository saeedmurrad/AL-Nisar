پچھلا رکوع حضرت ابراہیم علیہ السلام کی خلت اور اللہ کے احاطے پر ختم ہوا تھا؛ اب گفتگو پھر گھر اور کمزوروں کے حقوق کی طرف لوٹتی ہے، گویا یہ بتانا مقصود ہو کہ خلیل کی ملت کا راستہ یتیم اور عورت کے حق سے ہو کر گزرتا ہے: وَيَسْتَفْتُوْنَكَ فِي النِّسَآءِ ۭ قُلِ اللّٰهُ يُفْتِيْكُمْ فِيْهِنَّ ۙ وَمَا يُتْلٰى عَلَيْكُمْ فِي الْكِتٰبِ فِيْ يَتٰمَى النِّسَآءِ الّٰتِيْ لَا تُؤْتُوْنَهُنَّ مَا كُتِبَ لَهُنَّ وَتَرْغَبُوْنَ اَنْ تَنْكِحُوْهُنَّ۔ “اور لوگ آپ صلی اللہ علیہ وسلم سے عورتوں کے بارے میں حکم پوچھتے ہیں؛ کہہ دیجیے: اللہ تمہیں ان کے بارے میں حکم دیتا ہے، اور وہ آیات بھی جو تمہیں کتاب میں ان یتیم لڑکیوں کے بارے میں سنائی جاتی ہیں جنہیں تم وہ نہیں دیتے جو ان کے لیے مقرر کیا گیا ہے اور چاہتے ہو کہ ان سے نکاح کر لو۔” صحیح بخاری میں ام المؤمنین حضرت عائشہ صدیقہ رضی اللہ عنہا کی تفسیر کا مفہوم ہے کہ یتیم لڑکی کسی سرپرست کی پرورش میں ہوتی اور اس کے مال میں شریک ہوتی؛ سرپرست اس کے مال اور حسن کی وجہ سے اس سے نکاح کرنا چاہتا مگر پورا مہر نہ دیتا، اور اگر اسے پسند نہ کرتا تو مال کی خاطر اسے کسی اور سے نکاح بھی نہ کرنے دیتا۔ قرآن نے دونوں راستے بند کیے، اور یہ وہی حکم ہے جو سورت کی ابتدا میں آیا تھا: وَاِنْ خِفْتُمْ اَلَّا تُقْسِطُوْا فِي الْيَتٰمٰى۔ اہلِ معرفت کہتے ہیں کہ سوال لوگوں نے حضورِ اکرم صلی اللہ علیہ وسلم سے کیا تھا، مگر جواب کی نسبت حق تعالیٰ نے اپنی طرف فرمائی: قُلِ اللّٰهُ يُفْتِيْكُمْ۔ عورت کا معاملہ اتنا نازک اور اتنا معزز ہے کہ اس کا فیصلہ کسی رسم یا کسی سرپرست کی مرضی پر نہیں چھوڑا گیا۔ پھر مَا كُتِبَ لَهُنَّ کے الفاظ دیکھیے: ان کا حق لکھا جا چکا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر عینِ ثابتہ کا حق اس کی استعداد کے ساتھ علمِ الٰہی میں ثبت ہے؛ جو کسی کا لکھا ہوا حق روکتا ہے وہ حق تعالیٰ کی تقسیم میں دخل دیتا ہے۔

پھر اسی آیت کا اگلا حصہ کمزوروں کی پوری صف کو سامنے لے آتا ہے: وَالْمُسْتَضْعَفِيْنَ مِنَ الْوِلْدَانِ ۙ وَاَنْ تَقُوْمُوْا لِلْيَتٰمٰى بِالْقِسْطِ ۭ وَمَا تَفْعَلُوْا مِنْ خَيْرٍ فَاِنَّ اللّٰهَ كَانَ بِهٖ عَلِيْمًا۔ “اور کمزور بچوں کے بارے میں بھی، اور یہ کہ یتیموں کے ساتھ انصاف پر قائم رہو؛ اور جو بھلائی بھی تم کرو گے، اللہ اسے خوب جانتا ہے۔” جاہلیت میں میراث صرف انہیں ملتی جو ہتھیار اٹھا سکتے؛ عورتیں اور چھوٹے بچے محروم رہتے۔ اس سورت نے ابتدا ہی میں ان کے حصے مقرر کیے، اور یہاں پھر یاد دلایا کہ وہ حکم کتاب میں پڑھا جاتا ہے اور پڑھا جاتا رہے گا۔ تَقُوْمُوْا کا لفظ قابلِ غور ہے: صرف حق دے دینا کافی نہیں، انصاف کے لیے کھڑے ہونا ہے، یتیم کا وکیل بننا ہے۔ صحیح بخاری کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے شہادت کی اور درمیانی انگلی سے اشارہ کر کے فرمایا: میں اور یتیم کی کفالت کرنے والا جنت میں یوں ہوں گے۔ اور آیت کا آخری جملہ اس بھلائی کے لیے ہے جو کوئی نہیں دیکھتا: یتیم کے سر پر رکھا ہاتھ، رات کو اٹھ کر اس کی خبر لینا؛ وہ سب اللہ کے علم میں ہے۔ شیخِ اکبر کے نزدیک ہر سرپرست اسمِ رب کا ایک مظہر ہے، کیونکہ ربوبیت پرورش اور تربیت ہی کا نام ہے؛ جو یتیم کی پرورش کرتا ہے وہ اس اسم کا رنگ لیتا ہے، اور جو یتیم کا حق کھاتا ہے وہ اس منصب سے خیانت کرتا ہے جس پر حق نے اسے بٹھایا تھا۔

اب وہ آیت جو گھر کے اندر بگڑتے رشتے کا علاج بتاتی ہے: وَاِنِ امْرَاَةٌ خَافَتْ مِنْۢ بَعْلِهَا نُشُوْزًا اَوْ اِعْرَاضًا فَلَا جُنَاحَ عَلَيْهِمَآ اَنْ يُّصْلِحَا بَيْنَهُمَا صُلْحًا ۭ وَالصُّلْحُ خَيْرٌ۔ “اور اگر کسی عورت کو اپنے شوہر کی طرف سے زیادتی یا بے رخی کا اندیشہ ہو تو ان دونوں پر کوئی گناہ نہیں کہ آپس میں کسی طرح صلح کر لیں، اور صلح بہتر ہے۔” پہلی بات یہ کہ قرآن نے عورت کے اندیشے کو اہمیت دی اور پہل کا اختیار اسے دیا؛ اسے کسی کے فیصلے کا انتظار کرنے والی نہیں بنایا۔ دوسری بات یہ کہ نشوز کا لفظ یہاں شوہر کے لیے آیا، جیسے آیت ۳۴ میں بیوی کے لیے آیا تھا؛ یعنی زیادتی کا امکان دونوں طرف ہے اور جواب دہی بھی دونوں کی ہے۔ صحیح بخاری میں حضرت عائشہ صدیقہ رضی اللہ عنہا کی تفسیر کا مفہوم ہے کہ یہ اس عورت کے بارے میں ہے جس کا شوہر اس کی طرف مائل نہ رہے اور جدا کرنا چاہے، اور وہ کہے کہ مجھے اپنے نکاح میں رکھو اور میرے بعض حقوق سے تم آزاد ہو۔ اور ام المؤمنین حضرت سودہ رضی اللہ عنہا نے اپنی باری حضرت عائشہ صدیقہ رضی اللہ عنہا کو ہبہ کر دی، اور جامع ترمذی کی ایک روایت میں اسی کو اس آیت کا پس منظر بتایا گیا۔ یہ ان کا اپنا آزادانہ فیصلہ تھا، اس آرزو سے کہ وہ حضورِ اکرم صلی اللہ علیہ وسلم کے گھرانے میں امہات المؤمنین کے ساتھ رہیں؛ اور آپ صلی اللہ علیہ وسلم نے ان کی خواہش کا احترام کیا۔ علما نے واضح کیا کہ یہ رعایت عورت کا اختیار ہے؛ اگر دباؤ سے لی جائے تو صلح نہیں، ظلم ہے۔

پھر وہ جملہ جو اس رکوع کا دل ہے اور اس کے ساتھ نفس کی ایک کمزوری کا اعتراف: وَاُحْضِرَتِ الْاَنْفُسُ الشُّحَّ ۭ وَاِنْ تُحْسِنُوْا وَتَتَّقُوْا فَاِنَّ اللّٰهَ كَانَ بِمَا تَعْمَلُوْنَ خَبِيْرًا۔ “اور نفسوں کے سامنے حرص حاضر رکھی گئی ہے؛ اور اگر تم بھلائی کرو اور تقویٰ اختیار کرو تو اللہ تمہارے سب کاموں سے پوری طرح باخبر ہے۔” وَالصُّلْحُ خَيْرٌ ایک عام اصول ہے جو گھر سے نکل کر ہر جھگڑے پر حاوی ہے۔ سنن ابی داؤد اور جامع ترمذی کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: کیا میں تمہیں وہ عمل نہ بتاؤں جو نفلی روزے، نماز اور صدقے کے درجے سے بڑھ کر ہے؟ آپس کے تعلقات کو درست کرنا۔ اور شح وہ حرص ہے جو اپنا حق چھوڑنے نہیں دیتی؛ عورت اپنے حق پر اور مرد اپنے میلان پر اڑا رہتا ہے۔ آیت نے کسی ایک کو نہیں، سب نفسوں کو اس کا ذمہ دار کہا۔ سورۂ حشر میں اس سے بچنے والوں کو فلاح کی خوشخبری دی گئی: وَمَنْ يُّوْقَ شُحَّ نَفْسِهٖ فَاُولٰۤئِكَ هُمُ الْمُفْلِحُوْنَ۔ مفسرین کہتے ہیں کہ احسان اور تقویٰ کا خطاب خاص طور پر شوہر کو ہے: اگر اسے حق حاصل بھی ہو تو اس سے بڑھ کر بھلائی کرے۔ شیخِ اکبر کے مکتب میں صلح اسمِ السلام کا ایک ظہور ہے، اور دو حقوں کے درمیان وہ برزخ ہے جہاں کوئی دوسرے کو مٹاتا نہیں۔ اہلِ معرفت کہتے ہیں کہ اُحْضِرَتْ مجہول ہے: حرص نفس کے سامنے رکھ دی گئی ہے تاکہ آزمائش ہو؛ وہ بندے کی مجبوری نہیں، اس کا امتحان ہے، اور اسے چھوڑ دینا بندے کا اپنا انتخاب۔

اب قرآن انسانی دل کے بارے میں ایک ایسی سچائی کہتا ہے جسے کوئی انسانی قانون اتنی دیانت سے نہیں کہتا: وَلَنْ تَسْتَطِيْعُوْۤا اَنْ تَعْدِلُوْا بَيْنَ النِّسَآءِ وَلَوْ حَرَصْتُمْ۔ “اور تم ہرگز اس پر قادر نہ ہو گے کہ بیویوں کے درمیان پورا پورا عدل کرو، خواہ تم کتنا ہی چاہو۔” یہ آیت سورت کی تیسری آیت کی شرط کو ختم نہیں کرتی: فَاِنْ خِفْتُمْ اَلَّا تَعْدِلُوْا فَوَاحِدَةً؛ جسے عدل نہ کر سکنے کا اندیشہ ہو اس کے لیے ایک ہی بیوی کا حکم اپنی جگہ قائم ہے۔ علما نے دونوں آیتوں کو یوں جمع کیا کہ جو عدل انسان کے اختیار میں ہے، یعنی نفقہ، رہائش، باری، حسنِ سلوک اور برتاؤ، وہ فرض ہے؛ اور جو اختیار سے باہر ہے، یعنی دل کا میلان اور محبت کی کمی بیشی، اس پر گرفت نہیں۔ سنن ابی داؤد کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم باری میں پورا عدل فرماتے اور پھر دعا کرتے: اے اللہ، یہ میری تقسیم ہے اس میں جس کا میں مالک ہوں، تو مجھے اس میں ملامت نہ فرما جس کا تو مالک ہے اور میں نہیں۔ اور صحیح مسلم کی روایت کا مفہوم ہے کہ بنی آدم کے دل رحمٰن کی انگلیوں میں سے دو انگلیوں کے درمیان ہیں، وہ انہیں جدھر چاہے پھیرتا ہے۔ اہلِ معرفت کہتے ہیں کہ قلب کا نام ہی تقلب سے ہے؛ جو چیز ہر آن پلٹتی ہو اس پر پورا قابو کون رکھ سکتا ہے؟ شیخِ اکبر کے مکتب میں اپنے عجز کا اعتراف ادب کا پہلا زینہ ہے؛ جس نے کامل عدل کا دعویٰ کیا اس نے اپنے دل کو پہچانا ہی نہیں۔

مگر عجز کا اعتراف ظلم کا بہانہ نہیں بن سکتا، اس لیے فوراً حد کھینچ دی گئی: فَلَا تَمِيْلُوْا كُلَّ الْمَيْلِ فَتَذَرُوْهَا كَالْمُعَلَّقَةِ ۭ وَاِنْ تُصْلِحُوْا وَتَتَّقُوْا فَاِنَّ اللّٰهَ كَانَ غَفُوْرًا رَّحِيْمًا۔ “تو ایک ہی طرف پورے کے پورے نہ جھک جاؤ کہ دوسری کو بیچ میں لٹکتا چھوڑ دو؛ اور اگر تم اصلاح کرو اور تقویٰ اختیار کرو تو اللہ بڑا بخشنے والا، نہایت رحم والا ہے۔” معلقہ وہ ہے جو نہ بیوی کے حقوق پاتی ہو اور نہ آزاد ہو کر نئی زندگی شروع کر سکے؛ قرآن نے اس حالت کو حرام ٹھہرایا۔ سنن ابی داؤد اور جامع ترمذی کی روایت کا مفہوم ہے کہ جس کی دو بیویاں ہوں اور وہ ایک ہی کی طرف جھک جائے، وہ قیامت کے دن اس حال میں آئے گا کہ اس کا ایک پہلو جھکا ہوا ہوگا۔ لفظ كُلَّ الْمَيْلِ پر غور کیجیے: دل کا کچھ جھکاؤ معاف کیا گیا، مگر پورا جھک جانا، یعنی برتاؤ اور حقوق میں بھی ایک طرف ہو جانا، منع ہوا۔ اہلِ معرفت یہاں ایک لطیف بات دیکھتے ہیں: پچھلے جملے میں کہا گیا کہ تم پورا عدل نہ کر سکو گے، اور اس جملے کا اختتام غفور پر ہوا۔ گویا جو کمی بندے کے بس سے باہر تھی، اسے حق کی مغفرت ڈھانپ لیتی ہے، بشرطیکہ بندہ اپنے بس کی حد تک اصلاح اور تقویٰ میں کوتاہی نہ کرے۔ شیخِ اکبر کے نزدیک یہی سبقتِ رحمت ہے: حکم انسان کی طاقت کے مطابق رکھا گیا، اور طاقت سے آگے کا خلا رحمت سے بھر دیا گیا۔

اور اگر صلح کی ہر کوشش کے بعد بھی ساتھ نہ نبھ سکے: وَاِنْ يَّتَفَرَّقَا يُغْنِ اللّٰهُ كُلًّا مِّنْ سَعَتِهٖ ۭ وَكَانَ اللّٰهُ وَاسِعًا حَكِيْمًا۔ “اور اگر دونوں جدا ہو جائیں تو اللہ اپنی وسعت سے ہر ایک کو بے نیاز کر دے گا، اور اللہ بڑی وسعت والا، حکمت والا ہے۔” اس آیت میں جدائی کے بعد کی زندگی کو بڑی عزت سے دیکھا گیا ہے۔ وعدہ دونوں کے لیے ہے: كُلًّا، ہر ایک کو؛ عورت کو کوئی بے سہارا اور کم تر نہیں سمجھ سکتا، اس کے لیے بھی وہی وسعت ہے جو مرد کے لیے۔ طلاق جائز ہے مگر انجام نہیں؛ قرآن نے دوسری جگہ جدائی کو بھی بھلائی کے ساتھ رکھنے کا حکم دیا: اَوْ تَسْرِيْحٌۢ بِاِحْسَانٍ۔ اہلِ معرفت کہتے ہیں کہ جو یہ سمجھے کہ اس کا رزق، اس کا سکون یا اس کی عزت کسی ایک انسان سے بندھی ہے، اس نے واسع کو تنگ سمجھا۔ شیخِ اکبر کے مکتب میں یہ اصول معروف ہے کہ تجلی میں تکرار نہیں؛ حق کی ہر عطا نئی ہے، اور ایک دروازہ بند ہونے کے بعد وہ وسعت ختم نہیں ہوتی جس سے دروازے کھلتے ہیں۔ اور اسمِ واسع کے ساتھ اسمِ حکیم آیا: وسعت بے حساب ہے مگر بے حکمت نہیں؛ جو کچھ اس نے لیا وہ حکمت سے لیا، اور جو دے گا وہ بھی حکمت سے دے گا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں جدائی کے احکام اپنی جگہ ہیں، اور باطن میں یہ اطمینان کہ بندے کا اصل سہارا کبھی جدا نہیں ہوتا۔

اب احکام کے بعد وہ وصیت آتی ہے جو تمام آسمانی کتابوں میں مشترک ہے: وَلِلّٰهِ مَا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ ۭ وَلَقَدْ وَصَّيْنَا الَّذِيْنَ اُوْتُوا الْكِتٰبَ مِنْ قَبْلِكُمْ وَاِيَّاكُمْ اَنِ اتَّقُوا اللّٰهَ ۭ وَاِنْ تَكْفُرُوْا فَاِنَّ لِلّٰهِ مَا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ ۭ وَكَانَ اللّٰهُ غَنِيًّا حَمِيْدًا۔ “اور اللہ ہی کا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمین میں ہے؛ اور ہم نے تم سے پہلے کتاب پانے والوں کو بھی اور تمہیں بھی یہی تاکید کی کہ اللہ سے ڈرو؛ اور اگر تم انکار کرو تو بے شک اللہ ہی کا ہے جو کچھ آسمانوں اور زمین میں ہے، اور اللہ بے نیاز، ہر تعریف کا سزاوار ہے۔” تقویٰ تمام انبیاء علیہم السلام کی امتوں کے لیے ایک ہی وصیت ہے؛ شریعتیں بدلیں، یہ وصیت نہ بدلی۔ اور دیکھیے کہ تقویٰ کا لفظ پچھلی دو آیتوں میں بھی آیا تھا، گھر کی صلح اور عدل کے ساتھ؛ اب وہی لفظ پوری انسانیت کی وصیت بن کر لوٹا، یعنی گھر کا تقویٰ دین کے تقویٰ سے الگ نہیں۔ اس مکتب میں تقویٰ کو وقایہ سے سمجھایا جاتا ہے: بندہ حق کو اپنی ڈھال بنائے، اور ادب یہ ہے کہ ہر کمال کو حق کی طرف منسوب کرے اور ہر کوتاہی کو اپنی طرف۔ پھر وَاِنْ تَكْفُرُوْا: انکار کا امکان کھلا رکھا گیا۔ حق چاہتا تو سب کو مجبور کر دیتا، مگر اس نے بندے کو انتخاب کا وقار بخشا؛ اور انکار سے اس کی بے نیازی میں کوئی کمی نہیں آتی، نقصان صرف انکار کرنے والے کا ہے۔ حمید وہ ہے جو تعریف کا سزاوار ہے خواہ کوئی تعریف کرے یا نہ کرے۔

اور تیسری بار وہی اعلان: وَلِلّٰهِ مَا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ ۭ وَكَفٰى بِاللّٰهِ وَكِيْلًا۔ “اور اللہ ہی کا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمین میں ہے، اور کارساز کے طور پر اللہ کافی ہے۔” مفسرین کہتے ہیں کہ یہ جملہ ہر بار ایک نئے معنی کے ساتھ جڑا: پہلی بار تقویٰ کی وصیت کے ساتھ، دوسری بار بے نیازی کے ساتھ، اور تیسری بار کارسازی کے ساتھ۔ نکاح، جدائی اور میراث کے احکام کے فوراً بعد اس اعلان کی تکرار ایک گہری تعلیم ہے: کوئی شوہر بیوی کا مالک نہیں، کوئی بیوی شوہر کی مالک نہیں، کوئی سرپرست یتیم کے مال کا مالک نہیں؛ ملک صرف اسی کا ہے۔ جس نے یہ جان لیا اس کے نفس کی وہ حرص ڈھیلی پڑ جاتی ہے جس کا ذکر صلح والی آیت میں تھا۔ اور وکیل کے ذکر میں ہر اس کے لیے تسلی ہے جو اکیلا رہ گیا: جدا ہونے والی عورت، بے سہارا یتیم، کمزور بچہ؛ جس کا کوئی کارساز نہ رہا، اللہ اس کا کارساز ہے۔ شیخِ اکبر کے مکتب میں توکل کا مطلب کوئی نیا انتظام نہیں، اس حقیقت کا اقرار ہے کہ وہ پہلے ہی سے ہر چیز کا کارساز ہے؛ بندہ صرف اپنے اس وہم سے دست بردار ہوتا ہے کہ معاملات اس کے ہاتھ میں تھے۔ یہی توحیدِ افعال کا ثمر ہے، اور یہی وہ سکون ہے جو تدبیر کے بعد دل میں اترتا ہے، تدبیر کی جگہ نہیں لیتا۔

پھر بے نیازی کا ایک اور پہلو: اِنْ يَّشَاْ يُذْهِبْكُمْ اَيُّهَا النَّاسُ وَيَاْتِ بِاٰخَرِيْنَ ۭ وَكَانَ اللّٰهُ عَلٰى ذٰلِكَ قَدِيْرًا۔ “لوگو! اگر وہ چاہے تو تمہیں لے جائے اور دوسروں کو لے آئے، اور اللہ اس پر پوری قدرت رکھتا ہے۔” ظاہر میں یہ تنبیہ ہے: کوئی قوم یہ نہ سمجھے کہ دین کی امانت اس کی جاگیر ہے؛ اگر وہ منہ موڑے تو اللہ اس کی جگہ اور لوگ لے آئے گا۔ اور یہاں بھی وہی حاکم اصول نمایاں ہے: وہ قادر ہے، چاہتا تو ہٹا دیتا یا زبردستی سیدھا کر دیتا، مگر اس نے مہلت اور اختیار دیا تاکہ لوٹنا بندے کا اپنا لوٹنا ہو۔ اہلِ معرفت اس آیت میں ایک اور حقیقت بھی دیکھتے ہیں جو ہر لمحے جاری ہے۔ شیخِ اکبر کے مکتب میں تجددِ امثال کا اصول یہ ہے کہ حق ہر آن خلق کو نئے سرے سے وجود دیتا ہے؛ ایک صورت جاتی ہے اور اس جیسی دوسری آتی ہے، اور تسلسل کی وجہ سے ہم اسے وہی سمجھتے ہیں۔ سورۂ قٓ میں اسی کی طرف اشارہ ہے: بَلْ هُمْ فِيْ لَبْسٍ مِّنْ خَلْقٍ جَدِيْدٍ۔ اس نظر سے لے جانا اور لے آنا ایک مستقبل کی دھمکی ہی نہیں، ہر سانس کی حقیقت بھی ہے۔ ظاہر کی تنبیہ اپنی جگہ قائم ہے، اور یہ اشارہ اس کی گہرائی ہے: جس کا وجود ہر لمحہ عطا ہو رہا ہو وہ کس بات پر اکڑے؟ یہی فقر کی جڑ ہے۔

رکوع کا اختتام بندے کی ہمت کو بلند کرنے والی آیت پر ہوتا ہے: مَنْ كَانَ يُرِيْدُ ثَوَابَ الدُّنْيَا فَعِنْدَ اللّٰهِ ثَوَابُ الدُّنْيَا وَالْاٰخِرَةِ ۭ وَكَانَ اللّٰهُ سَمِيْعًا بَصِيْرًا۔ “جو دنیا کا بدلہ چاہتا ہے تو اللہ کے پاس دنیا کا بدلہ بھی ہے اور آخرت کا بھی، اور اللہ سب کچھ سننے والا، سب کچھ دیکھنے والا ہے۔” ظاہر میں یہ ان لوگوں کو تنبیہ ہے جو دین کے کاموں میں بھی صرف دنیا کا فائدہ دیکھتے تھے: جب دونوں جہانوں کا خزانہ ایک ہی در پر ہے تو اپنی طلب کو آدھے پر کیوں روکتے ہو؟ اسی لیے قرآن نے مومن کو یہ دعا سکھائی: رَبَّنَآ اٰتِنَا فِي الدُّنْيَا حَسَنَةً وَّفِي الْاٰخِرَةِ حَسَنَةً۔ جامع ترمذی کی روایت کا مفہوم ہے کہ جس کی فکر آخرت ہو، اللہ اس کے دل میں بے نیازی رکھ دیتا ہے، اس کے بکھرے کام سمیٹ دیتا ہے، اور دنیا اس کے پاس جھک کر آتی ہے۔ اہلِ معرفت کہتے ہیں کہ اصل خزانہ دنیا یا آخرت کا بدلہ نہیں، عِنْدَ اللّٰهِ ہے؛ جو اس کو چاہے جس کے پاس سب کچھ ہے، وہ سب کچھ پا لیتا ہے۔ شیخِ اکبر کے نزدیک یہ بھی سرِّ قدر کا ایک رخ ہے: حق بندے کو اس کی طلب اور استعداد کے مطابق دیتا ہے؛ جس نے صرف دنیا مانگی اسے وہی ملی، اور جس نے حق کو مانگا اسے سب ملا۔ اور سمیع و بصیر پر اختتام: وہ نیت کی آہٹ سنتا ہے اور عمل کی صورت دیکھتا ہے۔ یوں یتیم کے حق سے شروع ہونے والا یہ رکوع ہمت کی بلندی پر ختم ہوا، اور درمیان میں صلح، عدل، جدائی کا وقار اور تقویٰ کی وصیت؛ ظاہر کا ہر حکم اپنی جگہ قائم ہے اور اشارہ اس کی گہرائی کھولتا ہے۔
