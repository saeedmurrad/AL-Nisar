پچھلے رکوع میں اہلِ کتاب کا یہ مطالبہ گزرا کہ آسمان سے ایک لکھی ہوئی کتاب اتار لائی جائے؛ یہ رکوع اس مطالبے کا جواب کسی مناظرے سے نہیں، ایک سلسلے سے دیتا ہے: اِنَّآ اَوْحَيْنَآ اِلَيْكَ كَمَآ اَوْحَيْنَآ اِلٰى نُوْحٍ وَّالنَّبِيّٖنَ مِنْ بَعْدِهٖ ۚ وَاَوْحَيْنَآ اِلٰۤى اِبْرٰهِيْمَ وَاِسْمٰعِيْلَ وَاِسْحٰقَ وَيَعْقُوْبَ وَالْاَسْبَاطِ وَعِيْسٰى وَاَيُّوْبَ وَيُوْنُسَ وَهٰرُوْنَ وَسُلَيْمٰنَ ۚ وَاٰتَيْنَا دَاوٗدَ زَبُوْرًا۔ “بے شک ہم نے آپ کی طرف اسی طرح وحی بھیجی جس طرح نوح علیہ السلام اور ان کے بعد کے نبیوں کی طرف بھیجی تھی، اور ہم نے ابراہیم، اسماعیل، اسحاق، یعقوب، اولادِ یعقوب، عیسیٰ، ایوب، یونس، ہارون اور سلیمان علیہم السلام کی طرف وحی بھیجی، اور داود علیہ السلام کو ہم نے زبور عطا کی۔” ظاہر میں بات صاف ہے: حضورِ اکرم صلی اللہ علیہ وسلم پر اترنے والی وحی کوئی نئی چیز نہیں، یہ وہی کلام ہے جو حضرت نوح علیہ السلام سے لے کر حضرت عیسیٰ علیہ السلام تک ہر سچے نبی کے دل پر اترتا رہا؛ تو جو ان میں سے کسی ایک کی وحی مانے اور اس آخری وحی کا انکار کرے، وہ اپنے ہی اقرار کو جھٹلاتا ہے۔ اسباط سے مراد حضرت یعقوب علیہ السلام کی اولاد کے وہ گھرانے ہیں جن میں نبی اٹھائے گئے، اور زبور حمد و مناجات کا وہ صحیفہ ہے جو حضرت داود علیہ السلام کو ملا۔ ترتیب زمانی نہیں، اور اہلِ علم کہتے ہیں کہ یہی اس بات کی دلیل ہے کہ وحی پانے میں سب ایک صف میں ہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہ آیت گویا فصوص الحکم کی فہرست ہے: ہر نبی ایک کلمہ ہے جس میں ایک خاص حکمت نگینے کی طرح جڑی ہے۔ وحی ایک، ظرف بہت؛ اور ہر ظرف نے اپنے عینِ ثابت کی استعداد کے مطابق لیا۔ حرفِ تشبیہ سرچشمے کی یکسانی بتاتا ہے، مرتبے کی برابری نہیں؛ کیونکہ اس مکتب میں حضورِ اکرم صلی اللہ علیہ وسلم وہ جامع کلمہ ہیں جن میں یہ سب حکمتیں اکٹھی ہو گئیں، اور حکمتِ فردیہ کا مفہوم یہی جامعیت ہے۔

پھر فرمایا کہ یہ فہرست مکمل نہیں: وَرُسُلًا قَدْ قَصَصْنٰهُمْ عَلَيْكَ مِنْ قَبْلُ وَرُسُلًا لَّمْ نَقْصُصْهُمْ عَلَيْكَ ۗ۔ “اور کچھ رسول ایسے ہیں جن کا حال ہم اس سے پہلے آپ کو سنا چکے ہیں، اور کچھ رسول ایسے ہیں جن کا حال ہم نے آپ کو نہیں سنایا۔” ظاہر میں اس سے دو باتیں نکلتی ہیں۔ ایک یہ کہ رسولوں کی گنتی ہمارے علم سے بڑی ہے، اور ایک مشہور روایت میں انبیا کی تعداد ایک لاکھ سے زیادہ بتائی گئی ہے؛ دوسری یہ کہ کسی قوم کے بارے میں یہ دعویٰ کرنا درست نہیں کہ حق تعالیٰ نے اسے کبھی کوئی ہادی نہیں بھیجا۔ مومن کا ادب یہ ہے کہ جن کے نام قرآن نے لیے ان پر تفصیل سے ایمان لائے، اور جن کے نام نہیں لیے ان پر اجمال سے؛ نہ کسی نبی کا انکار کرے، نہ کسی غیر نبی کو بے دلیل نبی کہے۔ اہلِ معرفت کہتے ہیں کہ جن رسولوں کے قصے بیان ہوئے وہ اسماءِ الٰہیہ کے ظاہر مظاہر ہیں، اور جن کے قصے بیان نہیں ہوئے وہ اس بات کی یاد دہانی ہیں کہ حق کے مظاہر ہماری گنتی میں نہیں سماتے۔ اس آیت میں عاجزی کا ایک سبق بھی ہے: جو علم ہمیں نہیں دیا گیا اس پر حکم لگانا ادب کے خلاف ہے، اور اہلِ اللہ اسی لیے اس خاموشی کو بھی کلام کا حصہ سمجھتے ہیں جسے حق نے بیان نہیں فرمایا۔ ظاہر کا ایمان اور باطن کی حیرت یہاں ایک ہو جاتے ہیں: ہم مانتے ہیں، مگر گھیر نہیں سکتے۔

اور اس سلسلے کے بیچ ایک خاص شرف کا ذکر آیا: وَكَلَّمَ اللّٰهُ مُوْسٰى تَكْلِيْمًا۔ “اور اللہ نے موسیٰ علیہ السلام سے کلام فرمایا، جیسا کلام فرمانے کا حق ہے۔” عربی قاعدے میں فعل کے بعد اسی کا مصدر لانا تاکید ہے؛ اہلِ سنت نے اسی سے یہ نکالا کہ یہ کلام حقیقی تھا، مجاز یا محض الہام نہیں۔ مگر اسی سانس میں تنزیہ لازم ہے: حق تعالیٰ کا کلام اس کی ازلی صفت ہے، ہماری طرح زبان، حلق، حرف اور آواز کا محتاج نہیں، نہ کسی سمت میں قید ہے۔ سورۂ الشوریٰ میں ہم نے پڑھا تھا: لَيْسَ كَمِثْلِهٖ شَيْءٌ ۚ وَهُوَ السَّمِيْعُ الْبَصِيْرُ؛ وہ سنتا اور بولتا ہے، مگر کسی کی مانند نہیں۔ اور سورۂ القصص میں گزرا کہ ندا درخت کی جانب سے آئی مگر کلام درخت میں اترا نہیں؛ درخت جہت تھا، بولنے والا نہیں۔ شیخِ اکبر کے نزدیک یہ تنزیہ اور تشبیہ کا جمع ہے: “کلام فرمایا” میں قرب کی تشبیہ ہے، اور تکلیما کی تاکید کے ساتھ بے مثلی کی تنزیہ۔ حکمتِ علویہ میں اس مکتب کا مفہوم یہ ہے کہ حضرت موسیٰ علیہ السلام اپنے گھر والوں کے لیے آگ لینے نکلے تھے، اور حق نے ان کی حاجت ہی کی صورت میں تجلی فرمائی؛ وہ آگ ڈھونڈنے گئے اور کلام پا لیا۔ اہلِ ذوق کہتے ہیں کہ جو بندہ اپنی سچی حاجت لے کر نکلتا ہے، حق اس کی حاجت ہی کو ملاقات کا دروازہ بنا دیتا ہے۔

اب رسولوں کے بھیجنے کی حکمت بیان ہوئی: رُسُلًا مُّبَشِّرِيْنَ وَمُنْذِرِيْنَ لِئَلَّا يَكُوْنَ لِلنَّاسِ عَلَى اللّٰهِ حُجَّةٌ بَعْدَ الرُّسُلِ ۗ وَكَانَ اللّٰهُ عَزِيْزًا حَكِيْمًا۔ “یہ سب رسول خوش خبری دینے والے اور ڈرانے والے بنا کر بھیجے گئے، تاکہ رسولوں کے آنے کے بعد لوگوں کے پاس اللہ کے مقابلے میں کوئی حجت نہ رہے؛ اور اللہ زبردست ہے، حکمت والا ہے۔” یہ آیت حق کے عدل کا اعلان ہے: کسی کو پکڑا نہیں جاتا جب تک بات اس تک پہنچ نہ جائے، اور کوئی کل یہ نہ کہہ سکے کہ مجھے تو بتایا ہی نہیں گیا تھا۔ ترتیب پر غور کیجیے: بشارت پہلے، ڈراوا بعد میں؛ اہلِ معرفت کے نزدیک یہ سبقتِ رحمت کا عکس ہے کہ پیغام کا پہلا لفظ امید ہے۔ پھر دو اسم آئے: عزیز اور حکیم۔ عزت کا تقاضا یہ ہو سکتا تھا کہ وہ بے خبر کو بھی پکڑ لے، کوئی اس سے پوچھنے والا نہیں؛ مگر حکمت اور رحمت نے یہ پسند کیا کہ پہلے رسول بھیجے، دلیل کھولے، راستہ دکھائے، اور پھر انتخاب بندے پر چھوڑ دے۔ شیخِ اکبر کے مکتب میں یہی بندے کے اختیار کی عزت ہے: حق زبردستی ہدایت پر لا سکتا تھا، مگر اس نے دعوت کو چنا تاکہ جو آئے اپنے رخ سے آئے، اور جو منہ پھیرے اس کا منہ پھیرنا اسی کا ہو۔ اہلِ ذوق اس میں یہ اشارہ بھی دیکھتے ہیں کہ دل کے اندر بھی ایک بشارت دینے والا اور ایک ڈرانے والا بیٹھا ہے، مگر اندر کی یہ آواز باہر کے رسول کا بدل نہیں، اس کی تصدیق ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

اور جب منکر گواہی نہ دیں تو سب سے بڑی گواہی سامنے آتی ہے: لٰكِنِ اللّٰهُ يَشْهَدُ بِمَآ اَنْزَلَ اِلَيْكَ اَنْزَلَهٗ بِعِلْمِهٖ ۚ وَالْمَلٰۤئِكَةُ يَشْهَدُوْنَ ۗ وَكَفٰى بِاللّٰهِ شَهِيْدًا۔ “لیکن اللہ گواہی دیتا ہے اس پر جو اس نے آپ کی طرف اتارا، اسے اپنے علم سے اتارا ہے، اور فرشتے بھی گواہی دیتے ہیں؛ اور اللہ گواہی کے لیے کافی ہے۔” حرفِ “لیکن” ایک تسلی ہے: اگر یہ لوگ نہیں مانتے تو آپ کا غم نہیں، کیونکہ جس نے بھیجا وہ خود گواہ ہے۔ اور حق کی گواہی کیا ہے؟ یہی کلام اپنی بلاغت، اپنی سچائی اور اپنی تاثیر کے ساتھ؛ اور وہ معجزات جو اس کے لانے والے کے ہاتھ پر ظاہر ہوئے۔ “اسے اپنے علم سے اتارا” کا ایک ظاہری معنی یہ ہے کہ اس میں وہ علم ہے جو اسی کے پاس تھا، اور دوسرا یہ کہ اس نے پوری آگاہی کے ساتھ اسے اتارا، کوئی لفظ بے خبری میں نہیں آیا۔ اہلِ معرفت کہتے ہیں کہ یہاں گواہ اور گواہی اور جس پر گواہی ہے، تینوں کا سرچشمہ ایک ہے: کلام اسی کا، علم اسی کا، گواہی اسی کی؛ قرآن اپنی دلیل آپ ہے۔ شیخِ اکبر کے نزدیک اسمِ شہید کا ایک مظہر وہ دل ہے جو قرآن پڑھتے ہوئے اس کی سچائی کو چکھ لیتا ہے؛ یہ ذوق اس گواہی کا ایک ذرہ ہے جو حق نے اپنے کلام پر دی۔ اسی لیے قاری جتنا اپنا دل صاف کرتا ہے، قرآن کی گواہی اس کے اندر اتنی ہی روشن ہوتی ہے۔

اب دوسرا رخ: اِنَّ الَّذِيْنَ كَفَرُوْا وَصَدُّوْا عَنْ سَبِيْلِ اللّٰهِ قَدْ ضَلُّوْا ضَلٰلًا بَعِيْدًا۔ “بے شک جن لوگوں نے کفر کیا اور اللہ کی راہ سے روکا، وہ گمراہی میں بہت دور نکل گئے۔” یہاں فعل کی نسبت پر رکیے: فعل کا فاعل وہ خود ہیں، وہ خود بھٹکے۔ پہلے انکار، پھر دوسروں کو روکنا، اور ان دونوں کا نتیجہ دوری؛ دوری کسی نے ان پر لادی نہیں، یہ ان کے اپنے قدموں کا فاصلہ ہے۔ اسی طرح آگے فرمایا: اِنَّ الَّذِيْنَ كَفَرُوْا وَظَلَمُوْا لَمْ يَكُنِ اللّٰهُ لِيَغْفِرَ لَهُمْ وَلَا لِيَهْدِيَهُمْ طَرِيْقًا اِلَّا طَرِيْقَ جَهَنَّمَ خٰلِدِيْنَ فِيْهَآ اَبَدًا ۚ وَكَانَ ذٰلِكَ عَلَى اللّٰهِ يَسِيْرًا۔ “بے شک جنہوں نے کفر کیا اور ظلم کیا، اللہ ایسا نہیں کہ انہیں بخش دے اور نہ یہ کہ انہیں کوئی راستہ دکھائے، سوائے جہنم کے راستے کے، جس میں وہ ہمیشہ ہمیشہ رہیں گے؛ اور یہ اللہ پر آسان ہے۔” اہلِ تفسیر کہتے ہیں کہ مراد وہ ہیں جو کفر اور ظلم پر اڑے رہے اور اسی حال میں دنیا سے گئے، ورنہ توبہ کا دروازہ آخری سانس تک کھلا ہے۔ یہ رکوع یہاں اس اصول کو دہراتا ہے جو اسی سورت کی ایک آیت میں قرآن نے خود بیان کیا: نُوَلِّهٖ مَا تَوَلّٰى وَنُصْلِهٖ جَهَنَّمَ، یعنی ہم اسے اسی طرف پھیر دیتے ہیں جدھر وہ خود پھرا۔

یہی اس رکوع میں حاکم اصول ہے، اور اسے صاف لفظوں میں کہنا ضروری ہے: حق تعالیٰ کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت کا طالب ہو اسے کبھی گمراہ نہیں کرتا۔ جو ڈھونڈتا ہے اسے ہدایت ملتی ہے؛ اور جو منہ پھیرتا ہے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے۔ “راستہ نہ دکھانا سوائے جہنم کے راستے کے” کا مطلب یہ نہیں کہ کسی بھلائی چاہنے والے کو دھکیل کر آگ کی طرف موڑ دیا گیا؛ مطلب یہ ہے کہ جس راستے کو بندے نے عمر بھر چنا، حق نے اسی کو وجود دے دیا، اور اسی کی منزل اس کے سامنے آ گئی۔ اہلِ تفسیر نے لکھا ہے کہ یہاں جہنم کے راستے کو “ہدایت” کے لفظ سے تعبیر کرنا اس طرز پر ہے کہ جس نے روشنی کو ٹھکرایا اس کے لیے یہی اس کا “راستہ” ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے، فیصلہ اس چیز کے اپنے حال کے تابع ہے جس پر فیصلہ ہوا۔ حق کا علم بندے کے حال کو جانتا ہے، اسے بناتا نہیں کہ بندہ مجبور ہو جائے؛ اور اگر وہ چاہتا تو سب کو زبردستی ہدایت پر جمع کر دیتا، مگر اس نے انتخاب کی عزت کو باقی رکھا۔ “یہ اللہ پر آسان ہے” کا مطلب یہ ہے کہ اس کا عدل کسی رکاوٹ کا محتاج نہیں؛ بندے نے جو راہ چنی، اس کی منزل تک پہنچانا اس کی قدرت پر کچھ بھاری نہیں۔ اور یاد رہے کہ ان ہی آیات کے ساتھ اگلی آیت پوری انسانیت کو پکار رہی ہے؛ دروازہ بند کرنے سے پہلے دعوت دی جاتی ہے۔

اور وہ دعوت یہ ہے: يٰۤاَيُّهَا النَّاسُ قَدْ جَآءَكُمُ الرَّسُوْلُ بِالْحَقِّ مِنْ رَّبِّكُمْ فَاٰمِنُوْا خَيْرًا لَّكُمْ ۚ وَاِنْ تَكْفُرُوْا فَاِنَّ لِلّٰهِ مَا فِي السَّمٰوٰتِ وَالْاَرْضِ ۚ وَكَانَ اللّٰهُ عَلِيْمًا حَكِيْمًا۔ “اے لوگو! یہ رسول تمہارے رب کی طرف سے حق لے کر تمہارے پاس آ گئے ہیں، تو ایمان لاؤ، یہ تمہارے لیے بہتر ہے؛ اور اگر تم انکار کرو تو آسمانوں اور زمین میں جو کچھ ہے سب اللہ ہی کا ہے؛ اور اللہ سب جاننے والا، حکمت والا ہے۔” خطاب “اے لوگو” سے ہے، کسی ایک قوم سے نہیں؛ حضورِ اکرم صلی اللہ علیہ وسلم کی رسالت سب کے لیے ہے۔ اور “تمہارے رب کی طرف سے” کہہ کر یاد دلایا کہ یہ پیغام کسی اجنبی کا نہیں، اسی کا ہے جس نے تمہیں پالا۔ “یہ تمہارے لیے بہتر ہے” میں فائدے کا رخ بندے کی طرف ہے: ایمان سے حق کو کچھ نہیں ملتا، بندے کو ملتا ہے۔ اسی لیے انکار کی صورت میں یہ نہیں کہا کہ تم حق کا نقصان کرو گے، بلکہ یہ کہا کہ آسمان و زمین سب اسی کے ہیں؛ یعنی وہ غنی ہے، تمہارا انکار اس کی بادشاہی میں سے کچھ نہیں گھٹاتا۔ اہلِ معرفت کہتے ہیں کہ یہ غنا ہی دعوت کی پاکیزگی کی دلیل ہے: جو پکارنے والا خود کسی چیز کا محتاج نہ ہو، اس کی پکار سراسر رحمت ہوتی ہے۔ اس مکتب میں عالم کا ہر ذرہ اسی کا مظہر ہے، تو جو انکار کرتا ہے وہ بھی اسی کی ملکیت میں رہ کر انکار کرتا ہے؛ اور علیم و حکیم کے اسماء یہ بتاتے ہیں کہ وہ ہر دل کا رخ جانتا ہے اور ہر ایک کے ساتھ اس کے رخ کے مطابق حکمت سے معاملہ کرتا ہے۔

اب رکوع اپنے سب سے نازک اور سب سے بلند مقام پر پہنچتا ہے: يٰۤاَهْلَ الْكِتٰبِ لَا تَغْلُوْا فِيْ دِيْنِكُمْ وَلَا تَقُوْلُوْا عَلَى اللّٰهِ اِلَّا الْحَقَّ ۚ۔ “اے اہلِ کتاب! اپنے دین میں حد سے نہ بڑھو، اور اللہ کے بارے میں حق کے سوا کچھ نہ کہو۔” خطاب کے ادب پر غور کیجیے: انہیں “کتاب والے” کہہ کر پکارا گیا، یعنی ان کی نسبت آسمانی کتاب سے یاد دلائی گئی۔ غلو کا مطلب ہے کسی چیز کو اس کی حد سے آگے لے جانا، اور حضرت عیسیٰ علیہ السلام کے معاملے میں غلو دونوں طرف ہوا: ایک گروہ نے ان کی شان گھٹائی اور حضرت مریم علیہا السلام پر بہتان باندھا، دوسرے گروہ نے انہیں بندگی کے مقام سے اٹھا کر خدائی کے مقام پر بٹھا دیا۔ قرآن دونوں کو رد کرتا ہے اور درمیانی سچائی بتاتا ہے۔ اور یہ تنبیہ صرف ان کے لیے نہیں؛ حضورِ اکرم صلی اللہ علیہ وسلم نے اپنی امت کو بھی اسی سے روکا، صحیح بخاری کی حدیث کا مفہوم ہے کہ میری تعریف میں ویسا مبالغہ نہ کرو جیسا نصاریٰ نے حضرت عیسیٰ ابنِ مریم علیہ السلام کے بارے میں کیا، میں تو اللہ کا بندہ ہوں، پس کہو: اللہ کا بندہ اور اس کا رسول۔ اہلِ معرفت کہتے ہیں کہ غلو محبت کا بگاڑ ہے، محبت نہیں: سچی محبت محبوب کو اسی مقام پر رکھتی ہے جو حق نے اسے دیا، اور اس سے اوپر لے جانا دراصل اسے اس کے اصل شرف سے محروم کرنا ہے، کیونکہ نبی کا سب سے بڑا شرف یہی ہے کہ وہ کامل بندہ ہے۔ اور “اللہ پر حق کے سوا کچھ نہ کہو” میں یہ ادب ہے کہ ذاتِ حق کے بارے میں ہر بات وحی کی سند سے کہی جائے، خیال اور جذبے کی بنیاد پر نہیں۔

پھر وہ سچائی بیان ہوئی جس پر غلو کا علاج ہے: اِنَّمَا الْمَسِيْحُ عِيْسَى ابْنُ مَرْيَمَ رَسُوْلُ اللّٰهِ وَكَلِمَتُهٗ ۚ اَلْقٰهَآ اِلٰى مَرْيَمَ وَرُوْحٌ مِّنْهُ۔ “مسیح عیسیٰ ابنِ مریم علیہ السلام تو بس اللہ کے رسول ہیں اور اس کا کلمہ ہیں جسے اس نے مریم علیہا السلام کی طرف ڈالا، اور اس کی طرف سے ایک روح ہیں۔” پہلا لفظ انما ہے، یعنی بس یہی، اس سے زیادہ نہیں۔ نسبت ماں کی طرف ہے: ابنِ مریم، یعنی حضرت مریم علیہا السلام کے بیٹے؛ اور یہ نسبت خود اعلان ہے کہ وہ ایک خاتون کے بطن سے پیدا ہوئے، کھاتے پیتے انسان تھے۔ پھر سب سے پہلے “اللہ کے رسول” کہا، کیونکہ ان کا اصل مقام رسالت ہے۔ “کلمہ” کا ظاہری معنی جمہور مفسرین کے نزدیک یہ ہے کہ وہ کلمۂ کن سے بغیر باپ کے پیدا ہوئے، جیسا کہ فرمایا: اِنَّ مَثَلَ عِيْسٰى عِنْدَ اللّٰهِ كَمَثَلِ اٰدَمَ؛ اور حضرت آدم علیہ السلام تو نہ باپ رکھتے تھے نہ ماں، پھر بھی کوئی انہیں خدا نہیں کہتا۔ شیخِ اکبر کے مکتب میں ہر موجود حق کا ایک کلمہ ہے، کیونکہ ہر شے کن سے ہے؛ مگر عام طور پر یہ کلمہ اسباب کے پردے میں چھپا رہتا ہے۔ حضرت عیسیٰ علیہ السلام میں باپ کا واسطہ اٹھا لیا گیا تو کن بے پردہ ظاہر ہوا، اس لیے انہیں خاص طور پر کلمہ کہا گیا۔ حکمتِ نبویہ میں اس مکتب کا مفہوم یہ ہے کہ ان کا وجود حضرت جبریل علیہ السلام کے پھونکے ہوئے نفخ اور حضرت مریم علیہا السلام کی پاکیزگی سے بنا، اور ان کے ہاتھ پر جو مردے زندہ ہوئے وہ ہر بار “اللہ کے حکم سے” ہوئے؛ قرآن کا یہ قید بار بار لگانا ہی وہ دیوار ہے جو مظہر کو ظاہر سمجھ لینے کی غلطی سے روکتی ہے۔ اور حضرت مریم علیہا السلام کا ذکر یہاں پوری عزت کے ساتھ ہے: کلمہ “ان کی طرف ڈالا گیا”، یعنی ان کی پاکیزگی اس امانت کی اہل ٹھہری۔

اب “اور اس کی طرف سے ایک روح” پر پوری تنزیہ کے ساتھ ٹھہرنا ضروری ہے، کیونکہ یہیں سب سے بڑی لغزش ہوئی۔ عربی میں حرفِ من کبھی حصہ بتاتا ہے اور کبھی آغاز اور سرچشمہ؛ یہاں جمہور اہلِ علم کے نزدیک دوسرا معنی ہے: روح اس کی طرف سے آئی، اس کا حصہ نہیں۔ اس کی دلیل خود قرآن میں ہے: وَسَخَّرَ لَكُمْ مَّا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ جَمِيْعًا مِّنْهُ؛ آسمان و زمین کی ہر چیز “اس کی طرف سے” ہے، اور کوئی عقل والا یہ نہیں کہتا کہ زمین کے پتھر خدا کے ٹکڑے ہیں۔ اور حضرت آدم علیہ السلام کے بارے میں فرمایا: وَنَفَخْتُ فِيْهِ مِنْ رُّوْحِيْ؛ یہ اضافتِ تشریفی ہے، جیسے کعبے کو بیت اللہ اور اونٹنی کو ناقۃ اللہ کہا گیا: نسبت عزت کی ہے، جزئیت کی نہیں۔ پس حضرت عیسیٰ علیہ السلام پیدا کی ہوئی روح ہیں جسے حق نے عزت کی نسبت بخشی۔ یہاں یہ بھی صاف کہنا لازم ہے کہ شیخِ اکبر کا مکتب حلول اور اتحاد دونوں کا سخت منکر ہے: نہ حق کسی بدن میں اترتا ہے، نہ کوئی مخلوق حق بن جاتی ہے۔ اس مکتب میں ہر شے حق کا مظہر ہے، اور ٹھیک اسی لیے کسی ایک شخص کو خدا کہنا توحید نہیں، توحید کا الٹ ہے: یہ لامحدود کو ایک محدود صورت میں قید کرنا ہے۔ اہلِ معرفت کہتے ہیں کہ جس نے حضرت عیسیٰ علیہ السلام کے معجزوں میں حق کی قدرت دیکھی اس نے ٹھیک دیکھا، مگر جس نے کہا کہ قدرت والا یہی ہے اس نے آئینے کو چہرہ سمجھ لیا۔ تنزیہ کہتی ہے کہ وہ کسی صورت میں نہیں سماتا، اور تشبیہ کہتی ہے کہ ہر صورت اسی کی نشانی ہے؛ اور ان دونوں کے جمع کا نام ادب ہے۔

اور اب وہ نتیجہ جس پر آیت ختم ہوتی ہے: فَاٰمِنُوْا بِاللّٰهِ وَرُسُلِهٖ ۚ وَلَا تَقُوْلُوْا ثَلٰثَةٌ ۚ اِنْتَهُوْا خَيْرًا لَّكُمْ ۚ اِنَّمَا اللّٰهُ اِلٰهٌ وَّاحِدٌ ۚ سُبْحٰنَهٗۤ اَنْ يَّكُوْنَ لَهٗ وَلَدٌ ۘ لَهٗ مَا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ ۗ وَكَفٰى بِاللّٰهِ وَكِيْلًا۔ “تو اللہ پر اور اس کے سب رسولوں پر ایمان لاؤ، اور یہ نہ کہو کہ تین ہیں؛ باز آ جاؤ، یہ تمہارے لیے بہتر ہے۔ اللہ تو بس ایک ہی معبود ہے؛ وہ اس سے پاک ہے کہ اس کا کوئی بیٹا ہو؛ آسمانوں اور زمین میں جو کچھ ہے سب اسی کا ہے؛ اور اللہ کارساز ہونے میں کافی ہے۔” رد کا انداز دیکھیے: گالی نہیں، طعنہ نہیں، بلکہ “باز آ جاؤ، یہ تمہارے لیے بہتر ہے”، یعنی منع کرنے میں بھی خیر خواہی۔ دلیل تین قدموں میں ہے۔ پہلا: معبود ایک ہے، اور یہ ایک گنتی کا ایک نہیں جس کے بعد دو اور تین آئیں؛ سورۂ الاخلاص میں ہم نے دیکھا کہ وہ احد اور صمد ہے، نہ اس نے جنا نہ وہ جنا گیا۔ دوسرا: لفظِ سبحانہ، یعنی بیٹا ہونا محتاجی، ہم جنس ہونے اور تقسیم کو چاہتا ہے، اور وہ ان سب سے پاک ہے۔ تیسرا: آسمان و زمین کی ہر چیز اس کی ملکیت ہے، حضرت عیسیٰ علیہ السلام بھی ان ہی میں ہیں، اور جو ملکیت ہو وہ بیٹا اور شریک نہیں ہو سکتا۔ آخر میں اسمِ وکیل بتاتا ہے کہ اسے نہ کسی وارث کی حاجت ہے نہ کسی مددگار کی۔ شیخِ اکبر کے نزدیک وحدتِ حق عدد کی وحدت نہیں، یہ وہ یکتائی ہے جس سے سب اعداد نکلتے ہیں اور جو خود کسی گنتی میں نہیں آتی؛ اسی لیے تین کہنا اس یکتائی کو گنتی کی صف میں اتار دینا ہے۔ اور ادب کا تقاضا یہ ہے کہ ہم یہ بات قرآن ہی کے لہجے میں کہیں: عقیدے کا رد، اشخاص کی تحقیر نہیں، اور آج کی کسی قوم پر کوئی عمومی حکم نہیں۔
