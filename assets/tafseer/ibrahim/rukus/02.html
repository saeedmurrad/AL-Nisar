وَاِذْ تَاَذَّنَ رَبُّكُمْ لَئِنْ شَكَرْتُمْ لَاَزِيْدَنَّكُمْ وَلَئِنْ كَفَرْتُمْ اِنَّ عَذَابِيْ لَشَدِيْدٌ۔ اور جب تمھارے رب نے اعلان کیا کہ اگر تم شکر کرو گے تو میں تمھیں اور دوں گا، اور اگر ناشکری کرو گے تو میرا عذاب سخت ہے۔ یہ آیت اس سورت کا ایک ستون ہے، اور شیخِ اکبر کے مکتب میں شکر کی حقیقت اسی سے کھلتی ہے۔ شکر کیا ہے؟ اہلِ معرفت کہتے ہیں: نعمت کو منعم کی طرف لوٹانا۔ یعنی نعمت کو دیکھ کر دینے والے کو دیکھنا، اور نعمت کو اسی کام میں لگانا جس کے لیے دی گئی۔ اور زیادتی کا وعدہ اس لیے ہے کہ شکر دل کو کشادہ کرتا ہے۔ سورۂ رعد میں وادیاں اپنی گنجائش کے مطابق بھرتی تھیں؛ یہاں بتایا گیا کہ گنجائش کیسے بڑھتی ہے۔ شکر وادی کو گہرا کرتا ہے۔ جو نعمت پر شکر کرے اس کی استعداد بڑھ جاتی ہے، اور بڑھی ہوئی استعداد پر بڑا فیض اترتا ہے۔ یہ لازیدنکم کا باطنی معنی ہے: زیادتی صرف نعمت کی نہیں، نعمت لینے کی صلاحیت کی۔ اور ناشکری وادی کو پاٹ دیتی ہے؛ اس پر وہی بارش برسے تو بہ جاتی ہے، ٹھہرتی نہیں۔ عذاب کی شدت اہلِ ذوق کے نزدیک یہی ہے کہ فیض سامنے ہو اور دل اسے لینے کے قابل نہ رہے۔

شیخِ اکبر شکر کے تین درجے بتاتے ہیں۔ زبان کا شکر، جو حمد ہے۔ عمل کا شکر، جو نعمت کو منعم کی رضا میں لگانا ہے۔ اور قلب کا شکر، جو یہ دیکھنا ہے کہ نعمت میں دینے والا خود جلوہ فرما ہے۔ تیسرا درجہ معرفت ہے، اور اسی درجے پر شکر اور مشاہدہ ایک ہو جاتے ہیں۔ اور یہاں ایک نکتہ اور ہے: آیت میں شکر کا مقابل ناشکری کو نہیں، کفر کو کہا گیا۔ اہلِ معرفت کے نزدیک یہ اشارہ ہے کہ کفر کی اصل نعمت کو چھپانا ہے، کیونکہ کفر کا لغوی معنی ہی چھپانا ہے۔ جس نے نعمت دیکھی اور منعم کو نہ دیکھا اس نے منعم کو چھپا دیا، اور یہ چھپانا ہی کفر کی جڑ ہے۔ اسی لیے فرمایا: اِنْ تَكْفُرُوْۤا اَنْتُمْ وَمَنْ فِي الْاَرْضِ جَمِيْعًا فَاِنَّ اللّٰهَ لَغَنِيٌّ حَمِيْدٌ۔ اگر تم اور زمین کے سب لوگ ناشکری کریں تو اللہ بے نیاز ہے، حمد والا ہے۔ غنی: اسے شکر کی ضرورت نہیں۔ حمید: وہ حمد کیا ہوا ہے خواہ کوئی حمد کرے یا نہ کرے، کیونکہ ہر موجود اپنے وجود سے اس کی حمد کر رہا ہے۔ شیخِ اکبر کے نزدیک شکر بندے کی ضرورت ہے، حق کی نہیں۔ بندہ شکر کر کے خود کھلتا ہے، حق کو کچھ نہیں دیتا۔

پھر پچھلی قوموں کا ذکر: نوح، عاد اور ثمود، اور ان کے بعد والے جنہیں صرف اللہ جانتا ہے۔ رسول ان کے پاس نشانیاں لے کر آئے، فَرَدُّوْۤا اَيْدِيَهُمْ فِيْۤ اَفْوَاهِهِمْ۔ تو انہوں نے اپنے ہاتھ اپنے منہ میں لوٹا لیے۔ مفسرین نے اس کے کئی معنی بیان کیے ہیں: غصے سے انگلیاں کاٹنا، یا رسولوں کے منہ پر ہاتھ رکھ کر خاموش کرانا، یا اپنے منہ پر ہاتھ رکھ کر تعجب کرنا۔ اہلِ تاویل کے نزدیک تینوں میں ایک قدرِ مشترک ہے: ہاتھ اور منہ کا مل جانا، یعنی عمل اور کلام کا بند ہو جانا۔ جو حق کو سنے اور نہ مانے، اس کے پاس نہ کوئی دلیل رہتی ہے نہ کوئی عمل، وہ ہاتھ منہ میں ڈال کر بیٹھ جاتا ہے۔ اور وہ کہتے ہیں: اِنَّا لَفِيْ شَكٍّ مِّمَّا تَدْعُوْنَنَآ اِلَيْهِ۔ ہم تو اس بارے میں شک میں ہیں جس کی طرف تم ہمیں بلاتے ہو۔

اور رسولوں کا جواب اس سورت کا ایک روشن ترین مقام ہے: قَالَتْ رُسُلُهُمْ اَفِي اللّٰهِ شَكٌّ فَاطِرِ السَّمٰوٰتِ وَالْاَرْضِ۔ ان کے رسولوں نے کہا: کیا اللہ کے بارے میں شک ہے، جو آسمانوں اور زمین کا پیدا کرنے والا ہے؟ یہ سوال دلیل نہیں، حیرت ہے۔ شیخِ اکبر کے نزدیک حق وہ ہے جس کا انکار ممکن ہی نہیں، کیونکہ انکار کرنے والا خود اسی کے وجود سے قائم ہے۔ جو کہے کہ مجھے اللہ میں شک ہے، وہ اپنے شک کے وجود کو کس سے مانگ رہا ہے؟ شک بھی ایک موجود ہے اور ہر موجود اسی کے فیض سے ہے۔ اہلِ معرفت کہتے ہیں کہ حق میں شک اس لیے ناممکن ہے کہ وہ سب سے زیادہ ظاہر ہے، اور اسی ظہور کی شدت سے چھپا ہوا ہے۔ آنکھ سب کچھ دیکھتی ہے اور خود کو نہیں دیکھتی، اسی لیے نہیں کہ وہ چھپی ہے، بلکہ اس لیے کہ وہ خود دیکھنے والی ہے۔ حق بھی اسی طرح ہے: وہ اتنا ظاہر ہے کہ اس کے سوا کچھ ظاہر نہیں، اور اسی لیے ظاہر نظر نہیں آتا۔ اور فاطر کا لفظ بھی معنی خیز ہے: پھاڑ کر نکالنے والا، جیسے بیج پھٹتا ہے اور درخت نکلتا ہے۔ عالم عدم کے پردے کو پھاڑ کر نکالا گیا ہے، اور جو اس پھٹنے کو دیکھ لے وہ پھاڑنے والے میں شک نہیں کر سکتا۔

يَدْعُوْكُمْ لِيَغْفِرَ لَكُمْ مِّنْ ذُنُوْبِكُمْ وَيُؤَخِّرَكُمْ اِلٰۤى اَجَلٍ مُّسَمًّى۔ وہ تمھیں بلاتا ہے تاکہ تمھارے گناہ بخش دے اور تمھیں ایک مقررہ مدت تک مہلت دے۔ غور کیجیے: بلانے والا وہ ہے، اور بلانے کا مقصد بخشنا ہے۔ سورۂ رعد میں کہا گیا تھا کہ سچی پکار اسی کی ہے، اور یہاں پکار کا مقصد بتا دیا: مغفرت۔ اہلِ معرفت کہتے ہیں کہ حق بندے کو اپنی طرف اس لیے نہیں بلاتا کہ اس کا حساب لے، بلکہ اس لیے کہ اسے بخش دے۔ حساب اُس کا ہے جو نہ آئے۔ جو آ گیا وہ بخشش کے لیے آیا۔ اور قوم کا جواب: اِنْ اَنْتُمْ اِلَّا بَشَرٌ مِّثْلُنَا۔ تم تو ہمارے جیسے انسان ہو۔ یہ وہی اعتراض ہے جو ہر زمانے میں نبوت اور ولایت پر ہوا: یہ ہمارے جیسا ہے، تو یہ کیسے حق کا واسطہ ہو سکتا ہے؟ اور رسولوں کا جواب شیخِ اکبر کے مکتب میں تنزیہ اور تشبیہ کے جمع کا اعلان ہے: اِنْ نَّحْنُ اِلَّا بَشَرٌ مِّثْلُكُمْ وَلٰكِنَّ اللّٰهَ يَمُنُّ عَلٰى مَنْ يَّشَاءُ مِنْ عِبَادِهٖ۔ ہاں، ہم تمھارے جیسے انسان ہیں، مگر اللہ اپنے بندوں میں سے جس پر چاہتا ہے احسان فرماتا ہے۔ رسولوں نے بشریت کا انکار نہیں کیا۔ انہوں نے کہا: بشر ہونا اور منتِ حق کا مظہر ہونا ایک ہی وجود میں جمع ہو سکتا ہے۔ کامل زمین پر چلتا ہے اور اس کا دل عرش پر ہے۔ جو بشریت دیکھ کر نور کا انکار کرے وہ صورت پر رک گیا؛ جو نور دیکھ کر بشریت کا انکار کرے وہ بھی صورت پر رک گیا، صرف الٹی طرف سے۔

اور رکوع کا اختتام توکل اور صبر پر: وَمَا لَنَآ اَلَّا نَتَوَكَّلَ عَلَى اللّٰهِ وَقَدْ هَدٰنَا سُبُلَنَا ۚ وَلَنَصْبِرَنَّ عَلٰى مَآ اٰذَيْتُمُوْنَا ۚ وَعَلَى اللّٰهِ فَلْيَتَوَكَّلِ الْمُتَوَكِّلُوْنَ۔ اور ہم اللہ پر بھروسا کیوں نہ کریں جب اس نے ہمیں ہماری راہیں دکھا دیں، اور ہم تمھاری اذیت پر ضرور صبر کریں گے، اور بھروسا کرنے والوں کو اللہ ہی پر بھروسا کرنا چاہیے۔ "ہماری راہیں" جمع ہے، سبلنا۔ اہلِ معرفت کے نزدیک یہ اشارہ ہے کہ راہیں بہت ہیں، اور ہر رسول کی اور ہر سالک کی اپنی راہ ہے، اور سب حق کی طرف ہیں۔ یہ سورۂ رعد کے ربِّ خاص کا مضمون ہے: راستے مخلوق کی سانسوں کے برابر۔ اور توکل کی دلیل یہ دی گئی کہ اس نے راہ دکھائی۔ جس نے راہ دکھائی وہ راہ پر چلانے کا بھی ذمہ دار ہے۔ اور صبر اذیت پر ہے، یعنی سالک کو راہ پر اذیت ملے گی، اور یہ اذیت راہ کی نفی نہیں، راہ کا حصہ ہے۔ شیخِ اکبر کے نزدیک توکل کا کمال یہ ہے کہ بندہ اسباب چھوڑے نہیں مگر اسباب پر تکیہ نہ کرے؛ وہ چلے اور جانے کہ چلانے والا کوئی اور ہے۔ اور آیت کا آخری جملہ عام ہے: توکل کرنے والوں کو اللہ ہی پر توکل کرنا چاہیے۔ یعنی توکل کا مقام کسی نبی کا خاص نہیں، ہر وہ بندہ جسے راہ دکھائی گئی ہو اس کا فرض ہے۔
