یہ پوری سورت ایک رسول کی وہ روداد ہے جو اس نے صدیوں کی محنت کے بعد اپنے رب کے حضور پیش کی، اور اس کا آغاز بھیجنے والے کی زبان سے ہوتا ہے: اِنَّآ اَرْسَلْنَا نُوْحًا اِلٰى قَوْمِهٖۤ اَنْ اَنْذِرْ قَوْمَكَ مِنْ قَبْلِ اَنْ يَّاْتِيَهُمْ عَذَابٌ اَلِيْمٌ۔ “ہم نے نوح کو اس کی قوم کی طرف بھیجا کہ اپنی قوم کو خبردار کر دو، اس سے پہلے کہ ان پر دردناک عذاب آ جائے۔” حدیثِ شفاعت کا مفہوم ہے کہ قیامت کے دن لوگ حضرت نوح علیہ السلام سے عرض کریں گے کہ آپ اہلِ زمین کی طرف بھیجے گئے پہلے رسول ہیں۔ تو یہ سورت گویا رسالت کی پہلی روداد ہے۔ آیت کا سب سے لطیف ٹکڑا مِنْ قَبْلِ ہے: خبرداری عذاب سے پہلے رکھی گئی۔ اہلِ معرفت کہتے ہیں کہ یہی سبقتِ رحمت ہے؛ حق تعالیٰ نے پکڑ سے پہلے پکار رکھی، تاکہ جو پکڑا جائے وہ اپنی بے پروائی سے پکڑا جائے، بے خبری سے نہیں۔ پھر قوم کو قَوْمِهٖ اور قَوْمَكَ کہا گیا: وہ ان کے اپنے لوگ تھے، اجنبی نہیں۔ حضرت نوح علیہ السلام نے بھی پیغام اسی اپنائیت سے کھولا: قَالَ يٰقَوْمِ اِنِّيْ لَكُمْ نَذِيْرٌ مُّبِيْنٌ۔ “انہوں نے کہا: اے میری قوم! میں تمہیں صاف صاف خبردار کرنے والا ہوں۔” يٰقَوْمِ میں “میری” کی نسبت ہے اور لَكُمْ میں “تمہارے بھلے کے لیے” کی خیرخواہی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں نبی وہ آئینہ ہے جس میں حق کی رحمت مخلوق کی طرف رخ کرتی ہے، اس لیے اس کا ڈرانا بھی رحمت ہی کی ایک صورت ہے۔ اور مُبِيْنٌ کہہ کر بتا دیا کہ بات کھلی ہے؛ جو نہ سمجھے وہ ابہام کی وجہ سے نہیں، اعراض کی وجہ سے نہ سمجھے گا۔

پیغام کا خلاصہ تین لفظوں میں: اَنِ اعْبُدُوا اللّٰهَ وَاتَّقُوْهُ وَاَطِيْعُوْنِ۔ “کہ اللہ کی عبادت کرو، اس سے ڈرو اور میری اطاعت کرو۔” عبادت حق کا حق ہے، تقویٰ دل کی پہرے داری ہے، اور اطاعت اس واسطے کا ادب ہے جس کے ذریعے حق کی بات پہنچی۔ اہلِ معرفت کہتے ہیں کہ یہ ایک ہی سفر کے تین قدم ہیں: پہلے رخ درست ہو، پھر دل نگہبانی کرے، پھر قدم رسول کے نقشِ قدم پر پڑیں۔ اطاعت کو اپنی طرف منسوب کرنا خود نمائی نہیں، کیونکہ رسول کی اطاعت حق ہی کی اطاعت ہے۔ اس کے بعد وعدہ آیا: يَغْفِرْ لَكُمْ مِّنْ ذُنُوْبِكُمْ وَيُؤَخِّرْكُمْ اِلٰۤى اَجَلٍ مُّسَمًّى ۭ اِنَّ اَجَلَ اللّٰهِ اِذَا جَآءَ لَا يُؤَخَّرُ لَوْ كُنْتُمْ تَعْلَمُوْنَ۔ “وہ تمہارے گناہ بخش دے گا اور تمہیں ایک مقررہ مدت تک مہلت دے گا؛ بے شک اللہ کی مقررہ گھڑی جب آ جاتی ہے تو ٹالی نہیں جاتی، کاش تم جانتے۔” مفسرین کہتے ہیں کہ یہاں دو مدتیں ہیں: ایک وہ جو ایمان کی صورت میں انہیں ملتی، کہ عذاب سے ہلاک ہوئے بغیر اپنی طبعی عمر پوری کرتے؛ اور ایک اللہ کی وہ آخری گھڑی جو کسی کے لیے نہیں ٹلتی۔ شیخِ اکبر کے نزدیک ہر شے کی مدت مُسَمًّى ہے، یعنی علمِ الٰہی میں اس کا نام رکھا جا چکا ہے؛ اور سرِّ قدر یہ ہے کہ یہ علم اس شے کے اپنے حال کے مطابق ہے۔ حق نے جانا کہ کون کس طرف رخ کرے گا، اور حکم اسی رخ کے مطابق لکھا۔ اسی لیے آخر میں فرمایا: کاش تم جانتے؛ وہ جاننا جو آدمی کا رخ بدل دے، محض خبر کا جاننا نہیں۔

اب منظر بدلتا ہے۔ قوم سے خطاب ختم ہوا، اور رسول اپنے رب کے سامنے اپنی عمر کا حساب رکھتا ہے: قَالَ رَبِّ اِنِّيْ دَعَوْتُ قَوْمِيْ لَيْلًا وَّنَهَارًا۔ “عرض کیا: اے میرے رب! میں نے اپنی قوم کو رات دن پکارا۔” یہ رات دن کتنے لمبے تھے، اس کی خبر دوسرے مقام پر دی گئی: فَلَبِثَ فِيْهِمْ اَلْفَ سَنَةٍ اِلَّا خَمْسِيْنَ عَامًا۔ “تو وہ ان میں پچاس کم ایک ہزار برس رہے۔” ساڑھے نو سو برس کی صبح و شام، اور ہر صبح وہی پکار۔ شیخِ اکبر نے فصوص الحکم میں حضرت نوح علیہ السلام کے کلمے کو حکمتِ سبوحیہ سے موسوم کیا، اور سبوح تنزیہ کا لفظ ہے: ہر نقص اور ہر مشابہت سے پاک۔ اس باب کی جو بات یہاں بیان کرنی ہے اس کا مفہوم یہ ہے کہ کامل دعوت تنزیہ اور تشبیہ دونوں کو جمع کرتی ہے، جیسے قرآن نے ایک ہی آیت میں دونوں کو جمع کیا: لَيْسَ كَمِثْلِهٖ شَيْءٌ ۚ وَهُوَ السَّمِيْعُ الْبَصِيْرُ۔ پہلے حصے میں حق ہر مثل سے پاک ہے، اور دوسرے میں وہ سننے دیکھنے والا ہے۔ اور حضرت نوح علیہ السلام کی دعوت نے رات اور دن، چھپ کر اور کھل کر، دونوں رخ جمع کیے۔ اہلِ ذوق کہتے ہیں کہ رات کی پکار انسان کے باطن، اس کی عقل اور روح سے خطاب تھی جب وہ خلوت میں ہو، اور دن کی پکار اس کے ظاہر سے جب وہ مجلس میں ہو۔ یوں انسان کا کوئی گوشہ ایسا نہ رہا جس تک پیغام نہ پہنچا ہو۔

پھر وہ جملہ جس میں صدیوں کا درد سمٹ آیا: فَلَمْ يَزِدْهُمْ دُعَآءِيْۤ اِلَّا فِرَارًا۔ “تو میری پکار نے ان کے بھاگنے ہی کو بڑھایا۔” یہاں ایک نازک بات سمجھنی ضروری ہے۔ پکار نے فرار پیدا نہیں کیا؛ پکار نے وہ ظاہر کیا جو پہلے سے اندر تھا۔ سورج ایک ہی ہے، موم کو پگھلا دیتا ہے اور گیلی مٹی کو سخت کر دیتا ہے؛ فرق سورج میں نہیں، چیز کے اپنے مزاج میں ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ دعوت ایک تھی، استعدادیں مختلف تھیں، اور ہر ایک نے اپنی استعداد کے مطابق جواب دیا۔ اور یہ بھی دیکھیے کہ فرار کا لفظ آیا۔ بھاگنا ہمیشہ کسی چیز سے ہوتا ہے، اور یہ لوگ رحمت سے بھاگ رہے تھے۔ قرآن نے صحیح فرار کا رخ بھی بتایا ہے: فَفِرُّوْۤا اِلَى اللّٰهِ۔ “تو اللہ کی طرف دوڑو۔” اہلِ معرفت کہتے ہیں کہ ہر انسان کسی نہ کسی طرف بھاگ رہا ہے؛ ایک اللہ سے بھاگتا ہے اور ایک اللہ کی طرف۔ قدم دونوں کے تیز ہیں، رخ مختلف ہیں، اور منزل رخ سے طے ہوتی ہے۔ سالک کے لیے یہ آیت آئینہ ہے: جب نصیحت سن کر دل میں بے چینی اٹھے اور جی چاہے کہ موضوع بدل دیا جائے، تو وہی لمحہ پہچاننے کا ہے کہ قدم کس طرف اٹھ رہے ہیں۔

اب وہ آیت آتی ہے جو اس سورت کا مرکزی اصول ہے، اور اسے اصول ہی کے طور پر پڑھنا چاہیے: وَاِنِّيْ كُلَّمَا دَعَوْتُهُمْ لِتَغْفِرَ لَهُمْ جَعَلُوْۤا اَصَابِعَهُمْ فِيْۤ اٰذَانِهِمْ وَاسْتَغْشَوْا ثِيَابَهُمْ وَاَصَرُّوْا وَاسْتَكْبَرُوا اسْتِكْبَارًا۔ “اور میں نے جب بھی انہیں پکارا تاکہ تو انہیں بخش دے، انہوں نے اپنی انگلیاں اپنے کانوں میں ٹھونس لیں، اپنے کپڑے اپنے اوپر لپیٹ لیے، اڑ گئے اور بڑا ہی تکبر کیا۔” یہ ہے اصول: کان انہوں نے خود بند کیے۔ اس آیت کے ہر فعل کا فاعل وہی ہیں: جَعَلُوْا، اسْتَغْشَوْا، اَصَرُّوْا، اسْتَكْبَرُوْا۔ ایک فعل بھی ایسا نہیں جس میں ان کے بہرے پن کو حق کی طرف زبردستی کے طور پر منسوب کیا گیا ہو۔ دوسری طرف دعوت کا مقصد دیکھیے: لِتَغْفِرَ لَهُمْ، “تاکہ تو انہیں بخش دے۔” ایک طرف بخشش کا دروازہ کھولا جا رہا تھا، اور دوسری طرف انگلیاں کانوں میں جا رہی تھیں۔ حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت چاہے اسے کبھی گمراہ نہیں کرتا۔ جو مانگتا ہے اسے ہدایت ملتی ہے؛ اور جو منہ پھیرتا ہے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے۔ یہ قوم ہدایت سے محروم نہیں کی گئی؛ اس نے ہدایت کو سننے سے انکار کیا، اور سننے سے انکار کرنے والا وہ سن ہی کیسے سکتا ہے جو اسے پکار رہا ہو۔

اسی اصول کی روشنی میں قرآن کے ان مقامات کو پڑھنا چاہیے جہاں دلوں پر مہر اور کانوں پر ڈاٹ کا ذکر آیا ہے۔ وہ مہر ابتدا نہیں، انجام ہے؛ اس انگلی کا انجام جو بار بار کان میں رکھی گئی۔ شیخِ اکبر کے مکتب میں یہ بات یوں سمجھائی جاتی ہے کہ وجود دینے والا حق ہے، مگر جو صورت وجود میں آتی ہے وہ بندے کے اپنے رخ اور اپنی طلب کی صورت ہوتی ہے۔ فیضِ مقدس ہر چیز کو وہی دیتا ہے جو اس کی عینِ ثابتہ اپنی استعداد کی زبان سے مانگتی ہے۔ جس نے اپنی انگلیوں سے کہا کہ مجھے نہیں سننا، حق نے اس کی اس چاہت کو وجود دے دیا: نہ سننا اس کا حال بن گیا۔ حکم محکوم کے مطابق؛ فیصلہ اسی رخ پر ہوا جو بندے نے خود چنا۔ اور اسی میں انسان کی عزت بھی ہے۔ حق چاہتا تو سب کو مجبوراً راہ پر لے آتا، مگر اس نے انسان کو اختیار کا شرف دیا اور اس شرف کو واپس نہیں لیا۔ جو قوم ساڑھے نو سو برس پکاری گئی، اس کے بارے میں کوئی یہ نہیں کہہ سکتا کہ اسے موقع نہیں ملا۔ پکار ہر روز تھی؛ انکار بھی ہر روز تھا؛ اور ہر روز کا انکار ایک نئی اینٹ تھی اس دیوار کی جو آخر میں ان کے اور حق کے درمیان کھڑی ہو گئی۔

اب ان چار افعال کو ایک ایک کر کے دیکھیے، کیونکہ ان میں انکار کے مراتب کی پوری تصویر ہے۔ پہلا مرتبہ کان بند کرنا ہے: سننے سے انکار۔ دوسرا کپڑے لپیٹنا: اسْتَغْشَوْا کا صیغہ طلب کا ہے، یعنی انہوں نے خود ڈھکنا چاہا، کہ نہ دیکھیں اور نہ دیکھے جائیں؛ وہی پردہ جو دوسرے مقامات پر آنکھوں پر پردے کے نام سے آیا، یہاں ان کی اپنی طلب کی صورت میں نظر آتا ہے۔ تیسرا مرتبہ اصرار ہے: ارادہ سخت ہو جاتا ہے، اور ضد عادت بن جاتی ہے۔ چوتھا اور سب کی جڑ تکبر ہے، اور اسے مصدر کے ساتھ دہرا کر کہا گیا: اسْتَكْبَرُوا اسْتِكْبَارًا، بڑا ہی تکبر۔ یہی ابلیس کی پرانی بیماری ہے: اپنے آپ کو بڑا دیکھنا۔ اہلِ ذوق کہتے ہیں کہ انگلیاں حق کی دی ہوئی وہ قوت تھیں جن سے پکڑا اور کیا جاتا ہے؛ انہیں بند کرنے کا آلہ بنا لیا گیا۔ اور کپڑے انسان کی اپنی عادتیں اور اپنا نفس ہیں؛ جو اپنے ہی کپڑوں میں منہ چھپا لے، وہ اپنے نفس کے سوا کچھ نہیں دیکھتا۔ سالک کو اپنی انگلیاں دیکھنی چاہییں: جب نصیحت آتی ہے تو کون سا بہانہ کان میں رکھ لیتا ہے، اور کون سی دلیل کپڑے کی طرح سر پر کھینچ لیتا ہے۔

اس کے بعد طریقِ دعوت کی ترقی: ثُمَّ اِنِّيْ دَعَوْتُهُمْ جِهَارًا۔ “پھر میں نے انہیں بلند آواز سے پکارا۔” ثُمَّ اِنِّيْۤ اَعْلَنْتُ لَهُمْ وَاَسْرَرْتُ لَهُمْ اِسْرَارًا۔ “پھر میں نے انہیں علانیہ بھی سمجھایا اور چپکے چپکے بھی سمجھایا۔” ثُمَّ کا دو بار آنا بتاتا ہے کہ یہ مرحلے تھے: پہلے مجمعوں میں بلند آواز، پھر عام اعلان، پھر ایک ایک کے پاس تنہائی میں۔ داعی کا کمال یہ ہے کہ ایک طریقہ کام نہ کرے تو دوسرا اختیار کرے، اور پیغام وہی رہے۔ کسی کا دل مجمع میں کھلتا ہے، کسی کا تنہائی میں؛ کوئی بلند آواز سے جاگتا ہے، اور کوئی سرگوشی سے۔ شیخِ اکبر کے مکتب میں ہر اسم کا اپنا مظہر ہے اور ہر دل کا اپنا دروازہ؛ کامل داعی ہر استعداد کو اس کے اپنے دروازے سے پکارتا ہے۔ اہلِ ذوق کہتے ہیں کہ علانیہ پکار انسان کے ظاہر اور اس کی معاشرت سے خطاب ہے، اور پوشیدہ پکار اس کے سرّ سے، یعنی اس لطیفے سے جو قلب اور روح سے بھی گہرا ہے۔ جو بات مجمع میں نہیں کہی جا سکتی وہ کان میں کہی جاتی ہے، اور بعض باتیں صرف کان ہی سن سکتا ہے۔ مگر یہ سارا تنوع اس قوم کے لیے بے کار رہا، کیونکہ دروازے داعی کی طرف سے کھلے تھے اور قفل اندر سے لگا تھا۔

اب حضرت نوح علیہ السلام وہ نسخہ بتاتے ہیں جو انہوں نے قوم کے ہاتھ میں رکھا: فَقُلْتُ اسْتَغْفِرُوْا رَبَّكُمْ ۭ اِنَّهٗ كَانَ غَفَّارًا۔ “پھر میں نے کہا: اپنے رب سے معافی مانگو، بے شک وہ بہت بخشنے والا ہے۔” يُّرْسِلِ السَّمَآءَ عَلَيْكُمْ مِّدْرَارًا۔ “وہ تم پر آسمان سے موسلا دھار بارش برسائے گا۔” غَفَّارًا مبالغے کا صیغہ ہے: وہ ایک بار نہیں، بار بار بخشنے والا ہے؛ اور كَانَ بتاتا ہے کہ یہ اس کی ہمیشہ کی شان ہے، کوئی نیا وعدہ نہیں۔ بعض مفسرین نے لکھا ہے کہ یہ قوم برسوں سے قحط اور بے اولادی میں مبتلا تھی، اسی لیے حضرت نوح علیہ السلام نے بخشش کو بارش سے جوڑا۔ اس پر حضرت حسن بصری رحمۃ اللہ علیہ کا واقعہ مشہور ہے، جس کا مفہوم یہ ہے کہ ایک شخص نے ان سے قحط کی شکایت کی تو فرمایا: استغفار کرو؛ دوسرے نے فقر کی شکایت کی، فرمایا: استغفار کرو؛ تیسرے نے اولاد کی خواہش ظاہر کی، فرمایا: استغفار کرو؛ چوتھے نے باغ کے سوکھنے کا ذکر کیا، فرمایا: استغفار کرو۔ پاس بیٹھے لوگوں نے حیرت سے پوچھا کہ سب کو ایک ہی جواب؟ تو انہوں نے یہی آیات پڑھ دیں۔ گویا ان کے نزدیک استغفار ایک کنجی تھی جو کئی دروازوں پر لگتی ہے۔ مِدْرَارًا کا مطلب ہے لگاتار برسنے والا؛ نہ ایک چھینٹا کہ زمین کی پیاس جاگے اور بجھے نہیں، بلکہ ایسی بارش جو زمین کو سیراب کر دے۔

وعدہ آگے بڑھتا ہے: وَّيُمْدِدْكُمْ بِاَمْوَالٍ وَّبَنِيْنَ وَيَجْعَلْ لَّكُمْ جَنّٰتٍ وَّيَجْعَلْ لَّكُمْ اَنْهٰرًا۔ “اور مال اور بیٹوں سے تمہاری مدد کرے گا، اور تمہارے لیے باغ بنا دے گا اور تمہارے لیے نہریں جاری کر دے گا۔” ترتیب پر غور کیجیے: پہلے آسمان سے بارش، پھر گھر میں مال اور اولاد، پھر زمین پر باغ اور نہریں۔ رحمت اوپر سے اترتی ہے، گھر کو بھرتی ہے، اور پھر زمین تک پھیل جاتی ہے۔ یہ ظاہری وعدہ پورا سچ ہے، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ ایک حدیث کا مفہوم، جسے امام ابو داؤد نے روایت کیا، یہ ہے کہ جو استغفار کو لازم پکڑ لے اللہ اس کے لیے ہر تنگی سے نکلنے کا راستہ اور ہر غم سے کشادگی بنا دیتا ہے، اور اسے وہاں سے رزق دیتا ہے جہاں سے اسے گمان بھی نہ ہو۔ اور حضرت عمر رضی اللہ عنہ کے بارے میں مفسرین نے نقل کیا ہے کہ وہ بارش کی دعا کے لیے نکلے اور استغفار کے سوا کچھ نہ کیا، اور جب پوچھا گیا تو یہی آیات پڑھیں، گویا فرمایا کہ میں نے بارش کو آسمان کی کنجیوں سے طلب کیا ہے۔ اہلِ معرفت کہتے ہیں کہ گناہ رزق اور بندے کے درمیان ایک رکاوٹ ہے؛ استغفار اس رکاوٹ کو ہٹاتا ہے، اور جو رحمت پہلے سے برسنے کو تیار تھی وہ راستہ پا لیتی ہے۔ استغفار رزق کی قیمت نہیں، رکاوٹ کا ہٹنا ہے۔

اب اس وعدے کا باطن۔ غفر کے اصل معنی ڈھانپنے کے ہیں؛ مِغْفَر وہ خود ہے جو سر کو ڈھانپ کر بچاتا ہے۔ شیخِ اکبر کے مکتب میں عام بندے کا استغفار یہ ہے کہ اس کے گناہ ڈھانپ دیے جائیں، اور عارف کا استغفار یہ ہے کہ اس کا “میں” ڈھانپ دیا جائے، وہ دعویٰ کہ وہ اپنے بل پر کچھ ہے۔ جب یہ پردہ ڈھک جاتا ہے تو حق کا نور سامنے آ جاتا ہے۔ اور غَفَّار کا مبالغہ تجددِ امثال کی یاد دلاتا ہے: بندہ ہر آن نئی تخلیق میں ہے، تو ہر آن اسے نئی بخشش کی ضرورت ہے، اور وہ ہر آن نئی بخشش دینے والا ہے۔ اسی طرح اہلِ تاویل کہتے ہیں کہ روح کے آسمان سے دل کی زمین پر واردات کی بارش اترتی ہے؛ اموال وہ علوم اور اعمال ہیں جن سے باطن غنی ہوتا ہے؛ بنین وہ نتیجے اور احوال ہیں جو ذکر اور فکر سے پیدا ہوتے ہیں؛ جنات احوال کے باغ ہیں؛ اور انہار وہ علوم ہیں جو دل سے پھوٹ کر بہنے لگتے ہیں۔ یہ سب ظاہری بارش اور ظاہری مال کی نفی نہیں؛ وہ بھی سچ ہیں، اور یہ ان کا اندرونی رخ ہے۔ جس دل کا آسمان گناہ کے بادل سے بند ہو، اس پر یہ بارش نہیں اترتی؛ استغفار وہ ہوا ہے جو بادل چھانٹ دیتی ہے۔

یہاں ایک سوال اٹھتا ہے: جو معصوم ہو وہ استغفار کیوں کرے؟ ایک صحیح حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم دن میں ستر بار سے زیادہ استغفار کرتے تھے، اور ایک دوسری روایت کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم دن میں سو بار استغفار فرماتے۔ اہلِ معرفت کہتے ہیں کہ کامل کا استغفار گناہ سے نہیں، ترقی سے ہوتا ہے۔ سالک جب ایک مقام سے اگلے مقام پر چڑھتا ہے تو پچھلا مقام اسے نئی روشنی کے مقابلے میں ناقص دکھائی دیتا ہے، اور وہ اس پر بھی معافی مانگتا ہے۔ یوں استغفار عبدیت کی دائمی حالت بن جاتا ہے: بندہ کبھی خود کو حق سے بے نیاز نہیں دیکھتا۔ پھر حضرت نوح علیہ السلام کی حکمت دیکھیے: انہوں نے قوم کو اس چیز سے پکارا جس کی انہیں سب سے زیادہ طلب تھی، یعنی بارش، مال اور اولاد۔ اہلِ ذوق کہتے ہیں کہ یہی ادبِ دعوت ہے: جہاں لوگ کھڑے ہوں وہاں سے بات شروع کی جائے، اور انہیں دکھایا جائے کہ جو وہ چاہتے ہیں اس کا دروازہ بھی اسی گھر میں کھلتا ہے جس کی طرف بلایا جا رہا ہے۔ دنیا کی طلب کو حق کی طرف موڑ دینا اسے ٹھکرا دینے سے زیادہ کامل تربیت ہے۔

پھر ایک ایسا سوال جو دل کو پکڑ لیتا ہے: مَا لَكُمْ لَا تَرْجُوْنَ لِلّٰهِ وَقَارًا۔ “تمہیں کیا ہو گیا ہے کہ تم اللہ کی عظمت کا لحاظ نہیں رکھتے؟” وَقَدْ خَلَقَكُمْ اَطْوَارًا۔ “حالانکہ اس نے تمہیں طرح طرح کے مرحلوں سے گزار کر پیدا کیا۔” مفسرین نے لَا تَرْجُوْنَ کے دو معنی بیان کیے ہیں: ایک یہ کہ تم اللہ کی عظمت سے نہیں ڈرتے، کیونکہ رجاء بعض اہلِ عرب کے محاورے میں خوف کے معنی میں بھی آتا ہے؛ اور دوسرا یہ کہ تم اللہ سے عزت اور وقار کی امید نہیں رکھتے۔ دونوں معنی ایک دوسرے کو مکمل کرتے ہیں: جو اس کی عظمت کو نہ مانے وہ اس سے عزت کی امید بھی نہیں رکھتا۔ پھر دلیل آئی: تم مرحلہ بہ مرحلہ بنے ہو۔ نطفہ، پھر جما ہوا خون، پھر گوشت کا لوتھڑا، پھر ہڈیاں، پھر ان پر گوشت، پھر ایک بالکل نئی صورت؛ اور پیدائش کے بعد بچپن، جوانی اور بڑھاپا۔ کسی مرحلے میں تمہارا اختیار نہ تھا، ہر مرحلے پر بنانے والا ہاتھ کوئی اور تھا۔ جو یہ دیکھ لے کہ وہ ایک بنانے والے کے ہاتھ میں ابھی بن رہا ہے، اس کے دل میں اس بنانے والے کا وقار خود بخود پیدا ہو جاتا ہے۔

اہلِ معرفت کے نزدیک اَطْوَارًا کا ایک اندرونی رخ بھی ہے: نفس کے مراتب۔ قرآن نے نفس کو کہیں برائی پر اکسانے والا کہا، کہیں ملامت کرنے والا، اور کہیں اطمینان پانے والا؛ یہ ایک ہی نفس کے اطوار ہیں جن سے سالک گزرتا ہے۔ اسی طرح اہلِ تصوف لطائف کا ذکر کرتے ہیں: قلب، روح، سرّ، خفی اور اخفیٰ؛ گویا ظاہری اطوار کے بعد باطن کے اطوار شروع ہوتے ہیں۔ شیخِ اکبر کے مکتب میں اس سے بھی گہری بات یہ ہے کہ انسان کی پہلی منزل علمِ الٰہی میں اس کی عینِ ثابتہ ہے؛ فیضِ اقدس نے اسے اس کی استعداد دی، اور فیضِ مقدس نے اسے وجود دیا۔ پھر تجددِ امثال کی رو سے خلق کبھی رکتی نہیں: انسان ہر آن ایک نئے طور میں ہے، ہر سانس ایک نئی تخلیق ہے۔ جو اس حقیقت کو دیکھ لے، اس کے لیے وقار اور امید دونوں جمع ہو جاتے ہیں: وقار اس لیے کہ وہ ہر لمحہ حق کے ہاتھ میں ہے، اور امید اس لیے کہ جو آج تک ہر طور میں سنوارتا آیا ہے وہ اگلے طور میں بھی چھوڑے گا نہیں۔ یوں رجاء کے دونوں معنی یہاں ایک ہو جاتے ہیں۔ اور یہ بھی کہ کوئی شخص اپنے موجودہ طور پر فیصلہ نہ کرے؛ ابھی کام جاری ہے۔

اب نظر کو اوپر اٹھایا گیا: اَلَمْ تَرَوْا كَيْفَ خَلَقَ اللّٰهُ سَبْعَ سَمٰوٰتٍ طِبَاقًا۔ “کیا تم نے نہیں دیکھا کہ اللہ نے کس طرح سات آسمان تہ بہ تہ بنائے؟” وَّجَعَلَ الْقَمَرَ فِيْهِنَّ نُوْرًا وَّجَعَلَ الشَّمْسَ سِرَاجًا۔ “اور ان میں چاند کو روشنی بنایا اور سورج کو چراغ بنایا۔” حضرت نوح علیہ السلام نے پہلے انسان کو اپنے اندر کے اطوار دکھائے، پھر باہر کے آسمان؛ دعوت انفس سے آفاق کی طرف گئی۔ سورج کو سِرَاج کہا، یعنی وہ چراغ جو خود جلتا ہے اور اپنی روشنی رکھتا ہے؛ اور چاند کو نُوْر کہا، یعنی وہ روشنی جو دوسرے سے لے کر پھیلاتا ہے۔ اہلِ ذوق کہتے ہیں کہ دل چاند ہے: اس کی اپنی کوئی روشنی نہیں، وہ اتنا ہی روشن ہوتا ہے جتنا ہدایت کے سورج کے سامنے رہے۔ جب دنیا کی زمین بیچ میں آ جائے تو گرہن لگ جاتا ہے، اور جب پورا رخ سورج کی طرف ہو تو چودھویں کا چاند بن جاتا ہے۔ شیخِ اکبر کے مکتب میں سارا عالم حق کا سایہ ہے، اور اس کی ہر روشنی اسی اصل نور کا عکس۔ اور یہ بھی کہ چاند کو فِيْهِنَّ کہا، ان سب آسمانوں میں؛ اہلِ تاویل کہتے ہیں کہ دل کی روشنی جب صاف ہو تو باطن کی ہر تہ تک پہنچتی ہے۔

پھر انسان کو اپنی اصل یاد دلائی گئی: وَاللّٰهُ اَنْۢبَتَكُمْ مِّنَ الْاَرْضِ نَبَاتًا۔ “اور اللہ نے تمہیں زمین سے ایک خاص طرح اگایا۔” اہلِ زبان ایک باریک نکتہ بیان کرتے ہیں: فعل اَنْۢبَتَ ہے، یعنی اس نے اگایا، مگر مصدر نَبَاتًا لایا گیا جو خود اگنے کا مصدر ہے۔ گویا فرمایا: اس نے تمہیں اگایا اور تم اگے۔ اہلِ معرفت کہتے ہیں کہ اس ایک ترکیب میں فعل اور قبولیت دونوں جمع ہیں: اگانے والا حق ہے، اور اگنا بیج کی اپنی استعداد کے مطابق ہے۔ شیخِ اکبر کے مکتب میں یہی وہ نکتہ ہے کہ حق ہر چیز کو اس کی استعداد کے مطابق وجود دیتا ہے۔ اور انسان کو پودا کہنے میں تربیت کا پورا نقشہ ہے۔ پودا ایک دن میں درخت نہیں بنتا؛ اسے پانی چاہیے، روشنی چاہیے، اچھی مٹی چاہیے، اور وقت چاہیے۔ اہلِ ذوق کہتے ہیں کہ سالک کا پانی ذکر ہے، اس کی روشنی صالحین کی صحبت ہے، اس کی مٹی عاجزی ہے، اور اس کا وقت صبر ہے۔ جو پودا پانی سے محروم رہے وہ مرجھا جاتا ہے، خواہ مٹی کتنی ہی زرخیز ہو؛ اور جو دل ذکر سے محروم رہے وہ سوکھ جاتا ہے، خواہ استعداد کتنی ہی بلند ہو۔ اسی لیے ذکر کو روز کا معمول بنایا جاتا ہے، کیونکہ پودے کو ایک دن بہت سا پانی دے کر پھر بھول جانا اسے سیراب نہیں کرتا۔

پودے کی کہانی آگے چلتی ہے: ثُمَّ يُعِيْدُكُمْ فِيْهَا وَيُخْرِجُكُمْ اِخْرَاجًا۔ “پھر وہ تمہیں اسی میں لوٹا دے گا اور پھر تمہیں ایک خاص طرح نکال لائے گا۔” ظاہر میں یہ موت اور قبر ہے، پھر قیامت کا اٹھنا؛ اور اِخْرَاجًا کی تاکید بتاتی ہے کہ یہ نکالنا یقینی ہے، اس میں کوئی شک نہیں۔ جو زمین سے اگا وہ زمین میں لوٹے گا، اور جو زمین میں لوٹا وہ پھر نکالا جائے گا، جیسے دانہ زمین میں دب کر ہی اگتا ہے۔ اہلِ معرفت کے نزدیک یہاں فنا اور بقا کا اشارہ ہے۔ بیج جب تک اپنی سخت چھال میں بند رہے، درخت نہیں بنتا؛ اسے مٹی میں دفن ہونا اور اپنی پہلی صورت کھونا پڑتا ہے۔ اسی طرح جو سالک اس زندگی میں عاجزی کی زمین میں لوٹ جائے، اپنے “میں” کی چھال توڑ دے، اسے بقا کی صورت میں نکالا جاتا ہے۔ اہلِ تصوف میں ایک قول مشہور ہے: مرنے سے پہلے مر جاؤ؛ یعنی نفس کی خواہشوں کو اس زندگی میں فنا کر دو۔ یہ جسمانی موت کی تمنا نہیں، بلکہ نفس کی موت ہے جس کے بعد دل کی زندگی شروع ہوتی ہے۔ اور جو اپنی خاک ساری کو پہچان لے، اسے قبر کی زمین اجنبی نہیں لگتی؛ وہ اسی زمین میں لوٹ رہا ہوتا ہے جس میں اس نے اپنا سر پہلے ہی سجدے میں رکھ دیا تھا۔

رکوع زمین کے بچھونے پر ختم ہوتا ہے: وَاللّٰهُ جَعَلَ لَكُمُ الْاَرْضَ بِسَاطًا۔ “اور اللہ نے تمہارے لیے زمین کو بچھونا بنایا۔” لِّتَسْلُكُوْا مِنْهَا سُبُلًا فِجَاجًا۔ “تاکہ تم اس کے کشادہ راستوں پر چلو۔” فِجَاج پہاڑوں کے درمیان کھلے راستوں کو کہتے ہیں؛ زمین کو اس طرح بچھایا گیا کہ پہاڑوں کے بیچ بھی راستے کھلے رہیں۔ اہلِ تصوف کے ہاں بساط ایک اصطلاح ہے: قرب کا وہ فرش جس پر بندہ اپنے رب کے سامنے ہوتا ہے، اور اس فرش کا ادب ادبِ بساط کہلاتا ہے۔ اہلِ ذوق کہتے ہیں کہ شریعت وہ بساط ہے جو حق نے بندوں کے لیے بچھائی، اور اس پر چلنے کے راستے کشادہ ہیں۔ اہلِ تصوف میں مشہور قول ہے کہ اللہ تک پہنچنے کے راستے مخلوق کے سانسوں کی تعداد کے برابر ہیں؛ مگر سب راستے اسی ایک بساط پر ہیں، اس سے باہر نہیں۔ اور غور کیجیے کہ لِتَسْلُكُوْا اسی مادے سے ہے جس سے سلوک اور سالک کے لفظ بنے ہیں۔ گویا زمین بچھانے کا مقصد ہی یہ تھا کہ انسان سالک بنے، چلے، اور ہر راستے پر اپنے رب کی نشانیاں دیکھے۔ یوں حضرت نوح علیہ السلام کی دعوت نفس کے اطوار سے آسمان کی تہوں تک اور پھر زمین کے راستوں تک پھیل گئی: ہر طرف ایک ہی پیغام تھا، اور جو کان بند نہ کرے وہ ہر طرف سے اسے سن سکتا ہے۔
