انبیا علیہم السلام کے قصوں کی جو لڑی حضرت نوح علیہ السلام سے شروع ہوئی تھی، اس کا آخری دانہ حضرت موسیٰ علیہ السلام کا ذکر ہے، اور وہ بہت مختصر لایا گیا: وَلَقَدْ اَرْسَلْنَا مُوْسٰى بِاٰيٰتِنَا وَسُلْطٰنٍ مُّبِيْنٍ اِلٰى فِرْعَوْنَ وَمَلَا۟ئِهٖ فَاتَّبَعُوْۤا اَمْرَ فِرْعَوْنَ ۚ وَمَاۤ اَمْرُ فِرْعَوْنَ بِرَشِيْدٍ۔ “اور یقیناً ہم نے موسیٰ علیہ السلام کو اپنی نشانیوں اور کھلی سند کے ساتھ فرعون اور اس کے سرداروں کی طرف بھیجا، تو وہ فرعون کے حکم پر چلے، اور فرعون کا حکم راستی والا نہ تھا۔” مفسرین کے نزدیک اٰيٰت سے مراد وہ معجزات ہیں جو حضرت موسیٰ علیہ السلام کو عطا ہوئے، جیسے عصا اور روشن ہاتھ، اور سُلْطٰنٍ مُّبِيْنٍ سے مراد وہ کھلی دلیل اور غلبہ ہے جس کے سامنے کوئی حجت نہ ٹھہر سکی۔ قابلِ غور بات یہ ہے کہ آیت نے سرداروں کا جرم یہ بتایا کہ انہوں نے نشانیاں دیکھ کر بھی فرعون کا حکم مانا؛ گویا ان کے سامنے دو راستے کھلے تھے، ایک دلیل کا اور ایک حکم کا، اور انہوں نے خود حکم کو دلیل پر ترجیح دی۔ رَشِيْد وہ ہے جو راہِ راست پر ہو اور دوسروں کو بھی اسی پر لے جائے؛ فرعون کا حکم نہ خود راست تھا نہ کسی کو راستی تک پہنچا سکتا تھا۔ اہلِ معرفت کہتے ہیں کہ یہاں دو سلطنتیں آمنے سامنے ہیں: نبی کا سلطان دلیل کا سلطان ہے جو دل کو قائل کرتا اور انتخاب کی آزادی باقی رکھتا ہے، اور جابر کا سلطان زور کا سلطان ہے جو آزادی چھین کر اطاعت مانگتا ہے۔ جس نے دلیل کو چھوڑ کر زور کے آگے سر جھکایا، اس نے اپنا اختیار خود دوسرے کے ہاتھ میں دیا، اور اختیار کا یہ سونپ دینا بھی اسی کا اپنا فعل تھا۔

شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس آیت کا ایک اشاری رخ بیان کیا جاتا ہے، اس وضاحت کے ساتھ کہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر یہی ہے کہ یہ ایک تاریخی واقعہ ہے اور فرعون اور اس کے سردار وہی ہیں جن کا ذکر ہوا۔ اہلِ تاویل کہتے ہیں کہ ہر انسان کے اندر بھی ایک تخت ہے جس پر کبھی نفس بیٹھ جاتا ہے؛ قرآن نے اسی نفس کو امارہ کہا، یعنی بہت حکم چلانے والا، اور یہ لفظ اسی مادے سے ہے جس سے اَمْرَ فِرْعَوْنَ کا اَمْر ہے۔ نفسِ امارہ کا حکم بھی کبھی رَشِيْد نہیں ہوتا، اور اس کے سردار، یعنی خواہش، غصہ، حرص اور نمود کی قوتیں، اسی کے حکم پر چلتی ہیں۔ دل کے پاس حضرت موسیٰ علیہ السلام کی طرح ایک روشن دلیل آتی ہے، وحی کی، عقلِ سلیم کی، ضمیر کی؛ اور سوال یہی ہوتا ہے کہ یہ قوتیں دلیل کے پیچھے چلیں گی یا حکم کے۔ یہ تمثیل ہے، کسی شخص یا گروہ پر حکم نہیں؛ اس کا مقصد صرف یہ ہے کہ قاری قصے کو اپنے اندر بھی پڑھے۔ اور اس تمثیل کا سبق یہ ہے کہ اطاعت اپنی ذات میں فضیلت نہیں؛ فضیلت یہ ہے کہ اطاعت رَشِيْد کی ہو۔ اندھی پیروی خواہ باہر کے حاکم کی ہو یا اندر کے نفس کی، دونوں کا انجام ایک ہے، جیسا کہ اگلی آیت بتاتی ہے۔

پھر انجام کا منظر: يَقْدُمُ قَوْمَهٗ يَوْمَ الْقِيٰمَةِ فَاَوْرَدَهُمُ النَّارَ ۗ وَبِئْسَ الْوِرْدُ الْمَوْرُوْدُ۔ “وہ قیامت کے دن اپنی قوم کے آگے آگے ہوگا اور انہیں آگ پر لا اتارے گا، اور کیا ہی برا گھاٹ ہے جس پر اترا جائے۔” ظاہر معنی یہی ہے، اور اسی پر ٹھہرنا ادب ہے: جس طرح دنیا میں فرعون ان کا پیشوا بنا اور وہ اس کے پیچھے چلے، اسی طرح قیامت کے دن وہ ان کے آگے آگے چلے گا اور انہیں آگ تک لے جائے گا۔ فَاَوْرَدَ ماضی کا صیغہ ہے، حالانکہ واقعہ آئندہ کا ہے؛ عربی میں یہ اسلوب اس بات کی قطعیت بتاتا ہے کہ گویا یہ ہو چکا۔ الْوِرْد عرب کے ہاں پانی کا وہ گھاٹ ہے جہاں پیاسے قافلے اترتے ہیں؛ پیاسا گھاٹ کی امید پر چلتا ہے، اور یہاں جس گھاٹ پر یہ قافلہ اترا وہ آگ ہے۔ یہ قرآن کا وہ تلخ طنز ہے جو دل کو جھنجھوڑ دیتا ہے۔ اور اس کے ساتھ: وَاُتْبِعُوْا فِيْ هٰذِهٖ لَعْنَةً وَّيَوْمَ الْقِيٰمَةِ ۗ بِئْسَ الرِّفْدُ الْمَرْفُوْدُ۔ “اور اس دنیا میں بھی لعنت ان کے پیچھے لگا دی گئی اور قیامت کے دن بھی؛ کیا ہی برا انعام ہے جو انہیں دیا گیا۔” رِفْد عطیہ اور سہارا ہے؛ جنہوں نے فرعون سے انعام کی امید رکھی تھی، ان کے حصے میں دو لعنتیں آئیں، ایک یہاں اور ایک وہاں۔ صحیح بخاری اور صحیح مسلم میں حضرت انس رضی اللہ عنہ سے مروی ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ آدمی اسی کے ساتھ ہوگا جس سے وہ محبت رکھتا ہے۔ اہلِ معرفت کہتے ہیں کہ دنیا میں جس کے پیچھے چلو، آخرت میں وہی تمہارا پیشوا ہوگا؛ اس لیے پیشوا کا انتخاب سوچ کر کرو۔ اور اہلِ ذوق ایک لطیف نکتہ یہاں سے لیتے ہیں: سالکوں کی اصطلاح میں روز کا معمول بھی وِرد کہلاتا ہے، یعنی وہ گھاٹ جس پر بندہ ہر روز لوٹ کر آتا ہے۔ ہر شخص کا کوئی نہ کوئی ورد ہے، جس پر اس کا دل روز اترتا ہے؛ کسی کا ذکر ہے، کسی کی خواہش۔ انجام کا گھاٹ اسی روز کے گھاٹ کی صورت ہے۔

قصے ختم ہوئے تو ان سب پر ایک جامع تبصرہ آیا: ذٰلِكَ مِنْ اَنْۢبَآءِ الْقُرٰى نَقُصُّهٗ عَلَيْكَ مِنْهَا قَآئِمٌ وَّحَصِيْدٌ۔ “یہ بستیوں کی خبروں میں سے ہیں جو ہم آپ کو سناتے ہیں؛ ان میں سے کچھ اب تک کھڑی ہیں اور کچھ کٹی ہوئی کھیتی کی طرح مٹ چکی ہیں۔” مفسرین کہتے ہیں کہ قَآئِمٌ وہ بستیاں ہیں جن کے کھنڈر اور نشان باقی ہیں، اور حَصِيْدٌ وہ ہیں جو کٹی فصل کی طرح زمین کے برابر ہو گئیں کہ ان کا نشان بھی نہ رہا۔ دونوں عبرت ہیں: کھڑے کھنڈر آنکھ کو سبق دیتے ہیں اور مٹی ہوئی بستیاں کان کو؛ ایک دیکھ کر عبرت ہے، دوسری سن کر۔ نَقُصُّ کا مادہ قَصّ ہے، جس کی اصل نقشِ قدم کے پیچھے چلنا ہے؛ قرآن قصہ اس لیے نہیں سناتا کہ سننے والا داستان سے لطف اٹھائے، بلکہ اس لیے کہ وہ ان قدموں کے نشانوں کے پیچھے چلتا ہوا اس مقام تک پہنچے جہاں ہر قوم نے اپنا رخ چنا تھا۔ اہلِ معرفت اس سے یہ نکتہ لیتے ہیں کہ دل بھی ایک بستی ہے؛ اس میں بھی کچھ عمارتیں کھڑی ہیں، جیسے ایمان کی بنیاد، اور کچھ فصلیں کٹ چکی ہیں، جیسے وہ نیک ارادے جو غفلت نے کاٹ ڈالے۔ جو اپنی بستی کے کھنڈر دیکھ لے اور انہیں پھر سے آباد کرنے لگے، اس کے لیے یہ قصے خبر سے بڑھ کر آئینہ بن جاتے ہیں۔ حضورِ اکرم صلی اللہ علیہ وسلم کو یہ خبریں سنائی گئیں تو یہ آپ صلی اللہ علیہ وسلم کی تسلی بھی تھیں اور امت کے لیے تنبیہ بھی۔

اب وہ آیت آتی ہے جو اس سورت کا حاکم اصول ہے، اور اسے اسی حیثیت سے پڑھنا چاہیے: وَمَا ظَلَمْنٰهُمْ وَلٰكِنْ ظَلَمُوْۤا اَنْفُسَهُمْ فَمَاۤ اَغْنَتْ عَنْهُمْ اٰلِهَتُهُمُ الَّتِيْ يَدْعُوْنَ مِنْ دُوْنِ اللّٰهِ مِنْ شَيْءٍ لَّمَّا جَآءَ اَمْرُ رَبِّكَ ۗ وَمَا زَادُوْهُمْ غَيْرَ تَتْبِيْبٍ۔ “اور ہم نے ان پر ظلم نہیں کیا، بلکہ انہوں نے خود اپنی جانوں پر ظلم کیا؛ تو جب آپ کے رب کا حکم آ پہنچا تو ان کے وہ معبود، جنہیں وہ اللہ کے سوا پکارتے تھے، ان کے کچھ کام نہ آئے، اور انہوں نے ان کی تباہی کے سوا کچھ نہ بڑھایا۔” یہ اصول ہے، اور اس سورت کے ہر قصے پر اس کی مہر لگی ہے۔ حضرت نوح علیہ السلام کی قوم کو ساڑھے نو سو برس دعوت دی گئی؛ قومِ عاد کے پاس حضرت ہود علیہ السلام آئے؛ ثمود کو اونٹنی کی کھلی نشانی دی گئی؛ قومِ لوط کو بار بار ڈرایا گیا؛ اہلِ مدین کو حضرت شعیب علیہ السلام نے خیر خواہی سے سمجھایا؛ اور فرعون کے پاس حضرت موسیٰ علیہ السلام نشانیاں لے کر پہنچے۔ ہر جگہ پہلے رسول، پھر دلیل، پھر مہلت، پھر اعراض، اور اس کے بعد انجام۔ انجام دعوت کے بعد آیا، اعراض کے نتیجے میں آیا، اور کسی قوم پر بے خبری کی حالت میں نہیں آیا۔ ظلم کی اصل تعریف علما یہ کرتے ہیں کہ کسی چیز کو اس کی جگہ کے سوا کہیں اور رکھ دینا؛ اور سب سے بڑا ظلم یہ ہے کہ عبادت، جو صرف حق کا حق ہے، ایسی ہستیوں کو دے دی جائے جو کسی کے کچھ کام نہیں آ سکتیں۔ اس طرح انہوں نے خود اپنی جانوں کو ان کی جگہ سے ہٹایا، اور یہی ان کا اپنے اوپر ظلم تھا۔

صحیح مسلم میں حضرت ابوذر رضی اللہ عنہ سے ایک حدیثِ قدسی مروی ہے جس کا مفہوم یہ ہے کہ حق تعالیٰ فرماتا ہے: اے میرے بندو! میں نے ظلم کو اپنے اوپر حرام کر لیا ہے اور تمہارے درمیان بھی اسے حرام ٹھہرایا ہے؛ اور اسی حدیث کے آخر میں ہے کہ یہ تمہارے ہی اعمال ہیں جنہیں میں تمہارے لیے گن رکھتا ہوں، پھر تمہیں ان کا پورا بدلہ دوں گا؛ تو جو بھلائی پائے وہ اللہ کی حمد کرے، اور جو اس کے سوا کچھ پائے وہ اپنے سوا کسی کو ملامت نہ کرے۔ یہ حدیث اس آیت کی شرح ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کا خلاصہ ہے: حکم محکوم کے مطابق ہوتا ہے، یعنی حق تعالیٰ کا فیصلہ اس چیز کی حقیقت کے موافق ہوتا ہے جس پر فیصلہ ہو رہا ہے، اور حق کا علم معلوم کے تابع ہے؛ وہ بندے کو وہی دیتا ہے جو بندے کی اپنی حقیقت اور اس کا اپنا رخ مانگتا ہے۔ ان قوموں نے منہ پھیرا، تو منہ پھیرنا ہی ان کی گمراہی تھی؛ اور حق نے اس رخ کو وجود دیا جو انہوں نے خود چنا تھا۔ اس میں زبردستی کا کوئی شائبہ نہیں؛ اگر وہ چاہتا تو سب کو مجبور کر کے ایک راستے پر ڈال دیتا، مگر اس نے مجبور نہ کرنا چنا تاکہ انتخاب کی عزت باقی رہے۔ اس لیے یہ کہنا کہ گمراہی اوپر سے ان پر مسلط کی گئی، اس آیت کے صریح خلاف ہے۔ وَمَا ظَلَمْنٰهُمْ پہلے آیا اور وَلٰكِنْ ظَلَمُوْۤا اَنْفُسَهُمْ بعد میں، تاکہ پہلے حق کی ذات سے ہر شبہہ دور ہو، پھر ذمہ داری اپنی جگہ پر رکھ دی جائے۔ اہلِ معرفت کہتے ہیں کہ یہ آیت ان تمام آیات کی کنجی ہے جن میں گمراہی کا ذکر حق کی طرف منسوب ہو کر آتا ہے: وہاں بھی نسبت فیصلے کی ہے، زبردستی کی نہیں، اور فیصلہ بندے کے اپنے اعراض کے بعد آتا ہے۔

آیت کا دوسرا حصہ معبودوں کی حقیقت کھولتا ہے: فَمَاۤ اَغْنَتْ عَنْهُمْ اٰلِهَتُهُمُ۔ جن کو انہوں نے مشکل کے لیے سنبھال کر رکھا تھا، وہ عین مشکل کے وقت کچھ کام نہ آئے؛ بلکہ وَمَا زَادُوْهُمْ غَيْرَ تَتْبِيْبٍ، انہوں نے صرف ان کا خسارہ بڑھایا۔ تَتْبِيْب کا مادہ تَبَاب ہے، یعنی ہلاکت اور ٹوٹا؛ جھوٹا سہارا صرف بے فائدہ نہیں ہوتا، وہ نقصان کو دوگنا کر دیتا ہے، کیونکہ آدمی اس پر بھروسا کر کے سچے سہارے کی طرف جانا چھوڑ دیتا ہے۔ شیخِ اکبر کے مکتب میں عالم کو حق کا ظل کہا جاتا ہے، یعنی سایہ؛ سایہ اپنی اصل کے ساتھ قائم ہے اور اس کے بغیر کچھ نہیں۔ جو دل اصل کو چھوڑ کر سایے پر ٹیک لگائے، اس کا ٹیک ایسی دیوار پر ہے جس کی اپنی کوئی بنیاد نہیں؛ اور جب لَمَّا جَآءَ اَمْرُ رَبِّكَ کا لمحہ آتا ہے تو ہر سایہ اپنی اصل میں لوٹ جاتا ہے اور سہارا لینے والا خالی ہاتھ رہ جاتا ہے۔ اہلِ ذوق کہتے ہیں کہ ہر دور کے اپنے اٰلِهَة ہیں: کسی کا مال، کسی کا منصب، کسی کی شہرت، اور کسی کا اپنا نفس؛ یہ بات اپنے دل کو ٹٹولنے کے لیے ہے، کسی دوسرے پر حکم لگانے کے لیے نہیں۔ جو کچھ بھی حق کے سوا سہارا بن جائے، آزمائش کی گھڑی میں وہ یہی کرتا ہے: کچھ بچاتا نہیں، کچھ بڑھاتا نہیں، سوائے خسارے کے۔

پھر حق کی گرفت کی شان بیان ہوئی: وَكَذٰلِكَ اَخْذُ رَبِّكَ اِذَاۤ اَخَذَ الْقُرٰى وَهِيَ ظٰلِمَةٌ ۗ اِنَّ اَخْذَهٗۤ اَلِيْمٌ شَدِيْدٌ۔ “اور آپ کے رب کی پکڑ ایسی ہی ہوتی ہے جب وہ بستیوں کو پکڑتا ہے اس حال میں کہ وہ ظلم کر رہی ہوں؛ بے شک اس کی پکڑ دردناک اور سخت ہے۔” صحیح بخاری اور صحیح مسلم میں حضرت ابو موسیٰ اشعری رضی اللہ عنہ سے مروی ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ اللہ ظالم کو ڈھیل دیتا رہتا ہے، یہاں تک کہ جب اسے پکڑتا ہے تو پھر اسے چھوٹنے نہیں دیتا؛ پھر آپ صلی اللہ علیہ وسلم نے یہی آیت تلاوت فرمائی۔ آیت میں ایک شرط ہے جس پر غور ضروری ہے: وَهِيَ ظٰلِمَةٌ، یعنی پکڑ اس حال سے بندھی ہے۔ جو بستی ظلم سے پلٹ آئے، اس کا حکم بدل جاتا ہے؛ اسی سورت سے پہلے سورۂ یونس میں حضرت یونس علیہ السلام کی قوم کا ذکر گزرا: اِلَّا قَوْمَ يُوْنُسَ ۚ لَمَّاۤ اٰمَنُوْا كَشَفْنَا عَنْهُمْ عَذَابَ الْخِزْيِ۔ انہوں نے رخ بدلا تو عذاب ٹل گیا۔ اہلِ معرفت کہتے ہیں کہ ڈھیل، جسے علما املاء اور امہال کہتے ہیں، اپنی ذات میں رحمت ہے، کیونکہ رحمت غضب پر سبقت رکھتی ہے؛ مہلت کا ہر دن لوٹ آنے کی ایک اور دعوت ہے، اور دروازہ اس وقت تک کھلا ہے جب تک مہلت باقی ہے۔ مگر جو ڈھیل کو رضامندی سمجھ بیٹھے، وہ اسی مہلت کو اپنے خلاف حجت بنا لیتا ہے۔ حق کا نام اَلْحَلِيْم پہلے جلوہ کرتا ہے، اور جو حلم کی قدر نہ کرے اس پر شَدِيْدُ الْعِقَاب کا ظہور ہوتا ہے؛ ترتیب یہ ہے، اور اس ترتیب میں بھی رحمت کی سبقت ہے۔

اب نظر دنیا کی بستیوں سے اٹھ کر آخرت کے دن پر جاتی ہے: اِنَّ فِيْ ذٰلِكَ لَاٰيَةً لِّمَنْ خَافَ عَذَابَ الْاٰخِرَةِ ۗ ذٰلِكَ يَوْمٌ مَّجْمُوْعٌ لَّهُ النَّاسُ وَذٰلِكَ يَوْمٌ مَّشْهُوْدٌ۔ “بے شک اس میں اس کے لیے بڑی نشانی ہے جو آخرت کے عذاب سے ڈرے؛ وہ ایسا دن ہے جس کے لیے سب لوگ اکٹھے کیے جائیں گے، اور وہ ایسا دن ہے جس میں سب حاضر ہوں گے۔” اور: وَمَا نُؤَخِّرُهٗۤ اِلَّا لِاَجَلٍ مَّعْدُوْدٍ۔ “اور ہم اسے صرف ایک گنی ہوئی مدت تک کے لیے مؤخر کر رہے ہیں۔” آیت نے کہا کہ یہ قصے نشانی ہیں، مگر صرف اس کے لیے جو آخرت سے ڈرتا ہو۔ ایک ہی قصہ ایک قاری کے لیے تاریخ ہے اور دوسرے کے لیے آیت؛ فرق قصے میں نہیں، پڑھنے والے کی استعداد میں ہے۔ شیخِ اکبر کے نزدیک ہر شے کو اپنی استعداد کے مطابق ہی فیض ملتا ہے؛ جس کے دل میں آخرت کا خوف کھلا ہے، اس میں عبرت اترنے کی جگہ ہے، اور جس نے یہ دریچہ بند کر رکھا ہے، اس کے لیے کھنڈر صرف پتھر ہیں۔ مَشْهُوْدٌ کے معنی ہیں جس میں حاضری ہو اور جو آنکھوں کے سامنے ہو؛ اس دن ہر چھپی چیز شہود میں آ جائے گی۔ اہلِ معرفت کہتے ہیں کہ عارف اسی دن کو آج مراقبے میں جیتا ہے؛ وہ اپنے ہر عمل کو اس طرح دیکھتا ہے جیسے وہ اس دن کے سامنے پیش ہو رہا ہو۔ اور مَعْدُوْد میں ایک لطیف اشارہ ہے: مدت گنی ہوئی ہے، یعنی سانسوں میں ناپی ہوئی؛ شیخِ اکبر کے مکتب میں تجددِ امثال کی تعلیم ہے کہ وجود ہر آن نئے سرے سے عطا ہوتا ہے، اور ہر نیا آن اس گنتی میں سے ایک کم کر دیتا ہے۔ جو اپنی سانسوں کو گنتی میں سمجھے، وہ انہیں ضائع نہیں کرتا۔

پھر اس دن کی ہیبت: يَوْمَ يَاْتِ لَا تَكَلَّمُ نَفْسٌ اِلَّا بِاِذْنِهٖ ۚ فَمِنْهُمْ شَقِيٌّ وَّسَعِيْدٌ۔ “جس دن وہ آئے گا، کوئی جان اس کی اجازت کے بغیر بات نہ کر سکے گی؛ پھر ان میں سے کوئی بدبخت ہوگا اور کوئی نیک بخت۔” اس دن زبانیں اسی کی اجازت سے کھلیں گی، شفاعت بھی اسی کی اجازت سے ہوگی۔ جامع ترمذی میں حضرت عمر رضی اللہ عنہ سے مروی ہے کہ جب یہ آیت اتری تو انہوں نے عرض کیا کہ یا رسول اللہ صلی اللہ علیہ وسلم! پھر ہم کس بنیاد پر عمل کریں، کسی ایسی بات پر جس سے فراغت ہو چکی یا ایسی بات پر جس سے فراغت نہیں ہوئی؟ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ اس بات پر جس سے فراغت ہو چکی اور جس پر قلم چل چکے، لیکن ہر ایک کے لیے وہ آسان کر دیا جاتا ہے جس کے لیے وہ پیدا کیا گیا۔ اور صحیحین کی ایک حدیث میں اسی مضمون کے ساتھ یہ حکم ہے کہ عمل کرو، کیونکہ ہر ایک کے لیے آسانی ہے؛ پھر آپ صلی اللہ علیہ وسلم نے سورۂ اللیل کی آیت تلاوت فرمائی جس میں ہے: فَسَنُيَسِّرُهٗ لِلْيُسْرٰى۔ اس حدیث کو حاکم اصول کی روشنی میں پڑھنا ضروری ہے۔ لکھا جا چکنا علم کا لکھا جانا ہے، اور علم مجبور نہیں کرتا؛ حق تعالیٰ کا یہ جاننا کہ بندہ کیا چنے گا، بندے کو اس انتخاب پر مجبور نہیں کرتا۔ شیخِ اکبر کے نزدیک علم معلوم کے تابع ہے: حق نے بندے کو ویسا ہی جانا جیسا وہ اپنی حقیقت میں ہے، اور اسی کے مطابق لکھا؛ تحریر نے حقیقت نہیں بنائی، حقیقت کو ظاہر کیا۔ اسی لیے حدیث کا آخری حکم بیٹھ رہنا نہیں بلکہ عمل ہے، اور آسانی اسی راستے میں ملتی ہے جس کی طرف بندہ خود قدم اٹھائے۔ اہلِ ذوق ایک اور نکتہ لیتے ہیں: لَا تَكَلَّمُ نَفْسٌ اِلَّا بِاِذْنِهٖ کا ادب جو آج سیکھ لے، کہ اس کی زبان حق کی رضا کے بغیر نہ کھلے، وہ کل بے خوف ہوگا۔

پھر دونوں گروہوں کا انجام الگ الگ بیان ہوا: فَاَمَّا الَّذِيْنَ شَقُوْا فَفِي النَّارِ لَهُمْ فِيْهَا زَفِيْرٌ وَّشَهِيْقٌ۔ “تو جو بدبخت ہوئے وہ آگ میں ہوں گے، وہاں ان کے لیے چیخنا اور دھاڑنا ہوگا۔” خٰلِدِيْنَ فِيْهَا مَا دَامَتِ السَّمٰوٰتُ وَالْاَرْضُ اِلَّا مَا شَآءَ رَبُّكَ ۗ اِنَّ رَبَّكَ فَعَّالٌ لِّمَا يُرِيْدُ۔ “وہ اس میں ہمیشہ رہیں گے جب تک آسمان اور زمین قائم ہیں، مگر جو آپ کا رب چاہے؛ بے شک آپ کا رب جو چاہے کر گزرنے والا ہے۔” زَفِيْر سانس کا درد سے باہر نکلنا ہے اور شَهِيْق اس کا کھنچ کر اندر جانا؛ مفسرین نے اسے شدید کرب کی کیفیت کہا ہے۔ مَا دَامَتِ السَّمٰوٰتُ وَالْاَرْضُ کے بارے میں مفسرین کے دو قول ہیں: ایک یہ کہ عرب ہمیشگی کو اسی محاورے سے ادا کرتے تھے، یعنی جب تک آسمان و زمین ہیں، مراد ہمیشہ؛ اور دوسرا یہ کہ اس سے آخرت کے آسمان و زمین مراد ہیں، جیسا کہ سورۂ ابراہیم میں ہے: يَوْمَ تُبَدَّلُ الْاَرْضُ غَيْرَ الْاَرْضِ وَالسَّمٰوٰتُ، اور وہ ہمیشہ رہنے والے ہیں۔ اِلَّا مَا شَآءَ رَبُّكَ کے استثنا کے بارے میں جمہور مفسرین کہتے ہیں کہ اس سے اہلِ توحید میں سے وہ گنہگار مراد ہیں جو اپنے گناہوں کی وجہ سے آگ میں داخل ہوں گے، پھر حضورِ اکرم صلی اللہ علیہ وسلم کی شفاعت اور حق کی رحمت سے نکالے جائیں گے، جیسا کہ صحیح احادیثِ شفاعت میں آیا ہے؛ اور خُلُود کا حکم کافروں کے لیے ہے، جیسا کہ اہلِ سنت کا عقیدہ ہے۔ بعض نے یہ بھی کہا کہ استثنا یہ بتانے کے لیے ہے کہ ہر حال میں حکم اسی کی مشیت سے ہے، اس پر کوئی چیز لازم نہیں کی جا سکتی۔ آیت کا آخری جملہ سورۂ البروج کی اس صفت کو دہراتا ہے جو وہاں گزری: فَعَّالٌ لِّمَا يُرِيْدُ۔ اور یہاں ادب یہ ہے کہ اسی ظاہر پر ٹھہرا جائے جو مفسرین نے بیان کیا۔

پھر نیک بختوں کا ذکر: وَاَمَّا الَّذِيْنَ سُعِدُوْا فَفِي الْجَنَّةِ خٰلِدِيْنَ فِيْهَا مَا دَامَتِ السَّمٰوٰتُ وَالْاَرْضُ اِلَّا مَا شَآءَ رَبُّكَ ۗ عَطَآءً غَيْرَ مَجْذُوْذٍ۔ “اور جو نیک بخت بنائے گئے وہ جنت میں ہوں گے، اس میں ہمیشہ رہیں گے جب تک آسمان اور زمین قائم ہیں، مگر جو آپ کا رب چاہے؛ یہ ایسی عطا ہے جو کبھی کٹنے والی نہیں۔” یہاں کا استثنا مفسرین کے نزدیک اس مدت کی طرف اشارہ ہے جو جنت میں داخلے سے پہلے گزرتی ہے، جیسے میدانِ حشر کا قیام، یا بعض گنہگار اہلِ ایمان کا جنت میں دیر سے داخل ہونا؛ اور فوراً بعد عَطَآءً غَيْرَ مَجْذُوْذٍ نے ہر شبہے کا دروازہ بند کر دیا کہ جنت کی نعمت کبھی منقطع نہ ہوگی۔ اب ایک نکتہ جو اہلِ تفسیر نے بڑی لطافت سے بیان کیا ہے: بدبختوں کے لیے فعل معروف آیا، شَقُوْا، یعنی وہ خود بدبخت ہوئے؛ اور نیک بختوں کے لیے مشہور قرأت میں فعل مجہول آیا، سُعِدُوْا، یعنی انہیں نیک بخت بنایا گیا۔ بدبختی بندے کی اپنی کمائی ہے اور اس کی نسبت اسی کی طرف ہے؛ نیک بختی حق کی عطا ہے اور اس کی نسبت عطا کرنے والے کی طرف۔ یہ وہی حاکم اصول ہے جو آیت ۱۰۱ میں گزرا، اب صرف ایک فعل کے صیغے میں سمٹ آیا ہے۔ اور اسی لیے آیت کے آخر میں اجر نہیں بلکہ عَطَآء کہا گیا: جنت عمل کی قیمت نہیں، فضل کی بخشش ہے، اگرچہ عمل اس کا راستہ ہے۔ صحیحین کی حدیث میں ہے، جس کا مفہوم یہ ہے کہ کسی کو اس کا عمل جنت میں داخل نہیں کرے گا جب تک اللہ اسے اپنی رحمت سے ڈھانپ نہ لے۔

شیخِ اکبر کے مکتب میں غَيْرَ مَجْذُوْذٍ کو خلقِ جدید کی تعلیم سے جوڑا جاتا ہے۔ مَجْذُوْذ وہ ہے جو کاٹ دیا جائے اور منقطع ہو جائے؛ اور جنت کی عطا اس لیے منقطع نہیں ہوتی کہ اس کی ہر نعمت ہر آن نئے سرے سے عطا ہوتی ہے۔ اہلِ معرفت کہتے ہیں کہ دنیا کی لذت اس لیے ماند پڑ جاتی ہے کہ آدمی اسے پرانی سمجھنے لگتا ہے؛ جنت میں تجددِ امثال کا ظہور کامل ہوگا، ہر لمحہ نئی عطا، نئی تجلی، نیا شہود، اس لیے وہاں اکتاہٹ نہیں۔ اور اس مکتب میں سعادت کی اصل لذت دیدار ہے؛ باقی سب نعمتیں اسی کے گرد ہیں۔ سعید کون ہے؟ اہلِ ذوق کہتے ہیں کہ سعید وہ ہے جس نے یہاں عطا کو عطا جانا، اپنا کمال نہ سمجھا؛ اور جس نے عطا کو اپنی کمائی سمجھ لیا، وہ شکر کے دروازے سے دور ہو گیا۔ یہ تقسیم، شقی اور سعید، اسی دن کھلے گی، مگر اس کے بیج آج بوئے جا رہے ہیں؛ ہر رخ، ہر انتخاب، ہر توبہ اس کھیتی کی ایک کیاری ہے۔

رکوع کا اختتام حضورِ اکرم صلی اللہ علیہ وسلم کو خطاب پر ہوتا ہے: فَلَا تَكُ فِيْ مِرْيَةٍ مِّمَّا يَعْبُدُ هٰۤؤُلَآءِ ۗ مَا يَعْبُدُوْنَ اِلَّا كَمَا يَعْبُدُ اٰبَآؤُهُمْ مِّنْ قَبْلُ ۗ وَاِنَّا لَمُوَفُّوْهُمْ نَصِيْبَهُمْ غَيْرَ مَنْقُوْصٍ۔ “تو آپ ان چیزوں کے بارے میں کسی شک میں نہ رہیں جنہیں یہ لوگ پوجتے ہیں؛ یہ تو بس اسی طرح پوجتے ہیں جیسے ان سے پہلے ان کے باپ دادا پوجتے تھے، اور ہم انہیں ان کا حصہ پورا پورا دینے والے ہیں، بغیر کسی کمی کے۔” مفسرین کہتے ہیں کہ خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے اور مراد امت ہے؛ آپ صلی اللہ علیہ وسلم کو کبھی کوئی شک نہ تھا، یہ قرآن کا وہ اسلوب ہے جس میں پیغام امام کو مخاطب کر کے پوری جماعت تک پہنچایا جاتا ہے۔ ان کی عبادت کی بنیاد کوئی دلیل نہیں، صرف یہ ہے کہ باپ دادا یوں کرتے تھے؛ یہی تقلیدِ محض ہے جو حق کی جستجو کا دروازہ بند کر دیتی ہے۔ نَصِيْبَهُمْ غَيْرَ مَنْقُوْصٍ کے بارے میں مفسرین کے دو قول ہیں: آخرت میں ان کے عمل کی پوری جزا، یا دنیا میں ان کا رزق اور مہلت جو انہیں کفر کے باوجود پوری دی جاتی ہے؛ اور دونوں میں حق کا عدل اور اس کی رحمت ظاہر ہے۔ اہلِ معرفت اس میں سرِّ قدر کا ایک اور رخ دیکھتے ہیں: ہر ایک کو وہی حصہ ملتا ہے جو اس کے اپنے رخ نے مانگا، نہ اس سے کم نہ اس سے زیادہ ظلم کے طور پر؛ البتہ نیکی کا حصہ حق اپنے فضل سے بڑھا دیتا ہے، کیونکہ وہاں عدل کے اوپر فضل ہے۔ اور اہلِ ذوق اس آیت سے اپنے لیے یہ سبق لیتے ہیں کہ ایمان والا بھی ورثے میں ملے ہوئے ایمان پر شکر کرے، مگر وہاں رک نہ جائے؛ علم الیقین سے عین الیقین کی طرف سفر جاری رکھے، تاکہ جو بات اس نے سنی ہے وہ اس کے دل کا مشاہدہ بن جائے۔ اس کا مقصد کسی کے ایمان کو کم تر کہنا نہیں، بلکہ ہر ایک کو آگے بڑھنے کی دعوت دینا ہے۔
