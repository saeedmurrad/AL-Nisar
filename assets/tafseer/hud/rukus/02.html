یہ رکوع انسان کے دل کا آئینہ لے کر شروع ہوتا ہے: وَلَئِنْ اَذَقْنَا الْاِنْسَانَ مِنَّا رَحْمَةً ثُمَّ نَزَعْنٰهَا مِنْهُ ۚ اِنَّهٗ لَيَئُوْسٌ كَفُوْرٌ۔ “اور اگر ہم انسان کو اپنی طرف سے کسی رحمت کا مزہ چکھائیں، پھر اسے اس سے چھین لیں، تو وہ نہایت مایوس اور بڑا ناشکرا ہو جاتا ہے۔” لفظ اَذَقْنَا پر غور کیجیے: رحمت دی نہیں گئی کہ ملکیت بن جائے، چکھائی گئی؛ دنیا کی ہر نعمت ایک ذائقہ ہے، پورا دسترخوان نہیں۔ اور مِنَّا یاد دلاتا ہے کہ ذائقہ بھی ہماری طرف سے تھا۔ پھر نَزَعْنٰهَا، یعنی کھینچ کر نکال لینا، جیسے کوئی چیز جڑ سے اکھاڑی جائے؛ یہ لفظ بتاتا ہے کہ انسان نعمت سے کس قدر چمٹ جاتا ہے کہ اس کا جانا اکھڑنے جیسا لگتا ہے۔ اس کا ردِ عمل دو لفظوں میں ہے: يَئُوْسٌ، یعنی ایسا مایوس جو نعمت کے لوٹنے کی امید ہی چھوڑ دے، اور كَفُوْرٌ، یعنی ایسا ناشکرا جو پچھلی ساری عطا بھول جائے۔ اہلِ معرفت کہتے ہیں کہ مایوسی اسی کو گھیرتی ہے جو ذائقے کو اپنا حق سمجھ بیٹھا ہو؛ جس نے جانا کہ ذائقہ عطا تھا، وہ اس کے جانے پر بھی دینے والے سے امید رکھتا ہے۔ اہلِ تصوف کی اصطلاح میں ذوق وہ پہلا چکھنا ہے جو سالک کو کسی حقیقت کا ملتا ہے، اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کہا جاتا ہے کہ ذوق کے بعد کبھی اس کا چھن جانا بھی تربیت ہے، تاکہ سالک ذائقے کا نہیں، ذائقہ دینے والے کا طالب بنے۔ جو حال کے چھن جانے پر مایوس ہو، اس نے حال کو پوجا تھا۔

اور اس کا الٹا رخ: وَلَئِنْ اَذَقْنٰهُ نَعْمَآءَ بَعْدَ ضَرَّآءَ مَسَّتْهُ لَيَقُوْلَنَّ ذَهَبَ السَّيِّاٰتُ عَنِّيْ ؕ اِنَّهٗ لَفَرِحٌ فَخُوْرٌ۔ “اور اگر ہم اسے کسی تکلیف کے بعد جو اسے پہنچی تھی، نعمت کا مزہ چکھائیں تو وہ ضرور کہتا ہے کہ میری ساری سختیاں مجھ سے دور ہو گئیں؛ بے شک وہ اترانے والا، شیخی بگھارنے والا ہے۔” یہاں تکلیف کے لیے مَسَّتْهُ آیا، یعنی صرف چھو گئی؛ مصیبت بھی پوری نہیں آئی تھی، اسے صرف چھوا تھا۔ اور اس کی زبان پر جو جملہ آیا وہ قابلِ غور ہے: “سختیاں مجھ سے چلی گئیں”، گویا وہ خود سے چلی گئیں، کسی نے انہیں دور نہیں کیا۔ اس جملے میں نہ دور کرنے والے کا ذکر ہے نہ شکر کا؛ اور اس کے بعد کی حالت یقینی ہے کہ اب کبھی نہ آئیں گی۔ یہاں فَرِحٌ وہ خوشی ہے جو اترانے میں بدل جائے، اور فَخُوْرٌ وہ جو نعمت کو اپنی قابلیت کا تمغہ بنا لے۔ اہلِ علم بتاتے ہیں کہ قرآن میں فرح دو طرح کا ہے: ایک اللہ کے فضل پر خوش ہونا جو محمود ہے، اور دوسرا نعمت پر اترانا جو مذموم ہے؛ فرق یہ ہے کہ پہلی خوشی دینے والے کی طرف دیکھتی ہے اور دوسری اپنی طرف۔ اہلِ معرفت کے نزدیک قبض اور بسط دونوں حق کے اسماء کے احوال ہیں جو باری باری دل پر آتے ہیں؛ جو قبض میں ٹوٹ جائے اور بسط میں پھول جائے وہ تلوین میں ہے، یعنی احوال کا کھلونا، اور جو دونوں میں دینے والے کو دیکھتا رہے وہ تمکین میں ہے۔

اب استثنا آتا ہے جو ساری تصویر بدل دیتا ہے: اِلَّا الَّذِيْنَ صَبَرُوْا وَعَمِلُوا الصّٰلِحٰتِ ؕ اُولٰۤئِكَ لَهُمْ مَّغْفِرَةٌ وَّاَجْرٌ كَبِيْرٌ۔ “سوائے ان لوگوں کے جنہوں نے صبر کیا اور نیک عمل کیے؛ انہی کے لیے مغفرت ہے اور بڑا اجر۔” یہ استثنا دونوں حالتوں پر ہے: صبر کرنے والا نہ تنگی میں مایوس ہوتا ہے نہ کشادگی میں اترتا ہے۔ اکثر لوگ صبر کو صرف مصیبت سے جوڑتے ہیں، مگر یہ آیت بتاتی ہے کہ نعمت پر بھی صبر لازم ہے، یعنی نفس کو اترانے سے روکنا۔ حضرت عبدالرحمٰن بن عوف رضی اللہ عنہ سے منقول ہے جس کا مفہوم یہ ہے کہ ہمیں تنگی سے آزمایا گیا تو ہم نے صبر کیا، اور کشادگی سے آزمایا گیا تو ہم صبر نہ کر سکے؛ یہ ان کی انکساری تھی، اور اس میں یہ سبق ہے کہ نعمت کی آزمائش مصیبت کی آزمائش سے زیادہ باریک ہے۔ صحیح مسلم کی روایت کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: مومن کا معاملہ عجیب ہے، اس کے لیے ہر حال میں خیر ہے؛ اگر خوشی پہنچے تو شکر کرتا ہے اور یہ اس کے لیے بہتر ہے، اور اگر تکلیف پہنچے تو صبر کرتا ہے اور یہ بھی اس کے لیے بہتر ہے۔ گویا يَئُوْسٌ اور كَفُوْرٌ کا علاج صبر ہے، اور فَرِحٌ فَخُوْرٌ کا علاج شکر؛ اور یہ دونوں ایک ہی حقیقت کے دو چہرے ہیں: ہر حال میں دینے والے کو دیکھنا۔

صبر کے ساتھ عمل کا ذکر کیوں آیا؟ اہلِ معرفت کہتے ہیں کہ صبر محض رک جانا نہیں، رک کر کچھ کرنا ہے؛ جو صرف شکایت سے رکا رہے مگر عمل سے خالی ہو، اس کا صبر جمود ہے۔ شیخِ اکبر کے مکتب میں صبر کی ایک لطیف تعریف ملتی ہے، جس کا مفہوم یہ ہے کہ صبر نفس کو غیرِ حق کے سامنے شکایت سے روکنا ہے، اور اللہ کے حضور اپنا دکھ رکھ دینا صبر کے خلاف نہیں بلکہ عین عبدیت ہے۔ اسی لیے حضرت یعقوب علیہ السلام نے، جن کا صبر صبرِ جمیل کہلایا، فرمایا: اِنَّمَاۤ اَشْكُوْا بَثِّيْ وَحُزْنِيْۤ اِلَى اللّٰهِ؛ شکایت تھی، مگر صرف اسی سے۔ اس مکتب میں کہا جاتا ہے کہ جو مصیبت میں اپنے رب کو نہ پکارے وہ گویا قہر کے مقابلے پر کھڑا ہو گیا، اور یہ ادب کے خلاف ہے؛ بندگی کا حسن یہ ہے کہ بندہ اپنی کمزوری کو مانے اور اسی کے دروازے پر رکھے۔ پھر انعام: مَّغْفِرَةٌ وَّاَجْرٌ كَبِيْرٌ۔ مغفرت پہلے ہے، کیونکہ صبر کرنے والا بھی کمی سے خالی نہیں، اور اجر بعد میں؛ اور اجر کو كَبِيْرٌ کہا، یعنی ان کے صبر کے اندازے سے کہیں بڑا۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں صبر اور عمل کا بدلہ جنت ہے، اور باطن میں صبر کرنے والا اسی دنیا میں تمکین کا سکون پا لیتا ہے۔

اب خطاب حضورِ اکرم صلی اللہ علیہ وسلم کی طرف مڑتا ہے: فَلَعَلَّكَ تَارِكٌۢ بَعْضَ مَا يُوْحٰۤى اِلَيْكَ وَضَآئِقٌۢ بِهٖ صَدْرُكَ اَنْ يَّقُوْلُوْا لَوْلَاۤ اُنْزِلَ عَلَيْهِ كَنْزٌ اَوْ جَآءَ مَعَهٗ مَلَكٌ ؕ اِنَّمَاۤ اَنْتَ نَذِيْرٌ ؕ وَاللّٰهُ عَلٰى كُلِّ شَيْءٍ وَّكِيْلٌ۔ “تو کہیں ایسا تو نہیں کہ آپ صلی اللہ علیہ وسلم اس وحی کا کچھ حصہ چھوڑ دینے والے ہوں جو آپ کی طرف بھیجی جاتی ہے، اور اس سے آپ کا سینہ تنگ ہو، اس بات پر کہ وہ کہتے ہیں: اس پر کوئی خزانہ کیوں نہ اتارا گیا، یا اس کے ساتھ کوئی فرشتہ کیوں نہ آیا؟ آپ صلی اللہ علیہ وسلم تو بس ڈرانے والے ہیں، اور اللہ ہر چیز کا کارساز ہے۔” یہاں ادب کا پہلا اصول یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے وحی کا ایک حرف بھی کبھی نہیں چھوڑا؛ آپ صلی اللہ علیہ وسلم تبلیغ میں معصوم تھے۔ مفسرین کہتے ہیں کہ یہاں لَعَلَّ شک کے لیے نہیں، تسلی اور تقویت کے لیے ہے: گویا فرمایا جا رہا ہے کہ ان کی باتوں کا بوجھ ہمارے علم میں ہے، مگر ان کی باتیں آپ صلی اللہ علیہ وسلم کو پیغام سے ذرا بھی نہ ہٹائیں۔ سینے کی تنگی ایک انسانی کیفیت تھی جو قوم کی ضد پر طبعی طور پر پیدا ہوتی ہے، اور یہ آیت اس کیفیت پر رحمت کا مرہم ہے۔ اِنَّمَاۤ اَنْتَ نَذِيْرٌ میں بوجھ اتار دیا گیا: نتیجے کی ذمہ داری آپ صلی اللہ علیہ وسلم پر نہیں، پہنچانے کی ذمہ داری ہے؛ اور وَكِيْلٌ، یعنی کام بنانے والا، اللہ ہے۔

اہلِ معرفت کہتے ہیں کہ منکروں نے خزانہ مانگا اور یہ نہ دیکھا کہ خزانہ ان کے سامنے کھڑا ہے۔ وہ زمین کا دفینہ چاہتے تھے، اور جو اترا وہ آسمان کا خزانہ تھا: وحی، جس کے ایک کلمے کے سامنے دنیا کے سب دفینے ہیچ ہیں۔ اہلِ تصوف میں مشہور اثر ہے کہ حق فرماتا ہے: میں ایک چھپا ہوا خزانہ تھا، میں نے چاہا کہ پہچانا جاؤں؛ اور اہلِ ذوق کہتے ہیں کہ اس پہچان کا سب سے کامل مظہر حضورِ اکرم صلی اللہ علیہ وسلم کی ذاتِ مبارکہ ہے، جنہیں شیخِ اکبر کے مکتب میں انسانِ کامل اور اسمِ جامع کا مظہر کہا جاتا ہے۔ وہ کہتے تھے کہ اس پر خزانہ کیوں نہ اترا، حالانکہ حضورِ اکرم صلی اللہ علیہ وسلم خود معرفت کے خزانے کا دروازہ تھے۔ اسی طرح انہوں نے فرشتہ مانگا جو ان کی آنکھوں کے سامنے چلے؛ حالانکہ حضرت جبریل علیہ السلام بار بار آپ صلی اللہ علیہ وسلم کے قلبِ مبارک پر اترتے تھے۔ آنکھ کا فرشتہ دیکھنے کی خواہش اس دل کی علامت ہے جو باطن کے دروازے بند کر چکا ہو۔ سالک کے لیے اس میں یہ سبق ہے کہ لوگ اکثر حق کے پیغام سے پہلے اس کے ساتھ دنیا کی چمک مانگتے ہیں، اور داعی کا ادب یہ ہے کہ پیغام کو اس چمک کے لیے نہ بدلے؛ اس کا کام نذیر ہونا ہے، اور کارسازی اللہ کے ہاتھ میں ہے۔

پھر منکروں کا دعویٰ اور اس کا جواب: اَمْ يَقُوْلُوْنَ افْتَرٰىهُ ؕ قُلْ فَاْتُوْا بِعَشْرِ سُوَرٍ مِّثْلِهٖ مُفْتَرَيٰتٍ وَّادْعُوْا مَنِ اسْتَطَعْتُمْ مِّنْ دُوْنِ اللّٰهِ اِنْ كُنْتُمْ صٰدِقِيْنَ۔ “کیا وہ کہتے ہیں کہ اس نے اسے گھڑ لیا ہے؟ کہہ دیجیے: تو تم بھی اس جیسی دس سورتیں گھڑی ہوئی لے آؤ، اور اللہ کے سوا جسے بلا سکتے ہو بلا لو، اگر تم سچے ہو۔” فَاِلَّمْ يَسْتَجِيْبُوْا لَكُمْ فَاعْلَمُوْۤا اَنَّمَاۤ اُنْزِلَ بِعِلْمِ اللّٰهِ وَاَنْ لَّاۤ اِلٰهَ اِلَّا هُوَ ۚ فَهَلْ اَنْتُمْ مُّسْلِمُوْنَ۔ “پھر اگر وہ تمہاری بات کا جواب نہ دیں تو جان لو کہ یہ اللہ کے علم کے ساتھ اتارا گیا ہے، اور یہ کہ اس کے سوا کوئی معبود نہیں؛ تو کیا تم فرمانبردار ہوتے ہو؟” مفسرین نے بتایا کہ قرآن کی تحدی، یعنی مقابلے کی للکار، مرحلہ وار آئی: پورے قرآن جیسا، پھر دس سورتیں، پھر ایک سورت؛ اور یہاں مُفْتَرَيٰتٍ کا لفظ لا کر نرمی کی انتہا کر دی: سچی نہ سہی، گھڑی ہوئی ہی لے آؤ، صرف بیان میں اس جیسی ہوں۔ عرب فصاحت کے بادشاہ تھے اور پھر بھی عاجز رہے۔ اس کی وجہ بِعِلْمِ اللّٰهِ میں بتائی گئی: یہ کلام اپنے کہنے والے کے علم کو اٹھائے ہوئے ہے۔ شیخِ اکبر کے نزدیک ہر کلام اپنے متکلم کے مرتبے کا آئینہ ہے؛ انسان کا کلام اس کے محدود علم کا اثر رکھتا ہے، اور یہ کلام اس علم کا اثر رکھتا ہے جو ہر شے کو گھیرے ہوئے ہے۔ اسی لیے اس کا ہر لفظ ہر زمانے کے پڑھنے والے پر نیا کھلتا ہے، اور کوئی گھڑنے والا ایسی گہرائی نہیں گھڑ سکتا۔ اور آخر میں فَهَلْ اَنْتُمْ مُّسْلِمُوْنَ: سخت للکار کے بعد بھی دعوت کا لہجہ نرم ہے؛ دلیل دکھا کر دروازہ کھلا رکھا گیا، کسی کو زبردستی اندر نہیں دھکیلا گیا۔

اب دو طرح کے ارادوں کا بیان شروع ہوتا ہے: مَنْ كَانَ يُرِيْدُ الْحَيٰوةَ الدُّنْيَا وَزِيْنَتَهَا نُوَفِّ اِلَيْهِمْ اَعْمَالَهُمْ فِيْهَا وَهُمْ فِيْهَا لَا يُبْخَسُوْنَ۔ “جو کوئی دنیا کی زندگی اور اس کی زینت چاہتا ہے، ہم اس کے اعمال کا بدلہ اسی دنیا میں پورا دے دیتے ہیں اور اس میں ان کی کوئی کمی نہیں کی جاتی۔” اُولٰۤئِكَ الَّذِيْنَ لَيْسَ لَهُمْ فِي الْاٰخِرَةِ اِلَّا النَّارُ ۖ وَحَبِطَ مَا صَنَعُوْا فِيْهَا وَبٰطِلٌ مَّا كَانُوْا يَعْمَلُوْنَ۔ “یہی وہ لوگ ہیں جن کے لیے آخرت میں آگ کے سوا کچھ نہیں، اور جو کچھ انہوں نے دنیا میں بنایا وہ برباد ہو گیا، اور جو کچھ وہ کرتے رہے وہ بے کار ہے۔” ان آیات کا عدل دیکھیے: دنیا چاہنے والے کا بھی حق نہیں مارا جاتا؛ اس کی محنت کا پھل اسے یہیں پورا ملتا ہے، شہرت، دولت یا تعریف کی صورت میں۔ مگر اس نے جو مانگا وہی پایا، اور آخرت کو اس نے مانگا ہی نہ تھا۔ صحیح مسلم میں حضرت ابوہریرہ رضی اللہ عنہ کی روایت کا مفہوم یہ ہے کہ قیامت کے دن سب سے پہلے ایک عالم، ایک شہید اور ایک سخی کا فیصلہ ہوگا جنہوں نے دکھاوے کے لیے عمل کیا تھا؛ ان سے کہا جائے گا کہ تم نے یہ اس لیے کیا تھا کہ لوگ کہیں، سو کہہ دیا گیا۔ امام ترمذی رحمۃ اللہ علیہ کی روایت کے مطابق جب یہ حدیث حضرت امیر معاویہ رضی اللہ عنہ کو سنائی گئی تو وہ بہت روئے اور پھر یہی آیات تلاوت کیں۔

اہلِ معرفت کہتے ہیں کہ ان آیات کا محور لفظ يُرِيْدُ ہے، یعنی ارادہ۔ عمل کی صورت ایک ہو سکتی ہے اور اس کی حقیقت دو؛ دو آدمی ایک جیسی نماز پڑھتے ہیں اور ایک کی نماز آسمان تک جاتی ہے، دوسرے کی لوگوں کی نظروں تک۔ حدیث کا مفہوم ہے کہ اعمال کا دار و مدار نیتوں پر ہے؛ اور شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ عمل اپنے مقصود کی صورت اختیار کر لیتا ہے: جس کا مقصود دنیا ہو اس کا عمل دنیا کی صورت بن جاتا ہے اور دنیا کے ساتھ ختم ہو جاتا ہے، اور جس کا مقصود حق ہو اس کا عمل باقی رہتا ہے کیونکہ اس کا مقصود باقی ہے۔ لفظ حَبِطَ کی اصل اس جانور کی حالت ہے جو بری گھاس کھا کر پھول جائے اور پھر ہلاک ہو جائے؛ دکھاوے کا عمل بھی باہر سے پھولا ہوا اور اندر سے کھوکھلا ہوتا ہے۔ اور زِيْنَتَهَا یاد دلاتا ہے کہ دنیا کو سجایا ہی اس لیے گیا کہ آزمائش ہو؛ پچھلے رکوع کی آیت میں آسمان و زمین کی تخلیق کا مقصد بہترین عمل کی آزمائش بتایا گیا تھا، اور یہاں بتایا کہ جو زینت پر رک گیا وہ اس امتحان میں ناکام رہا۔ یہ اشاری معنی ظاہر کی نفی نہیں: ظاہر میں یہ حکم کافروں اور دکھاوا کرنے والوں کا ہے؛ اور باطن میں ہر سالک کے لیے تنبیہ ہے کہ وہ اپنی نیتوں کو پرکھتا رہے۔

اب اس کے مقابل وہ شخص آتا ہے جس کا ارادہ صاف ہے: اَفَمَنْ كَانَ عَلٰى بَيِّنَةٍ مِّنْ رَّبِّهٖ وَيَتْلُوْهُ شَاهِدٌ مِّنْهُ۔ “تو کیا وہ شخص جو اپنے رب کی طرف سے ایک روشن دلیل پر ہو، اور اس کے پیچھے اس کی طرف سے ایک گواہ بھی آ رہا ہو [دنیا چاہنے والے کی طرح ہو سکتا ہے]؟” جواب لفظوں میں نہیں دیا گیا، کیونکہ وہ خود ظاہر ہے۔ مفسرین کے نزدیک اس سے مراد حضورِ اکرم صلی اللہ علیہ وسلم ہیں، یا آپ صلی اللہ علیہ وسلم اور آپ کے ساتھ ایمان والے۔ بَيِّنَةٍ کے بارے میں کہا گیا کہ یہ قرآن ہے، یا وہ فطرت اور عقل کی روشنی جو حق کو پہچانتی ہے۔ يَتْلُوْهُ کے دو معنی ہیں: اس کی تلاوت کرتا ہے، یا اس کے پیچھے آتا ہے؛ اور شَاهِدٌ مِّنْهُ کے بارے میں اقوال ہیں کہ اس سے مراد حضرت جبریل علیہ السلام ہیں جو قرآن پڑھ کر سناتے تھے، یا خود قرآن جو حق کی گواہی دیتا ہے، یا حضورِ اکرم صلی اللہ علیہ وسلم کی زبانِ مبارک جو اس کی تلاوت کرتی تھی۔ ان سب اقوال میں تضاد نہیں؛ یہ ایک ہی حقیقت کے مختلف رخ ہیں: باطن کی روشنی، اور اس کے ساتھ باہر سے آنے والی گواہی جو اس روشنی کی تصدیق کرتی ہے۔

شیخِ اکبر کے نزدیک یہ آیت معرفت کے صحیح راستے کا نقشہ ہے۔ بَيِّنَةٍ مِّنْ رَّبِّهٖ وہ اندرونی وضوح ہے جو دل کو ربِّ خاص کی طرف سے ملتا ہے؛ ہر بندے کا اپنے رب سے ایک خاص رشتہ ہے جس سے اس کے دل میں روشنی اترتی ہے، اور اسی کو اس مکتب میں کشف اور فتح کہا جاتا ہے۔ مگر آیت فوراً کہتی ہے کہ اس روشنی کے پیچھے ایک گواہ ہے، اور وہ گواہ مِّنْهُ ہے، یعنی اسی کی طرف سے، بندے کے نفس کی طرف سے نہیں۔ شیخِ اکبر بار بار اس اصول پر زور دیتے ہیں، جس کا مفہوم یہ ہے کہ جو کشف شریعت کے خلاف ہو وہ رد ہے، اور کشف کی صحت کا معیار کتاب و سنت ہے؛ ولی کا ذوق نبی کی لائی ہوئی شریعت کے تابع ہے۔ یوں بینہ اور شاہد ایک دوسرے کی حفاظت کرتے ہیں: باطن کی روشنی کے بغیر ظاہر کی گواہی رسم بن جاتی ہے، اور ظاہر کی گواہی کے بغیر باطن کی روشنی نفس کا دھوکا ہو سکتی ہے۔ پھر تیسرا گواہ: وَمِنْ قَبْلِهٖ كِتٰبُ مُوْسٰۤى اِمَامًا وَّرَحْمَةً ؕ اُولٰۤئِكَ يُؤْمِنُوْنَ بِهٖ۔ “اور اس سے پہلے موسیٰ علیہ السلام کی کتاب تھی جو رہنما اور رحمت تھی؛ یہی لوگ اس پر ایمان لاتے ہیں۔” حضرت موسیٰ علیہ السلام کی کتاب کا ذکر یہاں اس لیے ہے کہ وحی کا سلسلہ ایک ہے، اور پچھلی کتاب نے اس کتاب کی آمد کی خبر دی تھی۔ اہلِ ذوق کہتے ہیں کہ سالک کے پاس بھی تین گواہ ہیں: اپنے دل کی روشنی، وحی کی گواہی، اور اہلِ حق کی وہ روایت جو اس سے پہلے چلی آ رہی ہے؛ جب تینوں ایک بات کہیں تو قدم یقین پر پڑتا ہے۔

آیت کا آخری حصہ: وَمَنْ يَّكْفُرْ بِهٖ مِنَ الْاَحْزَابِ فَالنَّارُ مَوْعِدُهٗ ۚ فَلَا تَكُ فِيْ مِرْيَةٍ مِّنْهُ ۗ اِنَّهُ الْحَقُّ مِنْ رَّبِّكَ وَلٰكِنَّ اَكْثَرَ النَّاسِ لَا يُؤْمِنُوْنَ۔ “اور ان گروہوں میں سے جو اس کا انکار کرے تو آگ اس کے وعدے کی جگہ ہے؛ پس آپ صلی اللہ علیہ وسلم اس کے بارے میں کسی شک میں نہ رہیں، یہ آپ کے رب کی طرف سے حق ہے، لیکن اکثر لوگ ایمان نہیں لاتے۔” یہ خطاب ظاہر میں حضورِ اکرم صلی اللہ علیہ وسلم سے ہے، اور مفسرین متفق ہیں کہ آپ صلی اللہ علیہ وسلم کو کبھی کوئی شک نہ تھا؛ مقصود امت ہے، جیسے بادشاہ وزیر سے کہے اور سنانا رعایا کو ہو۔ اور آخری جملہ ایک بڑا اصول دیتا ہے: حق کی پہچان اکثریت سے نہیں ہوتی۔ اہلِ معرفت کہتے ہیں کہ جو سالک لوگوں کی گنتی دیکھ کر حق کو پرکھتا ہے، وہ بینہ سے محروم ہے؛ بینہ والا تنہا ہو تب بھی روشنی میں ہے۔ مِرْيَةٍ وہ شک ہے جو بار بار جھگڑنے سے پیدا ہوتا ہے؛ گویا فرمایا گیا کہ منکروں کے جھگڑے کو اپنے دل میں راہ نہ دیجیے۔ شیخِ اکبر کے مکتب میں یقین کے درجے بیان ہوتے ہیں: علم الیقین، عین الیقین اور حق الیقین؛ اور بینہ اور شاہد کا جمع ہونا علم کو عین کی طرف لے جاتا ہے، جہاں شک کی گنجائش نہیں رہتی۔

اب ظلم کی انتہا کا ذکر ہے: وَمَنْ اَظْلَمُ مِمَّنِ افْتَرٰى عَلَى اللّٰهِ كَذِبًا ؕ اُولٰۤئِكَ يُعْرَضُوْنَ عَلٰى رَبِّهِمْ وَيَقُوْلُ الْاَشْهَادُ هٰۤؤُلَاۤءِ الَّذِيْنَ كَذَبُوْا عَلٰى رَبِّهِمْ ۚ اَلَا لَعْنَةُ اللّٰهِ عَلَى الظّٰلِمِيْنَ۔ “اور اس سے بڑھ کر ظالم کون ہے جو اللہ پر جھوٹ گھڑے؟ یہ لوگ اپنے رب کے سامنے پیش کیے جائیں گے، اور گواہ کہیں گے: یہی ہیں وہ جنہوں نے اپنے رب پر جھوٹ باندھا تھا؛ سنو! ظالموں پر اللہ کی لعنت ہے۔” اللہ پر جھوٹ باندھنا سب سے بڑا ظلم اس لیے ہے کہ یہ حقیقت کے سرچشمے کو گدلا کرنا ہے؛ جو بندوں پر جھوٹ بولتا ہے وہ ایک تعلق خراب کرتا ہے، اور جو اللہ پر جھوٹ بولتا ہے وہ ہر تعلق کی بنیاد خراب کرتا ہے۔ الْاَشْهَادُ کے بارے میں مفسرین کہتے ہیں کہ اس سے مراد فرشتے، انبیا علیہم السلام اور اہلِ ایمان ہیں۔ صحیح بخاری میں حضرت عبداللہ بن عمر رضی اللہ عنہما کی روایت کا مفہوم یہ ہے کہ قیامت کے دن اللہ مومن کو اپنے قریب کرے گا، اس پر اپنا پردہ ڈالے گا اور اس سے اس کے گناہوں کا اقرار کرائے گا، پھر فرمائے گا کہ میں نے دنیا میں ان پر پردہ ڈالا اور آج تمہیں بخشتا ہوں؛ اور کافروں اور منافقوں کو سب کے سامنے پکارا جائے گا، اور اسی موقع پر یہی آیت پڑھی گئی۔ اہلِ معرفت کہتے ہیں کہ یہاں غفر بمعنی ستر کا راز پورا کھلتا ہے: مومن کو پردے میں حساب ملتا ہے اور منکر کو گواہوں کے سامنے؛ جس نے دنیا میں اپنے رب سے پردہ داری کی دعا مانگی، اسے آخرت میں پردہ ملا۔

پھر ان ظالموں کی پہچان: الَّذِيْنَ يَصُدُّوْنَ عَنْ سَبِيْلِ اللّٰهِ وَيَبْغُوْنَهَا عِوَجًا ؕ وَهُمْ بِالْاٰخِرَةِ هُمْ كٰفِرُوْنَ۔ “جو اللہ کی راہ سے روکتے ہیں اور اس میں ٹیڑھ تلاش کرتے ہیں، اور وہی آخرت کے منکر ہیں۔” اُولٰۤئِكَ لَمْ يَكُوْنُوْا مُعْجِزِيْنَ فِي الْاَرْضِ وَمَا كَانَ لَهُمْ مِّنْ دُوْنِ اللّٰهِ مِنْ اَوْلِيَآءَ۔ “یہ لوگ زمین میں [اللہ کو] عاجز کرنے والے نہ تھے، اور اللہ کے سوا ان کا کوئی حمایتی نہ تھا۔” يَبْغُوْنَهَا عِوَجًا کا مطلب ہے کہ سیدھی راہ میں ٹیڑھ ڈھونڈتے ہیں، یعنی اس پر اعتراض گھڑتے اور اسے ٹیڑھا دکھاتے ہیں؛ اور اس سے یہ اشارہ بھی نکلتا ہے کہ وہ ایسی راہ چاہتے ہیں جو ان کی خواہشوں کے مطابق مڑ جائے۔ اہلِ معرفت کہتے ہیں کہ نفس سیدھی راہ کو اس لیے ٹیڑھا کہتا ہے کہ وہ خود ٹیڑھا ہے؛ ٹیڑھے آئینے میں سیدھی لکیر بھی ٹیڑھی دکھائی دیتی ہے۔ یہاں ترتیب پر غور لازم ہے، کیونکہ اگلی بات کا فہم اسی پر ہے: پہلے ان کا اپنا فعل بیان ہوا، روکنا، ٹیڑھ ڈھونڈنا، آخرت کا انکار؛ اور پھر اس کا انجام۔ مُعْجِزِيْنَ کا مطلب یہ ہے کہ وہ کہیں بھاگ کر اللہ کی پکڑ سے باہر نہیں نکل سکتے تھے، اور جن کو انہوں نے حمایتی سمجھ رکھا تھا وہ اس دن کسی کام کے نہیں۔

اب وہ جملہ جو اس رکوع میں حاکم اصول کا مقام ہے: يُضٰعَفُ لَهُمُ الْعَذَابُ ؕ مَا كَانُوْا يَسْتَطِيْعُوْنَ السَّمْعَ وَمَا كَانُوْا يُبْصِرُوْنَ۔ “ان کے لیے عذاب دگنا کیا جائے گا؛ وہ نہ سن سکتے تھے اور نہ دیکھتے تھے۔” عذاب دگنا اس لیے کہ وہ خود بھی بھٹکے اور دوسروں کو بھی روکا۔ اور “سن نہیں سکتے تھے” کا مطلب یہ ہرگز نہیں کہ اللہ نے زبردستی ان کے کان بند کر دیے تھے اور وہ سننا چاہتے تھے مگر روک دیے گئے۔ مفسرین کہتے ہیں کہ حق سے ان کی نفرت اس حد کو پہنچ گئی تھی کہ وہ اسے سننا برداشت ہی نہ کر سکتے تھے، جیسے کوئی کہے کہ میں اس کی بات سن ہی نہیں سکتا؛ کان سلامت تھے، دل نے سننے سے انکار کر دیا تھا۔ اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت کا طالب ہو اسے کبھی گمراہ نہیں کرتا؛ جو ڈھونڈتا ہے وہ پاتا ہے، اور جو منہ پھیرتا ہے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے۔ پچھلی آیت نے ان کا اعراض بتا دیا تھا: راہ سے روکنا اور اس میں ٹیڑھ ڈھونڈنا؛ اور یہ آیت اس اعراض کا پھل بتاتی ہے: جس نے بار بار سننے سے منہ پھیرا اس سے سننے کی سکت جاتی رہی، اور جس نے بار بار آنکھ بند کی وہ دیکھنے کا عادی نہ رہا۔ یہی اصول اسی سورت میں آگے صاف لفظوں میں آتا ہے: وَمَا ظَلَمْنٰهُمْ وَلٰكِنْ ظَلَمُوْۤا اَنْفُسَهُمْ؛ “اور ہم نے ان پر ظلم نہیں کیا بلکہ انہوں نے خود اپنی جانوں پر ظلم کیا۔”

شیخِ اکبر کے نزدیک یہ سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے۔ حق بندے کو وہی دیتا ہے جو بندے کی اپنی حقیقت اور اس کا اپنا رخ طلب کرتا ہے؛ جس نے سننے سے انکار کو اپنا رخ بنایا، حق نے اس کے اسی رخ کو وجود دیا، اور یہ عدل ہے، جبر نہیں۔ اسی لیے قرآن میں قومِ ثمود کے بارے میں فرمایا: فَاسْتَحَبُّوا الْعَمٰى عَلَى الْهُدٰى، یعنی انہوں نے اندھے پن کو ہدایت پر ترجیح دی؛ اندھا پن ان کا اپنا پسند کیا ہوا تھا۔ اور اگر وہ چاہتا تو سب کو زبردستی ہدایت پر لے آتا، جیسا کہ فرمایا: فَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ؛ مگر اس نے انسان کو اختیار کی عزت دی اور جبر کو اختیار نہ کیا۔ اختیار کی یہ عزت ہی اس امتحان کی جان ہے جس کا ذکر پچھلے رکوع میں لِيَبْلُوَكُمْ کے لفظ سے ہوا تھا؛ جہاں زبردستی ہو وہاں امتحان نہیں رہتا۔ اہلِ معرفت کہتے ہیں کہ سننے کی سکت ایک امانت ہے جو استعمال سے بڑھتی اور ترک سے گھٹتی ہے؛ جو ایک بار حق سن کر منہ موڑ لے، اگلی بار اس کے لیے سننا اور بھاری ہو جاتا ہے، یہاں تک کہ وہ وقت آتا ہے جب وہ سن ہی نہیں سکتا۔ سالک کے لیے اس میں ڈر کا مقام ہے: نصیحت کو ٹالنے کی عادت دل کی سماعت کو کم کرتی ہے، اور نصیحت قبول کرنے کی عادت اسے بڑھاتی ہے۔

پھر خسارے کی انتہا: اُولٰۤئِكَ الَّذِيْنَ خَسِرُوْۤا اَنْفُسَهُمْ وَضَلَّ عَنْهُمْ مَّا كَانُوْا يَفْتَرُوْنَ۔ “یہی ہیں جنہوں نے اپنی جانوں کو خسارے میں ڈالا، اور جو کچھ وہ گھڑتے تھے وہ ان سے گم ہو گیا۔” لَا جَرَمَ اَنَّهُمْ فِي الْاٰخِرَةِ هُمُ الْاَخْسَرُوْنَ۔ “یقیناً یہی لوگ آخرت میں سب سے زیادہ خسارے والے ہیں۔” تاجر مال کھوتا ہے تو کچھ کھوتا ہے، مگر جو خود کو کھو دے اس نے سب کچھ کھو دیا، کیونکہ اب کوئی بچا ہی نہیں جو دوبارہ کمائے۔ اور خَسِرُوْۤا اَنْفُسَهُمْ کا فاعل وہ خود ہیں؛ کسی نے ان سے ان کی جان نہیں چھینی، انہوں نے اسے خود ہارا۔ پھر وَضَلَّ عَنْهُمْ: جن معبودوں اور سہاروں کو انہوں نے گھڑ رکھا تھا، وہ گم ہو گئے، جیسے دھوپ نکلنے پر دھند۔ شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ جو کچھ حق کے سوا مستقل ہستی کا دعویٰ کرے وہ خیال کی طرح ہے، جس کی اپنی کوئی بنیاد نہیں؛ اس دن ہر وہ خیال جو حق کے مقابل کھڑا کیا گیا تھا، پگھل جائے گا، اور جس نے اس پر بھروسا کیا تھا وہ خالی ہاتھ رہ جائے گا۔ لَا جَرَمَ کا مطلب ہے کہ اس میں کوئی شک نہیں، یہ لازمی نتیجہ ہے؛ اور الْاَخْسَرُوْنَ اسمِ تفضیل ہے: ہر خسارے والے سے بڑھ کر خسارے والے۔

اب اس رکوع کی روشن آیت: اِنَّ الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَاَخْبَتُوْۤا اِلٰى رَبِّهِمْ ۙ اُولٰۤئِكَ اَصْحٰبُ الْجَنَّةِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ۔ “بے شک جو لوگ ایمان لائے، نیک عمل کیے اور اپنے رب کے آگے عاجزی سے جھک گئے، وہی جنت والے ہیں، وہ اس میں ہمیشہ رہیں گے۔” اَخْبَتُوْۤا کی اصل خَبْت ہے، یعنی نرم، ہموار اور نشیبی زمین۔ اخبات کا مطلب ہے دل کا اپنے رب کے سامنے اس طرح پست اور نرم ہو جانا جیسے نشیبی زمین، اور ساتھ ہی اس پر مطمئن ہو کر ٹک جانا۔ مفسرین نے اس کے معنی عاجزی، خشوع، اطمینان اور رجوع بیان کیے ہیں، اور اِلٰى کا حرف بتاتا ہے کہ یہ عاجزی رخ رکھتی ہے: وہ اپنے رب کی طرف جھکے اور اسی پر ٹھہر گئے۔ آیت کی ترتیب تین منزلیں دکھاتی ہے: ایمان دل کا اقرار ہے، عملِ صالح اعضا کی گواہی، اور اخبات وہ کیفیت جو ان دونوں پر تاج کی طرح رکھی جاتی ہے۔ ایمان اور عمل کے بعد بھی اگر دل اکڑا رہے تو عمل کا نور اندر نہیں اترتا؛ اخبات وہ نرمی ہے جو عمل کو دل میں جذب کر دیتی ہے۔

اخبات کی یہ تصویر، نشیبی زمین، اہلِ معرفت کے ہاں بڑی گہری ہے۔ بارش اونچے ٹیلوں پر بھی برستی ہے اور نشیبی زمین پر بھی، مگر ٹیلا پانی کو بہا دیتا ہے اور نشیب اسے جمع کر لیتا ہے؛ پھر وہیں سبزہ اگتا ہے۔ اہلِ ذوق کہتے ہیں کہ تجلی کی بارش سب پر برستی ہے، اور اسے وہی دل روکتا ہے جو نیچا ہو۔ پچھلے رکوع میں عرش کو پانی پر بتایا گیا تھا اور زمین پر چلنے والے ہر جاندار کو اللہ کے رزق کا محتاج؛ اب بتایا کہ اس پانی اور اس رزق کو وہی سمیٹتا ہے جو زمین کی طرح جھک جائے۔ سورۂ الحج میں فرمایا: وَبَشِّرِ الْمُخْبِتِيْنَ، اور پھر ان کی صفات بیان ہوئیں، جن کا مفہوم یہ ہے کہ اللہ کا ذکر ہو تو ان کے دل لرز جاتے ہیں، مصیبت پر صابر رہتے ہیں، نماز قائم کرتے ہیں اور دیے ہوئے میں سے خرچ کرتے ہیں۔ یوں اخبات صرف دل کی کیفیت نہیں، اس سے صبر، نماز اور سخاوت پھوٹتی ہے۔ اور اس رکوع کے آغاز سے موازنہ کیجیے: وہاں فَرِحٌ فَخُوْرٌ تھا جو نعمت پا کر اونچا ہو جاتا ہے، اور یہاں مُخْبِت ہے جو نعمت پا کر اور نیچا ہو جاتا ہے؛ پہلا ٹیلا ہے جس پر کچھ نہیں ٹھہرتا، دوسرا وادی ہے جس میں سب کچھ ٹھہرتا ہے۔

شیخِ اکبر کے مکتب میں عاجزی کی کئی صورتیں الگ الگ پہچانی جاتی ہیں: خشوع جو اعضا اور آواز پر ظاہر ہوتا ہے، تواضع جو بندوں کے ساتھ برتاؤ میں ہوتی ہے، اور اخبات جو رب کی طرف دل کا ٹھہرا ہوا جھکاؤ ہے۔ اخبات میں اطمینان کا پہلو بھی ہے، اس لیے یہ لمحاتی حال نہیں بلکہ ٹھہرا ہوا مقام ہے، اور اسے تمکین کے قریب سمجھا جاتا ہے۔ اہلِ معرفت کہتے ہیں کہ جو بندہ اپنے آپ کو حق کے سامنے جتنا نیچا کرتا ہے، اس کی اپنی ہستی کا دعویٰ اتنا ہی مٹتا ہے؛ یہ فنا کا دروازہ ہے، اور اس فنا کے بعد جو بقا ملتی ہے وہ حق کے ساتھ بقا ہے۔ اسی لیے جزا میں خٰلِدُوْنَ آیا: جس نے اپنی خودی کو جھکا دیا، اسے وہ گھر ملا جس میں فنا نہیں۔ اس آیت کو پچھلے رکوع کی ایک تصویر کے مقابل بھی رکھیے: وہاں کچھ لوگ يَثْنُوْنَ صُدُوْرَهُمْ تھے، یعنی رب سے چھپنے کے لیے سینے موڑتے تھے؛ اور یہاں مُخْبِت ہیں جو اپنا سینہ رب کے سامنے کھول کر جھک جاتے ہیں۔ ایک چھپتا ہے اور دوسرا اپنے آپ کو پیش کر دیتا ہے؛ ایک کا انجام رسوائی ہے اور دوسرے کا دائمی گھر۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر میں ایمان، عمل اور اخبات جنت کا راستہ ہیں، اور باطن میں یہی تینوں دل کی جنت کا دروازہ۔

رکوع ایک تمثیل پر ختم ہوتا ہے: مَثَلُ الْفَرِيْقَيْنِ كَالْاَعْمٰى وَالْاَصَمِّ وَالْبَصِيْرِ وَالسَّمِيْعِ ؕ هَلْ يَسْتَوِيٰنِ مَثَلًا ؕ اَفَلَا تَذَكَّرُوْنَ۔ “دونوں گروہوں کی مثال ایسی ہے جیسے ایک اندھا اور بہرا ہو اور ایک دیکھنے اور سننے والا؛ کیا یہ دونوں مثال میں برابر ہو سکتے ہیں؟ تو کیا تم نصیحت حاصل نہیں کرتے؟” مفسرین کہتے ہیں کہ یہاں دو ہی آدمی ہیں: ایک جس میں اندھا پن اور بہرا پن دونوں جمع ہیں، اور دوسرا جس میں بینائی اور سماعت دونوں؛ اسی لیے آگے يَسْتَوِيٰنِ تثنیہ کا صیغہ آیا۔ یہ تمثیل بیسویں آیت سے جڑی ہے: جو سن نہیں سکتے تھے اور دیکھتے نہیں تھے، وہی اس مثال کا پہلا آدمی ہیں؛ اور یہ اندھا پن وہی ہے جو انہوں نے اپنے اعراض سے خود چنا۔ دوسری طرف وہ ہیں جن کا ذکر بینہ اور شاہد کے ساتھ ہوا، اور جو اخبات والے ہیں۔ حدیثِ قربِ نوافل کا مفہوم یہ ہے کہ بندہ نوافل کے ذریعے قریب ہوتا جاتا ہے یہاں تک کہ اللہ فرماتا ہے: میں اس کا کان بن جاتا ہوں جس سے وہ سنتا ہے اور اس کی آنکھ جس سے وہ دیکھتا ہے۔ شیخِ اکبر کے نزدیک اس کا مطلب حلول یا اتحاد ہرگز نہیں، حق ان سے پاک ہے؛ مطلب یہ ہے کہ بندے کی سماعت اور بینائی حق کی تائید اور نور سے دیکھنے سننے لگتی ہیں، اور اسے ہر شے میں حق کی نشانیاں نظر آتی ہیں۔ آیت کا الْبَصِيْرِ وَالسَّمِيْعِ یہی ہے: وہ بندہ جس کے حواس اس کے رب کی طرف کھل گئے۔ اور اَفَلَا تَذَكَّرُوْنَ آخری دستک ہے: تذکر یاد کرنا ہے، گویا یہ فرق انسان کی فطرت میں پہلے سے لکھا ہوا ہے، اسے صرف یاد کرنا ہے؛ اور یاد کرنے کا دروازہ آخری لمحے تک کھلا ہے۔
