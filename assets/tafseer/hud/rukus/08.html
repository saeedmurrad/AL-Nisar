حضرت لوط علیہ السلام کی بستی کے بعد سلسلۂ رسالت مدین پہنچتا ہے: وَاِلٰى مَدْيَنَ اَخَاهُمْ شُعَيْبًا ۗ قَالَ يٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَيْرُهٗ ۗ وَلَا تَنْقُصُوا الْمِكْيَالَ وَالْمِيْزَانَ اِنِّيْۤ اَرٰىكُمْ بِخَيْرٍ وَّاِنِّيْۤ اَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ مُّحِيْطٍ۔ “اور مدین کی طرف ہم نے ان کے بھائی شعیب علیہ السلام کو بھیجا۔ انہوں نے کہا: اے میری قوم! اللہ کی عبادت کرو، اس کے سوا تمہارا کوئی معبود نہیں؛ اور ناپ اور تول میں کمی نہ کرو۔ میں تمہیں خوشحال دیکھ رہا ہوں، اور مجھے تم پر ایک گھیر لینے والے دن کے عذاب کا ڈر ہے۔” حضرت نوح، حضرت ہود اور حضرت صالح علیہم السلام کی دعوت کی طرح یہاں بھی پہلا کلمہ توحید ہے؛ مگر دوسرا کلمہ اس قوم کی مخصوص بیماری کی نشاندہی کرتا ہے: ناپ تول میں کمی۔ مفسرین بیان کرتے ہیں کہ مدین تجارتی راستوں پر آباد تھا، اور اس قوم کا فساد بازار میں تھا۔ یہ ترتیب بتاتی ہے کہ توحید عقیدے کے حجرے میں بند نہیں رہتی، ترازو تک پہنچتی ہے۔ اہلِ معرفت کہتے ہیں کہ جو ناپ تول میں کمی کرتا ہے اس نے دل میں حرص کو ایک چھوٹا سا معبود بنا رکھا ہے، اس لیے توحید کی دعوت اور ترازو کی اصلاح ایک ہی دعوت کے دو رخ ہیں۔ اَرٰىكُمْ بِخَيْرٍ میں ایک لطیف دلیل ہے: تم خوشحال ہو، تمہیں کمی کرنے کی کوئی مجبوری نہیں؛ نعمت کی حالت میں خیانت ناشکری کی انتہا ہے۔ اور اَخَاهُمْ کا لفظ، جو پچھلے قصوں میں بھی آیا، بتاتا ہے کہ نبی اپنی قوم میں اجنبی بن کر نہیں، بھائی بن کر آتا ہے؛ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہ اس اصول کا مظہر ہے کہ حق کا پیغام ہمیشہ قبول کرنے والے کی زبان اور اس کے مانوس پیکر میں آتا ہے، تاکہ کسی کے پاس اجنبیت کا عذر نہ رہے۔

پھر وہی بات مثبت پیرائے میں دہرائی گئی: وَيٰقَوْمِ اَوْفُوا الْمِكْيَالَ وَالْمِيْزَانَ بِالْقِسْطِ وَلَا تَبْخَسُوا النَّاسَ اَشْيَآءَهُمْ وَلَا تَعْثَوْا فِي الْاَرْضِ مُفْسِدِيْنَ۔ “اور اے میری قوم! ناپ اور تول انصاف کے ساتھ پورا کرو، لوگوں کو ان کی چیزیں گھٹا کر نہ دو، اور زمین میں فساد پھیلاتے نہ پھرو۔” پہلے فرمایا تھا کہ کمی نہ کرو، اب فرمایا کہ پورا دو؛ منع اور حکم دونوں ایک ساتھ، تاکہ کوئی درمیانی گنجائش نہ رہے۔ تَبْخَسُوا کے بارے میں مفسرین کہتے ہیں کہ یہ صرف ناپ تول تک محدود نہیں: لوگوں کی ہر چیز کو اس کی قدر سے کم کر کے دینا یا دیکھنا، ان کی محنت، ان کے حق اور ان کی خوبی کو گھٹانا، سب اسی میں داخل ہے۔ سورۂ المطففین میں ہم پڑھ چکے: وَيْلٌ لِّلْمُطَفِّفِيْنَ؛ اور یہاں اسی مرض میں مبتلا ایک تاریخی قوم دکھائی گئی ہے۔ پھر حضرت شعیب علیہ السلام نے وہ جملہ فرمایا جو کمائی کا پورا زاویہ بدل دیتا ہے: بَقِيَّتُ اللّٰهِ خَيْرٌ لَّكُمْ اِنْ كُنْتُمْ مُّؤْمِنِيْنَ ۚ وَمَاۤ اَنَا عَلَيْكُمْ بِحَفِيْظٍ۔ “اللہ کا بچایا ہوا تمہارے لیے بہتر ہے اگر تم ایمان والے ہو، اور میں تم پر نگہبان نہیں ہوں۔” مفسرین کہتے ہیں کہ بَقِيَّتُ اللّٰهِ سے مراد وہ حلال نفع ہے جو پورا ناپنے تولنے کے بعد اللہ کی اجازت سے بچ رہے؛ وہ تھوڑا ہو تب بھی اس مال سے بہتر ہے جو کمی کر کے کمایا جائے۔ بعض نے اس سے اللہ کی اطاعت اور اس کا باقی رہنے والا اجر بھی مراد لیا۔ اور وَمَاۤ اَنَا عَلَيْكُمْ بِحَفِيْظٍ میں نبی کا ادب ہے: میں تمہارا داروغہ نہیں، تم پر زبردستی نہیں کر سکتا؛ ہدایت تمہارا اپنا انتخاب ہے۔

اہلِ معرفت بَقِيَّتُ اللّٰهِ میں فنا اور بقا کا پورا سفر دیکھتے ہیں، اس شرط کے ساتھ کہ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر میں حلال نفع کی برتری ہی مقصود ہے۔ حرص بڑھانے کا نام ہے اور بقیہ بچ رہنے کا۔ جو شخص لوگوں کا حق کاٹ کر اپنا ڈھیر بڑھاتا ہے وہ فانی کو جمع کر رہا ہے؛ اور جو پورا دے کر جو بچے اس پر راضی ہو جاتا ہے وہ باقی سے جڑ رہا ہے۔ سورۂ النحل کا جملہ یہاں روشنی ڈالتا ہے: مَا عِنْدَكُمْ يَنْفَدُ وَمَا عِنْدَ اللّٰهِ بَاقٍ۔ شیخِ اکبر کے نزدیک سالک کا سفر فنا سے بقا تک ہے: پہلے نفس کی وہ صفات فنا ہوں جو اپنے آپ کو مالک سمجھتی ہیں، پھر جو بچے وہ اللہ کا بقیہ ہے، یعنی بندے میں وہ جو حق کے ساتھ باقی ہے۔ اس مکتب میں ترازو بھی ایک بڑی علامت ہے: کائنات کا ہر وجود ایک میزان پر قائم ہے، ہر اسم اپنی مقدار کے ساتھ ظاہر ہوتا ہے، اور عدل اسی میزان کی پاسداری ہے۔ جو بازار میں ترازو جھکاتا ہے وہ اپنے حصے کے دائرے میں اس میزان کو توڑتا ہے؛ اور جو ترازو سیدھا رکھتا ہے وہ اپنے چھوٹے سے دائرے میں اسمِ العدل کا مظہر بنتا ہے۔ وَمَاۤ اَنَا عَلَيْكُمْ بِحَفِيْظٍ اسی کا تتمہ ہے: نبی ترازو دکھاتا ہے، ہاتھ پکڑ کر تلواتا نہیں۔

قوم کا جواب طنز میں ڈوبا ہوا تھا، مگر اس طنز میں بھی ایک سچائی چھپی تھی: قَالُوْا يٰشُعَيْبُ اَصَلٰوتُكَ تَاْمُرُكَ اَنْ نَّتْرُكَ مَا يَعْبُدُ اٰبَآؤُنَاۤ اَوْ اَنْ نَّفْعَلَ فِيْۤ اَمْوَالِنَا مَا نَشٰٓؤُا ۗ اِنَّكَ لَاَنْتَ الْحَلِيْمُ الرَّشِيْدُ۔ “وہ بولے: اے شعیب (علیہ السلام)! کیا تمہاری نماز تمہیں یہ حکم دیتی ہے کہ ہم ان کو چھوڑ دیں جنہیں ہمارے باپ دادا پوجتے تھے، یا اپنے مالوں میں جیسا چاہیں ویسا نہ کریں؟ تم تو بڑے بردبار اور سمجھ دار ہو!” مفسرین لکھتے ہیں کہ حضرت شعیب علیہ السلام کثرت سے نماز پڑھتے تھے، اس لیے قوم نے نماز ہی کو طنز کا نشانہ بنایا۔ مگر انجانے میں انہوں نے ایک حقیقت کہہ دی: ہاں، سچی نماز واقعی حکم دیتی ہے؛ قرآن نے دوسری جگہ بتایا کہ وہ بے حیائی اور برائی سے روکتی ہے۔ ان کی اصل غلطی دوسرے جملے میں ہے: فِيْۤ اَمْوَالِنَا مَا نَشٰٓؤُا۔ یہ وہ دعویٰ ہے جو دین کو عبادت گاہ میں بند کر کے بازار کو اس سے آزاد رکھنا چاہتا ہے۔ اہلِ معرفت صَلٰوة کو صِلَة یعنی جوڑ سے مناسبت دیتے ہیں؛ جو نماز بندے کو حق سے واقعی جوڑ دے وہ اس کے مال، اس کے ترازو اور اس کے معاملات تک ضرور پہنچتی ہے، کیونکہ جس کا تعلق مالک سے جڑ جائے وہ اپنے آپ کو مال کا مالک نہیں، امین سمجھنے لگتا ہے۔ مَا نَشٰٓؤُا کا دعویٰ مشیت کا وہ دعویٰ ہے جو اصل مالک کو بھلا دیتا ہے۔ اور اِنَّكَ لَاَنْتَ الْحَلِيْمُ الرَّشِيْدُ کے بارے میں مفسرین کہتے ہیں کہ یہ طنز تھا، یا یہ ان کا اعتراف تھا کہ تمہارا حلم اور رشد تو ہم سب جانتے ہیں، پھر یہ باتیں کیوں؟ دونوں صورتوں میں مخالف کی زبان سے نبی کی صفت کی گواہی نکلی۔

اب وہ آیت آتی ہے جو اس رکوع کا دل ہے، اور جسے ہم چند حصوں میں ٹھہر ٹھہر کر پڑھیں گے: قَالَ يٰقَوْمِ اَرَءَيْتُمْ اِنْ كُنْتُ عَلٰى بَيِّنَةٍ مِّنْ رَّبِّيْ وَرَزَقَنِيْ مِنْهُ رِزْقًا حَسَنًا ۗ وَمَاۤ اُرِيْدُ اَنْ اُخَالِفَكُمْ اِلٰى مَاۤ اَنْهٰىكُمْ عَنْهُ۔ “شعیب علیہ السلام نے کہا: اے میری قوم! بتاؤ تو، اگر میں اپنے رب کی طرف سے ایک روشن دلیل پر ہوں اور اس نے مجھے اپنی طرف سے اچھا رزق دیا ہو (تو کیا میں تمہاری خاطر حق چھوڑ دوں؟)؛ اور میں یہ نہیں چاہتا کہ جس بات سے تمہیں روکتا ہوں خود اس کے خلاف کروں۔” جواب میں نہ طنز ہے نہ غصہ؛ ایک سوال ہے جو سننے والے کو سوچنے کی دعوت دیتا ہے۔ بَيِّنَةٍ مِّنْ رَّبِّيْ وہ اندرونی یقین اور نبوت کی روشنی ہے جس پر داعی کھڑا ہوتا ہے؛ اور رِزْقًا حَسَنًا کے بارے میں مفسرین کہتے ہیں کہ اس سے مراد نبوت اور علم بھی ہے اور حلال مال بھی، جو حضرت شعیب علیہ السلام کو قوم کے مال اور ان کی کم تول سے بے نیاز رکھتا تھا۔ داعی کی پہلی شرط یہ ہے کہ وہ جس بات کی دعوت دے اس پر خود کھڑا ہو، اور دوسری یہ کہ وہ لوگوں کے مال کا محتاج نہ ہو۔ پھر وَمَاۤ اُرِيْدُ اَنْ اُخَالِفَكُمْ: قول اور عمل کی یکسانی۔ اہلِ معرفت کہتے ہیں کہ نصیحت کا اثر الفاظ سے نہیں، کہنے والے کے حال سے ہوتا ہے؛ جس کی زبان ایک بات کہے اور قدم دوسری طرف چلے، اس کی بات کانوں تک پہنچتی ہے، دلوں تک نہیں۔ شیخِ اکبر کے مکتب میں یہ صدق کا مقام ہے: ظاہر اور باطن کا ایک ہو جانا، یہاں تک کہ داعی کی ذات خود دعوت بن جائے۔

آیت کا اگلا حصہ داعی کی نیت کا آئینہ ہے: اِنْ اُرِيْدُ اِلَّا الْاِصْلَاحَ مَا اسْتَطَعْتُ۔ “میں تو بس اپنی استطاعت بھر اصلاح ہی چاہتا ہوں۔” اِنْ اور اِلَّا کا حصر بتاتا ہے کہ اس دعوت میں کوئی دوسری خواہش شامل نہیں: نہ اقتدار، نہ شہرت، نہ قوم پر برتری، نہ ان کے مال میں حصہ؛ صرف اصلاح۔ اور اصلاح کا لفظ خود ایک ادب رکھتا ہے: اصلاح کسی چیز کو مٹانا نہیں، اسے اس کی درست حالت پر لوٹانا ہے۔ نبی قوم کو مٹانے نہیں آتا، سنوارنے آتا ہے۔ پھر مَا اسْتَطَعْتُ: اپنی طاقت بھر۔ یہ قید بہت گہری ہے۔ داعی اپنی کوشش کا ذمہ دار ہے، نتیجے کا نہیں؛ وہ اتنا کرے جتنا اس کے بس میں ہے، اور جو بس سے باہر ہے اسے اپنے رب کے سپرد کرے۔ جو مصلح اس قید کو بھول جائے وہ یا تو مایوسی میں ٹوٹ جاتا ہے یا جبر کی طرف بڑھنے لگتا ہے؛ اور دونوں راستے انبیا علیہم السلام کی سنت سے ہٹے ہوئے ہیں۔ شیخِ اکبر کے نزدیک یہاں استعداد کا راز ہے: ہر وجود اپنی استعداد کے مطابق فیض قبول کرتا ہے، اور داعی کی استطاعت اور سننے والے کی استعداد جہاں ملتی ہیں وہیں ہدایت کا حصہ بنتا ہے۔ داعی بارش کی طرح برستا ہے؛ زمین کتنا جذب کرے، یہ زمین کا حال ہے۔ اور یہی اس سلسلے کا بنیادی اصول ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور اس کے نبی بھی کسی کو زبردستی ہدایت پر نہیں لاتے؛ دعوت عام ہے اور قبولیت بندے کا اپنا رخ۔

پھر وہ جملہ جو داعی کو اپنی کوشش کے دعوے سے بچا لیتا ہے: وَمَا تَوْفِيْقِيْۤ اِلَّا بِاللّٰهِ۔ “اور مجھے توفیق صرف اللہ ہی سے ہے۔” ابھی فرمایا تھا کہ میں اصلاح چاہتا ہوں، اور فوراً بتا دیا کہ چاہنا بھی اور کر پانا بھی اسی کی عطا ہے۔ توفیق کا لفظ موافقت سے ہے: بندے کے ارادے اور اسباب کا خیر کے مطابق ہو جانا۔ علما نے لکھا ہے کہ توفیق یہ ہے کہ اللہ بندے میں طاعت کی قدرت پیدا فرمائے اور اس کے لیے راستے آسان کر دے۔ اہلِ معرفت کہتے ہیں کہ توفیق حق کی طرف سے بندے کے رخ کا جواب ہے: بندہ طلب کا ایک قدم اٹھاتا ہے تو توفیق اس سے کہیں بڑھ کر آگے آتی ہے؛ اور حدیثِ قدسی کا مفہوم بھی یہی ہے، جو صحیح بخاری اور صحیح مسلم میں حضرت ابو ہریرہ رضی اللہ عنہ سے مروی ہے، کہ بندہ بالشت بھر قریب آئے تو حق تعالیٰ ہاتھ بھر قریب آتا ہے۔ اس میں جبر نہیں، جواب ہے؛ اور یہی ہدایت کے اصول کا دوسرا رخ ہے: جو طلب کرتا ہے اسے توفیق ملتی ہے۔ شیخِ اکبر کے نزدیک ہر فعل کا حقیقی موجد حق ہے اور بندہ اس فعل کا مظہر؛ مگر یہ بات شریعت کی ذمہ داری کو ختم نہیں کرتی، بلکہ بندے سے دعوے کو ختم کرتی ہے۔ مصلح کامیاب ہو تو کہے وَمَا تَوْفِيْقِيْۤ اِلَّا بِاللّٰهِ، اور کامیاب نہ ہو تب بھی یہی کہے؛ کامیابی میں یہ جملہ غرور سے بچاتا ہے اور ناکامی میں مایوسی سے۔ یہی وجہ ہے کہ اہلِ دل اس جملے کو اپنے ہر کام کی ابتدا اور انتہا میں دہراتے آئے ہیں۔

اور آیت ختم ہوتی ہے دو ستونوں پر: عَلَيْهِ تَوَكَّلْتُ وَاِلَيْهِ اُنِيْبُ۔ “میں نے اسی پر بھروسا کیا اور اسی کی طرف رجوع کرتا ہوں۔” دونوں جملوں میں عَلَيْهِ اور اِلَيْهِ پہلے آئے ہیں، جو عربی میں حصر کا فائدہ دیتے ہیں: بھروسا صرف اسی پر، رجوع صرف اسی کی طرف۔ اب صیغوں پر غور کیجیے، جو اس آیت کا ایک نہایت لطیف نکتہ ہے: تَوَكَّلْتُ ماضی ہے، اور اُنِيْبُ مضارع۔ اہلِ ذوق کہتے ہیں کہ توکل ایک طے شدہ فیصلہ ہے جو ایک بار دل میں جم جائے تو قائم رہتا ہے، اس لیے ماضی کے صیغے میں آیا؛ اور انابت ایک جاری عمل ہے جو ہر لمحہ تازہ ہوتا ہے، اس لیے مضارع میں۔ شیخِ اکبر کے نزدیک ہر آن خلقِ جدید ہے، اور جو ہر آن نیا ہو اس کا رجوع بھی ہر آن نیا ہونا چاہیے۔ یاد کیجیے کہ پچھلے رکوع میں حضرت ابراہیم علیہ السلام کو مُّنِيْبٌ کہا گیا تھا، اور یہاں حضرت شعیب علیہ السلام خود اپنے بارے میں اُنِيْبُ فرما رہے ہیں؛ گویا انابت انبیا علیہم السلام کی مشترک میراث ہے۔ ترتیب بھی قابلِ غور ہے: پہلے اصلاح کی کوشش، پھر استطاعت کا اعتراف، پھر توفیق کی نسبت اللہ کی طرف، اور آخر میں توکل۔ یعنی توکل کوشش کے بعد آتا ہے، اس کی جگہ نہیں لیتا۔ اہلِ معرفت کہتے ہیں کہ توکل اسباب کو چھوڑنا نہیں، اسباب کو ان کی جگہ پر رکھ کر دل کو مسبب الاسباب سے جوڑنا ہے؛ اسباب اس کے فعل کے پردے ہیں، اور متوکل پردے کو پہچانتا ہے مگر پردے پر بھروسا نہیں کرتا۔

اس ایک آیت کو اگر ایک نظر میں دیکھا جائے تو یہ ہر داعی اور ہر مصلح کا پورا دستور ہے: باطن میں رب کی طرف سے روشن دلیل، معاش میں رزقِ حسن کی بے نیازی، عمل میں قول کی مطابقت، نیت میں صرف اصلاح، کوشش میں استطاعت کی حد کا ادب، نتیجے میں توفیق کی نسبت اللہ کی طرف، دل میں توکل، اور ہر سانس میں انابت۔ اہلِ سیر و تفسیر میں مشہور ہے کہ حضرت شعیب علیہ السلام کو ان کے حسنِ کلام اور قوتِ استدلال کی وجہ سے خطیب الانبیاء کہا گیا؛ اور یہ آیت اس لقب کی روشن مثال ہے۔ مگر اس خطابت کی تاثیر لفظوں کے حسن میں نہیں، ان آٹھ منزلوں میں ہے جن سے یہ کلام نکلا۔ اہلِ معرفت کہتے ہیں کہ سالک کا پہلا میدانِ اصلاح اس کا اپنا نفس ہے؛ وہ پہلے اپنے اندر کی بستی میں ترازو سیدھا کرے، پھر باہر کسی سے بات کرے۔ اور جب اپنی اصلاح میں بھی کچھ پیش رفت ہو تو یہی کہے: وَمَا تَوْفِيْقِيْۤ اِلَّا بِاللّٰهِ؛ اور جب ٹھوکر کھائے تو یہی کہے: وَاِلَيْهِ اُنِيْبُ۔ شیخِ اکبر کے مکتب میں یہ جمع کا مقام ہے: کوشش بندے کی طرف سے ظاہر، توفیق حق کی طرف سے حقیقت، اور دونوں کا ملنا وہ مقام جہاں بندہ نہ اپنی کوشش کا منکر ہوتا ہے نہ اس پر نازاں۔ یہ وہ توازن ہے جسے کھو کر کوئی جبر کی طرف گرتا ہے اور کوئی خود پسندی کی طرف۔

پھر حضرت شعیب علیہ السلام نے تاریخ کو آئینہ بنا کر قوم کے سامنے رکھا: وَيٰقَوْمِ لَا يَجْرِمَنَّكُمْ شِقَاقِيْۤ اَنْ يُّصِيْبَكُمْ مِّثْلُ مَاۤ اَصَابَ قَوْمَ نُوْحٍ اَوْ قَوْمَ هُوْدٍ اَوْ قَوْمَ صٰلِحٍ ۗ وَمَا قَوْمُ لُوْطٍ مِّنْكُمْ بِبَعِيْدٍ۔ “اور اے میری قوم! میری مخالفت تمہیں اس پر آمادہ نہ کر دے کہ تم پر بھی ویسی ہی آفت آئے جیسی نوح علیہ السلام کی قوم، یا ہود علیہ السلام کی قوم، یا صالح علیہ السلام کی قوم پر آئی؛ اور لوط علیہ السلام کی قوم تو تم سے کچھ دور بھی نہیں۔” شِقَاقٌ کا مطلب ہے ایک طرف ہو جانا، دوسرے کے مقابل الگ کنارہ پکڑ لینا۔ حضرت شعیب علیہ السلام فرما رہے ہیں کہ میری ذات سے ضد تمہیں حق سے ضد تک نہ لے جائے؛ بہت سے لوگ بات سے نہیں، کہنے والے کی مخالفت میں راہ کھو دیتے ہیں۔ وَمَا قَوْمُ لُوْطٍ مِّنْكُمْ بِبَعِيْدٍ کے بارے میں مفسرین کہتے ہیں کہ وہ زمانے میں بھی قریب تھے اور جگہ میں بھی، اور بعض نے کہا عمل کی نوعیت میں بھی؛ مدین والے ان کے انجام کی خبر جانتے تھے۔ یہ سب تاریخی قومیں ہیں، اور یہاں ان کا ذکر ایک قوم کو اس کے اپنے قریبی ماضی کے آئینے میں دکھانے کے لیے ہے، کسی آج کی قوم پر چسپاں کرنے کے لیے نہیں۔ اہلِ معرفت کہتے ہیں کہ تاریخ اللہ کی سنتوں کی کتاب ہے: جو دوسروں کے انجام سے سبق لے وہ ان کے انجام سے بچ جاتا ہے، اور جو صرف کہانی سنے وہ خود کہانی بن جاتا ہے۔ سالک کے لیے قریب کا یہ لفظ اندر کی طرف اشارہ ہے: جن کیفیات نے پچھلی قوموں کو ہلاک کیا، یعنی ضد، حرص اور تکبر، وہ کسی دل سے دور نہیں؛ اس لیے نگرانی اپنے دل کی کرنی ہے۔

اور اس کے فوراً بعد، تنبیہ کے بیچ میں، رحمت کا دروازہ کھول دیا گیا: وَاسْتَغْفِرُوْا رَبَّكُمْ ثُمَّ تُوْبُوْۤا اِلَيْهِ ۗ اِنَّ رَبِّيْ رَحِيْمٌ وَّدُوْدٌ۔ “اور اپنے رب سے بخشش مانگو، پھر اس کی طرف رجوع کرو؛ بے شک میرا رب بڑا مہربان، بڑی محبت والا ہے۔” یہ وہی ترتیب ہے جو اس سورت کے آغاز سے بار بار گونج رہی ہے: پہلی آیات میں، حضرت ہود علیہ السلام کی دعوت میں، حضرت صالح علیہ السلام کی دعوت میں، اور اب یہاں: پہلے استغفار، پھر توبہ۔ استغفار پچھلے کا علاج ہے اور توبہ آئندہ کا رخ؛ استغفار داغ دھوتا ہے اور توبہ قدم موڑتی ہے۔ مگر اس آیت کا خاص جوہر آخری لفظ ہے: وَدُوْدٌ۔ یہ فَعُوْلٌ کے وزن پر ہے، جو مبالغہ بھی دیتا ہے اور دونوں معنی بھی رکھتا ہے: بہت محبت کرنے والا بھی، اور وہ جس سے بہت محبت کی جائے۔ حضرت شعیب علیہ السلام نے اپنی قوم کو، جو ناپ تول میں کمی کر رہی تھی اور ان پر طنز کر رہی تھی، یہ نہیں کہا کہ میرا رب سخت گرفت کرنے والا ہے؛ کہا کہ میرا رب رحیم ہے، ودود ہے۔ گویا توبہ کا دروازہ محض معافی کے صحن میں نہیں کھلتا، محبت کے گھر میں کھلتا ہے۔ سورۂ البروج میں یہ اسم مغفرت کے ساتھ آیا: وَهُوَ الْغَفُوْرُ الْوَدُوْدُ؛ اور یہاں رحمت کے ساتھ۔ دونوں جگہ پیغام ایک ہے: لوٹنے والا معاف ہی نہیں کیا جاتا، محبوب بھی بنایا جاتا ہے۔

شیخِ اکبر کے مکتب میں محبت وجود کی جڑ ہے۔ اہلِ تصوف میں مشہور اثر ہے کہ میں ایک چھپا ہوا خزانہ تھا، میں نے چاہا کہ پہچانا جاؤں، تو میں نے مخلوق کو پیدا کیا؛ محدثین نے اسے حدیث کی حیثیت سے ثابت نہیں مانا، مگر اہلِ معرفت اس کے مفہوم کو محبت اور معرفت کی بنیاد سمجھتے ہیں۔ اس روشنی میں الْوَدُوْدُ وہ اسم ہے جس کی طرف تخلیق کا آغاز منسوب ہے اور جس پر واپسی کی انتہا ہے۔ اہلِ لغت کہتے ہیں کہ وُدّ وہ محبت ہے جو ظاہر ہو اور قائم رہے، جو عمل میں اپنا اثر دکھائے۔ اس مکتب کے نزدیک بندے کی توبہ دراصل اس محبت کا جواب ہے جو اس سے پہلے موجود تھی؛ قرآن نے دوسری جگہ بھی اللہ کی محبت کو بندوں کی محبت سے پہلے ذکر فرمایا ہے۔ یعنی بندہ لوٹتا اس لیے ہے کہ اسے پہلے سے چاہا گیا تھا۔ یہ بات ہدایت کے اصول کو اور روشن کرتی ہے: الْوَدُوْدُ کسی کو اپنے دروازے سے دھکیلتا نہیں؛ جو دور ہوتا ہے وہ اپنے اعراض سے دور ہوتا ہے، اور جو پلٹتا ہے وہ پاتا ہے کہ دروازہ کبھی بند ہی نہیں تھا۔ اس مکتب میں تخلق باسماء اللہ کا مطلب یہ ہے کہ بندہ بھی اپنی بساط بھر ودود بنے، اپنی محبت میں مستقل، جیسے حضرت شعیب علیہ السلام اپنی ضدی قوم کے لیے بھی آخری لمحے تک محبت کی زبان بولتے رہے۔

اس آیت میں ایک اور لطیف بات “میرا رب” کا لفظ ہے: اِنَّ رَبِّيْ، نہ کہ تمہارا رب۔ پہلے جملے میں فرمایا رَبَّكُمْ، کہ اپنے رب سے بخشش مانگو؛ اور جب محبت کا ذکر آیا تو فرمایا رَبِّيْ۔ اہلِ ذوق کہتے ہیں کہ حضرت شعیب علیہ السلام یہاں اپنا ذاتی مشاہدہ بیان کر رہے ہیں: میں نے اپنے رب کو رحیم اور ودود پایا ہے، تم بھی آؤ اور پاؤ۔ شیخِ اکبر کے نزدیک ہر بندے کا ایک ربِّ خاص ہے، یعنی ربوبیت کی وہ خاص نسبت جس کے ذریعے حق اس بندے کی پرورش کرتا ہے؛ اور داعی کا کام یہ ہے کہ اپنے ربِّ خاص کی جو صفت اس نے چکھی ہے، اس کی خبر دوسروں کو دے۔ حضرت شعیب علیہ السلام نے رحمت اور محبت چکھی تھی، اس لیے انہوں نے اسی کی دعوت دی۔ یہ اس بات کی بھی تعلیم ہے کہ دین کی دعوت میں اللہ کو صرف سختی اور گرفت کی صورت میں پیش کرنا انبیا علیہم السلام کا طریقہ نہیں؛ وہ عذاب سے ڈراتے بھی ہیں، مگر دروازہ ہمیشہ رحمت اور محبت کی طرف کھولتے ہیں۔ سالک کے لیے بھی یہی سبق ہے: اپنے گناہ کے بوجھ تلے دب کر وہ یہ نہ سمجھے کہ اس کا رب اس سے روٹھ کر دور جا بیٹھا ہے؛ وہ رحیم ہے، ودود ہے، اور استغفار کا پہلا قدم ہی اس محبت تک پہنچنے کا راستہ ہے۔

مگر قوم نے دروازے کی طرف دیکھنے کے بجائے دیوار کھڑی کر دی: قَالُوْا يٰشُعَيْبُ مَا نَفْقَهُ كَثِيْرًا مِّمَّا تَقُوْلُ وَاِنَّا لَنَرٰىكَ فِيْنَا ضَعِيْفًا ۚ وَلَوْلَا رَهْطُكَ لَرَجَمْنٰكَ ۖ وَمَاۤ اَنْتَ عَلَيْنَا بِعَزِيْزٍ۔ “وہ بولے: اے شعیب (علیہ السلام)! تمہاری بہت سی باتیں ہماری سمجھ میں نہیں آتیں، اور ہم تمہیں اپنے درمیان کمزور دیکھتے ہیں؛ اور اگر تمہارا کنبہ نہ ہوتا تو ہم تمہیں سنگسار کر دیتے، اور تم ہم پر کچھ بھاری نہیں ہو۔” مَا نَفْقَهُ کا اعتراف خود ان کے حال کی گواہی ہے۔ حضرت شعیب علیہ السلام کا کلام صاف اور روشن تھا؛ نہ سمجھ پانا الفاظ کا قصور نہیں تھا، سننے والوں کے رخ کا نتیجہ تھا۔ یہ وہی اصول ہے جو اس سورت کی بیسویں آیت میں گزرا کہ وہ سن نہیں سکتے تھے اور دیکھتے نہیں تھے؛ یعنی جو دل پہلے سے منہ موڑ لے اس کے لیے سب سے واضح بات بھی بے معنی ہو جاتی ہے۔ یہ اللہ کی طرف سے کوئی زبردستی کا پردہ نہیں، ان کے اپنے اعراض کا پردہ ہے؛ حکم محکوم کے مطابق۔ ضَعِيْفًا میں وہی پیمانہ ہے جس سے دنیا لوگوں کو تولتی ہے: جس کے پیچھے جتھا نہ ہو وہ کمزور۔ اور لَوْلَا رَهْطُكَ: ہمیں تمہاری عزت کا پاس نہیں، تمہارے خاندان کا لحاظ ہے۔ اہلِ معرفت کہتے ہیں کہ یہ وہ نظر ہے جو صرف ظاہری اسباب کو دیکھتی ہے؛ اسے نبی میں چھپی قوت نظر نہیں آتی، صرف قبیلے کی گنتی نظر آتی ہے۔ یہ وہی ترازو ہے جس میں وہ اپنا مال تولتے تھے، اب اسی میں انسانوں کو تولنے لگے، اور دونوں جگہ ڈنڈی ماری۔

حضرت شعیب علیہ السلام کا جواب اس پورے رکوع کی سب سے کاٹ دار مگر سب سے باوقار بات ہے: قَالَ يٰقَوْمِ اَرَهْطِيْۤ اَعَزُّ عَلَيْكُمْ مِّنَ اللّٰهِ ۗ وَاتَّخَذْتُمُوْهُ وَرَآءَكُمْ ظِهْرِيًّا ۗ اِنَّ رَبِّيْ بِمَا تَعْمَلُوْنَ مُحِيْطٌ۔ “شعیب علیہ السلام نے کہا: اے میری قوم! کیا میرا کنبہ تمہارے نزدیک اللہ سے زیادہ عزت والا ہے؟ اور تم نے اسے پسِ پشت ڈال رکھا ہے؛ بے شک میرا رب تمہارے سب کاموں کو گھیرے ہوئے ہے۔” حضرت شعیب علیہ السلام نے اپنی ذات کا دفاع نہیں کیا، اللہ کی عزت کا سوال اٹھایا: تم چند لوگوں کے لحاظ سے رک گئے، مگر جس کا پیغام میں لایا ہوں اس کا تمہیں کوئی لحاظ نہیں۔ ظِهْرِيًّا وہ چیز ہے جو پیٹھ کے پیچھے ڈال دی جائے اور بھلا دی جائے۔ اور پھر جملہ ختم ہوا مُحِيْطٌ پر، اور یہی جواب کا راز ہے۔ اہلِ معرفت کہتے ہیں کہ انسان اللہ کو پیٹھ پیچھے ڈال سکتا ہے، مگر اللہ کسی کی پیٹھ کے پیچھے نہیں ہوتا؛ وہ ہر طرف سے محیط ہے۔ جسے تم نے پیچھے پھینکا وہ تمہیں آگے سے بھی گھیرے ہوئے ہے۔ شیخِ اکبر کے مکتب میں احاطہ کا مطلب حلول یا اتحاد نہیں، بلکہ علم، قدرت اور قیومیت کا ہر چیز پر قائم ہونا ہے، پوری تنزیہ کے ساتھ۔ بندے کا بھلانا صرف بندے کی اپنی سمت بدلتا ہے، حق کے احاطے میں کوئی کمی نہیں لاتا۔ سالک کے لیے یہاں ایک کسوٹی ہے: کہیں ایسا تو نہیں کہ وہ لوگوں کی ناراضی سے اللہ کی ناراضی سے زیادہ ڈرتا ہو؟ جس دن لوگوں کا لحاظ اللہ کے لحاظ پر غالب آ جائے، اسی دن دل میں اَرَهْطِيْۤ اَعَزُّ کا سوال کھڑا ہو جاتا ہے۔

آخر میں حضرت شعیب علیہ السلام نے معاملہ ان کے اپنے اختیار پر چھوڑ دیا: وَيٰقَوْمِ اعْمَلُوْا عَلٰى مَكَانَتِكُمْ اِنِّيْ عَامِلٌ ۗ سَوْفَ تَعْلَمُوْنَ ۙ مَنْ يَّاْتِيْهِ عَذَابٌ يُّخْزِيْهِ وَمَنْ هُوَ كَاذِبٌ ۗ وَارْتَقِبُوْۤا اِنِّيْ مَعَكُمْ رَقِيْبٌ۔ “اور اے میری قوم! تم اپنی جگہ اپنا کام کیے جاؤ، میں اپنا کام کر رہا ہوں؛ جلد تم جان لو گے کہ کس پر رسوا کرنے والا عذاب آتا ہے اور کون جھوٹا ہے؛ اور انتظار کرو، میں بھی تمہارے ساتھ انتظار کرنے والا ہوں۔” یہ جملہ اجازت نہیں، تنبیہ ہے؛ مگر اس تنبیہ میں انسان کے اختیار کی عزت بھی ہے۔ نبی نے ان کا ہاتھ زبردستی نہیں موڑا، کہا: تم اپنی راہ پر رہو، میں اپنی راہ پر؛ انجام فیصلہ کر دے گا۔ یہی وہ اصول ہے جس پر یہ سلسلہ قائم ہے: اللہ زبردستی گمراہ نہیں کرتا، بندے کو اس کا رخ چننے دیتا ہے، اور پھر اسی رخ کو وجود دیتا ہے۔ عَلٰى مَكَانَتِكُمْ یعنی اپنی حالت، اپنے مقام اور اپنی استطاعت کے مطابق۔ شیخِ اکبر کے نزدیک ہر وجود اپنی عینِ ثابتہ کے تقاضے کے مطابق ظاہر ہوتا ہے، اور اس مکتب میں یہ اصول معروف ہے کہ علم معلوم کے تابع ہوتا ہے؛ یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہو۔ وَارْتَقِبُوْۤا اِنِّيْ مَعَكُمْ رَقِيْبٌ میں نبی کا سکون دیکھیے: وہ جلدی میں نہیں، نہ انتقام کی خواہش میں؛ وہ بس انتظار کرنے والے ہیں، اس یقین کے ساتھ کہ وقت حق کا ساتھ دیتا ہے۔ سالک بھی جب اپنی بات کہہ چکے تو نتیجے کی جلدی نہ کرے؛ وہ عامل بھی رہے اور منتظر بھی۔

پھر وعدے کا وقت آ گیا: وَلَمَّا جَآءَ اَمْرُنَا نَجَّيْنَا شُعَيْبًا وَّالَّذِيْنَ اٰمَنُوْا مَعَهٗ بِرَحْمَةٍ مِّنَّا وَاَخَذَتِ الَّذِيْنَ ظَلَمُوا الصَّيْحَةُ فَاَصْبَحُوْا فِيْ دِيَارِهِمْ جٰثِمِيْنَ۔ “اور جب ہمارا حکم آ پہنچا تو ہم نے شعیب علیہ السلام کو اور ان کے ساتھ ایمان لانے والوں کو اپنی رحمت سے بچا لیا، اور ظالموں کو ایک ہولناک چیخ نے آ پکڑا، تو وہ اپنے گھروں میں اوندھے پڑے رہ گئے۔” كَاَنْ لَّمْ يَغْنَوْا فِيْهَا ۗ اَلَا بُعْدًا لِّمَدْيَنَ كَمَا بَعِدَتْ ثَمُوْدُ۔ “گویا وہ ان میں کبھی بسے ہی نہ تھے۔ سن لو! مدین کے لیے دوری ہے، جیسے ثمود دور کر دیے گئے۔” نجات کو بِرَحْمَةٍ مِّنَّا کہا گیا: اہلِ ایمان بھی اپنے عمل کے زور پر نہیں، اللہ کی رحمت سے بچے۔ اور عذاب کو الَّذِيْنَ ظَلَمُوْا سے جوڑا گیا، یعنی ان کے اپنے ظلم سے۔ قرآن نے دوسری جگہ اسی عذاب کے ساتھ زمین کے لرزنے کا ذکر بھی کیا ہے؛ مفسرین کہتے ہیں کہ دونوں جمع ہوئے: ایک ہولناک آواز اور اس کے ساتھ زمین کا کانپنا۔ ہم اسی پر رکتے ہیں جو قرآن نے بتایا، اور اس کی تفصیل میں اسرائیلی روایات سے گریز کرتے ہیں۔ كَمَا بَعِدَتْ ثَمُوْدُ: حضرت صالح علیہ السلام کی قوم کا انجام بھی صیحہ ہی سے ہوا تھا، اور اب مدین اسی صف میں شامل ہو گیا۔ یہ سب ایک تاریخی قوم کا انجام ہے؛ اسے آج کی کسی قوم یا کسی شہر پر چسپاں کرنا قرآن کے منشا کے خلاف ہے۔ كَاَنْ لَّمْ يَغْنَوْا فِيْهَا میں دنیا کی پوری حقیقت سمٹ آئی: جن بازاروں میں ناپ تول کی کمی سے دولت جمع کی جا رہی تھی، وہ ایسے ہو گئے گویا کبھی آباد ہی نہ تھے؛ جو کچھ کم تول کر بچایا گیا تھا وہ بچا نہیں، اور بَقِيَّتُ اللّٰهِ کی بات سچ نکلی۔

حضرت شعیب علیہ السلام کے باب کو شیخِ اکبر نے حکمتِ قلبیہ کا نام دیا، اور شارحینِ فصوص بیان کرتے ہیں کہ اس نسبت میں نام کی رعایت بھی ہے، کیونکہ شُعَب شاخوں کو کہتے ہیں، اور دل اپنی کیفیات اور اعتقادات میں بے شمار شاخیں رکھتا ہے۔ اس باب کا مفہوم یہ ہے کہ عارف کا دل حق کی تجلیات کے لیے اتنا وسیع ہے کہ زمین و آسمان بھی اتنی وسعت نہیں رکھتے؛ اہلِ تصوف میں مشہور اثر ہے کہ میری زمین اور میرا آسمان مجھے نہیں سما سکے، مگر میرے مومن بندے کا دل مجھے سما گیا۔ اور قلب کو قلب اس لیے کہتے ہیں کہ وہ تقلب میں رہتا ہے، ہر تجلی کے ساتھ نئی کیفیت اختیار کرتا ہے، جیسے ہر لمحہ خلقِ جدید ہے۔ اس روشنی میں مدین کا قصہ دو دلوں کا قصہ بن جاتا ہے: ایک دل جو اتنا تنگ ہو گیا کہ ناپ اور تول کے چند دانوں میں سمٹ گیا، اور ایک دل جو اتنا وسیع تھا کہ ضد، طنز اور دھمکی کے جواب میں بھی رحیم اور ودود کی خبر دیتا رہا۔ جو دل نرم آواز کو سننے سے انکار کر دے، اسے آخرکار سخت آواز سننی پڑتی ہے؛ مدین والوں نے حضرت شعیب علیہ السلام کی دعوت پر کان بند کیے تو صیحہ ان کے کانوں تک پہنچی، مگر وقت گزر چکا تھا۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں ایک تاریخی قوم کا عبرت ناک انجام ہے، اور باطن میں ہر سالک کے لیے یہ سوال کہ اس کا دل ترازو کے پلڑے جتنا ہے یا حق کی تجلی جتنا۔ سیدھا ترازو سلامت دل کا ظاہر ہے، اور وَاِلَيْهِ اُنِيْبُ اس دل کا باطن۔
