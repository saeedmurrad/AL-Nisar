پچھلے رکوع نے بتایا تھا کہ کس دل سے دیا جائے؛ یہ رکوع بتاتا ہے کہ کیا دیا جائے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْۤا اَنْفِقُوْا مِنْ طَيِّبٰتِ مَا كَسَبْتُمْ وَمِمَّاۤ اَخْرَجْنَا لَكُمْ مِّنَ الْاَرْضِ ۚ وَلَا تَيَمَّمُوا الْخَبِيْثَ مِنْهُ تُنْفِقُوْنَ وَلَسْتُمْ بِاٰخِذِيْهِ اِلَّاۤ اَنْ تُغْمِضُوْا فِيْهِ ۭ وَاعْلَمُوْۤا اَنَّ اللّٰهَ غَنِيٌّ حَمِيْدٌ۔ “اے ایمان والو، اپنی کمائی کی پاکیزہ چیزوں میں سے خرچ کرو اور اس میں سے بھی جو ہم نے تمہارے لیے زمین سے نکالا، اور اس میں سے ردی چیز چھانٹ کر خرچ کرنے کا ارادہ نہ کرو، حالانکہ تم خود اسے لینے والے نہیں مگر یہ کہ اس میں آنکھیں بند کر لو؛ اور جان رکھو کہ اللہ بے نیاز ہے، ہر تعریف کے لائق۔” مفسرین لکھتے ہیں کہ بعض لوگ کھجوروں کے گھٹیا خوشے لا کر مسجد میں فقرا کے لیے لٹکا دیتے تھے، تب یہ آیت اتری۔ کسوٹی بڑی سادہ رکھی گئی: جو چیز تم خود آنکھ بند کیے بغیر قبول نہ کرو، وہ دوسرے کو اللہ کے نام پر کیسے دیتے ہو؟ اب دو لفظوں کا فرق دیکھیے: مَا كَسَبْتُمْ، جو تم نے کمایا، اور مَاۤ اَخْرَجْنَا، جو ہم نے نکالا۔ اہلِ معرفت کہتے ہیں کہ یہ جوڑ کسب کی حقیقت کھولتا ہے۔ کسان ہل چلاتا ہے، مگر دانے کو مٹی سے نکالنا اس کا کام نہیں؛ تاجر محنت کرتا ہے، مگر رزق کا اخراج کسی اور کا فعل ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں افعال کی نسبت اصل میں حق کی طرف ہے اور بندہ ان کا مظہر ہے؛ تو جو مال تمہارے ہاتھ میں آیا وہ پہلے اسی کا نکالا ہوا تھا۔ جب اسی کی چیز اسی کو لوٹا رہے ہو تو اس میں سے ردی چھانٹنا کیسا؟

یہ معیار ایک اور مقام پر یوں کھول دیا گیا: لَنْ تَنَالُوا الْبِرَّ حَتّٰى تُنْفِقُوْا مِمَّا تُحِبُّوْنَ۔ نیکی کی حقیقت تک رسائی اسی وقت ہے جب دل کی پسندیدہ چیز دی جائے۔ اہلِ ذوق اس آیت کو مال سے آگے بھی لے جاتے ہیں۔ بندہ اپنے رب کو کیا پیش کرتا ہے؟ دن کا تھکا ہوا آخری حصہ نماز کے لیے، توجہ کا بچا کھچا ٹکڑا ذکر کے لیے، اور زبان کی وہ بات جو کسی اور کام کی نہ رہی ہو؟ یہ بھی خبیث کا تیمم ہے، یعنی ردی کو چھانٹ کر نذر کرنا۔ اہلِ معرفت کہتے ہیں کہ دل کا بہترین لمحہ ہی اس کا طیب ہے، اور وہی بارگاہ کے لائق۔ پھر آیت دو اسماء پر رکی: غَنِيٌّ حَمِيْدٌ۔ غنی اس لیے کہ اسے تمہارے عمدہ مال کی بھی حاجت نہیں؛ یہ حکم تمہاری اپنی تربیت کے لیے ہے۔ اور حمید اس لیے کہ وہ تھوڑے کو بھی قبول کر کے اس پر قدر دانی فرماتا ہے۔ شیخِ اکبر کے نزدیک بندہ جو کچھ دیتا ہے وہ اس کے اپنے گمان کا آئینہ ہے: جو اپنے رب کو کریم جانتا ہے وہ عمدہ دیتا ہے، اور جو اسے ردی کا حق دار سمجھے، اس نے دراصل اپنے رب کو پہچانا ہی نہیں۔

اب وہ آواز سنائی جاتی ہے جو ہر دینے والے کے کان میں ہاتھ روکنے کو کہتی ہے: اَلشَّيْطٰنُ يَعِدُكُمُ الْفَقْرَ وَيَاْمُرُكُمْ بِالْفَحْشَآءِ ۚ وَاللّٰهُ يَعِدُكُمْ مَّغْفِرَةً مِّنْهُ وَفَضْلًا ۭ وَاللّٰهُ وَاسِعٌ عَلِيْمٌ۔ “شیطان تمہیں تنگ دستی سے ڈراتا ہے اور بے حیائی کا حکم دیتا ہے، اور اللہ تم سے اپنی بخشش اور فضل کا وعدہ کرتا ہے؛ اور اللہ وسعت والا، سب جاننے والا ہے۔” مفسرین کے نزدیک یہاں فحشاء میں بخل بھی داخل ہے، کیونکہ عرب کے ہاں حد سے بڑھی ہوئی کنجوسی کو بھی فحش کہا جاتا تھا۔ شیطان کا ہتھیار دیکھیے: وعدہ۔ وہ ہاتھ نہیں پکڑتا، کان میں کہتا ہے کہ دو گے تو کل کیا کھاؤ گے؟ سورۂ ابراہیم میں اس نے خود اقرار کیا تھا: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ، مجھے تم پر کوئی زور حاصل نہ تھا۔ یعنی یہاں بندے کے سامنے دو وعدے رکھے گئے ہیں، ایک تنگی کا اور ایک مغفرت اور فضل کا، اور انتخاب اسی کا ہے۔ جس وعدے پر وہ بھروسہ کرے، حق اسی رخ کو وجود دے دیتا ہے؛ یہی سرِّ قدر ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم کی حدیث کا مفہوم ہے کہ صدقے سے مال کم نہیں ہوتا (مسلم، بالمعنی)۔ شیطان کا وعدہ آنکھ کے حساب پر کھڑا ہے اور اللہ کا وعدہ اس حقیقت پر جو آنکھ سے پرے ہے۔

اس آیت میں ایک لطیف راز بھی ہے۔ شیطان جس فقر سے ڈراتا ہے وہ چیزوں کا فقر ہے، یعنی ہاتھ خالی ہو جانے کا خوف؛ اور اس خوف سے وہ ایک اور فقر کو چھپاتا ہے جو انسان کی اصل حقیقت ہے۔ سورۂ فاطر میں اعلان ہوا تھا: يٰۤاَيُّهَا النَّاسُ اَنْتُمُ الْفُقَرَآءُ اِلَى اللّٰهِ۔ شیخِ اکبر کے نزدیک بندہ ذاتاً فقیر ہے اور حق ذاتاً غنی؛ بندے کا کمال اپنی محتاجی کو پہچاننا ہے، اسے ڈھانپنا نہیں۔ جو شخص چیزوں کی محتاجی سے ڈرتا ہے وہ دراصل چیزوں کو اپنا سہارا بنا بیٹھا ہے، اور جو اللہ کی طرف اپنے فقر کو پہچان لیتا ہے وہ چیزوں کی کمی سے نہیں گھبراتا، کیونکہ اس کا سہارا کم نہیں ہوتا۔ اہلِ تصوف اسی کو فقر و تجرید کہتے ہیں: دل کا غیر کی وابستگی سے خالی ہو جانا۔ پھر اللہ کے وعدے کی ترتیب دیکھیے: پہلے مغفرت، پھر فضل؛ پہلے دل کی صفائی، پھر عطا کی فراوانی۔ اور آخر میں وَاسِعٌ عَلِيْمٌ، وہی دو نام جن پر سات سو دانوں والی آیت ختم ہوئی تھی۔ تنگی کے وسوسے کا جواب وسعت کا نام ہے، اور اندیشۂ کل کا جواب یہ کہ وہ تمہاری کل کی ضرورت بھی جانتا ہے۔

اور ان دو وعدوں کے درمیان پہچان کی جو قوت فیصلہ کرتی ہے، اسی کا ذکر اگلی آیت میں ہے: يُؤْتِي الْحِكْمَةَ مَنْ يَّشَآءُ ۚ وَمَنْ يُّؤْتَ الْحِكْمَةَ فَقَدْ اُوْتِيَ خَيْرًا كَثِيْرًا ۭ وَمَا يَذَّكَّرُ اِلَّاۤ اُولُوا الْاَلْبَابِ۔ “وہ جسے چاہتا ہے حکمت عطا کرتا ہے، اور جسے حکمت ملی اسے بہت بڑی بھلائی مل گئی؛ اور نصیحت وہی قبول کرتے ہیں جو عقل کے مغز والے ہیں۔” آیت کی جگہ پر غور کیجیے۔ شیطان کا وعدہ اور اللہ کا وعدہ آمنے سامنے رکھے گئے، اور فوراً حکمت کا ذکر آیا؛ کیونکہ حکمت ہی وہ نظر ہے جو دونوں وعدوں میں سے سچے کو پہچانتی ہے۔ مفسرین نے حکمت کے کئی معنی لکھے ہیں: قرآن کی سمجھ، سنت، قول و عمل میں درستی، اور اللہ کا خوف؛ اور ایک مشہور قول ہے کہ حکمت کی جڑ اللہ کا ڈر ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم کی حدیث کا مفہوم ہے کہ رشک صرف دو آدمیوں پر روا ہے: ایک وہ جسے اللہ نے مال دیا اور وہ اسے حق کی راہ میں لٹاتا ہے، اور دوسرا وہ جسے اللہ نے حکمت دی اور وہ اس کے مطابق فیصلہ کرتا اور اسے سکھاتا ہے (بخاری، بالمعنی)۔ یہ رکوع بھی انہی دو دولتوں کو ساتھ رکھتا ہے: مال کا خرچ اور حکمت کی عطا۔ گویا اصل مال حکمت ہے، اور حکمت والا ہی جانتا ہے کہ مال کہاں رکھا جائے۔

اہلِ معرفت حکمت کی تعریف یوں کرتے ہیں: ہر چیز کو اس کی جگہ رکھنا۔ الحکیم وہ ہے جو ہر شے کو اس کا حق اور اس کا مقام دیتا ہے، اور بندۂ حکیم اسی نام کا مظہر ہے: وہ سختی کو اس کی جگہ اور نرمی کو اس کی جگہ رکھتا ہے، دینے کو اس کے وقت اور روکنے کو اس کے وقت۔ شیخِ اکبر کی کتاب فصوص الحکم کا نام ہی حکمتوں کے نگینے ہے، اور اس میں ہر نبی کے ساتھ ایک حکمت منسوب ہے، گویا ہر نبی ایک انگوٹھی کا وہ خانہ ہے جس میں اس کے حال کے مطابق حکمت کا نگینہ جڑا گیا۔ اس مکتب میں حکمت برتن کی صورت لیتی ہے؛ ایک ہی نور ہر دل میں اس کی ساخت کے مطابق جلوہ گر ہوتا ہے۔ قرآن نے حضرت لقمان کے بارے میں فرمایا: وَلَقَدْ اٰتَيْنَا لُقْمٰنَ الْحِكْمَةَ، اور سورۂ لقمان کے صفحے پر حکمتِ احسانیہ کا ذکر گزر چکا ہے؛ وہاں بھی حکمت کا پہلا پھل شکر تھا۔ یہاں آیت میں لفظ يُؤْتَ مجہول ہے: حکمت دی جاتی ہے، کمائی نہیں جاتی۔ حکیم اپنی حکمت کو اپنا کارنامہ نہیں سمجھتا، اور یہی اس کی حکمت کی پہلی نشانی ہے۔ پھر خَيْرًا كَثِيْرًا فرمایا، بہت بھلائی، ساری بھلائی نہیں؛ حکمت والا اپنی حد جانتا ہے، اور اس مکتب میں علم کی انتہا حیرت ہے، دعویٰ نہیں۔

اب مَنْ يَّشَآءُ کو اس سورت کے اصول کے ساتھ پڑھیے۔ یہاں مشیت کا مطلب یہ نہیں کہ عطا اندھی ہے اور طالب محروم رہ جاتا ہے۔ قرآن کا وعدہ ہے: وَالَّذِيْنَ جَاهَدُوْا فِيْنَا لَنَهْدِيَنَّهُمْ سُبُلَنَا۔ جو اس کی راہ میں کوشش کرتا ہے اسے راہیں دکھائی جاتی ہیں۔ شیخِ اکبر کے نزدیک فیضِ اقدس ہر حقیقت کو اس کی استعداد دیتا ہے اور فیضِ مقدس اس استعداد کے مطابق عطا اتارتا ہے؛ مشیت علم کے ساتھ ہے، اور علم معلوم کی حقیقت کے مطابق۔ تو جس دل نے اپنا برتن طلب، صدق اور تقویٰ سے تیار کیا، حکمت اسی کی طرف بہتی ہے؛ اور جس نے برتن الٹا رکھا، اس نے بارش سے خود منہ موڑا۔ پھر آیت کا خاتمہ: وَمَا يَذَّكَّرُ اِلَّاۤ اُولُوا الْاَلْبَابِ۔ لُبّ چھلکے کے اندر کا مغز ہے۔ عقل والے وہ ہیں جو چیزوں کے چھلکے پر نہیں رکتے، مغز تک پہنچتے ہیں: مال کے چھلکے میں امانت، صدقے کے چھلکے میں قرب، اور فقیر کی صورت میں اللہ کا بھیجا ہوا موقع دیکھتے ہیں۔ اور يَذَّكَّرُ یاد آنا ہے، نیا سیکھنا نہیں۔ اہلِ معرفت کہتے ہیں کہ حکمت دل میں کوئی اجنبی چیز نہیں ڈالتی، اس فطرت کو جگاتی ہے جو پہلے سے اپنے رب کو جانتی تھی۔

پھر ہر خرچ اور ہر نذر کو ایک ہی گواہ کے سپرد کیا گیا: وَمَاۤ اَنْفَقْتُمْ مِّنْ نَّفَقَةٍ اَوْ نَذَرْتُمْ مِّنْ نَّذْرٍ فَاِنَّ اللّٰهَ يَعْلَمُهٗ ۭ وَمَا لِلظّٰلِمِيْنَ مِنْ اَنْصَارٍ۔ “اور تم جو کچھ بھی خرچ کرو یا جو بھی نذر مانو، اللہ اسے جانتا ہے؛ اور ظالموں کا کوئی مددگار نہیں۔” نذر یہ ہے کہ بندہ کسی ایسی نیکی کو اپنے اوپر لازم کر لے جو پہلے لازم نہ تھی؛ پھر اسے پورا کرنا واجب ہو جاتا ہے۔ مفسرین لکھتے ہیں کہ یہاں ظالم وہ ہیں جو مال ناحق جگہ خرچ کریں، واجب حق روک لیں، یا نذر مان کر پوری نہ کریں۔ اہلِ معرفت کہتے ہیں کہ يَعْلَمُهٗ میں دینے والے کی ساری تسلی ہے۔ جس عمل کو اللہ جانتا ہو اسے کسی اور گواہ کی حاجت نہیں؛ اور جو اس علم پر قانع ہو جائے اس کے لیے لوگوں کا دیکھنا یا نہ دیکھنا برابر ہو جاتا ہے۔ یہی بات اگلی آیت میں چھپا کر دینے کی فضیلت تک لے جاتی ہے۔ اور ظالموں کا مددگار نہ ہونا کوئی جبر نہیں: ظلم کا پہلا شکار ظالم کی اپنی جان ہے، اور جس نے اکیلے مددگار سے منہ پھیر لیا، اس نے مدد کا دروازہ اپنے ہاتھ سے بند کیا۔

اب ظاہر اور باطن میں دینے کا موازنہ ہوا: اِنْ تُبْدُوا الصَّدَقٰتِ فَنِعِمَّا هِيَ ۚ وَاِنْ تُخْفُوْهَا وَتُؤْتُوْهَا الْفُقَرَآءَ فَهُوَ خَيْرٌ لَّكُمْ ۭ وَيُكَفِّرُ عَنْكُمْ مِّنْ سَيِّاٰتِكُمْ ۭ وَاللّٰهُ بِمَا تَعْمَلُوْنَ خَبِيْرٌ۔ “اگر تم صدقے کھلے دو تو یہ بھی خوب ہے، اور اگر انہیں چھپا کر فقیروں کو دو تو یہ تمہارے لیے زیادہ بہتر ہے، اور وہ تمہاری کچھ برائیاں تم سے دور کر دے گا؛ اور اللہ تمہارے کاموں سے پوری طرح باخبر ہے۔” ظاہر کا حکم دونوں کو سراہتا ہے؛ مفسرین لکھتے ہیں کہ فرض زکوٰۃ کھلے دینا بہتر ہے تاکہ دوسروں کو ترغیب ہو اور تہمت نہ رہے، اور نفلی صدقہ چھپا کر دینا افضل ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم نے ان سات خوش نصیبوں میں جنہیں قیامت کے دن عرش کا سایہ ملے گا، اس شخص کو بھی گنا جو یوں چھپا کر صدقہ دے کہ اس کے بائیں ہاتھ کو خبر نہ ہو کہ دائیں نے کیا دیا (بخاری و مسلم، بالمعنی)۔ اہلِ ذوق کہتے ہیں کہ یہ بایاں ہاتھ نفس ہے: سب سے گہرا اخفا یہ ہے کہ نفس کو بھی خبر نہ ہو، کہ وہ اس عطا پر اپنا دعویٰ نہ رکھ سکے۔ شیخِ اکبر کے مکتب میں اس کی انتہا یہ ہے کہ بندہ اپنے آپ کو دینے والا ہی نہ دیکھے، اپنے فعل کو حق کے فعل میں فانی پائے۔ اور خفا میں لینے والے کی عزت بھی محفوظ رہتی ہے؛ اس کا ہاتھ کسی کی آنکھ کے سامنے نیچا نہیں ہوتا۔ آیت خَبِيْرٌ پر ختم ہوئی: وہ جو چھپا ہوا ہے اسی کو جاننے والا، جس کے علم میں پوشیدہ صدقہ بھی روشن ہے۔

اس کے بعد خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہوا: لَيْسَ عَلَيْكَ هُدٰىهُمْ وَلٰكِنَّ اللّٰهَ يَهْدِيْ مَنْ يَّشَآءُ ۭ وَمَا تُنْفِقُوْا مِنْ خَيْرٍ فَلِاَنْفُسِكُمْ ۭ وَمَا تُنْفِقُوْنَ اِلَّا ابْتِغَآءَ وَجْهِ اللّٰهِ ۭ وَمَا تُنْفِقُوْا مِنْ خَيْرٍ يُّوَفَّ اِلَيْكُمْ وَاَنْتُمْ لَا تُظْلَمُوْنَ۔ “ان کو راہ پر لے آنا آپ کے ذمے نہیں، بلکہ اللہ جسے چاہتا ہے راہ دکھاتا ہے؛ اور تم جو بھلائی بھی خرچ کرو وہ تمہارے اپنے لیے ہے، اور تم اللہ کی رضا کے سوا کسی اور غرض سے خرچ نہیں کرتے؛ اور جو بھلائی تم خرچ کرو گے وہ تمہیں پوری لوٹا دی جائے گی اور تم پر ظلم نہ ہوگا۔” مفسرین نے حضرت ابنِ عباس رضی اللہ عنہما سے نقل کیا ہے، جس کا مفہوم یہ ہے کہ بعض مسلمان اپنے غیر مسلم رشتہ داروں اور محتاجوں کو صدقہ دینے سے رکتے تھے، اس خیال سے کہ شاید یوں وہ اسلام لے آئیں؛ تب یہ آیت اتری۔ یعنی صدقہ دباؤ کا ذریعہ نہیں، اور بھوکے کی روٹی اس کے عقیدے سے مشروط نہیں۔ یہ وہی بات ہے جو چند آیات پہلے کہی گئی تھی: لَاۤ اِكْرَاهَ فِي الدِّيْنِ۔ دین میں زبردستی نہیں، نہ تلوار سے نہ روٹی سے۔

اور وَلٰكِنَّ اللّٰهَ يَهْدِيْ مَنْ يَّشَآءُ اس سورت کا اصول ہے، اسی کے ساتھ پڑھا جائے۔ حضورِ اکرم صلی اللہ علیہ وسلم کا منصب پہنچانا ہے، اور دلوں کو پھیرنا اللہ کا کام؛ مگر اللہ کی مشیت حکمت والی ہے، اندھی نہیں۔ وہ کسی کو زبردستی ہدایت پر نہیں لاتا اور کسی طالب کو ہدایت سے محروم نہیں کرتا؛ جو اس کی طرف رخ کرتا ہے اس تک ہدایت پہنچتی ہے، اور جو منہ موڑتا ہے اس کا منہ موڑنا ہی اس کی محرومی ہے۔ اگر وہ چاہتا تو سب کو مجبوراً سیدھا کر دیتا، مگر اس نے انسان کو انتخاب کی عزت دی۔ اسی لیے داعی کا کام دعوت اور خدمت ہے، نتیجے کا بوجھ اٹھانا نہیں؛ اور یہ بات داعی کے دل سے وہ بوجھ اتار دیتی ہے جو اس کا تھا ہی نہیں۔ پھر فَلِاَنْفُسِكُمْ فرمایا: تمہارا دیا ہوا تمہارے ہی لیے ہے۔ شیخِ اکبر کے نزدیک بندہ دوسرے کو دیتے ہوئے دراصل اپنی ہی حقیقت کو سنوارتا ہے، کیونکہ عمل اپنے کرنے والے پر لوٹتا ہے۔ اور ابْتِغَآءَ وَجْهِ اللّٰهِ اہلِ ایمان کی پہچان بتا دی گئی: ان کا خرچ صرف اس کے چہرے کی طلب میں ہے۔ اہلِ معرفت کے نزدیک وجہ اللہ وہ جہت ہے جس سے حق اپنے آپ کو ظاہر کرتا ہے، اور جو اس کا طالب ہو وہ لینے والے کے مذہب کو نہیں، اپنے رب کی رضا کو دیکھتا ہے۔ اور يُوَفَّ اِلَيْكُمْ: پورا لوٹایا جائے گا، ایک دانہ بھی کم نہ ہوگا۔

رکوع کی آخری آیت ان لوگوں کی تصویر کھینچتی ہے جو سب سے زیادہ حق دار ہیں اور سب سے کم نظر آتے ہیں: لِلْفُقَرَآءِ الَّذِيْنَ اُحْصِرُوْا فِيْ سَبِيْلِ اللّٰهِ لَا يَسْتَطِيْعُوْنَ ضَرْبًا فِي الْاَرْضِ ۚ يَحْسَبُهُمُ الْجَاهِلُ اَغْنِيَآءَ مِنَ التَّعَفُّفِ ۚ تَعْرِفُهُمْ بِسِيْمٰىهُمْ ۚ لَا يَسْاَلُوْنَ النَّاسَ اِلْحَافًا ۭ وَمَا تُنْفِقُوْا مِنْ خَيْرٍ فَاِنَّ اللّٰهَ بِهٖ عَلِيْمٌ۔ “یہ صدقے ان فقیروں کے لیے ہیں جو اللہ کی راہ میں ایسے گھر گئے ہیں کہ زمین میں چل پھر کر روزی نہیں کما سکتے؛ ناواقف آدمی ان کے سوال سے بچنے کی وجہ سے انہیں مال دار سمجھتا ہے؛ تم انہیں ان کے چہرے کی علامت سے پہچانتے ہو؛ وہ لوگوں سے لپٹ کر نہیں مانگتے؛ اور تم جو بھلائی بھی خرچ کرو، اللہ اسے خوب جانتا ہے۔” مفسرین لکھتے ہیں کہ اس کا اولین مصداق اصحابِ صفہ تھے، جو مسجدِ نبوی کے چبوترے پر علم اور خدمتِ دین کے لیے ٹھہرے رہتے تھے۔ حضورِ اکرم صلی اللہ علیہ وسلم کی حدیث کا مفہوم ہے کہ مسکین وہ نہیں جو در در پھرے اور ایک دو لقمے یا ایک دو کھجوریں اسے لوٹا دیں، بلکہ مسکین وہ ہے جس کے پاس اتنا نہ ہو جو اسے کافی ہو، نہ اس کا حال کسی کو معلوم ہو کہ اسے صدقہ دیا جائے، اور نہ وہ کھڑا ہو کر لوگوں سے مانگے (بخاری و مسلم، بالمعنی)۔ یعنی دینے والے کا فرض ہے کہ وہ ڈھونڈے، صرف انتظار نہ کرے۔

اہلِ معرفت اس آیت میں فقر کی سب سے روشن تصویر دیکھتے ہیں۔ ناواقف انہیں غنی سمجھتا ہے، اور ایک معنی میں وہ واقعی غنی ہیں: ان کی بے نیازی لوگوں سے ہے، کیونکہ ان کی محتاجی صرف اللہ سے ہے۔ یہی فقر و تجرید ہے، کہ ہاتھ خالی ہو اور دل بھرا ہوا۔ اَلتَّعَفُّف ان کا لباس ہے، اور سِيْمٰىهُمْ ان کا چہرہ۔ وہ نشان کیا ہے؟ مفسرین نے فاقے کی زردی اور خشوع لکھا ہے، اور اہلِ ذوق کہتے ہیں کہ اللہ کی طرف حاجت کا وہ نور جو چہرے پر اتر آتا ہے، جسے صرف وہی آنکھ پڑھتی ہے جو خود اس حاجت کو جانتی ہو؛ اسی لیے فرمایا تَعْرِفُهُمْ، تم پہچانتے ہو، الْجَاهِلُ نہیں۔ لَا يَسْاَلُوْنَ النَّاسَ اِلْحَافًا میں ادب کی ایک پوری دنیا ہے: ان کی زبان پر سوال نہیں، اور ان کی خاموشی بھی شکایت نہیں۔ شیخِ اکبر کے مکتب میں فقیر کا کمال یہ ہے کہ اس کا سوال صرف رب کے سامنے ہو؛ وہ مخلوق کو واسطہ مانتا ہے، مقصود نہیں۔ اور رکوع وہیں لوٹ آیا جہاں سے یہ سلسلہ شروع ہوا تھا: فَاِنَّ اللّٰهَ بِهٖ عَلِيْمٌ۔ جس دانے سے سات سو دانے اگنے تھے، اس کا بیج بونے والا بھی وہی جانتا ہے، اور اس بیج کی سب سے زرخیز زمین وہ خاموش ہاتھ ہے جو کبھی پھیلا نہیں۔
