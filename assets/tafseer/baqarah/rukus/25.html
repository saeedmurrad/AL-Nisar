پچھلا رکوع بیت اللہ کے دروازے پر ختم ہوا تھا؛ یہ رکوع اس سفر کے آداب سے شروع ہوتا ہے: اَلْحَجُّ اَشْهُرٌ مَّعْلُوْمٰتٌ ۚ فَمَنْ فَرَضَ فِيْهِنَّ الْحَجَّ فَلَا رَفَثَ وَلَا فُسُوْقَ ۙ وَلَا جِدَالَ فِي الْحَجِّ۔ حج کے چند جانے پہچانے مہینے ہیں؛ تو جو ان میں حج اپنے اوپر لازم کر لے، تو حج میں نہ شہوت کی بات ہو، نہ نافرمانی، نہ جھگڑا۔ جمہور کے نزدیک یہ مہینے شوال، ذوالقعدہ اور ذوالحجہ کے پہلے دس دن ہیں، اور "فَرَضَ" یہ ہے کہ احرام باندھ کر تلبیہ کہہ دیا۔ حدیث کا مفہوم ہے کہ جس نے حج کیا اور اس میں نہ شہوت کی بات کی نہ نافرمانی، وہ ایسے لوٹتا ہے جیسے آج ہی ماں کے پیٹ سے پیدا ہوا ہو (بخاری و مسلم)۔ اہلِ معرفت ان تین ممانعتوں میں تین پردے دیکھتے ہیں: رفث بدن کی خواہش کا پردہ ہے، فسوق نفس کی سرکشی کا، اور جدال عقل اور انا کی کشمکش کا۔ پہلے دو کو سب پہچانتے ہیں؛ تیسرا سب سے باریک ہے، کیونکہ جھگڑنے والا اکثر یہ سمجھتا ہے کہ وہ حق کی خاطر بول رہا ہے۔ حج میں وہ بحث بھی روک دی گئی جو جھگڑے تک لے جائے، چاہے آدمی حق پر ہو، کہ دل کو خاموشی چاہیے تاکہ لبیک کے بعد جواب سن سکے۔ اور اس مکتب میں ظاہر کا بیت اللہ باطن کے دل کی طرف اشارہ ہے؛ اہلِ تصوف میں یہ اثر مشہور ہے کہ حق کو زمین و آسمان نہ سما سکے اور مومن بندے کا دل سما گیا۔ تو جو حرم کو رفث، فسوق اور جدال سے پاک رکھتا ہے، وہ دراصل دل کو ان سے خالی کرنے کی مشق کرتا ہے۔ یہ اشاری معنی ظاہری حکم کو ذرا بھی نہیں ہلاتا؛ حج کے مناسک اپنی جگہ فرض ہیں، اور اشارہ ان کی روح کی خبر ہے۔

اسی آیت کا دوسرا حصہ اس رکوع کا مرکز ہے: وَمَا تَفْعَلُوْا مِنْ خَيْرٍ يَّعْلَمْهُ اللّٰهُ ۚ وَتَزَوَّدُوْا فَاِنَّ خَيْرَ الزَّادِ التَّقْوٰى وَاتَّقُوْنِ يٰٓاُولِي الْاَلْبَابِ۔ اور تم جو بھلائی بھی کرو گے اللہ اسے جانتا ہے؛ اور زادِ راہ ساتھ لو، پس بہترین زادِ راہ تقویٰ ہے؛ اور اے عقل والو! مجھ سے ڈرو۔ حضرت ابن عباس رضی اللہ عنہما کی روایت کا مفہوم ہے کہ اہلِ یمن حج کو آتے اور زادِ راہ ساتھ نہ لاتے، کہتے ہم متوکل ہیں، پھر مکہ پہنچ کر لوگوں سے مانگتے؛ تب یہ حکم اترا (بخاری)۔ یوں پہلا سبق یہ ملا کہ توکل اسباب چھوڑنے کا نام نہیں؛ حدیث کا مفہوم ہے کہ اونٹ کو باندھو پھر بھروسا کرو (ترمذی)۔ جو اسباب چھوڑ کر مخلوق کے آگے ہاتھ پھیلائے، اس نے توکل نہیں کیا، اپنا بوجھ دوسروں پر ڈال دیا۔ پھر آیت ایک ہی سانس میں راستہ بدل دیتی ہے: دنیا کے سفر کا توشہ لو، اور یاد رکھو کہ ایک اور سفر بھی ہے جس کا توشہ روٹی اور پانی نہیں، تقویٰ ہے۔ اور اس سے پہلے کا جملہ "وَمَا تَفْعَلُوْا مِنْ خَيْرٍ يَّعْلَمْهُ اللّٰهُ" اخلاص کی بنیاد ہے: بھلائی کو کسی گواہ کی ضرورت نہیں جب اسے جاننے والا وہ ہے۔ جو اس پر قانع ہو گیا کہ اللہ جانتا ہے، اس کا عمل لوگوں کی نگاہ سے آزاد ہو گیا۔ اور خطاب "يٰٓاُولِي الْاَلْبَابِ" پر ٹھہرا، یعنی ان سے جن کے پاس مغز ہے، کیونکہ تقویٰ کا راز وہی سمجھتا ہے جو چھلکے سے گزر کر مغز تک پہنچا۔

شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں تقویٰ کی جڑ وقایہ ہے، یعنی ڈھال بنانا، اور اس کے دو رخ بیان کیے جاتے ہیں جن کا مفہوم یہ ہے: ہر خوبی میں حق کو اپنی ڈھال بناؤ کہ اسے اسی کی طرف منسوب کرو، اور ہر نقص میں خود کو اس کی ڈھال بناؤ کہ اسے اپنی طرف منسوب کرو۔ یہی ادب حضرت ابراہیم علیہ السلام کی زبان پر تھا جب بیماری اپنی طرف اور شفا رب کی طرف منسوب کی۔ تو بہترین زاد وہ ادب ہے جو ہر حال میں حق کو حق کا مقام اور بندے کو بندے کا مقام دیتا ہے۔ اہلِ ذوق کہتے ہیں کہ اس سفر کی منزل ختم نہیں ہوتی، کیونکہ حق کی طرف سیر کے بعد حق میں سیر ہے جس کی کوئی انتہا نہیں؛ اس لیے یہ توشہ بھی کبھی ختم نہیں ہونا چاہیے۔ اور "اُولِي الْاَلْبَابِ" کا اشارہ یہ ہے کہ شریعت چھلکا ہے جو مغز کی حفاظت کرتا ہے؛ چھلکا توڑ کر مغز نکالنے والا اسے سڑا دیتا ہے، اور جو چھلکے ہی پر رک جائے وہ مغز سے محروم رہتا ہے۔ عقل والا دونوں کو سنبھالتا ہے۔ اور فقر کا راز بھی یہیں ہے: یمن کا وہ مسافر جو توشے کے بغیر آیا، لوگوں سے مانگنے پر مجبور ہوا؛ اسی طرح تقویٰ کے بغیر دل مخلوق کے در پر بھیک مانگتا ہے، اس کی تعریف اور توجہ کا محتاج رہتا ہے۔ تقویٰ والا فقیر ہے مگر صرف اس کا، اور یہی محتاجی اسے سب سے بے نیاز کر دیتی ہے۔

پھر کمائی اور عرفات کا ذکر آتا ہے: لَيْسَ عَلَيْكُمْ جُنَاحٌ اَنْ تَبْتَغُوْا فَضْلًا مِّنْ رَّبِّكُمْ ۚ فَاِذَآ اَفَضْتُمْ مِّنْ عَرَفٰتٍ فَاذْكُرُوا اللّٰهَ عِنْدَ الْمَشْعَرِ الْحَرَامِ ۚ وَاذْكُرُوْهُ كَمَا هَدٰىكُمْ ۚ وَاِنْ كُنْتُمْ مِّنْ قَبْلِهٖ لَمِنَ الضَّآلِّيْنَ۔ تم پر کوئی گناہ نہیں کہ اپنے رب کا فضل تلاش کرو؛ پھر جب تم عرفات سے لوٹو تو مشعرِ حرام کے پاس اللہ کو یاد کرو، اور اسے اس طرح یاد کرو جس طرح اس نے تمہیں ہدایت دی، اور بے شک تم اس سے پہلے راہ سے بھٹکے ہوئے تھے۔ ثُمَّ اَفِيْضُوْا مِنْ حَيْثُ اَفَاضَ النَّاسُ وَاسْتَغْفِرُوا اللّٰهَ ۚ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِيْمٌ۔ پھر تم بھی وہیں سے لوٹو جہاں سے سب لوگ لوٹتے ہیں، اور اللہ سے بخشش مانگو؛ بے شک اللہ بخشنے والا مہربان ہے۔ صحابہ حج کے موسم میں تجارت سے جھجکتے تھے، تو بتایا گیا کہ حلال کمائی بھی رب کا فضل ہے۔ اور قریش خود کو اہلِ حرم سمجھ کر عرفات نہ جاتے، مزدلفہ ہی میں ٹھہر جاتے؛ حکم ہوا کہ سب کے ساتھ وہیں سے لوٹو جہاں سے سب لوٹتے ہیں (بخاری، بمفہوم)۔ حرم میں نسب کا کوئی امتیاز نہیں۔ اہلِ معرفت کہتے ہیں کہ عرفات معرفت کا میدان ہے، اور حدیث کا مفہوم ہے کہ حج عرفہ ہی ہے (ترمذی)؛ اور افاضہ کے معنی چھلک کر بہنا ہیں۔ عارف معرفت کے مقام پر رک نہیں جاتا، اس سے ذکر کی طرف اور خلق کی طرف بہتا ہے؛ یہی فیض ہے۔ اور "كَمَا هَدٰىكُمْ" بتاتا ہے کہ ذکر بھی اسی صورت میں ہو جو اس نے سکھائی، اور اسی پیمانے پر جو ہدایت اس نے دی۔ "وَاِنْ كُنْتُمْ مِّنْ قَبْلِهٖ لَمِنَ الضَّآلِّيْنَ" اسی اصول کی تصدیق ہے کہ ہدایت عطا ہے جو بلانے والے کی پکار پر رخ کرنے والے کو ملی۔ اور سب سے بڑی عبادت کے بعد استغفار کا حکم اس لیے کہ "میں نے کیا" کا پردہ سب سے آخر میں اٹھتا ہے؛ مغفرت کے معنی ڈھانپنا ہیں، اور عارف اپنے عمل کو اپنی نظر سے ڈھانپے جانے کی دعا مانگتا ہے۔

مناسک پورے ہوئے تو ذکر کا معیار بدلا: فَاِذَا قَضَيْتُمْ مَّنَاسِكَكُمْ فَاذْكُرُوا اللّٰهَ كَذِكْرِكُمْ اٰبَآءَكُمْ اَوْ اَشَدَّ ذِكْرًا ۚ فَمِنَ النَّاسِ مَنْ يَّقُوْلُ رَبَّنَآ اٰتِنَا فِي الدُّنْيَا وَمَا لَهٗ فِي الْاٰخِرَةِ مِنْ خَلَاقٍ۔ پھر جب تم اپنے حج کے ارکان پورے کر چکو تو اللہ کو اس طرح یاد کرو جیسے اپنے باپ دادا کو یاد کرتے تھے، بلکہ اس سے بھی بڑھ کر؛ پھر لوگوں میں کوئی ایسا ہے جو کہتا ہے: اے ہمارے رب! ہمیں دنیا ہی میں دے دے، اور آخرت میں اس کا کوئی حصہ نہیں۔ مفسرین بتاتے ہیں کہ جاہلیت میں حج کے بعد منیٰ میں لوگ اپنے باپ دادا کے کارنامے گنواتے اور فخر کرتے تھے؛ قرآن نے اس محفل کو ذکرِ الٰہی کی محفل بنا دیا۔ اہلِ معرفت کہتے ہیں کہ باپ دادا کا ذکر اصل کا ذکر ہے، اور ہر وجود کی اصل اصلی وہ ہے جس کے علم میں اس کا عینِ ثابتہ تھا؛ تو جو ظاہری اصلوں کو یاد کرتا ہے اسے حقیقی اصل کو اس سے زیادہ یاد کرنا چاہیے۔ پھر آیت مانگنے والوں کی دو قسمیں دکھاتی ہے۔ پہلا مانگنے والا غلط دروازے پر نہیں گیا، رب ہی سے مانگا؛ مگر کم مانگا۔ دعا کا افق دل کے افق کی خبر دیتا ہے، اور ہر بندہ اپنی استعداد کے مطابق سوال کرتا ہے۔ اور "وَمَا لَهٗ فِي الْاٰخِرَةِ مِنْ خَلَاقٍ" اسی اصول کا اظہار ہے جس پر یہ سلسلہ قائم ہے: اسے آخرت کا حصہ اس لیے نہ ملا کہ اس نے اسے طلب ہی نہ کیا، اور حق نے اسی رخ کو پورا کیا جو بندے نے خود چنا۔ کسی دروازے کو اس پر بند نہیں کیا گیا تھا۔

اب وہ جامع دعا آتی ہے جو امت کی زبان پر سب سے زیادہ جاری ہے: وَمِنْهُمْ مَّنْ يَّقُوْلُ رَبَّنَآ اٰتِنَا فِي الدُّنْيَا حَسَنَةً وَّفِي الْاٰخِرَةِ حَسَنَةً وَّقِنَا عَذَابَ النَّارِ۔ اور ان میں سے کوئی ایسا ہے جو کہتا ہے: اے ہمارے رب! ہمیں دنیا میں بھی بھلائی دے اور آخرت میں بھی بھلائی دے، اور ہمیں آگ کے عذاب سے بچا۔ حضرت انس رضی اللہ عنہ سے روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم اکثر یہی دعا مانگا کرتے تھے (بخاری و مسلم)؛ اور خود حضرت انس رضی اللہ عنہ سے دعا کی درخواست کی جاتی تو یہی دعا کرتے اور فرماتے کہ اس میں سب کچھ آ گیا۔ مفسرین نے دنیا کی بھلائی میں علم، عبادت کی توفیق، عافیت، حلال رزق اور نیک رفیقِ حیات گنوائے ہیں، اور آخرت کی بھلائی میں مغفرت، آسان حساب اور جنت۔ مگر آیت نے "حَسَنَةً" کو نکرہ رکھا، کسی ایک چیز میں قید نہیں کیا؛ اہلِ ذوق کہتے ہیں کہ یہی دعا کا ادب ہے: بندہ بھلائی مانگے اور اس کی تعیین مانگے ہوئے رب پر چھوڑ دے، کیونکہ وہ جانتا ہے کہ کس کے لیے کیا بھلا ہے۔ جو اپنی بھلائی کی صورت خود متعین کر کے مانگتا ہے، وہ کبھی وہ چیز مانگ بیٹھتا ہے جو اس کے حق میں بھلی نہیں۔ اور پہلی قسم کے مانگنے والے کے مقابلے میں اس دعا کی خوبی یہ نہیں کہ اس نے دنیا چھوڑ دی؛ اس نے دنیا بھی مانگی، مگر اسے آخرت سے کاٹ کر نہیں مانگا۔ یہ ترکِ دنیا کی دعا نہیں، توازن کی دعا ہے۔

شیخِ اکبر کے مکتب میں "حَسَنَةً" حسن سے ہے، اور حسن وہی ہے جس کی طرف احسان کی حدیث اشارہ کرتی ہے: شہود کی آنکھ سے دیکھا گیا ہر حال حسین ہے۔ دنیا کی بھلائی یہ ہے کہ بندہ دنیا کو اسمائے الٰہیہ کا مظہر دیکھے، اس سے نفرت بھی نہ کرے اور اس میں ڈوب بھی نہ جائے؛ اور آخرت کی بھلائی یہ ہے کہ وہی دید وہاں کامل ہو۔ انسانِ کامل دونوں عالموں کے درمیان برزخ ہے، اور اس کی دعا بھی دونوں کو جوڑتی ہے؛ اسی لیے یہ دعا حضورِ اکرم صلی اللہ علیہ وسلم کی زبان پر سب سے زیادہ تھی، جن میں جامعیت اپنے کمال پر تھی۔ اور یہ مشہور قول کہ دنیا آخرت کی کھیتی ہے، اسی جامعیت کی ترجمانی کرتا ہے۔ "وَّقِنَا عَذَابَ النَّارِ" میں آگ کا عذاب حقیقی اور ظاہری ہے، اس پر ایمان لازم ہے؛ اور اس کے ساتھ اہلِ معرفت کہتے ہیں کہ سب سے جلانے والی آگ حجاب کی آگ ہے، یعنی محبوب سے پردے میں رہ جانا، جیسا سورۂ المطففین میں فرمایا: كَلَّآ اِنَّهُمْ عَنْ رَّبِّهِمْ يَوْمَئِذٍ لَّمَحْجُوْبُوْنَ۔ تو عارف کی زبان پر "وَّقِنَا" دوہری پناہ ہے: آگ سے بھی اور اس دوری سے بھی جو آگ سے زیادہ تکلیف دہ ہے۔ اور دعا کی ترتیب پر غور کرو: دنیا، پھر آخرت، پھر بچاؤ؛ گویا بندہ راستہ مانگتا ہے، منزل مانگتا ہے، اور راستے کے خطرے سے پناہ مانگتا ہے۔

پھر اس دعا کا ثمر اور منیٰ کے دن: اُولٰۤئِكَ لَهُمْ نَصِيْبٌ مِّمَّا كَسَبُوْا ۚ وَاللّٰهُ سَرِيْعُ الْحِسَابِ۔ یہی لوگ ہیں جن کے لیے ان کی کمائی میں سے حصہ ہے، اور اللہ جلد حساب لینے والا ہے۔ وَاذْكُرُوا اللّٰهَ فِيْٓ اَيَّامٍ مَّعْدُوْدٰتٍ ۚ فَمَنْ تَعَجَّلَ فِيْ يَوْمَيْنِ فَلَآ اِثْمَ عَلَيْهِ ۚ وَمَنْ تَاَخَّرَ فَلَآ اِثْمَ عَلَيْهِ ۙ لِمَنِ اتَّقٰى ۚ وَاتَّقُوا اللّٰهَ وَاعْلَمُوْٓا اَنَّكُمْ اِلَيْهِ تُحْشَرُوْنَ۔ اور گنتی کے چند دنوں میں اللہ کو یاد کرو؛ پھر جو دو دن میں جلدی کر لے اس پر کوئی گناہ نہیں، اور جو ٹھہر جائے اس پر بھی کوئی گناہ نہیں، اس کے لیے جو تقویٰ رکھے؛ اور اللہ سے ڈرو اور جان لو کہ تم اسی کی طرف جمع کیے جاؤ گے۔ "مِمَّا كَسَبُوْا" میں لطیف نکتہ ہے کہ دعا خود کمائی بنی؛ مانگنا بھی عمل ہے اور اس کا بھی حصہ ملتا ہے۔ "سَرِيْعُ الْحِسَابِ" بتاتا ہے کہ جو ایک ہی آن میں سب کو رزق دیتا ہے، وہ ایک ہی آن میں سب کا حساب بھی لے سکتا ہے۔ گنتی کے دن ایامِ تشریق ہیں، جن میں تکبیر اور رمی ہے؛ بارہ ذوالحجہ کو لوٹ جانے والا بھی بے گناہ، تیرہ تک ٹھہرنے والا بھی بے گناہ۔ اہلِ معرفت یہاں شریعت کی وسعت دیکھتے ہیں: رخصت اور عزیمت دونوں اسی کی طرف سے ہیں، صورتیں الگ اور روح ایک؛ فیصلہ کن بات "لِمَنِ اتَّقٰى" ہے، نہ کہ ایک دن کم یا زیادہ۔ اور حج کا یہ بیان "تُحْشَرُوْنَ" پر ختم ہوا، کیونکہ میدانِ عرفات کا وہ ہجوم، ایک لباس میں، بغیر امتیاز کے، حشر کے میدان کی پیشگی تصویر ہے۔

اب رکوع حاجیوں کی صف سے ہٹ کر انسانوں کی دو قسموں کی طرف آتا ہے: وَمِنَ النَّاسِ مَنْ يُّعْجِبُكَ قَوْلُهٗ فِي الْحَيٰوةِ الدُّنْيَا وَيُشْهِدُ اللّٰهَ عَلٰى مَا فِيْ قَلْبِهٖ ۙ وَهُوَ اَلَدُّ الْخِصَامِ۔ اور لوگوں میں کوئی ایسا ہے جس کی بات دنیا کی زندگی میں تمہیں بھلی لگتی ہے، اور وہ اپنے دل کی بات پر اللہ کو گواہ بناتا ہے، حالانکہ وہ سخت جھگڑالو ہے۔ وَاِذَا تَوَلّٰى سَعٰى فِي الْاَرْضِ لِيُفْسِدَ فِيْهَا وَيُهْلِكَ الْحَرْثَ وَالنَّسْلَ ۚ وَاللّٰهُ لَا يُحِبُّ الْفَسَادَ۔ اور جب وہ پیٹھ پھیرتا ہے تو زمین میں دوڑ دھوپ کرتا ہے کہ اس میں فساد پھیلائے اور کھیتی اور نسل کو برباد کرے، اور اللہ فساد کو پسند نہیں کرتا۔ وَاِذَا قِيْلَ لَهُ اتَّقِ اللّٰهَ اَخَذَتْهُ الْعِزَّةُ بِالْاِثْمِ فَحَسْبُهٗ جَهَنَّمُ ۚ وَلَبِئْسَ الْمِهَادُ۔ اور جب اس سے کہا جائے کہ اللہ سے ڈرو تو غرور اسے گناہ پر اکسا دیتا ہے؛ تو اس کے لیے جہنم کافی ہے، اور وہ بہت برا ٹھکانا ہے۔ بعض مفسرین نے اسے ایک خاص شخص کے بارے میں بتایا ہے جو میٹھی باتیں کرتا اور پھر کھیتیاں اور مویشی تباہ کرتا، مگر حکم اس وصف کا ہے، کسی قوم یا گروہ پر چسپاں کرنے کا نہیں۔ یہ آیات اسی اصول کو کھول دیتی ہیں: پکار آئی، "اتَّقِ اللّٰهَ"، اور اس نے خود منہ پھیرا؛ "اَخَذَتْهُ الْعِزَّةُ بِالْاِثْمِ" میں پکڑنے والی چیز اس کی اپنی جھوٹی عزت ہے، اور "فَحَسْبُهٗ" بتاتا ہے کہ انجام اسی رخ کا پورا ہونا ہے جو اس نے چنا۔ اہلِ معرفت کہتے ہیں کہ زبان اور دل کا فاصلہ نفاق ہے، اور سچی عزت صرف اللہ کی اور اس کی نسبت سے ہے، جیسا سورۂ المنافقون میں سنا: وَلِلّٰهِ الْعِزَّةُ وَلِرَسُوْلِهٖ وَلِلْمُؤْمِنِيْنَ۔ جھوٹی عزت نفس ہے جس نے ادھار کا نام اوڑھ لیا۔ اور ہر دل کے اندر ایک "اَلَدُّ الْخِصَامِ" چھپا ہے؛ سچائی کی پہچان یہ ہے کہ جب کوئی "اللہ سے ڈرو" کہے تو گردن جھک جائے، اکڑ نہ جائے۔

دوسری قسم پہلی کا عکس ہے: وَمِنَ النَّاسِ مَنْ يَّشْرِيْ نَفْسَهُ ابْتِغَآءَ مَرْضَاتِ اللّٰهِ ۚ وَاللّٰهُ رَءُوْفٌ بِالْعِبَادِ۔ اور لوگوں میں کوئی ایسا ہے جو اللہ کی رضا کی طلب میں اپنی جان بیچ دیتا ہے، اور اللہ بندوں پر نہایت شفیق ہے۔ روایات میں آتا ہے کہ حضرت صہیب رومی رضی اللہ عنہ ہجرت کو نکلے تو قریش نے روکا؛ انہوں نے اپنا سارا مال ان کے حوالے کر دیا کہ راستہ چھوڑ دو، اور جب مدینہ پہنچے تو حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم ہے: سودا نفع کا رہا۔ بعض مفسرین نے ہجرت کی رات حضرت علی رضی اللہ عنہ کے حضورِ اکرم صلی اللہ علیہ وسلم کے بستر پر سونے کا بھی ذکر کیا ہے؛ اور حکم ہر اس شخص کو شامل ہے جو حق کی خاطر اپنی جان اور مال کھپا دے۔ دونوں قسموں کو آمنے سامنے رکھو: پہلا آخرت کو باتوں کے بدلے بیچتا ہے، دوسرا اپنی جان کو رضا کے بدلے؛ پہلے کی بات بھلی اور عمل برا، دوسرے کی بات کم اور عمل کامل۔ پہلے کو "اللہ سے ڈرو" کہا گیا تو اکڑ گیا، دوسرے کو کسی نے کچھ نہ کہا اور وہ خود سب کچھ دے بیٹھا۔ "يَشْرِيْ" عربی کے ان الفاظ میں سے ہے جو خریدنے اور بیچنے دونوں کے لیے آتے ہیں؛ اہلِ ذوق کہتے ہیں کہ اس سودے میں دونوں معنی سچ ہیں: وہ جان بیچتا ہے اور رضا خریدتا ہے۔ اور قیمت "مَرْضَاتِ اللّٰهِ" رکھی گئی، جنت کا نام بھی نہیں لیا؛ یہ اس عاشق کا سودا ہے جو عطا سے آگے عطا کرنے والے کی رضا چاہتا ہے۔

اس آیت کا راز اہلِ معرفت کے نزدیک یہ ہے کہ بندہ جو جان بیچتا ہے وہ حقیقت میں اس کی ملکیت تھی ہی نہیں؛ حقیقی مالک وہی ہے، اور بندے کے پاس امانت تھی۔ پھر بھی حق نے اسے سودا کہا اور قیمت ادا کی، جیسا سورۂ التوبہ میں فرمایا: اِنَّ اللّٰهَ اشْتَرٰى مِنَ الْمُؤْمِنِيْنَ اَنْفُسَهُمْ وَاَمْوَالَهُمْ بِاَنَّ لَهُمُ الْجَنَّةَ۔ اپنی ہی دی ہوئی چیز کو خریدنا اور اس کی قیمت بھی دینا، یہی "رَءُوْفٌ بِالْعِبَادِ" کی تفسیر ہے؛ آیت کا خاتمہ کوئی الگ جملہ نہیں، سودے کی شرح ہے۔ شیخِ اکبر کے مکتب میں جان بیچنا فنا ہے: "میں" کا دعویٰ چھوڑ دینا اور یہ پہچان لینا کہ وجود اس کا ہے۔ اور اس سودے کے بعد بقا ہے: جان واپس ملتی ہے، مگر اب امانت کی حیثیت سے، اپنی ملکیت کے طور پر نہیں؛ اب بندہ جو کرتا ہے اسی کی رضا کی طلب میں کرتا ہے۔ اہلِ ذوق کہتے ہیں کہ جس نے جان بیچ دی اس کے پاس ڈرنے کو کچھ نہیں بچتا، کیونکہ جو چیز بک چکی اس کے چھن جانے کا خوف کیسا۔ اسی لیے ایسے لوگ نڈر بھی ہوتے ہیں اور نرم بھی۔ یہ ظاہر کی نفی نہیں: ظاہر میں یہ ہجرت، قربانی اور حق کی خاطر نقصان سہنا ہے، اور باطن میں اسی عمل کی روح، کہ بندہ اپنے آپ سے نکل کر اپنے رب کا ہو جائے۔

پھر پوری امت کو ایک جامع حکم ملا: يٰٓاَيُّهَا الَّذِيْنَ اٰمَنُوا ادْخُلُوْا فِي السِّلْمِ كَآفَّةً ۚ وَّلَا تَتَّبِعُوْا خُطُوٰتِ الشَّيْطٰنِ ۚ اِنَّهٗ لَكُمْ عَدُوٌّ مُّبِيْنٌ۔ اے ایمان والو! اسلام میں پورے کے پورے داخل ہو جاؤ، اور شیطان کے قدموں کی پیروی نہ کرو؛ بے شک وہ تمہارا کھلا دشمن ہے۔ بعض مفسرین نے بیان کیا کہ اہلِ کتاب میں سے اسلام لانے والے بعض حضرات اپنی سابقہ شریعت کی بعض پابندیوں کو بھی باقی رکھنا چاہتے تھے؛ تو حکم ہوا کہ اسلام میں پورے داخل ہو۔ "كَآفَّةً" کے دو معنی کیے گئے ہیں: تم سب کے سب داخل ہو، یا اسلام کے سب احکام میں داخل ہو؛ اور دونوں معنی ایک دوسرے کو مکمل کرتے ہیں۔ اسلام کو اپنی پسند کے حصوں میں بانٹ لینا، جو آسان لگے لے لینا اور جو بھاری لگے چھوڑ دینا، داخل ہونا نہیں، دروازے پر کھڑے رہنا ہے۔ اور "خُطُوٰتِ الشَّيْطٰنِ" کا لفظ بڑا گہرا ہے: شیطان کبھی ایک چھلانگ کی دعوت نہیں دیتا، ایک ایک قدم کی دعوت دیتا ہے۔ کوئی پہلے ہی دن بڑا گناہ نہیں کرتا؛ ایک چھوٹی رخصت، پھر ایک تاویل، پھر ایک عادت۔ اہلِ ذوق کہتے ہیں کہ سالک کی لغزش ہمیشہ ایک چھوٹے قدم سے شروع ہوتی ہے جسے وہ قدم ہی نہیں سمجھتا۔ اور "عَدُوٌّ مُّبِيْنٌ" یاد دلاتا ہے کہ اس کی دشمنی کھلی ہے، چھپی ہوئی صرف اس کی چال ہے۔

شیخِ اکبر کے مکتب میں "السِّلْمِ" اسلام بھی ہے اور سلامتی بھی، اور "كَآفَّةً" کا اشارہ یہ ہے کہ انسان اپنے سارے وجود کے ساتھ داخل ہو: بدن شریعت میں، نفس ریاضت میں، دل توحید میں، روح محبت میں، اور سرّ شہود میں۔ جو صرف زبان سے داخل ہوا، وہ اسی شخص کی طرح ہے جس کی بات بھلی لگتی تھی اور دل کہیں اور تھا۔ اور سلامتی کے معنی میں داخل ہونا یہ ہے کہ بندہ اپنے رب کی تقدیر سے صلح کر لے، اس کی مخلوق سے صلح کر لے، اور اپنے نفس کی کشمکش سے نکل کر اطمینان میں آ جائے؛ السلام اسی کا نام ہے، اور جو اس میں داخل ہوا وہ اس اسم کا مظہر بنا۔ پھر تنبیہ آئی: فَاِنْ زَلَلْتُمْ مِّنْۢ بَعْدِ مَا جَآءَتْكُمُ الْبَيِّنٰتُ فَاعْلَمُوْٓا اَنَّ اللّٰهَ عَزِيْزٌ حَكِيْمٌ۔ پھر اگر تم واضح دلیلیں آ جانے کے بعد بھی پھسل جاؤ تو جان لو کہ اللہ غالب، حکمت والا ہے۔ "مِّنْۢ بَعْدِ مَا جَآءَتْكُمُ الْبَيِّنٰتُ" بتاتا ہے کہ لغزش دلیل کی کمی سے نہیں، دلیل آ جانے کے بعد بندے کے اپنے قدم سے ہے؛ حجت پوری ہو چکی۔ "عَزِيْزٌ" یعنی اس کی گرفت سے کوئی نہیں نکلتا، اور "حَكِيْمٌ" یعنی اس کا ہر فیصلہ حکمت سے ہے، وہ بے موقع جلدی نہیں کرتا؛ اس کی حکمت ہی پھسلنے والے کو مہلت دیتی ہے کہ لوٹ آئے۔ تو یہ آیت دھمکی بھی ہے اور کھلا دروازہ بھی۔

رکوع ایک بڑے سوال پر ختم ہوتا ہے: هَلْ يَنْظُرُوْنَ اِلَّآ اَنْ يَّاْتِيَهُمُ اللّٰهُ فِيْ ظُلَلٍ مِّنَ الْغَمَامِ وَالْمَلٰۤئِكَةُ وَقُضِيَ الْاَمْرُ ۚ وَاِلَى اللّٰهِ تُرْجَعُ الْاُمُوْرُ۔ کیا یہ لوگ اسی کا انتظار کر رہے ہیں کہ اللہ بادلوں کے سائبانوں میں ان کے پاس آ جائے اور فرشتے بھی، اور فیصلہ ہی چکا دیا جائے؟ اور سب معاملات اللہ ہی کی طرف لوٹائے جاتے ہیں۔ اہلِ سنت کا طریقہ یہ ہے کہ ایسی آیات پر جیسی آئیں ویسا ایمان لایا جائے، کیفیت بیان کیے بغیر اور مخلوق سے تشبیہ دیے بغیر؛ حق ایک جگہ سے دوسری جگہ منتقل ہونے، جسم اور سمت سے پاک ہے۔ بعض مفسرین نے اس کی تاویل اس کے حکم اور فیصلے کے آنے سے کی ہے۔ شیخِ اکبر کے مکتب میں یہ قیامت کے دن کی تجلی ہے، اور حدیث کا مفہوم ہے کہ اس دن حق اپنی تجلی ایسی صورت میں فرمائے گا جسے لوگ پہچانیں گے (بخاری و مسلم)؛ مگر تنزیہ اپنی جگہ قائم ہے: صورت تجلی کا مظہر ہے، اس کی قید نہیں، اور جو کچھ ذہن میں آئے حق اس سے ورا ہے۔ تنزیہ اور تشبیہ کا یہی جمع اس مکتب کا امتیاز ہے۔ اور آیت کا سوال تنبیہ ہے: جو آج دلیل پر نہیں مانتا وہ کس چیز کا منتظر ہے؟ جب پردہ اٹھ جائے گا تو ماننا اختیار نہیں رہے گا، فیصلہ ہو چکا ہوگا۔ ایمان کی قیمت اسی لیے ہے کہ وہ غیب میں ہے۔ اور "وَاِلَى اللّٰهِ تُرْجَعُ الْاُمُوْرُ" پر یہ رکوع وہیں جا ٹھہرا جہاں حج کا بیان "تُحْشَرُوْنَ" پر ٹھہرا تھا: ہر سفر، ہر سودا، ہر لغزش اور ہر معاملہ آخرکار اسی کی طرف لوٹتا ہے۔
