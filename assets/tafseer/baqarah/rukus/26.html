پچھلے رکوع کی آخری پکار یہ تھی کہ ایمان والے پورے کے پورے سلامتی میں داخل ہو جائیں، اور یہ رکوع اسی پکار کو تاریخ کے آئینے میں رکھ کر شروع ہوتا ہے: سَلْ بَنِيْۤ اِسْرَآءِيْلَ كَمْ اٰتَيْنٰهُمْ مِّنْ اٰيَةٍۭ بَيِّنَةٍ ؕ وَمَنْ يُّبَدِّلْ نِعْمَةَ اللّٰهِ مِنْۢ بَعْدِ مَا جَآءَتْهُ فَاِنَّ اللّٰهَ شَدِيْدُ الْعِقَابِ۔ “بنی اسرائیل سے پوچھ لیجیے کہ ہم نے انہیں کتنی روشن نشانیاں عطا کیں؛ اور جو کوئی اللہ کی نعمت کو اس کے پاس آ جانے کے بعد بدل ڈالے تو بے شک اللہ سخت سزا دینے والا ہے۔” مفسرین کے نزدیک یہاں ایک خاص زمانے کی ایک خاص جماعت کے طرزِ عمل کی طرف اشارہ ہے، جس نے حضرت موسیٰ علیہ السلام کے ہاتھوں سمندر کا پھٹنا، من و سلویٰ اور پتھر سے پھوٹتے چشمے دیکھے۔ اس سے آج کی کسی قوم پر کوئی عمومی حکم نہیں نکلتا؛ سوال کا اصل رخ سننے والے کی طرف ہے کہ وہ اپنی نعمتوں کا حساب کرے۔ نعمت کو بدلنا یہ ہے کہ جو چیز شکر کے لیے ملی تھی اسے ناشکری، غرور یا غفلت کا سامان بنا لیا جائے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر نشانی حق کی ایک تجلی ہے، اور تجلی جس دل پر پڑتی ہے اس کی استعداد کے مطابق اسے روشن بھی کرتی ہے اور اس پر حجت بھی قائم کرتی ہے۔ اہلِ معرفت کہتے ہیں کہ سب سے بڑی نعمت ہدایت ہے، اور اسے بدلنے کی سب سے باریک صورت یہ ہے کہ بندہ ملی ہوئی روشنی کو اپنی ملکیت سمجھ بیٹھے؛ اسی لمحے روشنی پردہ بن جاتی ہے۔ اور سزا کی سختی اس بدلنے کا نتیجہ ہے جو بندے نے خود کیا، کوئی ابتدائی قہر نہیں۔ نعمت پہلے آئی، بدلنا بعد میں ہوا، اور حکم اسی ترتیب کے مطابق ہے؛ یہی سرِّ قدر کی پہلی جھلک ہے۔

پھر وہ نقشہ دکھایا گیا جو ہر زمانے میں دہرایا جاتا ہے: زُيِّنَ لِلَّذِيْنَ كَفَرُوا الْحَيٰوةُ الدُّنْيَا وَيَسْخَرُوْنَ مِنَ الَّذِيْنَ اٰمَنُوْا ۘ وَالَّذِيْنَ اتَّقَوْا فَوْقَهُمْ يَوْمَ الْقِيٰمَةِ ؕ وَاللّٰهُ يَرْزُقُ مَنْ يَّشَآءُ بِغَيْرِ حِسَابٍ۔ “انکار کرنے والوں کے لیے دنیا کی زندگی خوش نما بنا دی گئی ہے، اور وہ ایمان والوں کا مذاق اڑاتے ہیں؛ حالانکہ تقویٰ والے قیامت کے دن ان سے بلند ہوں گے، اور اللہ جسے چاہتا ہے بے حساب رزق دیتا ہے۔” مفسرین بیان کرتے ہیں کہ مکہ کے سردار نادار صحابۂ کرام رضی اللہ عنہم کی تنگ دستی پر ہنستے تھے اور اپنی دولت کو اپنے حق پر ہونے کی دلیل سمجھتے تھے۔ آراستگی کا فعل مجہول آیا ہے۔ اہلِ تفسیر کہتے ہیں کہ زینت دنیا کی ساخت میں آزمائش کے لیے رکھی گئی، جیسا کہ فرمایا: اِنَّا جَعَلْنَا مَا عَلَى الْاَرْضِ زِيْنَةً لَّهَا لِنَبْلُوَهُمْ؛ اور اسی زینت کو دل کا مقصود بنا دینا نفس کی پسند اور شیطان کا وسوسہ ہے۔ زینت سب کے سامنے ایک ہی تھی، فرق دیکھنے والے کے رخ کا ہے۔ شیخِ اکبر کے نزدیک عالم حق کے اسماء کا مظہر ہے؛ جو اسے آئینہ سمجھ کر دیکھے اسے اس میں جمال کی جھلک ملتی ہے، اور جو اسے منزل بنا لے اس کے لیے وہی آئینہ دیوار بن جاتا ہے۔ “فَوْقَهُمْ” کی بلندی ظاہری دولت کی نہیں، حقیقت کی ہے، جو اس دن کھلے گی جب ہر چیز اپنی اصل صورت میں ظاہر ہوگی۔ اور “بِغَيْرِ حِسَابٍ” بتاتا ہے کہ اس کی عطا بندے کے ترازو سے نہیں تلتی؛ اہلِ ذوق کہتے ہیں کہ بے حساب رزق میں سب سے لطیف رزق معرفت ہے، جس کا کوئی پیمانہ نہیں۔

اب وہ آیت آتی ہے جو انسانی تاریخ کو ایک جملے میں سمیٹتی ہے اور ہدایت و گمراہی کا اصول بھی کھول دیتی ہے: كَانَ النَّاسُ اُمَّةً وَّاحِدَةً ۚ فَبَعَثَ اللّٰهُ النَّبِيّٖنَ مُبَشِّرِيْنَ وَمُنْذِرِيْنَ ۙ وَاَنْزَلَ مَعَهُمُ الْكِتٰبَ بِالْحَقِّ لِيَحْكُمَ بَيْنَ النَّاسِ فِيْمَا اخْتَلَفُوْا فِيْهِ ؕ وَمَا اخْتَلَفَ فِيْهِ اِلَّا الَّذِيْنَ اُوْتُوْهُ مِنْۢ بَعْدِ مَا جَآءَتْهُمُ الْبَيِّنٰتُ بَغْيًۢا بَيْنَهُمْ۔ “سب لوگ ایک ہی امت تھے، پھر اللہ نے نبیوں کو خوش خبری دینے والے اور ڈرانے والے بنا کر بھیجا، اور ان کے ساتھ حق پر مبنی کتاب اتاری تاکہ وہ لوگوں کے درمیان ان باتوں کا فیصلہ کرے جن میں انہوں نے اختلاف کیا؛ اور اس میں اختلاف انہی لوگوں نے کیا جنہیں وہ کتاب دی گئی تھی، روشن دلیلیں آ جانے کے بعد، محض آپس کی سرکشی سے۔” حضرت عبداللہ بن عباس رضی اللہ عنہما سے منقول قول کا مفہوم یہ ہے کہ حضرت آدم اور حضرت نوح علیہما السلام کے درمیان کئی نسلیں حق کی شریعت پر رہیں، پھر اختلاف پیدا ہوا تو انبیاء علیہم السلام بھیجے گئے۔ آیت نے اختلاف کی جڑ صاف بتا دی: نہ دلیل کی کمی تھی، نہ روشنی کی؛ سبب “بَغْيًۢا بَيْنَهُمْ” تھا، یعنی ایک دوسرے پر بڑائی چاہنے کی سرکشی۔ اہلِ معرفت کے نزدیک وہ پہلی وحدت ہر دل میں اب بھی محفوظ ہے، کیونکہ وہی فِطْرَتَ اللّٰهِ الَّتِيْ فَطَرَ النَّاسَ عَلَيْهَا ہے۔ شیخِ اکبر کے مکتب میں کثرت بذاتِ خود عیب نہیں، کیونکہ اسماءِ الٰہیہ کی کثرت ہی سے عالم کا رنگا رنگ ظہور ہے؛ عیب وہ اختلاف ہے جو حق کو پہچان لینے کے بعد انا کی خاطر کیا جائے۔ مظاہر کا فرق رحمت ہے، اور دل کا سرکشی پر اڑ جانا حجاب۔

اسی آیت کا دوسرا حصہ وہ اصول ہے جس پر یہ پورا سلسلہ قائم ہے، اس لیے اسے ٹھہر کر پڑھنا چاہیے: فَهَدَى اللّٰهُ الَّذِيْنَ اٰمَنُوْا لِمَا اخْتَلَفُوْا فِيْهِ مِنَ الْحَقِّ بِاِذْنِهٖ ؕ وَاللّٰهُ يَهْدِيْ مَنْ يَّشَآءُ اِلٰى صِرَاطٍ مُّسْتَقِيْمٍ۔ “پھر اللہ نے اپنے حکم سے ایمان والوں کو اس حق کی راہ دکھا دی جس میں لوگوں نے اختلاف کیا تھا؛ اور اللہ جسے چاہتا ہے سیدھی راہ کی طرف ہدایت دیتا ہے۔” ترتیب پر غور کیجیے: روشن دلیلیں سب کے پاس آئیں؛ ایک گروہ نے سرکشی کو چنا اور اختلاف میں پڑا، دوسرے نے ایمان کو چنا اور اسے ہدایت ملی۔ ہدایت ان کو ملی جو اس کی طرف بڑھے، اور جو منہ پھیر گیا اس کا منہ پھیرنا ہی اس کی گمراہی تھی۔ یہی وہ اصول ہے جسے یہ سورت خود بیان کرتی ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور کسی ہدایت کے طالب کو گمراہ نہیں کرتا۔ “مَنْ يَّشَآءُ” میں اس کی مشیت حکمت والی ہے، اندھی نہیں، اور اس کی حکمت یہ ہے کہ طلب کرنے والا محروم نہ رہے؛ سورۂ العنکبوت کے صفحے پر یہ وعدہ گزر چکا: وَالَّذِيْنَ جَاهَدُوْا فِيْنَا لَنَهْدِيَنَّهُمْ سُبُلَنَا۔ شیخِ اکبر کے نزدیک یہ سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے، یعنی حق ہر چیز کو وہی وجود دیتا ہے جو اس کی اپنی حقیقت اور اس کے اپنے رخ کا تقاضا ہے۔ بندہ جس سمت کو چنتا ہے، حق اسی سمت کو وجود عطا کرتا ہے۔ اور “بِاِذْنِهٖ” یاد دلاتا ہے کہ ہدایت پانے والا بھی اپنی ہدایت کا مالک نہیں؛ طلب اس کی تھی، عطا حق کی۔ پس ایمان والا شکر کرے اور اختلاف کرنے والوں پر بڑائی نہ جتائے، کیونکہ یہی بڑائی تو وہ بیماری تھی جس نے پہلی امت کو بانٹ دیا۔

ہدایت مل جانے کے بعد کیا راستہ ہموار ہو جاتا ہے؟ اگلی آیت اس خوش فہمی کو توڑتی ہے: اَمْ حَسِبْتُمْ اَنْ تَدْخُلُوا الْجَنَّةَ وَلَمَّا يَاْتِكُمْ مَّثَلُ الَّذِيْنَ خَلَوْا مِنْ قَبْلِكُمْ ؕ مَسَّتْهُمُ الْبَاْسَآءُ وَالضَّرَّآءُ وَزُلْزِلُوْا۔ “کیا تم نے یہ سمجھ رکھا ہے کہ جنت میں داخل ہو جاؤ گے، حالانکہ ابھی تم پر ان لوگوں جیسے حالات نہیں آئے جو تم سے پہلے گزر چکے؟ انہیں سختیوں اور تکلیفوں نے آ لیا اور وہ ہلا ڈالے گئے۔” مفسرین کہتے ہیں کہ یہ آیت مدینہ کے ابتدائی دنوں کی تنگی میں صحابۂ کرام رضی اللہ عنہم کی تسلی کے لیے اتری، اور بعض نے اسے غزوۂ خندق کے حالات سے جوڑا۔ “الْبَاْسَآءُ” سے مراد مال اور معاش کی تنگی، “الضَّرَّآءُ” سے بیماری اور جسم کا دکھ، اور “وَزُلْزِلُوْا” وہ لرزہ ہے جو دل کی جڑ تک پہنچ جائے۔ اہلِ معرفت کے نزدیک یہ لرزہ بندے کو اسباب کے سہاروں سے ہلا کر مسبب الاسباب پر ٹکا دیتا ہے۔ جب تک دل کسی سبب پر ٹیک لگائے رہے، توکل زبان کی بات رہتا ہے؛ زلزلہ آتا ہے تو ہر ٹیک گرتی ہے اور وہی باقی رہتا ہے جو گرنے والا نہیں۔ شیخِ اکبر کے مکتب میں قبض و بسط دونوں تربیت کے ہاتھ ہیں: قبض میں بندہ اپنے فقر کو پہچانتا ہے اور بسط میں عطا کو۔ “مَثَلُ الَّذِيْنَ خَلَوْا” میں یہ تسلی بھی ہے کہ یہ راستہ نیا نہیں، ہر نبی اور اس کے ساتھیوں نے یہی گھاٹی پار کی۔ سورۂ الزلزال میں زمین کا لرزنا قیامت کی خبر تھا؛ یہاں دل کا لرزنا ایک نئی پیدائش کی خبر ہے، کیونکہ ہل چلائی ہوئی زمین ہی بیج قبول کرتی ہے۔

پھر اس لرزے کی انتہا بیان ہوئی: حَتّٰى يَقُوْلَ الرَّسُوْلُ وَالَّذِيْنَ اٰمَنُوْا مَعَهٗ مَتٰى نَصْرُ اللّٰهِ ؕ اَلَاۤ اِنَّ نَصْرَ اللّٰهِ قَرِيْبٌ۔ “یہاں تک کہ رسول اور ان کے ساتھ ایمان لانے والے پکار اٹھے کہ اللہ کی مدد کب آئے گی؟ سن لو، اللہ کی مدد قریب ہے۔” مفسرین واضح کرتے ہیں کہ “مَتٰى نَصْرُ اللّٰهِ” شک کا جملہ نہیں؛ رسول اور اہلِ ایمان کو وعدے پر کامل یقین تھا، سوال وقت کا تھا، وعدے کا نہیں۔ یہ ایک بے قرار دل کی دعا ہے جو اپنی بے بسی کو رب کے سامنے رکھ دیتا ہے۔ اہلِ ذوق کہتے ہیں کہ مدد اسی لمحے قریب آتی ہے جب بندہ اپنی ساری تدبیروں سے خالی ہو کر پکارتا ہے؛ اپنی طاقت پر ذرا سا بھروسا بھی باقی ہو تو پکار ادھوری رہتی ہے۔ جواب سوال کے ساتھ ہی جڑا ہوا ہے، درمیان میں کوئی فاصلہ نہیں، گویا مدد کی دوری صرف بندے کی نگاہ میں تھی۔ شیخِ اکبر کے نزدیک قرب مسافت کا نام نہیں؛ حق ہر حال میں بندے کے ساتھ ہے، اور اسی سورت نے کہا: وَاِذَا سَاَلَكَ عِبَادِيْ عَنِّيْ فَاِنِّيْ قَرِيْبٌ۔ پس مدد کا آنا دراصل پردے کا اٹھنا ہے: جو ساتھ تھا وہ دکھائی دینے لگتا ہے۔ سورۂ الانشراح کا وعدہ فَاِنَّ مَعَ الْعُسْرِ يُسْرًا بھی یہی کہتا ہے کہ آسانی تنگی کے بعد نہیں، اس کے ساتھ ہے۔ اہلِ معرفت کہتے ہیں کہ جو بندہ اس راز کو پا لے وہ سختی میں بھی شکایت نہیں کرتا، دعا کرتا ہے؛ اور دعا خود اس قرب کا پہلا پھل ہے۔

اس کے بعد سوالوں کا ایک سلسلہ شروع ہوتا ہے جو اس رکوع اور اگلے رکوعوں میں بار بار “يَسْـَٔلُوْنَكَ” کے ساتھ آتا ہے، اور پہلا سوال خرچ کا ہے: يَسْـَٔلُوْنَكَ مَاذَا يُنْفِقُوْنَ ؕ قُلْ مَاۤ اَنْفَقْتُمْ مِّنْ خَيْرٍ فَلِلْوَالِدَيْنِ وَالْاَقْرَبِيْنَ وَالْيَتٰمٰى وَالْمَسٰكِيْنِ وَابْنِ السَّبِيْلِ ؕ وَمَا تَفْعَلُوْا مِنْ خَيْرٍ فَاِنَّ اللّٰهَ بِهٖ عَلِيْمٌ۔ “لوگ آپ سے پوچھتے ہیں کہ کیا خرچ کریں؟ کہہ دیجیے: جو مال بھی تم خرچ کرو وہ ماں باپ، قرابت داروں، یتیموں، مسکینوں اور مسافروں کے لیے ہے؛ اور جو بھلائی بھی تم کرو اللہ اسے خوب جانتا ہے۔” سوال “کیا” کا تھا اور جواب “کس پر” کا آیا۔ مفسرین کہتے ہیں کہ اس میں ادب سکھایا گیا: خرچ کی چیز تو “مِّنْ خَيْرٍ” کہہ کر بتا دی کہ جو بھی پاک اور حلال ہو، اور اصل زور اس پر رکھا کہ وہ صحیح جگہ پہنچے۔ ترتیب میں ماں باپ سب سے پہلے ہیں، پھر قریب کے رشتے، پھر وہ جن کا کوئی سہارا نہیں۔ اہلِ معرفت کہتے ہیں کہ خرچ دائروں میں پھیلتا ہے، جیسے پانی میں پتھر گرنے سے لہریں: پہلے اپنی اصل، پھر اپنا خون، پھر انسانیت کی کمزور ترین کڑی۔ شیخِ اکبر کے مکتب میں انفاق تخلق باسماء اللہ کی ایک صورت ہے: بندہ اسمِ وہاب اور اسمِ رزاق کا مظہر بنتا ہے، حالانکہ دینے والا حقیقت میں وہی ہے اور بندہ صرف راستہ ہے۔ جو راستہ ہونے پر راضی ہو وہ احسان نہیں جتاتا، کیونکہ اس نے اپنا کچھ دیا ہی نہیں۔ اور “فَاِنَّ اللّٰهَ بِهٖ عَلِيْمٌ” دینے والے کو لوگوں کی داد سے بے نیاز کر دیتا ہے: جسے وہ دیکھ رہا ہے جو سب جانتا ہے، اسے کسی اور گواہ کی حاجت نہیں۔

اب وہ حکم آتا ہے جسے نفس سب سے زیادہ ناگوار جانتا ہے: كُتِبَ عَلَيْكُمُ الْقِتَالُ وَهُوَ كُرْهٌ لَّكُمْ۔ “تم پر جنگ فرض کی گئی، اور وہ تمہیں ناگوار ہے۔” اس آیت کو اس کے تاریخی پس منظر میں پڑھنا لازم ہے۔ مکہ میں تیرہ برس مسلمانوں کو ستایا گیا، ان میں سے کئی شہید کیے گئے، انہیں گھروں اور مال سے نکالا گیا، اور اس سارے عرصے میں ہاتھ روکنے اور صبر کرنے کا حکم تھا۔ مدینہ پہنچنے کے بعد بھی ان پر حملے جاری رہے، تب پہلی اجازت ان الفاظ میں آئی: اُذِنَ لِلَّذِيْنَ يُقٰتَلُوْنَ بِاَنَّهُمْ ظُلِمُوْا، یعنی اجازت ان کو جن سے لڑا جا رہا تھا، اس لیے کہ ان پر ظلم ہوا۔ پس یہ حکم دفاع کا تھا، حضورِ اکرم صلی اللہ علیہ وسلم کی قیادت میں، ایک منظم جماعت کے لیے، اپنی حدوں کے ساتھ؛ اسی سورت میں پہلے فرمایا جا چکا: وَلَا تَعْتَدُوْا ؕ اِنَّ اللّٰهَ لَا يُحِبُّ الْمُعْتَدِيْنَ۔ فقہا کے نزدیک یہ معاملہ جائز حکومت اور اس کی ذمہ داری کا ہے، کسی فرد یا گروہ کا اپنا فیصلہ نہیں، اور اس کی تفصیلات فقہ میں ہیں۔ قرآن نے انسانی فطرت کو جھٹلایا نہیں؛ صاف کہا کہ یہ تمہیں ناگوار ہے۔ اس نے جنگ کو خوشی کی چیز نہیں بنایا، ایک بوجھ کہا جو ظلم کو روکنے کے لیے اٹھایا گیا۔ اہلِ معرفت کہتے ہیں کہ یہ صاف گوئی خود ایک رحمت ہے: حق بندے کی کمزوری کو جانتا ہے، اسے چھپانے کا مطالبہ نہیں کرتا، بلکہ اس کمزوری کے ساتھ چلنے کا راستہ دکھاتا ہے۔

پھر وہ جملہ آتا ہے جو اس حکم سے آگے بڑھ کر پوری زندگی کا قاعدہ بن گیا: وَعَسٰۤى اَنْ تَكْرَهُوْا شَيْـًٔا وَّهُوَ خَيْرٌ لَّكُمْ ۚ وَعَسٰۤى اَنْ تُحِبُّوْا شَيْـًٔا وَّهُوَ شَرٌّ لَّكُمْ ؕ وَاللّٰهُ يَعْلَمُ وَاَنْتُمْ لَا تَعْلَمُوْنَ۔ “اور ہو سکتا ہے کہ تم کسی چیز کو ناپسند کرو اور وہ تمہارے لیے بہتر ہو، اور ہو سکتا ہے کہ تم کسی چیز کو پسند کرو اور وہ تمہارے لیے بری ہو؛ اور اللہ جانتا ہے، تم نہیں جانتے۔” انسان کی پسند اور ناپسند اس کے محدود علم پر قائم ہے؛ وہ لمحے کو دیکھتا ہے، انجام کو نہیں۔ اہلِ معرفت کے نزدیک یہ آیت رضا کا دروازہ ہے: بندہ اپنی ناپسند کو مٹا نہیں دیتا، مگر اسے آخری فیصلہ بھی نہیں بناتا۔ شیخِ اکبر کے مکتب میں اعیانِ ثابتہ کی بات اسی کو کھولتی ہے: ہر شے کی حقیقت حق کے علم میں ہے، اور جو کچھ پیش آتا ہے وہ اسی حقیقت کے مطابق ہے؛ بندہ اپنی عین کو پوری طرح نہیں جانتا، اس لیے اپنی بھلائی کو بھی پوری طرح نہیں جانتا۔ جلال کی تجلی کبھی جمال کو اپنے اندر چھپائے ہوتی ہے، جیسے کڑوی دوا شفا کو۔ اہلِ تصوف میں ایک قول مشہور ہے کہ ہم چھوٹے جہاد سے بڑے جہاد کی طرف لوٹے، یعنی نفس کے جہاد کی طرف؛ محدثین نے اس کی سند پر کلام کیا ہے، مگر اس کا مفہوم اہلِ سلوک کے ہاں مقبول رہا۔ نفس بھی ہر اس چیز کو ناپسند کرتا ہے جو اسے توڑ کر نیا بناتی ہے، اور ہر اس چیز کو چاہتا ہے جو اسے اس کے حال پر چھوڑ دے۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہری حکم اپنی جگہ اپنی شرطوں اور حدوں کے ساتھ ہے، اور باطنی سبق یہ ہے کہ “وَاللّٰهُ يَعْلَمُ وَاَنْتُمْ لَا تَعْلَمُوْنَ” کے سامنے دل جھک جائے اور اپنی پسند کو حق کے علم کے حوالے کر دے۔
