یہ رکوع اس صورت سے شروع ہوتا ہے جہاں رشتہ بندھا مگر گھر بسنے سے پہلے ہی ٹوٹ گیا: لَا جُنَاحَ عَلَيْكُمْ اِنْ طَلَّقْتُمُ النِّسَآءَ مَا لَمْ تَمَسُّوْهُنَّ اَوْ تَفْرِضُوْا لَهُنَّ فَرِيْضَةً ۚ وَّمَتِّعُوْهُنَّ ۚ عَلَى الْمُوْسِعِ قَدَرُهٗ وَعَلَى الْمُقْتِرِ قَدَرُهٗ ۚ مَتَاعًاۢ بِالْمَعْرُوْفِ ۚ حَقًّا عَلَى الْمُحْسِنِيْنَ۔ “تم پر کوئی گناہ نہیں اگر تم عورتوں کو ایسی حالت میں طلاق دو کہ نہ تم نے انہیں ہاتھ لگایا ہو اور نہ ان کا مہر مقرر کیا ہو؛ اور انہیں کچھ سامان دے کر رخصت کرو، خوش حال پر اس کی حیثیت کے مطابق اور تنگ دست پر اس کی حیثیت کے مطابق؛ یہ سامان معروف طریقے سے ہو، نیکی کرنے والوں پر یہ ایک حق ہے۔” عورت خالی ہاتھ اور ٹوٹے دل کے ساتھ رخصت نہ ہو؛ یہ تحفہ اس کے دل کی دلجوئی ہے اور اس بات کا اعلان کہ رشتے کا ٹوٹنا اس کی عزت کا ٹوٹنا نہیں۔ مقدار کی تفصیل فقہ کا موضوع ہے۔ معیار ہر ایک کی حیثیت رکھا گیا، تاکہ کسی پر بوجھ نہ ہو اور کوئی بخل کا بہانہ نہ ڈھونڈے۔ اہلِ معرفت کہتے ہیں کہ ہر عطا ایک اندازے سے ہے۔ حق کے خزانے بے حد ہیں، پھر بھی وہ فرماتا ہے: وَمَا نُنَزِّلُهٗۤ اِلَّا بِقَدَرٍ مَّعْلُوْمٍ۔ بندہ بھی اپنے اندازے سے دیتا ہے، اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں دینے کی قدر مقدار سے نہیں، اس احسان سے ہے جو اس کے ساتھ ہو۔ اور دیکھیے کہ ایک فیاضی کو محسنین پر حق کہہ دیا گیا: احسان کرنے والے کے لیے سخاوت اختیاری نہیں رہتی، اس کی پہچان بن جاتی ہے۔

دوسری صورت وہ ہے جس میں مہر مقرر ہو چکا تھا: وَاِنْ طَلَّقْتُمُوْهُنَّ مِنْ قَبْلِ اَنْ تَمَسُّوْهُنَّ وَقَدْ فَرَضْتُمْ لَهُنَّ فَرِيْضَةً فَنِصْفُ مَا فَرَضْتُمْ اِلَّاۤ اَنْ يَّعْفُوْنَ اَوْ يَعْفُوَا الَّذِيْ بِيَدِهٖ عُقْدَةُ النِّكَاحِ ۭ وَاَنْ تَعْفُوْۤا اَقْرَبُ لِلتَّقْوٰى۔ “اور اگر تم انہیں ہاتھ لگانے سے پہلے طلاق دو اور ان کا مہر مقرر کر چکے ہو تو جو مقرر کیا اس کا آدھا (دینا ہے)، سوائے اس کے کہ وہ عورتیں معاف کر دیں، یا وہ معاف کر دے جس کے ہاتھ میں نکاح کی گرہ ہے؛ اور تمہارا معاف کر دینا تقویٰ سے زیادہ قریب ہے۔” جس کے ہاتھ میں نکاح کی گرہ ہے، جمہور اور احناف کے نزدیک وہ شوہر ہے جو آدھے کے بجائے پورا دے دے، اور بعض کے نزدیک ولی؛ تفصیل فقہ میں ہے۔ مگر پہلی بات یہ دیکھیے کہ معاف کرنے کا ذکر سب سے پہلے عورت کا ہے۔ حق اس کا ہے، اختیار اس کا ہے، اور اس کی فیاضی کو قرآن نے پہلے سراہا۔ پھر وہ جملہ جو دونوں سے خطاب ہے: معاف کر دینا تقویٰ سے قریب تر ہے۔ تقویٰ کی جڑ بچاؤ ہے؛ اہلِ معرفت کہتے ہیں کہ جو اپنا حق چھوڑ دیتا ہے وہ دل کو حساب کے بوجھ سے بچا لیتا ہے، اور ہلکا دل حق کی طرف جلد اٹھتا ہے۔ حق تعالیٰ کا نام العفو ہے، یعنی مٹا دینے والا؛ جو بندہ اپنے حق کا ایک حصہ مٹا دیتا ہے وہ اس نام کا ایک عکس اپنے اندر روشن کر لیتا ہے۔

اور اب اس رکوع کا دل: وَلَا تَنْسَوُا الْفَضْلَ بَيْنَكُمْ ۭ اِنَّ اللّٰهَ بِمَا تَعْمَلُوْنَ بَصِيْرٌ۔ “اور آپس میں ایک دوسرے کے ساتھ بھلائی کو نہ بھولو؛ بے شک اللہ تمہارے کاموں کو دیکھ رہا ہے۔” غور کیجیے کہ یہ جملہ کہاں آیا۔ ابھی آدھے مہر کا حساب ہو رہا تھا، یعنی جدائی کی سب سے حسابی گھڑی؛ اور ٹھیک اسی گھڑی قرآن حساب سے اوپر کی ایک بات کہتا ہے: فضل کو نہ بھولو۔ رشتہ ختم ہو سکتا ہے، فضل کی یاد ختم نہیں ہونی چاہیے۔ جو مہربانیاں ہوئیں، جو عزت دی گئی، جو اچھے دن ساتھ گزرے، غصہ انہیں سب سے پہلے بھلاتا ہے؛ قرآن نے بھولنے کو ہی منع کر دیا۔ صحیح مسلم کی حدیث کا مفہوم ہے کہ مومن مرد مومن عورت سے بغض نہ رکھے؛ اگر اس کی کوئی عادت ناپسند ہو تو کوئی دوسری پسند آ جائے گی۔ یہی نظر جدائی کے وقت بھی باقی رہے۔ اور لفظ بینکم دیکھیے: فضل کسی ایک طرف کا نہیں، دونوں کے بیچ کا ہے؛ دونوں نے دیا، دونوں نے پایا۔ اہلِ معرفت کہتے ہیں کہ بھولنا نفس کا کام ہے اور یاد رکھنا ذکر کا؛ یوں طلاق کے احکام کے بیچ قرآن نے ذکر کی ایک مشق رکھ دی، تاکہ جدا ہونے والے جدا ہوں تو دل سیاہ کر کے نہیں، روشن رکھ کر۔

اس جملے کی ایک اور گہرائی عدل اور فضل کے فرق میں ہے۔ عدل یہ ہے کہ ہر ایک کو اس کا حق ملے، آدھا مہر آدھا ہی رہے؛ فضل یہ ہے کہ حق سے بڑھ کر دیا جائے۔ حدیثِ قدسی کا مفہوم صحیحین میں ہے کہ میری رحمت میرے غضب پر سبقت لے گئی۔ شیخِ اکبر کے مکتب میں سبقتِ رحمت کا مطلب یہ ہے کہ وجود خود ایک عطا ہے جو کسی استحقاق سے پہلے ملی؛ اعیان نے وجود کما کر نہیں پایا، نفَسِ رحمانی نے انہیں بخشا۔ حق کا معاملہ مخلوق سے فضل کا ہے، اور جو بندہ اس کے اخلاق کا رنگ لے وہ بھی حساب پر نہیں رکتا۔ پھر بینکم کا لفظ ایک برزخ کی طرف اشارہ کرتا ہے۔ اس مکتب میں برزخ وہ درمیانی حقیقت ہے جو دو چیزوں کو جدا بھی رکھتی ہے اور ملاتی بھی ہے؛ دو انسانوں کے بیچ بھی ایک ایسی جگہ ہوتی ہے جو کسی ایک کی ملکیت نہیں۔ نکاح ٹوٹ جائے تو بھی یہ برزخ بھلائی سے روشن رہ سکتا ہے: سلام میں، دعا میں، اچھی بات میں، اولاد کی خاطر تعاون میں۔ اور آخر میں بصیر کا نام: وہ رشتے کا آخری عمل بھی اسی طرح دیکھتا ہے جیسے پہلا دیکھا تھا۔

عین اسی جگہ، طلاق اور مہر کے احکام کے بیچ، نماز کا حکم آتا ہے: حٰفِظُوْا عَلَى الصَّلَوٰتِ وَالصَّلٰوةِ الْوُسْطٰى وَقُوْمُوْا لِلّٰهِ قٰنِتِيْنَ۔ “سب نمازوں کی حفاظت کرو، اور خاص طور پر درمیانی نماز کی، اور اللہ کے سامنے ادب اور عاجزی سے کھڑے ہو۔” مفسرین اس ترتیب کی حکمت یہ بتاتے ہیں کہ گھر کے جھگڑے اور مال کے حساب میں آدمی نماز کو نہ بھولے؛ نماز ہی وہ پہرہ دار ہے جو باقی سب احکام کی حفاظت کرتی ہے۔ درمیانی نماز کون سی ہے، اس میں کئی اقوال ہیں؛ سب سے مضبوط قول عصر کا ہے، کیونکہ صحیحین کی روایت کا مفہوم ہے کہ غزوۂ احزاب کے دن حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ انہوں نے ہمیں درمیانی نماز، یعنی عصر کی نماز، سے روک دیا۔ بعض علما نے فجر کا قول بھی لیا ہے، اور بعض نے کہا کہ اسے کچھ مخفی رکھا گیا تاکہ ہر نماز کی حفاظت ہو، جیسے لیلۃ القدر کو مخفی رکھا گیا۔ حافظوا کا صیغہ دو طرفہ عمل کا ہے؛ اہلِ تفسیر لکھتے ہیں کہ تم نماز کی حفاظت کرو تو نماز تمہاری حفاظت کرتی ہے، جیسا کہ فرمایا: اِنَّ الصَّلٰوةَ تَنْهٰى عَنِ الْفَحْشَآءِ وَالْمُنْكَرِ۔ جو گھر طلاق کے طوفان سے گزر رہا ہو، اسے سب سے پہلے اسی پہرے کی ضرورت ہے۔

اہلِ معرفت کے نزدیک وسطیٰ کا لفظ ایک راز کھولتا ہے۔ عربی میں وسط درمیان بھی ہے اور بہترین بھی۔ شیخِ اکبر کے مکتب میں دل انسان کے اندر کا برزخ ہے، روح اور نفس کے درمیان، جو دونوں طرف دیکھتا ہے۔ اس تاویل میں درمیانی نماز دل کی نماز ہے: وہ حضوری جو پانچوں نمازوں کے بیچ بھی جاری رہتی ہے، اور جس کے بارے میں فرمایا: الَّذِيْنَ هُمْ عَلٰى صَلَاتِهِمْ دَآئِمُوْنَ۔ پانچ نمازیں ستون ہیں اور یہ حضوری ان کے بیچ بہنے والی رو۔ صحیح مسلم کی حدیثِ قدسی کا مفہوم ہے کہ اللہ نے فرمایا: میں نے نماز کو اپنے اور اپنے بندے کے درمیان آدھا آدھا بانٹ دیا ہے۔ گویا نماز خود ایک درمیان ہے، بندے اور رب کے بیچ کا برزخ۔ پچھلی آیت میں میاں بیوی کے بیچ کے فضل کی بات تھی، اور یہاں بندے اور رب کے بیچ کی نماز کی؛ جس کا یہ دوسرا درمیان سلامت ہو، اس کا پہلا درمیان بھی بگڑنے نہیں پاتا۔ شیخِ اکبر نے فتوحاتِ مکیہ میں نماز کے اسرار پر طویل گفتگو کی ہے۔ مگر یہ یاد رہے کہ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ درمیانی نماز کی حفاظت پہلے وہی ظاہری نماز ہے، وقت پر اور پوری۔

اور آیت کا آخری حصہ: اللہ کے لیے قانت ہو کر کھڑے ہونا۔ صحیحین میں حضرت زید بن ارقم رضی اللہ عنہ کی روایت کا مفہوم ہے کہ ابتدا میں ہم نماز میں ایک دوسرے سے بات کر لیتے تھے، یہاں تک کہ یہ آیت اتری اور ہمیں خاموشی کا حکم دیا گیا۔ قنوت میں اطاعت بھی ہے، عاجزی بھی، دیر تک کھڑے رہنا بھی، اور خاموشی بھی۔ اہلِ ذوق کہتے ہیں کہ زبان کی خاموشی پہلا درجہ ہے؛ دوسرا درجہ دل کی خاموشی ہے، جب اندر کی باتیں بھی تھم جائیں اور خیالات کا شور بیٹھ جائے۔ شیخِ اکبر کے مکتب میں نماز مناجات ہے، یعنی رب سے سرگوشی؛ اور سرگوشی میں سننے کے لیے چپ رہنا پڑتا ہے۔ للہ کا لفظ بتاتا ہے کہ یہ کھڑا ہونا کسی دکھاوے کے لیے نہیں، صرف اس کے لیے ہے۔ جو بندہ اس قنوت کو پا لیتا ہے، اس کے لیے نماز ایک وقفہ نہیں رہتی، قرارگاہ بن جاتی ہے؛ اور وہی شخص گھر کے جھگڑے میں بھی اپنی زبان پر قابو رکھنا سیکھ لیتا ہے، کیونکہ وہ دن میں پانچ بار رب کے سامنے چپ رہنے کی مشق کرتا ہے۔

نماز کی اہمیت اتنی ہے کہ خوف میں بھی ساقط نہیں ہوتی: فَاِنْ خِفْتُمْ فَرِجَالًا اَوْ رُكْبَانًا ۚ فَاِذَاۤ اَمِنْتُمْ فَاذْكُرُوا اللّٰهَ كَمَا عَلَّمَكُمْ مَّا لَمْ تَكُوْنُوْا تَعْلَمُوْنَ۔ “پھر اگر تمہیں خوف ہو تو پیدل یا سواری پر (جیسے بن پڑے نماز پڑھ لو)؛ پھر جب امن ہو جائے تو اللہ کو اسی طرح یاد کرو جیسے اس نے تمہیں سکھایا جو تم نہیں جانتے تھے۔” صورت میں نرمی آ گئی، فرض اپنی جگہ رہا؛ خوف کی نماز کی تفصیل فقہ کا موضوع ہے۔ اور امن لوٹے تو یاد بھی اسی طریقے سے ہو جو سکھایا گیا؛ ذکر میں بھی ادب یہ ہے کہ بندہ اپنی پسند کی صورت نہ گھڑے، تعلیم کی پیروی کرے۔ پھر یہ احسان یاد دلایا کہ تم نہیں جانتے تھے؛ اہلِ معرفت کہتے ہیں کہ بندے کا ہر علم عطا ہے، کمائی نہیں۔ اس آیت میں ایک اور اشارہ بھی ہے: خوف ہو یا امن، حال بدلتا ہے، رخ نہیں بدلتا۔ شیخِ اکبر کے مکتب میں تلوین اور تمکین کا فرق یہی ہے؛ باہر کی حالتیں رنگ بدلتی ہیں، مگر تمکین والا دل ہر رنگ میں اسی ایک کی طرف رہتا ہے، پیدل ہو تو بھی اور سوار ہو تو بھی۔

پھر بیوہ اور مطلقہ کے لیے رخصت کا سامان: وَالَّذِيْنَ يُتَوَفَّوْنَ مِنْكُمْ وَيَذَرُوْنَ اَزْوَاجًا ۚ وَّصِيَّةً لِّاَزْوَاجِهِمْ مَّتَاعًا اِلَى الْحَوْلِ غَيْرَ اِخْرَاجٍ۔ “اور تم میں سے جو لوگ وفات پا جائیں اور بیویاں چھوڑ جائیں، وہ اپنی بیویوں کے لیے وصیت کر جائیں کہ ایک سال تک انہیں خرچ دیا جائے اور گھر سے نکالا نہ جائے۔” جمہور مفسرین کے نزدیک یہ حکم بعد میں عدت کی آیت اور وراثت کے حصوں سے منسوخ ہوا، اور بعض نے کہا کہ یہ بیوہ کے لیے ایک اختیار کی صورت میں باقی ہے؛ تفصیل فقہ کی ہے۔ مگر اس کی روح ہمیشہ باقی ہے: غم کے دنوں میں عورت کو اس کے گھر سے بے دخل نہ کیا جائے۔ اور پھر: وَلِلْمُطَلَّقٰتِ مَتَاعٌۢ بِالْمَعْرُوْفِ ۭ حَقًّا عَلَى الْمُتَّقِيْنَ۔ “اور طلاق یافتہ عورتوں کے لیے معروف طریقے سے کچھ سامان ہے، یہ پرہیزگاروں پر ایک حق ہے۔” اس رکوع کی ابتدا میں یہی سامان محسنین پر حق کہا گیا تھا، اور یہاں متقین پر۔ اہلِ معرفت کہتے ہیں کہ متقی اسے اس لیے دیتا ہے کہ چھوڑنے سے ڈرتا ہے، اور محسن اس لیے کہ دینے سے محبت رکھتا ہے؛ تقویٰ دروازہ ہے اور احسان منزل، اور دونوں عورت کی عزت پر آ کر ملتے ہیں۔

اور گھریلو احکام کا یہ طویل سلسلہ ایک جملے پر تمام ہوتا ہے: كَذٰلِكَ يُبَيِّنُ اللّٰهُ لَكُمْ اٰيٰتِهٖ لَعَلَّكُمْ تَعْقِلُوْنَ۔ “اسی طرح اللہ تمہارے لیے اپنی آیتیں کھول کر بیان کرتا ہے تاکہ تم سمجھو۔” ان سب احکام کو، نکاح، طلاق، عدت، دودھ پلانا، مہر اور سامان، آیات کہا گیا؛ وہی آیات جنہیں مذاق بنانے سے پہلے رکوع میں منع کیا گیا تھا۔ عقل کا لفظ عقال سے ہے، وہ رسی جس سے اونٹ کو باندھا جاتا ہے تاکہ بھاگ نہ جائے۔ عقل وہ ہے جو سمجھی ہوئی بات کو باندھ کر رکھے، تاکہ غصے کے وقت ہاتھ سے نہ نکل جائے۔ شیخِ اکبر کے مکتب میں عقل اور قلب کا فرق بیان ہوتا ہے: عقل باندھتی ہے اور حد قائم رکھتی ہے، قلب پلٹتا ہے اور ہر تجلی کو اس کی صورت میں قبول کرتا ہے۔ کامل بندہ دونوں کو جمع کرتا ہے: عقل سے حکم تھامتا ہے اور دل سے حکم دینے والے کو دیکھتا ہے۔ یوں یہ سلسلہ حدود سے شروع ہوا تھا اور عقل پر ختم ہوا؛ حد اور رسی دونوں ایک ہی رحمت کے نام ہیں، جو بندے کو اس کے اپنے نفس سے بچا کر اس کے رب کے قریب رکھتی ہے۔
