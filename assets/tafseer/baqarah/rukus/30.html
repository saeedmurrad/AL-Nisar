پچھلے رکوع میں شوہر کو روکا گیا تھا کہ عورت کو ستانے کے لیے نہ روکے؛ اب ولی اور خاندان کو روکا جاتا ہے کہ عورت کو اس کی جائز مرضی سے نہ روکیں: وَاِذَا طَلَّقْتُمُ النِّسَآءَ فَبَلَغْنَ اَجَلَهُنَّ فَلَا تَعْضُلُوْهُنَّ اَنْ يَّنْكِحْنَ اَزْوَاجَهُنَّ اِذَا تَرَاضَوْا بَيْنَهُمْ بِالْمَعْرُوْفِ۔ “اور جب تم عورتوں کو طلاق دو اور وہ اپنی عدت پوری کر لیں تو انہیں اپنے (سابقہ) شوہروں سے نکاح کرنے سے نہ روکو، جبکہ وہ آپس میں معروف طریقے سے راضی ہوں۔” صحیح بخاری کی روایت کا مفہوم ہے کہ حضرت معقل بن یسار رضی اللہ عنہ کی بہن کو ان کے شوہر نے طلاق دی اور عدت گزر گئی؛ پھر وہی شخص دوبارہ پیغام لایا اور بہن بھی راضی تھیں، مگر حضرت معقل رضی اللہ عنہ نے غیرت میں آ کر انکار کر دیا۔ یہ آیت اتری تو انہوں نے سر جھکا دیا اور نکاح کرا دیا۔ آیت کی کنجی باہمی رضامندی ہے۔ فیصلہ ان دونوں کا ہے جن کا گھر بسنا ہے، اور عورت کی رضا اس فیصلے کا مرکز ہے؛ ولی کی ذمہ داری حفاظت ہے، رکاوٹ نہیں۔ عضل کا اصل معنی تنگ کر دینا اور راستہ روک دینا ہے۔ اہلِ معرفت کہتے ہیں کہ یہ نفس کی پرانی بیماری ہے کہ وہ اپنے مجروح مان کی خاطر دوسروں کی بھلائی کا راستہ روک دیتا ہے۔ حق تعالیٰ کا ایک نام الواسع ہے، اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں تخلق باسماء اللہ کا مطلب یہ ہے کہ بندہ اپنی استعداد کے مطابق ان اسما کا رنگ لے؛ جس نے وسعت کا رنگ لیا وہ کسی کے لیے تنگی نہیں بنتا۔

پھر نصیحت کو ایمان سے جوڑ دیا گیا: ذٰلِكَ يُوْعَظُ بِهٖ مَنْ كَانَ مِنْكُمْ يُؤْمِنُ بِاللّٰهِ وَالْيَوْمِ الْاٰخِرِ ۭ ذٰلِكُمْ اَزْكٰى لَكُمْ وَاَطْهَرُ ۭ وَاللّٰهُ يَعْلَمُ وَاَنْتُمْ لَا تَعْلَمُوْنَ۔ “اس کی نصیحت اسے کی جاتی ہے جو تم میں سے اللہ اور آخرت کے دن پر ایمان رکھتا ہے؛ یہ تمہارے لیے زیادہ پاکیزہ اور زیادہ ستھرا ہے، اور اللہ جانتا ہے اور تم نہیں جانتے۔” گھر کے یہ احکام رسم و رواج کا معاملہ نہیں، ایمان کا امتحان ہیں۔ ازکیٰ کی جڑ میں بڑھنا اور پاک ہونا دونوں ہیں، اور اطہر میں میل سے صاف ہونا۔ جب بھائی اپنا مان ایک طرف رکھ دیتا ہے تو گھر بھی پاک ہوتا ہے اور اس کا اپنا دل بھی؛ نفس کا دعویٰ گھٹتا ہے اور روح بڑھتی ہے۔ پھر آیت کا آخری جملہ ادب کی ایک پوری راہ ہے۔ شیخِ اکبر کے مکتب میں حق ہر چیز کو اس کی حقیقت میں جانتا ہے، جیسی وہ اعیانِ ثابتہ میں ہے، اور بندہ صرف وہ جانتا ہے جو اس پر ظاہر ہو۔ اس فرق کو مان لینا جہالت نہیں، علم کا ایک مقام ہے؛ اس مکتب میں عارف وہ ہے جو اپنے علم کی حد جانتا ہے اور اس حد پر حیرت کے ساتھ کھڑا ہوتا ہے، دعوے کے ساتھ نہیں۔ حضرت معقل رضی اللہ عنہ نے مصلحت پوری طرح سمجھنے سے پہلے سر جھکا دیا؛ یہی وہ اطاعت ہے جو جانتی ہے کہ وہ نہیں جانتی، اور اسی میں پاکیزگی ہے۔

اب گفتگو ایک نئے وجود کی طرف مڑتی ہے جو جدائی کے بیچ بھی سب سے زیادہ حفاظت کا حق دار ہے: وَالْوَالِدٰتُ يُرْضِعْنَ اَوْلَادَهُنَّ حَوْلَيْنِ كَامِلَيْنِ لِمَنْ اَرَادَ اَنْ يُّتِمَّ الرَّضَاعَةَ ۭ وَعَلَى الْمَوْلُوْدِ لَهٗ رِزْقُهُنَّ وَكِسْوَتُهُنَّ بِالْمَعْرُوْفِ۔ “اور مائیں اپنے بچوں کو پورے دو سال دودھ پلائیں، اس کے لیے جو دودھ کی مدت پوری کرنا چاہے؛ اور جس کا بچہ ہے اس کے ذمے ان ماؤں کا کھانا اور کپڑا ہے، معروف طریقے سے۔” باپ کو یہاں اس نام سے پکارا گیا: وہ جس کے لیے بچہ پیدا ہوا۔ مفسرین لکھتے ہیں کہ نسب باپ کی طرف چلتا ہے، اس لیے خرچ بھی اسی پر ہے؛ ماں اپنے وجود سے دیتی ہے اور باپ اپنے مال سے، اور دونوں مل کر ایک زندگی کو سنبھالتے ہیں۔ ماں کا دودھ پلانا ایسی خدمت ہے جس کا حق ادا کیا جائے، اور وہ بھی معروف کے ساتھ، اس کی حیثیت اور عزت کے مطابق۔ ماں کا مقام کیا ہے، اس پر حدیثِ قدسی کا مفہوم سنن میں آیا ہے کہ اللہ نے فرمایا: میں رحمٰن ہوں، میں نے رحم کو پیدا کیا اور اس کا نام اپنے نام سے نکالا۔ اہلِ معرفت کہتے ہیں کہ بچے کی دنیا میں ماں اسمِ رحمٰن کا سب سے قریبی مظہر ہے، اور اسمِ رب کا بھی، کیونکہ رب کے معنی میں پرورش اور تربیت ہے، اور پرورش کی پہلی درس گاہ ماں کی گود ہے۔

اہلِ تاویل اس آیت میں ایک تمثیل بھی دیکھتے ہیں۔ صحیح بخاری کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے خواب میں دودھ کا پیالہ پیا اور بچا ہوا حضرت عمر رضی اللہ عنہ کو دے دیا، اور اس کی تعبیر علم فرمائی۔ اسی بنا پر اس مکتب میں دودھ علم کی صورت ہے، اور دودھ پلانا روح کی پرورش کی تصویر۔ پورے دو سال کا ذکر بتاتا ہے کہ پرورش کی ایک مکمل مدت ہوتی ہے؛ مرشد طالب کو وقت سے پہلے دودھ نہیں چھڑاتا، اور نہ اسے وہ غذا دیتا ہے جو ابھی اس کے ہضم کی نہیں۔ اہلِ ذوق کہتے ہیں کہ جیسے بچہ دودھ سے بڑھ کر روٹی تک پہنچتا ہے، ویسے طالب اشارے سے بڑھ کر حقیقت تک، مگر ترتیب سے۔ یہ واضح رہے کہ یہ محض تمثیل ہے؛ اس میں نہ ماں کے مقام کی کمی کا کوئی دعویٰ ہے نہ کسی کی برتری کا۔ بلکہ تمثیل کی جڑ یہی ہے کہ ماں کی پرورش اتنی بلند حقیقت ہے کہ اہلِ معرفت نے روحانی تربیت کو بھی اسی سے تشبیہ دی۔ اور ظاہر کا حکم اپنی جگہ پورا قائم ہے: بچے کا حق، ماں کا حق، باپ کی ذمہ داری؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر چند اصول جو گھر کے ٹوٹنے کے بعد بھی بچے کو بچاتے ہیں: لَا تُكَلَّفُ نَفْسٌ اِلَّا وُسْعَهَا ۚ لَا تُضَآرَّ وَالِدَةٌۢ بِوَلَدِهَا وَلَا مَوْلُوْدٌ لَّهٗ بِوَلَدِهٖ وَعَلَى الْوَارِثِ مِثْلُ ذٰلِكَ ۚ فَاِنْ اَرَادَا فِصَالًا عَنْ تَرَاضٍ مِّنْهُمَا وَتَشَاوُرٍ فَلَا جُنَاحَ عَلَيْهِمَا ۭ وَاِنْ اَرَدْتُّمْ اَنْ تَسْتَرْضِعُوْۤا اَوْلَادَكُمْ فَلَا جُنَاحَ عَلَيْكُمْ اِذَا سَلَّمْتُمْ مَّاۤ اٰتَيْتُمْ بِالْمَعْرُوْفِ ۭ وَاتَّقُوا اللّٰهَ وَاعْلَمُوْۤا اَنَّ اللّٰهَ بِمَا تَعْمَلُوْنَ بَصِيْرٌ۔ “کسی جان پر اس کی وسعت سے زیادہ بوجھ نہیں ڈالا جاتا؛ نہ ماں کو اس کے بچے کی وجہ سے تکلیف دی جائے اور نہ باپ کو اس کے بچے کی وجہ سے، اور وارث پر بھی ایسی ہی ذمہ داری ہے۔ پھر اگر دونوں باہمی رضامندی اور مشورے سے دودھ چھڑانا چاہیں تو ان پر کوئی گناہ نہیں؛ اور اگر تم اپنے بچوں کو کسی اور سے دودھ پلوانا چاہو تو بھی تم پر کوئی گناہ نہیں، جبکہ جو طے کیا وہ معروف طریقے سے ادا کر دو؛ اور اللہ سے ڈرو اور جان رکھو کہ اللہ تمہارے کاموں کو دیکھ رہا ہے۔” بچہ ماں باپ کے درمیان ہتھیار نہ بنے، یہ اس آیت کا مرکزی حکم ہے۔ اور دودھ چھڑانے جیسے چھوٹے فیصلے کو بھی باہمی مشورے سے باندھا گیا؛ شوریٰ کا اصول جھولے تک پہنچ گیا اور ماں کی رائے کو برابر وزن ملا۔ شیخِ اکبر کے مکتب میں وسعت کا جملہ استعداد کا راز کھولتا ہے: فیضِ اقدس ہر عین کو اس کی استعداد دیتا ہے اور فیضِ مقدس اسی استعداد کے مطابق اسے وجود اور حال عطا کرتا ہے؛ حق کسی سے وہ نہیں مانگتا جو اس میں رکھا نہ گیا ہو۔ اور آخر میں بصیر کا نام: جو گھر کے کمروں میں ہونے والے فیصلوں کو بھی دیکھتا ہے۔

بیوہ کا ذکر پورے وقار کے ساتھ آیا: وَالَّذِيْنَ يُتَوَفَّوْنَ مِنْكُمْ وَيَذَرُوْنَ اَزْوَاجًا يَّتَرَبَّصْنَ بِاَنْفُسِهِنَّ اَرْبَعَةَ اَشْهُرٍ وَّعَشْرًا ۚ فَاِذَا بَلَغْنَ اَجَلَهُنَّ فَلَا جُنَاحَ عَلَيْكُمْ فِيْمَا فَعَلْنَ فِيْۤ اَنْفُسِهِنَّ بِالْمَعْرُوْفِ ۭ وَاللّٰهُ بِمَا تَعْمَلُوْنَ خَبِيْرٌ۔ “اور تم میں سے جو لوگ وفات پا جائیں اور بیویاں چھوڑ جائیں، وہ بیویاں اپنے آپ کو چار مہینے دس دن روکے رکھیں؛ پھر جب وہ اپنی مدت پوری کر لیں تو اپنے بارے میں جو کچھ معروف طریقے سے کریں اس کا تم پر کوئی گناہ نہیں؛ اور اللہ تمہارے کاموں سے پوری طرح باخبر ہے۔” صحیح بخاری کی روایات کا مفہوم ہے کہ جاہلیت میں بیوہ کو پورا ایک سال سخت اور ذلت آمیز حالت میں گزارنا پڑتا تھا۔ اسلام نے اسے ایک باوقار سوگ میں بدل دیا: رشتے کا احترام بھی، اور ایک مقررہ حد بھی۔ مدت پوری ہو جائے تو اپنی زندگی کے فیصلے اس کے اپنے ہیں، معروف کے دائرے میں؛ خاندان کو اعتراض کا حق نہیں۔ اور دیکھیے کہ انتظار کو ان کی اپنی ذات کی طرف منسوب کیا گیا: وہ خود اپنے آپ کو روکے رکھتی ہیں؛ صبر ان کا ہے اور اس کا اجر بھی انہی کا۔ اہلِ معرفت کہتے ہیں کہ غم کی بھی ایک حد ہے۔ قبض آتا ہے تاکہ دل کو گہرائی دے، پھر بسط آتا ہے تاکہ زندگی لوٹ آئے؛ حق دل کو ہمیشہ کی تنگی میں نہیں چھوڑتا۔ سورۂ طلاق کا وعدہ یہی ہے: سَيَجْعَلُ اللّٰهُ بَعْدَ عُسْرٍ يُّسْرًا۔

اسی مدت کے دوران رشتے کی بات کے ادب بتائے گئے: وَلَا جُنَاحَ عَلَيْكُمْ فِيْمَا عَرَّضْتُمْ بِهٖ مِنْ خِطْبَةِ النِّسَآءِ اَوْ اَكْنَنْتُمْ فِيْۤ اَنْفُسِكُمْ ۭ عَلِمَ اللّٰهُ اَنَّكُمْ سَتَذْكُرُوْنَهُنَّ وَلٰكِنْ لَّا تُوَاعِدُوْهُنَّ سِرًّا اِلَّاۤ اَنْ تَقُوْلُوْا قَوْلًا مَّعْرُوْفًا۔ “اور تم پر کوئی گناہ نہیں کہ (عدت والی) عورتوں کو اشارے میں نکاح کا پیغام دو یا اسے اپنے دلوں میں چھپائے رکھو؛ اللہ جانتا ہے کہ تم ان کا ذکر کرو گے، مگر ان سے چھپ کر وعدہ نہ لو، سوائے اس کے کہ کوئی بھلی بات کہو۔” سوگ کی مدت میں کھلا پیغام یا خفیہ عہد و پیمان روا نہیں؛ یہ عورت کے غم کا احترام ہے اور گزرے ہوئے رشتے کا بھی۔ ہاں شائستہ اشارہ جائز رکھا گیا۔ پھر وہ جملہ جو انسانی فطرت کو کھلے دل سے مان لیتا ہے: اللہ جانتا ہے کہ تم ان کا ذکر کرو گے۔ شریعت فطرت کا انکار نہیں کرتی، اسے ادب سکھاتی ہے۔ شیخِ اکبر کے مکتب میں یہ سرِّ قدر کی ایک جھلک ہے: حق بندوں کو جیسا جانتا ہے، ویسا ہی ان کے لیے حکم دیتا ہے؛ حکم محکوم کی حقیقت کے مطابق ہوتا ہے۔ وہ جانتا ہے کہ دل میں خیال آئے گا، اس لیے خیال کو جرم نہیں بنایا؛ وہ جانتا ہے کہ زبان پھسل سکتی ہے، اس لیے بات کو معروف سے باندھ دیا۔

آخر میں حد اور وقت دونوں کا ذکر: وَلَا تَعْزِمُوْا عُقْدَةَ النِّكَاحِ حَتّٰى يَبْلُغَ الْكِتٰبُ اَجَلَهٗ ۭ وَاعْلَمُوْۤا اَنَّ اللّٰهَ يَعْلَمُ مَا فِيْۤ اَنْفُسِكُمْ فَاحْذَرُوْهُ ۚ وَاعْلَمُوْۤا اَنَّ اللّٰهَ غَفُوْرٌ حَلِيْمٌ۔ “اور نکاح کی گرہ باندھنے کا پختہ ارادہ نہ کرو جب تک مقررہ مدت پوری نہ ہو جائے؛ اور جان رکھو کہ اللہ جانتا ہے جو تمہارے دلوں میں ہے، سو اس سے ڈرتے رہو؛ اور جان رکھو کہ اللہ بخشنے والا، بردبار ہے۔” کتاب کا اپنی مدت کو پہنچنا یعنی لکھا ہوا وقت پورا ہونا۔ اہلِ معرفت کہتے ہیں کہ ہر کام کا ایک لکھا ہوا وقت ہے، اور جلدی بے ادبی ہے؛ سالک بھی فتح کو اس کے وقت سے پہلے نہیں چھینتا۔ پھر آیت بات کو عمل سے اٹھا کر دل تک لے جاتی ہے: اللہ وہ بھی جانتا ہے جو تمہارے دلوں میں ہے۔ اور فوراً بعد دو نام: غفور اور حلیم۔ ڈرو، مگر مایوس نہ ہو۔ شیخِ اکبر کے مکتب میں یہی جلال و جمال کا جمع ہے؛ بندہ دونوں ناموں کے بیچ کھڑا رہتا ہے، نہ بے خوف ہوتا ہے نہ ناامید۔ حلیم وہ ہے جو دیکھتا ہے اور سزا میں جلدی نہیں کرتا، تاکہ بندہ لوٹ آئے۔ اس رکوع کی ہر آیت اسی توازن پر ختم ہوئی ہے: حکم ظاہر میں صاف، اور اس کی جڑ دل کے اس دھیان میں کہ وہ دیکھ رہا ہے۔
