یہ رکوع ایک سوال سے کھلتا ہے جو ایک خاص واقعے سے پیدا ہوا: يَسْـَٔلُوْنَكَ عَنِ الشَّهْرِ الْحَرَامِ قِتَالٍ فِيْهِ ؕ قُلْ قِتَالٌ فِيْهِ كَبِيْرٌ ؕ وَصَدٌّ عَنْ سَبِيْلِ اللّٰهِ وَكُفْرٌۢ بِهٖ وَالْمَسْجِدِ الْحَرَامِ ۗ وَاِخْرَاجُ اَهْلِهٖ مِنْهُ اَكْبَرُ عِنْدَ اللّٰهِ ۚ وَالْفِتْنَةُ اَكْبَرُ مِنَ الْقَتْلِ۔ “لوگ آپ سے حرمت والے مہینے میں لڑنے کے بارے میں پوچھتے ہیں۔ کہہ دیجیے: اس میں لڑنا بڑا گناہ ہے؛ مگر اللہ کی راہ سے روکنا، اس کا انکار کرنا، مسجدِ حرام سے روکنا اور اس کے رہنے والوں کو وہاں سے نکال دینا اللہ کے نزدیک اس سے بھی بڑا ہے، اور فتنہ قتل سے بڑھ کر ہے۔” مفسرین کے مطابق ہجرت کے دوسرے برس حضرت عبداللہ بن جحش رضی اللہ عنہ کی قیادت میں ایک چھوٹی جماعت نخلہ کی طرف بھیجی گئی، اور وہاں رجب کے آخری دنوں میں ایک جھڑپ ہوئی جس میں ایک شخص مارا گیا۔ قریش نے شور مچایا کہ مسلمانوں نے حرمت والے مہینے کو پامال کیا، اور روایات کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے بھی اس پر ناپسندیدگی ظاہر فرمائی۔ آیت نے پہلے حرمت کو قائم رکھا: “قِتَالٌ فِيْهِ كَبِيْرٌ”؛ پھر ترازو کا دوسرا پلڑا دکھایا کہ جن لوگوں نے برسوں ایمان والوں کو ستایا، انہیں گھروں سے نکالا اور اللہ کے گھر کا راستہ روکا، ان کا جرم اس سے کہیں بڑا ہے۔ مفسرین کہتے ہیں کہ یہاں فتنہ سے مراد لوگوں کو ظلم کے زور سے ان کے دین سے پھیرنا ہے۔ اہلِ معرفت اس ترازو سے یہ سیکھتے ہیں کہ قرآن انکار نہیں کرتا، تولتا ہے؛ وہ اپنوں کی غلطی کو بھی غلطی کہتا ہے اور مخالف کے بڑے ظلم کو بھی چھپنے نہیں دیتا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: اہلِ تاویل کہتے ہیں کہ دل بھی ایک حرم ہے، اور اسے اس کی قبلہ رخی سے پھیر دینا ہر ظاہری زخم سے بڑا زخم ہے۔

پھر اس دشمنی کی اصل غرض بیان کی گئی: وَلَا يَزَالُوْنَ يُقَاتِلُوْنَكُمْ حَتّٰى يَرُدُّوْكُمْ عَنْ دِيْنِكُمْ اِنِ اسْتَطَاعُوْا ؕ وَمَنْ يَّرْتَدِدْ مِنْكُمْ عَنْ دِيْنِهٖ فَيَمُتْ وَهُوَ كَافِرٌ فَاُولٰۤئِكَ حَبِطَتْ اَعْمَالُهُمْ فِي الدُّنْيَا وَالْاٰخِرَةِ ۚ وَاُولٰۤئِكَ اَصْحٰبُ النَّارِ ۚ هُمْ فِيْهَا خٰلِدُوْنَ۔ “اور وہ تم سے لڑتے ہی رہیں گے یہاں تک کہ اگر ان کا بس چلے تو تمہیں تمہارے دین سے پھیر دیں؛ اور تم میں سے جو اپنے دین سے پھر جائے اور کفر ہی کی حالت میں مرے تو ایسے لوگوں کے اعمال دنیا اور آخرت میں اکارت گئے، اور وہی آگ والے ہیں، اس میں ہمیشہ رہیں گے۔” تاریخ کی روشنی میں یہ جنگ زمین یا مال کی نہیں تھی؛ مخالف کا مقصد ضمیر کی آزادی چھیننا تھا، اور یہی وہ ظلم تھا جس کے مقابلے میں دفاع کی اجازت ملی۔ آیت کی شرط پر غور کیجیے: “فَيَمُتْ وَهُوَ كَافِرٌ”۔ علما کہتے ہیں کہ اعمال کا اکارت ہونا اس حال پر موت سے جڑا ہے، اور موت تک توبہ کا دروازہ کھلا ہے۔ لفظ “حَبِطَتْ” کی اصل اس جانور کی حالت ہے جس کا پیٹ مضر گھاس کھا کر پھول جائے اور وہ ہلاک ہو جائے؛ باہر سے فربہی، اندر سے موت۔ اہلِ معرفت کے نزدیک یہ ان اعمال کی تصویر ہے جن کی جڑ ایمان سے کٹ گئی ہو: صورت بھری ہوئی، حقیقت خالی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں حکم انجام کے حال پر ہے، اور انجام بندے کے اپنے آخری رخ کا نام ہے؛ حق کسی کو زبردستی نہیں پھیرتا، جو خود پھرا اس کے پھرنے کو وجود ملا۔ اسی لیے سالک ہر عمل کے بعد حسنِ خاتمہ کی دعا کرتا ہے اور اپنے کسی عمل پر بھروسا نہیں کرتا۔

اس سخت تنبیہ کے فوراً بعد امید کا دروازہ کھولا گیا: اِنَّ الَّذِيْنَ اٰمَنُوْا وَالَّذِيْنَ هَاجَرُوْا وَجٰهَدُوْا فِيْ سَبِيْلِ اللّٰهِ ۙ اُولٰۤئِكَ يَرْجُوْنَ رَحْمَتَ اللّٰهِ ؕ وَاللّٰهُ غَفُوْرٌ رَّحِيْمٌ۔ “بے شک جو لوگ ایمان لائے اور جنہوں نے ہجرت کی اور اللہ کی راہ میں جدوجہد کی، وہی اللہ کی رحمت کے امیدوار ہیں؛ اور اللہ بخشنے والا، مہربان ہے۔” مفسرین کہتے ہیں کہ حضرت عبداللہ بن جحش رضی اللہ عنہ کے ساتھیوں کو فکر تھی کہ کیا انہیں اجر ملے گا، تو یہ آیت ان کی تسلی کے لیے اتری۔ تین قدم ذکر ہوئے: ایمان، ہجرت اور جہاد۔ حضورِ اکرم صلی اللہ علیہ وسلم کے ارشاد کا مفہوم ہے کہ اصل مہاجر وہ ہے جو اللہ کی منع کی ہوئی چیزوں کو چھوڑ دے (بخاری، بمفہوم)؛ پس ہجرت ایک بار کا سفر بھی ہے اور ہر روز کا ترک بھی۔ سب سے لطیف بات “يَرْجُوْنَ” ہے: ان تینوں مرتبوں کے بعد بھی وہ دعوے دار نہیں، امیدوار ہیں۔ حضورِ اکرم صلی اللہ علیہ وسلم کے ارشاد کا مفہوم ہے کہ کسی کو اس کا عمل جنت میں نہیں لے جائے گا جب تک اللہ کی رحمت اسے نہ ڈھانپ لے (بخاری و مسلم، بمفہوم)۔ شیخِ اکبر کے مکتب میں سبقتِ رحمت کا یہی مطلب ہے: رحمت عمل سے پہلے تھی، عمل کے ساتھ ہے اور عمل کے بعد بھی وہی سہارا ہے۔ عمل دروازے پر دستک ہے، دروازہ کھولنا اس کا کرم ہے۔ اور آیت کا اختتام “غَفُوْرٌ رَّحِيْمٌ” پر ہوا، کیونکہ کوشش کرنے والا بھی کوتاہیوں سے خالی نہیں، اور کوتاہی کا علاج مغفرت ہی ہے۔

اب وہ سوال آتا ہے جس نے ایک پورے معاشرے کی عادت بدل دی: يَسْـَٔلُوْنَكَ عَنِ الْخَمْرِ وَالْمَيْسِرِ ؕ قُلْ فِيْهِمَاۤ اِثْمٌ كَبِيْرٌ وَّمَنَافِعُ لِلنَّاسِ وَاِثْمُهُمَاۤ اَكْبَرُ مِنْ نَّفْعِهِمَا۔ “لوگ آپ سے شراب اور جوئے کے بارے میں پوچھتے ہیں۔ کہہ دیجیے: ان دونوں میں بڑا گناہ ہے اور لوگوں کے لیے کچھ فائدے بھی، مگر ان کا گناہ ان کے فائدے سے بہت بڑا ہے۔” عرب میں شراب اور جوا رگ و پے میں اترے ہوئے تھے، اس لیے شریعت نے حکمت سے درجہ بدرجہ قدم رکھا۔ روایات کا مفہوم ہے کہ حضرت عمر رضی اللہ عنہ دعا کرتے تھے کہ اے اللہ، شراب کے بارے میں ہمیں واضح حکم دے (ابو داود و ترمذی، بمفہوم)۔ پہلے یہ آیت آئی جس نے نفع اور نقصان کو تول کر دل کو تیار کیا؛ پھر نماز کے وقت نشے سے روکا گیا: لَا تَقْرَبُوا الصَّلٰوةَ وَاَنْتُمْ سُكٰرٰى؛ پھر سورۂ مائدہ میں قطعی حرمت آئی: اِنَّمَا الْخَمْرُ وَالْمَيْسِرُ وَالْاَنْصَابُ وَالْاَزْلَامُ رِجْسٌ مِّنْ عَمَلِ الشَّيْطٰنِ فَاجْتَنِبُوْهُ۔ اب حکم قطعی ہے: شراب اور ہر نشہ آور چیز حرام ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم کے ارشاد کا مفہوم ہے کہ ہر نشہ لانے والی چیز خمر ہے اور ہر خمر حرام ہے (مسلم، بمفہوم)۔ جوا ہر وہ داؤ ہے جس میں ایک کا نفع دوسرے کا نقصان ہو اور فیصلہ محنت سے نہیں، اتفاق سے ہو؛ اس کی نئی صورتیں بھی اسی حکم میں ہیں، اور تفصیل فقہ کا موضوع ہے۔ قرآن کی دیانت دیکھیے: اس نے فائدوں کا انکار نہیں کیا، انہیں گن کر بتایا کہ نقصان کا پلڑا بھاری ہے۔ یہ عقل کو مخاطب کرنے کا طریقہ ہے، اسے دبانے کا نہیں۔

اہلِ معرفت اس حکم کے ظاہر کو پوری طرح قائم رکھ کر اس کے اندر جھانکتے ہیں۔ “الْخَمْرِ” کی اصل ڈھانپنا ہے؛ خمر وہ ہے جو عقل پر پردہ ڈال دے۔ اہلِ تاویل کہتے ہیں کہ جو چیز بندے کو اپنے رب سے غافل کر دے اس میں ایک طرح کا نشہ ہے: مال کا نشہ، اقتدار کا نشہ، اپنی تعریف کا نشہ؛ یہ تشبیہ ہے، اور اس سے ظاہری شراب کی حرمت میں کوئی کمی نہیں آتی۔ “الْمَيْسِرِ” کی اصل آسانی ہے: بغیر محنت کے پا لینے کی ہوس۔ اہلِ سلوک کے نزدیک باطن میں جوا یہ ہے کہ نفس بغیر مجاہدے کے احوال اور مقامات چاہے، اور اتفاق کی امید پر راستے کی محنت چھوڑ دے۔ شیخِ اکبر کے مکتب کی شاعری میں شراب محبت اور معرفت کی علامت ہے، مگر یہ علامت حرام کو حلال نہیں کرتی؛ اہلِ معرفت اسی ظاہری حرام کو چھوڑ کر اس پاک شراب تک پہنچتے ہیں جس کا ذکر سورۂ الدہر میں گزرا: وَسَقٰىهُمْ رَبُّهُمْ شَرَابًا طَهُوْرًا۔ اس مکتب میں سکر اور صحو کی بحث بھی اسی سے جڑی ہے: محبت کا غلبہ کبھی سالک کو بے خود کر دیتا ہے، مگر کامل وہ ہے جو اس بے خودی سے ہوش میں لوٹ آئے، شریعت کے ہر حکم کو تھامے رہے اور اپنے حال کو دوسروں پر حجت نہ بنائے۔ حرام نشہ عقل کو مٹاتا ہے؛ معرفت کی سرشاری عقل کو روشن کر کے اسے اس کی حد بتا دیتی ہے۔

اسی آیت میں خرچ کا سوال دوبارہ آیا اور اس بار پیمانہ بتایا گیا: وَيَسْـَٔلُوْنَكَ مَاذَا يُنْفِقُوْنَ ؕ قُلِ الْعَفْوَ ؕ كَذٰلِكَ يُبَيِّنُ اللّٰهُ لَكُمُ الْاٰيٰتِ لَعَلَّكُمْ تَتَفَكَّرُوْنَ فِي الدُّنْيَا وَالْاٰخِرَةِ۔ “اور وہ آپ سے پوچھتے ہیں کہ کیا خرچ کریں؟ کہہ دیجیے: جو ضرورت سے بچ جائے۔ اسی طرح اللہ تمہارے لیے اپنی آیتیں کھول کر بیان کرتا ہے تاکہ تم دنیا اور آخرت دونوں کے بارے میں سوچو۔” مفسرین کے نزدیک “الْعَفْوَ” سے مراد وہ مال ہے جو اپنی اور اپنے گھر والوں کی ضرورت سے زائد ہو؛ نہ ایسا خرچ کہ گھر والے محتاج ہو جائیں، نہ ایسی پکڑ کہ زائد مال سینے پر بوجھ بن جائے۔ لفظ میں ایک لطافت بھی ہے: عفو کا معنی درگزر بھی ہے، گویا دینے والا جو دے وہ آسانی سے دے، دل کی تنگی کے بغیر۔ پھر “تَتَفَكَّرُوْنَ فِي الدُّنْيَا وَالْاٰخِرَةِ” نے سوچ کے دونوں کنارے ملا دیے۔ اہلِ معرفت کہتے ہیں کہ جو صرف دنیا کو سوچے وہ آخرت کو بھول جاتا ہے، اور جو دنیا سے آنکھ چرا کر آخرت کی بات کرے وہ اپنی ذمہ داریاں بھول جاتا ہے؛ کامل فکر دونوں کو ایک ساتھ دیکھتی ہے۔ شیخِ اکبر کے مکتب میں انسانِ کامل برزخ ہے جس میں دونوں جہان جمع ہیں؛ وہ دنیا میں رہتا ہے اور آخرت کو اپنے اندر لیے پھرتا ہے۔

پھر یتیموں کے بارے میں سوال آیا: وَيَسْـَٔلُوْنَكَ عَنِ الْيَتٰمٰى ؕ قُلْ اِصْلَاحٌ لَّهُمْ خَيْرٌ ؕ وَاِنْ تُخَالِطُوْهُمْ فَاِخْوَانُكُمْ ؕ وَاللّٰهُ يَعْلَمُ الْمُفْسِدَ مِنَ الْمُصْلِحِ ؕ وَلَوْ شَآءَ اللّٰهُ لَاَعْنَتَكُمْ ؕ اِنَّ اللّٰهَ عَزِيْزٌ حَكِيْمٌ۔ “اور وہ آپ سے یتیموں کے بارے میں پوچھتے ہیں۔ کہہ دیجیے: ان کی بھلائی کرنا بہتر ہے، اور اگر تم ان کے ساتھ مل جل کر رہو تو وہ تمہارے بھائی ہیں؛ اور اللہ بگاڑنے والے کو سنوارنے والے سے الگ پہچانتا ہے؛ اور اگر اللہ چاہتا تو تمہیں مشکل میں ڈال دیتا؛ بے شک اللہ زبردست، حکمت والا ہے۔” مفسرین بیان کرتے ہیں کہ جب یتیم کا مال کھانے پر سخت وعید اتری تو صحابۂ کرام رضی اللہ عنہم اتنے محتاط ہو گئے کہ یتیموں کا کھانا الگ پکانے لگے، اور بچا ہوا کھانا ضائع ہونے لگا؛ تب یہ آیت آئی اور فرمایا کہ انہیں بھائی بنا کر ساتھ رکھو، اصل چیز نیت اور بھلائی ہے۔ “فَاِخْوَانُكُمْ” نے یتیم کو احسان کے نیچے نہیں، برابری کے رشتے میں رکھا۔ “وَلَوْ شَآءَ اللّٰهُ لَاَعْنَتَكُمْ” میں وہی اصول ہے جو اس سلسلے میں بار بار آتا ہے: وہ سختی کر سکتا تھا اور اس نے نہ کی؛ اس کی قدرت کے ساتھ اس کا آسانی کو چننا رحمت ہے۔ شیخِ اکبر کے مکتب میں اسماء کا یہ جوڑ معنی رکھتا ہے: عزیز یعنی غالب، کہ کوئی اسے مجبور نہیں کر سکتا؛ اور حکیم، کہ وہ ہر چیز کو اس کی جگہ رکھتا ہے۔ اہلِ معرفت کہتے ہیں کہ ہر سالک بھی ایک یتیم ہے جو اپنے اصل سے جدا ہے، اور حق اس کی سرپرستی سختی سے نہیں، اصلاح سے کرتا ہے۔

اب نکاح کا معیار مقرر ہوا: وَلَا تَنْكِحُوا الْمُشْرِكٰتِ حَتّٰى يُؤْمِنَّ ؕ وَلَاَمَةٌ مُّؤْمِنَةٌ خَيْرٌ مِّنْ مُّشْرِكَةٍ وَّلَوْ اَعْجَبَتْكُمْ ۚ وَلَا تُنْكِحُوا الْمُشْرِكِيْنَ حَتّٰى يُؤْمِنُوْا ؕ وَلَعَبْدٌ مُّؤْمِنٌ خَيْرٌ مِّنْ مُّشْرِكٍ وَّلَوْ اَعْجَبَكُمْ۔ “اور مشرک عورتوں سے نکاح نہ کرو جب تک وہ ایمان نہ لائیں، اور ایک مومن باندی مشرک عورت سے بہتر ہے خواہ وہ تمہیں کتنی ہی پسند ہو؛ اور مشرک مردوں سے اپنی عورتوں کا نکاح نہ کرو جب تک وہ ایمان نہ لائیں، اور ایک مومن غلام مشرک مرد سے بہتر ہے خواہ وہ تمہیں کتنا ہی پسند ہو۔” آیت کا توازن قابلِ غور ہے: ایک ہی اصول عورت اور مرد دونوں پر یکساں لاگو ہوا، دونوں جملے ایک ہی سانچے میں ڈھلے۔ اور معیار حسن، دولت یا نسب نہیں، ایمان ہے؛ ایک باندی اور ایک غلام، جنہیں اس زمانے کا معاشرہ سب سے نیچے رکھتا تھا، ایمان کی بدولت ان سے بہتر ٹھہرے جن پر نگاہیں ٹھہرتی تھیں۔ یہ انسانی وقار کا اعلان ہے۔ اہلِ کتاب کی پاک دامن عورتوں سے نکاح کا حکم سورۂ مائدہ میں الگ آیا ہے، اور اس کی تفصیل فقہ میں ہے۔ اہلِ معرفت کہتے ہیں کہ “وَّلَوْ اَعْجَبَتْكُمْ” آنکھ کی پسند اور دل کی سلامتی کا فرق بتاتا ہے: آنکھ صورت پر ٹھہرتی ہے، دل حقیقت کو دیکھتا ہے؛ اور نکاح وہ رشتہ ہے جو دو دلوں کو ایک سمت میں چلاتا ہے، اس لیے اس میں سمت کا ایک ہونا اصل ہے۔

اور پھر وہ جملہ جو اس پورے رکوع کا خلاصہ ہے: اُولٰۤئِكَ يَدْعُوْنَ اِلَى النَّارِ ۚ وَاللّٰهُ يَدْعُوْۤا اِلَى الْجَنَّةِ وَالْمَغْفِرَةِ بِاِذْنِهٖ ۚ وَيُبَيِّنُ اٰيٰتِهٖ لِلنَّاسِ لَعَلَّهُمْ يَتَذَكَّرُوْنَ۔ “وہ لوگ آگ کی طرف بلاتے ہیں، اور اللہ اپنے حکم سے جنت اور مغفرت کی طرف بلاتا ہے، اور اپنی آیتیں لوگوں کے لیے کھول کر بیان کرتا ہے تاکہ وہ نصیحت پکڑیں۔” یہاں دو دعوتیں آمنے سامنے ہیں، اور دونوں کے درمیان بندہ کھڑا ہے۔ دعوت پر زور نہیں چلتا؛ جواب دینا بندے کا اپنا کام ہے۔ یہی اس سلسلے کا بنیادی اصول ہے: حق کسی کو زبردستی آگ کی طرف نہیں دھکیلتا؛ وہ خود جنت اور مغفرت کی طرف بلاتا ہے، آیتیں کھول کر بیان کرتا ہے، اور جو اس پکار سے منہ موڑے اس کا منہ موڑنا ہی اس کی گمراہی ہے۔ سورۂ ابراہیم کے صفحے پر شیطان کا اقرار گزر چکا کہ اس کا کام صرف بلانا تھا، زور اس کے پاس نہیں تھا۔ اہلِ معرفت کہتے ہیں کہ ترتیب میں جنت کے بعد مغفرت کا ذکر یہ بتاتا ہے کہ پکار گناہ گار کے لیے بھی ہے؛ وہ پاک لوگوں ہی کو نہیں بلاتا، آلودہ دامن کو بھی بلاتا ہے تاکہ اسے دھو دے۔ شیخِ اکبر کے نزدیک ہر شے حق کی پکار کا جواب دیتی ہے، مگر انسان کو یہ شرف ملا کہ وہ جواب اپنے اختیار سے دے؛ یہی اختیار اس کی عظمت بھی ہے اور اس کی امانت بھی۔ اور “لَعَلَّهُمْ يَتَذَكَّرُوْنَ” یاد دلاتا ہے کہ یہ پکار کوئی نئی بات نہیں، اسی فطرت کی یاد دہانی ہے جو دل میں پہلے سے لکھی ہے۔
