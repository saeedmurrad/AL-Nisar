یہ رکوع ایک عہد نامے سے کھلتا ہے۔ ادب کی بات پہلے کہہ دی جائے: یہاں جس عہد اور جس روگردانی کا ذکر ہے وہ ایک خاص تاریخی گروہ کا خاص طرزِ عمل ہے، جسے قرآن نے نام لے کر اور حدیں باندھ کر بیان کیا؛ اسے کسی قوم یا آج کے کسی گروہ پر عمومی حکم بنانا قرآن کے اپنے انصاف کے خلاف ہے۔ اس کا اصل رخ ہر اس دل کی طرف ہے جس سے کبھی کوئی عہد لیا گیا۔ وَاِذْ اَخَذْنَا مِيْثَاقَ بَنِيْۤ اِسْرَآءِيْلَ لَا تَعْبُدُوْنَ اِلَّا اللّٰهَ وَبِالْوَالِدَيْنِ اِحْسَانًا وَّذِي الْقُرْبٰى وَالْيَتٰمٰى وَالْمَسٰكِيْنِ۔ “اور جب ہم نے بنی اسرائیل سے پختہ عہد لیا کہ تم اللہ کے سوا کسی کی عبادت نہ کرو گے، اور ماں باپ کے ساتھ بھلائی کرو گے، اور قرابت داروں، یتیموں اور مسکینوں کے ساتھ بھی۔” مفسرین ایک لطیف نکتہ لکھتے ہیں: لَا تَعْبُدُوْنَ امر کے صیغے میں نہیں، خبر کے صیغے میں آیا؛ گویا حکم اتنا پختہ ہے کہ اسے ہو چکی بات کی طرح بیان کیا گیا۔ پھر ترتیب دیکھیے: پہلے توحید، پھر ماں باپ۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ماں باپ وہ ظاہری سبب ہیں جن کے ذریعے حق نے بندے کو وجود میں ظاہر کیا، اور بچپن میں ربوبیت، یعنی پالنے کی صفت، سب سے پہلے انہی کے ہاتھوں سے ظاہر ہوتی ہے۔ ان کے ساتھ احسان اس سبب کی قدر ہے جسے مسبّب الاسباب نے چنا۔ مگر عبادت صرف اللہ کی، اور احسان مظاہر کے ساتھ؛ یہی تنزیہ اور تشبیہ کا جمع ہے: معبود ایک ہے، اور اس کی نشانیاں ہر طرف۔ احسان کا لفظ حدیثِ احسان کی یاد دلاتا ہے، جس کا مفہوم ہے کہ اللہ کی عبادت یوں کرو گویا تم اسے دیکھ رہے ہو؛ اہلِ معرفت کہتے ہیں کہ جو رب کے سامنے احسان میں جیتا ہے، وہ ماں باپ کے سامنے بھی احسان سے پیش آتا ہے، کیونکہ وہ ان میں اسی کی عطا دیکھتا ہے۔ پھر دائرہ پھیلتا ہے: قرابت دار، یتیم، مسکین؛ گھر سے خاندان، خاندان سے بے سہارا، گویا محبت ایک مرکز سے لہروں کی طرح نکل کر سب تک پہنچتی ہے۔

عہد کی اگلی شقیں: وَقُوْلُوْا لِلنَّاسِ حُسْنًا وَّاَقِيْمُوا الصَّلٰوةَ وَاٰتُوا الزَّكٰوةَ۔ “اور لوگوں سے بھلی بات کہو، اور نماز قائم کرو، اور زکوٰۃ دو۔” لِلنَّاسِ، تمام لوگوں سے؛ مفسرین لکھتے ہیں کہ یہ حکم صرف اپنوں یا ہم مذہبوں تک محدود نہیں، ہر انسان کے ساتھ بات اچھی ہونی چاہیے۔ اور حُسْنًا مصدر ہے: یعنی بات بھلی نہ ہو بلکہ سراپا بھلائی ہو۔ شیخِ اکبر کے مکتب میں کلام کی اصل نَفَسُ الرَّحْمٰنِ ہے: ساری کائنات رحمٰن کے سانس سے کلمات کی صورت میں ظاہر ہوئی، اور انسان کا سانس بھی جب لفظ بنتا ہے تو اسی رحمت کی امانت لیے ہوتا ہے؛ جو زبان سے سختی نکالے وہ اس امانت میں خیانت کرتا ہے۔ پھر نماز اور زکوٰۃ: اہلِ معرفت کہتے ہیں کہ نماز بندے کا حق کی طرف رخ ہے اور زکوٰۃ مخلوق کی طرف؛ ایک اوپر کی طرف اٹھتی ہے اور دوسری اطراف میں پھیلتی ہے، اور دونوں مل کر بندگی کا پورا دائرہ بناتی ہیں۔ اور پھر انجام: ثُمَّ تَوَلَّيْتُمْ اِلَّا قَلِيْلًا مِّنْكُمْ وَاَنْتُمْ مُّعْرِضُوْنَ۔ “پھر تم پھر گئے، سوائے تم میں سے تھوڑے سے لوگوں کے، اور تم منہ موڑنے والے ہی ہو۔” قرآن کا عدل دیکھیے: اِلَّا قَلِيْلًا مِّنْكُمْ، تھوڑے لوگ عہد پر قائم رہے، اور ان کا حق بیان میں محفوظ رکھا گیا۔ پھر دو لفظ آئے: تَوَلَّيْتُمْ اور مُعْرِضُوْنَ۔ مفسرین کہتے ہیں کہ تولی پیٹھ پھیرنا ہے اور اعراض پہلو بدل لینا؛ ایک عمل ہے اور دوسرا دل کی مستقل حالت۔ یہی وہ اصول ہے جس پر یہ سارا بیان ٹکا ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا؛ جو منہ موڑے اس کا منہ موڑنا ہی اس کی گمراہی ہے۔

اب دوسرا عہد، جو انسانی جان اور گھر کی حرمت سے متعلق ہے: وَاِذْ اَخَذْنَا مِيْثَاقَكُمْ لَا تَسْفِكُوْنَ دِمَآءَكُمْ وَلَا تُخْرِجُوْنَ اَنْفُسَكُمْ مِّنْ دِيَارِكُمْ ثُمَّ اَقْرَرْتُمْ وَاَنْتُمْ تَشْهَدُوْنَ۔ “اور جب ہم نے تم سے پختہ عہد لیا کہ تم آپس میں ایک دوسرے کا خون نہ بہاؤ گے اور ایک دوسرے کو اپنے گھروں سے نہ نکالو گے، پھر تم نے اقرار کیا اور تم خود اس کے گواہ ہو۔” لفظوں کی بناوٹ پر غور کیجیے: دوسرے کے خون کو دِمَآءَكُمْ، تمہارا خون، کہا گیا؛ اور دوسرے کو نکالنے کو اَنْفُسَكُمْ، اپنے آپ کو نکالنا۔ مفسرین کہتے ہیں کہ ایک ملت کے افراد ایک جان کی طرح ہیں، اور حدیث کا مفہوم بھی یہی ہے کہ مومن آپس میں ایک جسم کی طرح ہیں کہ ایک عضو کو تکلیف ہو تو سارا جسم بے چین ہو جاتا ہے۔ اہلِ معرفت اس سے ایک قدم آگے دیکھتے ہیں: شیخِ اکبر کے مکتب میں انسان کی حقیقت ایک ہے اور افراد اس کی بہت سی صورتیں؛ جو دوسرے پر ہاتھ اٹھاتا ہے وہ دراصل اپنی ہی حقیقت کو زخمی کرتا ہے۔ یہ نکتہ دوسروں کے حقوق کو مٹاتا نہیں، انہیں اور پختہ کرتا ہے۔ پھر اَقْرَرْتُمْ وَاَنْتُمْ تَشْهَدُوْنَ: اقرار بھی تمہارا، گواہی بھی تمہاری۔ اہلِ ذوق کہتے ہیں کہ ہر روح نے اپنے رب کے سامنے ایک ازلی اقرار کیا تھا، اور شریعت کا ہر عہد اسی پرانے اقرار کی یاد دہانی ہے؛ اس لیے عہد شکنی دراصل اپنی ہی گواہی کو جھٹلانا ہے، کسی اجنبی حکم کو ٹھکرانا نہیں۔ اور جس دل نے اپنی گواہی جھٹلا دی، اس کے خلاف کسی اور گواہ کی ضرورت نہیں رہتی۔

پھر وہ واقعہ جس میں یہ عہد ٹوٹا: ثُمَّ اَنْتُمْ هٰۤؤُلَآءِ تَقْتُلُوْنَ اَنْفُسَكُمْ وَتُخْرِجُوْنَ فَرِيْقًا مِّنْكُمْ مِّنْ دِيَارِهِمْ تَظٰهَرُوْنَ عَلَيْهِمْ بِالْاِثْمِ وَالْعُدْوَانِ ۭ وَاِنْ يَّاْتُوْكُمْ اُسٰرٰى تُفٰدُوْهُمْ وَهُوَ مُحَرَّمٌ عَلَيْكُمْ اِخْرَاجُهُمْ۔ “پھر تم ہی وہ ہو کہ آپس میں ایک دوسرے کو قتل کرتے ہو، اور اپنے ہی ایک گروہ کو ان کے گھروں سے نکالتے ہو، گناہ اور زیادتی کے ساتھ ان کے خلاف ایک دوسرے کی پشت پناہی کرتے ہو؛ اور اگر وہ قیدی بن کر تمہارے پاس آئیں تو فدیہ دے کر چھڑا لیتے ہو، حالانکہ ان کا نکالنا ہی تم پر حرام تھا۔” مفسرین اس کا تاریخی پس منظر بیان کرتے ہیں: اسلام سے پہلے مدینہ میں اوس اور خزرج آپس میں لڑتے تھے، اور وہاں آباد بعض قبیلے ایک کے حلیف تھے اور بعض دوسرے کے؛ جنگ میں حلیفوں کا ساتھ دیتے ہوئے وہ اپنے ہی ہم مذہبوں کو قتل کرتے اور گھروں سے نکالتے، اور جنگ ختم ہونے پر انہی کے قیدیوں کا فدیہ ادا کرتے اور کہتے کہ ہماری کتاب میں قیدی چھڑانے کا حکم ہے۔ یہ ایک خاص بستی کے خاص قبیلوں کا خاص عمل تھا، اور قرآن نے اسے اسی حد میں رکھا ہے۔ اس میں تضاد صاف تھا: ایک حکم کو نبھایا، اور اسی کتاب کے دو حکم توڑ دیے۔ اہلِ معرفت کہتے ہیں کہ نفس کی ایک چال یہ ہے کہ وہ گناہ کی پوری عمارت کھڑی کر کے اس پر نیکی کی ایک تختی لگا دیتا ہے، تاکہ ضمیر کو تسلی رہے؛ قیدی کا فدیہ نیکی تھا، مگر وہ اس ظلم کا پردہ بن گیا جس نے اسے قیدی بنایا تھا۔

اور اب وہ سوال جو اس رکوع کا آئینہ ہے، اور جس میں ہر پڑھنے والا اپنا چہرہ دیکھ سکتا ہے: اَفَتُؤْمِنُوْنَ بِبَعْضِ الْكِتٰبِ وَتَكْفُرُوْنَ بِبَعْضٍ ۚ فَمَا جَزَآءُ مَنْ يَّفْعَلُ ذٰلِكَ مِنْكُمْ اِلَّا خِزْيٌ فِي الْحَيٰوةِ الدُّنْيَا ۚ وَيَوْمَ الْقِيٰمَةِ يُرَدُّوْنَ اِلٰۤى اَشَدِّ الْعَذَابِ ۭ وَمَا اللّٰهُ بِغَافِلٍ عَمَّا تَعْمَلُوْنَ۔ “تو کیا تم کتاب کے کچھ حصے پر ایمان رکھتے ہو اور کچھ کا انکار کرتے ہو؟ تو تم میں سے جو ایسا کرے اس کی سزا اس کے سوا کیا ہے کہ دنیا کی زندگی میں رسوائی ہو، اور قیامت کے دن وہ سخت ترین عذاب کی طرف لوٹائے جائیں؛ اور اللہ اس سے غافل نہیں جو تم کرتے ہو۔” یہ سوال ہر مسلمان کو اپنے آپ سے پوچھنا ہے: کیا میں وہ حکم مانتا ہوں جو آسان ہے اور وہ چھوڑ دیتا ہوں جو قیمت مانگتا ہے؟ نماز پڑھ کر جھوٹ بولنا، روزہ رکھ کر حق مارنا، یہ سب اسی ٹکڑے ٹکڑے دین کی صورتیں ہیں۔ شیخِ اکبر کے مکتب میں کتاب ایک ہے جیسے حق ایک ہے، اور اس کا ہر حکم کسی نہ کسی اسمِ الٰہی کی تجلی ہے؛ جو ایک حکم کو رد کرتا ہے وہ اپنے آپ کو اس اسم کے جلوے سے محروم کرتا ہے۔ انسانِ کامل کی شان جامعیت ہے، یعنی سارے اسماء کو قبول کرنا؛ اور ٹکڑوں میں بٹا ہوا ایمان اس جامعیت کا الٹ ہے۔ اور دیکھیے کہ سزا کس پر رکھی گئی: مَنْ يَّفْعَلُ ذٰلِكَ مِنْكُمْ، تم میں سے جو ایسا کرے۔ حکم فعل پر ہے، شناخت پر نہیں؛ جو کرے، وہ پائے، خواہ کوئی ہو۔ یہی قرآن کا عدل ہے، اور یہی اس بات کی دلیل کہ یہ آیت کسی قوم کا طعنہ نہیں، ایک عمل کا میزان ہے۔

اور آخر میں اس سارے طرزِ عمل کی جڑ کھولی گئی: اُولٰۤئِكَ الَّذِيْنَ اشْتَرَوُا الْحَيٰوةَ الدُّنْيَا بِالْاٰخِرَةِ فَلَا يُخَفَّفُ عَنْهُمُ الْعَذَابُ وَلَا هُمْ يُنْصَرُوْنَ۔ “یہی وہ لوگ ہیں جنہوں نے آخرت کے بدلے دنیا کی زندگی خرید لی، تو نہ ان سے عذاب ہلکا کیا جائے گا اور نہ ان کی مدد کی جائے گی۔” ساری بات ایک سودے میں سمٹ آئی۔ حلیفوں کا ساتھ، قبیلے کی عزت، جنگ کا فائدہ: یہ سب دنیا کی قیمتیں تھیں، اور ادا آخرت سے کی گئیں۔ الدنیا کا مادہ قرب ہے، یعنی جو قریب ہے؛ اور نفس کی فطرت ہے کہ قریب کو دور پر ترجیح دے۔ اہلِ معرفت کہتے ہیں کہ عارف آخرت کو بھی قریب دیکھتا ہے، اس لیے اس کا سودا الٹا ہوتا ہے: وہ فانی دے کر باقی لیتا ہے۔ اور وَلَا هُمْ يُنْصَرُوْنَ میں ایک تلخ انصاف ہے: انہوں نے دنیا میں حلیفوں کی مدد کے لیے عہد توڑا تھا، اور وہاں کوئی حلیف کام نہ آئے گا؛ جس نے مددگار کا حکم چھوڑ کر مدد ڈھونڈی، اسے مدد نہیں ملتی۔ مگر جب تک سانس باقی ہے یہ سودا پلٹا جا سکتا ہے، اور اس کا دوسرا رخ سورۂ توبہ میں کھلا ہے: اِنَّ اللّٰهَ اشْتَرٰى مِنَ الْمُؤْمِنِيْنَ اَنْفُسَهُمْ وَاَمْوَالَهُمْ بِاَنَّ لَهُمُ الْجَنَّةَ، بے شک اللہ نے ایمان والوں سے ان کی جانیں اور ان کے مال اس بدلے میں خرید لیے کہ ان کے لیے جنت ہے۔ ایک بازار میں آدمی آخرت بیچتا ہے، دوسرے میں خود حق خریدار ہے؛ اور سالک کا سارا سفر پہلے بازار سے نکل کر دوسرے بازار تک پہنچنا ہے، جہاں بیچنے والا اپنی جان دے کر بھی گھاٹے میں نہیں رہتا۔
