یہ رکوع قرآن کی ان جامع آیات میں سے ایک سے کھلتا ہے جنہیں مفسرین دین کا خلاصہ کہتے ہیں: لَيْسَ الْبِرَّ اَنْ تُوَلُّوْا وُجُوْهَكُمْ قِبَلَ الْمَشْرِقِ وَالْمَغْرِبِ وَلٰكِنَّ الْبِرَّ مَنْ اٰمَنَ بِاللّٰهِ وَالْيَوْمِ الْاٰخِرِ وَالْمَلٰۤئِكَةِ وَالْكِتٰبِ وَالنَّبِيّٖنَ۔ نیکی یہی نہیں کہ تم اپنے چہرے مشرق یا مغرب کی طرف پھیر لو، بلکہ نیکی اس کی ہے جو اللہ پر، آخرت کے دن پر، فرشتوں پر، کتاب پر اور نبیوں پر ایمان لائے۔ پس منظر یہ ہے کہ تحویلِ قبلہ کے بعد قبلے کی بحث نے بہت طول پکڑا، اور بعض لوگوں نے سمت ہی کو دین کا مرکز بنا لیا۔ آیت نے بتایا کہ قبلہ برحق ہے اور اس کا حکم قائم ہے، مگر سمت اپنی ذات میں نیکی نہیں؛ نیکی وہ حقیقت ہے جس کی طرف سمت اشارہ کرتی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہاں "تُوَلُّوْا" کا لفظ اسی سورت کی اس آیت کی یاد دلاتا ہے: فَاَيْنَمَا تُوَلُّوْا فَثَمَّ وَجْهُ اللّٰهِ۔ وہ کسی جہت میں محدود نہیں، اس لیے نیکی بھی کسی جہت میں بند نہیں ہو سکتی؛ بدن قبلے کی طرف اس لیے پھرتا ہے کہ دل قبلہ مقرر کرنے والے کی طرف پھرے۔ اور دیکھیے کہ "الْبِرَّ" کی تعریف کسی عمل سے نہیں، ایک شخص سے کی گئی: "مَنْ اٰمَنَ"۔ اہلِ ذوق کہتے ہیں کہ نیکی کوئی رسم نہیں جسے ادا کر کے فارغ ہو جائیں؛ یہ ایک انسان کا حال ہے۔ نیکی وہ ہے جو آدمی خود بن جائے۔

ایمان کے پانچ ارکان ایک خاص ترتیب سے آئے: اللہ، آخرت کا دن، فرشتے، کتاب اور نبی۔ اہلِ معرفت اس ترتیب میں پورا دائرہ دیکھتے ہیں۔ پہلے مبدا اور معاد: جس سے سب کچھ آیا اور جس کی طرف سب کچھ لوٹے گا۔ پھر وہ سلسلہ جس کے ذریعے مبدا کا پیغام بندے تک پہنچتا ہے: فرشتہ اسے لاتا ہے، کتاب اسے اٹھاتی ہے، اور نبی اسے اپنی زندگی میں مجسم کرتا ہے۔ "الْكِتٰبِ" واحد آیا اور مفسرین کہتے ہیں کہ اس سے جنسِ کتاب مراد ہے، یعنی تمام آسمانی کتابیں؛ شیخِ اکبر کے مکتب میں یہ اشارہ ہے کہ کلام کی حقیقت ایک ہے اور اس کے نزول بہت، جیسے ایک ہی نور مختلف شیشوں سے مختلف رنگوں میں گزرے۔ پھر "النَّبِيّٖنَ" جمع آیا، کہ کسی ایک نبی پر ایمان اور دوسرے کا انکار اس نیکی کی نفی ہے۔ ایمان کو عمل سے پہلے رکھا گیا، کیونکہ عمل کی قیمت اس کی جڑ سے ہے؛ یہ وہی بات ہے جو سورت کے آغاز میں "يُؤْمِنُوْنَ بِالْغَيْبِ" کے ساتھ آئی تھی۔ اہلِ ذوق کہتے ہیں کہ ان پانچوں میں سے کوئی بھی آنکھ سے دکھائی نہیں دیتا؛ نیکی کی بنیاد ہی غیب پر بھروسا ہے، اور جس دل نے غیب پر بھروسا سیکھ لیا اس کے لیے ہاتھ کھولنا آسان ہو جاتا ہے۔ اسی لیے اگلا جملہ مال کا ہے۔

ایمان کے بعد پہلا عمل جو گنوایا گیا وہ مال کا دینا ہے: وَاٰتَى الْمَالَ عَلٰى حُبِّهٖ ذَوِي الْقُرْبٰى وَالْيَتٰمٰى وَالْمَسٰكِيْنَ وَابْنَ السَّبِيْلِ ۙ وَالسَّآئِلِيْنَ وَفِي الرِّقَابِ۔ اور مال دے، اس کی محبت کے باوجود، رشتہ داروں کو، یتیموں کو، مسکینوں کو، مسافر کو، مانگنے والوں کو، اور گردنیں چھڑانے میں۔ "عَلٰى حُبِّهٖ" کی ضمیر میں مفسرین کے دو قول ہیں: مال کی محبت کے باوجود، یا اللہ کی محبت میں؛ اور دونوں معنی ایک ساتھ آیت میں سمائے ہوئے ہیں۔ اسی سورت میں کچھ پہلے فرمایا گیا تھا وَالَّذِيْنَ اٰمَنُوْۤا اَشَدُّ حُبًّا لِّلّٰهِ؛ اہلِ معرفت کہتے ہیں کہ یہاں اسی دعوے کا امتحان ہے۔ جو محبوب چیز دوسری محبت کی خاطر چھوڑ دی جائے وہ بتا دیتی ہے کہ دل میں کون سی محبت بڑی ہے۔ سورۂ آلِ عمران میں یہی بات قاعدہ بن کر آئی: لَنْ تَنَالُوا الْبِرَّ حَتّٰى تُنْفِقُوْا مِمَّا تُحِبُّوْنَ۔ دینے والوں کی فہرست قریب سے شروع ہو کر پھیلتی جاتی ہے: پہلے اپنا خون، پھر بے سہارا بچہ، پھر محتاج، پھر اجنبی مسافر، پھر ہر مانگنے والا، اور آخر میں وہ جس کی گردن کسی اور کے ہاتھ میں ہو۔ اہلِ ذوق اسے وسعتِ قلب کا نقشہ کہتے ہیں: دل کا دائرہ رشتے سے نکل کر انسانیت تک پھیلتا ہے۔ "وَفِي الرِّقَابِ" اپنے زمانے کی غلامی کے پس منظر میں آزادی کا دروازہ تھا، جسے شریعت نے کفاروں اور صدقات کے ذریعے کھلا رکھا۔ اور چونکہ زکوٰۃ کا ذکر آگے الگ آ رہا ہے، اس لیے بعض مفسرین نے یہاں سے یہ نتیجہ نکالا کہ مال میں زکوٰۃ کے علاوہ بھی حق ہوتا ہے۔

پھر عبادت، عہد اور صبر کا ذکر آیا، اور آیت اپنی انتہا کو پہنچی: وَاَقَامَ الصَّلٰوةَ وَاٰتَى الزَّكٰوةَ ۚ وَالْمُوْفُوْنَ بِعَهْدِهِمْ اِذَا عٰهَدُوْا ۚ وَالصّٰبِرِيْنَ فِي الْبَاْسَآءِ وَالضَّرَّآءِ وَحِيْنَ الْبَاْسِ ۭ اُولٰۤئِكَ الَّذِيْنَ صَدَقُوْا ۭ وَاُولٰۤئِكَ هُمُ الْمُتَّقُوْنَ۔ اور نماز قائم کرے اور زکوٰۃ دے؛ اور جب عہد کریں تو اپنا عہد پورا کرنے والے، اور خاص طور پر تنگ دستی میں، تکلیف میں اور جنگ کے وقت صبر کرنے والے؛ یہی لوگ سچے ہیں اور یہی پرہیزگار ہیں۔ نحو کے علما ایک نکتہ بتاتے ہیں: "وَالْمُوْفُوْنَ" حالتِ رفع میں ہے، مگر "وَالصّٰبِرِيْنَ" حالتِ نصب میں آیا؛ عربی بلاغت میں اسے نصب علی المدح کہتے ہیں، یعنی اعراب بدل کر کسی صفت کو الگ سے نمایاں کرنا۔ گویا صبر کو اس فہرست میں سے اٹھا کر تاج کی طرح سر پر رکھ دیا گیا۔ مفسرین کہتے ہیں کہ "الْبَاْسَآءِ" فقر و تنگی ہے، "الضَّرَّآءِ" بیماری اور جسمانی تکلیف، اور "حِيْنَ الْبَاْسِ" میدانِ جنگ کی سختی؛ مال، بدن اور جان، تینوں کے امتحان۔ پھر دو گواہیاں: "صَدَقُوْا" اور "الْمُتَّقُوْنَ"۔ اہلِ معرفت کے نزدیک صدق یہ ہے کہ دعوے اور حال میں فاصلہ نہ رہے؛ جس نے ایمان کا دعویٰ کیا اور مال، عہد اور صبر سے اس کی تصدیق کر دی، اس کا ظاہر اس کے باطن کا آئینہ بن گیا۔

اس آیت کو تقویٰ پر ختم کیا گیا، اور یہیں سے اس حصے کی ایک لڑی شروع ہوتی ہے: چند آیات میں قصاص کی حکمت "لَعَلَّكُمْ تَتَّقُوْنَ" پر ختم ہوگی، وصیت "حَقًّا عَلَى الْمُتَّقِيْنَ" پر، اور روزے کا مقصد بھی "لَعَلَّكُمْ تَتَّقُوْنَ" بتایا جائے گا۔ شیخِ اکبر کے مکتب میں نیکی کے تین دائرے ہیں جو اس ایک آیت میں جمع ہو گئے: ایک عقیدے کا دائرہ، جو دل میں ہے؛ ایک عطا کا دائرہ، جو ہاتھ میں ہے؛ اور ایک استقامت کا دائرہ، جو وقت کے ساتھ آزمایا جاتا ہے۔ جب یہ تینوں ایک مرکز پر آ جائیں تو بندہ اسمِ بَرّ کے اخلاق سے آراستہ ہوتا ہے؛ اس مکتب میں اسے تخلق باسماء اللہ کہا جاتا ہے۔ مگر ادب لازم ہے: بندہ نیکی کا مظہر ہے، اس کا مالک نہیں؛ اسی لیے آیت نے نیکی کو بندے کی صفت بنا کر بھی اس کا سرچشمہ ایمان باللہ ہی کو قرار دیا۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر یہ بتاتا ہے کہ نیک آدمی کیا کرتا ہے، اور باطن یہ کہ وہ ایسا کیوں کر پاتا ہے۔

اب نئی امت کی شریعت کا وہ حصہ شروع ہوتا ہے جو جان کی حفاظت سے متعلق ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا كُتِبَ عَلَيْكُمُ الْقِصَاصُ فِي الْقَتْلٰى ۭ اَلْحُرُّ بِالْحُرِّ وَالْعَبْدُ بِالْعَبْدِ وَالْاُنْثٰى بِالْاُنْثٰى۔ اے ایمان والو! مقتولوں کے بارے میں تم پر برابری کا بدلہ فرض کیا گیا: آزاد کے بدلے آزاد، غلام کے بدلے غلام، اور عورت کے بدلے عورت۔ مفسرین بتاتے ہیں کہ جاہلیت میں طاقتور قبیلے اپنے ایک مقتول کے بدلے کئی جانیں مانگتے، یا اپنے غلام کے بدلے دوسرے قبیلے کا آزاد اور اپنی عورت کے بدلے ان کا مرد طلب کرتے تھے؛ یہ آیت اسی زیادتی کو ختم کرنے آئی کہ بدلہ صرف قاتل سے ہے، اس سے آگے نہیں۔ جمہور فقہا کے نزدیک قصاص میں مرد اور عورت کی جان برابر ہے، جیسا کہ سورۂ المائدہ میں فرمایا: اَنَّ النَّفْسَ بِالنَّفْسِ۔ قصاص کسی فرد کا ذاتی انتقام نہیں؛ اس کا نفاذ ثبوت کے بعد عدالت اور جائز حکومت کے ہاتھ سے ہوتا ہے، اور اس کی تفصیل فقہ کے ذمے ہے۔ "كُتِبَ عَلَيْكُمْ" کا صیغہ اس حصے میں تین بار آیا: قصاص کے لیے، وصیت کے لیے، اور روزے کے لیے۔ اہلِ ذوق کہتے ہیں کہ یہ تین تحریریں انسان کے تین مرحلوں پر لکھی گئیں: زندگی کی حفاظت، موت کا انتظام، اور نفس کی تربیت۔

پھر قصاص کے ساتھ ہی معافی کا دروازہ کھول دیا: فَمَنْ عُفِيَ لَهٗ مِنْ اَخِيْهِ شَيْءٌ فَاتِّبَاعٌۢ بِالْمَعْرُوْفِ وَاَدَآءٌ اِلَيْهِ بِاِحْسَانٍ ۭ ذٰلِكَ تَخْفِيْفٌ مِّنْ رَّبِّكُمْ وَرَحْمَةٌ ۭ فَمَنِ اعْتَدٰى بَعْدَ ذٰلِكَ فَلَهٗ عَذَابٌ اَلِيْمٌ۔ پھر جس قاتل کے لیے اس کے بھائی کی طرف سے کچھ معاف کر دیا جائے تو بھلے طریقے سے خون بہا کا مطالبہ ہو اور خوش اسلوبی سے اس کی ادائیگی؛ یہ تمہارے رب کی طرف سے تخفیف اور رحمت ہے؛ پھر جو اس کے بعد زیادتی کرے اس کے لیے دردناک عذاب ہے۔ غور کیجیے کہ مقتول کے وارث کو قاتل کا "اَخِيْهِ" کہا گیا؛ مفسرین کہتے ہیں کہ قتل جیسا جرم بھی ایمان کا رشتہ نہیں کاٹتا، اور بھائی کہہ کر دل کو معافی کی طرف مائل کیا گیا۔ حضرت عبد اللہ بن عباس رضی اللہ عنہما سے صحیح بخاری میں منقول ہے، جس کا مفہوم یہ ہے کہ بنی اسرائیل میں قصاص تھا اور خون بہا نہ تھا، تو اس امت کو یہ تخفیف دی گئی۔ اب وارث کے سامنے تین راہیں ہیں: قصاص، خون بہا، یا پوری معافی۔ شیخِ اکبر کے مکتب میں یہ بندے کو اسماء کے درمیان اختیار دینا ہے: قصاص اسمِ عدل کا مظہر ہے اور معافی اسمِ عفو کا؛ عدل ملامت کے قابل نہیں، مگر عفو اونچا مقام ہے، کیونکہ رحمت غضب پر سبقت لے گئی۔ پھر حکمت بیان ہوئی: وَلَكُمْ فِي الْقِصَاصِ حَيٰوةٌ يّٰۤاُولِي الْاَلْبَابِ لَعَلَّكُمْ تَتَّقُوْنَ۔ اور اے عقل والو! تمہارے لیے قصاص میں زندگی ہے، تاکہ تم بچو۔ بظاہر قصاص ایک موت ہے، مگر قرآن اسے زندگی کہتا ہے، کیونکہ جب قاتل جانتا ہے کہ بدلہ یقینی ہے تو بہت سی جانیں بچ جاتی ہیں۔ اہلِ معرفت ایک باطنی قصاص بھی دیکھتے ہیں: جو اپنے نفس سے اس کی زیادتیوں کا حساب لے، اس کا دل زندہ ہو جاتا ہے۔ اور خطاب "اُولِي الْاَلْبَابِ" سے ہے، یعنی مغز والوں سے، جو چھلکے پر نہیں رکتے۔

جان کی حفاظت کے بعد اب موت کے انتظام کا حکم آیا: كُتِبَ عَلَيْكُمْ اِذَا حَضَرَ اَحَدَكُمُ الْمَوْتُ اِنْ تَرَكَ خَيْرًا الْوَصِيَّةُ لِلْوَالِدَيْنِ وَالْاَقْرَبِيْنَ بِالْمَعْرُوْفِ ۚ حَقًّا عَلَى الْمُتَّقِيْنَ۔ تم پر فرض کیا گیا کہ جب تم میں سے کسی کی موت آ پہنچے، اگر وہ مال چھوڑ رہا ہو، تو ماں باپ اور رشتہ داروں کے لیے دستور کے مطابق وصیت کرے؛ یہ پرہیزگاروں پر لازم ہے۔ مفسرین بتاتے ہیں کہ یہ حکم سورۂ النساء میں میراث کے حصے مقرر ہونے سے پہلے کا ہے؛ جب حصے مقرر ہو گئے تو جمہور کے نزدیک وارث کے لیے وصیت باقی نہ رہی۔ حضورِ اکرم صلی اللہ علیہ وسلم کا ارشاد، جو سنن ابی داود اور جامع ترمذی میں ہے، اس کا مفہوم یہ ہے کہ وارث کے لیے کوئی وصیت نہیں۔ اب وصیت غیر وارث رشتہ داروں اور نیکی کے کاموں کے لیے ہے، اور تہائی مال تک محدود؛ حضرت سعد بن ابی وقاص رضی اللہ عنہ کی حدیث میں، جو صحیح بخاری میں ہے، اس کا مفہوم یہ ہے کہ تہائی کی اجازت ہے اور تہائی بھی بہت ہے۔ تفصیل فقہ کے ذمے ہے۔ قرآن نے یہاں مال کو "خَيْرًا" کہا؛ اہلِ ذوق کہتے ہیں کہ مال اپنی ذات میں بھلائی ہے جب وسیلہ رہے، اور اس کا شر دل کی وابستگی میں ہے۔ اور "حَضَرَ اَحَدَكُمُ الْمَوْتُ" میں موت کو حاضر ہونے والا کہا گیا۔ اہلِ تصوف میں مشہور قول ہے: مرنے سے پہلے مر جاؤ؛ یعنی وصیت کا شعور ابھی پیدا کرو، اس سے پہلے کہ موت حاضر ہو۔ جس نے اپنے مال کو امانت سمجھ کر جینا سیکھ لیا، اس کی ساری زندگی ایک وصیت بن جاتی ہے۔

پھر وصیت کی حفاظت کا حکم: فَمَنْۢ بَدَّلَهٗ بَعْدَمَا سَمِعَهٗ فَاِنَّمَاۤ اِثْمُهٗ عَلَى الَّذِيْنَ يُبَدِّلُوْنَهٗ ۭ اِنَّ اللّٰهَ سَمِيْعٌ عَلِيْمٌ۔ پھر جو اسے سن لینے کے بعد بدل ڈالے تو اس کا گناہ انہی پر ہے جو اسے بدلتے ہیں؛ بے شک اللہ سب کچھ سنتا، سب کچھ جانتا ہے۔ وصیت کرنے والا اپنا فرض ادا کر کے بری ہو گیا؛ بوجھ اسی پر ہے جس نے بدلا۔ یہ قرآن کا عام اصول ہے کہ ہر جان اپنا ہی بوجھ اٹھاتی ہے۔ مگر ہر تبدیلی خیانت نہیں: فَمَنْ خَافَ مِنْ مُّوْصٍ جَنَفًا اَوْ اِثْمًا فَاَصْلَحَ بَيْنَهُمْ فَلَاۤ اِثْمَ عَلَيْهِ ۭ اِنَّ اللّٰهَ غَفُوْرٌ رَّحِيْمٌ۔ ہاں جسے وصیت کرنے والے کی طرف سے کسی بے جا جھکاؤ یا گناہ کا اندیشہ ہو اور وہ ان کے درمیان صلح کرا دے تو اس پر کوئی گناہ نہیں؛ بے شک اللہ بخشنے والا، رحم کرنے والا ہے۔ مفسرین کہتے ہیں کہ "جَنَفًا" وہ جھکاؤ ہے جو بھول یا غلطی سے ہو، اور "اِثْمًا" وہ جو جان بوجھ کر ہو؛ دونوں صورتوں میں اصلاح جائز ہے۔ فرق نیت کا ہے: ایک بدلنا فساد ہے اور ایک بدلنا اصلاح۔ اب دونوں آیتوں کے اختتام پر غور کیجیے۔ بدلنے والے کے ذکر کے بعد "سَمِيْعٌ عَلِيْمٌ" آیا، کہ اس نے وصیت کے الفاظ بھی سنے ہیں اور بدلنے والے کی نیت بھی جانتا ہے؛ اور اصلاح کرنے والے کے ذکر کے بعد "غَفُوْرٌ رَّحِيْمٌ" آیا، کہ اگر اصلاح میں کوئی کمی رہ جائے تو مغفرت اسے ڈھانپ لے۔ شیخِ اکبر کے مکتب میں ہر آیت اس اسمِ الٰہی پر ختم ہوتی ہے جس کا حکم اس مقام پر جاری ہے؛ اسماء اپنے مظاہر کے مطابق آتے ہیں، اور جو ان کو پہچان لے وہ آیت کی روح تک پہنچ جاتا ہے۔
