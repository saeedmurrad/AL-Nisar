یہ رکوع ایک عجیب اعتراض کے جواب سے کھلتا ہے۔ مفسرین نے روایات کی بنیاد پر لکھا ہے کہ مدینہ کے یہود میں سے کچھ لوگوں نے حضورِ اکرم صلی اللہ علیہ وسلم سے پوچھا کہ آپ کے پاس وحی کون لاتا ہے، اور جب حضرت جبریل علیہ السلام کا نام سنا تو کہنے لگے کہ وہ تو ہمارا دشمن ہے، سخت احکام لے کر اترتا ہے؛ میکائیل ہوتے تو ہم مان لیتے۔ جواب آیا: قُلْ مَنْ كَانَ عَدُوًّا لِّجِبْرِيْلَ فَاِنَّهٗ نَزَّلَهٗ عَلٰى قَلْبِكَ بِاِذْنِ اللّٰهِ مُصَدِّقًا لِّمَا بَيْنَ يَدَيْهِ وَهُدًى وَّبُشْرٰى لِلْمُؤْمِنِيْنَ۔ “کہہ دیجیے: جو کوئی جبریل کا دشمن ہو، تو اسی نے تو یہ کلام اللہ کے اذن سے آپ کے دل پر اتارا ہے، جو اپنے سے پہلے کی کتابوں کی تصدیق کرتا ہے اور ایمان والوں کے لیے ہدایت اور خوشخبری ہے۔” حضرت جبریل علیہ السلام کا تعارف ان کی اپنی حیثیت سے نہیں، ان کی امانت سے کرایا گیا: وہ لانے والے ہیں، بھیجنے والا کوئی اور ہے، اس لیے قاصد سے بیر دراصل بھیجنے والے سے بیر ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں “عَلٰى قَلْبِكَ” کے الفاظ پر خاص توجہ ہے: کلام پہلے دل پر اترا، زبان پر بعد میں آیا۔ دل کو قلب اسی لیے کہا گیا کہ وہ تقلب کرتا، یعنی پلٹتا رہتا ہے، اور جو دل ہر حال میں حق کی طرف پلٹے وہی وحی کا ظرف بنتا ہے۔ سورۂ الشعراء میں یہی بات یوں آئی کہ روح الامین اسے آپ کے قلب پر لے کر اترے۔ اہلِ ذوق کہتے ہیں کہ قاریِ قرآن بھی اپنے درجے میں اسی سنت کا وارث ہے: جب تک آیت دل پر نہ اترے، تلاوت صرف زبان کا عمل رہتی ہے۔

پھر بات کو پھیلا کر فرمایا: مَنْ كَانَ عَدُوًّا لِّلّٰهِ وَمَلٰٓئِكَتِهٖ وَرُسُلِهٖ وَجِبْرِيْلَ وَمِيْكٰىلَ فَاِنَّ اللّٰهَ عَدُوٌّ لِّلْكٰفِرِيْنَ۔ “جو کوئی اللہ کا، اس کے فرشتوں کا، اس کے رسولوں کا اور جبریل و میکائیل کا دشمن ہو، تو اللہ ایسے کافروں کا دشمن ہے۔” عام ذکر کے بعد جبریل و میکائیل علیہما السلام کا نام الگ لینا ان کی عزت کا اظہار ہے، اور یہ اعلان بھی کہ فرشتوں کے درمیان تفریق کی کوئی گنجائش نہیں۔ اہلِ تاویل کہتے ہیں کہ جبریل علیہ السلام وحی اور علم کے امین ہیں اور میکائیل علیہ السلام، جیسا کہ مفسرین نے ذکر کیا، بارش اور رزق کے امور پر مقرر ہیں؛ جو ایک کو قبول کرے اور دوسرے سے بیر رکھے وہ گویا روٹی کو مانتا ہے اور ہدایت کو ٹھکراتا ہے، حالانکہ دونوں ایک ہی رب کی عطا ہیں۔ اب ایک باریکی دیکھیے: آیت نے یہ نہیں کہا کہ اللہ ان کا دشمن ہے، بلکہ فرمایا کہ اللہ کافروں کا دشمن ہے۔ عداوت کا حکم ایک صفت پر لگا جو انہوں نے خود اختیار کی، کسی نسل یا گروہ کی ذات پر نہیں۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کا خلاصہ ہے: حکم محکوم کے حال کے مطابق لگتا ہے، اور جو اپنی صفت بدل لے اس کا حکم بھی بدل جاتا ہے۔ دروازہ بند نہیں ہوا، صرف رخ کا نتیجہ بتایا گیا ہے۔

اس کے بعد فرمایا: وَلَقَدْ اَنْزَلْنَآ اِلَيْكَ اٰيٰتٍ بَيِّنٰتٍ ۚ وَمَا يَكْفُرُ بِهَآ اِلَّا الْفٰسِقُوْنَ۔ “اور بے شک ہم نے آپ کی طرف روشن آیتیں اتاری ہیں، اور ان کا انکار وہی کرتے ہیں جو اطاعت سے نکل چکے ہیں۔” یہ اسی اصول کی تکرار ہے جو اس سورت کے آغاز میں طے ہو چکا: وَمَا يُضِلُّ بِهٖٓ اِلَّا الْفٰسِقِيْنَ۔ آیتوں میں کمی نہیں، وہ بیّنات ہیں؛ انکار اس کی طرف سے ہے جو پہلے ہی دائرۂ اطاعت سے باہر قدم رکھ چکا۔ فسق کا اصل معنی پھل کا اپنے چھلکے سے باہر نکل آنا ہے؛ یعنی گمراہی کوئی اوپر سے مسلط کی ہوئی چیز نہیں، بندے کا اپنا نکلنا ہے۔ پھر فرمایا: اَوَكُلَّمَا عٰهَدُوْا عَهْدًا نَّبَذَهٗ فَرِيْقٌ مِّنْهُمْ ۚ بَلْ اَكْثَرُهُمْ لَا يُؤْمِنُوْنَ۔ “کیا ایسا نہیں کہ جب کبھی انہوں نے کوئی عہد باندھا تو ان میں سے ایک گروہ نے اسے اٹھا پھینکا؟ بلکہ ان میں سے اکثر ایمان ہی نہیں رکھتے۔” قرآن کی احتیاط دیکھیے: ایک فریق، اکثر؛ سب نہیں۔ یہ اُس زمانے کے مخصوص طرزِ عمل کا بیان ہے، آج کی کسی قوم پر عام حکم نہیں۔ اور فرمایا: وَلَمَّا جَآءَهُمْ رَسُوْلٌ مِّنْ عِنْدِ اللّٰهِ مُصَدِّقٌ لِّمَا مَعَهُمْ نَبَذَ فَرِيْقٌ مِّنَ الَّذِيْنَ اُوْتُوا الْكِتٰبَ ۙ كِتٰبَ اللّٰهِ وَرَآءَ ظُهُوْرِهِمْ كَاَنَّهُمْ لَا يَعْلَمُوْنَ۔ “اور جب ان کے پاس اللہ کی طرف سے ایک رسول آیا جو ان کے پاس موجود کتاب کی تصدیق کرنے والا تھا، تو اہلِ کتاب کے ایک گروہ نے اللہ کی کتاب کو اپنی پیٹھ پیچھے یوں پھینک دیا گویا وہ جانتے ہی نہیں۔” اہلِ معرفت کہتے ہیں کہ جس چیز کو پیٹھ پیچھے ڈالا جائے، چہرہ لازماً کسی اور طرف ہوتا ہے؛ کتاب پیچھے گئی تو خواہش سامنے آ گئی۔ اور جو علم عمل اور حال نہ بنے، قرآن اسے نہ جاننے کے برابر گنتا ہے۔

اب وہ چیز بتائی گئی جس کے پیچھے وہ لگے: وَاتَّبَعُوْا مَا تَتْلُوا الشَّيٰطِيْنُ عَلٰى مُلْكِ سُلَيْمٰنَ ۚ وَمَا كَفَرَ سُلَيْمٰنُ وَلٰكِنَّ الشَّيٰطِيْنَ كَفَرُوْا يُعَلِّمُوْنَ النَّاسَ السِّحْرَ ۚ۔ “اور وہ اس چیز کے پیچھے لگ گئے جو شیاطین حضرت سلیمان کی بادشاہی کے زمانے میں پڑھا کرتے تھے؛ حالانکہ سلیمان علیہ السلام نے کفر نہیں کیا، بلکہ کفر تو شیاطین نے کیا جو لوگوں کو جادو سکھاتے تھے۔” ربط پر غور کیجیے: اللہ کی کتاب پیٹھ پیچھے گئی تو اس کی خالی جگہ کسی اور تلاوت نے بھر دی۔ انسان خالی نہیں رہتا؛ جو وحی کو چھوڑے وہ کسی اور آواز کے پیچھے چل پڑتا ہے۔ مفسرین لکھتے ہیں کہ بعض لوگوں نے یہ تہمت پھیلائی تھی کہ حضرت سلیمان علیہ السلام کی سلطنت جادو کے زور پر قائم تھی؛ قرآن نے صاف اعلان کر کے ان کی برأت کر دی۔ ان کی بادشاہی دعا اور عطا تھی: وَهَبْ لِيْ مُلْكًا لَّا يَنْۢبَغِيْ لِاَحَدٍ مِّنْۢ بَعْدِيْ۔ شیخِ اکبر حکمتِ رحمانیہ میں حضرت سلیمان علیہ السلام کے بارے میں جو کچھ فرماتے ہیں اس کا مفہوم یہ ہے کہ ان کی تسخیر کا امتیاز یہ تھا کہ وہ حق کے امر سے تھی، کسی نفسانی زور سے نہیں۔ اس مکتب میں یہی فرق کی کسوٹی ہے: تسخیر بالامر حق کے اسم رحمٰن کا ظہور ہے اور بندگی میں ڈھلی ہوئی ہے، جبکہ سحر تخییل اور فریب ہے جو دلوں کو حق سے پھیرتا ہے۔ ایک کی جڑ عبدیت ہے اور دوسرے کی جڑ دعویٰ۔

یہاں آیت ایک نازک مقام پر پہنچتی ہے: وَمَآ اُنْزِلَ عَلَى الْمَلَكَيْنِ بِبَابِلَ هَارُوْتَ وَمَارُوْتَ ۚ وَمَا يُعَلِّمٰنِ مِنْ اَحَدٍ حَتّٰى يَقُوْلَآ اِنَّمَا نَحْنُ فِتْنَةٌ فَلَا تَكْفُرْ ۚ۔ “اور (وہ اس کے پیچھے بھی لگے) جو بابل میں دو فرشتوں ہاروت اور ماروت پر اتارا گیا تھا، اور وہ دونوں کسی کو کچھ نہ سکھاتے تھے جب تک یہ نہ کہہ دیتے کہ ہم تو محض آزمائش ہیں، پس تو کفر نہ کر۔” بعض پرانی کتابوں میں ان دو فرشتوں کے بارے میں قصے ملتے ہیں کہ وہ زہرہ نام کی ایک عورت کے سبب گناہ میں پڑے اور وہ عورت ستارہ بن گئی۔ محقق علما اور مفسرین نے ان قصوں کو بے اصل اسرائیلیات قرار دے کر رد کر دیا ہے، کیونکہ فرشتوں کی شان خود قرآن نے بیان کی ہے: لَا يَعْصُوْنَ اللّٰهَ مَآ اَمَرَهُمْ وَيَفْعَلُوْنَ مَا يُؤْمَرُوْنَ۔ ہم صرف وہی کہتے ہیں جو آیت کہتی ہے: وہ دونوں آزمائش تھے، اور کسی کو کچھ بتانے سے پہلے صاف خبردار کرتے تھے۔ اہلِ معرفت کہتے ہیں کہ فتنہ کا اصل معنی سونے کو آگ میں ڈالنا ہے تاکہ کھرا کھوٹے سے الگ ہو جائے۔ آگ کھوٹ پیدا نہیں کرتی، صرف ظاہر کرتی ہے۔ اسی طرح آزمائش کسی کے دل میں کفر نہیں ڈالتی، دل میں جو پہلے سے ہے اسے سامنے لے آتی ہے۔ اور “فَلَا تَكْفُرْ” کی تنبیہ خود گواہ ہے کہ اختیار سننے والے کے ہاتھ میں تھا؛ راستہ واضح کر دیا گیا، انتخاب اس کا اپنا رہا۔ شیخِ اکبر کے مکتب میں یہی سرِّ قدر ہے: آزمائش آئینہ ہے، اور آئینے میں وہی دکھائی دیتا ہے جو سامنے کھڑا ہو۔

پھر انجام بتایا: فَيَتَعَلَّمُوْنَ مِنْهُمَا مَا يُفَرِّقُوْنَ بِهٖ بَيْنَ الْمَرْءِ وَزَوْجِهٖ ۚ وَمَا هُمْ بِضَآرِّيْنَ بِهٖ مِنْ اَحَدٍ اِلَّا بِاِذْنِ اللّٰهِ ۚ وَيَتَعَلَّمُوْنَ مَا يَضُرُّهُمْ وَلَا يَنْفَعُهُمْ ۚ۔ “پھر بھی لوگ ان سے وہ چیز سیکھتے جس سے میاں اور بیوی میں جدائی ڈال دیں، حالانکہ وہ اس کے ذریعے اللہ کے اذن کے بغیر کسی کو کوئی نقصان نہیں پہنچا سکتے تھے؛ اور وہ وہ چیز سیکھتے تھے جو انہیں نقصان دیتی اور کوئی نفع نہ دیتی۔” اہلِ سنت کے نزدیک سحر کی ایک حقیقت ہے، مگر اس کی کوئی مستقل طاقت نہیں۔ یہاں اذن سے مراد اذنِ تکوینی ہے، یعنی اللہ کے پیدا کیے بغیر کوئی اثر وجود میں نہیں آتا؛ یہ رضا اور پسند کا اذن نہیں۔ اسی لیے مومن جادو سے نہیں ڈرتا بلکہ اسی کی پناہ لیتا ہے جس کے اذن کے بغیر کچھ نہیں ہوتا، جیسا کہ سورۂ الفلق نے گرہوں میں پھونکنے والیوں کے شر سے پناہ مانگنا سکھایا۔ شیخِ اکبر کے نزدیک یہ توحیدِ افعال کا درس ہے: اسباب پردے ہیں اور فاعلِ حقیقی حق ہے۔ اور دیکھیے کہ جن چیزوں کا ذکر ہوا ان میں سب سے بڑا نقصان یہ بتایا گیا کہ وہ اسے توڑتے تھے جسے اللہ نے محبت اور رحمت سے جوڑا تھا۔ اہلِ ذوق کہتے ہیں کہ جس علم کا پھل دلوں میں جدائی ہو وہ اسی نسل سے ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم کی دعا کا مفہوم، جو صحیح مسلم میں ہے، یہ ہے کہ اے اللہ، میں ایسے علم سے تیری پناہ مانگتا ہوں جو نفع نہ دے۔

آیت کا آخری حصہ ایک عجیب تضاد سامنے رکھتا ہے: وَلَقَدْ عَلِمُوْا لَمَنِ اشْتَرٰىهُ مَا لَهٗ فِي الْاٰخِرَةِ مِنْ خَلَاقٍ ۚ وَلَبِئْسَ مَا شَرَوْا بِهٖٓ اَنْفُسَهُمْ ۚ لَوْ كَانُوْا يَعْلَمُوْنَ۔ “اور وہ خوب جانتے تھے کہ جو اس کا خریدار بنا اس کے لیے آخرت میں کوئی حصہ نہیں، اور کیا ہی بری چیز ہے جس کے بدلے انہوں نے اپنی جانیں بیچ ڈالیں، کاش وہ جانتے۔” پہلے فرمایا کہ وہ جانتے تھے، پھر فرمایا کاش وہ جانتے۔ اہلِ معرفت کہتے ہیں کہ پہلا علم خبر کا علم ہے اور دوسرا وہ علم جو یقین بن کر عمل میں اتر جائے؛ سورۂ التکاثر میں علم الیقین اور عین الیقین کا جو فرق گزرا، یہ اسی کی تصویر ہے۔ اور قیمت پر غور کیجیے: انہوں نے کوئی چیز نہیں، اپنا آپ بیچ دیا۔ پھر رحمت فوراً دروازہ کھولتی ہے: وَلَوْ اَنَّهُمْ اٰمَنُوْا وَاتَّقَوْا لَمَثُوْبَةٌ مِّنْ عِنْدِ اللّٰهِ خَيْرٌ ۚ لَوْ كَانُوْا يَعْلَمُوْنَ۔ “اور اگر وہ ایمان لاتے اور تقویٰ اختیار کرتے تو اللہ کے ہاں کا بدلہ کہیں بہتر تھا، کاش وہ جانتے۔” اتنی باتوں کے بعد بھی دعوت باقی ہے؛ یہی حکم کا اصول ہے کہ حق کسی طالب کو زبردستی محروم نہیں کرتا۔ مثوبہ کا مادہ ثوب ہے، یعنی لوٹنا؛ شیخِ اکبر کے مکتب میں اجر دراصل بندے کے اپنے عمل کی حقیقت کا اس کی طرف لوٹ آنا ہے، اور اللہ کے ہاں کی قید بتاتی ہے کہ اس لوٹنے کی آخری منزل خود اسی کی بارگاہ ہے۔
