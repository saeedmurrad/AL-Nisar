پچھلے رکوع میں بنی اسرائیل سے خطاب عہد کی یاد دہانی سے شروع ہوا تھا؛ یہ رکوع اسی پکار کو دہرا کر نعمتوں اور آزمائشوں کی ایک طویل داستان کھولتا ہے: يٰبَنِيْۤ اِسْرَآءِيْلَ اذْكُرُوْا نِعْمَتِيَ الَّتِيْۤ اَنْعَمْتُ عَلَيْكُمْ وَاَنِّيْ فَضَّلْتُكُمْ عَلَى الْعٰلَمِيْنَ۔ اے بنی اسرائیل، میری اس نعمت کو یاد کرو جو میں نے تم پر فرمائی، اور اس بات کو کہ میں نے تمہیں جہان والوں پر فضیلت دی۔ آغاز ہی میں ایک ادب پیشِ نظر رہے: قرآن یہاں ایک خاص تاریخی جماعت کے خاص طرزِ عمل سے مخاطب ہے، اور اس بیان کو آج کی کسی قوم یا گروہ پر عام حکم بنا دینا اس کلام کے مزاج کے خلاف ہے۔ یہ واقعات ہر پڑھنے والے کے لیے آئینہ ہیں، اور سب سے پہلے اس امت کے لیے جو انہیں تلاوت کرتی ہے۔ جمہور مفسرین کے نزدیک یہ فضیلت اپنے زمانے کے لوگوں پر تھی اور نبوت و کتاب کی امانت سے بندھی تھی؛ یہ نسل کا کوئی دائمی امتیاز نہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر فضیلت عطا ہے، اور عطا ہمیشہ لینے والے کی استعداد کے مطابق اترتی ہے؛ مگر جس آئینے پر زیادہ نور پڑا ہو، اس سے صفائی کا تقاضا بھی زیادہ ہوتا ہے۔ اسی لیے پہلا حکم "اذْكُرُوْا" ہے۔ اہلِ معرفت کہتے ہیں کہ نعمت کی یاد دراصل منعم کی یاد ہے؛ جو نعمت کو یاد رکھے اور دینے والے کو بھول جائے، وہ نعمت ہی کو اپنا قبلہ بنا لیتا ہے، اور اسی رکوع میں آگے آئے گا کہ بھولی ہوئی نعمت کس طرح سونے کا بچھڑا بن گئی۔ یہ بھی دیکھیے کہ "نِعْمَتِيَ" واحد ہے: ساری نعمتیں ایک ہی عطا کی شاخیں ہیں، اور وہ اصل عطا وجود کی ہے جو رحمت سے ملا۔

نعمت کی یاد کے فوراً بعد اس دن کی یاد دلائی گئی جب نسبتیں کام نہیں آتیں: وَاتَّقُوْا يَوْمًا لَّا تَجْزِيْ نَفْسٌ عَنْ نَّفْسٍ شَيْـًٔا وَّلَا يُقْبَلُ مِنْهَا شَفَاعَةٌ وَّلَا يُؤْخَذُ مِنْهَا عَدْلٌ وَّلَا هُمْ يُنْصَرُوْنَ۔ اور اس دن سے بچو جب کوئی جان کسی جان کے ذرا کام نہ آئے گی، نہ اس کی طرف سے کوئی سفارش قبول کی جائے گی، نہ اس سے کوئی بدلہ لیا جائے گا، اور نہ ان کی مدد کی جائے گی۔ مفسرین نے واضح کیا ہے کہ اہلِ سنت کے نزدیک شفاعت برحق ہے، مگر حق تعالیٰ کے اذن سے، جیسا کہ آیت الکرسی میں فرمایا: مَنْ ذَا الَّذِيْ يَشْفَعُ عِنْدَهٗۤ اِلَّا بِاِذْنِهٖ۔ کون ہے جو اس کے حضور اس کی اجازت کے بغیر سفارش کرے۔ یہاں جس سفارش کی نفی ہے وہ وہ سفارش ہے جس پر لوگ ایمان و عمل سے بے پروا ہو کر بھروسا کیے بیٹھے تھے: یہ گمان کہ بزرگوں کی نسبت ہی کافی ہے۔ اہلِ معرفت کہتے ہیں کہ اس دن ہر جان اپنی حقیقت کے ساتھ کھڑی ہوگی۔ شیخِ اکبر کے نزدیک ہر بندے کا ایک عینِ ثابتہ ہے، یعنی علمِ الٰہی میں اس کی اپنی حقیقت، اور اس دن وہی حقیقت پوری طرح ظاہر ہو جائے گی؛ کسی دوسرے کا عین اس کی جگہ نہیں لے سکتا، جس طرح ایک آئینے کی صفائی دوسرے آئینے کو روشن نہیں کرتی۔ "عَدْلٌ" یہاں فدیہ اور بدلے کے معنی میں ہے: اس دن کوئی چیز جان کا بدل نہیں بنتی، کیونکہ جان کے سوا کوئی چیز جان کے برابر نہیں۔ یہ آیت سالک کو اس وہم سے نکالتی ہے کہ اس کی جگہ کوئی اور چل لے گا؛ رہنما راستہ دکھاتا ہے، قدم بندے کو خود اٹھانا ہوتا ہے۔

اب نعمتوں کی تفصیل شروع ہوتی ہے، اور پہلی نعمت غلامی سے نجات ہے: وَاِذْ نَجَّيْنٰكُمْ مِّنْ اٰلِ فِرْعَوْنَ يَسُوْمُوْنَكُمْ سُوْٓءَ الْعَذَابِ يُذَبِّحُوْنَ اَبْنَآءَكُمْ وَيَسْتَحْيُوْنَ نِسَآءَكُمْ ۭ وَفِيْ ذٰلِكُمْ بَلَآءٌ مِّنْ رَّبِّكُمْ عَظِيْمٌ۔ اور جب ہم نے تمہیں فرعون والوں سے نجات دی جو تمہیں بدترین عذاب چکھاتے تھے، تمہارے بیٹوں کو ذبح کرتے اور تمہاری عورتوں کو زندہ رہنے دیتے تھے، اور اس میں تمہارے رب کی طرف سے بڑی آزمائش تھی۔ یہ ایک تاریخی ظلم کا بیان ہے، اور قرآن اسے سیدھے لفظوں میں سناتا ہے: ایک جابر حکومت ایک کمزور قوم کی نسل مٹانے پر تلی ہوئی تھی۔ مفسرین کہتے ہیں کہ "بَلَآءٌ" کے دو معنی یہاں جمع ہیں: سختی کی آزمائش بھی، اور نجات کی نعمت بھی۔ اہلِ معرفت کہتے ہیں کہ حق تعالیٰ بندے کو دو طرح آزماتا ہے: تنگی سے، کہ صبر ظاہر ہو؛ اور کشادگی سے، کہ شکر ظاہر ہو؛ اور دوسری آزمائش اکثر پہلی سے کٹھن نکلتی ہے، جیسا کہ اس جماعت کی آگے کی داستان بتاتی ہے۔ پھر فرمایا: وَاِذْ فَرَقْنَا بِكُمُ الْبَحْرَ فَاَنْجَيْنٰكُمْ وَاَغْرَقْنَآ اٰلَ فِرْعَوْنَ وَاَنْتُمْ تَنْظُرُوْنَ۔ اور جب ہم نے تمہارے لیے سمندر کو پھاڑ دیا، پھر تمہیں بچا لیا اور فرعون والوں کو ڈبو دیا اور تم دیکھ رہے تھے۔ "بِكُمْ" کی باء پر اہلِ ذوق ٹھہرتے ہیں: سمندر تمہارے سبب اور تمہاری خاطر پھٹا، گویا اسباب کا سارا نظام ایک کمزور قافلے کے لیے راستہ دے گیا۔ اور آخری کلمہ قابلِ غور ہے: "وَاَنْتُمْ تَنْظُرُوْنَ"۔ انہوں نے حق کا فعل آنکھوں سے دیکھا، مگر آنکھ کا دیکھنا دل کے دیکھنے کی ضمانت نہیں؛ اسی لیے کچھ ہی عرصے بعد یہی دیکھنے والے بچھڑے کے گرد جمع ہو گئے۔ اہلِ معرفت کہتے ہیں کہ عین الیقین اگر دل میں جڑ نہ پکڑے تو منظر گزر جاتا ہے اور غفلت لوٹ آتی ہے۔

پھر وہ ملاقات آئی جس کی خاطر یہ سارا سفر تھا: وَاِذْ وٰعَدْنَا مُوْسٰۤى اَرْبَعِيْنَ لَيْلَةً ثُمَّ اتَّخَذْتُمُ الْعِجْلَ مِنْۢ بَعْدِهٖ وَاَنْتُمْ ظٰلِمُوْنَ۔ اور جب ہم نے موسیٰ سے چالیس راتوں کا وعدہ ٹھہرایا، پھر تم نے ان کے بعد بچھڑے کو معبود بنا لیا، اور تم ظلم کرنے والے تھے۔ حضرت موسیٰ علیہ السلام کوہِ طور پر کتاب لینے گئے اور قوم کو حضرت ہارون علیہ السلام کے سپرد کر گئے۔ "وٰعَدْنَا" باہمی وعدے کا صیغہ ہے: ملاقات کا وعدہ دونوں طرف سے تھا؛ حق نے بلایا اور بندہ حاضر ہوا۔ اہلِ معرفت کہتے ہیں کہ ہر قرب اسی طرح دوطرفہ ہے، اور حدیثِ قدسی کا مفہوم ہے کہ جو میری طرف چل کر آتا ہے میں اس کی طرف دوڑ کر آتا ہوں۔ چالیس راتوں سے اہلِ طریق نے خلوت کے اربعین کی اصل لی ہے؛ اور "لَيْلَةً" کہا گیا، دن نہیں، کیونکہ رات خلوت اور مناجات کا وقت ہے۔ مگر اسی آیت میں دوسرا منظر بھی ہے: "مِنْۢ بَعْدِهٖ"۔ رہنما کی غیر حاضری کا خلا تھا جس میں بچھڑا ابھرا۔ حضرت ہارون علیہ السلام نے اپنا فرض پوری طرح ادا کیا اور انہیں صاف بتا دیا: يٰقَوْمِ اِنَّمَا فُتِنْتُمْ بِهٖ۔ اے میری قوم، تم اس کے ذریعے آزمائش میں ڈالے گئے ہو۔ سورۂ طٰہٰ میں سامری کے بچھڑے کا ذکر گزر چکا ہے، اور وہاں یہ بھی آیا تھا کہ یہ بچھڑا قوم کے زیورات کے بوجھ سے ڈھالا گیا۔ "ظٰلِمُوْنَ" کا لفظ اس عمل کی حقیقت بتاتا ہے: ظلم کسی چیز کو اس کے مقام کے سوا کہیں رکھ دینا ہے، اور عبادت کو مخلوق کی ایک ڈھلی ہوئی صورت پر رکھ دینا سب سے بڑی بے محل نسبت ہے۔

اب اس بچھڑے کو اندر کی آنکھ سے دیکھیے، یہ یاد رکھتے ہوئے کہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ تاویل کہتے ہیں کہ ہر سالک کے اندر ایک صورت گر بیٹھا ہے جو اس کی اپنی چمکتی چیزیں، یعنی علم، عبادت کی شہرت، مرتبہ اور کرامت کی پسند، خواہش کی آگ میں پگھلا کر ایک صورت ڈھال لیتا ہے۔ سورۂ اعراف میں اس صورت کا وصف آیا ہے: عِجْلًا جَسَدًا لَّهٗ خُوَارٌ۔ ایک بچھڑا، نرا دھڑ، جس میں سے گائے کی سی آواز نکلتی تھی۔ جسم ہے، آواز ہے، مگر روح نہیں۔ نفس کا بچھڑا بھی ایسا ہی ہوتا ہے: اس میں دین کی آواز سنائی دیتی ہے، مگر رخ اپنی ذات کی طرف ہوتا ہے۔ اور یہ ہمیشہ "مِنْۢ بَعْدِهٖ" بنتا ہے، یعنی اس وقت جب بندہ صحبت اور ذکر سے دور ہو جائے۔ پھر بھی رحمت کا دروازہ بند نہ ہوا: ثُمَّ عَفَوْنَا عَنْكُمْ مِّنْۢ بَعْدِ ذٰلِكَ لَعَلَّكُمْ تَشْكُرُوْنَ۔ پھر ہم نے اس کے بعد بھی تمہیں معاف کر دیا تاکہ تم شکر گزار بنو۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا مظہر ہے: رحمت ہر چیز سے پہلے ہے، گناہ سے بھی پہلے، اس لیے گناہ اسے پیچھے نہیں چھوڑ سکتا۔ اور معافی کا مقصد شکر بتایا گیا۔ اہلِ معرفت کہتے ہیں کہ بچھڑے کی پرستش دراصل شکر کی الٹی صورت تھی: نعمت کا سونا لے کر اسی کو معبود بنا لینا۔ شکر یہ ہے کہ نعمت میں دینے والے کو دیکھا جائے؛ اور معافی اسی لیے ملی کہ آنکھ نعمت سے ہٹ کر منعم پر ٹک جائے۔

بچھڑے کے قصے کے عین درمیان یہ آیت رکھی گئی: وَاِذْ اٰتَيْنَا مُوْسَى الْكِتٰبَ وَالْفُرْقَانَ لَعَلَّكُمْ تَهْتَدُوْنَ۔ اور جب ہم نے موسیٰ کو کتاب اور فرقان عطا کیا تاکہ تم ہدایت پاؤ۔ مفسرین کہتے ہیں کہ فرقان سے مراد وہی تورات ہے جو حق و باطل میں فرق کرتی ہے، اور بعض نے اسے وہ معجزات اور دلائل کہا جو سچ کو جھوٹ سے جدا کر دیتے ہیں۔ اس ترتیب میں ایک لطیف علاج پوشیدہ ہے۔ بچھڑے کی پرستش کی جڑ التباس تھی: خالق کی عبادت کو مخلوق کی ایک صورت سے گڈمڈ کر دینا۔ اس کا علاج فرقان ہے، یعنی جدا کر کے دیکھنے والی نظر۔ شیخِ اکبر کے مکتب میں قرآن اور فرقان دو مقاموں کے نام بھی ہیں: قرآن جمع کا مقام ہے جس میں بندہ ہر شے کو حق کی نشانی اور اس کی تجلی کا مظہر دیکھتا ہے، اور فرقان فرق کا مقام ہے جس میں وہ رب کو رب اور بندے کو بندہ جانتا ہے۔ کامل معرفت دونوں کو ساتھ رکھتی ہے؛ جس کے پاس فرق نہ ہو وہ صورت کو معبود بنا بیٹھتا ہے، اور جس کے پاس جمع نہ ہو وہ عالم کو نشانیوں سے خالی سمجھ لیتا ہے۔ اہلِ معرفت کہتے ہیں کہ یہی تنزیہ و تشبیہ کا جمع ہے، اور بچھڑے والوں نے تنزیہ کھو کر ایک صورت کو بت بنا لیا۔ پھر "لَعَلَّكُمْ تَهْتَدُوْنَ"؛ کتاب دے دی گئی، راستہ روشن کر دیا گیا، اب ہدایت اسی کو ملتی ہے جو اس روشنی کی طرف رخ کرے۔ جو طلب کرے اسے ملتی ہے؛ جو منہ پھیرے، اس کا منہ پھیرنا ہی اس کی محرومی ہے۔

پھر حضرت موسیٰ علیہ السلام لوٹے اور قوم سے فرمایا: وَاِذْ قَالَ مُوْسٰى لِقَوْمِهٖ يٰقَوْمِ اِنَّكُمْ ظَلَمْتُمْ اَنْفُسَكُمْ بِاتِّخَاذِكُمُ الْعِجْلَ فَتُوْبُوْۤا اِلٰى بَارِئِكُمْ فَاقْتُلُوْۤا اَنْفُسَكُمْ ۭ ذٰلِكُمْ خَيْرٌ لَّكُمْ عِنْدَ بَارِئِكُمْ ۭ فَتَابَ عَلَيْكُمْ ۭ اِنَّهٗ هُوَ التَّوَّابُ الرَّحِيْمُ۔ اور جب موسیٰ نے اپنی قوم سے کہا: اے میری قوم، تم نے بچھڑے کو معبود بنا کر اپنی جانوں پر ظلم کیا ہے، تو اپنے پیدا کرنے والے کی طرف توبہ کرو اور اپنے آپ کو قتل کرو؛ یہ تمہارے پیدا کرنے والے کے نزدیک تمہارے لیے بہتر ہے؛ پھر اس نے تمہاری توبہ قبول کر لی، بے شک وہی بہت توبہ قبول کرنے والا، نہایت مہربان ہے۔ خطاب "يٰقَوْمِ" سے شروع ہوا: سخت ترین لمحے میں بھی نبی کی زبان پر اپنائیت ہے۔ ظاہر میں جمہور مفسرین کے نزدیک یہ حکم اس شریعت میں اس خاص جرم کی سزا اور توبہ کی شرط تھا، اور منقول ہے کہ جو بچھڑے کی پرستش سے بچے رہے انہیں قصورواروں پر یہ حکم نافذ کرنے کو کہا گیا؛ پھر حق تعالیٰ نے توبہ قبول فرما کر اسے روک دیا۔ ہماری شریعت میں صاف ممانعت ہے: وَلَا تَقْتُلُوْۤا اَنْفُسَكُمْ۔ اور اپنے آپ کو قتل نہ کرو۔ اس لیے اشاری معنی کسی طرح جسم کو نقصان پہنچانے کی اجازت نہیں دیتا۔ اہلِ معرفت کے نزدیک باطن میں جس چیز کا قتل مطلوب ہے وہ نفس کا دعویٰ ہے، یعنی "میں" کا وہ بت جو بچھڑے ڈھالتا ہے؛ اسی کو فنا کہتے ہیں، جو جسم کی نہیں، دعوے کی موت ہے۔ اسمِ الٰہی کا انتخاب دیکھیے: "بَارِئِكُمْ"۔ باری وہ ہے جو بغیر کسی نمونے کے صورت عطا کرے۔ تم نے اپنے ہاتھ سے ایک صورت ڈھالی؛ لوٹو اس کی طرف جس نے خود تمہیں ڈھالا۔ اور آخر میں "فَتَابَ عَلَيْكُمْ"۔ اہلِ ذوق کہتے ہیں کہ بندے کی توبہ سے پہلے حق کی توجہ ہوتی ہے، جیسا کہ سورۂ توبہ میں فرمایا: ثُمَّ تَابَ عَلَيْهِمْ لِيَتُوْبُوْا۔ پھر اس نے ان پر رجوع فرمایا تاکہ وہ رجوع کریں۔

اس رکوع کی اگلی لغزش ایک اور طرح کی ہے: وَاِذْ قُلْتُمْ يٰمُوْسٰى لَنْ نُّؤْمِنَ لَكَ حَتّٰى نَرَى اللّٰهَ جَهْرَةً فَاَخَذَتْكُمُ الصّٰعِقَةُ وَاَنْتُمْ تَنْظُرُوْنَ۔ اور جب تم نے کہا: اے موسیٰ، ہم تمہاری بات ہرگز نہ مانیں گے جب تک اللہ کو کھلم کھلا نہ دیکھ لیں؛ تو تمہیں کڑک نے آ پکڑا اور تم دیکھ رہے تھے۔ بہت سے مفسرین کہتے ہیں کہ یہ وہ منتخب لوگ تھے جو حضرت موسیٰ علیہ السلام کے ساتھ میقات پر گئے تھے۔ ان کا قصور دیدار کی آرزو نہ تھا؛ قصور یہ تھا کہ ایمان کو دیدار کی شرط سے باندھ دیا، گویا حق کو اپنی شرطوں کا پابند بنانا چاہا۔ خود حضرت موسیٰ علیہ السلام نے بھی عرض کیا تھا: رَبِّ اَرِنِيْۤ اَنْظُرْ اِلَيْكَ۔ اے میرے رب، مجھے دکھا کہ میں تجھے دیکھوں۔ مگر وہ عرض شوق اور ادب سے تھی، شرط اور مطالبے سے نہیں۔ شیخِ اکبر کے نزدیک تجلی دیکھنے والے کے ظرف کے مطابق ہوتی ہے؛ جو ظرف تیار نہ ہو وہ تجلی کی تاب نہیں لاتا، اور یہی "الصّٰعِقَةُ" کا اشاری راز ہے۔ آخرت میں اہلِ ایمان کو دیدار نصیب ہوگا، جیسا کہ سورۂ قیامہ میں گزرا، مگر وہ عطا ہے، مطالبہ نہیں۔ پھر رحمت لوٹی: ثُمَّ بَعَثْنٰكُمْ مِّنْۢ بَعْدِ مَوْتِكُمْ لَعَلَّكُمْ تَشْكُرُوْنَ۔ پھر ہم نے تمہاری موت کے بعد تمہیں اٹھا کھڑا کیا تاکہ تم شکر کرو۔ جمہور کے نزدیک یہ حقیقی موت کے بعد حقیقی زندگی تھی۔ اہلِ ذوق اس میں یہ اشارہ بھی دیکھتے ہیں کہ جس کا دعویٰ تجلی کی کڑک سے مر جائے، اسے نئی زندگی ملتی ہے؛ اور شیخِ اکبر کے مکتب میں یہی تجددِ امثال کا ذوق ہے کہ ہر لمحہ پرانا مٹتا اور نیا اٹھتا ہے۔

پھر صحرا کی نعمتیں گنوائی گئیں: وَظَلَّلْنَا عَلَيْكُمُ الْغَمَامَ وَاَنْزَلْنَا عَلَيْكُمُ الْمَنَّ وَالسَّلْوٰى ۭ كُلُوْا مِنْ طَيِّبٰتِ مَا رَزَقْنٰكُمْ ۭ وَمَا ظَلَمُوْنَا وَلٰكِنْ كَانُوْۤا اَنْفُسَهُمْ يَظْلِمُوْنَ۔ اور ہم نے تم پر بادلوں کا سایہ کیا اور تم پر من و سلویٰ اتارا؛ ہماری دی ہوئی پاکیزہ چیزیں کھاؤ؛ اور انہوں نے ہمارا کچھ نہیں بگاڑا، بلکہ وہ اپنی ہی جانوں پر ظلم کرتے رہے۔ صحرائے سینا کی تپتی دھوپ میں بادل سائبان بنے، اور رزق بغیر کھیتی اور محنت کے اترا۔ مفسرین کے نزدیک من ایک میٹھی چیز تھی جو شبنم کی طرح اترتی، اور سلویٰ ایک پرندہ تھا۔ اہلِ معرفت کہتے ہیں کہ یہ بے واسطہ رزق کی تربیت تھی: قوم کو دکھایا گیا کہ رزق کا اصل سرچشمہ اسباب نہیں، رزاق ہے۔ شیخِ اکبر کے مکتب میں سارا عالم حق کا ظل ہے، یعنی اس کے وجود کا سایہ؛ اور "وَظَلَّلْنَا" میں یہ اشارہ ملتا ہے کہ بندے پر پڑنے والا ہر سایہ دراصل اسی کی رحمت کا سایہ ہے۔ پھر آیت کا آخری حصہ اس سورت کے بنیادی اصول کی صاف گواہی ہے: "وَمَا ظَلَمُوْنَا وَلٰكِنْ كَانُوْۤا اَنْفُسَهُمْ يَظْلِمُوْنَ"۔ نافرمانی حق کا کچھ نہیں بگاڑتی؛ نقصان اسی جان کا ہوتا ہے جو منہ پھیرتی ہے۔ یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہے، اور جو اندھیرا بندے پر آتا ہے وہ اس کے اپنے رخ کا سایہ ہے، اوپر سے مسلط کی ہوئی کوئی زبردستی نہیں۔ اسی لیے آیت نے "اَنْفُسَهُمْ" کو فعل سے پہلے رکھا: ظلم کا نشانہ صرف وہی تھے۔ اور خطاب سے غائب کی طرف یہ پلٹنا بھی معنی رکھتا ہے: جو نعمت کے سائے میں رہ کر منہ پھیرے، وہ گویا حضوری کی مجلس سے خود اٹھ جاتا ہے۔

پھر شہر میں داخلے کا حکم آیا: وَاِذْ قُلْنَا ادْخُلُوْا هٰذِهِ الْقَرْيَةَ فَكُلُوْا مِنْهَا حَيْثُ شِئْتُمْ رَغَدًا وَّادْخُلُوا الْبَابَ سُجَّدًا وَّقُوْلُوْا حِطَّةٌ نَّغْفِرْ لَكُمْ خَطٰيٰكُمْ ۭ وَسَنَزِيْدُ الْمُحْسِنِيْنَ۔ اور جب ہم نے کہا: اس بستی میں داخل ہو جاؤ اور اس میں جہاں سے چاہو فراغت سے کھاؤ، اور دروازے میں جھکتے ہوئے داخل ہو اور کہو کہ ہمارے بوجھ اتار دے؛ ہم تمہاری خطائیں بخش دیں گے، اور نیکی کرنے والوں کو اور زیادہ دیں گے۔ اکثر مفسرین کے نزدیک یہ بستی بیت المقدس یا اس کے نواح کی تھی۔ حکم دو باتوں کا تھا: بدن جھکا ہوا ہو اور زبان پر عاجزی کا کلمہ۔ اہلِ معرفت کہتے ہیں کہ ہر نئی منزل کا دروازہ اسی طرح کھلتا ہے: فتح کے لمحے میں بندہ سر اٹھا کر نہیں، جھکا کر داخل ہو۔ اس کی کامل مثال حضورِ اکرم صلی اللہ علیہ وسلم کا فتحِ مکہ کے دن سر جھکائے داخل ہونا ہے، جیسا کہ سیرت میں منقول ہے۔ "وَسَنَزِيْدُ الْمُحْسِنِيْنَ" میں احسان کا وعدہ ہے، اور حدیثِ احسان کا مفہوم ہے کہ اللہ کی عبادت ایسے کرو گویا اسے دیکھ رہے ہو۔ مگر کیا ہوا؟ فَبَدَّلَ الَّذِيْنَ ظَلَمُوْا قَوْلًا غَيْرَ الَّذِيْ قِيْلَ لَهُمْ فَاَنْزَلْنَا عَلَى الَّذِيْنَ ظَلَمُوْا رِجْزًا مِّنَ السَّمَآءِ بِمَا كَانُوْا يَفْسُقُوْنَ۔ تو ظالموں نے اس بات کو جو ان سے کہی گئی تھی ایک اور بات سے بدل دیا، تو ہم نے ظالموں پر آسمان سے عذاب اتارا اس لیے کہ وہ نافرمانی کرتے تھے۔ صحیح بخاری کی روایت کا مفہوم ہے کہ انہوں نے جھکنے کے بجائے گھسٹتے ہوئے داخلہ کیا اور کلمۂ عاجزی کو ایک ہلکے بے معنی لفظ سے بدل دیا۔ آیت نے "الَّذِيْنَ ظَلَمُوْا" کہہ کر خود حد کھینچ دی: سزا صرف بدلنے والوں پر تھی، سب پر نہیں۔ اور "بِمَا كَانُوْا يَفْسُقُوْنَ" نے سبب بتا دیا: عذاب ان کے اپنے فسق کا ثمر تھا، کوئی ناگہانی جبر نہیں۔ اہلِ ذوق کہتے ہیں کہ نفس کی پرانی عادت یہی ہے: حکم کا لفظ رکھ لیتا ہے اور اس کی روح بدل دیتا ہے؛ اور ہر پڑھنے والا اپنے دل سے پوچھے کہ اس نے کس "حِطَّةٌ" کو کس لفظ سے بدل رکھا ہے۔
