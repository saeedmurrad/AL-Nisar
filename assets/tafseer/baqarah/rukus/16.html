پچھلا رکوع خلیل کی دعا پر ختم ہوا تھا؛ یہ رکوع ایک سوال سے کھلتا ہے کہ ایسی ملت سے کون منہ موڑ سکتا ہے: وَمَنْ يَّرْغَبُ عَنْ مِّلَّةِ اِبْرٰهٖمَ اِلَّا مَنْ سَفِهَ نَفْسَهٗ ۚ وَلَقَدِ اصْطَفَيْنٰهُ فِي الدُّنْيَا ۚ وَاِنَّهٗ فِي الْاٰخِرَةِ لَمِنَ الصّٰلِحِيْنَ۔ “اور ابراہیم کی ملت سے کون بے رغبتی کرے گا، سوائے اس کے جس نے خود کو نادان بنا لیا؟ اور ہم نے اسے دنیا میں چن لیا، اور آخرت میں وہ یقیناً صالحین میں سے ہے۔” مفسرین "سَفِهَ نَفْسَهٗ" کے کئی معنی بیان کرتے ہیں: اپنی جان کو ہلکا اور بے قدر کر دیا، یا اپنی ذات سے ناواقف رہا۔ دوسرا معنی اہلِ معرفت کے ہاں بہت گہرا ہے۔ اہلِ تصوف میں مشہور اثر ہے کہ جس نے اپنے نفس کو پہچانا اس نے اپنے رب کو پہچانا؛ تو جو اپنی ذات سے نادان رہا وہ اپنے رب سے بھی نادان رہا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ملتِ ابراہیم کوئی باہر سے لادا ہوا بوجھ نہیں، انسان کی اصل حقیقت کی طرف واپسی ہے؛ اس لیے اس سے منہ موڑنا گویا اپنے آپ سے منہ موڑنا ہے۔ اور اس ملت کی جڑ کیا ہے؟ اگلی آیت بتاتی ہے: اِذْ قَالَ لَهٗ رَبُّهٗۤ اَسْلِمْ ۙ قَالَ اَسْلَمْتُ لِرَبِّ الْعٰلَمِيْنَ۔ “جب اس کے رب نے اس سے فرمایا: فرماں بردار ہو جا، تو اس نے عرض کیا: میں سارے جہانوں کے رب کا فرماں بردار ہو گیا۔” حکم اور جواب کے درمیان کوئی وقفہ نہیں، کوئی سوال نہیں، کوئی شرط نہیں۔ اور غور کیجیے کہ جواب میں "تیرا فرماں بردار" نہیں کہا بلکہ "لِرَبِّ الْعٰلَمِيْنَ" کہا۔ اہلِ ذوق کہتے ہیں کہ خلیل نے اپنی تسلیم کو سارے جہانوں کے رب کے سپرد کیا، یعنی انہوں نے ہر جہان میں اسی کی ربوبیت دیکھی: آگ میں بھی، چھری میں بھی، بے آب وادی میں بھی۔ اسی لیے آگ ان کے لیے گلزار ہوئی؛ جو ہر شے میں رب کو دیکھ لے، اس کے لیے کوئی شے اجنبی نہیں رہتی۔

یہ کلمہ صرف زبان کا اقرار نہ رہا، نسلوں کی امانت بن گیا: وَوَصّٰى بِهَاۤ اِبْرٰهٖمُ بَنِيْهِ وَيَعْقُوْبُ ۚ يٰبَنِيَّ اِنَّ اللّٰهَ اصْطَفٰى لَكُمُ الدِّيْنَ فَلَا تَمُوْتُنَّ اِلَّا وَاَنْتُمْ مُّسْلِمُوْنَ۔ “اور اسی بات کی وصیت ابراہیم نے اپنے بیٹوں کو کی اور یعقوب نے بھی: اے میرے بیٹو! اللہ نے تمہارے لیے یہی دین چن لیا ہے، تو تمہیں موت نہ آئے مگر اس حال میں کہ تم فرماں بردار ہو۔” وصیت مال کی نہیں، اسلام کی ہے؛ اور دین کو "اصْطَفٰى" کہا گیا، یعنی یہ بندوں کی ایجاد نہیں، حق کا چنا ہوا تحفہ ہے۔ پھر یہ حکم کہ موت اسلام ہی کی حالت میں آئے، ایک عجیب نکتہ رکھتا ہے: موت کا وقت بندے کے اختیار میں نہیں، تو اس حکم کا عملی مطلب یہ ہوا کہ ہر لمحہ اسلام میں گزرے، کیونکہ کوئی بھی لمحہ آخری ہو سکتا ہے۔ شیخِ اکبر کے مکتب میں تجددِ امثال کی رو سے ہر آن ایک نئی تخلیق ہے، اور ہر آن ایک نیا انجام بھی؛ اس لیے سالک ہر سانس کو آخری سانس جان کر تسلیم کی تجدید کرتا ہے۔ اہلِ تصوف میں مشہور اثر ہے: مرنے سے پہلے مر جاؤ؛ اور اہلِ ذوق اس آیت میں اسی کی طرف اشارہ دیکھتے ہیں کہ جس کی خودی تسلیم میں فنا ہو چکی ہو، اس کی موت جب بھی آئے اسلام ہی پر آتی ہے۔ مگر یہ اشاری معنی ظاہر کی نفی نہیں؛ ظاہر حکم یہی ہے کہ آدمی ایمان پر جیے اور ایمان پر مرے، اور یہی اصل ہے۔

پھر منظر بدلتا ہے اور ہم حضرت یعقوب علیہ السلام کے بسترِ مرگ کے پاس کھڑے ہیں: اَمْ كُنْتُمْ شُهَدَآءَ اِذْ حَضَرَ يَعْقُوْبَ الْمَوْتُ ۙ اِذْ قَالَ لِبَنِيْهِ مَا تَعْبُدُوْنَ مِنْۢ بَعْدِيْ۔ “کیا تم اس وقت موجود تھے جب یعقوب کو موت آئی، جب اس نے اپنے بیٹوں سے پوچھا: میرے بعد تم کس کی عبادت کرو گے؟” رخصت ہوتے باپ کی فکر یہ نہیں کہ اولاد کیا کھائے گی، فکر یہ ہے کہ کس کو پوجے گی۔ قَالُوْا نَعْبُدُ اِلٰهَكَ وَاِلٰهَ اٰبَآئِكَ اِبْرٰهٖمَ وَاِسْمٰعِيْلَ وَاِسْحٰقَ اِلٰهًا وَّاحِدًا ۚ وَّنَحْنُ لَهٗ مُسْلِمُوْنَ۔ “انہوں نے کہا: ہم آپ کے معبود اور آپ کے آبا ابراہیم، اسماعیل اور اسحاق کے معبود کی عبادت کریں گے، جو ایک ہی معبود ہے، اور ہم اسی کے فرماں بردار ہیں۔” مفسرین کہتے ہیں کہ حضرت اسماعیل علیہ السلام چچا تھے، مگر عرب چچا کو بھی باپ کہتے ہیں، اور صحیح مسلم کی روایت کا مفہوم ہے کہ آدمی کا چچا اس کے باپ کی مانند ہے۔ اب اس مکتب کا ذوق دیکھیے: فصوص الحکم میں ان چاروں انبیاء علیہم السلام کی حکمتیں الگ الگ ہیں: حضرت ابراہیم کی حکمتِ مہیمیہ، حضرت اسماعیل کی حکمتِ علیہ، حضرت اسحاق کی حکمتِ حقیہ اور حضرت یعقوب کی حکمتِ روحیہ۔ ہر نبی میں اسماءِ الٰہیہ کا ایک الگ رنگ ظاہر ہوا، مگر بیٹوں نے سب کو گنوا کر کہا: "اِلٰهًا وَّاحِدًا"۔ مظاہر بہت، معبود ایک؛ یہی وحدت میں کثرت اور کثرت میں وحدت کا راز ہے۔ اور پھر وہ قاعدہ: تِلْكَ اُمَّةٌ قَدْ خَلَتْ ۚ لَهَا مَا كَسَبَتْ وَلَكُمْ مَّا كَسَبْتُمْ۔ “وہ ایک جماعت تھی جو گزر چکی؛ اس کے لیے اس کی کمائی ہے اور تمہارے لیے تمہاری کمائی۔” بزرگوں کی بزرگی ان کی کمائی ہے؛ تمہیں اپنی کمائی خود کرنی ہے۔

اب وہ دعوے سامنے آتے ہیں جو اُس زمانے میں کیے جاتے تھے: وَقَالُوْا كُوْنُوْا هُوْدًا اَوْ نَصٰرٰى تَهْتَدُوْا ۚ قُلْ بَلْ مِلَّةَ اِبْرٰهٖمَ حَنِيْفًا ۚ وَمَا كَانَ مِنَ الْمُشْرِكِيْنَ۔ “اور انہوں نے کہا: یہودی یا نصرانی ہو جاؤ تو ہدایت پا جاؤ گے۔ کہہ دیجیے: بلکہ ابراہیم کی ملت، جو یکسو تھے، اور وہ مشرکوں میں سے نہ تھے۔” یہ ایک خاص تاریخی گفتگو کا جواب ہے جو مدینہ میں ہوئی؛ اسے کسی قوم کے بارے میں عمومی حکم بنانا قرآن کے انصاف کے خلاف ہے۔ جواب کی روح یہ ہے کہ ہدایت کسی نام یا گروہ کی ملکیت نہیں، اس ملت کا نام ہے جو گروہ بندی سے پہلے تھی۔ حنیف وہ ہے جو ہر طرف سے ہٹ کر ایک کی طرف جھک جائے؛ اہلِ معرفت کہتے ہیں کہ حنیفیت دل کا وہ میلان ہے جو ماسوا سے پھر کر حق کی طرف ٹھہر جائے۔ پھر امت کو اس کا کلمہ سکھایا گیا: قُوْلُوْۤا اٰمَنَّا بِاللّٰهِ وَمَاۤ اُنْزِلَ اِلَيْنَا وَمَاۤ اُنْزِلَ اِلٰۤى اِبْرٰهٖمَ وَاِسْمٰعِيْلَ وَاِسْحٰقَ وَيَعْقُوْبَ وَالْاَسْبَاطِ وَمَاۤ اُوْتِيَ مُوْسٰى وَعِيْسٰى وَمَاۤ اُوْتِيَ النَّبِيُّوْنَ مِنْ رَّبِّهِمْ ۚ لَا نُفَرِّقُ بَيْنَ اَحَدٍ مِّنْهُمْ ۚ وَنَحْنُ لَهٗ مُسْلِمُوْنَ۔ “کہو: ہم ایمان لائے اللہ پر، اور اس پر جو ہماری طرف اتارا گیا، اور جو ابراہیم، اسماعیل، اسحاق، یعقوب اور ان کی اولاد کی طرف اتارا گیا، اور جو موسیٰ اور عیسیٰ کو دیا گیا، اور جو سب نبیوں کو ان کے رب کی طرف سے دیا گیا؛ ہم ان میں سے کسی کے درمیان فرق نہیں کرتے، اور ہم اسی کے فرماں بردار ہیں۔” یہ امتِ محمدیہ کا امتیاز ہے کہ وہ سب انبیاء علیہم السلام پر ایمان رکھتی ہے۔ شیخِ اکبر کے نزدیک ہر نبی حق کا ایک کلمہ ہے جس میں ایک خاص حکمت ظاہر ہوئی، اور حضورِ اکرم صلی اللہ علیہ وسلم ان سب حکمتوں کے جامع ہیں؛ اسی لیے آپ کی امت کسی نبی کا انکار نہیں کر سکتی، کہ ایک کا انکار اس جامعیت کو توڑنا ہے۔

فَاِنْ اٰمَنُوْا بِمِثْلِ مَاۤ اٰمَنْتُمْ بِهٖ فَقَدِ اهْتَدَوْا ۚ وَاِنْ تَوَلَّوْا فَاِنَّمَا هُمْ فِيْ شِقَاقٍ ۚ فَسَيَكْفِيْكَهُمُ اللّٰهُ ۚ وَهُوَ السَّمِيْعُ الْعَلِيْمُ۔ “پھر اگر وہ بھی اسی طرح ایمان لائیں جیسے تم ایمان لائے ہو تو یقیناً ہدایت پا گئے؛ اور اگر منہ پھیریں تو وہ بس مخالفت میں پڑے ہیں؛ سو ان کے مقابلے میں اللہ تمہیں کافی ہے، اور وہی سننے والا، جاننے والا ہے۔” یہ آیت ہدایت اور گمراہی کے اس اصول کو صاف لفظوں میں بیان کرتی ہے جس پر یہ پورا سلسلہ قائم ہے۔ ہدایت کا دروازہ کسی پر بند نہیں: جو بھی ایمان لائے، "فَقَدِ اهْتَدَوْا"، وہ ہدایت پا گیا۔ اور گمراہی کیا ہے؟ آیت اسے کسی بیرونی جبر کا نام نہیں دیتی؛ وہ کہتی ہے "وَاِنْ تَوَلَّوْا"، اگر وہ خود منہ پھیر لیں۔ منہ پھیرنا ہی شقاق ہے، اور یہی ان کی گمراہی ہے۔ اسی سورت کی چھبیسویں آیت نے یہ اصول پہلے ہی طے کر دیا تھا: وَمَا يُضِلُّ بِهٖۤ اِلَّا الْفٰسِقِيْنَ؛ یعنی گمراہی انہی کے حصے میں آتی ہے جو پہلے ہی اطاعت کے دائرے سے نکل چکے ہوں۔ اللہ کسی ہدایت کے طالب کو زبردستی گمراہ نہیں کرتا؛ جو مانگتا ہے پاتا ہے، اور جو منہ موڑتا ہے اس کا منہ موڑنا ہی اس کی محرومی ہے۔ شیخِ اکبر کے نزدیک یہ سرِّ قدر ہے: حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ پھر حضورِ اکرم صلی اللہ علیہ وسلم کو تسلی: "فَسَيَكْفِيْكَهُمُ اللّٰهُ"۔ اہلِ ذوق کہتے ہیں کہ سالک کا کام انکار کرنے والوں سے الجھنا نہیں، اپنی راہ پر ٹھہرے رہنا ہے؛ جو حق پر بھروسہ کرے، حق اس کی طرف سے کافی ہو جاتا ہے۔

اب اس رکوع کا مرکزی جملہ: صِبْغَةَ اللّٰهِ ۚ وَمَنْ اَحْسَنُ مِنَ اللّٰهِ صِبْغَةً۔ “اللہ کا رنگ (اختیار کرو)، اور اللہ کے رنگ سے بہتر کس کا رنگ ہے؟” صبغہ رنگنے کو کہتے ہیں۔ مفسرین بیان کرتے ہیں کہ اُس زمانے میں نصاریٰ اپنے نومولود بچوں کو ایک خاص پانی میں غوطہ دیتے اور سمجھتے کہ اب وہ پاک اور ان کے دین میں رنگ گیا؛ قرآن نے فرمایا کہ اصل رنگ وہ ہے جو اللہ چڑھاتا ہے، پانی کا نہیں، فطرت اور ایمان کا۔ اکثر مفسرین کے نزدیک صبغۃ اللہ سے مراد اللہ کا دین ہے، اور بعض کے نزدیک وہ فطرت جس پر انسان پیدا کیا گیا، جس کا ذکر سورۂ الروم میں ہے: فِطْرَتَ اللّٰهِ الَّتِيْ فَطَرَ النَّاسَ عَلَيْهَا۔ صحیح بخاری کی روایت کا مفہوم ہے کہ ہر بچہ فطرت پر پیدا ہوتا ہے، پھر اس کے ماں باپ اسے دوسرے رنگ میں رنگ دیتے ہیں۔ یوں یہ دونوں قول ایک ہی حقیقت کے دو رخ ہیں: دین اسی فطرت کی یاد دہانی ہے جو پہلے سے دل کی تہ میں موجود ہے۔ رنگ کی مثال کیوں دی گئی؟ اس لیے کہ رنگ چیز کی سطح پر نہیں رہتا، تار تار میں اتر جاتا ہے؛ رنگا ہوا کپڑا ہر طرف سے ایک ہی رنگ دکھاتا ہے۔ اسی طرح ایمان جب سچا ہو تو آدمی کی بات، نظر، چال اور خاموشی سب ایک ہی رنگ میں رنگی ہوتی ہیں۔ اور "وَمَنْ اَحْسَنُ" میں حدیثِ احسان کی گونج سنائی دیتی ہے: سب سے اچھا رنگ وہ ہے جس میں بندہ یوں عبادت کرے گویا اپنے رب کو دیکھ رہا ہو۔

شیخِ اکبر کے مکتب میں اللہ کے رنگ کا مطلب تخلق باسماء اللہ ہے: بندے پر اسماءِ الٰہیہ کا رنگ چڑھ جائے، وہ رحیم کے رنگ سے رحم کرے، حلیم کے رنگ سے برداشت کرے، کریم کے رنگ سے دے۔ مگر یہ رنگ صفات کا رنگ ہے، ذات کا نہیں؛ رنگا ہوا کپڑا رنگریز نہیں بن جاتا۔ حضرت جنید بغدادی رحمۃ اللہ علیہ سے منسوب ایک مشہور قول ہے، جسے شیخِ اکبر بھی نقل کرتے ہیں، کہ پانی کا رنگ اس کے برتن کا رنگ ہوتا ہے۔ اہلِ ذوق اس سے یہ سمجھتے ہیں کہ جو دل اپنے ذاتی رنگ سے خالی ہو جائے وہ ہر تجلی کا رنگ قبول کر لیتا ہے؛ گویا اللہ کا سب سے گہرا رنگ بے رنگی ہے، یعنی اپنی پسند اور اپنی ضد کا رنگ اتر جانا۔ اسی لیے شیخِ اکبر کے نزدیک کامل کی تمکین تلوین ہی میں ہے: وہ ہر آن نئی تجلی کے ساتھ نیا رنگ قبول کرتا ہے، مگر حق کی طرف رخ ایک ہی رہتا ہے۔ پھر آیت کا اختتام اس رنگ کی پہچان بتاتا ہے: وَّنَحْنُ لَهٗ عٰبِدُوْنَ۔ “اور ہم اسی کی عبادت کرنے والے ہیں۔” اللہ کے رنگ کی نشانی دعویٰ نہیں، عبدیت ہے۔ جتنا گہرا رنگ چڑھتا ہے، بندہ اتنا ہی جھکتا ہے؛ اور اگر کسی رنگ سے آدمی میں بڑائی پیدا ہو تو وہ اللہ کا رنگ نہیں، نفس کا رنگ ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں؛ ظاہر یہی ہے کہ اللہ کا دین سب سے اچھا رنگ ہے، اور باطن اسی ظاہر کی گہرائی ہے۔

قُلْ اَتُحَآجُّوْنَنَا فِي اللّٰهِ وَهُوَ رَبُّنَا وَرَبُّكُمْ ۚ وَلَنَاۤ اَعْمَالُنَا وَلَكُمْ اَعْمَالُكُمْ ۚ وَنَحْنُ لَهٗ مُخْلِصُوْنَ۔ “کہہ دیجیے: کیا تم اللہ کے بارے میں ہم سے جھگڑتے ہو، حالانکہ وہی ہمارا رب بھی ہے اور تمہارا رب بھی؟ ہمارے لیے ہمارے اعمال ہیں اور تمہارے لیے تمہارے اعمال، اور ہم اسی کے لیے خالص ہیں۔” اللہ کسی ایک گروہ کی جاگیر نہیں؛ وہ سب کا رب ہے، اور فیصلہ نسبت سے نہیں، عمل اور اخلاص سے ہوگا۔ اب ذرا ان تین لفظوں کی ترتیب دیکھیے جن پر ان آیات کے جملے ختم ہوئے: پہلے "مُسْلِمُوْنَ"، پھر "عٰبِدُوْنَ"، پھر "مُخْلِصُوْنَ"۔ اہلِ معرفت اس میں مقامات کی سیڑھی دیکھتے ہیں: پہلے تسلیم، پھر عبدیت، پھر اخلاص؛ یعنی پہلے بندہ جھکتا ہے، پھر بندگی اس کی عادت بنتی ہے، پھر اس بندگی سے غیر کا ہر شائبہ نکل جاتا ہے۔ پھر سوال: اَمْ تَقُوْلُوْنَ اِنَّ اِبْرٰهٖمَ وَاِسْمٰعِيْلَ وَاِسْحٰقَ وَيَعْقُوْبَ وَالْاَسْبَاطَ كَانُوْا هُوْدًا اَوْ نَصٰرٰى ۚ قُلْ ءَاَنْتُمْ اَعْلَمُ اَمِ اللّٰهُ۔ “یا تم کہتے ہو کہ ابراہیم، اسماعیل، اسحاق، یعقوب اور ان کی اولاد یہودی یا نصرانی تھے؟ کہہ دیجیے: کیا تم زیادہ جانتے ہو یا اللہ؟” یہ نام تو بعد میں پڑے، وہ انبیاء ان سے پہلے گزر چکے تھے؛ تو انہیں کسی بعد کے گروہ کا رکن کہنا تاریخ کے بھی خلاف ہے اور علم کے بھی۔

وَمَنْ اَظْلَمُ مِمَّنْ كَتَمَ شَهَادَةً عِنْدَهٗ مِنَ اللّٰهِ۔ “اور اس سے بڑھ کر ظالم کون ہے جو اس گواہی کو چھپائے جو اس کے پاس اللہ کی طرف سے ہے؟” ظاہری معنی میں یہ وہ علم ہے جو ان کی کتابوں میں حضرت ابراہیم علیہ السلام کی حنیفیت اور آنے والے نبی کے بارے میں موجود تھا۔ اہلِ معرفت کہتے ہیں کہ ہر جان کے پاس اللہ کی طرف سے ایک گواہی امانت رکھی گئی ہے، وہی جو عہدِ الست کے دن دی گئی تھی: اَلَسْتُ بِرَبِّكُمْ ۚ قَالُوْا بَلٰى۔ سب سے بڑا ظلم یہ ہے کہ آدمی اپنے اندر کی اس گواہی کو غفلت اور ضد کے پردوں میں چھپا دے۔ شیخِ اکبر کے نزدیک یہ گواہی بندے کی عینِ ثابتہ میں نقش ہے، اور ذکر اسی کو یاد دلانے کا نام ہے۔ پھر رکوع اسی جملے پر ختم ہوتا ہے جو پہلے آ چکا: تِلْكَ اُمَّةٌ قَدْ خَلَتْ ۚ لَهَا مَا كَسَبَتْ وَلَكُمْ مَّا كَسَبْتُمْ۔ “وہ ایک جماعت تھی جو گزر چکی؛ اس کے لیے اس کی کمائی ہے اور تمہارے لیے تمہاری کمائی۔” مفسرین کہتے ہیں کہ یہ تکرار تاکید کے لیے ہے؛ اہلِ ذوق کہتے ہیں کہ بزرگوں پر تکیہ کرنے کی عادت اتنی گہری ہے کہ اسے ایک بار کہنا کافی نہ تھا۔ نسبت عزت ہے، مگر نجات نہیں؛ نجات اپنی کمائی سے ہے، اور ہر جان اپنے رب کے سامنے اپنا حساب خود دے گی۔
