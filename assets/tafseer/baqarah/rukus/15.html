یہ رکوع بنی اسرائیل سے ہونے والے طویل خطاب کو اسی نقطے پر سمیٹتا ہے جہاں سے وہ شروع ہوا تھا، اور پھر سارے بیان کا رخ حضرت ابراہیم علیہ السلام کی طرف موڑ دیتا ہے، جن سے سب اہلِ کتاب اپنا رشتہ جوڑتے تھے: يٰبَنِيْۤ اِسْرَآءِيْلَ اذْكُرُوْا نِعْمَتِيَ الَّتِيْۤ اَنْعَمْتُ عَلَيْكُمْ وَاَنِّيْ فَضَّلْتُكُمْ عَلَى الْعٰلَمِيْنَ۔ “اے بنی اسرائیل! میری وہ نعمت یاد کرو جو میں نے تم پر کی، اور یہ کہ میں نے تمہیں جہان والوں پر فضیلت دی۔” یہی بات سینتالیسویں اور اڑتالیسویں آیت میں تقریباً انہی لفظوں میں آ چکی ہے؛ مفسرین کہتے ہیں کہ خطاب کے آغاز اور انجام پر ایک ہی یاددہانی رکھ کر قرآن نے ساری داستان کو ایک دائرے میں باندھ دیا۔ یہ فضیلت اس زمانے کے اہلِ عالم پر تھی، نسل کا کوئی دائمی امتیاز نہیں، اور اگلی آیت فوراً بتا دیتی ہے کہ نسبت آخرت میں کام نہ آئے گی: وَاتَّقُوْا يَوْمًا لَّا تَجْزِيْ نَفْسٌ عَنْ نَّفْسٍ شَيْئًا وَّلَا يُقْبَلُ مِنْهَا عَدْلٌ وَّلَا تَنْفَعُهَا شَفَاعَةٌ۔ “اور اس دن سے ڈرو جب کوئی جان کسی جان کے کچھ کام نہ آئے گی، نہ اس سے کوئی بدلہ قبول کیا جائے گا اور نہ کوئی سفارش اسے نفع دے گی۔” اہلِ معرفت کے نزدیک نعمت کی یاد اور حساب کا خوف یہاں ایک سانس میں اس لیے ہیں کہ نعمت کو نسبت سمجھ لینا ہی غفلت کی جڑ ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر جان اپنے ربِّ خاص کے سامنے اکیلی کھڑی ہے؛ اس کا رشتہ حق سے براہِ راست ہے، اور کوئی دوسرا اس کی جگہ نہیں لے سکتا۔ سفارش کی نفی یہاں ان کے حق میں ہے جو اسے موروثی حق سمجھ بیٹھے تھے؛ وہ شفاعت جو اذن سے ہو، اہلِ ایمان کے لیے برحق ہے۔ یوں رکوع کی دہلیز ہی پر بتا دیا گیا کہ جس امامت کا ذکر آگے آ رہا ہے وہ نسب سے نہیں، حال سے ملتی ہے۔

اب وہ آیت جس سے اس سورت کا ایک نیا باب کھلتا ہے: وَاِذِ ابْتَلٰۤى اِبْرٰهٖمَ رَبُّهٗ بِكَلِمٰتٍ فَاَتَمَّهُنَّ۔ “اور جب ابراہیم کو اس کے رب نے چند باتوں میں آزمایا تو اس نے انہیں پورا کر دکھایا۔” ان کلمات کی تعیین میں مفسرین کے کئی اقوال ہیں: بعض کے نزدیک فطرت کی سنتیں اور طہارت کے احکام، بعض کے نزدیک مناسکِ حج، اور بعض کے نزدیک وہ بڑی آزمائشیں جن سے حضرت ابراہیم علیہ السلام گزرے: آگ، ہجرت، بے آب وادی میں اہل و عیال کو بسانا اور بیٹے کی قربانی۔ لفظ پر غور کیجیے: آزمانے والا "رَبُّهٗ" ہے، یعنی پرورش کرنے والا؛ یہاں آزمائش سزا نہیں، تربیت ہے۔ اور "فَاَتَمَّهُنَّ" بتاتا ہے کہ ابتلا کا مقصد کمی ڈھونڈنا نہیں بلکہ تکمیل تک پہنچانا تھا۔ پھر انعام کا اعلان ہوا: قَالَ اِنِّيْ جَاعِلُكَ لِلنَّاسِ اِمَامًا۔ “فرمایا: میں تجھے سب انسانوں کا امام بنانے والا ہوں۔” یہاں "لِلنَّاسِ" عام ہے، اسی لیے بعد کی سب ملتیں انہیں اپنا پیشوا مانتی ہیں۔ شیخِ اکبر کے نزدیک امامت وہ مقام ہے جہاں بندہ اپنے رب کے اسماء کا ایسا صاف مظہر بن جائے کہ دوسرے اس میں راستہ دیکھ سکیں۔ فصوص الحکم میں حضرت ابراہیم علیہ السلام کا باب حکمتِ مہیمیہ ہے، اور وہاں شیخ خلیل کی وجہِ تسمیہ تخلل سے بیان کرتے ہیں، جس کا مفہوم یہ ہے کہ انہوں نے ان صفات کو اپنے اندر سمو لیا جن سے ذاتِ الٰہیہ موصوف ہے، جیسے رنگ کپڑے کے تار تار میں اتر جاتا ہے۔ مگر یہ سمونا مظہر ہونے کے معنی میں ہے؛ عبد عبد رہتا ہے اور رب رب، اور تنزیہ اپنی جگہ قائم ہے۔ اہلِ ذوق کہتے ہیں: جو کلمات کو پورا کرتا ہے اس میں اسماء پورے ظاہر ہوتے ہیں، اور جس میں اسماء ظاہر ہوں وہی لوگوں کا امام ہے۔

پھر باپ کے دل کی فطری تمنا زبان پر آئی: قَالَ وَمِنْ ذُرِّيَّتِيْ۔ “عرض کیا: اور میری اولاد میں سے بھی؟” اور جواب ایسا آیا جو ہر زمانے کا قاعدہ بن گیا: قَالَ لَا يَنَالُ عَهْدِي الظّٰلِمِيْنَ۔ “فرمایا: میرا عہد ظالموں تک نہیں پہنچتا۔” مفسرین کہتے ہیں کہ یہ دعا رد نہیں ہوئی بلکہ تقسیم ہوئی: اولاد میں نیک بھی ہوں گے اور ظالم بھی، اور امامت نیکوں کو ملے گی؛ چنانچہ نبوت ان کی نسل میں حضرت اسحاق اور حضرت اسماعیل علیہما السلام کی دونوں شاخوں میں جاری رہی۔ قابلِ غور یہ ہے کہ فعل عہد کی طرف منسوب ہے: عہد ان تک نہیں پہنچتا۔ دروازہ حق کی طرف سے کسی پر زبردستی بند نہیں کیا گیا؛ ظالم نے خود اپنے اور عہد کے درمیان دیوار اٹھائی۔ ظلم کا لغوی معنی ہے کسی چیز کو اس کی جگہ سے ہٹا کر رکھنا، اور سب سے بڑا ظلم یہ ہے کہ بندہ اپنے دل کو، جو معرفتِ حق کے لیے بنا ہے، غیر کے سپرد کر دے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے؛ جو بندہ خود اپنا رخ ظلم کی طرف موڑ لے، عطا اس پر اسی رخ کے مطابق اترتی ہے۔ نہ کوئی کسی کو جبراً محروم کرتا ہے، نہ خون کا رشتہ کسی کو استحقاق دلاتا ہے۔ اور یہی اس پورے خطاب کا نچوڑ ہے جو بنی اسرائیل سے ہوا: ابراہیمی نسبت کا دعویٰ کافی نہیں، ابراہیمی حال درکار ہے۔ اہلِ تاویل کہتے ہیں کہ ہر سالک کے اندر بھی ایک اولاد ہے، یعنی وہ اعمال اور خیالات جو اس سے جنم لیتے ہیں؛ ان میں سے جو ظلم سے آلودہ ہوں وہ امامت کے نور کے وارث نہیں بنتے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

وَاِذْ جَعَلْنَا الْبَيْتَ مَثَابَةً لِّلنَّاسِ وَاَمْنًا۔ “اور جب ہم نے اس گھر کو لوگوں کے لیے بار بار لوٹ کر آنے کی جگہ اور امن کا مقام بنایا۔” مثابہ کی اصل لوٹنا ہے؛ جو ایک بار اس گھر کو دیکھ لیتا ہے، اس کا دل بار بار اسی کی طرف پلٹتا ہے۔ پھر حکم: وَاتَّخِذُوْا مِنْ مَّقَامِ اِبْرٰهٖمَ مُصَلًّى۔ “اور ابراہیم کے کھڑے ہونے کی جگہ کو نماز کی جگہ بنا لو۔” اور تاکید: وَعَهِدْنَاۤ اِلٰۤى اِبْرٰهٖمَ وَاِسْمٰعِيْلَ اَنْ طَهِّرَا بَيْتِيَ لِلطَّآئِفِيْنَ وَالْعٰكِفِيْنَ وَالرُّكَّعِ السُّجُوْدِ۔ “اور ہم نے ابراہیم اور اسماعیل کو تاکید کی کہ میرے گھر کو طواف کرنے والوں، اعتکاف کرنے والوں اور رکوع و سجدہ کرنے والوں کے لیے پاک رکھو۔” ظاہری معنی میں یہ کعبہ ہے: اس کی حرمت، اس کا امن اور اس کی پاکیزگی؛ اور یہی اصل ہے۔ اہلِ تاویل کہتے ہیں کہ اس ظاہر کے ساتھ ایک باطن بھی ہے: مومن کا دل بھی حق کا گھر ہے۔ اہلِ تصوف میں مشہور اثر ہے کہ زمین و آسمان مجھے نہ سما سکے اور مومن بندے کا دل مجھے سما گیا؛ اور شیخِ اکبر اس وسعتِ قلب کو تجلیات کی قبولیت کے معنی میں لیتے ہیں، کسی طرح کے حلول کے معنی میں ہرگز نہیں۔ "بَيْتِيَ" میں اضافت شرف کی ہے، جیسے ناقۃ اللہ؛ گھر حق کا ٹھکانا نہیں، اس کی یاد کا مرکز ہے۔ اس رمز میں طہارت یہ ہے کہ دل کو اغیار کے نقوش سے خالی کیا جائے، اور طواف کرنے والے، معتکف اور رکوع و سجود والے گویا ذکر کے مراتب ہیں: کبھی دل گرد گھومتا ہے، کبھی ٹھہر جاتا ہے، کبھی جھکتا ہے، اور کبھی اپنی ہستی بھلا کر سجدے میں گر جاتا ہے۔ اور مقامِ ابراہیم کو مصلیٰ بنانے میں یہ اشارہ بھی ہے کہ سالک وہاں کھڑا ہو جہاں خلیل کھڑے تھے: تسلیم کے قدم پر۔

وَاِذْ قَالَ اِبْرٰهٖمُ رَبِّ اجْعَلْ هٰذَا بَلَدًا اٰمِنًا وَّارْزُقْ اَهْلَهٗ مِنَ الثَّمَرٰتِ مَنْ اٰمَنَ مِنْهُمْ بِاللّٰهِ وَالْيَوْمِ الْاٰخِرِ۔ “اور جب ابراہیم نے دعا کی: اے میرے رب! اس کو امن والا شہر بنا دے، اور اس کے باشندوں میں سے جو اللہ اور آخرت کے دن پر ایمان لائیں انہیں پھلوں کی روزی دے۔” قَالَ وَمَنْ كَفَرَ فَاُمَتِّعُهٗ قَلِيْلًا ثُمَّ اَضْطَرُّهٗۤ اِلٰى عَذَابِ النَّارِ۔ “فرمایا: اور جو کفر کرے گا اسے بھی میں تھوڑا سا فائدہ اٹھانے دوں گا، پھر اسے آگ کے عذاب کی طرف کھینچ لاؤں گا۔” مفسرین نے یہاں ایک لطیف ادب دیکھا ہے: ابھی امامت کے بارے میں جواب ملا تھا کہ عہد ظالموں تک نہیں پہنچتا، تو حضرت ابراہیم علیہ السلام نے روزی کی دعا میں بھی خود ہی ایمان کی قید لگا دی۔ مگر جواب آیا کہ روزی کا معاملہ امامت جیسا نہیں؛ دنیا کا رزق منکر کو بھی ملے گا۔ شیخِ اکبر کے مکتب میں یہ رحمانیت اور رحیمیت کا فرق ہے: اسمِ رحمٰن کی رحمت عام ہے جو ہر موجود کو وجود اور روزی دیتی ہے، اور اسمِ رحیم کی رحمت خاص ہے جو ایمان اور قرب کی صورت میں اترتی ہے۔ امامت رحمتِ خاصہ ہے اس لیے اس میں شرط ہے، اور روزی رحمتِ عامہ ہے اس لیے اس میں شرط نہیں۔ یہی سبقتِ رحمت ہے کہ دنیا کا دسترخوان انکار کرنے والے کے لیے بھی بچھا رہتا ہے۔ اور انجام کی خبر بھی غور طلب ہے: تھوڑا سا فائدہ دراصل مہلت ہے، اور مہلت پلٹ آنے کا دروازہ ہے۔ جو اس دروازے سے نہ پلٹا، اس کا انجام کسی جبر کا نتیجہ نہیں، اس کے اپنے اختیار کیے ہوئے رخ کا تسلسل ہے۔ اہلِ معرفت کہتے ہیں کہ امن والا شہر دل کی وہ حالت بھی ہے جہاں خوف کے بجائے سکینت بسے، اور اس کے پھل وہ معارف ہیں جو ایمان والے دل کو نصیب ہوتے ہیں۔

اب وہ منظر جسے قرآن فعلِ مضارع میں بیان کرتا ہے، گویا ہم اسے آج بھی دیکھ رہے ہوں: وَاِذْ يَرْفَعُ اِبْرٰهٖمُ الْقَوَاعِدَ مِنَ الْبَيْتِ وَاِسْمٰعِيْلُ ۚ رَبَّنَا تَقَبَّلْ مِنَّا ۚ اِنَّكَ اَنْتَ السَّمِيْعُ الْعَلِيْمُ۔ “اور جب ابراہیم اور اسماعیل اس گھر کی بنیادیں اٹھا رہے تھے (اور دعا کر رہے تھے): اے ہمارے رب! ہم سے قبول فرما، بے شک تو ہی سب کچھ سننے والا، سب کچھ جاننے والا ہے۔” باپ پتھر رکھتے ہیں، بیٹا پتھر لا کر دیتا ہے، اور دونوں کی زبان پر ایک ہی بات ہے۔ حیرت یہ ہے کہ زمین کا سب سے مقدس گھر بنانے والے قبولیت کی بھیک مانگ رہے ہیں؛ عمل کتنا ہی بڑا ہو، اس کا قبول ہونا حق کے ہاتھ میں ہے۔ اہلِ معرفت کے نزدیک یہی عمل کو اپنا نہ سمجھنے کا ادب ہے۔ شیخِ اکبر کے مکتب میں فاعلِ حقیقی حق ہے اور بندہ اس کے فعل کا مظہر؛ اسی لیے عارف اپنے عمل پر ناز نہیں کرتا، اسے پیش کر کے قبولیت مانگتا ہے۔ اور دعا "السَّمِيْعُ الْعَلِيْمُ" پر ختم ہوئی: سننے والا دعا کو سنتا ہے، اور جاننے والا اس نیت کو جانتا ہے جو پتھروں کے بیچ چھپی ہے۔ پھر دعا آگے بڑھی: رَبَّنَا وَاجْعَلْنَا مُسْلِمَيْنِ لَكَ وَمِنْ ذُرِّيَّتِنَاۤ اُمَّةً مُّسْلِمَةً لَّكَ۔ “اے ہمارے رب! ہم دونوں کو اپنا فرماں بردار بنا، اور ہماری اولاد میں سے بھی ایک امت اپنی فرماں بردار بنا۔” وہ پہلے ہی فرماں بردار تھے، پھر بھی اسلام مانگا؛ مفسرین کہتے ہیں کہ یہ ثبات اور زیادتی کی دعا ہے۔ اہلِ ذوق کہتے ہیں کہ اسلام کوئی منزل نہیں جس پر پہنچ کر ٹھہر جائیں: تجددِ امثال کی رو سے بندے کو ہر آن نیا وجود ملتا ہے، تو ہر آن نئی تسلیم بھی درکار ہے۔

وَاَرِنَا مَنَاسِكَنَا وَتُبْ عَلَيْنَا ۚ اِنَّكَ اَنْتَ التَّوَّابُ الرَّحِيْمُ۔ “اور ہمیں ہماری عبادت کے طریقے دکھا، اور ہم پر رجوع فرما؛ بے شک تو ہی بہت رجوع فرمانے والا، نہایت رحم کرنے والا ہے۔” یہاں "اَرِنَا" کا لفظ قابلِ غور ہے: سکھا نہیں فرمایا، دکھا فرمایا۔ مفسرین کہتے ہیں کہ حج کے مناسک حضرت جبریل علیہ السلام نے مقام بہ مقام دکھا کر بتائے۔ اہلِ معرفت کہتے ہیں کہ عبادت کی حقیقت سننے سے نہیں، دکھائے جانے سے کھلتی ہے؛ یہ کشف کی طلب ہے، یعنی وہ آنکھ جو عمل کے ظاہر میں اس کا باطن دیکھ لے۔ رہی توبہ کی دعا، تو انبیاء علیہم السلام معصوم ہیں، ان کی توبہ گناہ سے نہیں۔ اہلِ معرفت کے نزدیک ان کی توبہ ایک مقام سے بلند تر مقام کی طرف رجوع ہے، اور یہ امت کے لیے تعلیم بھی ہے کہ بڑے سے بڑا عمل کر کے بھی بندہ رجوع ہی کی زبان میں بات کرے۔ توبہ کا اصل معنی لوٹنا ہے: بندے کی توبہ حق کی طرف لوٹنا ہے، اور حق کی توبہ بندے کی طرف رحمت کے ساتھ متوجہ ہونا۔ اسی لیے دعا "وَتُبْ عَلَيْنَا" ہے، یعنی تو ہماری طرف رحمت سے متوجہ ہو؛ کیونکہ بندے کا لوٹنا بھی اسی وقت ممکن ہے جب پہلے وہ اس کی طرف متوجہ ہو۔ شیخِ اکبر کے نزدیک ہر حرکت جو بندے کی طرف سے ظاہر ہوتی ہے، اس سے پہلے حق کی طرف سے ایک توجہ ہوتی ہے؛ یہ جبر نہیں، دعوت ہے، اور بندہ جب اسے قبول کر لیتا ہے تو التواب الرحیم کا مظہر بنتا ہے۔

اور اب وہ دعا جس کی قبولیت نے تاریخ کا رخ بدل دیا: رَبَّنَا وَابْعَثْ فِيْهِمْ رَسُوْلًا مِّنْهُمْ يَتْلُوْا عَلَيْهِمْ اٰيٰتِكَ وَيُعَلِّمُهُمُ الْكِتٰبَ وَالْحِكْمَةَ وَيُزَكِّيْهِمْ ۚ اِنَّكَ اَنْتَ الْعَزِيْزُ الْحَكِيْمُ۔ “اے ہمارے رب! اور ان میں انہی میں سے ایک رسول بھیج جو انہیں تیری آیات پڑھ کر سنائے، انہیں کتاب اور حکمت سکھائے اور انہیں پاک کرے؛ بے شک تو ہی غالب، حکمت والا ہے۔” "فِيْهِمْ" کا اشارہ اسی امتِ مسلمہ کی طرف ہے جو حضرت اسماعیل علیہ السلام کی نسل سے مانگی گئی تھی، اور صدیوں بعد یہ دعا حضورِ اکرم صلی اللہ علیہ وسلم کی صورت میں قبول ہوئی۔ مسند احمد کی روایت کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم نے فرمایا: میں اپنے باپ ابراہیم کی دعا ہوں، عیسیٰ کی بشارت ہوں، اور اپنی والدہ کا وہ خواب ہوں جو انہوں نے دیکھا۔ اسی وادی میں، انہی بنیادوں کے سائے میں، اسی شہر میں جس کے لیے امن مانگا گیا تھا، وہ رسول مبعوث ہوئے۔ دعا میں رسول کے چار کام گنوائے گئے: آیات کی تلاوت، کتاب کی تعلیم، حکمت کی تعلیم اور تزکیہ۔ اہلِ معرفت کہتے ہیں کہ آیات وہ نشانیاں ہیں جو کان تک پہنچتی ہیں، کتاب وہ احکام ہیں جو عقل میں اترتے ہیں، حکمت وہ فہم ہے جو دل میں جڑ پکڑتا ہے کہ ہر چیز کو اس کی جگہ پر رکھا جائے، اور تزکیہ وہ ہے جو نفس کو اندر سے بدل دیتا ہے۔ اور دعا "الْعَزِيْزُ الْحَكِيْمُ" پر ختم ہوئی: عزیز کہ اس دعا کی قبولیت میں کوئی رکاوٹ نہیں بن سکتا، اور حکیم کہ وہ قبولیت کا ٹھیک وقت جانتا ہے، خواہ اس میں صدیاں لگ جائیں۔

مفسرین نے ایک لطیف نکتہ دیکھا ہے۔ حضرت ابراہیم علیہ السلام کی دعا میں تزکیہ سب سے آخر میں ہے، مگر جب قبولیت کا ذکر آیا تو اسی سورت کی ایک سو اکیاونویں آیت میں، سورۂ آلِ عمران میں اور سورۂ الجمعہ میں تزکیہ کو تعلیم سے پہلے رکھا گیا: يَتْلُوْا عَلَيْهِمْ اٰيٰتِهٖ وَيُزَكِّيْهِمْ وَيُعَلِّمُهُمُ الْكِتٰبَ وَالْحِكْمَةَ۔ اہلِ معرفت کہتے ہیں کہ خلیل نے علم کی ترتیب سے مانگا اور حق نے تربیت کی ترتیب سے عطا کیا: آئینہ صاف ہو تو صورت اترتی ہے؛ دل پاک ہو تو کتاب اور حکمت اس میں ٹھہرتی ہیں۔ شیخِ اکبر کے مکتب میں اس دعا کا ایک اور پہلو بھی ہے۔ ترمذی کی روایت کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم سے پوچھا گیا کہ آپ کے لیے نبوت کب ثابت ہوئی، تو فرمایا: جب آدم ابھی روح اور جسم کے درمیان تھے۔ اس مکتب میں حقیقتِ محمدیہ تمام مظاہر سے پہلے ہے، اور حضرت ابراہیم علیہ السلام کی دعا ایک ازلی حقیقت کے زمانی ظہور کی طلب تھی۔ فصوص الحکم میں حضورِ اکرم صلی اللہ علیہ وسلم کا باب حکمتِ فردیہ ہے، جس کا مفہوم یہ ہے کہ آپ نوعِ انسانی میں سب سے کامل وجود ہیں، اسی لیے معاملہ آپ سے شروع ہوا اور آپ ہی پر ختم ہوا۔ یوں خلیل نے بنیاد اٹھاتے ہوئے جس رسول کو مانگا، وہی اس گھر کی معنوی بنیاد تھے؛ اور جس امت کو مانگا، وہ اسی رسول کی امت بنی۔
