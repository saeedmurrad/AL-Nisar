كَذَّبَتْ قَوْمُ نُوْحٍ الْمُرْسَلِيْنَ۔ نوح کی قوم نے رسولوں کو جھٹلایا۔ اِذْ قَالَ لَهُمْ اَخُوْهُمْ نُوْحٌ اَلَا تَتَّقُوْنَ۔ جب ان کے بھائی نوح نے ان سے کہا: کیا تم ڈرتے نہیں؟ حضرت نوح علیہ السلام ایک رسول تھے، پھر بھی قرآن نے "الْمُرْسَلِيْنَ" جمع فرمایا۔ ظاہری تفسیر یہ ہے کہ سب رسولوں کا پیغام ایک ہے، اس لیے ایک کا انکار سب کا انکار ہے۔ اہلِ معرفت اسی میں ایک گہرائی دیکھتے ہیں: رسالت ایک ہی حقیقت ہے جو زمانوں میں الگ الگ مظاہر میں جلوہ گر ہوئی، اور جو ایک مظہر کو رد کرتا ہے وہ اس نور کو رد کرتا ہے جو سب میں ایک ہے۔ پھر لفظ "اَخُوْهُمْ" دیکھیے: نہ حاکم، نہ اجنبی، بلکہ بھائی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں نبی ایک برزخ ہے؛ ایک رخ حق کی طرف جس سے وہ لیتا ہے، اور ایک رخ مخلوق کی طرف جس سے وہ انہی کی زبان اور انہی کی بشریت میں دیتا ہے۔ اور دعوت کا آغاز حکم سے نہیں، سوال سے ہوا: "اَلَا تَتَّقُوْنَ"؛ کیونکہ تقویٰ زبردستی نہیں اگایا جاتا، اسے دل میں جگایا جاتا ہے۔ حکمتِ سبوحیہ میں شیخِ اکبر نے جو بات کھولی ہے اس کا مفہوم یہ ہے کہ حضرت نوح علیہ السلام کی دعوت میں تنزیہ غالب تھی، اس لیے کہ ان کی قوم مورتیوں کی تشبیہ میں ڈوبی ہوئی تھی؛ اور کامل معرفت تنزیہ و تشبیہ کو جمع کرتی ہے۔ یہ حضرت نوح علیہ السلام کے علم کی کمی نہیں، قوم کے حال کا تقاضا تھا: دوا مرض کے مطابق دی جاتی ہے۔

اِنِّيْ لَكُمْ رَسُوْلٌ اَمِيْنٌ۔ میں تمہارے لیے ایک امانت دار رسول ہوں۔ فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ۔ تو اللہ سے ڈرو اور میری اطاعت کرو۔ یہاں سے وہ دعوت شروع ہوتی ہے جو اس سورت میں ہر نبی کی زبان پر انہی الفاظ میں لوٹ کر آئے گی۔ "اَمِيْنٌ" کی دلیل قوم خود جانتی تھی: جس نے برسوں ان کے لین دین میں خیانت نہ کی، وہ رب کے پیغام میں کیا خیانت کرے گا۔ اہلِ معرفت کے نزدیک رسول کی امانت یہ ہے کہ جو اترا ویسا ہی پہنچا دے؛ نہ اپنی طرف سے کچھ بڑھائے نہ گھٹائے، گویا اس کا دل ایسا صاف آئینہ ہو جس پر اپنا کوئی رنگ نہ ہو۔ یہی ادبِ اسرار کی جڑ ہے۔ اس مکتب میں تقویٰ کی ایک لطیف تعبیر یوں بیان ہوتی ہے کہ تقویٰ "وقایہ" یعنی ڈھال بنانا ہے: خوبیوں میں حق کو اپنی ڈھال بنا لو کہ ہر خوبی اسی کی طرف منسوب ہو، اور کوتاہیوں میں اپنے آپ کو ڈھال بنا لو کہ ہر کمی اپنی طرف منسوب ہو۔ یہی ادب اسی سورت میں حضرت ابراہیم علیہ السلام نے بیماری کو اپنی طرف اور شفا کو رب کی طرف منسوب کر کے دکھایا تھا۔ اور "وَاَطِيْعُوْنِ" اپنی ذات کی طرف بلانا نہیں؛ قرآن نے خود کھول دیا: مَنْ يُّطِعِ الرَّسُوْلَ فَقَدْ اَطَاعَ اللّٰهَ۔ رسول وہ مقام ہے جہاں امرِ الٰہی سنائی دینے لگتا ہے۔

وَمَآ اَسْـَٔلُكُمْ عَلَيْهِ مِنْ اَجْرٍ ۚ اِنْ اَجْرِيَ اِلَّا عَلٰى رَبِّ الْعٰلَمِيْنَ۔ اور میں اس پر تم سے کوئی اجر نہیں مانگتا؛ میرا اجر تو بس رب العالمین کے ذمے ہے۔ فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ۔ تو اللہ سے ڈرو اور میری اطاعت کرو۔ داعی کی آزادی اس کی بے نیازی میں ہے۔ جو بلائے جانے والوں سے کچھ چاہتا ہے وہ ان کا اسیر ہو جاتا ہے، اور اسیر کی بات میں وہ قوت نہیں رہتی جو آزاد کی بات میں ہوتی ہے۔ یہ انبیاء کا فقر ہے: ہاتھ خالی، دل بھرا ہوا۔ یہ بھی دیکھیے کہ یہاں "رَبِّيْ" نہیں فرمایا بلکہ "رَبِّ الْعٰلَمِيْنَ"؛ یعنی وہ جو دینے والوں کا بھی رب ہے اور سننے والوں کا بھی۔ جس کا اجر سب جہانوں کے رب پر ہو، اسے کسی جہان کی حاجت نہیں رہتی۔ پھر "فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ" دوبارہ آیا۔ پہلی بار اس کی بنیاد امانت تھی، دوسری بار بے غرضی؛ دو دلیلیں اور ایک مطالبہ۔ اس مکتب کا مشہور اصول ہے کہ تجلی میں تکرار نہیں، اور اسی کو تجددِ امثال کہتے ہیں: ایک ہی لفظ دوسری بار ایک نئے دل پر اترتا ہے اور نیا معنی لاتا ہے۔ اسی لیے اس سورت میں ایک ہی دعوت ہر نبی کے ساتھ لوٹتی ہے، اور ہر بار تازہ ہوتی ہے۔

قَالُوْۤا اَنُؤْمِنُ لَكَ وَاتَّبَعَكَ الْاَرْذَلُوْنَ۔ وہ بولے: کیا ہم تجھ پر ایمان لے آئیں حالانکہ تیری پیروی تو کم تر لوگوں نے کی ہے؟ غور کیجیے کہ سرداروں نے پیغام کا رد نہیں کیا، پیروی کرنے والوں کو دیکھا۔ انہوں نے حق کو سماجی مرتبے کے ترازو پر تولا، اور یہ ترازو ہی غلط تھا۔ اہلِ معرفت کہتے ہیں کہ یہ صورت پر رک جانا ہے، وہی بیماری جو سورۂ الحجر میں ابلیس کے ہاں دیکھی گئی کہ اس نے مٹی دیکھی اور اس میں پھونکی گئی روح نہ دیکھی۔ جنہیں انہوں نے "الْاَرْذَلُوْنَ" کہا، وہ وہ لوگ تھے جن کے دلوں پر مال اور منصب کے پردے کم تھے؛ ظاہر کی تنگی ان کے باطن کا ہلکا پن بن گئی، اور ہلکا برتن جلد بھرتا ہے۔ یاد رہے کہ یہ لفظ قرآن نے سرداروں کی زبان سے نقل کیا ہے، یہ حق تعالیٰ کا فیصلہ نہیں؛ قرآن کے ترازو میں یہی لوگ سب سے بھاری نکلے۔ اور سرداروں کا غرور ہی ان کا پردہ تھا؛ یہ پردہ ان پر ڈالا نہیں گیا، انہوں نے خود بُنا، اور حق نے اسی سمت کو وجود دیا جو انہوں نے خود چنی۔

قَالَ وَمَا عِلْمِيْ بِمَا كَانُوْا يَعْمَلُوْنَ۔ نوح نے کہا: مجھے کیا خبر کہ وہ کیا کرتے رہے ہیں۔ اِنْ حِسَابُهُمْ اِلَّا عَلٰى رَبِّيْ لَوْ تَشْعُرُوْنَ۔ ان کا حساب تو بس میرے رب کے ذمے ہے، کاش تم سمجھ رکھتے۔ سرداروں کا اشارہ یہ تھا کہ یہ لوگ کسی دنیاوی غرض سے آئے ہیں یا ان کا ماضی پست ہے۔ نبی کا جواب ادب کا پورا باب ہے: میں ظاہر کو قبول کرتا ہوں، باطن میرے رب کے سپرد ہے۔ مرشد کا کام طالب کے ماضی کی تلاشی لینا نہیں؛ جو دروازے پر آ گیا، اس کا آنا ہی اس کی سند ہے۔ یہاں "رَبِّيْ" فرمایا، اور شیخِ اکبر کے نزدیک ہر عین کا ایک ربِّ خاص ہے، یعنی وہ اسم جو اس کی پرورش کرتا ہے؛ ہر ایک کا حساب اسی رب کے پاس ہے جو اس کی حقیقت کو جانتا ہے، کسی دوسرے کی نظر کو اس میں دخل نہیں۔ اور "لَوْ تَشْعُرُوْنَ" میں کمی عقل کی نہیں، شعور کی بتائی گئی؛ وہ باریک باطنی احساس جو چیزوں کی قیمت ان کی صورت سے آگے دیکھتا ہے۔ عجیب لطف یہ ہے کہ اس سورت کا نام "الشعراء" بھی اسی مادے ش ع ر سے ہے؛ گویا پوری سورت پوچھتی ہے کہ تمہارا شعور کس چیز کو محسوس کرتا ہے۔

وَمَآ اَنَا بِطَارِدِ الْمُؤْمِنِيْنَ۔ اور میں ایمان والوں کو دھتکارنے والا نہیں۔ اِنْ اَنَا اِلَّا نَذِيْرٌ مُّبِيْنٌ۔ میں تو بس کھول کر ڈرانے والا ہوں۔ یہ اس رکوع کا دل ہے اور فقر کی عزت کا اعلان۔ سرداروں کی شرط یہ تھی کہ غریبوں کو ہٹاؤ تو ہم آئیں؛ نبی نے سرداروں کا آنا کھو دینا گوارا کیا مگر ایک فقیر کو در سے ہٹانا گوارا نہ کیا۔ یہی سبق سورۂ الکہف میں حضورِ اکرم صلی اللہ علیہ وسلم کو دیا گیا: وَاصْبِرْ نَفْسَكَ مَعَ الَّذِيْنَ يَدْعُوْنَ رَبَّهُمْ بِالْغَدٰوةِ وَالْعَشِيِّ يُرِيْدُوْنَ وَجْهَهٗ؛ اور مشہور حدیث کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم نے مسکینوں کے ساتھ جینے اور انہی کے ساتھ اٹھائے جانے کی دعا کی۔ شیخِ اکبر کے مکتب میں فقر بندے کی اصل حقیقت ہے: ہر موجود ہر لمحہ اپنے وجود میں حق کا محتاج ہے۔ ظاہر کا فقیر اس باطنی فقر کا آئینہ ہے جو ہر شے کی حقیقت ہے؛ جو فقیر کو دھتکارتا ہے وہ گویا اپنی ہی حقیقت کا آئینہ توڑتا ہے۔ اور "نَذِيْرٌ مُّبِيْنٌ" بتاتا ہے کہ نبی کا کام چھانٹنا نہیں، خبردار کرنا ہے؛ دروازہ سب کے لیے کھلا ہے، جو آ جائے وہ قبول ہے۔

قَالُوْا لَئِنْ لَّمْ تَنْتَهِ يٰنُوْحُ لَتَكُوْنَنَّ مِنَ الْمَرْجُوْمِيْنَ۔ وہ بولے: اے نوح! اگر تو باز نہ آیا تو ضرور سنگسار کیے جانے والوں میں سے ہو گا۔ قَالَ رَبِّ اِنَّ قَوْمِيْ كَذَّبُوْنِ۔ نوح نے عرض کیا: اے میرے رب! میری قوم نے مجھے جھٹلا دیا۔ جب دلیل ختم ہوتی ہے تو دھمکی شروع ہوتی ہے؛ پتھر اسی کے ہاتھ میں آتا ہے جس کی حجت ٹوٹ چکی ہو۔ اور یہ دعوت ایک دو دن کی نہ تھی: فَلَبِثَ فِيْهِمْ اَلْفَ سَنَةٍ اِلَّا خَمْسِيْنَ عَامًا۔ اتنی طویل دعوت کے جواب میں سنگساری کی دھمکی ملی، مگر حضرت نوح علیہ السلام نے پلٹ کر دھمکی نہ دی؛ رخ رب کی طرف کر لیا۔ اور دیکھیے کہ اب بھی "قَوْمِيْ" فرمایا، "میری قوم"؛ جھٹلائے جانے کے بعد بھی رشتہ نہیں کاٹا۔ ان کی عرض میں نہ شکوہ ہے نہ مطالبۂ انتقام، صرف حال کی خبر ہے۔ اہلِ معرفت کہتے ہیں کہ اللہ سے شکایت، اللہ کی شکایت نہیں؛ یہ عین عبدیت ہے، جیسے سورۂ یوسف میں حضرت یعقوب علیہ السلام نے فرمایا: اِنَّمَآ اَشْكُوْا بَثِّيْ وَحُزْنِيْۤ اِلَى اللّٰهِ۔ دل کا بوجھ مخلوق کے سامنے رکھنا ذلت ہے، اور حق کے سامنے رکھنا قرب۔

فَافْتَحْ بَيْنِيْ وَبَيْنَهُمْ فَتْحًا وَّنَجِّنِيْ وَمَنْ مَّعِيَ مِنَ الْمُؤْمِنِيْنَ۔ تو میرے اور ان کے درمیان دو ٹوک فیصلہ فرما دے، اور مجھے اور میرے ساتھ کے ایمان والوں کو نجات دے۔ فَاَنْجَيْنٰهُ وَمَنْ مَّعَهٗ فِي الْفُلْكِ الْمَشْحُوْنِ۔ تو ہم نے اسے اور اس کے ساتھ والوں کو بھری ہوئی کشتی میں بچا لیا۔ ثُمَّ اَغْرَقْنَا بَعْدُ الْبٰقِيْنَ۔ پھر اس کے بعد باقی رہ جانے والوں کو ہم نے غرق کر دیا۔ ظاہر میں "فَتْحًا" فیصلہ ہے، اور اہلِ معرفت کی زبان میں اسی مادے سے "فتح" دل کے کھلنے کو کہتے ہیں؛ باہر کا فیصلہ حق و باطل کو جدا کرتا ہے، اور اندر کی فتح سالک میں نفس اور رب کی پکار کو جدا کر دیتی ہے۔ نبی نے نجات اکیلے اپنے لیے نہ مانگی، "وَمَنْ مَّعِيَ" ساتھ رکھا۔ اہلِ تاویل کشتی کو امرِ الٰہی اور شریعت و طریقت کی تمثیل کہتے ہیں اور طوفان کو خواہشوں اور طبیعت کے سیلاب کی؛ جو امر کی کشتی میں سوار ہو گیا، وہی پانی اسے اٹھاتا ہے جو دوسروں کو ڈبوتا ہے۔ "الْمَشْحُوْنِ" یعنی بھری ہوئی؛ جو دل ذکر سے بھر جائے اس میں طوفان کے لیے جگہ نہیں رہتی۔ اور "الْبٰقِيْنَ" وہ تھے جو اپنی ہی جگہ رک گئے؛ انہیں غرق کی طرف گھسیٹا نہیں گیا، وہ وہیں ٹھہرے جہاں ٹھہرنا انہوں نے خود چنا۔ کشتی سب کے سامنے بنی تھی، اور حضرت نوح علیہ السلام کے بیٹے سے کہا گیا: لَا عَاصِمَ الْيَوْمَ مِنْ اَمْرِ اللّٰهِ اِلَّا مَنْ رَّحِمَ؛ اور وہ رحمت کشتی کی صورت میں سب کو پیش کی گئی تھی۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ طوفان ایک حقیقی واقعہ اور حقیقی عذاب تھا۔

اِنَّ فِيْ ذٰلِكَ لَاٰيَةً ۭ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِيْنَ۔ بے شک اس میں ایک نشانی ہے، اور ان میں سے اکثر ایمان لانے والے نہ تھے۔ وَاِنَّ رَبَّكَ لَهُوَ الْعَزِيْزُ الرَّحِيْمُ۔ اور یقیناً تیرا رب ہی غالب، نہایت رحم والا ہے۔ چوتھی بار یہ ٹیپ کا بند لوٹا، اور حضرت نوح علیہ السلام کے قصے کے بعد اس کا رنگ پانی کا رنگ ہے۔ ایک ہی پانی تھا: ڈبونے میں وہ العزیز کا جلال تھا، اور کشتی کو اٹھانے میں الرحیم کا جمال۔ تجلی ایک، قبول کرنے والے دو؛ شیخِ اکبر کے نزدیک تجلی اپنا رنگ محل کی استعداد سے لیتی ہے، اور اہلِ تصوف میں مشہور قول ہے کہ پانی کا رنگ برتن کا رنگ ہوتا ہے۔ اور "اَكْثَرُهُمْ" کی گواہی دوسری جگہ یوں ہے: وَمَآ اٰمَنَ مَعَهٗۤ اِلَّا قَلِيْلٌ۔ ساڑھے نو سو برس پیغام پہنچتا رہا، تو ایمان نہ لانا ان کا اپنا رخ تھا، ایمان روکا نہیں گیا تھا۔ یہاں سرِّ قدر سامنے ہے: حکم محکوم کے مطابق ہوا؛ جو کشتی کی طرف بڑھا اسے رحیمیت ملی، جو پہاڑ کی طرف بھاگا اسے عزت کا جلال ملا۔ اور "رَبَّكَ" میں خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے: یہ قصہ آپ صلی اللہ علیہ وسلم کی تسلی ہے کہ اکثریت کا انکار حق کی کمی نہیں، اور آپ کا رب وہی ہے جو غالب بھی ہے اور رحیم بھی۔
