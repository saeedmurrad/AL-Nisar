كَذَّبَتْ عَادٌ الْمُرْسَلِيْنَ۔ عاد نے رسولوں کو جھٹلایا۔ اِذْ قَالَ لَهُمْ اَخُوْهُمْ هُوْدٌ اَلَا تَتَّقُوْنَ۔ جب ان کے بھائی ہود نے ان سے کہا: کیا تم ڈرتے نہیں؟ قصہ انہی الفاظ سے کھلا جن سے حضرت نوح علیہ السلام کا قصہ کھلا تھا، مگر سننے والے بدل گئے۔ عاد وہ قوم تھی جسے طوفان کے بعد زمین کی وراثت ملی: جَعَلَكُمْ خُلَفَآءَ مِنْ بَعْدِ قَوْمِ نُوْحٍ۔ انہیں زمین ملی، سبق نہ ملا؛ ڈوبنے والوں کی جگہ تو لی، ان کے انجام سے عبرت نہ لی۔ ان کی پہچان طاقت تھی، اور طاقتور کو تقویٰ کی سب سے زیادہ حاجت ہوتی ہے، کیونکہ کمزور کو اس کی کمزوری روک لیتی ہے اور طاقتور کو صرف تقویٰ۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ نے حکمتِ احدیہ میں حضرت ہود علیہ السلام ہی کے اس قول کو بنیاد بنایا ہے: مَا مِنْ دَآبَّةٍ اِلَّا هُوَ اٰخِذٌۢ بِنَاصِيَتِهَا؛ جس کا مفہوم یہ ہے کہ ہر جاندار کی پیشانی ایک ہی ہاتھ میں ہے اور اس گرفت میں کوئی شریک نہیں۔ اس قوم کے سامنے، جو اپنی قوت پر نازاں تھی، اس نبی کو بھیجا گیا جس کی حکمت یہ تھی کہ قوت کا مالک ایک ہے؛ تمہارا زور بھی ادھار ہے اور تمہاری پیشانی بھی اسی کے ہاتھ میں۔

اِنِّيْ لَكُمْ رَسُوْلٌ اَمِيْنٌ، فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ، وَمَآ اَسْـَٔلُكُمْ عَلَيْهِ مِنْ اَجْرٍ ۚ اِنْ اَجْرِيَ اِلَّا عَلٰى رَبِّ الْعٰلَمِيْنَ۔ میں تمہارے لیے امانت دار رسول ہوں، تو اللہ سے ڈرو اور میری اطاعت کرو؛ اور میں اس پر تم سے کوئی اجر نہیں مانگتا، میرا اجر تو بس رب العالمین کے ذمے ہے۔ وہی دعوت حرف بحرف، مگر اس قصے میں اس کی دھار نئی ہے۔ عاد اپنے آپ کو زمین کا مالک سمجھتے تھے؛ ان کے سامنے "رَبِّ الْعٰلَمِيْنَ" کہنا یہ یاد دلانا تھا کہ جہانوں کا رب کوئی اور ہے۔ وہ جبر سے لیتے تھے، اور یہ نبی کچھ نہیں لیتا؛ وہ دوسروں کا حق کھا کر بڑھتے تھے، اور یہ اپنا حق بھی چھوڑ کر آیا۔ "اَمِيْنٌ" کا مقابل اس رکوع میں "جَبَّارِيْنَ" ہے: ایک امانت پہنچاتا ہے، دوسرا امانت چھینتا ہے۔ اہلِ معرفت کہتے ہیں کہ اسی ایک دعوت کا بار بار لوٹنا خود ایک نشانی ہے کہ حق کا پیغام نہیں بدلتا، دلوں کے مرض بدلتے ہیں؛ اور ہر مرض پر وہی کلمہ نئی دوا بن کر اترتا ہے۔

اَتَبْنُوْنَ بِكُلِّ رِيْعٍ اٰيَةً تَعْبَثُوْنَ۔ کیا تم ہر اونچی جگہ پر ایک یادگار بناتے ہو، محض بے مقصد؟ اب دعوت عمومی اصول سے ان کے خاص مرض پر آئی۔ وہ ہر بلندی پر عمارت کھڑی کرتے تھے، کسی ضرورت کے لیے نہیں، اپنی شان دکھانے کے لیے۔ قرآن نے ان عمارتوں کو "اٰيَةً" کہا، اور یہ لفظ چونکا دیتا ہے: ساری کائنات حق کی آیات سے بھری ہوئی تھی، اور وہ اپنی آیات کھڑی کرنے میں لگے تھے۔ اہلِ معرفت کے نزدیک انسان خود حق کی سب سے جامع آیت ہے، اسمِ جامع کا مظہر؛ جو اپنے اندر کی آیت سے غافل ہو، وہ باہر پتھر کی آیتیں بناتا ہے تاکہ اپنے ہونے کا یقین کر سکے۔ "تَعْبَثُوْنَ" یعنی بے مقصد کھیل؛ عبث وہ عمل ہے جس کا رخ کسی حقیقت کی طرف نہ ہو۔ شیخِ اکبر کے مکتب میں کوئی شے حق کے ہاں عبث نہیں، ہر شے کسی اسم کا مظہر ہے؛ عبث تو بندے کی نیت میں پیدا ہوتا ہے جب وہ مظہر کو اس کے ظاہر کرنے والے سے کاٹ دیتا ہے۔ سالک کے لیے سبق یہ ہے کہ اپنے ہر کام سے پوچھے: یہ کس کی نشانی بن رہا ہے، میری یا اس کی؟

وَتَتَّخِذُوْنَ مَصَانِعَ لَعَلَّكُمْ تَخْلُدُوْنَ۔ اور تم مضبوط محل اور قلعے بناتے ہو گویا تمہیں ہمیشہ رہنا ہے۔ یہ اس رکوع کا مرکزی نکتہ ہے: ہمیشگی کا فریب۔ "لَعَلَّكُمْ" میں ان کے دل کی امید کھل گئی؛ وہ پتھر کو مضبوط کر کے اپنی بقا خریدنا چاہتے تھے۔ یہی بیماری سورۂ الکہف کے باغ والے میں تھی جس نے کہا: مَآ اَظُنُّ اَنْ تَبِيْدَ هٰذِهٖۤ اَبَدًا۔ شیخِ اکبر کے نزدیک خلقِ جدید کا راز اس فریب کو جڑ سے کاٹتا ہے: کوئی شے دو لمحے ایک حال پر نہیں رہتی، ہر لمحہ وجود نئے سرے سے عطا ہوتا ہے؛ پھر جو اپنے ہی لمحے کا مالک نہیں، وہ ہمیشگی کا مالک کیسے بنے؟ بقا صرف اسی کی ہے: كُلُّ شَيْءٍ هَالِكٌ اِلَّا وَجْهَهٗ۔ اور سورۂ النحل نے یہی اصول یوں کہا تھا: مَا عِنْدَكُمْ يَنْفَدُ وَمَا عِنْدَ اللّٰهِ بَاقٍ۔ اہلِ ذوق کہتے ہیں کہ خلود کی پیاس انسان میں سچی ہے، کیونکہ اس کی روح باقی سے آئی ہے؛ غلطی پیاس میں نہیں، پانی کی جگہ میں ہے۔ جو اسے پتھر میں ڈھونڈے وہ پیاسا مرتا ہے، اور جو اپنے آپ کو باقی کے سپرد کر دے، فنا کے بعد اسی کی بقا سے باقی ہو جاتا ہے۔

وَاِذَا بَطَشْتُمْ بَطَشْتُمْ جَبَّارِيْنَ۔ اور جب کسی پر ہاتھ ڈالتے ہو تو سراسر جابر بن کر ڈالتے ہو۔ فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ۔ تو اللہ سے ڈرو اور میری اطاعت کرو۔ تکبر تین منزلوں میں اترا: نمائش کی عمارت، ہمیشگی کی امید، اور پھر کمزور پر بے رحم گرفت۔ جو اپنے آپ کو ہمیشہ رہنے والا سمجھ لے، وہ دوسروں کو کچلنے میں جھجکتا نہیں۔ "الْجَبَّارُ" حق تعالیٰ کا اسم ہے، اور عربی میں "جبر" کا ایک اصل معنی ٹوٹی ہڈی کو جوڑنا ہے۔ اہلِ معرفت کہتے ہیں کہ تخلق باسماء اللہ میں اس اسم سے بندے کا حصہ یہ ہے کہ ٹوٹے دلوں کو جوڑے؛ عاد نے اسی اسم کا الٹا لباس پہنا اور جوڑنے کی جگہ توڑنے لگے۔ اسم جب اپنے محل اور ادب سے ہٹ کر بندے میں ظاہر ہو تو کمال کی جگہ ظلم بن جاتا ہے۔ اسی لیے فوراً بعد "فَاتَّقُوا اللّٰهَ وَاَطِيْعُوْنِ" لوٹا؛ اس بار یہ دعوت جبر کا علاج بن کر آئی۔ جابر کی دوا تقویٰ ہے، کیونکہ تقویٰ اسے یاد دلاتا ہے کہ اس کے اوپر بھی ایک ہاتھ ہے، اور اسی ہاتھ میں اس کی پیشانی ہے۔

وَاتَّقُوا الَّذِيْۤ اَمَدَّكُمْ بِمَا تَعْلَمُوْنَ۔ اور اس سے ڈرو جس نے تمہیں وہ سب کچھ دیا جو تم جانتے ہو۔ اَمَدَّكُمْ بِاَنْعَامٍ وَّبَنِيْنَ۔ اس نے تمہیں مویشیوں اور بیٹوں سے نوازا۔ وَجَنّٰتٍ وَّعُيُوْنٍ۔ اور باغوں اور چشموں سے۔ یہاں تقویٰ کو عطا سے جوڑا گیا: ڈرو اس سے جو دیتا ہے۔ پہلے اجمال آیا، "بِمَا تَعْلَمُوْنَ"، تم خود جانتے ہو؛ پھر تفصیل، تاکہ گنتی کرتے کرتے دل دینے والے تک پہنچ جائے۔ "اَمَدَّ" کا لفظ مسلسل رسد کو کہتے ہیں، ایک بار کی بخشش کو نہیں۔ شیخِ اکبر کے نزدیک ہر موجود ہر لمحہ نفَسِ رحمانی سے وجود کی رسد پا رہا ہے؛ اگر یہ رسد ایک لمحہ رکے تو کچھ باقی نہ رہے۔ عاد اپنے مویشیوں، بیٹوں اور باغوں کو اپنی کمائی سمجھ بیٹھے تھے، حالانکہ یہ سب ایک جاری عطا کی لہریں تھیں۔ نعمت کو دیکھنا اور دینے والے کو نہ دیکھنا حجاب ہے؛ نعمت میں دینے والے کو دیکھنا شہود ہے۔ اہلِ ذوق "عُيُوْنٍ" میں دل کے ان چشموں کی تمثیل بھی دیکھتے ہیں جن سے معرفت پھوٹتی ہے؛ یہ تمثیل ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہری چشمے بھی حقیقی نعمت تھے جن پر شکر واجب تھا۔

اِنِّيْۤ اَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ عَظِيْمٍ۔ مجھے تم پر ایک بڑے دن کے عذاب کا ڈر ہے۔ قَالُوْا سَوَآءٌ عَلَيْنَاۤ اَوَعَظْتَ اَمْ لَمْ تَكُنْ مِّنَ الْوٰعِظِيْنَ۔ وہ بولے: ہمارے لیے برابر ہے، تو نصیحت کرے یا نصیحت کرنے والوں میں سے نہ ہو۔ نبی کا ڈر اپنے لیے نہیں، "عَلَيْكُمْ"، تمہارے لیے ہے؛ یہ وہ شفقت ہے جو انکار کے بعد بھی ختم نہیں ہوتی۔ اور قوم کا جواب اس رکوع میں ہدایت و ضلالت کا اصول کھول دیتا ہے؛ اسے اصول سمجھ کر پڑھیے۔ سورۂ البقرہ میں حق تعالیٰ نے ایک گروہ کے بارے میں فرمایا: سَوَآءٌ عَلَيْهِمْ ءَاَنْذَرْتَهُمْ اَمْ لَمْ تُنْذِرْهُمْ لَا يُؤْمِنُوْنَ۔ یہاں وہی "سَوَآءٌ" بندوں نے اپنی زبان سے اپنے بارے میں کہا: ہمارے لیے برابر ہے۔ یہ ظاہر کرتا ہے کہ وہ برابری ان پر زبردستی نہیں ڈالی گئی تھی؛ وہ ان کا اپنا اعلان تھا، ان کا اپنا بند کیا ہوا دروازہ۔ حق نے پہلے نبی بھیجا، نصیحت دی، ڈرایا؛ جب انہوں نے خود کہہ دیا کہ نصیحت اور خاموشی ہمارے لیے ایک ہیں، تو حکم اسی حال کے مطابق ہوا۔ یہی سرِّ قدر ہے: حکم محکوم کے مطابق؛ اور یہی اعراض ہے جو خود ضلالت ہے۔

اِنْ هٰذَاۤ اِلَّا خُلُقُ الْاَوَّلِيْنَ۔ یہ تو بس پہلے لوگوں کی پرانی روش ہے۔ وَمَا نَحْنُ بِمُعَذَّبِيْنَ۔ اور ہمیں عذاب ہونے والا نہیں۔ مفسرین نے "هٰذَا" کے دو معنی لکھے ہیں: ہمارا یہ طریقہ باپ دادا کی روش ہے جسے ہم نہیں چھوڑیں گے، یا تمہاری یہ باتیں پرانے لوگوں کی رسم ہیں۔ دونوں صورتوں میں دلیل ایک ہے: پرانا ہونا۔ اہلِ معرفت کہتے ہیں کہ یہ تقلید کا پردہ ہے۔ حق کی تجلی ہر لمحہ نئی ہے، اور جو ہر نئی پکار کو "پرانی بات" کہہ کر ٹال دے، وہ خلقِ جدید کے سامنے آنکھیں بند کر کے بیٹھا ہے۔ سورۂ المؤمنون میں اسی حال کی تصویر تھی: كُلُّ حِزْبٍۢ بِمَا لَدَيْهِمْ فَرِحُوْنَ۔ پھر "وَمَا نَحْنُ بِمُعَذَّبِيْنَ" وہ بے خوفی ہے جو ایمان کا امن نہیں، غفلت کا امن ہے۔ قرآن نے اس امن کو یوں پرکھا: فَلَا يَاْمَنُ مَكْرَ اللّٰهِ اِلَّا الْقَوْمُ الْخٰسِرُوْنَ۔ مومن کا امن حق کی پناہ سے ہوتا ہے اور اس کے ساتھ خوف و امید دونوں رہتے ہیں؛ غافل کا امن اپنے زور سے ہوتا ہے اور وہ زور ادھار کا ہے۔

فَكَذَّبُوْهُ فَاَهْلَكْنٰهُمْ ۭ اِنَّ فِيْ ذٰلِكَ لَاٰيَةً ۭ وَمَا كَانَ اَكْثَرُهُمْ مُّؤْمِنِيْنَ۔ تو انہوں نے اسے جھٹلایا، پس ہم نے انہیں ہلاک کر دیا؛ بے شک اس میں ایک نشانی ہے، اور ان میں سے اکثر ایمان لانے والے نہ تھے۔ وَاِنَّ رَبَّكَ لَهُوَ الْعَزِيْزُ الرَّحِيْمُ۔ اور یقیناً تیرا رب ہی غالب، نہایت رحم والا ہے۔ پانچویں بار یہ بند لوٹا، اور عاد کے بعد اس کا معنی قوت کا معنی ہے۔ وہ کہتے تھے: مَنْ اَشَدُّ مِنَّا قُوَّةً؛ اور جواب وہیں دیا گیا کہ جس نے انہیں پیدا کیا وہ ان سے کہیں زیادہ قوت والا ہے۔ یہاں "الْعَزِيْزُ" کا مطلب ہے وہ جس کی عزت تک کوئی ادھار کی قوت نہیں پہنچ سکتی۔ اور "الرَّحِيْمُ" کا جلوہ دوسری جگہ کھول کر بتایا گیا: نَجَّيْنَا هُوْدًا وَّالَّذِيْنَ اٰمَنُوْا مَعَهٗ بِرَحْمَةٍ مِّنَّا۔ اور ایک لطیف بات: انہوں نے ہر بلندی پر "اٰيَةً" بنائی تھی تاکہ ہمیشہ یاد رکھے جائیں؛ وہ یاد تو رکھے گئے، مگر خود "لَاٰيَةً" بن کر، عبرت کی نشانی بن کر۔ جو اپنی نشانی بنانا چاہے وہ آخرکار حق کی نشانی بنتا ہے، یا رحمت کی یا عبرت کی؛ اور یہ انتخاب بندے کا ہے۔ اور خطاب پھر حضورِ اکرم صلی اللہ علیہ وسلم سے ہے: "رَبَّكَ"، تیرا رب؛ جو قوت والوں کے غرور کو توڑنے میں غالب ہے، اور ماننے والوں کو بچانے میں رحیم۔
