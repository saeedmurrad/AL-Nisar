پچھلا رکوع گہوارے سے شروع ہو کر ماں کی گود پر ختم ہوا تھا؛ یہ رکوع جوانی سے شروع ہو کر ہجرت پر ختم ہوتا ہے، اور اس کا پہلا جملہ ہی عطا کا جملہ ہے: وَلَمَّا بَلَغَ اَشُدَّهٗ وَاسْتَوٰۤى اٰتَيْنٰهُ حُكْمًا وَّعِلْمًا ۭ وَكَذٰلِكَ نَجْزِي الْمُحْسِنِيْنَ۔ اور جب وہ اپنی پوری قوت کو پہنچے اور بھرپور ہو گئے تو ہم نے انہیں حکم اور علم عطا کیا، اور نیکو کاروں کو ہم یوں ہی بدلہ دیتے ہیں۔ مفسرین کہتے ہیں کہ اشد جسم کی قوت کی انتہا ہے اور استوا عقل و مزاج کا اعتدال؛ اور یہاں حکم سے مراد دانائی اور فیصلے کی درستی ہے، کیونکہ رسالت تو برسوں بعد طور کے دامن میں ملنی تھی۔ سورۂ یوسف میں یہی عطا حضرت یوسف علیہ السلام کے لیے انہی لفظوں میں آئی تھی: اٰتَيْنٰهُ حُكْمًا وَّعِلْمًا؛ مگر حضرت موسیٰ علیہ السلام کے ذکر میں استوا کا اضافہ ہے، گویا یہاں پختگی کی تکمیل پر زور ہے، کہ آگے بوجھ بھی بھاری ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں لفظ اٰتینا پر نظر ٹھہرتی ہے: علم یہاں کمایا ہوا نہیں، دیا ہوا ہے؛ بندہ محل ہے اور معلمِ حقیقی حق تعالیٰ۔ اور یہ عطا احسان کی جزا ہے، وہی احسان جسے حدیثِ احسان نے یوں کھولا کہ تم اللہ کی عبادت ایسے کرو گویا اسے دیکھ رہے ہو۔ جو عمل میں حق کو دیکھنے لگے اسے چیزوں میں حکمت دیکھنے کی آنکھ دے دی جاتی ہے۔ اور ادب کا ایک نکتہ یہیں سے نکلتا ہے: اگلی آیت کے واقعے سے عین پہلے حق سبحانہ خود انہیں محسنین میں شمار فرماتا ہے، تو جس کی گواہی آسمان سے محسن ہونے کی آئی ہو اس کے کسی فعل کو جرم کہنا کسی کو زیبا نہیں۔

اب وہ واقعہ آتا ہے جسے ادب کی پوری احتیاط سے پڑھنا لازم ہے: وَدَخَلَ الْمَدِيْنَةَ عَلٰى حِيْنِ غَفْلَةٍ مِّنْ اَهْلِهَا فَوَجَدَ فِيْهَا رَجُلَيْنِ يَقْتَتِلٰنِ۔ اور وہ شہر میں ایسے وقت داخل ہوئے جب اس کے باشندے غفلت میں تھے، تو وہاں دو آدمیوں کو باہم لڑتے پایا۔ ایک ان کی اپنی قوم بنی اسرائیل میں سے تھا اور دوسرا فرعون کے لوگوں میں سے ایک قبطی، جو مظلوم کو دبا رہا تھا؛ مظلوم نے مدد کو پکارا: فَوَكَزَهٗ مُوْسٰى فَقَضٰى عَلَيْهِ۔ تو موسیٰ علیہ السلام نے اسے ایک گھونسا مارا اور اس کا کام تمام ہو گیا۔ وکز بند مٹھی کی اس ضرب کو کہتے ہیں جس سے عادتاً کوئی نہیں مرتا؛ ارادہ چھڑانے اور ظالم کو ہٹانے کا تھا، جان لینے کا ہرگز نہیں۔ یہ واقعہ نبوت سے پہلے کا ہے، بلا ارادہ پیش آیا، اور انبیائے کرام علیہم السلام معصوم ہیں؛ اسے گناہ یا جرم کہنا ادب کے خلاف ہے۔ سورۂ الشعراء میں خود آپ نے فرعون کے سامنے فرمایا کہ وہ بات اس وقت ہوئی جب انجام کی خبر نہ تھی۔ پھر آپ کی زبان سے نکلا: قَالَ هٰذَا مِنْ عَمَلِ الشَّيْطٰنِ ۭ اِنَّهٗ عَدُوٌّ مُّضِلٌّ مُّبِيْنٌ۔ کہا: یہ شیطان کے کام سے ہے، بے شک وہ کھلا بہکانے والا دشمن ہے۔ بعض اہلِ تفسیر کے نزدیک اشارہ اس جھگڑے کی طرف ہے جسے شیطان نے بھڑکایا تھا۔ یہ نبی کا ادب ہے کہ ہر کمی کو اپنی یا دشمن کی طرف منسوب کرے اور حق کی طرف صرف خیر کو۔ اور مُضِلّ کا لفظ اصول بھی بتا دیتا ہے: شیطان کا کام صرف بلاوا ہے، زبردستی اس کے بس میں نہیں۔ اہلِ تاویل اس شہر کو وجودِ انسانی کی بستی بھی کہتے ہیں، جس میں روح کی جماعت اور نفس کی جماعت باہم الجھتی رہتی ہیں؛ مگر یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

اس کے بعد وہ دعا آئی جو ہر سالک کی زبان کا ورد ہونی چاہیے: قَالَ رَبِّ اِنِّيْ ظَلَمْتُ نَفْسِيْ فَاغْفِرْ لِيْ فَغَفَرَ لَهٗ ۭ اِنَّهٗ هُوَ الْغَفُوْرُ الرَّحِيْمُ۔ عرض کیا: اے میرے رب! میں نے اپنی جان پر زیادتی کی، سو مجھے بخش دے؛ تو اس نے انہیں بخش دیا، بے شک وہی بخشنے والا، مہربان ہے۔ انبیاء کی زبان میں ظلمِ نفس کا مطلب وہ نہیں جو عام لوگوں کی زبان میں ہے؛ اہلِ تصوف میں یہ قول مشہور ہے کہ نیکوں کی نیکیاں مقربوں کے حق میں کوتاہیاں شمار ہوتی ہیں۔ جو جتنا قریب ہو اس کا پیمانہ اتنا ہی باریک ہو جاتا ہے۔ شیخِ اکبر کے نزدیک مغفرت کی اصل ستر ہے، یعنی ڈھانپ لینا: حق بندے کو اس کمی کے اثر سے بھی ڈھانپ لیتا ہے اور اس کی اپنی نظر سے بھی۔ اور فَغَفَرَ کی فا پر غور کیجیے: مانگنے اور ملنے کے درمیان کوئی فاصلہ نہیں۔ سورۂ طٰہٰ میں حق سبحانہ نے اسی واقعے کو اپنے احسانات کی فہرست میں رکھا: وَقَتَلْتَ نَفْسًا فَنَجَّيْنٰكَ مِنَ الْغَمِّ؛ یعنی یہ واقعہ عتاب کے باب میں نہیں، غم سے نجات کی نعمت کے باب میں لکھا گیا۔ پھر شکر سے پیدا ہونے والا عہد آیا: قَالَ رَبِّ بِمَاۤ اَنْعَمْتَ عَلَيَّ فَلَنْ اَكُوْنَ ظَهِيْرًا لِّلْمُجْرِمِيْنَ۔ عرض کیا: اے میرے رب! اس نعمت کے سبب جو تو نے مجھ پر کی، میں ہرگز مجرموں کا پشت پناہ نہ بنوں گا۔ ظہیر ظہر سے ہے، یعنی پیٹھ؛ اہلِ معرفت کہتے ہیں کہ شکر کی حقیقت یہ ہے کہ نعمت کو دینے والے کے رخ پر خرچ کیا جائے۔ جس نے نعمت کا سرچشمہ دیکھ لیا وہ ظالم کی پیٹھ نہیں بن سکتا۔

اگلی صبح کا منظر سالک کے حال کی تصویر ہے: فَاَصْبَحَ فِي الْمَدِيْنَةِ خَآئِفًا يَّتَرَقَّبُ۔ پھر صبح کو وہ شہر میں ڈرتے ہوئے، چوکنا ہو کر نکلے۔ ترقب کا مادہ رقب ہے، اور یہی مراقبہ کا مادہ ہے؛ اہلِ ذوق کہتے ہیں کہ طالب کا حال یہی ہے: خوف بھی ہو اور آنے والے کا انتظار بھی، دل کی آنکھ کھلی رہے کہ اب کیا ظاہر ہوتا ہے۔ یہ خوف طبعی ہے اور توکل کے منافی نہیں۔ فَاِذَا الَّذِي اسْتَنْصَرَهٗ بِالْاَمْسِ يَسْتَصْرِخُهٗ ۭ قَالَ لَهٗ مُوْسٰۤى اِنَّكَ لَغَوِيٌّ مُّبِيْنٌ۔ تو اچانک وہی شخص جس نے کل ان سے مدد مانگی تھی، آج پھر فریاد کو پکار رہا تھا؛ موسیٰ علیہ السلام نے اس سے کہا: تو یقیناً کھلا بہکا ہوا ہے۔ پھر جب آپ نے دشمن کو روکنا چاہا تو آواز اٹھی: قَالَ يٰمُوْسٰۤى اَتُرِيْدُ اَنْ تَقْتُلَنِيْ كَمَا قَتَلْتَ نَفْسًۢا بِالْاَمْسِ۔ بولا: اے موسیٰ! کیا تم مجھے بھی مار ڈالنا چاہتے ہو جیسے کل تم نے ایک جان کو مار ڈالا؟ اکثر مفسرین کے نزدیک یہ بات اسی اسرائیلی نے کہی جو آپ کی ڈانٹ سن کر سمجھا کہ ہاتھ اسی پر اٹھ رہا ہے، اور بعض کہتے ہیں کہ قبطی نے کہی۔ اور ساتھ یہ طعنہ: اِنْ تُرِيْدُ اِلَّاۤ اَنْ تَكُوْنَ جَبَّارًا فِي الْاَرْضِ وَمَا تُرِيْدُ اَنْ تَكُوْنَ مِنَ الْمُصْلِحِيْنَ۔ تم تو بس یہی چاہتے ہو کہ زمین میں زبردست بن کر رہو، اور اصلاح کرنے والوں میں سے ہونا نہیں چاہتے۔ اہلِ معرفت یہاں ایک سبق پڑھتے ہیں: کل جس زبان نے مدد مانگی تھی آج اسی نے راز فاش کر دیا۔ مخلوق کا سہارا آخرکار تمہارے خلاف فریاد بن جاتا ہے، تاکہ دل صرف حق پر ٹکنا سیکھ لے۔ اور دنیا کی آنکھ مصلح کو جبار کہتی ہے، جبکہ آسمان اسے پہلے ہی محسن کہہ چکا تھا۔

اب مدد وہاں سے آئی جہاں نگاہ نہ جاتی تھی: وَجَآءَ رَجُلٌ مِّنْ اَقْصَا الْمَدِيْنَةِ يَسْعٰى۔ اور شہر کے پرلے سرے سے ایک آدمی دوڑتا ہوا آیا۔ قَالَ يٰمُوْسٰۤى اِنَّ الْمَلَاَ يَاْتَمِرُوْنَ بِكَ لِيَقْتُلُوْكَ فَاخْرُجْ اِنِّيْ لَكَ مِنَ النّٰصِحِيْنَ۔ کہا: اے موسیٰ! سردار تمہارے بارے میں مشورے کر رہے ہیں کہ تمہیں قتل کر دیں، سو نکل جاؤ، میں تمہارا خیر خواہ ہوں۔ دربار کے قریب والے سازش کر رہے تھے اور خیر خواہ شہر کے دور ترین کنارے سے دوڑا آیا؛ اہلِ تاویل کہتے ہیں کہ باطن کی بستی میں بھی ناصح کی آواز اکثر اسی دور کے کنارے سے آتی ہے جسے ہم غیب کہتے ہیں۔ فَخَرَجَ مِنْهَا خَآئِفًا يَّتَرَقَّبُ ۫ قَالَ رَبِّ نَجِّنِيْ مِنَ الْقَوْمِ الظّٰلِمِيْنَ۔ تو وہ وہاں سے ڈرتے ہوئے، چوکنا ہو کر نکل کھڑے ہوئے؛ عرض کیا: اے میرے رب! مجھے ظالم قوم سے نجات دے۔ خائفاً یترقب کا دوبارہ آنا بتاتا ہے کہ وہی حال ہجرت میں بھی ساتھ رہا؛ نہ زادِ راہ، نہ رفیق، نہ راستہ؛ ساتھ تھی تو صرف دعا۔ سورۂ الشعراء میں آپ نے خود فرمایا: فَفَرَرْتُ مِنْكُمْ لَمَّا خِفْتُكُمْ فَوَهَبَ لِيْ رَبِّيْ حُكْمًا؛ یعنی یہی فرار عطا کا دروازہ بنا۔ شیخِ اکبر حکمتِ علویہ میں اس فرار پر ٹھہرتے ہیں، جس کا مفہوم یہ ہے کہ زبان پر خوف کا ذکر آیا مگر حرکت کی جڑ نجات کی محبت تھی، اور عالم کی ہر حرکت اپنی اصل میں محبت کی حرکت ہے۔ کوئی سلوک ہجرت کے بغیر شروع نہیں ہوتا، اور اس سفر کا پہلا لفظ نَجِّنِيْ ہے، جس کا جواب اگلے رکوع میں ایک بزرگ کی زبان سے آئے گا۔
