یہ سورت ایک ایسی خبر سے کھلتی ہے جو سننے والے کے قدموں تلے کی زمین ہلا دیتی ہے: اِذَا زُلْزِلَتِ الْاَرْضُ زِلْزَالَهَا ۝ وَاَخْرَجَتِ الْاَرْضُ اَثْقَالَهَا ۝ وَقَالَ الْاِنْسَانُ مَا لَهَا۔ “جب زمین اپنی پوری جنبش کے ساتھ ہلا ڈالی جائے گی، اور زمین اپنے سارے بوجھ باہر نکال پھینکے گی، اور انسان کہے گا: اسے کیا ہو گیا ہے؟” مفسرین کے نزدیک یہ قیامت کے آغاز کا بیان ہے۔ زلزال کی نسبت خود زمین کی طرف کی گئی، یعنی وہ زلزلہ جو اس کے شایانِ شان ہے اور اس کی آخری حد ہے؛ دنیا کے زلزلے کسی ایک خطے کو ہلاتے ہیں، یہ پوری زمین کو اس کی جڑوں تک ہلا دے گا۔ اثقال، ثقل کی جمع ہے، یعنی بوجھ۔ اکثر مفسرین کے نزدیک اس سے مراد وہ مردے ہیں جو اس کے پیٹ میں امانت کی طرح رکھے گئے، اور بعض نے اس میں وہ خزانے بھی شامل کیے جو اس نے چھپا رکھے ہیں؛ اسی ضمن میں بعض مفسرین نے صحیح مسلم کی وہ حدیث ذکر کی ہے جس کا مفہوم ہے کہ زمین اپنے جگر کے ٹکڑے سونے چاندی کے ستونوں کی صورت میں اگل دے گی۔ اور انسان سے مراد بعض کے نزدیک ہر وہ شخص ہے جو یہ منظر دیکھے گا، اور بعض کے نزدیک وہ منکر جو دوبارہ اٹھائے جانے کو نہیں مانتا تھا اور اب حیرت سے پوچھتا ہے۔ یہی ظاہری معنی ہے اور یہی اصل ہے؛ آگے جو اشارات آئیں گے وہ اس کی نفی نہیں، اس کی گہرائی ہیں۔

اہلِ معرفت کے نزدیک اس بڑے زلزلے سے پہلے ایک چھوٹا زلزلہ ہے جو ہر سالک پر اسی دنیا میں آتا ہے۔ انسان کا نفس بھی ایک زمین ہے: نیچے کی طرف جھکی ہوئی، بوجھل، اور جو کچھ اس میں دفن کیا جائے اسے خاموشی سے چھپا لینے والی۔ برسوں کی خواہشیں، پوشیدہ تکبر، تعریف کی بھوک، پرانی رنجشیں، سب اس کی تہوں میں دبے رہتے ہیں، اور سطح پر سکون دکھائی دیتا ہے۔ بندہ اس سکون کو پاکیزگی سمجھ بیٹھتا ہے۔ پھر کوئی سچا وارد اترتا ہے، یا کوئی آزمائش آتی ہے، یا سچی توبہ دل کو جھنجھوڑتی ہے، تو یہ زمین ہلتی ہے اور اپنے اثقال باہر پھینک دیتی ہے؛ وہ سب سامنے آ جاتا ہے جس کا بندے کو گمان بھی نہ تھا۔ اہلِ ذوق کہتے ہیں کہ یہ ہلنا عذاب نہیں، رحمت ہے، کیونکہ جو چیز یہاں نکل آئے وہ دھوئی جا سکتی ہے، اور جو وہاں نکلے گی وہ صرف دکھائی جائے گی۔ مگر اثقال میں صرف کچرا نہیں، خزانے بھی ہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک انسان نسخۂ جامعہ ہے، جس میں اسماءِ الٰہیہ کے آثار امانت رکھے گئے ہیں، اور یہ خزانے بھی نفس کی مٹی تلے دبے رہتے ہیں۔ جب زمین ہلتی ہے تو دونوں نکلتے ہیں: کھوٹ بھی، تاکہ چھوڑا جائے، اور سونا بھی، تاکہ پہچانا جائے۔

اب اس سوال پر ٹھہریے جو انسان کی زبان سے نکلتا ہے: اسے کیا ہو گیا؟ وہ ساری عمر اسی زمین پر چلتا رہا، اسی سے کھاتا رہا، اسی پر سجدہ کرتا رہا، مگر اسے ایک بے جان فرش سمجھتا رہا؛ آج اسے ایک اجنبی کی طرح دیکھ کر پوچھتا ہے۔ اہلِ تاویل کہتے ہیں کہ یہی سوال غافل کی زبان پر اس وقت بھی آتا ہے جب اس کے اپنے اندر کا زلزلہ اٹھتا ہے: مجھے کیا ہو گیا ہے؟ وہ اپنے نفس کو نہیں پہچانتا، اس لیے اس کی جنبش اسے بیگانی لگتی ہے۔ اہلِ تصوف میں ایک اثر مشہور ہے کہ جس نے اپنے نفس کو پہچانا اس نے اپنے رب کو پہچانا؛ اور جو اپنی زمین سے بے خبر رہا وہ اپنے ہی زلزلے پر حیران کھڑا رہ گیا۔ مگر شیخِ اکبر کے مکتب میں یہ حیرت انجام نہیں، دروازہ ہے۔ غور کیجیے کہ سوال انسان نے کیا اور جواب اگلی آیت میں خود زمین دیتی ہے؛ یعنی خبر جاننے والے کی طرف سے نہیں، معلوم کی طرف سے آتی ہے۔ اس مکتب کا ایک بنیادی اصول ہے کہ علم معلوم کے تابع ہے: حق تعالیٰ ہر شے کو ویسا ہی جانتا ہے جیسی وہ ہے، اور اسی کے مطابق اس پر حکم کرتا ہے۔ چنانچہ اس دن انسان کو اپنی حقیقت کسی اور کی زبانی نہیں، اپنی ہی زمین کی زبانی سننی ہوگی۔

پھر وہ گھڑی آتی ہے جب خاموش زمین بولتی ہے: يَوْمَئِذٍ تُحَدِّثُ اَخْبَارَهَا ۝ بِاَنَّ رَبَّكَ اَوْحٰى لَهَا۔ “اس دن وہ اپنی ساری خبریں بیان کر دے گی، اس لیے کہ تمہارے رب نے اسے یہی حکم بھیجا ہوگا۔” ان خبروں کی تفسیر خود حضورِ اکرم صلی اللہ علیہ وسلم نے فرما دی۔ امام ترمذی رحمۃ اللہ علیہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے یہ آیت تلاوت فرمائی اور پوچھا: جانتے ہو اس کی خبریں کیا ہیں؟ صحابہ نے عرض کیا: اللہ اور اس کا رسول بہتر جانتے ہیں۔ فرمایا: اس کی خبریں یہ ہیں کہ وہ ہر بندے اور ہر بندی پر اس عمل کی گواہی دے گی جو اس نے اس کی پشت پر کیا، کہے گی کہ فلاں نے فلاں دن یہ کیا اور یہ کیا۔ تو زمین کی خبر کوئی تاریخ یا جغرافیہ نہیں، انسان ہی کی سرگزشت ہے؛ اس کے پاس اپنی کوئی کہانی نہیں، جو کچھ ہے وہ اس پر چلنے والوں کا ہے۔ اور وحی یہاں نبوت والی وحی نہیں؛ مفسرین کے نزدیک اس سے مراد حکم دینا، اشارہ کرنا اور بولنے کی اجازت دینا ہے، جیسے شہد کی مکھی کی طرف وحی کا ذکر آیا۔ ایک لطیف نکتہ یہ ہے کہ اس ہولناک منظر میں بھی رب کا ذکر رَبَّكَ کے لفظ سے ہوا، یعنی حضورِ اکرم صلی اللہ علیہ وسلم کا رب: ہیبت کے بیچ قربت کا ایک دریچہ۔

شیخِ اکبر کے نزدیک زمین کا یہ بولنا کوئی نئی زندگی نہیں جو اسے اس دن دی جائے گی؛ وہ ہمیشہ سے زندہ تھی، ہمیشہ سے دیکھ رہی تھی، ہمیشہ سے اپنے رب کی تسبیح کر رہی تھی۔ اس مکتب میں جمادات، نباتات اور حیوانات سب حقیقی معنی میں زندہ اور ناطق ہیں، اور ان کی تسبیح محض زبانِ حال کا استعارہ نہیں بلکہ ایک حقیقی کلام ہے جسے اہلِ کشف سنتے ہیں۔ قرآن نے یہی فرمایا: وَاِنْ مِّنْ شَيْءٍ اِلَّا يُسَبِّحُ بِحَمْدِهٖ وَلٰكِنْ لَّا تَفْقَهُوْنَ تَسْبِيْحَهُمْ۔ تو اس دن جو بدلے گا وہ زمین نہیں، ہمارے کانوں پر پڑا ہوا پردہ ہے۔ جس دن پردہ اٹھے گا، ہر ذرہ وہی سنائے گا جو اس نے دیکھا تھا۔ اہلِ ذوق اسی سے ادب کا سبق لیتے ہیں: جو جانتا ہے کہ اس کے قدموں تلے کی مٹی گواہ ہے، وہ اس پر اکڑ کر نہیں چلتا، اور اس پر چھپ کر گناہ نہیں کرتا، کیونکہ اس پر کوئی جگہ چھپی ہوئی نہیں۔ مگر یہ گواہی صرف خوف کی بات نہیں۔ زمین کا وہ ٹکڑا جس پر کسی نے رات کے اندھیرے میں سجدہ کیا، جہاں کسی نے چپکے سے کسی یتیم کا ہاتھ تھاما، جہاں کسی کے آنسو گرے، وہ بھی بولے گا۔ اسی لیے اہلِ دل کہتے ہیں کہ ہر نئی جگہ پر ایک سجدہ چھوڑ جاؤ، تاکہ زمین کے بہت سے ٹکڑے تمہارے حق میں گواہ بنیں۔

زمین کی گواہی کے بعد منظر انسانوں کی طرف پلٹتا ہے: يَوْمَئِذٍ يَّصْدُرُ النَّاسُ اَشْتَاتًا ۙ لِّيُرَوْا اَعْمَالَهُمْ۔ “اس دن لوگ الگ الگ ٹولیوں میں لوٹیں گے، تاکہ انہیں ان کے اعمال دکھائے جائیں۔” صدر عربی میں اس لوٹنے کو کہتے ہیں جو گھاٹ سے پانی لے کر واپسی پر ہو، جیسے ورود گھاٹ پر اترنا ہے۔ اہلِ معرفت کے نزدیک یہ تعبیر خود ایک اشارہ ہے: دنیا ایک گھاٹ تھی، ہر شخص اس پر اترا، اپنی استعداد اور اپنی نیت کے برتن میں پانی بھرا، اور اب وہی برتن لے کر لوٹ رہا ہے۔ اشتات، یعنی بکھرے ہوئے اور الگ الگ: کوئی کسی کا حساب نہیں بانٹتا، ہر ایک اپنے حال کی ٹولی میں ہے۔ لیروا مجہول کا صیغہ ہے، یعنی انہیں دکھایا جائے گا؛ مفسرین کے نزدیک یا تو اعمال نامے دکھائے جائیں گے یا اعمال کی جزا۔ شیخِ اکبر کے مکتب میں اس کا ایک گہرا رخ تجسمِ اعمال ہے: عمل جو یہاں ایک گزر جانے والی حرکت تھا، وہاں ایک صورت اختیار کر کے سامنے آئے گا، اور بندہ اپنے ہی کیے ہوئے سے ملاقات کرے گا۔ ایک حدیثِ قدسی کا مفہوم، جسے امام مسلم رحمۃ اللہ علیہ نے روایت کیا، یہ ہے کہ یہ تمہارے ہی اعمال ہیں جنہیں میں تمہارے لیے گن رکھتا ہوں، پھر تمہیں ان کا پورا بدلہ دیتا ہوں؛ تو جو بھلائی پائے وہ اللہ کی حمد کرے، اور جو کچھ اور پائے وہ اپنے سوا کسی کو ملامت نہ کرے۔ اور سورۂ القیامہ نے اسی کی جڑ بتا دی تھی: بَلِ الْاِنْسَانُ عَلٰى نَفْسِهٖ بَصِيْرَةٌ۔

اور اب وہ دو آیتیں جن پر سورت ختم ہوتی ہے اور جو اس پورے باب کا اصول ہیں: فَمَنْ يَّعْمَلْ مِثْقَالَ ذَرَّةٍ خَيْرًا يَّرَهٗ ۝ وَمَنْ يَّعْمَلْ مِثْقَالَ ذَرَّةٍ شَرًّا يَّرَهٗ۔ “تو جس نے ذرہ برابر نیکی کی ہوگی وہ اسے دیکھ لے گا، اور جس نے ذرہ برابر برائی کی ہوگی وہ اسے دیکھ لے گا۔” یہی اصول ہے: ہر عمل بندے کا اپنا ہے اور لوٹ کر اسی کے پاس آتا ہے؛ نہ کسی کا عمل کسی اور کے کھاتے میں جاتا ہے، نہ کسی پر وہ ڈالا جاتا ہے جو اس نے نہیں کیا۔ امام بخاری رحمۃ اللہ علیہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم سے گدھوں کے بارے میں پوچھا گیا تو فرمایا کہ ان کے بارے میں مجھ پر اس یکتا اور جامع آیت کے سوا کچھ نازل نہیں ہوا، اور پھر یہی آیتیں تلاوت فرمائیں۔ اسی لیے انہیں جامعہ فاذہ کہا جاتا ہے: یکتا، بے نظیر، اور ہر خیر و شر کو سمیٹ لینے والی۔ اہلِ معرفت ایک لطیف ربط کی طرف اشارہ کرتے ہیں: سورت کے آغاز میں زمین نے اپنے اثقال نکالے، اور انجام پر انسان اپنا مثقال دیکھتا ہے؛ دونوں ایک ہی مادے ثقل سے ہیں۔ جو بوجھ اس نے زمین میں بھی دفن کیا اور اپنے نفس میں بھی، وہی وزن بن کر اس کے سامنے آتا ہے۔ اور شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ حق تعالیٰ کسی کو زبردستی کسی راہ پر نہیں ڈالتا؛ بندہ جس رخ پر چلتا ہے، حق اسی رخ کو وجود دیتا ہے، اور اس دن اسی رخ کا پھل اس کی جھولی میں ڈال دیا جاتا ہے۔

ذرے کا لفظ اس سورت کا سب سے چھوٹا اور سب سے بھاری لفظ ہے۔ اہلِ لغت کے نزدیک عرب اسے چیونٹی کے ننھے بچے یا روشنی میں تیرتے غبار کے ذرے کے لیے بولتے تھے، یعنی وہ چیز جس کا ترازو میں کوئی وزن محسوس نہ ہو۔ اسی سے اہلِ دل یہ ادب لیتے ہیں کہ کسی چھوٹی نیکی کو حقیر نہ جانو۔ امام بخاری رحمۃ اللہ علیہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: آگ سے بچو، چاہے کھجور کا آدھا ٹکڑا دے کر، اور جس کے پاس یہ بھی نہ ہو وہ اچھی بات کہہ کر۔ جس ہاتھ میں آدھی کھجور کے سوا کچھ نہ ہو، اس کے لیے بھی دروازہ کھلا ہے۔ سورۂ لقمان میں یہی بات ایک باپ کی زبان سے اپنے بیٹے کو سکھائی گئی: يٰبُنَيَّ اِنَّهَآ اِنْ تَكُ مِثْقَالَ حَبَّةٍ مِّنْ خَرْدَلٍ۔ شیخِ اکبر کے مکتب میں اس کی ایک اور گہرائی ہے: حق تعالیٰ کی نگاہ میں کچھ بھی حقیر نہیں، کیونکہ ہر ذرہ کسی نہ کسی اسمِ الٰہی کا مظہر ہے، اور اسمِ لطیف ہی کا تقاضا ہے کہ اس کا علم باریک ترین چیز تک پہنچے۔ جو عمل اخلاص سے ہوا، وہ ذرہ ہونے کے باوجود شکور کی بارگاہ میں پہنچتا ہے، اور شکور تھوڑے کو بہت کر کے لوٹاتا ہے۔ اور یہی نگاہ دوسری طرف بھی ہے: چھوٹے گناہ کو بھی ہلکا نہ جانو، کہ پہاڑ ذروں ہی سے بنتے ہیں۔

اہلِ ذوق کہتے ہیں کہ اس سورت کی ترتیب خود ایک سفر ہے: یہ پوری زمین کے ہلنے سے شروع ہوتی ہے اور ایک ذرے پر ختم ہوتی ہے؛ سب سے بڑی چیز سے سب سے چھوٹی چیز تک، تاکہ بتا دے کہ حساب میں نہ بڑا گم ہوتا ہے نہ چھوٹا۔ اور جو اس دن کے بڑے زلزلے سے امن چاہتا ہے، اس کا راستہ یہ ہے کہ آج اپنی زمین کو خود ہلائے۔ حضرت عمر رضی اللہ عنہ سے منسوب مشہور اثر کا مفہوم ہے کہ اپنا حساب کر لو اس سے پہلے کہ تمہارا حساب کیا جائے۔ اہلِ تصوف اسی کو محاسبہ کہتے ہیں: رات کو سونے سے پہلے دن بھر کے ذروں کو تولنا، نیکی پر شکر اور لغزش پر استغفار۔ پھر امید کی بات یہ ہے کہ جو زمین گناہ کی گواہ ہے وہی توبہ کی بھی گواہ ہے؛ جس جگہ بندے نے لغزش کی، اسی جگہ اس کے آنسو بھی گرے۔ امام ابنِ ماجہ رحمۃ اللہ علیہ کی روایت کا مفہوم ہے کہ گناہ سے توبہ کرنے والا ایسا ہے جیسے اس نے گناہ کیا ہی نہ ہو۔ تو یہ سورت ڈرانے کے لیے نہیں، جگانے کے لیے ہے: ہر ذرہ محفوظ ہے، اس لیے ہر ذرہ قیمتی ہے۔ حق کی رحمت اس کے غضب پر سبقت رکھتی ہے، اور جو آج اپنا رخ اس کی طرف کر لے، کل اس کی زمین اس کے حق میں بولے گی۔
