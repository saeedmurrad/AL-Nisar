وَلَقَدْ اَرْسَلْنَا نُوْحًا اِلٰى قَوْمِهٖ فَقَالَ يٰقَوْمِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَيْرُهٗ ۭ اَفَلَا تَتَّقُوْنَ۔ اور ہم نے نوح کو ان کی قوم کی طرف بھیجا، تو انہوں نے کہا: اے میری قوم! اللہ کی عبادت کرو، اس کے سوا تمہارا کوئی معبود نہیں؛ تو کیا تم ڈرتے نہیں؟ فَقَالَ الْمَلَؤُا الَّذِيْنَ كَفَرُوْا مِنْ قَوْمِهٖ مَا هٰذَآ اِلَّا بَشَرٌ مِّثْلُكُمْ ۙ يُرِيْدُ اَنْ يَّتَفَضَّلَ عَلَيْكُمْ۔ تو ان کی قوم کے کافر سرداروں نے کہا: یہ تو بس تم ہی جیسا ایک بشر ہے، چاہتا ہے کہ تم پر بڑائی حاصل کرے؛ وَلَوْ شَآءَ اللّٰهُ لَاَنْزَلَ مَلٰۤئِكَةً۔ اور اگر اللہ چاہتا تو فرشتے اتار دیتا؛ مَا سَمِعْنَا بِهٰذَا فِيْۤ اٰبَآئِنَا الْاَوَّلِيْنَ۔ ہم نے یہ بات اپنے اگلے باپ دادا میں کبھی نہیں سنی۔ پہلا رکوع اس جملے پر ختم ہوا تھا: وَعَلَيْهَا وَعَلَى الْفُلْكِ تُحْمَلُوْنَ؛ کشتی کا ذکر آیا تو کشتی والے پیغمبر کا قصہ کھل گیا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ نے فصوص الحکم میں حضرت نوح علیہ السلام کی حکمت کو حکمتِ سبوحیہ کہا ہے، جس کا مفہوم یہ ہے کہ ان کی دعوت میں تنزیہ کا رنگ غالب تھا، کیونکہ ان کی قوم حق کو پتھر کی صورتوں میں قید کر بیٹھی تھی؛ اور کامل معرفت تنزیہ و تشبیہ کے جمع میں ہے۔ اسی حکمت کے ذیل میں تقویٰ کی جو تعبیر ملتی ہے اس کا مفہوم یہ ہے کہ تقویٰ وقایہ یعنی ڈھال سے ہے: خوبیوں میں حق کو اپنی ڈھال بناؤ، اور کوتاہیوں میں اپنے آپ کو حق کے ادب کی ڈھال بناؤ کہ کوئی عیب اس کی طرف منسوب نہ ہو۔ اور سرداروں کا اعتراض دیکھیے: انہوں نے "مِّثْلُكُمْ" کی صورت دیکھی اور نبوت کا باطن نہ دیکھا۔ اہلِ معرفت کے نزدیک نبی ایک برزخ ہے: صورت میں بشر، اور باطن میں وحی کا امین۔ "يَّتَفَضَّلَ" کا الزام دراصل ان کے اپنے دل کا عکس تھا؛ بڑائی کی خواہش ان میں تھی، اسے پیغمبر پر ڈال دیا۔ اور "وَلَوْ شَآءَ اللّٰهُ" کہہ کر وہ خود اس اصول تک پہنچ گئے جس سے غافل تھے: ہاں، وہ چاہتا تو فرشتے اتار کر دلوں کو مجبور کر دیتا، مگر اس نے ہدایت کو ایک انسان کی زبان میں رکھا تاکہ ماننا بندے کا اپنا انتخاب رہے اور اختیار کی کرامت باقی رہے۔

اِنْ هُوَ اِلَّا رَجُلٌۢ بِهٖ جِنَّةٌ فَتَرَبَّصُوْا بِهٖ حَتّٰى حِيْنٍ۔ یہ تو بس ایک آدمی ہے جسے جنون ہو گیا ہے، سو ایک مدت تک اس کا انتظار کر دیکھو۔ قَالَ رَبِّ انْصُرْنِيْ بِمَا كَذَّبُوْنِ۔ نوح نے عرض کیا: اے میرے رب! میری مدد فرما، کیونکہ انہوں نے مجھے جھٹلا دیا ہے۔ "جِنَّةٌ" کا مادہ پوشیدگی ہے؛ جن، جنین، جنت اور مِجَن یعنی ڈھال سب اسی سے ہیں کہ ان میں کچھ چھپا ہوتا ہے۔ اہلِ ذوق کہتے ہیں کہ جو حقیقت پردے والوں کی آنکھ سے چھپی ہو، اس کے حامل کو وہ دیوانہ کہتے ہیں؛ ان کے نزدیک عقل وہی ہے جو ان کی عادت کے دائرے میں بند رہے، اور جو اس دائرے سے باہر کی خبر دے وہ ان کی نظر میں ہوش سے باہر ہے۔ یہی طعنہ اسی سورت میں آگے حضورِ اکرم صلی اللہ علیہ وسلم کے بارے میں بھی دہرایا جائے گا، کیونکہ منکر کی دلیل ہر زمانے میں ایک ہی رہتی ہے۔ پھر حضرت نوح علیہ السلام کی دعا کو دیکھیے: صدیوں کی دعوت کے بعد، جب ہر دروازہ آزما لیا گیا، تب یہ مختصر سا جملہ نکلا۔ شیخِ اکبر کے مکتب میں ہر بندے کا ایک ربِّ خاص ہے، یعنی وہ اسمِ الٰہی جو اس کی پرورش کرتا ہے؛ نبی جب "رَبِّ" کہتا ہے تو اسی پرورش کرنے والے اسم کو پکارتا ہے جس نے اسے اس منصب کے لیے تیار کیا۔ اور مدد اپنی ذات کے انتقام کے لیے نہیں مانگی گئی، "بِمَا كَذَّبُوْنِ" کہہ کر سبب کو ظاہر کر دیا گیا: فیصلہ ان کے اپنے جھٹلانے کے مطابق ہو۔ یہ سرِّ قدر کی صاف تعبیر ہے کہ حکم محکوم کے حال کے تابع ہے؛ جس نے صدیوں تک بلاوے پر کان بند رکھے، اس کا انجام اسی اعراض کی صورت ہے، کوئی بیرونی جبر نہیں۔

فَاَوْحَيْنَآ اِلَيْهِ اَنِ اصْنَعِ الْفُلْكَ بِاَعْيُنِنَا وَوَحْيِنَا فَاِذَا جَآءَ اَمْرُنَا وَفَارَ التَّنُّوْرُ۔ تو ہم نے ان کی طرف وحی بھیجی کہ ہماری آنکھوں کے سامنے اور ہماری وحی کے مطابق کشتی بناؤ، پھر جب ہمارا حکم آ جائے اور تنور ابل پڑے؛ فَاسْلُكْ فِيْهَا مِنْ كُلٍّ زَوْجَيْنِ اثْنَيْنِ وَاَهْلَكَ اِلَّا مَنْ سَبَقَ عَلَيْهِ الْقَوْلُ مِنْهُمْ ۚ وَلَا تُخَاطِبْنِيْ فِي الَّذِيْنَ ظَلَمُوْا ۚ اِنَّهُمْ مُّغْرَقُوْنَ۔ تو اس میں ہر قسم کے جوڑے میں سے دو دو رکھ لو اور اپنے گھر والوں کو بھی، سوائے ان کے جن کے بارے میں پہلے بات طے ہو چکی؛ اور ظالموں کے بارے میں مجھ سے بات نہ کرنا، وہ ضرور ڈبوئے جائیں گے۔ فَاِذَا اسْتَوَيْتَ اَنْتَ وَمَنْ مَّعَكَ عَلَى الْفُلْكِ فَقُلِ الْحَمْدُ لِلّٰهِ الَّذِيْ نَجّٰىنَا مِنَ الْقَوْمِ الظّٰلِمِيْنَ۔ پھر جب تم اور تمہارے ساتھی کشتی پر جم کر بیٹھ جاؤ تو کہنا: ساری تعریف اللہ کی جس نے ہمیں ظالم قوم سے نجات دی۔ سورۂ طٰہٰ میں حضرت موسیٰ علیہ السلام کے بارے میں فرمایا گیا تھا: وَلِتُصْنَعَ عَلٰى عَيْنِيْ؛ وہاں آنکھ ایک تھی اور بنایا جانے والا خود پیغمبر تھا، یہاں "بِاَعْيُنِنَا" جمع ہے اور بنائی جانے والی وہ کشتی ہے جس میں ہر جنس کا جوڑا سمانا ہے۔ اہلِ ذوق کہتے ہیں کہ جس عمل میں سارا عالم سوار ہو اس پر سارے اسماء کی نگاہ ہوتی ہے۔ ہاتھ بندے کا، آنکھ حق کی، نقشہ وحی کا: یہی کسب اور خلق کا جمع ہے۔ اور تنور کا ابلنا اس بات کی علامت ہے کہ جب امر آتا ہے تو اسباب اپنی عادت چھوڑ دیتے ہیں؛ آگ کی جگہ سے پانی پھوٹتا ہے، تاکہ معلوم ہو کہ سبب پردہ ہے، فاعل امر ہے۔ "زَوْجَيْنِ" میں اشارہ ہے کہ وجود جوڑوں پر قائم ہے؛ شیخِ اکبر کے مکتب میں فاعل اور قابل کا ملاپ ہر پیدائش کی بنیاد ہے۔ رہا "سَبَقَ عَلَيْهِ الْقَوْلُ" تو اہلِ معرفت کے نزدیک یہ پہلے سے طے شدہ بات اس علم کا نام ہے جو بندے کی حقیقت کو ویسا ہی جانتا ہے جیسی وہ ہے؛ علم معلوم کے تابع ہے، معلوم کو کفر پر مجبور نہیں کرتا۔ حضرت نوح علیہ السلام کے گھر کا جو فرد باہر رہا، وہ نسب کے باوجود اپنے انکار سے باہر رہا، اور کشتی میں بلانے کی آواز اس تک بھی پہنچی تھی۔ کشتی پر پہلا کلمہ حمد کا ہے، فخر کا نہیں؛ اور "اسْتَوَيْتَ" میں اہلِ ذوق تلوین کے بعد تمکین کا اشارہ دیکھتے ہیں: موجیں اٹھتی رہیں، مگر سوار جم کر بیٹھا رہا۔

وَقُلْ رَّبِّ اَنْزِلْنِيْ مُنْزَلًا مُّبٰرَكًا وَّاَنْتَ خَيْرُ الْمُنْزِلِيْنَ۔ اور کہو: اے میرے رب! مجھے برکت والی جگہ اتار، اور تو سب سے بہتر اتارنے والا ہے۔ اِنَّ فِيْ ذٰلِكَ لَاٰيٰتٍ وَّاِنْ كُنَّا لَمُبْتَلِيْنَ۔ بے شک اس میں بہت سی نشانیاں ہیں، اور ہم تو آزمائش کرنے والے ہیں ہی۔ کشتی پر سوار ہوتے وقت حمد سکھائی گئی اور اترتے وقت دعا؛ گویا ہر داخلہ اور ہر نکلنا حق سے مانگ کر ہو۔ سورۂ بنی اسرائیل میں یہی ادب حضورِ اکرم صلی اللہ علیہ وسلم کو سکھایا گیا تھا: وَقُلْ رَّبِّ اَدْخِلْنِيْ مُدْخَلَ صِدْقٍ وَّاَخْرِجْنِيْ مُخْرَجَ صِدْقٍ۔ اہلِ سلوک کے نزدیک ہر حال اور ہر مقام ایک منزل ہے، اور فتوحاتِ مکیہ میں منازل کے لیے ایک پوری فصل باندھی گئی ہے۔ سالک خود کسی منزل میں قدم نہیں رکھتا، اسے اتارا جاتا ہے؛ اسی لیے دعا "اَنْزِلْنِيْ" ہے، یہ نہیں کہ میں اتروں۔ اور عربی میں مُنْزِل مہمان کو ٹھہرانے والے میزبان کو بھی کہتے ہیں؛ بندہ مہمان ہے اور "خَيْرُ الْمُنْزِلِيْنَ" بہترین میزبان۔ برکت کے معنی ٹھہراؤ اور بڑھوتری دونوں ہیں، یعنی وہ جگہ جہاں خیر رکے بھی اور پھلے بھی۔ شیخِ اکبر کے نزدیک طوفان کے بعد کی زمین تجددِ امثال کی ایک کھلی تصویر ہے: پرانی صورت ڈوب گئی اور نئی صورت ابھر آئی؛ اور حقیقت میں یہ ہر سانس کا معاملہ ہے کہ عالم ہر لمحہ خلقِ جدید میں ہے، اس لیے یہ دعا ہر سانس کی دعا ہے۔ پھر "لَمُبْتَلِيْنَ" فرمایا: آزمائش اس لیے نہیں کہ حق کو کچھ معلوم ہو، بلکہ اس لیے کہ بندے کی استعداد میں جو چھپا ہے وہ ظاہر ہو اور بندہ خود اپنے آپ کو پہچان لے۔ آزمائش نتیجہ پیدا نہیں کرتی، اسے سامنے لاتی ہے۔ ظاہری قصہ اپنی جگہ قطعی تاریخ ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

ثُمَّ اَنْشَاْنَا مِنْۢ بَعْدِهِمْ قَرْنًا اٰخَرِيْنَ۔ پھر ان کے بعد ہم نے ایک اور نسل اٹھائی۔ فَاَرْسَلْنَا فِيْهِمْ رَسُوْلًا مِّنْهُمْ اَنِ اعْبُدُوا اللّٰهَ مَا لَكُمْ مِّنْ اِلٰهٍ غَيْرُهٗ ۭ اَفَلَا تَتَّقُوْنَ۔ تو ہم نے انہی میں سے ایک رسول ان میں بھیجا کہ اللہ کی عبادت کرو، اس کے سوا تمہارا کوئی معبود نہیں؛ تو کیا تم ڈرتے نہیں؟ اکثر مفسرین کے نزدیک یہ قومِ عاد ہے جن کی طرف حضرت ہود علیہ السلام بھیجے گئے، اور بعض نے اسے قومِ ثمود اور حضرت صالح علیہ السلام کا قصہ کہا ہے؛ نام نہ لینے میں یہ حکمت ہے کہ نقشہ ہر زمانے پر چسپاں ہو جائے۔ "اَنْشَاْنَا" وہی فعل ہے جو اسی سورت میں انسان کی تخلیق کے آخری مرحلے پر آیا تھا: ثُمَّ اَنْشَاْنٰهُ خَلْقًا اٰخَرَ۔ اہلِ معرفت کہتے ہیں کہ فرد کی پیدائش اور قوموں کا اٹھنا ایک ہی انشاء کے دو دائرے ہیں؛ تاریخ بھی خلقِ جدید ہے جس میں ہر نسل ایک نئی صورت ہے اور اس کے پیچھے وہی ایک نفَسِ رحمانی۔ پھر دیکھیے کہ اس رسول کی دعوت حرف بہ حرف وہی ہے جو حضرت نوح علیہ السلام کی تھی؛ زمانہ بدلا، زبان بدلی، کلمہ نہ بدلا، کیونکہ توحید ایک ہے اور انسان کی بھول بار بار لوٹتی ہے، اسی لیے یاد دہانی بھی بار بار آتی ہے۔ "رَسُوْلًا مِّنْهُمْ" کے ساتھ "فِيْهِمْ" فرمایا، یعنی ان کے بیچ، صرف ان کی طرف نہیں؛ اہلِ ذوق کہتے ہیں کہ رسول ان کے بیچ ان کا اپنا آئینہ بن کر کھڑا تھا، ان کی بولی بولتا، ان کا کھانا کھاتا، تاکہ کسی کے پاس یہ عذر نہ رہے کہ پیغام اجنبی تھا۔ یوں ہدایت ہر طالب کے دروازے تک پہنچا دی گئی؛ اب قبول کرنا یا منہ موڑنا ان کا اپنا رخ ہے۔
