وَقَالَ الْمَلَاُ مِنْ قَوْمِهِ الَّذِيْنَ كَفَرُوْا وَكَذَّبُوْا بِلِقَآءِ الْاٰخِرَةِ وَاَتْرَفْنٰهُمْ فِي الْحَيَاةِ الدُّنْيَا۔ اور اس کی قوم کے ان سرداروں نے، جنہوں نے کفر کیا اور آخرت کی ملاقات کو جھٹلایا اور جنہیں ہم نے دنیا کی زندگی میں آسودگی دے رکھی تھی، کہا: مَا هٰذَآ اِلَّا بَشَرٌ مِّثْلُكُمْ ۙ يَاْكُلُ مِمَّا تَاْكُلُوْنَ مِنْهُ وَيَشْرَبُ مِمَّا تَشْرَبُوْنَ۔ یہ تو بس تم ہی جیسا ایک بشر ہے، وہی کھاتا ہے جو تم کھاتے ہو اور وہی پیتا ہے جو تم پیتے ہو۔ وَلَئِنْ اَطَعْتُمْ بَشَرًا مِّثْلَكُمْ اِنَّكُمْ اِذًا لَّخٰسِرُوْنَ۔ اور اگر تم نے اپنے جیسے ایک بشر کی اطاعت کر لی تو تم یقیناً گھاٹے میں رہو گے۔ آیت نے سرداروں کی تین صفتیں ایک سانس میں گنوائیں: کفر، آخرت کی ملاقات کا انکار، اور آسودگی؛ گویا تیسری نے پہلی دو کو پالا۔ "اَتْرَفْنٰهُمْ" کی نسبت حق کی طرف ہے، کیونکہ نعمت اسی کی عطا تھی؛ مگر نعمت کو پردہ بنا لینا ان کا اپنا کام تھا۔ اہلِ معرفت کے نزدیک ایک ہی عطا کسی کے لیے شکر کا زینہ بنتی ہے اور کسی کے لیے غفلت کا لحاف؛ فرق عطا میں نہیں، لینے والے کے رخ میں ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کھانا پینا بندگی اور فقر کی علامت ہے: جو محتاج ہے وہ کھاتا ہے، اور جو بے نیاز ہے وہ نہ کھاتا ہے نہ پیتا ہے۔ تو جس بات کو منکروں نے رسول کا عیب سمجھا وہی اس کی سچائی کی گواہی تھی کہ وہ اپنے لیے الوہیت کا دعویٰ نہیں کرتا، اپنی عبدیت کو کھلے بندوں ظاہر کرتا ہے۔ اور "لَّخٰسِرُوْنَ" کا الٹا ترازو دیکھیے: انہیں ایک بشر کی اطاعت میں گھاٹا نظر آیا، اپنے نفس اور اپنی خواہش کی اطاعت میں نہیں؛ حالانکہ اطاعت سے کوئی خالی نہیں، سوال صرف یہ ہے کہ کس کی۔

اَيَعِدُكُمْ اَنَّكُمْ اِذَا مِتُّمْ وَكُنْتُمْ تُرَابًا وَّعِظَامًا اَنَّكُمْ مُّخْرَجُوْنَ۔ کیا یہ تم سے وعدہ کرتا ہے کہ جب تم مر جاؤ گے اور مٹی اور ہڈیاں ہو جاؤ گے تو تم نکالے جاؤ گے؟ هَيْهَاتَ هَيْهَاتَ لِمَا تُوْعَدُوْنَ۔ بہت دور، بہت دور ہے وہ بات جس کا تم سے وعدہ کیا جاتا ہے۔ اِنْ هِيَ اِلَّا حَيَاتُنَا الدُّنْيَا نَمُوْتُ وَنَحْيَا وَمَا نَحْنُ بِمَبْعُوْثِيْنَ۔ زندگی تو بس یہی ہماری دنیا کی زندگی ہے، ہم مرتے ہیں اور جیتے ہیں، اور ہم دوبارہ اٹھائے جانے والے نہیں۔ پہلے رکوع میں مٹی اور ہڈیاں انسان کی پیدائش کی سیڑھیاں تھیں: مٹی کا خلاصہ، پھر نطفہ، پھر ہڈیاں، پھر ان پر گوشت؛ جس نے مٹی سے پہلی بار نکالا اس کے لیے دوسری بار نکالنا کیوں دور ہو؟ "مُّخْرَجُوْنَ" میں زمین ماں کے پیٹ کی طرح ہے جس سے بچہ نکالا جاتا ہے۔ اور "هَيْهَاتَ" کی تکرار بتاتی ہے کہ دوری بات میں نہیں، ان کی نظر میں تھی؛ جس آنکھ نے کبھی غیب کی طرف دیکھا ہی نہ ہو، اسے ہر غیب دور لگتا ہے۔ پھر عجیب بات یہ ہے کہ ان کی زبان پر وہ جملہ آیا جس میں حقیقت چھپی تھی: "نَمُوْتُ وَنَحْيَا"۔ شیخِ اکبر کے نزدیک تجددِ امثال کا مطلب یہی ہے کہ عالم ہر لمحہ فنا ہوتا اور نیا پیدا ہوتا ہے؛ گویا ہم ہر سانس میں مرتے اور جیتے ہیں، اور جو اس پر قادر ہے کہ ہر آن نئی صورت دے، اس کے لیے قیامت کی نئی صورت کوئی اجنبی بات نہیں۔ اہلِ ذوق کہتے ہیں کہ منکر نے اپنے انکار کے لفظوں میں اقرار کی دلیل رکھ دی، مگر سنی نہیں۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں یہ موت کے بعد زندگی کا صاف انکار ہے اور اس کا جواب بھی صاف ہے۔

اِنْ هُوَ اِلَّا رَجُلٌ افْتَرٰى عَلَى اللّٰهِ كَذِبًا وَّمَا نَحْنُ لَهٗ بِمُؤْمِنِيْنَ۔ یہ تو بس ایک آدمی ہے جس نے اللہ پر جھوٹ گھڑا ہے، اور ہم اس پر ایمان لانے والے نہیں۔ قَالَ رَبِّ انْصُرْنِيْ بِمَا كَذَّبُوْنِ۔ رسول نے عرض کیا: اے میرے رب! میری مدد فرما، کیونکہ انہوں نے مجھے جھٹلا دیا ہے۔ پچھلی قوم نے حضرت نوح علیہ السلام کو دیوانہ کہا تھا، اس قوم نے اپنے رسول کو جھوٹا کہا؛ الزام بدلا، انکار نہ بدلا۔ اور "وَّمَا نَحْنُ لَهٗ بِمُؤْمِنِيْنَ" میں فیصلہ ان کا اپنا ہے: انہوں نے دلیل سنے بغیر دروازہ بند کرنے کا اعلان کیا۔ یہی وہ اعراض ہے جس کے بعد جو کچھ ہوا وہ اسی رخ کا پھل تھا۔ اب رسول کی دعا دیکھیے: لفظ بہ لفظ وہی جو حضرت نوح علیہ السلام کی زبان پر تھی۔ اہلِ معرفت کہتے ہیں کہ انبیاء کی دعائیں ایک ہی سرچشمے سے نکلتی ہیں، اس لیے ان کے لفظ بھی ایک ہو جاتے ہیں؛ ہر پیغمبر پچھلے پیغمبر کی وراثت اٹھاتا ہے۔ اگر یہ قومِ عاد ہے تو فصوص میں حضرت ہود علیہ السلام کی حکمت کو حکمتِ احدیہ کہا گیا ہے، جس کا مفہوم یہ ہے کہ ہر جاندار کی پیشانی حق کے ہاتھ میں ہے اور کوئی اس کی گرفت سے باہر نہیں۔ اسی لیے رسول نے اپنی قوت پر نہیں، اسی کی نصرت پر بھروسا کیا جس کے ہاتھ میں سب کی پیشانیاں ہیں۔ اور "بِمَا كَذَّبُوْنِ" پھر سرِّ قدر کی یاد دہانی ہے: انجام ان کے جھٹلانے کے مطابق ہوگا، اس سے زیادہ نہیں۔

قَالَ عَمَّا قَلِيْلٍ لَّيُصْبِحُنَّ نٰدِمِيْنَ۔ فرمایا: تھوڑی ہی دیر میں یہ ضرور پچھتاتے رہ جائیں گے۔ فَاَخَذَتْهُمُ الصَّيْحَةُ بِالْحَقِّ فَجَعَلْنٰهُمْ غُثَآءً ۚ فَبُعْدًا لِّلْقَوْمِ الظّٰلِمِيْنَ۔ تو انہیں حق کے ساتھ ایک چیخ نے آ پکڑا، اور ہم نے انہیں کوڑا کرکٹ بنا دیا؛ سو دوری ہو ظالم قوم کے لیے۔ "عَمَّا قَلِيْلٍ" اس بات کا اعلان ہے کہ جس مہلت کو وہ لمبا سمجھ رہے تھے وہ مختصر تھی۔ ندامت اگر وقت پر ہو تو توبہ کا دروازہ ہے؛ اہلِ سلوک کے نزدیک ندامت توبہ کی جان ہے۔ مگر "لَّيُصْبِحُنَّ نٰدِمِيْنَ" اس ندامت کی خبر ہے جو دروازہ بند ہونے کے بعد آئے۔ پھر "بِالْحَقِّ" فرمایا: وہ چیخ ظلم نہیں تھی، عدل تھی؛ ان کے اپنے کیے کا ٹھیک ٹھیک بدلہ۔ "غُثَآءً" وہ جھاگ اور کوڑا ہے جو سیلاب اپنے اوپر اٹھا لاتا ہے۔ سورۂ رعد میں حق اور باطل کی مثال میں فرمایا گیا تھا: فَاَمَّا الزَّبَدُ فَيَذْهَبُ جُفَآءً۔ اہلِ ذوق کہتے ہیں کہ جو قوم اپنے وجود کی جڑ حق سے کاٹ لے، وہ سیلاب کی سطح پر تیرتا جھاگ بن جاتی ہے؛ اوپر نظر آتی ہے، مگر اس میں نہ وزن ہوتا ہے نہ ٹھہراؤ۔ اور "فَبُعْدًا" اسی دوری کی صورت ہے جسے انہوں نے خود چنا تھا؛ شیخِ اکبر کے نزدیک بُعد کوئی مسافت نہیں، دل کا حق سے منہ موڑ لینا ہے۔ جو قریب آنا چاہے اسے دور نہیں کیا جاتا؛ دوری صرف اس کا حصہ ہے جس نے دوری مانگی۔

ثُمَّ اَنْشَاْنَا مِنْۢ بَعْدِهِمْ قُرُوْنًا اٰخَرِيْنَ۔ پھر ان کے بعد ہم نے اور کئی نسلیں اٹھائیں۔ مَا تَسْبِقُ مِنْ اُمَّةٍ اَجَلَهَا وَمَا يَسْتَاْخِرُوْنَ۔ کوئی قوم اپنے مقررہ وقت سے نہ آگے بڑھ سکتی ہے نہ پیچھے رہ سکتی ہے۔ ثُمَّ اَرْسَلْنَا رُسُلَنَا تَتْرَا ۭ كُلَّمَا جَآءَ اُمَّةً رَّسُوْلُهَا كَذَّبُوْهُ۔ پھر ہم نے پے در پے اپنے رسول بھیجے؛ جب بھی کسی قوم کے پاس اس کا رسول آیا، انہوں نے اسے جھٹلایا؛ فَاَتْبَعْنَا بَعْضَهُمْ بَعْضًا وَّجَعَلْنٰهُمْ اَحَادِيْثَ ۚ فَبُعْدًا لِّقَوْمٍ لَّا يُؤْمِنُوْنَ۔ تو ہم نے ایک کو دوسرے کے پیچھے لگا دیا اور انہیں کہانیاں بنا دیا؛ سو دوری ہو ان لوگوں کے لیے جو ایمان نہیں لاتے۔ پہلے "قَرْنًا" مفرد تھا، اب "قُرُوْنًا" جمع؛ قصہ تفصیل سے اجمال کی طرف آیا، کیونکہ نقشہ سمجھ میں آ چکا۔ شیخِ اکبر کے مکتب میں ہر شے کی ایک ازلی حقیقت ہے جسے عینِ ثابتہ کہتے ہیں، اور اس کا ظہور اپنے مقررہ وقت پر ہوتا ہے؛ "اَجَلَهَا" اسی قدرِ معلوم کا نام ہے، جو کسی کی جلدی سے آگے نہیں آتا اور کسی کی خواہش سے پیچھے نہیں جاتا۔ "تَتْرَا" کا مطلب ہے ایک کے بعد ایک، بیچ میں وقفہ رکھ کر؛ رحمت نے بلانا نہیں چھوڑا، ہر نسل کو اس کا اپنا رسول ملا۔ پھر وہ ترتیب دیکھیے جو اصول کو کھول دیتی ہے: پہلے "كَذَّبُوْهُ"، پھر "فَاَتْبَعْنَا"؛ انہوں نے انکار میں ایک دوسرے کی پیروی کی تو انجام میں بھی ایک دوسرے کے پیچھے لگا دیے گئے۔ یہ حکم محکوم کے مطابق ہے: جو راہ انہوں نے خود چنی، حق نے اسی کو وجود دیا۔ اور "اَحَادِيْثَ" سب سے کڑوا انجام ہے: جو قوم حق سے نسبت نہ رکھے، اس کی ساری شان و شوکت آخر میں ایک کہانی رہ جاتی ہے جو لوگ سناتے ہیں اور آگے بڑھ جاتے ہیں۔ اہلِ معرفت کہتے ہیں کہ باقی وہی رہتا ہے جو باقی سے جڑا ہو۔

ثُمَّ اَرْسَلْنَا مُوْسٰى وَاَخَاهُ هٰرُوْنَ ۙ بِاٰيٰتِنَا وَسُلْطٰنٍ مُّبِيْنٍ۔ پھر ہم نے موسیٰ اور ان کے بھائی ہارون کو اپنی نشانیوں اور کھلی دلیل کے ساتھ بھیجا۔ اِلٰى فِرْعَوْنَ وَمَلَا۟ئِهٖ فَاسْتَكْبَرُوْا وَكَانُوْا قَوْمًا عَالِيْنَ۔ فرعون اور اس کے سرداروں کی طرف، تو انہوں نے تکبر کیا اور وہ سرکش اور اونچا بننے والے لوگ تھے۔ فَقَالُوْۤا اَنُؤْمِنُ لِبَشَرَيْنِ مِثْلِنَا وَقَوْمُهُمَا لَنَا عٰبِدُوْنَ۔ تو کہنے لگے: کیا ہم اپنے ہی جیسے دو انسانوں پر ایمان لے آئیں، جبکہ ان کی قوم ہماری غلام ہے؟ نام لیے بغیر قوموں کا ذکر ہوا تو اب ایک نام آیا جس کا قصہ پڑھنے والا خوب جانتا ہے۔ "بِاٰيٰتِنَا" اور "سُلْطٰنٍ مُّبِيْنٍ" کو الگ ذکر کیا گیا: نشانیاں آنکھ کے لیے تھیں، اور کھلی دلیل دل کے لیے؛ اہلِ تاویل کہتے ہیں کہ سلطانِ مبین وہ غلبہ ہے جو حق کے سامنے دل کو بے دلیل نہیں چھوڑتا۔ اور حضرت ہارون علیہ السلام کو ساتھ رکھا گیا؛ فصوص میں ان کی حکمت حکمتِ امامیہ ہے، جس کا مفہوم یہ ہے کہ ان کا وجود حق کی رحمت کی بارگاہ سے تھا، یعنی وہ حضرت موسیٰ علیہ السلام کو رحمت کے طور پر عطا ہوئے۔ جلال کی دعوت کے ساتھ جمال کا رفیق، یہی کامل جوڑ ہے۔ پھر "عَالِيْنَ" دیکھیے: شیخِ اکبر کے مکتب میں حقیقی علو صرف العلی کا ہے، اور بندے کی بلندی اس کی عبدیت میں ہے؛ جو بندہ اپنے لیے ذاتی بلندی کا دعویٰ کرے وہ ایک ایسی صفت چھیننا چاہتا ہے جو اس کی نہیں۔ اور "عٰبِدُوْنَ" کا لفظ ان کی الٹی سمجھ کو کھول دیتا ہے: وہ انسانوں کی غلامی کو اپنی عزت سمجھتے تھے، حالانکہ عبادت کا حق دار صرف ایک ہے۔ انہوں نے بھی پچھلی قوموں کی طرح "مِثْلِنَا" کی صورت دیکھی، اس کے پیچھے کی حقیقت نہ دیکھی۔

فَكَذَّبُوْهُمَا فَكَانُوْا مِنَ الْمُهْلَكِيْنَ۔ تو انہوں نے دونوں کو جھٹلایا، اور ہلاک کیے جانے والوں میں سے ہو گئے۔ وَلَقَدْ اٰتَيْنَا مُوْسَى الْكِتٰبَ لَعَلَّهُمْ يَهْتَدُوْنَ۔ اور ہم نے موسیٰ کو کتاب دی تاکہ وہ لوگ ہدایت پائیں۔ وَجَعَلْنَا ابْنَ مَرْيَمَ وَاُمَّهٗۤ اٰيَةً وَّاٰوَيْنٰهُمَاۤ اِلٰى رَبْوَةٍ ذَاتِ قَرَارٍ وَّمَعِيْنٍ۔ اور ہم نے ابنِ مریم اور ان کی والدہ کو ایک نشانی بنایا، اور ان دونوں کو ایک بلند جگہ پر ٹھکانا دیا جو ٹھہرنے کے لائق تھی اور جہاں چشمہ بہتا تھا۔ پھر وہی ترتیب: پہلے "فَكَذَّبُوْهُمَا"، پھر ہلاکت؛ فعل ان کا، انجام اسی فعل کی صورت۔ اور فوراً بعد "لَعَلَّهُمْ يَهْتَدُوْنَ" آیا: ڈوبنے والے ڈوب گئے، مگر کتاب کا دروازہ باقیوں کے لیے کھلا رہا۔ یہی اصول ہے کہ حق کسی طالب کو گمراہ نہیں کرتا؛ ہدایت ہر اس شخص کے لیے رکھی گئی ہے جو اس کی طرف رخ کرے۔ آخری آیت میں "اٰيَةً" مفرد ہے، حالانکہ ذکر دو کا ہے: بیٹا باپ کے بغیر اور ماں شوہر کے بغیر، دونوں مل کر ایک ہی معجزہ ہیں؛ حضرت مریم علیہا السلام کی پاکیزگی اور حضرت عیسیٰ علیہ السلام کی ولادت ایک دوسرے سے جدا نہیں۔ فصوص میں حضرت عیسیٰ علیہ السلام کی حکمت حکمتِ نبویہ ہے، جس کا مفہوم یہ ہے کہ ان کا وجود روح کی پھونک سے بشری صورت میں ہوا، اسی لیے ان کے دم میں زندہ کرنے کی تاثیر رکھی گئی۔ سورۂ مریم میں اسی وادی کی جھلک تھی: قَدْ جَعَلَ رَبُّكِ تَحْتَكِ سَرِيًّا۔ اور "رَبْوَةٍ" دیکھیے: یہ سلسلہ پچھلے رکوع میں طوفان سے شروع ہوا تھا اور انتہا ایک ایسی بلندی پر ہے جہاں سیلاب نہیں پہنچتا۔ سورۂ ابراہیم میں کلمۂ خبیثہ کا درخت ایسا تھا: اجْتُثَّتْ مِنْ فَوْقِ الْاَرْضِ مَا لَهَا مِنْ قَرَارٍ؛ یہاں "ذَاتِ قَرَارٍ" ہے۔ اہلِ معرفت کے نزدیک یہ تمکین کا مقام ہے، جہاں دل احوال کی موجوں کے بیچ ٹھہر جاتا ہے؛ اور "مَعِيْنٍ" وہ زندہ چشمہ ہے جو اسی ٹھہراؤ سے پھوٹتا ہے۔ جسے لوگوں نے ستایا، اسے حق نے "وَّاٰوَيْنٰهُمَاۤ" کی پناہ دی؛ ہر ستائے ہوئے طالب کے لیے دل کی ایک ربوہ ہے جہاں قرار بھی ہے اور چشمہ بھی۔
