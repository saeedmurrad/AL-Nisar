انبیائے کرام کے سلسلے کی آخری کڑی یوں شروع ہوتی ہے: وَاِنَّ يُوْنُسَ لَمِنَ الْمُرْسَلِيْنَ اِذْ اَبَقَ اِلَى الْفُلْكِ الْمَشْحُوْنِ۔ اور بے شک یونس بھی رسولوں میں سے تھے، جب وہ نکل کر بھری ہوئی کشتی کی طرف جا پہنچے۔ ترتیب خود ادب سکھاتی ہے: واقعہ بیان کرنے سے پہلے حق تعالیٰ نے حضرت یونس علیہ السلام کی رسالت کی گواہی دی، تاکہ پڑھنے والا جو کچھ آگے سنے اس منصب کی روشنی میں سنے۔ "اَبَقَ" عربی میں اس غلام کے لیے آتا ہے جو آقا کی اجازت کے بغیر چلا جائے۔ واقعہ یہ تھا کہ حضرت یونس علیہ السلام نے قوم کو عذاب سے ڈرایا، قوم نہ مانی، اور آپ اللہ ہی کی خاطر ان پر ناراض ہو کر صریح اذنِ الٰہی آنے سے پہلے بستی سے نکل گئے۔ امت کا اجماع ہے کہ انبیاء معصوم ہیں؛ یہ گناہ نہ تھا بلکہ ترکِ اولیٰ تھا، یعنی افضل یہ تھا کہ حکم کا انتظار کیا جاتا۔ یہ سخت لفظ کیوں آیا؟ اہلِ معرفت کہتے ہیں کہ قرب جتنا بڑھتا ہے، پیمانہ اتنا باریک ہو جاتا ہے؛ اہلِ تصوف میں مشہور قول ہے کہ نیکوں کی نیکیاں مقربین کے نزدیک کوتاہیاں شمار ہوتی ہیں۔ اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس لفظ کے اندر ایک اعزاز بھی چھپا ہے: "اَبَقَ" صرف غلام کے لیے بولا جاتا ہے، گویا حق نے اس لفظ ہی سے گواہی دے دی کہ یونس اس کا اپنا بندہ ہے، اور بندگی ہی سب سے بڑا لقب ہے۔ یاد رہے کہ یہاں جو اشارہ بیان ہو رہا ہے وہ ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر کشتی میں وہ گھڑی آئی جب بوجھ ہلکا کیے بغیر سب کے ڈوبنے کا اندیشہ تھا: فَسَاهَمَ فَكَانَ مِنَ الْمُدْحَضِيْنَ فَالْتَقَمَهُ الْحُوْتُ وَهُوَ مُلِيْمٌ۔ تو قرعہ ڈالا گیا اور وہ ہارنے والوں میں نکلے، پھر مچھلی نے انہیں نگل لیا اور وہ اپنے آپ کو ملامت کر رہے تھے۔ "مُلِيْمٌ" کے بارے میں ادب یہ ہے کہ اس سے مراد وہ ہے جس سے اپنے بلند مرتبے کے اعتبار سے ایسا کام ہوا جس پر نرم سی گرفت ہو سکتی تھی، اور بہت سے اہلِ معرفت نے اسے یوں پڑھا کہ وہ خود اپنے آپ کو ملامت کرنے والے تھے۔ اپنے نفس پر یہ نگرانی کوئی عیب نہیں، مقربین کا خاص مقام ہے۔ شیخِ اکبر کے نزدیک یہاں ایک لطیف بات ہے: جو حکم پہلے صریح اذن کی صورت میں نہ آیا تھا، وہ اب قرعے کی زبان سے آیا۔ قرعہ بظاہر اتفاق ہے، مگر اتفاق نام کی کوئی چیز نہیں؛ قرعہ تقدیر کا آئینہ ہے۔ حضرت یونس علیہ السلام نے بلا تامل اس فیصلے کو قبول کیا، اور یہی کامل تسلیم ترکِ اولیٰ کا تدارک بن گئی۔ پھر لفظ "الْتَقَمَ" پر غور کیجیے: مچھلی نے لقمے کی طرح نگلا، چبایا نہیں۔ اہلِ ذوق کہتے ہیں کہ پیٹ قبر نہ بنا، خلوت خانہ بنا؛ نبی کا قید خانہ بھی حق کی بارگاہ میں حضوری کا حجرہ ہوتا ہے۔

اب وہ آیت آتی ہے جس نے نجات کا راز کھول دیا: فَلَوْلَاۤ اَنَّهٗ كَانَ مِنَ الْمُسَبِّحِيْنَ لَلَبِثَ فِيْ بَطْنِهٖۤ اِلٰى يَوْمِ يُبْعَثُوْنَ۔ پس اگر وہ تسبیح کرنے والوں میں سے نہ ہوتے تو اس کے پیٹ میں اس دن تک رہتے جب لوگ اٹھائے جائیں گے۔ "كَانَ" ماضی کا صیغہ ہے: وہ پہلے سے تسبیح کرنے والوں میں تھے۔ آسانی کے دنوں کی تسبیح نے تنگی کے دن دستگیری کی؛ حدیث کا مفہوم بھی یہی ہے کہ خوشحالی میں اللہ کو پہچانو، وہ سختی میں تمہیں پہچانے گا۔ اور پیٹ کے اندر جو تسبیح ادا ہوئی وہ سورۂ الانبیاء میں محفوظ ہے، جسے ہم ذکرِ یونسی کے نام سے پڑھ چکے: لَاۤ اِلٰهَ اِلَّاۤ اَنْتَ سُبْحٰنَكَ اِنِّيْ كُنْتُ مِنَ الظّٰلِمِيْنَ۔ اس میں توحید ہے، تنزیہ ہے اور اعتراف ہے؛ اور "ظالمین" یہاں ایک نبی کی انکساری ہے جو معمولی کمی کو بھی اپنی طرف منسوب کرتا ہے، گناہ کا اقرار نہیں۔ اہلِ معرفت کہتے ہیں کہ رات، سمندر اور مچھلی کے تین اندھیروں کی تہہ سے حق کو ہر تاریکی سے پاک قرار دینا تنزیہ کی معراج ہے۔ اور حکمتِ نفسیہ جو فصوص میں حضرت یونس علیہ السلام کا باب ہے، اس کا مفہوم یہ ہے کہ انسانی نشأت حق کی بنائی ہوئی ہے اور اس کی حفاظت اس کے نزدیک بڑی قدر رکھتی ہے؛ مچھلی کے پیٹ میں جسمِ نبی کا سلامت رہنا اسی عنایت کی نشانی ہے۔

پھر رحمت کا دوسرا مرحلہ: فَنَبَذْنٰهُ بِالْعَرَآءِ وَهُوَ سَقِيْمٌ وَاَنْۢبَتْنَا عَلَيْهِ شَجَرَةً مِّنْ يَّقْطِيْنٍ۔ پھر ہم نے انہیں ایک چٹیل میدان میں ڈال دیا اور وہ نڈھال تھے، اور ہم نے ان پر کدو کی ایک بیل اگا دی۔ پہلے فعل حضرت یونس علیہ السلام کی طرف منسوب تھا: وہ خود گئے تھے؛ اب فعل حق کی طرف ہے: "ہم نے ڈالا"، "ہم نے اگایا"۔ اہلِ ذوق کہتے ہیں کہ یہی سالک کا سفر ہے: پہلے وہ اپنے قدم سے چلتا ہے، پھر اسے اٹھا کر لایا جاتا ہے، اور جو لایا جائے وہ پہنچا ہوا ہوتا ہے۔ "الْعَرَآءِ" وہ زمین ہے جس پر نہ درخت ہو نہ سایہ؛ اس مکتب میں یہ تجرید کی تمثیل ہے، جہاں سارے سہارے اتر جائیں اور صرف حق کی دستگیری رہ جائے۔ پھر "عَلَيْهِ" کا لفظ: بیل ان کے اوپر اگائی گئی، جیسے رحمت نے ڈھانپ لیا ہو۔ مفسرین لکھتے ہیں کہ کدو کے چوڑے پتے سایہ دیتے ہیں اور مکھی اس پر نہیں بیٹھتی؛ نڈھال جسم کے لیے یہی موزوں تھا۔ اور روایات میں آتا ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم کو کدو مرغوب تھا۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا منظر ہے: کمزوری ابھی دور نہ ہوئی تھی کہ سایہ پہلے پہنچ گیا۔

اور اب منصب کی بحالی کا اعلان: وَاَرْسَلْنٰهُ اِلٰى مِائَةِ اَلْفٍ اَوْ يَزِيْدُوْنَ فَاٰمَنُوْا فَمَتَّعْنٰهُمْ اِلٰى حِيْنٍ۔ اور ہم نے انہیں ایک لاکھ یا اس سے زیادہ لوگوں کی طرف بھیجا، تو وہ ایمان لے آئے، پھر ہم نے انہیں ایک مدت تک فائدہ اٹھانے دیا۔ یہ جملہ خود بتا رہا ہے کہ جو کچھ پیش آیا وہ گناہ نہ تھا؛ رسالت لوٹائی گئی اور ایسی کامیابی کے ساتھ لوٹائی گئی کہ پوری قوم ایمان لے آئی۔ "اَوْ يَزِيْدُوْنَ" یعنی دیکھنے والا ایک لاکھ کہے گا، مگر حق کے علم میں گنتی اس سے آگے ہے۔ قومِ یونس کی خصوصیت سورۂ یونس میں یوں بیان ہوئی: فَلَوْلَا كَانَتْ قَرْيَةٌ اٰمَنَتْ فَنَفَعَهَاۤ اِيْمَانُهَاۤ اِلَّا قَوْمَ يُوْنُسَ؛ وہ واحد قوم جس نے عذاب کے آثار دیکھ کر سچے دل سے رجوع کیا اور اس کا ایمان قبول ہوا۔ یہاں ہدایت کا اصول روشن ہے: جو رخ پھیر کر حق کی طرف آئے، اسے واپس نہیں لوٹایا جاتا؛ دروازہ آخری لمحے تک کھلا ہے، اور رحمت بندے کے رجوع کی منتظر رہتی ہے۔ اہلِ معرفت کہتے ہیں کہ "اِلٰى حِيْنٍ" دنیا کی ہر نعمت پر لکھا ہوا ہے: یہ مدت والی ہے، اور جو باقی ہے وہ اللہ کے پاس ہے۔

انبیاء کے قصوں کے بعد سورت مشرکینِ مکہ کی طرف پلٹتی ہے: فَاسْتَفْتِهِمْ اَلِرَبِّكَ الْبَنَاتُ وَلَهُمُ الْبَنُوْنَ اَمْ خَلَقْنَا الْمَلٰۤئِكَةَ اِنَاثًا وَّهُمْ شٰهِدُوْنَ۔ ان سے پوچھیے: کیا آپ کے رب کے لیے بیٹیاں ہیں اور ان کے لیے بیٹے؟ یا ہم نے فرشتوں کو عورتیں بنایا اور یہ وہاں موجود تھے؟ واضح رہے کہ یہاں بیٹی کی تحقیر ہرگز مقصود نہیں؛ بیٹی اللہ کی رحمت ہے اور حضورِ اکرم صلی اللہ علیہ وسلم نے بیٹیوں کی پرورش پر جنت کی بشارت دی۔ دلیل ان کے اپنے پیمانے سے دی جا رہی ہے: جو چیز وہ خود اپنے لیے پسند نہ کرتے تھے، وہی رب کی طرف منسوب کرتے تھے؛ تضاد ان کا ہے، اور اصل خرابی یہ کہ اولاد کی نسبت ہی حق کی شان کے خلاف ہے۔ "وَّهُمْ شٰهِدُوْنَ" پر اہلِ معرفت ٹھہرتے ہیں: عالمِ غیب کی بات یا تو وحی سے کہی جاتی ہے یا اس کشف سے جو حق عطا کرے۔ شیخِ اکبر کے نزدیک ملکوت کے بارے میں ظن سے حکم لگانا اندھے کا رنگوں کی تعریف کرنا ہے؛ جس نے دیکھا نہیں اور جسے بتایا نہیں گیا، اس کی بات افترا ہے۔

پھر اس دعوے کی جڑ کھولی گئی: اَلَاۤ اِنَّهُمْ مِّنْ اِفْكِهِمْ لَيَقُوْلُوْنَ وَلَدَ اللّٰهُ وَاِنَّهُمْ لَكٰذِبُوْنَ اَصْطَفَى الْبَنَاتِ عَلَى الْبَنِيْنَ۔ سن لو! یہ اپنی من گھڑت سے کہتے ہیں کہ اللہ نے اولاد بنائی، اور یقیناً یہ جھوٹے ہیں۔ کیا اس نے بیٹوں پر بیٹیوں کو ترجیح دی؟ "اِفْك" اس بات کو کہتے ہیں جو اپنی اصل سے الٹ دی گئی ہو۔ اہلِ معرفت کہتے ہیں کہ ولادت کا تصور تین باتیں چاہتا ہے: ایک جز کا کل سے جدا ہونا، باپ اور بیٹے کا ہم جنس ہونا، اور جانشین کی حاجت۔ حق سبحانہ ان تینوں سے پاک ہے: لَمْ يَلِدْ وَلَمْ يُوْلَدْ۔ یہاں ایک نازک فرق سمجھنا ضروری ہے۔ شیخِ اکبر کے مکتب میں وحدتِ وجود کا مطلب یہ نہیں کہ عالم حق کے ٹکڑے ہیں؛ عالم اس کے اسماء کا مظہر ہے، جیسے آئینے میں صورت ہو۔ تجلی میں کچھ جدا نہیں ہوتا اور کچھ گھٹتا نہیں، جبکہ ولادت میں جدائی اور تقسیم ہے۔ اسی لیے یہی مکتب ولادت کی نسبت کو سب سے زیادہ سختی سے رد کرتا ہے: ظہور ایک بات ہے اور تولّد بالکل دوسری۔

اس کے بعد چار مختصر سوال، جیسے عقل کے چار دروازے ایک ایک کر کے کھٹکھٹائے جا رہے ہوں: مَا لَكُمْ كَيْفَ تَحْكُمُوْنَ اَفَلَا تَذَكَّرُوْنَ اَمْ لَكُمْ سُلْطٰنٌ مُّبِيْنٌ فَاْتُوْا بِكِتٰبِكُمْ اِنْ كُنْتُمْ صٰدِقِيْنَ۔ تمہیں کیا ہو گیا، تم کیسا فیصلہ کرتے ہو؟ کیا تم غور نہیں کرتے؟ یا تمہارے پاس کوئی کھلی دلیل ہے؟ تو اپنی کتاب لے آؤ اگر تم سچے ہو۔ پہلا سوال فیصلے کی صحت کا ہے، دوسرا یاد دہانی کا، تیسرا دلیل کا اور چوتھا وحی کا۔ گویا حق کے بارے میں کوئی بات کہنے کے دو ہی راستے ہیں: سلیم عقل جو حد میں رہے، یا کتاب جو خود حق کی طرف سے آئے۔ شیخِ اکبر کے نزدیک ذاتِ حق کے بارے میں عقل زیادہ سے زیادہ یہ جان سکتی ہے کہ وہ کیا نہیں ہے؛ وہ کیا ہے، یہ وہی بتا سکتا ہے۔ اس لیے "فَاْتُوْا بِكِتٰبِكُمْ" محض الزامی مطالبہ نہیں، ایک اصول کی طرف اشارہ ہے: اللہ کی صفت وہی معتبر ہے جو اس نے خود بیان کی۔ یہی بات آگے "الْمُخْلَصِيْنَ" کے استثنا میں کھلنے والی ہے۔

پھر ایک اور گمان کا ذکر: وَجَعَلُوْا بَيْنَهٗ وَبَيْنَ الْجِنَّةِ نَسَبًا وَلَقَدْ عَلِمَتِ الْجِنَّةُ اِنَّهُمْ لَمُحْضَرُوْنَ۔ اور انہوں نے اس کے اور جنوں کے درمیان رشتہ داری ٹھہرائی، حالانکہ جن خوب جانتے ہیں کہ وہ خود حاضر کیے جائیں گے۔ مفسرین لکھتے ہیں کہ بعض مشرکین نے فرشتوں کو بیٹیاں اور جنوں کے سرداروں کو ان کی مائیں کہا۔ قرآن کا جواب حیرت انگیز ہے: جنہیں تم رشتہ دار بناتے ہو وہ خود جانتے ہیں کہ وہ پیشی کے لیے حاضر کیے جانے والے بندے ہیں۔ جو جواب دہ ہو وہ رشتہ دار کیسے ہوا؟ شیخِ اکبر کے مکتب میں خالق اور مخلوق کے درمیان ایک ہی نسبت ہے: ربوبیت اور مربوبیت کی، یعنی ایک طرف عطا اور دوسری طرف احتیاج۔ اہلِ معرفت کہتے ہیں کہ بندے کا سب سے سچا "نسب" اس کا فقر ہے۔ جو اپنے فقر کو پہچان لے وہ حق سے قریب ہوا، اور جو حق سے خونی رشتہ جوڑنا چاہے وہ اپنے فقر کا انکار کر کے دور ہو گیا۔

اب تنزیہ کا اعلان اور ایک استثنا: سُبْحٰنَ اللّٰهِ عَمَّا يَصِفُوْنَ اِلَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ۔ اللہ پاک ہے ان باتوں سے جو یہ بیان کرتے ہیں، سوائے اللہ کے چنے ہوئے خالص بندوں کے۔ مفسرین نے استثنا کے دو مفہوم لکھے ہیں: یا یہ کہ مخلَص بندے پیشی کے عذاب سے مستثنیٰ ہیں، یا یہ کہ وہ ان بیان کرنے والوں سے الگ ہیں جن کی توصیف سے اللہ پاک ہے۔ دوسرا مفہوم اہلِ معرفت کے نزدیک بڑا گہرا ہے: مخلَص بندے جو کچھ اللہ کے بارے میں کہتے ہیں وہ درست ہے، کیونکہ وہ اپنی طرف سے نہیں کہتے؛ وہ وہی کہتے ہیں جو اس نے اپنی کتاب اور اپنے رسولوں کی زبان سے اپنے بارے میں فرمایا۔ شیخِ اکبر کے نزدیک ان کی توصیف دراصل حق کی اپنی توصیف ہے جو ان کی زبان پر جاری ہوئی۔ اور "مُخْلَص" اسمِ مفعول ہے: وہ خود خالص نہیں بنے، انہیں خالص بنایا گیا۔ یہ فقر کا مقام ہے، جہاں بندہ اپنے اخلاص کو بھی اپنی کمائی نہیں سمجھتا۔ اسی مکتب میں یہ بھی کہا گیا ہے کہ وہ تنزیہ بھی جو صرف عقل سے گھڑی جائے ایک طرح کی حد بندی ہے؛ کامل تنزیہ وہ ہے جو حق کی اپنی تعلیم کے تابع ہو۔

اور یہاں سورت خود ہدایت و گمراہی کا اصول بیان کرتی ہے، اس لیے اسے اصول کی حیثیت سے پڑھنا چاہیے: فَاِنَّكُمْ وَمَا تَعْبُدُوْنَ مَاۤ اَنْتُمْ عَلَيْهِ بِفٰتِنِيْنَ اِلَّا مَنْ هُوَ صَالِ الْجَحِيْمِ۔ پس تم اور جن کی تم پوجا کرتے ہو، اس کے خلاف کسی کو بہکا نہیں سکتے، سوائے اس کے جو جہنم میں جا پڑنے والا ہے۔ یعنی گمراہ کرنے والوں کا زور صرف اسی پر چلتا ہے جس کا رخ پہلے ہی آگ کی طرف ہو چکا ہو۔ کوئی طاقت کسی طالبِ ہدایت کو زبردستی نہیں بہکا سکتی۔ یہی بات سورۂ ابراہیم میں شیطان کی زبان سے ادا ہوئی: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ۔ "صَالِ" اسمِ فاعل ہے: وہ جو خود آگ میں داخل ہونے اور اس کی تپش سہنے والا ہے؛ راستہ اس کا اپنا چنا ہوا ہے۔ اللہ کسی کو زبردستی گمراہ نہیں کرتا اور کسی ہدایت کے طلبگار کو گمراہ نہیں کرتا؛ جو اعراض کرے، اس کا اعراض ہی اس کی گمراہی ہے، اور حق اسی سمت کو وجود دیتا ہے جو بندے نے خود چنی۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے اپنے حال کے مطابق۔

اور اب اس رکوع کی، بلکہ پوری سورت کی کلید: وَمَا مِنَّاۤ اِلَّا لَهٗ مَقَامٌ مَّعْلُوْمٌ۔ اور ہم میں سے ہر ایک کا ایک مقرر مقام ہے۔ جمہور مفسرین کے نزدیک یہ فرشتوں کا کلام ہے جو مشرکین کے دعوے کے جواب میں اپنی بندگی کا اقرار کر رہے ہیں: ہم اللہ کی بیٹیاں نہیں، اس کے بندے ہیں، اور ہر ایک اپنی جگہ کھڑا ہے؛ کوئی سجدے میں، کوئی رکوع میں، کوئی قیام میں، اور کوئی اپنی حد سے آگے نہیں بڑھتا۔ اہلِ تصوف میں ایک روایت مشہور ہے کہ معراج کی رات حضرت جبرئیل علیہ السلام سدرۃ المنتہیٰ پر رک گئے اور عرض کیا کہ اگر میں ایک قدم بھی آگے بڑھوں تو جل جاؤں، اور حضورِ اکرم صلی اللہ علیہ وسلم آگے تشریف لے گئے۔ یہ اپنی حد کو پہچاننے کا ادب ہے، اور اہلِ معرفت کہتے ہیں کہ اپنے مقام کو جاننا عبدیت کا پہلا سبق ہے۔ جو اپنا مقام جانتا ہے وہ نہ اس سے نیچے گر کر ناشکری کرتا ہے نہ اس سے اوپر اچھل کر دعویٰ کرتا ہے۔ سورت کا آغاز صف بستہ فرشتوں کی قسم سے ہوا تھا؛ اب وہی صفیں خود بول رہی ہیں اور بتا رہی ہیں کہ صف کی خوبی یہی ہے کہ ہر کوئی اپنی جگہ پر ہو۔

شیخِ اکبر کے نزدیک یہ آیت صرف فرشتوں کا حال نہیں، پورے وجود کا قانون ہے۔ ہر شے کا ایک عینِ ثابتہ ہے، یعنی اس کی وہ حقیقت جو وجود میں آنے سے پہلے علمِ الٰہی میں ثابت تھی، اور اس عین کی ایک استعداد ہے جس سے آگے وہ کچھ قبول نہیں کرتی۔ فیضِ اقدس نے ہر عین کو اس کی استعداد عطا کی، اور فیضِ مقدس اسی استعداد کے مطابق اسے وجود اور کمالات دیتا ہے۔ "مَعْلُوْمٌ" کا لفظ یہی کہتا ہے: یہ مقام حق کے علم میں معلوم ہے۔ اور اس مکتب کا ایک بنیادی اصول ہے کہ علم معلوم کے تابع ہوتا ہے: حق ہر شے کو ویسا ہی جانتا ہے جیسی وہ اپنی حقیقت میں ہے، اور اسی کے مطابق حکم کرتا ہے۔ اس سے جبر لازم نہیں آتا؛ جاننا مجبور کرنا نہیں۔ یہی سرِّ قدر ہے جس کا ذکر ابھی آیت ۱۶۳ کے تحت ہوا: حکم محکوم کے مطابق ہے، اور کوئی شخص قیامت کے دن یہ نہیں کہہ سکے گا کہ میرے ساتھ میری حقیقت کے خلاف معاملہ ہوا۔ اہلِ ذوق کہتے ہیں کہ جو اس راز کو پا لے اسے نہ کسی کی عطا پر حسد رہتا ہے نہ اپنی محرومی کا شکوہ؛ وہ جانتا ہے کہ ہر ایک کو وہی ملا جس کا وہ ظرف رکھتا تھا، اور ظرف بھی اسی کی عطا ہے۔

مگر یہاں انسان کی ایک خصوصیت بھی سامنے آتی ہے۔ اہلِ معرفت کہتے ہیں کہ فرشتہ اپنے مقام پر قائم ہے؛ جو جس مرتبے میں پیدا ہوا اسی میں رہتا ہے، اور یہ اس کی کمی نہیں، اس کی نوع کا کمال ہے۔ انسان کو حق نے مقامات میں سفر کرنے والا بنایا: وہ توبہ سے زہد، زہد سے توکل، توکل سے رضا اور رضا سے فنا و بقا تک ترقی کرتا ہے۔ وہ تلوین سے گزر کر تمکین تک پہنچتا ہے، اور اسی سفر میں امانت کا بوجھ اٹھاتا ہے۔ شیخِ اکبر نے سورۂ الاحزاب کے الفاظ يٰۤاَهْلَ يَثْرِبَ لَا مُقَامَ لَكُمْ سے اشارۃً "مقامِ لا مقام" کا مفہوم لیا ہے، یعنی کامل محمدی وارث کسی ایک مقام میں مقید نہیں رہتا؛ ظاہری طور پر یہ آیت غزوۂ خندق کے منافقین کے بارے میں ہے اور وہی اس کا اصل معنی ہے، یہ صرف اشارہ ہے جو ظاہر کی نفی نہیں کرتا۔ پھر بھی ادب یہ ہے کہ انسان کی یہ حرکت بھی "مقامِ معلوم" سے باہر نہیں؛ اس کا ہر قدم حق کے علم میں پہلے سے ہے۔ اور یہی نعمت خطرہ بھی ہے: جو اوپر جا سکتا ہے وہ نیچے بھی گر سکتا ہے۔ فرشتہ نہ گرتا ہے نہ چڑھتا ہے؛ انسان کے سامنے دونوں راستے کھلے ہیں، اور یہی اس کے اختیار کی عزت ہے۔

فرشتوں کا اقرار جاری ہے: وَاِنَّا لَنَحْنُ الصَّآفُّوْنَ وَاِنَّا لَنَحْنُ الْمُسَبِّحُوْنَ۔ اور بے شک ہم ہی صف باندھنے والے ہیں، اور بے شک ہم ہی تسبیح کرنے والے ہیں۔ سورت کا دائرہ مکمل ہو رہا ہے: پہلی آیت میں "وَالصّٰٓفّٰتِ صَفًّا" کی قسم کھائی گئی تھی، اب صفیں خود اپنا تعارف کرا رہی ہیں۔ تاکید پر تاکید، "اِنَّا" اور "لَنَحْنُ"، فخر کے لیے نہیں بلکہ مشرکین کے دعوے کے رد کے لیے ہے: ہم معبود نہیں، عبادت گزار ہیں۔ حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: تم ایسی صفیں کیوں نہیں باندھتے جیسی فرشتے اپنے رب کے حضور باندھتے ہیں؟ وہ پہلی صفیں پوری کرتے ہیں اور صف میں خلا نہیں چھوڑتے۔ شیخِ اکبر کے مکتب میں نماز کی صف عالمِ ملکوت کا عکس ہے؛ جب بندے کندھے سے کندھا ملاتے ہیں تو وہ فرشتوں کی ترتیب میں شامل ہو جاتے ہیں۔ صف ترتیب ہے، یعنی ہر ایک کا اپنی جگہ پر ہونا، اور تسبیح اس ترتیب کی روح ہے۔ اہلِ ذوق کہتے ہیں کہ سالک کے دل میں بھی دو کام ہیں: خیالات کو صف بستہ کرنا اور پھر دل کو تسبیح میں لگا دینا۔

اب کفارِ مکہ کی ایک پرانی تمنا یاد دلائی جاتی ہے: وَاِنْ كَانُوْا لَيَقُوْلُوْنَ لَوْ اَنَّ عِنْدَنَا ذِكْرًا مِّنَ الْاَوَّلِيْنَ لَكُنَّا عِبَادَ اللّٰهِ الْمُخْلَصِيْنَ فَكَفَرُوْا بِهٖ فَسَوْفَ يَعْلَمُوْنَ۔ اور یہ لوگ کہا کرتے تھے کہ اگر ہمارے پاس پہلے لوگوں جیسی کوئی نصیحت ہوتی تو ہم ضرور اللہ کے خالص بندے ہوتے۔ پھر جب وہ آ گئی تو انہوں نے اس کا انکار کر دیا، سو عنقریب جان لیں گے۔ اصول یہاں پوری وضاحت سے ہے: ہدایت مانگی گئی، ہدایت دی گئی، اور انہوں نے خود منہ موڑا۔ گمراہی ان پر مسلط نہیں کی گئی؛ ان کا اعراض ہی ان کی گمراہی تھا، اور حق نے اسی رخ کو وجود دیا جو انہوں نے چنا۔ اہلِ معرفت ایک اور نکتہ دیکھتے ہیں: وہ "الْمُخْلَصِيْنَ" کا مقام چاہتے تھے مگر طلب کے بغیر؛ ان کا خیال تھا کہ کتاب ہاتھ آ جائے تو اخلاص خود بخود آ جائے گا۔ حالانکہ اخلاص اسی کو عطا ہوتا ہے جو آئی ہوئی ہدایت کے آگے سر جھکائے۔ دلچسپ یہ ہے کہ یہ لفظ اس سورت میں بار بار لوٹتا ہے، اور ہر بار یاد دلاتا ہے کہ خالص بنانا حق کا کام ہے اور حاضر ہونا بندے کا۔

اس کے مقابلے میں رسولوں کے لیے ازلی وعدہ: وَلَقَدْ سَبَقَتْ كَلِمَتُنَا لِعِبَادِنَا الْمُرْسَلِيْنَ اِنَّهُمْ لَهُمُ الْمَنْصُوْرُوْنَ وَاِنَّ جُنْدَنَا لَهُمُ الْغٰلِبُوْنَ۔ اور ہمارے بھیجے ہوئے بندوں کے حق میں ہماری بات پہلے ہی طے ہو چکی ہے کہ یقیناً انہی کی مدد کی جائے گی، اور یقیناً ہمارا لشکر ہی غالب رہے گا۔ "سَبَقَتْ" بتاتا ہے کہ یہ فیصلہ ازل میں ہو چکا؛ شیخِ اکبر کے نزدیک "كَلِمَتُنَا" وہی کلمۂ "کن" ہے جو علمِ الٰہی میں ہر شے کا انجام لکھ چکا۔ پھر "لِعِبَادِنَا" کا لفظ: مدد رسولوں کو ان کی بندگی کی بنا پر ملتی ہے، اپنی قوت کی بنا پر نہیں۔ "جُنْدَنَا" یعنی وہ جو ہمارے ہیں؛ لشکر کی نسبت حق کی طرف ہے، اس لیے غلبہ بھی اسی کا ہے۔ اہلِ تاویل کہتے ہیں کہ یہ غلبہ کبھی میدان میں ظاہر ہوتا ہے اور کبھی دلیل اور انجام میں؛ کبھی رسول بظاہر آزمائش میں ہوتا ہے مگر آخری بات اسی کی رہتی ہے۔ حضرت یونس علیہ السلام کا قصہ ابھی گزرا: مچھلی کے پیٹ سے نکل کر ایک لاکھ سے زائد کے ایمان تک؛ یہی "الْمَنْصُوْرُوْنَ" کی ایک تفسیر ہے۔ اہلِ ذوق کہتے ہیں کہ دل کے میدان میں بھی یہی قانون ہے: جو خیالات حق کے لشکر ہیں وہ آخرکار نفس کے لشکروں پر غالب آتے ہیں۔

پھر حضورِ اکرم صلی اللہ علیہ وسلم کو تسلی اور صبر کا حکم: فَتَوَلَّ عَنْهُمْ حَتّٰى حِيْنٍ وَاَبْصِرْهُمْ فَسَوْفَ يُبْصِرُوْنَ اَفَبِعَذَابِنَا يَسْتَعْجِلُوْنَ فَاِذَا نَزَلَ بِسَاحَتِهِمْ فَسَآءَ صَبَاحُ الْمُنْذَرِيْنَ۔ تو کچھ مدت تک ان سے رخ پھیر لیجیے، اور انہیں دیکھتے رہیے، عنقریب یہ بھی دیکھ لیں گے۔ کیا یہ ہمارے عذاب کی جلدی مچا رہے ہیں؟ پس جب وہ ان کے آنگن میں اترے گا تو ان لوگوں کی صبح بہت بری ہوگی جنہیں ڈرایا جا چکا تھا۔ "تَوَلَّ" یہاں ترکِ دعوت نہیں، ان کی ایذا اور بحث سے درگزر ہے۔ اور دو دیکھنے ہیں: ایک حضورِ اکرم صلی اللہ علیہ وسلم کا دیکھنا جو ابھی ہے، اور ایک ان کا دیکھنا جو بعد میں ہوگا۔ اہلِ معرفت کہتے ہیں کہ نبی کی بصیرت علم الیقین اور عین الیقین کی ہے جو انجام کو آج دیکھ لیتی ہے، اور منکر کو عین الیقین تب ملتا ہے جب وہ کسی کام کا نہیں رہتا۔ روایات میں آتا ہے کہ خیبر کی صبح حضورِ اکرم صلی اللہ علیہ وسلم نے اسی آیت کے مضمون سے فرمایا کہ جب ہم کسی قوم کے آنگن میں اترتے ہیں تو ڈرائے ہوئے لوگوں کی صبح بری ہوتی ہے۔ قومِ یونس اور ان لوگوں کا فرق یہی تھا: وہ صبح آنے سے پہلے لوٹ آئے تھے۔

اور حکم دہرایا گیا، مگر ایک لفظ کم کر کے: وَتَوَلَّ عَنْهُمْ حَتّٰى حِيْنٍ وَاَبْصِرْ فَسَوْفَ يُبْصِرُوْنَ۔ اور کچھ مدت تک ان سے رخ پھیر لیجیے، اور دیکھتے رہیے، عنقریب یہ بھی دیکھ لیں گے۔ پہلی بار "اَبْصِرْهُمْ" تھا، یعنی انہیں دیکھیے؛ اس بار صرف "اَبْصِرْ"، یعنی دیکھیے۔ مفسرین کہتے ہیں کہ پہلی بار دنیا کے انجام کی طرف اشارہ تھا اور دوسری بار آخرت کے انجام کی طرف۔ اہلِ معرفت اس حذف میں ایک وسعت دیکھتے ہیں: پہلی نگاہ مخالفین پر تھی، دوسری نگاہ کسی ایک چیز میں مقید نہیں؛ حضورِ اکرم صلی اللہ علیہ وسلم سے فرمایا جا رہا ہے کہ پورے منظر کو دیکھیے، حق کے افعال کو، اس کی حکمت کو، اس کے وعدے کے پورے ہونے کو۔ شیخِ اکبر کے مکتب میں انسانِ کامل وہ ہے جس کی نگاہ کسی جزئی پر رک کر اٹکتی نہیں؛ وہ ہر مظہر میں ظاہر ہونے والے کو دیکھتا ہے۔ اہلِ ذوق کہتے ہیں کہ سالک کی نگاہ بھی پہلے مخلوق پر ہوتی ہے، کون اس کے ساتھ ہے اور کون مخالف؛ پھر مفعول گر جاتا ہے اور صرف دیکھنا رہ جاتا ہے۔

اور اب سورت کا اختتام، تین کلمات میں جو قرآن کی بلیغ ترین مہروں میں سے ہیں۔ پہلا کلمہ تنزیہ کا ہے: سُبْحٰنَ رَبِّكَ رَبِّ الْعِزَّةِ عَمَّا يَصِفُوْنَ۔ پاک ہے آپ کا رب، عزت کا رب، ان باتوں سے جو یہ بیان کرتے ہیں۔ "رَبِّكَ" پر غور کیجیے: آپ کا رب، جس نے آپ کو پالا، جس نے سورت بھر کے انبیاء کو نجات دی، اور جو اب آپ کو تسلی دے رہا ہے۔ اہلِ معرفت اسے ربِّ خاص کی نسبت کہتے ہیں: ہر بندے کا اپنا ایک رشتہ اپنے رب سے ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم کا رشتہ سب سے کامل ہے۔ پھر "رَبِّ الْعِزَّةِ": عزت کا ایک معنی ہے ناقابلِ رسائی ہونا۔ شیخِ اکبر کے نزدیک ذاتِ حق اپنی عزت میں ہر ادراک سے بلند ہے؛ عقل وہاں نہیں پہنچتی، اور جو کچھ پہنچتا ہے وہ اس کی تجلی ہے، ذات نہیں۔ اسی لیے "عَمَّا يَصِفُوْنَ" ظاہر میں مشرکین کے اقوال کا رد ہے، اور اشارے میں ہر اس توصیف سے تنزیہ جو بندہ اپنی عقل سے گھڑے۔ یہاں "الْمُخْلَصِيْنَ" والا استثنا دوبارہ نہیں آیا، گویا آخری لفظ خالص تنزیہ کا ہو، جہاں ہر بیان کرنے والا حیرت میں ٹھہر جائے۔ اور یہ حیرت جہالت نہیں، معرفت کی انتہا ہے۔

دوسرا کلمہ سلام ہے: وَسَلٰمٌ عَلَى الْمُرْسَلِيْنَ۔ اور سلام ہو تمام رسولوں پر۔ اس سورت میں سلام ایک ٹیپ کا بند تھا: حضرت نوح علیہ السلام پر سلام، حضرت ابراہیم علیہ السلام پر سلام، حضرت موسیٰ و ہارون علیہما السلام پر سلام، حضرت الیاس علیہ السلام پر سلام۔ ہر بار ایک نبی کا نام تھا؛ اب سلام سب کو سمیٹ لیتا ہے، جن کا ذکر ہوا اور جن کا نہیں ہوا۔ تنزیہ کے فوراً بعد رسولوں پر سلام کیوں؟ اہلِ معرفت کہتے ہیں کہ رسول ہی وہ ہیں جنہوں نے حق کی وہ صفت بیان کی جو اس نے خود اپنے لیے پسند کی؛ اس لیے ان کا بیان "عَمَّا يَصِفُوْنَ" کی زد سے باہر ہے، اور سلامتی ان کا حق ہے۔ شیخِ اکبر کے مکتب میں رسول تنزیہ اور تشبیہ کے درمیان برزخ ہیں: وہ حق کو بے مثل بھی بتاتے ہیں اور اس کی قربت، رحمت اور سمع و بصر کی خبر بھی دیتے ہیں۔ جو ان کے راستے سے حق کو پہچانے، اس کی معرفت بھی سلامت رہتی ہے، نہ خشک تنزیہ میں کھوتی ہے نہ تجسیم میں گرتی ہے۔ اور سلام کا لفظ بتاتا ہے کہ رسولوں کے دشمن جو کچھ کہیں، حق کے ہاں ان کا مقام محفوظ ہے؛ یہی "الْمَنْصُوْرُوْنَ" کا آخری مظہر ہے۔

اور تیسرا کلمہ حمد ہے، جس پر سورت تمام ہوتی ہے: وَالْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِيْنَ۔ اور سب تعریف اللہ کے لیے ہے جو تمام جہانوں کا رب ہے۔ تسبیح نقص کی نفی ہے اور حمد کمال کا اثبات؛ ایک کہتی ہے کہ وہ کیا نہیں ہے، دوسری کہتی ہے کہ ہر خوبی اسی کی ہے۔ شیخِ اکبر کے نزدیک یہی تنزیہ و تشبیہ کا جمع ہے: "سُبْحٰنَ" میں ذات کی بلندی ہے اور "الْحَمْدُ" میں ہر مظہر کے کمال کا اسی کی طرف لوٹنا، اور ان دونوں کے درمیان رسولوں پر سلام، جیسے دونوں کناروں کو ملانے والا پل۔ پھر "رَبِّ الْعٰلَمِيْنَ": سورت فرشتوں کی صفوں سے چلی، شیاطین، اہلِ جنت، اہلِ دوزخ اور انبیاء کے جہانوں سے گزری، اور ہر جہان کو ایک رب کی ربوبیت میں سمیٹ کر ختم ہوئی۔ روایات میں ان تین آیات کی ایک خاص فضیلت آتی ہے: حضرت علی رضی اللہ عنہ سے منقول ہے، جس کا مفہوم یہ ہے کہ جو چاہے کہ قیامت کے دن اسے پورے پیمانے سے اجر ملے، وہ اپنی مجلس کے آخر میں یہی کلمات پڑھے؛ اور حضورِ اکرم صلی اللہ علیہ وسلم کا معمول بھی مجلس کے اختتام پر تسبیح و حمد اور استغفار کا تھا۔ اہلِ ذوق کہتے ہیں کہ ہر مجلس، ہر دن اور ہر عمر کا بہترین اختتام یہی تین باتیں ہیں: اپنی کہی ہوئی باتوں سے حق کو پاک قرار دینا، اس تک پہنچانے والوں پر سلام بھیجنا، اور ہر نعمت کی تعریف اس کے مالک کو لوٹا دینا۔ ظاہر معنی اپنی جگہ مقدم ہے، اور یہ اشارے اسی کی گہرائی ہیں۔
