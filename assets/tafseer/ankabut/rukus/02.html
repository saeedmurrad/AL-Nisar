پہلے رکوع نے بتایا تھا کہ ایمان کا دعویٰ آزمائش کے بغیر نہیں چھوڑا جاتا؛ اب قرآن اسی آزمائش کی تاریخ کھولتا ہے، اور سب سے پہلے اس نبی کو سامنے لاتا ہے جس کی آزمائش سب سے طویل تھی: وَلَقَدْ اَرْسَلْنَا نُوْحًا اِلٰى قَوْمِهٖ فَلَبِثَ فِيْهِمْ اَلْفَ سَنَةٍ اِلَّا خَمْسِيْنَ عَامًا ۭ فَاَخَذَهُمُ الطُّوْفَانُ وَهُمْ ظٰلِمُوْنَ۔ اور یقیناً ہم نے حضرت نوح علیہ السلام کو ان کی قوم کی طرف بھیجا، تو وہ ان میں پچاس کم ایک ہزار برس رہے، پھر ان لوگوں کو طوفان نے آ پکڑا اور وہ ظالم تھے۔ حضرت نوح علیہ السلام کی ساڑھے نو صدیاں اس سورت کے پہلے سبق کی سب سے بڑی شرح ہیں: داعی کا صبر بھی آزمائش ہے اور مدعو کی مہلت بھی۔ اہلِ بلاغت نے یہ لطیفہ لکھا ہے کہ ایک ہی آیت میں سال کے لیے دو لفظ آئے؛ سَنَة جو عربی میں اکثر سختی کے برسوں کے لیے آتا ہے، اور عَام جو کشادگی کے برسوں کے لیے؛ اور ہزار کو سَنَة کہہ کر بتا دیا کہ دعوت کا زمانہ مشقت کا زمانہ تھا۔ پھر آخری کلمہ دیکھیے: وَهُمْ ظٰلِمُوْنَ، یعنی طوفان نے انہیں اس حال میں پکڑا کہ وہ ظلم پر قائم تھے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہی سرِّ قدر ہے: حکم محکوم کے حال کے مطابق اترتا ہے۔ طوفان نے انہیں ظالم نہیں بنایا؛ وہ نو صدیوں تک اپنا رخ خود چنتے رہے، اور جب پکڑ آئی تو اسی رخ پر آئی جس پر وہ کھڑے تھے۔ مہلت اتنی طویل رکھی گئی کہ کوئی یہ نہ کہہ سکے کہ اسے سننے کا موقع نہ ملا، اور یہی حق تعالیٰ کی رحمت کا وہ رخ ہے جو عذاب سے پہلے آتا ہے۔

پھر نجات کا ذکر: فَاَنْجَيْنٰهُ وَاَصْحٰبَ السَّفِيْنَةِ وَجَعَلْنٰهَاۤ اٰيَةً لِّلْعٰلَمِيْنَ۔ پھر ہم نے انہیں اور کشتی والوں کو بچا لیا، اور اسے سارے جہانوں کے لیے ایک نشانی بنا دیا۔ ظاہر میں کشتی لکڑی اور کیلوں کی تھی، اور یہ ظاہر اپنی جگہ محکم ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ تاویل کہتے ہیں کہ ہر زمانے میں ایک طوفان ہے، خواہشوں اور غفلت کا سیلاب، اور ہر زمانے میں ایک کشتی ہے، یعنی شریعت جس کے تختے احکام ہیں اور جس کا ناخدا نبی ہے۔ دیکھیے نجات صرف نبی کے لیے نہیں کہی گئی، اَصْحٰبَ السَّفِيْنَةِ کے لیے بھی کہی گئی؛ اہلِ ذوق اسی سے صحبت کا راز نکالتے ہیں کہ نجات اکثر ساتھ سے ملتی ہے، جو صادق کے ساتھ سوار ہو گیا وہ اس کے ساتھ پار اتر گیا۔ فصوص الحکم میں حضرت نوح علیہ السلام کے باب کا نام حکمتِ سبوحیہ ہے، اور اس کا مفہوم یہ ہے کہ حق کی معرفت نہ خالص تنزیہ میں قید ہے نہ خالص تشبیہ میں؛ صرف پاکی بیان کرنے والا بھی اسے ایک حد میں باندھ دیتا ہے اور صرف مشابہت دیکھنے والا بھی؛ کامل معرفت دونوں کا جمع ہے۔ اہلِ ذوق کے نزدیک کشتی اسی جمع کی تمثیل بن جاتی ہے: پانی پر ہے مگر پانی میں ڈوبی نہیں؛ موجوں کے بیچ ہے مگر موجوں میں گم نہیں۔ عارف بھی یوں ہی مخلوق کے درمیان رہتا ہے اور مخلوق میں تحلیل نہیں ہوتا۔ اور لِّلْعٰلَمِيْنَ کہہ کر اس نشانی کو ایک قوم کی تاریخ سے نکال کر ہر عالم اور ہر دل کی میراث بنا دیا۔

اب حضرت ابراہیم علیہ السلام آتے ہیں، اور ان کی دعوت چند لفظوں میں سمٹی ہوئی ہے: وَاِبْرٰهِيْمَ اِذْ قَالَ لِقَوْمِهِ اعْبُدُوا اللّٰهَ وَاتَّقُوْهُ ۭ ذٰلِكُمْ خَيْرٌ لَّكُمْ اِنْ كُنْتُمْ تَعْلَمُوْنَ۔ اور حضرت ابراہیم علیہ السلام کو بھیجا، جب انہوں نے اپنی قوم سے کہا: اللہ کی عبادت کرو اور اس سے ڈرو؛ یہ تمہارے لیے بہتر ہے اگر تم جانو۔ پھر بت پرستی کی حقیقت کھولی: اِنَّمَا تَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ اَوْثَانًا وَّتَخْلُقُوْنَ اِفْكًا۔ تم اللہ کو چھوڑ کر محض بتوں کو پوجتے ہو اور ایک جھوٹ گھڑتے ہو۔ اِنَّ الَّذِيْنَ تَعْبُدُوْنَ مِنْ دُوْنِ اللّٰهِ لَا يَمْلِكُوْنَ لَكُمْ رِزْقًا فَابْتَغُوْا عِنْدَ اللّٰهِ الرِّزْقَ وَاعْبُدُوْهُ وَاشْكُرُوْا لَهٗ ۭ اِلَيْهِ تُرْجَعُوْنَ۔ بے شک جنہیں تم اللہ کے سوا پوجتے ہو وہ تمہارے لیے رزق کے مالک نہیں؛ تو رزق اللہ کے پاس تلاش کرو، اسی کی عبادت کرو اور اسی کا شکر کرو؛ اسی کی طرف تم لوٹائے جاؤ گے۔ تَخْلُقُوْنَ اِفْكًا میں عجیب لطافت ہے: بت تراشنے کو خلق سے تعبیر کیا، گویا بت پرست خالق کی نقل کرتا ہے مگر اس کی تخلیق جھوٹ نکلتی ہے۔ اہلِ معرفت کہتے ہیں کہ ہر بت پہلے خیال میں تراشا جاتا ہے، پھر پتھر میں؛ اور جو صورت بندہ اپنے وہم سے گھڑ کر اس پر ٹھہر جائے، وہی اس کا وثن ہے، خواہ وہ دولت ہو، رتبہ ہو یا اپنی ہی رائے۔ پھر دیکھیے کہ رِزْقًا پہلے نکرہ آیا اور پھر الرِّزْقَ معرفہ: وہ تمہیں رزق کا ایک ذرہ بھی نہیں دے سکتے، اور سارا رزق اللہ کے پاس ہے؛ اور عِنْدَ اللّٰهِ کو پہلے لا کر طلب کا دروازہ ایک ہی کر دیا۔ شیخِ اکبر کے نزدیک رزق صرف روٹی نہیں؛ علم، حال اور تجلی بھی رزق ہیں، اور ہر عینِ ثابتہ کو اس کی استعداد کے مطابق ملتا ہے۔ ترتیب بھی سلوک کی ترتیب ہے: پہلے طلب، پھر عبادت، پھر شکر، اور انجام رجوع۔

دعوت کے بعد داعی کی ذمہ داری کی حد بیان ہوئی: وَاِنْ تُكَذِّبُوْا فَقَدْ كَذَّبَ اُمَمٌ مِّنْ قَبْلِكُمْ ۭ وَمَا عَلَى الرَّسُوْلِ اِلَّا الْبَلٰغُ الْمُبِيْنُ۔ اور اگر تم جھٹلاؤ تو تم سے پہلے بھی بہت سی امتیں جھٹلا چکی ہیں، اور رسول کے ذمے صاف صاف پہنچا دینے کے سوا کچھ نہیں۔ بعض مفسرین کے نزدیک یہاں سے چند آیات حضرت ابراہیم علیہ السلام کے قصے کے بیچ میں اہلِ مکہ سے براہِ راست خطاب ہیں، اور بعض انہیں حضرت ابراہیم علیہ السلام ہی کا کلام کہتے ہیں؛ دونوں صورتوں میں سبق ایک ہے۔ رسول کا کام پہنچانا ہے، منوانا نہیں۔ یہ اختیار کی حرمت ہے: حق تعالیٰ چاہتا تو سب کو زبردستی سیدھا کر دیتا، مگر اس نے دعوت کا راستہ رکھا تاکہ جو آئے اپنے رخ سے آئے۔ پھر دلیل کا رخ آفاق کی طرف موڑا: اَوَلَمْ يَرَوْا كَيْفَ يُبْدِئُ اللّٰهُ الْخَلْقَ ثُمَّ يُعِيْدُهٗ ۭ اِنَّ ذٰلِكَ عَلَى اللّٰهِ يَسِيْرٌ۔ کیا انہوں نے نہیں دیکھا کہ اللہ کس طرح مخلوق کو پہلی بار پیدا کرتا ہے، پھر اسے دوبارہ لوٹاتا ہے؟ بے شک یہ اللہ پر آسان ہے۔ فعل مضارع میں ہے: يُبْدِئُ، وہ پیدا کرتا ہے، اب بھی، ہر لمحہ۔ شیخِ اکبر کے نزدیک یہی تجددِ امثال کا راز ہے: عالم ہر آن اپنی ذات میں معدوم ہو کر نئے سرے سے وجود پاتا ہے، اور یہ تسلسل اتنا باریک ہے کہ آنکھ اسے ایک ہی چیز سمجھتی ہے۔ ابتدا اور اعادہ دو دور کے واقعے نہیں، ہر سانس میں جاری ہیں۔ سورۂ النمل میں تخت کے پلک جھپکتے حاضر ہو جانے پر جو خلقِ جدید کی بات گزری تھی، یہاں وہی اصول پوری کائنات پر پھیلا ہوا ہے۔ اور جسے ہر لمحے کی نئی پیدائش کا شعور ہو جائے، اس کے لیے دوسری زندگی کا انکار ممکن نہیں رہتا۔

اگلی آیت آنکھ کو سفر پر بھیجتی ہے: قُلْ سِيْرُوْا فِي الْاَرْضِ فَانْظُرُوْا كَيْفَ بَدَاَ الْخَلْقَ ثُمَّ اللّٰهُ يُنْشِئُ النَّشْاَةَ الْاٰخِرَةَ ۭ اِنَّ اللّٰهَ عَلٰى كُلِّ شَيْءٍ قَدِيْرٌ۔ کہہ دیجیے: زمین میں چلو پھرو، پھر دیکھو کہ اس نے مخلوق کو کیسے پہلی بار پیدا کیا، پھر اللہ ہی دوسری پیدائش اٹھائے گا؛ بے شک اللہ ہر چیز پر قادر ہے۔ ایک باریکی پر ٹھہریے: پہلی پیدائش کے ذکر میں فاعل ضمیر میں چھپا رہا، بَدَاَ، مگر دوسری پیدائش پر نام ظاہر ہو گیا: ثُمَّ اللّٰهُ يُنْشِئُ۔ اہلِ ذوق کہتے ہیں کہ پہلی پیدائش اسباب کے پردے میں دکھائی دیتی ہے؛ ماں، باپ، بیج اور بارش کے پردے میں؛ اسی لیے اس کا مشاہدہ زمین کی سیر سے ہوتا ہے۔ دوسری پیدائش میں اسباب کے پردے اٹھ جاتے ہیں اور فاعل بے پردہ سامنے آتا ہے۔ اہلِ تاویل اس سیر کو باطن کی سیر بھی کہتے ہیں: تمہاری اپنی مٹی بھی ایک زمین ہے، اس میں چلو تو دیکھو گے کہ تمہاری ہر حالت کیسے شروع ہوئی اور کس کے ہاتھ سے۔ اور النَّشْاَةَ الْاٰخِرَةَ کا ایک عکس اسی دنیا میں بھی ہے؛ اہلِ تصوف میں مشہور اثر ہے کہ جو دو بار پیدا نہ ہو وہ آسمانوں کی بادشاہی میں داخل نہیں ہوتا؛ پہلی ولادت ماں سے، دوسری نفس کی موت کے بعد معرفت میں۔ شیخِ اکبر کے مکتب میں یہ دوسری پیدائش فنا کے بعد بقا ہے: بندہ وہی رہتا ہے، مگر اب اپنی ہستی سے نہیں، حق کی عطا کردہ ہستی سے دیکھتا ہے۔ یہ اشارہ ظاہر کی جگہ نہیں لیتا؛ قیامت کی نشأت اپنی جگہ برحق ہے، اور یہ اس کا دنیا میں پڑنے والا سایہ ہے۔ اور آیت کا خاتمہ قدرت پر ہے، کیونکہ جو پہلی بار بنا سکا اس کے لیے دوبارہ بنانا کیا مشکل۔

اب وہ آیت جو مشیت کا بیان کرتی ہے: يُعَذِّبُ مَنْ يَّشَاءُ وَيَرْحَمُ مَنْ يَّشَاءُ ۚ وَاِلَيْهِ تُقْلَبُوْنَ۔ وہ جسے چاہے عذاب دے اور جس پر چاہے رحم کرے، اور اسی کی طرف تم پلٹائے جاؤ گے۔ اس آیت کو سرِّ قدر کی روشنی میں پڑھنا لازم ہے، ورنہ مشیت جبر دکھائی دے گی۔ شیخِ اکبر کے نزدیک علم معلوم کے تابع ہے: حق تعالیٰ ہر عین کو ویسا ہی جانتا ہے جیسا وہ اپنی حقیقت میں ہے، اور حکم اسی علم کے مطابق صادر ہوتا ہے۔ تو مَنْ يَّشَاءُ کا مطلب یہ نہیں کہ کسی طالب کو دروازے سے لوٹا دیا جاتا ہے؛ مطلب یہ ہے کہ جس نے اپنا رخ رحمت کی طرف کیا اسے رحمت ملتی ہے، اور جس نے منہ موڑا اس کا منہ موڑنا ہی اس کا عذاب بن جاتا ہے؛ حق اس رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ اور حدیثِ قدسی کا مفہوم ہے کہ میری رحمت میرے غضب پر سبقت لے گئی؛ اس لیے رحمت کی طرف بڑھنے والا کبھی خالی نہیں لوٹتا۔ پھر تُقْلَبُوْنَ: یہ وہی مادہ ہے جس سے قلب بنا؛ اہلِ معرفت کہتے ہیں کہ دل کو قلب اسی لیے کہتے ہیں کہ وہ ہر آن پلٹتا ہے، اور اس کا آخری پلٹنا اسی کی طرف ہے۔ پھر فرار کا ہر راستہ بند کر دیا: وَمَاۤ اَنْتُمْ بِمُعْجِزِيْنَ فِي الْاَرْضِ وَلَا فِي السَّمَآءِ ۚ وَمَا لَكُمْ مِّنْ دُوْنِ اللّٰهِ مِنْ وَّلِيٍّ وَّلَا نَصِيْرٍ۔ اور تم نہ زمین میں اسے عاجز کر سکتے ہو نہ آسمان میں، اور اللہ کے سوا تمہارا نہ کوئی کارساز ہے نہ مددگار۔ اہلِ معرفت کہتے ہیں کہ اس سے بھاگنے کی کوئی جگہ نہیں، مگر اس کی طرف بھاگنے کا دروازہ ہمیشہ کھلا ہے: فَفِرُّوْۤا اِلَى اللّٰهِ۔ تو اللہ کی طرف دوڑو۔ جس کا ولی وہی ہو اسے کسی اور ولی کی حاجت نہیں، اور یہی بات اگلے رکوع کی ہجرت کا دروازہ کھولتی ہے۔
