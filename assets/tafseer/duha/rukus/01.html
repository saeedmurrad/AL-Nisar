یہ تسلی کی سورت ہے، اور اس کا پس منظر وحی کا ایک مختصر وقفہ ہے۔ صحیح بخاری میں حضرت جندب رضی اللہ عنہ کی روایت کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم کی طبیعت کچھ ناساز ہوئی اور آپ دو یا تین رات قیام نہ فرما سکے، اور انہی دنوں وحی کچھ مدت رکی رہی؛ اس پر معاندین میں سے ایک عورت نے طعنہ دیا کہ آپ کا ساتھ چھوٹ گیا۔ قرآن نے اس طعنے کا جواب دلیل سے پہلے قسم سے دیا: وَالضُّحٰى۔ قسم ہے دن چڑھے کی روشنی کی؛ وَالَّيْلِ اِذَا سَجٰى۔ اور رات کی جب وہ ٹھہر کر چھا جائے۔ "الضُّحٰى" دن کا وہ حصہ ہے جب سورج بلند ہو کر روشنی پوری طرح پھیل جاتی ہے مگر دوپہر کی تپش ابھی نہیں آتی؛ اور "سَجٰى" کا مطلب ہے سکون سے ڈھانپ لینا، جیسے سمندر ٹھہر جائے اور ہوائیں تھم جائیں۔ مفسرین لکھتے ہیں کہ ان دو قسموں میں خود جواب کی دلیل چھپی ہے: رات کا آنا دن کی موت نہیں، اور روشنی کا کچھ دیر اوجھل ہونا سورج کی جدائی نہیں۔ جس طرح رات کا سکون اگلی صبح کی تیاری ہے، اسی طرح وحی کا وقفہ اگلے نزول کا شوق بڑھانے کے لیے تھا، ترکِ تعلق کے لیے ہرگز نہیں۔ ظاہر یہی ہے اور یہی مقدم ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کائنات کا ہر منظر اسماءِ الٰہیہ کا مظہر ہے؛ دن اسمِ ظاہر کی تجلی ہے اور رات اسمِ باطن کی، اور دونوں ایک ہی حق کے دو رخ ہیں۔ پس جو رات میں ہے وہ بھی حق کے ساتھ ہے؛ صرف تجلی کا رنگ بدلا ہے، تجلی کرنے والا نہیں۔

اہلِ معرفت کے نزدیک یہ دو قسمیں دل کے دو موسموں کی قسمیں ہیں۔ سالک کے دل پر کبھی چاشت طلوع ہوتی ہے: کشادگی، ذوق، انوار کی بارش اور عبادت میں لذت؛ اسے بسط کہتے ہیں۔ اور کبھی رات اترتی ہے: تنگی، خاموشی، وارد کا رک جانا اور ذوق کا چھپ جانا؛ اسے قبض کہتے ہیں۔ نوآموز قبض کو دوری سمجھ کر گھبراتا ہے اور بسط کو قرب سمجھ کر اتراتا ہے۔ یہ سورت دونوں غلطیوں کا علاج ہے، کیونکہ حق تعالیٰ نے دونوں کی قسم کھائی؛ یعنی دونوں اس کی نشانیاں ہیں، اور جس چیز کی قسم کھائی جائے وہ محترم ہوتی ہے۔ رات کی قسم "اِذَا سَجٰى" کی قید کے ساتھ ہے: وہ رات جو سکون لائے، وحشت نہیں۔ اہلِ ذوق کہتے ہیں کہ قبض اگر ادب کے ساتھ گزارا جائے تو وہ ایسی ہی ٹھہری ہوئی رات بن جاتا ہے، جس میں دل کی ہوائیں تھم جاتی ہیں اور اندر کا سمندر ساکن ہو کر اپنی گہرائی دکھاتا ہے۔ شیخِ اکبر کے نزدیک قبض اور بسط دونوں تلوین کی حالتیں ہیں، اور کامل وہ ہے جو تمکین پا لے: جس کا دل ہر حال میں حق کے ساتھ ٹھہرا رہے، خواہ وارد آئے یا رکے۔ حضورِ اکرم صلی اللہ علیہ وسلم تمکین کے سب سے بلند مقام پر تھے؛ وحی کا وقفہ آپ کے لیے وحشت کا قبض نہ تھا بلکہ شوق کی شدت تھا، اور شوق خود قرب کی ایک صورت ہے، دوری کی نہیں۔

ایک باریک نکتہ ترتیب میں ہے۔ سورۂ اللیل میں رات کا ذکر پہلے آیا اور دن کا بعد میں، اور یہاں چاشت پہلے ہے اور رات بعد میں۔ مفسرین کے ہاں یہ لطیف بات ملتی ہے کہ تسلی کی اس سورت میں روشنی کو مقدم کیا گیا تاکہ پہلا لفظ ہی دل کو کشادہ کر دے۔ شیخِ اکبر کے مکتب میں یہ سبقتِ رحمت کا اظہار ہے: حدیث کا مفہوم ہے کہ حق تعالیٰ کی رحمت اس کے غضب پر سبقت رکھتی ہے، اور اس مکتب میں وجود کا آغاز جمال سے ہے، جلال اس کے بعد اور اسی کے تابع آتا ہے۔ پھر یہ بھی دیکھو کہ چاشت دن کا پورا حصہ نہیں، اس کا ایک ٹکڑا ہے، اور رات کی قسم پوری رات کے ساتھ ہے۔ اہلِ تاویل کہتے ہیں کہ اس میں اشارہ ہے کہ بسط کی گھڑی چھوٹی ہو تب بھی ساری رات پر بھاری ہے، کیونکہ ایک تجلی برسوں کے قبض کا بدل ہو جاتی ہے۔ اور خلقِ جدید کے اصول پر، جسے شیخِ اکبر تجددِ امثال کہتے ہیں، نہ کوئی رات ہمیشہ رات رہتی ہے نہ کوئی دن ہمیشہ دن؛ ہر آن حق کی نئی تجلی ہے اور عالم ہر لمحہ نئے سرے سے وجود پاتا ہے۔ جو اسے سمجھ لے وہ رات میں مایوس نہیں ہوتا، کیونکہ جانتا ہے کہ ہر آن نیا فیض آ رہا ہے، اور اگلی آن کی چاشت اسی رات کے بطن میں تیار ہو رہی ہے۔

اب قسم کا جواب آتا ہے: مَا وَدَّعَكَ رَبُّكَ وَمَا قَلٰى۔ تمہارے رب نے نہ تمہیں رخصت کیا، اور نہ وہ ناراض ہوا۔ "وَدَّعَكَ" تودیع سے ہے، یعنی کسی کو الوداع کہنا، اور الوداع وہی کہتا ہے جو جدا ہو رہا ہو؛ اور "قَلٰى" شدید ناپسندیدگی اور بیزاری کو کہتے ہیں۔ آیت نے دونوں کی نفی کر دی: نہ جدائی ہوئی، نہ ناراضی۔ یہ بات قطعی ہے اور اس پر امت کا ایمان ہے کہ حق تعالیٰ نے اپنے حبیب صلی اللہ علیہ وسلم کو ایک لمحے کے لیے بھی نہیں چھوڑا؛ وحی کا وقفہ حکمت تھا، ترک نہیں۔ مفسرین نے ایک لطیف ادب کی طرف توجہ دلائی ہے: پہلے فعل "وَدَّعَكَ" میں خطاب کی ضمیر موجود ہے، مگر دوسرے فعل کے ساتھ ضمیرِ خطاب نہیں لائی گئی، صرف "وَمَا قَلٰى" فرمایا۔ ایک وجہ آیات کے فواصل کی ہم آہنگی ہے، اور دوسری وجہ جسے اہلِ علم نے ذکر کیا یہ ہے کہ ناراضی کا لفظ، نفی کی صورت میں بھی، محبوب کی ذات سے براہِ راست نہ جوڑا جائے۔ یہ اکرام کا اسلوب ہے۔ اہلِ معرفت کہتے ہیں کہ جو کلام محبوب سے ناراضی کی نفی کرتے ہوئے بھی اس کی طرف وہ لفظ نہیں پھیرتا، وہ خود محبت کی گواہی ہے۔ اور یہاں "رَبُّكَ" کا لفظ پرورش اور تربیت کی نسبت یاد دلاتا ہے: جو پال رہا ہو وہ بیچ راہ میں نہیں چھوڑتا، اور جس کی تربیت ابھی جاری ہو اس سے ناراضی کا گمان تربیت کے مزاج کے خلاف ہے۔

شیخِ اکبر کے نزدیک ہر موجود کا ایک ربِّ خاص ہے، یعنی وہ اسمِ الٰہی جس کے سائے میں اس کی تربیت ہوتی ہے؛ اور اس مکتب میں حضورِ اکرم صلی اللہ علیہ وسلم کا ربِّ خاص اسمِ جامع "اللہ" ہے، کیونکہ آپ انسانِ کامل ہیں جن میں تمام اسماء کے مظاہر جمع ہیں۔ اس لیے "رَبُّكَ" میں وہ رب مراد ہے جس کی ربوبیت تمام اسماء کو گھیرے ہوئے ہے؛ اس سے جدائی کا تصور ہی ممکن نہیں، کیونکہ جو اسمِ جامع کا مظہر ہو اس پر ایک اسم کی تجلی کا رکنا دوسرے اسم کی تجلی کا آنا ہے۔ وحی کا وقفہ اسی معنی میں ایک دوسری طرح کا حضور تھا: کلام رکا، متکلم نہیں ہٹا۔ اہلِ ذوق کہتے ہیں کہ محبوب کو کبھی خاموشی سے بھی پالا جاتا ہے؛ خاموشی میں شوق پکتا ہے اور ظرف کشادہ ہوتا ہے، تاکہ اگلا کلام پہلے سے زیادہ گنجائش پائے۔ اسی لیے وقفے کے بعد جو کلام آیا وہ یہی سراپا تسلی سورت تھی۔ اور ہمارے لیے سبق یہ ہے کہ سالک کے ورد میں ذوق رک جائے یا دعا کا جواب دیر سے آئے تو وہ اسے "مَا وَدَّعَكَ" کی روشنی میں دیکھے؛ تاخیر ترک نہیں، اور خاموشی ناراضی نہیں۔ جو دل اس یقین پر ٹھہر جائے، اس کی رات بھی "اِذَا سَجٰى" والی رات بن جاتی ہے، جس کا سکون صبح کی خبر دیتا ہے۔

پھر فرمایا: وَلَلْاٰخِرَةُ خَيْرٌ لَّكَ مِنَ الْاُوْلٰى۔ اور یقیناً بعد والی تمہارے لیے پہلی سے بہتر ہے۔ ظاہر معنی میں "الْاٰخِرَةُ" آخرت ہے اور "الْاُوْلٰى" دنیا؛ یعنی جو اکرام آپ کے لیے آخرت میں رکھا گیا ہے وہ دنیا کی ہر نعمت سے بڑھ کر ہے۔ مگر مفسرین کی ایک جماعت نے لفظ کی وسعت سے یہ معنی بھی لیا کہ آپ کی ہر آنے والی حالت پچھلی حالت سے بہتر ہے: ہر دن پچھلے دن سے، ہر مرحلہ پچھلے مرحلے سے بلند۔ سیرت اس کی گواہ ہے: مکہ کی تنہائی سے مدینہ کی جماعت تک، اور مدینہ سے فتحِ مکہ تک، آپ کا ہر قدم اوپر کی طرف تھا۔ صوفیہ نے اسی کو باطنی معنوں میں لیا: آپ کی معرفت میں کوئی ٹھہراؤ نہیں، ہر لمحہ عروج ہے۔ شیخِ اکبر کے مکتب میں اس کی بنیاد یہ ہے کہ حق کی تجلی کبھی مکرر نہیں ہوتی؛ ہر آن نئی تجلی ہے، اور ہر نئی تجلی نیا علم ساتھ لاتی ہے، کیونکہ بندے کی استعداد ہر تجلی سے کشادہ ہو کر اگلی کے لیے زیادہ ظرف بن جاتی ہے۔ اس طرح "الْاٰخِرَةُ" صرف موت کے بعد کا جہان نہیں، ہر بعد ہے، اور "الْاُوْلٰى" ہر پہلے ہے۔ آیت کے شروع کا لام تاکید یہ بتاتا ہے کہ یہ وعدہ قطعی ہے۔ وحی کے وقفے کے پس منظر میں یہ آیت کہتی ہے: وقفہ نیچے اترنا نہیں تھا، اگلی بلندی سے پہلے کا ٹھہراؤ تھا۔

اس کے بعد وہ آیت آتی ہے جسے بہت سے اہلِ علم نے قرآن کی سب سے زیادہ امید دلانے والی آیات میں شمار کیا: وَلَسَوْفَ يُعْطِيْكَ رَبُّكَ فَتَرْضٰى۔ اور عنقریب تمہارا رب تمہیں اتنا عطا فرمائے گا کہ تم راضی ہو جاؤ گے۔ شروع کا لام تاکید کا ہے اور لفظ سوف ایک پھیلے ہوئے مستقبل کا؛ یعنی عطا یقینی بھی ہے اور مسلسل بھی، دنیا میں بھی اور آخرت میں بھی۔ دنیا میں فتوحات، دین کا غلبہ اور امت کی کثرت؛ اور آخرت میں حوضِ کوثر، مقامِ محمود اور شفاعتِ کبریٰ۔ صحیح مسلم کی روایت کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے ایک بار وہ آیات تلاوت فرمائیں جن میں حضرت ابراہیم علیہ السلام اور حضرت عیسیٰ علیہ السلام کی اپنی امتوں کے حق میں عرضداشت ہے، پھر ہاتھ اٹھا کر میری امت، میری امت فرمایا اور رو پڑے؛ تب اللہ تعالیٰ نے حضرت جبریل علیہ السلام کو بھیجا اور پیغام دیا کہ ہم تمہیں تمہاری امت کے بارے میں راضی کریں گے اور تمہیں رنجیدہ نہیں کریں گے۔ اسی لیے اہلِ علم کہتے ہیں کہ اس آیت کی رضا کا بڑا حصہ امت کی بخشش اور شفاعت سے جڑا ہے۔ غور کرو کہ یہاں عطا کی مقدار بیان نہیں ہوئی، اس کی انتہا بیان ہوئی ہے: "فَتَرْضٰى"۔ عطا کا پیمانہ محبوب کی رضا ٹھہرا۔ یہ وہ اکرام ہے جس کی مثال کسی اور خطاب میں نہیں ملتی۔

شیخِ اکبر کے نزدیک رضا دو طرفہ ہے: بندہ رب سے راضی ہو اور رب بندے سے؛ اور کامل بندے کی رضا وہ ہے جس میں اس کی اپنی چاہت حق کی چاہت میں فنا ہو جائے۔ یہاں ایک عجیب لطف ہے: عام بندوں سے کہا جاتا ہے کہ اپنی مرضی کو رب کی مرضی کے تابع کرو، اور یہاں رب اپنی عطا کو حبیب کی رضا تک پہنچانے کا وعدہ فرماتا ہے۔ اس کا راز یہ ہے کہ حبیب کی رضا خود رب کی رضا کا آئینہ ہے؛ جس کا دل حق کی مرضی سے ہٹ کر کچھ نہ چاہے، اس کی رضا تک پہنچنا عین حق کی مرضی ہے۔ اور حضورِ اکرم صلی اللہ علیہ وسلم کی چاہت کیا ہے؟ امت کی نجات اور مخلوق پر رحمت، کیونکہ آپ کو سارے جہانوں کے لیے رحمت بنا کر بھیجا گیا۔ اس لیے یہ آیت، اہلِ معرفت کے نزدیک، سبقتِ رحمت کا وعدہ بھی ہے جو ایک محبوب ذات کے واسطے سے پوری مخلوق تک پہنچے گا۔ اہلِ ذوق کہتے ہیں کہ سالک کے لیے اس آیت میں تربیت کا ایک دروازہ ہے: وہ عطا کو گننے کے بجائے رضا کو مقصد بنائے۔ جو عطا کے پیچھے بھاگے وہ کبھی سیر نہیں ہوتا، اور جو رضا پا لے اس کے لیے تھوڑا بھی بہت ہے۔ "فَتَرْضٰى" کا فا بتاتا ہے کہ رضا عطا کا نتیجہ بھی ہے اور اس کی منزل بھی؛ عطا راستہ ہے اور رضا ٹھکانا۔

پھر ماضی کی تین نعمتیں یاد دلائی گئیں تاکہ مستقبل کے وعدے پر یقین اور پختہ ہو جائے: اَلَمْ يَجِدْكَ يَتِيْمًا فَاٰوٰى۔ کیا اس نے تمہیں یتیم نہیں پایا، پھر ٹھکانا دیا؟ وَوَجَدَكَ ضَآلًّا فَهَدٰى۔ اور تمہیں راہ کا جویا اور شریعت کی تفصیلات سے ناواقف پایا، تو راہ دکھا دی۔ وَوَجَدَكَ عَآئِلًا فَاَغْنٰى۔ اور تمہیں تنگ دست پایا، تو غنی کر دیا۔ والدِ گرامی ولادت سے پہلے رخصت ہو چکے تھے اور والدۂ ماجدہ بچپن میں؛ پھر دادا عبد المطلب نے، اور ان کے بعد چچا ابو طالب نے کفالت کی؛ یہ "فَاٰوٰى" کی ظاہری صورت ہے۔ اور حضرت خدیجہ رضی اللہ عنہا کی رفاقت، تجارت کی برکت اور سب سے بڑھ کر دل کی بے نیازی "فَاَغْنٰى" کی صورت ہے؛ حدیث کا مفہوم ہے کہ اصل غنا مال کی کثرت نہیں، دل کا غنا ہے۔ ساتویں آیت کے بارے میں ادب کی بات پوری صراحت سے کہنی ضروری ہے: یہاں "ضَآلًّا" کا مطلب گمراہی، خطا یا گناہ ہرگز نہیں۔ اہلِ حق کا عقیدہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نبوت سے پہلے بھی اور بعد میں بھی شرک اور گناہ سے محفوظ تھے، اور قرآن خود گواہی دیتا ہے: مَا ضَلَّ صَاحِبُكُمْ وَمَا غَوٰى۔ یہاں مراد وہی ہے جو دوسری جگہ فرمایا: مَا كُنْتَ تَدْرِيْ مَا الْكِتٰبُ وَلَا الْاِيْمَانُ؛ یعنی وحی سے پہلے کتاب اور ایمان کے احکام کی تفصیل آپ کے علم میں نہ تھی، آپ فطرتِ سلیمہ پر حق کے طالب تھے، اور حق نے وحی کے ذریعے راہ کی پوری تفصیل عطا کر دی۔

اہلِ علم نے "ضَآلًّا" کی جو توجیہات ذکر کیں وہ سب اسی ادب کے دائرے میں ہیں۔ ایک قول یہ ہے کہ بچپن میں آپ مکہ کی گھاٹیوں میں راستہ بھول گئے تھے، اور اللہ تعالیٰ نے آپ کو دادا تک پہنچا دیا؛ یہاں لفظ صرف راستہ نہ ملنے کے معنی میں ہے۔ دوسرا لطیف قول عربی محاورے سے ہے: عرب صحرا میں اکیلے کھڑے درخت کو بھی ضالّہ کہتے تھے، جس سے مسافر راستہ پہچانتے ہیں؛ یعنی حق نے آپ کو اپنی قوم میں یکتا اور تنہا پایا، تو آپ کے ذریعے دوسروں کو راہ دکھائی۔ اور اہلِ ذوق نے کہا کہ عربی میں یہ لفظ محبت کی وارفتگی کے لیے بھی آیا ہے؛ یعنی حق نے آپ کو اپنی طلب میں وارفتہ پایا، تو وصال کی راہ کھول دی۔ پھر یہ بھی دیکھو کہ "فَاٰوٰى"، "فَهَدٰى" اور "فَاَغْنٰى" کے بعد مفعول کی ضمیر نہیں آئی۔ مفسرین کہتے ہیں کہ اس میں عموم کا اشارہ ہے: آپ کو ٹھکانا دیا اور آپ کے ذریعے بے ٹھکانوں کو ٹھکانا دیا؛ آپ کو راہ دکھائی اور آپ کے ذریعے عالم کو؛ آپ کو غنی کیا اور آپ کے ذریعے فقیروں کو۔ شیخِ اکبر کے مکتب میں انسانِ کامل حق کے فیض کا برزخ ہے: جو اس پر اترتا ہے وہ اس سے مخلوق تک بہتا ہے۔ اہلِ تصوف یتیم کے لفظ میں بھی ایک اشارہ دیکھتے ہیں: عرب بے مثل موتی کو درۂ یتیمہ کہتے ہیں؛ آپ وہ یکتا گوہر ہیں جس کی پرورش کسی ظاہری واسطے کے سہارے کے بجائے براہِ راست رب کی نگرانی میں ہوئی۔

تین نعمتوں کے بعد تین حکم آتے ہیں، اور ہر حکم ایک نعمت کا شکر ہے: فَاَمَّا الْيَتِيْمَ فَلَا تَقْهَرْ۔ پس یتیم کو نہ دباؤ؛ وَاَمَّا السَّآئِلَ فَلَا تَنْهَرْ۔ اور مانگنے والے کو نہ جھڑکو؛ وَاَمَّا بِنِعْمَةِ رَبِّكَ فَحَدِّثْ۔ اور اپنے رب کی نعمت کا خوب چرچا کرو۔ ترتیب پر غور کرو: جو خود یتیم رہا اور جسے ٹھکانا ملا، اسے یتیم کی دل شکنی سے روکا گیا؛ جسے تنگ دستی کے بعد غنا ملا، اسے سائل کو جھڑکنے سے روکا گیا؛ اور جسے راہ کی جستجو کے بعد ہدایت ملی، اسے حکم ہوا کہ اس ہدایت کی نعمت کو بیان کرے اور آگے پہنچائے۔ مفسرین نے یہ بھی لکھا کہ "السَّآئِلَ" میں مال مانگنے والا بھی شامل ہے اور علم اور راہ پوچھنے والا بھی؛ اس طرح سائل والا حکم "فَهَدٰى" کے مقابل بھی آتا ہے: جسے طلب پر راہ ملی، وہ طالب کو کبھی نہ لوٹائے۔ "تَقْهَرْ" قہر سے ہے، یعنی کسی کمزور پر غلبہ جتا کر اس کا حق دبانا؛ اور "تَنْهَرْ" نہر سے، یعنی ڈانٹ کر دھتکار دینا۔ حکم صرف ظلم سے نہیں، لہجے کی سختی سے بھی روکتا ہے۔ اور حضورِ اکرم صلی اللہ علیہ وسلم کی سیرت اس کی کامل تفسیر ہے: صحیح روایات کا مفہوم ہے کہ آپ سے جب بھی کچھ مانگا گیا آپ نے انکار نہ فرمایا، اور یتیم کی کفالت کرنے والے کو جنت میں اپنی قربت کی بشارت دی۔

یہاں اس سورت کا اصول کھلتا ہے، اور اسے اصول کے طور پر یاد رکھنا چاہیے: طالب کو کبھی نہ دھتکارا جائے۔ جو حق کی راہ پوچھنے آئے، اس کا سوال خود حق کی طرف ایک قدم ہے، اور اس قدم کو جھڑکی سے روک دینا اس دروازے کو بند کرنا ہے جو رب نے کھولا تھا۔ قرآن نے ایک اور مقام پر اسی ادب کی تعلیم دی: وَاَمَّا مَنْ جَآءَكَ يَسْعٰى، کہ جو دوڑ کر طلب میں آئے اس کی قدر کی جائے۔ شیخِ اکبر کے نزدیک بندے کا کمال تخلق باسماء اللہ ہے، یعنی حق کے اسماء کا رنگ اپنے اخلاق میں لینا؛ اور حق کی سنت یہی ہے کہ وہ کسی طالب کو نہیں لوٹاتا۔ جو ہدایت مانگتا ہے وہ پاتا ہے، اور جو منہ پھیرتا ہے وہ اپنا رخ خود اختیار کرتا ہے، اور حق اسی رخ کو وجود دے دیتا ہے؛ یہی سرِّ قدر ہے کہ حکم محکوم کے حال کے مطابق ہوتا ہے، اور کسی کو زبردستی گمراہ نہیں کیا جاتا۔ پس جس نبی کو "فَهَدٰى" کی نعمت ملی، اسے "فَلَا تَنْهَرْ" کا حکم ملا، تاکہ امت کا ہر عالم اور ہر مرشد جان لے کہ طالب کا سوال، کتنا ہی سادہ یا بار بار کا کیوں نہ ہو، رب کی امانت ہے۔ اہلِ معرفت کہتے ہیں کہ جو استاد طالب کو جھڑکتا ہے وہ دراصل اپنے سینے کی تنگی کا اعلان کرتا ہے؛ اور وسعتِ قلب کی پہلی نشانی یہ ہے کہ سوال سن کر دل کشادہ ہو، تنگ نہیں۔

آخری حکم "فَحَدِّثْ" شکر کی سب سے بلند صورت سکھاتا ہے۔ نعمت کا چرچا فخر نہیں، اعتراف ہے؛ فخر میں بندہ نعمت کو اپنی طرف منسوب کرتا ہے، اور تحدیث میں دینے والے کی طرف۔ اسی لیے فرمایا "بِنِعْمَةِ رَبِّكَ"، اپنی نعمت نہیں فرمایا۔ مفسرین کہتے ہیں کہ سب سے بڑی نعمت نبوت اور قرآن ہے، اس لیے اس حکم کا پہلا مصداق تبلیغ ہے: جو ہدایت ملی اسے بیان کرو۔ شیخِ اکبر کے مکتب میں شکر کا کمال یہ ہے کہ بندہ ہر نعمت میں منعم کو دیکھے، اور نعمت کو ایک تجلی جانے جس میں حق اپنے کرم کا اظہار کر رہا ہے؛ تب نعمت کا بیان دراصل حق کے اسماء کا بیان بن جاتا ہے۔ اور یہی نکتہ سورت کے آغاز سے جا ملتا ہے: جس طرح چاشت اپنی روشنی چھپا نہیں سکتی، اسی طرح وہ دل جس پر نعمت کی صبح طلوع ہو، اسے اپنی ذات میں قید نہیں رکھتا۔ اہلِ ذوق کہتے ہیں کہ تحدیثِ نعمت کا ادب یہ ہے کہ بات ایسے کی جائے کہ سننے والے کا دل رب کی طرف کھنچے، کہنے والے کی طرف نہیں۔ اہلِ مکہ کے قراء سے منقول ہے کہ اس سورت سے آخرِ قرآن تک سورتوں کے اختتام پر تکبیر کہی جائے؛ گویا شکر کا یہ حکم تلاوت میں بھی ایک آواز بن گیا۔ اور اگلی سورت اسی تسلی کو آگے بڑھاتی ہے، جس میں سینے کی کشادگی کا ذکر ہے۔
