امام شافعی رحمۃ اللہ علیہ کا قول منقول ہے، جس کا مفہوم یہ ہے کہ اگر لوگ صرف اسی ایک سورت میں غور کر لیں تو یہ ان کے لیے کافی ہو جائے۔ سورت ایک قسم سے کھلتی ہے: وَالْعَصْرِ۔ “زمانے کی قسم۔” مفسرین نے لفظ عصر کے کئی معنی بیان کیے ہیں۔ ایک قول میں اس سے مراد پورا زمانہ ہے، دن رات کی وہ گردش جس میں انسان کی عمر خرچ ہوتی ہے۔ دوسرے قول میں دن کا آخری حصہ، یعنی سہ پہر، اور اسی نسبت سے نمازِ عصر۔ تیسرے قول میں حضورِ اکرم صلی اللہ علیہ وسلم کا عہدِ مبارک مراد ہے؛ اور صحیح بخاری کی ایک حدیث کا مفہوم یہ ہے کہ پچھلی امتوں کے مقابلے میں اس امت کی مدت ایسی ہے جیسے عصر سے مغرب تک کا وقت۔ اہلِ لغت یاد دلاتے ہیں کہ عصر کی اصل نچوڑنے میں ہے، جیسے کپڑا نچوڑا جاتا ہے یا پھل سے رس نکالا جاتا ہے۔ گویا زمانہ ایک شکنجہ ہے جس میں ہر جان دبائی جا رہی ہے، اور سوال صرف یہ ہے کہ اس سے رس کیا نکلتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں حق تعالیٰ جس چیز کی قسم کھائے وہ ایک آیت ہوتی ہے، اور قسم اس آیت کی طرف دل کو موڑنے کا نام ہے۔ یہ سب معانی ایک دوسرے کی نفی نہیں کرتے؛ ظاہر اپنی جگہ مقدم ہے، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

اب اسی قسم کو اہلِ تصوف کی نگاہ سے دیکھیے۔ ان کے ہاں سالک کا سب سے قیمتی سرمایہ وقت ہے، اور اہلِ تصوف میں مشہور قول ہے کہ صوفی ابن الوقت ہوتا ہے، یعنی وقت کا بیٹا۔ وہ نہ گزرے ہوئے لمحے پر ماتم کرتا ہے، نہ آنے والے کی فکر میں حال کو گنواتا ہے، بلکہ اس لمحے میں حق کی طرف سے جو حال وارد ہے اسی کا حق ادا کرتا ہے۔ شیخِ اکبر کے نزدیک یہ بات محض نصیحت نہیں، وجود کی حقیقت ہے۔ اس مکتب میں عالم ہر آن نئے سرے سے پیدا ہوتا ہے، جسے خلقِ جدید اور تجددِ امثال کہا گیا: ہر لمحہ پچھلی صورت فنا ہوتی ہے اور اس کی مثل نئی صورت وجود پاتی ہے۔ اسی کی گواہی میں یہ آیت پڑھی جاتی ہے: كُلَّ يَوْمٍ هُوَ فِيْ شَاْنٍ۔ “وہ ہر روز ایک نئی شان میں ہے۔” پس ہر لمحہ ایک نئی تخلیق ہے اور ہر نئی تخلیق ایک نیا خطاب۔ اس مکتب میں زمانہ اپنی ذات میں کوئی مستقل شے نہیں، حرکتوں کے درمیان ایک نسبت ہے؛ اس کی قیمت اس سے ہے جو اس میں بھرا جائے۔ اہلِ ذوق کہتے ہیں کہ عصر کی قسم دراصل اس لمحے کی قسم ہے جس میں تم یہ آیت پڑھ رہے ہو؛ اور اس سے آگے ایک مقام ہے جسے انہوں نے ابو الوقت کہا، یعنی وہ بندہ جو وقت کے بہاؤ میں بہتا نہیں بلکہ حق کے ساتھ حاضر رہ کر ہر وقت کو اس کا حکم دیتا ہے۔

عصر کا ایک معنی دن کا ڈھلتا حصہ ہے، اور اس میں ایک اور سبق چھپا ہے۔ سہ پہر میں سورج جھکنے لگتا ہے اور سائے لمبے ہونے لگتے ہیں؛ اہلِ معرفت کہتے ہیں کہ عمر کا سایہ بھی اسی طرح بڑھتا ہے اور آدمی کو خبر نہیں ہوتی کہ دن کتنا باقی ہے۔ بعض اسلاف سے منقول ہے، جس کا مفہوم یہ ہے کہ انہوں نے اس سورت کا مطلب ایک برف بیچنے والے سے سمجھا جو بازار میں پکار رہا تھا کہ رحم کرو اس پر جس کا سرمایہ پگھل رہا ہے۔ برف بکے یا نہ بکے، پگھلتی رہتی ہے؛ عمر بھی کام آئے یا نہ آئے، گھٹتی رہتی ہے۔ اسی پس منظر میں قسم کا جواب آتا ہے: اِنَّ الْاِنْسَانَ لَفِيْ خُسْرٍ۔ “بے شک انسان خسارے میں ہے۔” یہاں غور کی بات یہ ہے کہ خسارے کے لیے کچھ کرنا نہیں پڑتا؛ صرف کھڑے رہنا کافی ہے۔ سرمایہ وقت ہے، اور وقت خود بخود گزر رہا ہے۔ پس خسارہ کوئی ایسی سزا نہیں جو باہر سے کسی پر مسلط کی گئی ہو؛ یہ اس سرمائے کا فطری انجام ہے جو کسی تجارت میں نہ لگایا جائے۔ شیخِ اکبر کے مکتب میں ہر سانس نفَسِ رحمانی کی ایک موج ہے جو بندے کو عطا ہوتی ہے؛ جو سانس حق کے ساتھ گزرا وہ محفوظ ہو گیا، اور جو غفلت میں گزرا وہ بخشش تو تھا، مگر قبول نہ کیا گیا۔

اس آیت میں تاکید پر تاکید ہے: اِنَّ، پھر لام، پھر فِيْ۔ انسان خسارے کے کنارے پر نہیں، اس کے اندر ڈوبا ہوا ہے؛ اور خُسْر کی تنوین اس خسارے کی بڑائی بتاتی ہے۔ شیخِ اکبر کے نزدیک انسان دو طرح کا ہے: ایک انسانِ کامل، جو اسماءِ الٰہیہ کا جامع آئینہ بنا اور اپنی پوری استعداد کو ظاہر کر گیا؛ اور ایک وہ جسے اس مکتب میں انسانِ حیوان کہا گیا، جس کے پاس صورتِ انسانی تو ہے مگر وہ اسی صورت پر رک گیا۔ اہلِ معرفت کہتے ہیں کہ اصل خسارہ یہ ہے کہ آدمی کو جامعیت کی استعداد ملی، سارے اسماء کو قبول کرنے کی صلاحیت ملی، اور اس نے اسے صرف کھانے، سونے اور جمع کرنے پر خرچ کر دیا۔ قرآن نے بندے اور رب کے معاملے کو بار بار تجارت کہا ہے: اِنَّ اللّٰهَ اشْتَرٰى مِنَ الْمُؤْمِنِيْنَ اَنْفُسَهُمْ وَاَمْوَالَهُمْ۔ “بے شک اللہ نے مومنوں سے ان کی جانیں اور ان کے مال خرید لیے ہیں۔” جو نفس اللہ سے سودا نہیں کرتا وہ اپنے ہی ہاتھ میں ایک ایسا مال لیے بیٹھا ہے جس کی قیمت ہر لمحہ گر رہی ہے۔ وہ اپنی خواہش سے سودا کرتا ہے، اور خواہش وہ خریدار ہے جو قیمت ادا نہیں کرتا۔ اسی لیے یہ خسارہ ہر انسان کی ابتدائی حالت ہے، کوئی ٹھپا نہیں؛ اور اس سے نکلنے کا دروازہ اگلی آیت میں کھلتا ہے۔

اب وہ استثنا آتا ہے جس پر پوری سورت کا وزن ہے: اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَتَوَاصَوْا بِالْحَقِّ ۙ وَتَوَاصَوْا بِالصَّبْرِ۔ “سوائے ان لوگوں کے جو ایمان لائے، اور نیک عمل کیے، اور ایک دوسرے کو حق کی تلقین کی، اور ایک دوسرے کو صبر کی تلقین کی۔” یہی اس سورت کا اصول ہے، اور ہدایت و گمراہی کے باب میں اس سلسلے کا بنیادی اصول بھی۔ چاروں فعل بندوں کی طرف منسوب ہیں: ایمان لائے، عمل کیا، تلقین کی، تلقین کی۔ نجات کی چاروں شرطیں بندے کی اپنی ہیں، کوئی ان کے اور اس کے درمیان حائل نہیں۔ استثنا کسی نسل، کسی نام یا کسی خاندان کی فہرست نہیں، صفات کا بیان ہے؛ جو یہ صفات اختیار کرے وہ اس میں داخل ہے۔ کسی کو زبردستی خسارے میں نہیں ڈالا گیا، اور جو نکلنا چاہے اس کے لیے راستہ بند نہیں۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے؛ بندہ جو رخ اختیار کرتا ہے حق تعالیٰ اسی رخ کو وجود عطا فرماتا ہے۔ وہ چاہتا تو سب کو مجبور کر کے راہ پر لے آتا: وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ۔ “اور اگر وہ چاہتا تو تم سب کو ہدایت دے دیتا۔” مگر اس نے انسان کو اختیار کی عزت دی، اور اس عزت کا تقاضا یہ ہے کہ خسارے سے نکلنا بندے کا اپنا قدم ہو۔

پہلی دو شرطیں ایک دوسرے سے جڑی ہوئی ہیں۔ اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ۔ “سوائے ان کے جو ایمان لائے اور نیک عمل کیے۔” ایمان پہلے ہے، کیونکہ عمل بغیر ایمان کے بے روح جسم ہے؛ اور عمل ساتھ ہے، کیونکہ ایمان بغیر عمل کے بے پھل درخت ہے۔ شیخِ اکبر کے مکتب میں ایمان وہ نور ہے جو عقل کی حد سے آگے کی خبر کو قبول کر لیتا ہے، اور کشف جب آتا ہے تو ایمان کی تصدیق بن کر آتا ہے، اس کی جگہ لینے نہیں۔ اب لفظ الصّٰلِحٰت پر ٹھہریے۔ اس کی اصل صلاحیت اور مناسبت میں ہے: صالح وہ ہے جو اپنی جگہ کے لائق ہو۔ اہلِ معرفت کہتے ہیں کہ نیک عمل وہ ہے جو اپنے وقت کے مناسب ہو؛ نماز اپنے وقت میں، شکر نعمت کے وقت، صبر مصیبت کے وقت، توبہ لغزش کے فوراً بعد۔ یوں یہ لفظ سورت کی پہلی آیت سے جا ملتا ہے۔ عصر کی قسم کھا کر بتایا گیا کہ وقت سرمایہ ہے، اور الصّٰلِحٰت کہہ کر بتایا گیا کہ ہر وقت کا اپنا عمل ہے جو اسی کے لائق ہے۔ ابن الوقت ہونے کا یہی مطلب ہے کہ بندہ حکمِ وقت پہچانے اور اسی کو ادا کرے، نہ وہ جو گزر گیا اور نہ وہ جو ابھی آیا نہیں۔ اور جمع کا صیغہ بتاتا ہے کہ ایک عمل پر قناعت نہیں؛ ہر لمحہ نیا خلق ہے تو ہر لمحے کا صالح عمل بھی نیا ہے۔

تیسری شرط پر آ کر سورت فرد سے جماعت کی طرف مڑتی ہے: وَتَوَاصَوْا بِالْحَقِّ۔ “اور ایک دوسرے کو حق کی تلقین کی۔” تَوَاصَوْا باہمی عمل کا صیغہ ہے؛ ہر ایک تلقین کرنے والا بھی ہے اور تلقین قبول کرنے والا بھی۔ اس میں کوئی ہمیشہ کا واعظ نہیں جو خود سننے سے بے نیاز ہو، اور کوئی ہمیشہ کا سامع نہیں جس کی بات سنی نہ جائے۔ اہلِ معرفت کہتے ہیں کہ یہ راستہ اکیلے طے نہیں ہوتا؛ صحبت اس کا حصہ ہے۔ شیخِ اکبر کے مکتب میں ہر بندہ کسی اسم کا مظہر ہے، اور جو ایک میں کم ہے وہ دوسرے میں نمایاں ہے؛ اہلِ ایمان کی جماعت مل کر اس جامعیت کے قریب پہنچتی ہے جو ایک فرد میں مشکل ہے۔ پھر لفظ الْحَقّ پر غور کیجیے۔ ظاہر میں اس سے مراد سچائی، دین اور ثابت بات ہے؛ اور اہلِ ذوق کے نزدیک الحق اللہ کا نام بھی ہے، پس ایک دوسرے کو حق کی وصیت کرنا ایک دوسرے کو حق تعالیٰ کی یاد دلانا بھی ہے۔ سورۂ البلد میں یہی مضمون ایک اور رنگ میں آیا: ثُمَّ كَانَ مِنَ الَّذِيْنَ اٰمَنُوْا وَتَوَاصَوْا بِالصَّبْرِ وَتَوَاصَوْا بِالْمَرْحَمَةِ۔ “پھر وہ ان لوگوں میں سے ہو جو ایمان لائے اور ایک دوسرے کو صبر کی اور ایک دوسرے کو رحم کی تلقین کی۔” وہاں صبر کے ساتھ رحم ہے، یہاں صبر کے ساتھ حق؛ دونوں کو ملا کر پڑھیے تو سبق مکمل ہوتا ہے: حق کی بات رحم کے لہجے میں اور صبر کے ساتھ۔

آخری شرط سب سے آخر میں اس لیے آئی کہ اس کے بغیر باقی تین ٹھہرتی نہیں: وَتَوَاصَوْا بِالصَّبْرِ۔ “اور ایک دوسرے کو صبر کی تلقین کی۔” جو حق کہتا ہے اسے مخالفت ملتی ہے، اور جو حق سنتا ہے اسے اپنی خواہش چھوڑنی پڑتی ہے؛ دونوں کو صبر درکار ہے۔ صبر کے بارے میں شیخِ اکبر نے فصوص الحکم کی حکمتِ غیبیہ میں، حضرت ایوب علیہ السلام کے ذکر میں، جو بات بیان کی اس کا مفہوم یہ ہے کہ صبر یہ نہیں کہ بندہ اپنی تکلیف اللہ کے حضور بھی نہ کہے؛ صبر یہ ہے کہ غیر اللہ سے شکوہ نہ کرے۔ حضرت ایوب علیہ السلام نے اپنے رب ہی سے عرض کیا: اَنِّيْ مَسَّنِيَ الضُّرُّ وَاَنْتَ اَرْحَمُ الرّٰحِمِيْنَ۔ “مجھے تکلیف پہنچی ہے اور تو سب رحم کرنے والوں سے بڑھ کر رحم کرنے والا ہے۔” اور قرآن نے انہیں صابر کہا۔ امام طبرانی نے روایت کیا، جس کا مفہوم یہ ہے کہ صحابہ کرام رضی اللہ عنہم میں سے دو حضرات جب ملتے تو جدا ہونے سے پہلے ایک دوسرے کو یہ سورت سناتے، پھر سلام کر کے رخصت ہوتے۔ گویا ہر ملاقات کا اختتام اسی وصیت پر ہوتا تھا۔ اہلِ معرفت کہتے ہیں کہ سورت وقت سے شروع ہوئی اور صبر پر ختم ہوئی، اور یہ ترتیب خود ایک تعلیم ہے: وقت کے ساتھ بندے کا صحیح تعلق صبر ہے۔ جو صبر سے ہر لمحے کا حق ادا کرتا رہے، اس کے لیے وَالْعَصْرِ۔ “زمانے کی قسم” خسارے کی گواہی نہیں رہتی، کمائی کی گواہی بن جاتی ہے۔
