وَنُوْحًا اِذْ نَادٰى مِنْ قَبْلُ فَاسْتَجَبْنَا لَهٗ فَنَجَّيْنٰهُ وَاَهْلَهٗ مِنَ الْكَرْبِ الْعَظِيْمِ۔ اور نوح کو یاد کرو جب اس نے اس سے پہلے پکارا، تو ہم نے اس کی پکار قبول کی اور اسے اور اس کے گھر والوں کو بڑی بے چینی سے نجات دی۔ وَنَصَرْنٰهُ مِنَ الْقَوْمِ الَّذِيْنَ كَذَّبُوْا بِاٰيٰتِنَا ۭ اِنَّهُمْ كَانُوْا قَوْمَ سَوْءٍ فَاَغْرَقْنٰهُمْ اَجْمَعِيْنَ۔ اور ان لوگوں کے مقابلے میں اس کی مدد کی جنہوں نے ہماری آیتوں کو جھٹلایا تھا، بے شک وہ برے لوگ تھے تو ہم نے ان سب کو غرق کر دیا۔ یہ رکوع نداؤں کا رکوع ہے: ایک کے بعد ایک نبی پکارتا ہے، اور تقریباً ہر پکار کے بعد ایک ہی جواب آتا ہے کہ ہم نے قبول کیا۔ اہلِ معرفت کے نزدیک "من قبل" میں یہ اشارہ ہے کہ حضرت نوح علیہ السلام کی پکار زمانے میں بھی پہلے تھی اور صبر کی طوالت میں بھی سب سے آگے؛ صدیوں کی دعوت کے بعد جو ندا اٹھی وہ بے صبری نہیں، صبر کا نچوڑ تھی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ نے فصوص الحکم میں ان کے باب کو حکمتِ سبوحیہ کا نام دیا، جس کا مفہوم یہ ہے کہ حضرت نوح علیہ السلام کی دعوت کا غالب رنگ تنزیہ تھا، یعنی حق کو ان تمام صورتوں سے پاک ماننا جنہیں قوم نے معبود ٹھہرا لیا تھا، اور کامل معرفت تنزیہ و تشبیہ کے جمع میں ہے۔ رہی قوم کی غرقابی، تو آیت خود سبب بیان کرتی ہے: انہوں نے آیات کو جھٹلایا اور وہ برائی کو اپنا حال بنا چکے تھے۔ حکم ان کے اپنے حال کے مطابق صادر ہوا؛ یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہوتا ہے، اور حق کسی کو زبردستی ہلاکت کی راہ پر نہیں ڈالتا۔ اہلِ تاویل کہتے ہیں کہ کشتی شریعت کی صورت ہے اور طوفان احوال کا سمندر؛ جو سوار ہوا وہ پار لگا، اور جس نے اپنی تدبیر پر بھروسا کیا وہ اپنی ہی تدبیر میں ڈوبا۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ واقعہ اپنی جگہ حق ہے۔

وَدَاوٗدَ وَسُلَيْمٰنَ اِذْ يَحْكُمٰنِ فِي الْحَرْثِ اِذْ نَفَشَتْ فِيْهِ غَنَمُ الْقَوْمِ ۚ وَكُنَّا لِحُكْمِهِمْ شٰهِدِيْنَ۔ اور داؤد اور سلیمان کو یاد کرو جب وہ دونوں ایک کھیتی کے بارے میں فیصلہ کر رہے تھے جس میں رات کو کچھ لوگوں کی بکریاں جا پڑی تھیں، اور ہم ان کے فیصلے کو دیکھ رہے تھے۔ فَفَهَّمْنٰهَا سُلَيْمٰنَ ۚ وَكُلًّا اٰتَيْنَا حُكْمًا وَّعِلْمًا۔ تو ہم نے وہ معاملہ سلیمان کو سمجھا دیا، اور دونوں ہی کو ہم نے حکم اور علم عطا کیا تھا۔ ظاہر میں اس مقدمے میں حضرت سلیمان علیہ السلام کی رائے راجح ٹھہری، مگر قرآن فوراً ادب کی حد کھینچ دیتا ہے کہ حکم و علم دونوں کو ملا تھا؛ حضرت داؤد علیہ السلام کے مرتبے میں کوئی کمی نہیں۔ اہلِ معرفت کے نزدیک "سمجھا دیا" کا صیغہ بتاتا ہے کہ فہم کمائی نہیں، عطا ہے؛ ایک ہی واقعے میں حق جس قلب پر جو پہلو کھول دے۔ شیخِ اکبر کے مکتب میں ہر صحیح اجتہاد کسی اسمِ الٰہی کا مظہر ہے، اسی لیے فیصلے کو دونوں کی طرف منسوب کر کے حق نے خود کو دونوں پر شاہد کہا۔ پھر فرمایا: وَّسَخَّرْنَا مَعَ دَاوٗدَ الْجِبَالَ يُسَبِّحْنَ وَالطَّيْرَ ۭ وَكُنَّا فٰعِلِيْنَ۔ اور ہم نے پہاڑوں اور پرندوں کو داؤد کے ساتھ مسخر کر دیا کہ وہ تسبیح کرتے تھے، اور یہ سب کرنے والے ہم ہی تھے۔ سورۂ بنی اسرائیل میں گزرا: وَاِنْ مِّنْ شَيْءٍ اِلَّا يُسَبِّحُ بِحَمْدِهٖ۔ شیخِ اکبر کے نزدیک ہر شے زندہ اور تسبیح خواں ہے؛ حضرت داؤد علیہ السلام کی آواز نے وہ پوشیدہ تسبیح سنا دی جو پہاڑوں میں پہلے سے جاری تھی۔ فصوص میں ان کا باب حکمتِ وجودیہ ہے، جس کا مفہوم یہ ہے کہ انہیں جو کچھ ملا وہ خالص وہب اور عطا تھا، اور "ہم ہی کرنے والے تھے" اسی عطا کی مہر ہے۔

وَعَلَّمْنٰهُ صَنْعَةَ لَبُوْسٍ لَّكُمْ لِتُحْصِنَكُمْ مِّنْ بَاْسِكُمْ ۚ فَهَلْ اَنْتُمْ شٰكِرُوْنَ۔ اور ہم نے اسے تمہارے لیے زرہ بنانے کی صنعت سکھائی تاکہ وہ تمہیں تمہاری لڑائی کی سختی سے بچائے، تو کیا تم شکر کرنے والے ہو؟ وَلِسُلَيْمٰنَ الرِّيْحَ عَاصِفَةً تَجْرِيْ بِاَمْرِهٖۤ اِلَى الْاَرْضِ الَّتِيْ بٰرَكْنَا فِيْهَا ۭ وَكُنَّا بِكُلِّ شَيْءٍ عٰلِمِيْنَ۔ اور سلیمان کے لیے تند ہوا مسخر کی جو اس کے حکم سے اس سرزمین کی طرف چلتی تھی جس میں ہم نے برکت رکھی، اور آیت اس پر ختم ہوتی ہے کہ ہم ہر چیز کو جاننے والے تھے۔ وَمِنَ الشَّيٰطِيْنِ مَنْ يَّغُوْصُوْنَ لَهٗ وَيَعْمَلُوْنَ عَمَلًا دُوْنَ ذٰلِكَ ۚ وَكُنَّا لَهُمْ حٰفِظِيْنَ۔ اور سرکش جنوں میں سے کچھ اس کے لیے غوطے لگاتے اور اس کے سوا اور کام بھی کرتے، اور ہم ہی ان کے نگہبان تھے۔ اہلِ ذوق کہتے ہیں کہ ہنر بھی تعلیمِ الٰہی ہے، اسی لیے صنعت کا ذکر شکر کے سوال پر ختم ہوا؛ جو ہاتھ کی مہارت کو اپنی ملکیت سمجھے اس نے عطا کو چھپا لیا۔ زرہ کو وہ تقویٰ کی تمثیل کہتے ہیں جو انسان کو خود اس کی اپنی سختی سے بچاتی ہے۔ حضرت سلیمان علیہ السلام کا باب فصوص میں حکمتِ رحمانیہ ہے، جس کا مفہوم یہ ہے کہ ان کے خط کا آغاز رحمٰن و رحیم کے نام سے ہوا اور اس میں دو رحمتوں کا راز ہے: ایک رحمتِ امتنان جو محض عطا ہے، دوسری رحمتِ وجوب جو عمل پر وعدہ ہے؛ اور ان کی سلطنت پہلی کی جلوہ گاہ تھی۔ اس مکتب میں ساری کائنات تسخیرِ متبادل میں بندھی ہے کہ ہر چیز ضرورت کے رشتے سے دوسری کی خدمت میں ہے، مگر حضرت سلیمان علیہ السلام کو تسخیر بالامر ملی؛ پھر بھی ہوا کا چلنا ان کے حکم سے تھا اور ان کا حکم حق کے حکم سے۔ اہلِ تاویل جنوں کو نفس کی بے چین قوتوں کی تمثیل بھی بناتے ہیں جو قلب کے حکم میں آ کر معانی کے موتی نکالتی ہیں؛ یہ ظاہر کی نفی نہیں، اس پر ایک اضافی نظر ہے۔

وَاَيُّوْبَ اِذْ نَادٰى رَبَّهٗۤ اَنِّيْ مَسَّنِيَ الضُّرُّ وَاَنْتَ اَرْحَمُ الرّٰحِمِيْنَ۔ اور ایوب کو یاد کرو جب اس نے اپنے رب کو پکارا کہ مجھے تکلیف نے آ چھوا ہے، اور تو سب رحم کرنے والوں سے بڑھ کر رحم کرنے والا ہے۔ فَاسْتَجَبْنَا لَهٗ فَكَشَفْنَا مَا بِهٖ مِنْ ضُرٍّ وَّاٰتَيْنٰهُ اَهْلَهٗ وَمِثْلَهُمْ مَّعَهُمْ رَحْمَةً مِّنْ عِنْدِنَا وَذِكْرٰى لِلْعٰبِدِيْنَ۔ تو ہم نے اس کی پکار قبول کی، اس کی تکلیف دور کر دی، اور اسے اس کے گھر والے دیے اور ان کے ساتھ اتنے ہی اور، اپنی طرف سے رحمت کے طور پر اور عبادت گزاروں کے لیے یاد دہانی کے طور پر۔ اس ندا کا ادب دیکھو: حضرت ایوب علیہ السلام نے کچھ مانگا نہیں، صرف اپنا حال عرض کیا اور اپنے رب کی رحمت کا نام لیا۔ کریم کے دروازے پر حال کا بیان ہی سب سے مؤدب سوال ہے۔ "چھوا" کا لفظ بتاتا ہے کہ تکلیف نے انہیں چھوا، مغلوب نہیں کیا؛ اور اہلِ سنت کے نزدیک انبیاء ایسی بیماری سے محفوظ ہیں جو لوگوں کو ان سے دور کر دے۔ اللہ سے شکایت بے صبری نہیں، جیسے حضرت یعقوب علیہ السلام نے سورۂ یوسف میں کہا: اِنَّمَاۤ اَشْكُوْا بَثِّيْ وَحُزْنِيْۤ اِلَى اللّٰهِ۔ فصوص میں ان کا باب حکمتِ غیبیہ ہے، جس کا مفہوم یہ ہے کہ صبر غیر اللہ سے شکایت روکنے کا نام ہے، اللہ سے شکایت روکنے کا نہیں؛ اور اس مکتب میں تکلیف کے ہٹنے کی التجا نہ کرنا گویا قہرِ الٰہی کے سامنے اپنی برداشت پر اترانا ہے۔ اہلِ معرفت یہ بھی کہتے ہیں کہ دعا کا ادب حاجت کے مناسب نام سے پکارنا ہے: درد کے لیے ارحم الراحمین۔ اور جواب "اپنی طرف سے" آیا، یعنی سببوں کے راستے سے نہیں، بارگاہِ خاص سے؛ پھر "عبادت گزاروں کے لیے یاد دہانی" کہہ کر یہ دروازہ ہر بندے کے لیے کھول دیا۔

وَاِسْمٰعِيْلَ وَاِدْرِيْسَ وَذَا الْكِفْلِ ۭ كُلٌّ مِّنَ الصّٰبِرِيْنَ۔ اور اسماعیل اور ادریس اور ذوالکفل کو یاد کرو، یہ سب صبر کرنے والوں میں سے تھے۔ وَاَدْخَلْنٰهُمْ فِيْ رَحْمَتِنَا ۭ اِنَّهُمْ مِّنَ الصّٰلِحِيْنَ۔ اور ہم نے انہیں اپنی رحمت میں داخل کیا، بے شک وہ نیکوکاروں میں سے تھے۔ تین نام ایک سانس میں آئے اور سب پر ایک ہی صفت رکھی گئی، تاکہ کوئی ان میں درجہ بندی کر کے کسی کو کم نہ سمجھے۔ حضرت اسماعیل علیہ السلام کا باب فصوص میں حکمتِ علیہ ہے، جس کا مفہوم یہ ہے کہ وہ اپنے رب کے نزدیک پسندیدہ تھے، اور ہر بندہ اپنے ربِّ خاص کے نزدیک اسی طرح مرضی ہے جس اسم کا وہ مظہر ہے۔ حضرت ادریس علیہ السلام کا باب حکمتِ قدوسیہ ہے، جس کا مفہوم یہ ہے کہ ان کی بلندی مکان کی بھی تھی اور مکانت کی بھی، اور اصل رفعت مرتبے کی ہے۔ حضرت ذوالکفل علیہ السلام کے بارے میں مفسرین کہتے ہیں کہ انہوں نے ایک ذمہ داری اپنے سر لی اور اسے پورا نبھایا؛ اہلِ ذوق کہتے ہیں کہ عہد پر قائم رہنا صبر کی سب سے خاموش صورت ہے۔ پھر "انہیں اپنی رحمت میں داخل کیا" پر غور کرو: یہ نہیں فرمایا کہ رحمت ان میں داخل کی، بلکہ وہ رحمت میں داخل ہوئے، جیسے کوئی گھر میں داخل ہو کر اس کی چھت کے نیچے آ جائے۔ شیخِ اکبر کے نزدیک رحمت ہر چیز سے پہلے ہے اور ہر چیز کو گھیرے ہوئے ہے، اور صبر وہ دروازہ ہے جس سے بندہ اس گھیر کو شعور کے ساتھ پاتا ہے۔

وَذَا النُّوْنِ اِذْ ذَّهَبَ مُغَاضِبًا فَظَنَّ اَنْ لَّنْ نَّقْدِرَ عَلَيْهِ فَنَادٰى فِي الظُّلُمٰتِ اَنْ لَّاۤ اِلٰهَ اِلَّاۤ اَنْتَ سُبْحٰنَكَ ۖ اِنِّيْ كُنْتُ مِنَ الظّٰلِمِيْنَ۔ اور مچھلی والے کو یاد کرو جب وہ اپنی قوم سے ناراض ہو کر چل دیا اور سمجھا کہ ہم اس پر تنگی نہ کریں گے، پھر اس نے اندھیروں میں پکارا کہ تیرے سوا کوئی معبود نہیں، تو پاک ہے، بے شک میں ہی قصور واروں میں سے تھا۔ فَاسْتَجَبْنَا لَهٗ ۙ وَنَجَّيْنٰهُ مِنَ الْغَمِّ ۭ وَكَذٰلِكَ نُـْۨجِي الْمُؤْمِنِيْنَ۔ تو ہم نے اس کی پکار قبول کی اور اسے غم سے نجات دی، اور اسی طرح ہم ایمان والوں کو نجات دیتے ہیں۔ پہلے ادب کی بات: یہاں "نقدر" قدرت سے نہیں، قَدَر بمعنی تنگی سے ہے، جیسے رزق کی تنگی کے لیے یہی مادہ آتا ہے؛ مفسرین یہی بیان کرتے ہیں کہ حضرت یونس علیہ السلام نے یہ سمجھا کہ قوم کو چھوڑ کر نکلنے پر ان کی گرفت تنگ نہ کی جائے گی۔ کسی نبی کے دل میں اللہ کی قدرت پر شک کا گزر ممکن ہی نہیں۔ اور ان کی ناراضی قوم پر تھی، اللہ کی خاطر، رب پر ہرگز نہیں۔ اندھیرے تہ بہ تہ تھے: رات، سمندر اور مچھلی کا پیٹ۔ ان اندھیروں میں جو ذکر نکلا وہ کامل ذکر ہے: پہلے توحید، پھر تسبیح، پھر اپنے قصور کا اقرار؛ پاکی حق کی طرف اور کوتاہی اپنی طرف، یہی عبدیت کا کمال ہے۔ فصوص میں ان کا باب حکمتِ نفسیہ ہے، جس کا مفہوم یہ ہے کہ انسانی نشأت کی حرمت بہت بڑی ہے اور حق اس کی حفاظت کو عزیز رکھتا ہے؛ سو مچھلی کا پیٹ قید نہیں، حفاظت کا گھر بنا۔ حدیث کا مفہوم ہے کہ ذوالنون کی یہ دعا جو مسلمان بھی کسی حاجت میں پڑھے، اللہ اسے قبول فرماتا ہے۔ اور "اسی طرح ہم ایمان والوں کو نجات دیتے ہیں" کہہ کر یہ دروازہ ہر مومن کے لیے کھلا رکھا گیا؛ جو ڈھونڈتا ہے وہ پاتا ہے۔

وَزَكَرِيَّاۤ اِذْ نَادٰى رَبَّهٗ رَبِّ لَا تَذَرْنِيْ فَرْدًا وَّاَنْتَ خَيْرُ الْوٰرِثِيْنَ۔ اور زکریا کو یاد کرو جب اس نے اپنے رب کو پکارا: اے میرے رب! مجھے اکیلا نہ چھوڑ، اور تو سب سے بہتر وارث ہے۔ فَاسْتَجَبْنَا لَهٗ ۙ وَوَهَبْنَا لَهٗ يَحْيٰى وَاَصْلَحْنَا لَهٗ زَوْجَهٗ ۭ اِنَّهُمْ كَانُوْا يُسٰرِعُوْنَ فِي الْخَيْرٰتِ وَيَدْعُوْنَنَا رَغَبًا وَّرَهَبًا ۭ وَكَانُوْا لَنَا خٰشِعِيْنَ۔ تو ہم نے اس کی پکار قبول کی، اسے یحییٰ عطا کیا اور اس کی زوجہ کو اس کے لیے سنوار دیا؛ بے شک وہ سب نیکیوں میں جلدی کرتے تھے اور ہمیں شوق اور خوف کے ساتھ پکارتے تھے، اور ہمارے آگے عاجزی کرنے والے تھے۔ حضرت زکریا علیہ السلام وارث مانگتے ہیں اور اسی سانس میں حق کو بہترین وارث کہتے ہیں؛ اہلِ معرفت کے نزدیک یہ سبب مانگنے اور مسبب کو دیکھنے کا جمع ہے، دعا توحید کے منافی نہیں۔ سورۂ مریم میں اسی ندا کی کیفیت بیان ہوئی تھی: اِذْ نَادٰى رَبَّهٗ نِدَآءً خَفِيًّا۔ فصوص میں ان کا باب حکمتِ مالکیہ ہے، جس کا مفہوم یہ ہے کہ رحمتِ الٰہی وجود میں بھی ہر چیز کو گھیرے ہوئے ہے اور حکم میں بھی۔ "اس کی زوجہ کو سنوار دیا" میں ان محترم خاتون کی عزت ہے کہ حق نے خود ان کے حال کی اصلاح اپنے ذمے لی، اور پھر "وہ سب" کہہ کر پورے گھرانے کو ایک صف میں رکھا۔ شوق اور خوف دو پر ہیں: ایک جمال کی طرف کھینچتا ہے، دوسرا جلال کے آگے جھکاتا ہے؛ یہی قبض و بسط کا توازن ہے، اور دونوں کے بیچ خشوع وہ آشیانہ ہے جہاں دل ٹھہرتا ہے۔

وَالَّتِيْۤ اَحْصَنَتْ فَرْجَهَا فَنَفَخْنَا فِيْهَا مِنْ رُّوْحِنَا وَجَعَلْنٰهَا وَابْنَهَاۤ اٰيَةً لِّلْعٰلَمِيْنَ۔ اور اس پاک دامن خاتون کو یاد کرو جس نے اپنی عصمت کی حفاظت کی، تو ہم نے اس میں اپنی روح میں سے پھونکا اور اسے اور اس کے بیٹے کو جہانوں کے لیے ایک نشانی بنا دیا۔ اِنَّ هٰذِهٖۤ اُمَّتُكُمْ اُمَّةً وَّاحِدَةً ۖ وَّاَنَا رَبُّكُمْ فَاعْبُدُوْنِ۔ بے شک یہ تمہاری امت ایک ہی امت ہے، اور میں تمہارا رب ہوں، تو میری عبادت کرو۔ وَتَقَطَّعُوْۤا اَمْرَهُمْ بَيْنَهُمْ ۭ كُلٌّ اِلَيْنَا رٰجِعُوْنَ۔ اور لوگوں نے اپنے دین کو آپس میں ٹکڑے ٹکڑے کر لیا، سب ہماری ہی طرف لوٹنے والے ہیں۔ انبیاء کی اس صف میں حضرت مریم علیہا السلام کا ذکر ان کی پاک دامنی کے وصف سے آیا، اور یہی وصف ان کا تاج ہے۔ "اپنی روح" میں اضافت تشریف کی ہے، جیسے "اللہ کا گھر"؛ روح کو اپنی طرف منسوب کر کے حق نے اس کی عزت بتائی۔ حضرت عیسیٰ علیہ السلام کا باب فصوص میں حکمتِ نبویہ ہے، جس کا مفہوم یہ ہے کہ ان کا وجود روح کی پھونک سے تھا، اسی لیے ان کی پھونک اللہ کے اذن سے زندگی بنتی تھی۔ ماں اور بیٹے کو ایک ہی "نشانی" کہا، دو نہیں: معجزہ دونوں میں ایک ہے۔ پھر تمام انبیاء کے ذکر کے بعد اعلان ہوا کہ یہ سب ایک امت ہیں: ایک توحید، ایک رب، شریعتیں مختلف۔ ٹکڑے ٹکڑے کرنا لوگوں کا اپنا فعل ہے، آیت فعل کو انہی کی طرف منسوب کرتی ہے؛ حق نے انہیں نہیں بانٹا، انہوں نے خود اپنا رخ بانٹا، اور حکم ان کے اپنے رخ کے مطابق ہوا۔ اور سب کا لوٹنا اسی کی طرف ہے؛ شیخِ اکبر کے نزدیک ہر راہ کی انتہا وہی ہے، مگر پہنچنے والے کا حال اسی رخ کے مطابق ہوگا جو اس نے خود چنا۔
