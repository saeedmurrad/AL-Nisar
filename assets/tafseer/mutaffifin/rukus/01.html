یہ سورت ایک بازار سے شروع ہوتی ہے اور ایک چشمے پر ختم ہوتی ہے، اور اس کا پہلا لفظ ہی ایک للکار ہے: وَيْلٌ لِّلْمُطَفِّفِيْنَ ۙ۔ تباہی ہے ناپ تول میں کمی کرنے والوں کے لیے۔ حضرت عبداللہ بن عباس رضی اللہ عنہما سے بالمعنی منقول ہے کہ جب حضورِ اکرم صلی اللہ علیہ وسلم مدینہ تشریف لائے تو وہاں کے لوگ ناپ تول میں بدترین لوگوں میں سے تھے؛ یہ سورت اتری تو انہوں نے اپنا ناپ درست کر لیا۔ ظاہری حکم صاف اور قطعی ہے: تجارت میں پورا ناپنا اور پورا تولنا فرض ہے، اور اس میں کمی حرام اور بندوں کا حق مارنا ہے، جس کی تلافی صرف ندامت سے نہیں، حق لوٹانے سے ہوتی ہے۔ لفظ مطفف طفیف سے ہے، یعنی معمولی اور حقیر مقدار؛ ناپ میں کمی کرنے والا عموماً تھوڑا ہی مارتا ہے، اتنا کہ کسی کو محسوس نہ ہو۔ قرآن نے اسی تھوڑے پر "وَيْلٌ" کا لفظ رکھا، تاکہ معلوم ہو کہ حق کے ترازو میں چوری کی مقدار نہیں، چور کا رخ تولا جاتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ترازو اسمِ عدل کا مظہر ہے، اور آسمان و زمین اسی میزان پر قائم ہیں؛ جس نے ترازو میں ہاتھ کی صفائی دکھائی، اس نے دراصل اس اسم کے مظہر کو بگاڑا جس پر کائنات کا توازن ٹکا ہے۔ اسی لیے یہ معمولی سی کمی ایک بڑی بے ادبی ہے، اور چند دانوں کی چوری پوری میزان سے بغاوت۔

پھر قرآن خود اس کمی کی صورت بیان کرتا ہے: الَّذِيْنَ اِذَا اكْتَالُوْا عَلَى النَّاسِ يَسْتَوْفُوْنَ ۖ۔ وہ کہ جب لوگوں سے ناپ کر لیتے ہیں تو پورا لیتے ہیں۔ وَاِذَا كَالُوْهُمْ اَوْ وَّزَنُوْهُمْ يُخْسِرُوْنَ ۭ۔ اور جب انہیں ناپ کر یا تول کر دیتے ہیں تو گھٹا دیتے ہیں۔ غور کیجیے کہ عیب پورا لینے میں نہیں؛ اپنا حق پورا لینا جائز ہے۔ عیب اس دوہرے پیمانے میں ہے کہ لینے کا ناپ اور ہو اور دینے کا اور۔ یہی اس سورت کی کنجی ہے، اور یہیں سے اس کی باطنی وسعت کھلتی ہے۔ مگر ادب پہلے: ظاہری حکم اپنی جگہ لازم اور مقدم ہے؛ دکاندار کا ترازو، کارخانے کا پیمانہ، مزدور کی اجرت اور ملازم کا وقت، سب اسی آیت کے تحت ہیں۔ جو شخص اس آیت کا اشاری معنی پڑھ کر بھی اپنے کاروبار میں کم تولتا رہے، وہ ظاہر سے بھی محروم رہا اور باطن سے بھی۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ اسی گہرائی میں اہلِ معرفت کہتے ہیں کہ تطفیف صرف غلے اور سونے میں نہیں ہوتی۔ گفتگو میں بھی ہوتی ہے کہ اپنی بات کا پورا وزن مانگا جائے اور دوسرے کی بات آدھی سنی جائے؛ فیصلے میں بھی ہوتی ہے کہ اپنی غلطی کا عذر پورا قبول ہو اور دوسرے کی غلطی پر پوری گرفت ہو؛ اور عبادت میں بھی ہوتی ہے۔ حدیث میں بالمعنی آیا ہے کہ سب سے برا چور وہ ہے جو اپنی نماز میں سے چوری کرے، یعنی اس کے رکوع اور سجدے پورے نہ کرے۔

شیخِ اکبر کے نزدیک اس دوہرے پیمانے کی جڑ نفس کا یہ دعویٰ ہے کہ وہ مرکز ہے اور باقی سب اس کے گرد گھومتے ہیں۔ جو اپنے آپ کو مرکز سمجھتا ہے، اسے اپنا حق ہمیشہ بڑا دکھائی دیتا ہے اور دوسرے کا حق ہمیشہ چھوٹا۔ اس مکتب میں اس کی سب سے باریک صورت حق تعالیٰ کے ساتھ معاملہ ہے: بندہ وجود، رزق، صحت اور ہدایت سب کچھ اسی سے پورا پورا لیتا ہے، اور جب لوٹانے کی باری آتی ہے، یعنی شکر، ادب، حضوری اور بندگی کی، تو کمی کر دیتا ہے۔ دعا میں مطالبہ پورا، اور سجدے میں دل آدھا۔ اہلِ ذوق کہتے ہیں کہ عارف کا ترازو اس کے برعکس جھکتا ہے: وہ دوسروں کو دیتے وقت جھکتا ہوا تولتا ہے اور لیتے وقت اپنے حق میں سے بھی کچھ چھوڑ دیتا ہے، کیونکہ وہ جانتا ہے کہ اس کے پاس جو کچھ ہے وہ امانت ہے، ملکیت نہیں۔ یہ ایثار ہے، اور ایثار عدل سے اوپر کا درجہ ہے، مگر عدل کو گرا کر نہیں، اس پر قائم ہو کر۔ یہاں ایک لطیف نکتہ بھی ہے: آیت نے دینے کے لیے "يُخْسِرُوْنَ" کہا، یعنی وہ دوسروں کو خسارے میں ڈالتے ہیں۔ مگر اہلِ تاویل کہتے ہیں کہ اصل خسارہ خود ان کا ہے، کیونکہ جو ناپ میں سے چند دانے بچا لیتا ہے وہ اپنے دل میں سے امانت داری کا نور کھو دیتا ہے۔ ظاہر میں اس نے کمایا، باطن میں اس نے گنوایا؛ اور اگلی آیات بتائیں گی کہ یہ گنوانا کہاں جا کر ٹھہرتا ہے۔

اب اس بیماری کی جڑ پر انگلی رکھی جاتی ہے: اَلَا يَظُنُّ اُولٰۤئِكَ اَنَّهُمْ مَّبْعُوْثُوْنَ ۙ لِيَوْمٍ عَظِيْمٍ ۙ يَّوْمَ يَقُوْمُ النَّاسُ لِرَبِّ الْعٰلَمِيْنَ ۭ۔ کیا یہ لوگ گمان نہیں رکھتے کہ وہ اٹھائے جانے والے ہیں، ایک بڑے دن کے لیے، جس دن سب لوگ رب العالمین کے سامنے کھڑے ہوں گے؟ مفسرین نے یہاں "يَظُنُّ" پر توجہ دلائی ہے: یقین تو دور کی بات، اگر ان کے دل میں اس دن کا گمان بھی ہوتا تو یہی گمان انہیں کم تولنے سے روک دیتا۔ گویا ناپ کی ہر کمی آخرت کے انکار کی ایک چھوٹی سی گواہی ہے، چاہے زبان اقرار کر رہی ہو۔ حضرت عبداللہ بن عمر رضی اللہ عنہما کی روایت میں اسی آیت کے ذیل میں بالمعنی آیا ہے کہ اس دن لوگ اپنے پسینے میں کھڑے ہوں گے، یہاں تک کہ کسی کا پسینہ آدھے کانوں تک پہنچا ہوگا۔ شیخِ اکبر کے مکتب میں یہاں اسمِ رب کا انتخاب معنی خیز ہے: رب وہ ہے جو ہر شے کو اس کی حد تک پالتا اور اس کا ناپ مقرر کرتا ہے۔ جس نے دنیا میں ناپ بگاڑا، وہ اس دن اس کے سامنے کھڑا ہوگا جس نے ہر شے کا ناپ ٹھیک رکھا تھا۔ اور "النَّاسُ" کا عموم بتاتا ہے کہ اس قیام سے کوئی مستثنیٰ نہیں، نہ بیچنے والا نہ خریدنے والا۔ اہلِ ذوق کہتے ہیں کہ نماز کا قیام اسی قیام کی مشق ہے؛ جو یہاں اپنی مرضی سے رب کے سامنے کھڑا ہونا سیکھ لے، اسے وہاں کا کھڑا ہونا اجنبی نہیں لگے گا۔

اس تنبیہ کے بعد پردہ اٹھتا ہے: كَلَّاۤ اِنَّ كِتٰبَ الْفُجَّارِ لَفِيْ سِجِّيْنٍ ۭ۔ ہرگز نہیں، بے شک بدکاروں کا نامۂ اعمال سجین میں ہے۔ وَمَاۤ اَدْرٰىكَ مَا سِجِّيْنٌ ۭ كِتٰبٌ مَّرْقُوْمٌ ۭ۔ اور تمہیں کیا معلوم کہ سجین کیا ہے؟ ایک لکھا ہوا دفتر ہے۔ "كَلَّاۤ" ڈانٹ اور تردید کا لفظ ہے: ان کا یہ خیال کہ کوئی حساب نہیں، ہرگز درست نہیں۔ مفسرین کے نزدیک سجین سجن سے ہے، یعنی قید اور تنگی؛ ایک قول میں یہ پستی کی وہ جگہ ہے جہاں بدکاروں کے اعمال کا دفتر رکھا جاتا ہے، اور آیت ۹ کی رو سے یہ خود ایک لکھا ہوا دفتر ہے۔ "كِتٰبٌ مَّرْقُوْمٌ" یعنی صاف لکھا ہوا، جیسے کپڑے پر نقش؛ اس میں نہ کمی ہوگی نہ بیشی۔ ناپ میں کمی کرنے والے کو بتایا گیا کہ تمہارا اپنا ناپ ایسے دفتر میں ہے جس میں کوئی تطفیف نہیں ہوتی۔ شیخِ اکبر کے نزدیک بندے کی کتاب دراصل اس کی اپنی حقیقت کا ظہور ہے؛ جو کچھ لکھا گیا وہ وہی ہے جو اس نے کیا، اور اس نے وہی کیا جس کی طرف اس نے رخ کیا۔ یہ سرِّ قدر ہے: حکم محکوم کے مطابق۔ اہلِ معرفت یہاں صحیح مسلم کی وہ حدیث یاد کرتے ہیں جس کا مفہوم یہ ہے کہ دنیا مومن کے لیے قید خانہ ہے؛ مومن نے اسے قید خانہ سمجھ کر اس سے دل نہ لگایا تو اس کی کتاب بلندی کو گئی، اور فاجر نے اسی قید خانے کو اپنا گھر بنا لیا تو اس کی کتاب بھی تنگی میں جا ٹھہری۔ کتاب وہیں رہتی ہے جہاں دل رہتا تھا۔

یہاں سے سورت جھٹلانے والوں کی طرف مڑتی ہے: وَيْلٌ يَّوْمَئِذٍ لِّلْمُكَذِّبِيْنَ ۙ۔ اس دن جھٹلانے والوں کے لیے تباہی ہے۔ الَّذِيْنَ يُكَذِّبُوْنَ بِيَوْمِ الدِّيْنِ ۭ۔ جو بدلے کے دن کو جھٹلاتے ہیں۔ وَمَا يُكَذِّبُ بِهٖۤ اِلَّا كُلُّ مُعْتَدٍ اَثِيْمٍ ۙ۔ اور اسے کوئی نہیں جھٹلاتا مگر ہر حد سے بڑھنے والا گنہگار۔ اِذَا تُتْلٰى عَلَيْهِ اٰيٰتُنَا قَالَ اَسَاطِيْرُ الْاَوَّلِيْنَ ۭ۔ جب اس کے سامنے ہماری آیتیں پڑھی جاتی ہیں تو کہتا ہے: پہلوں کی کہانیاں ہیں۔ آیت ۱۲ ایک تشخیص ہے: انکار پہلے عقل میں نہیں، کردار میں پیدا ہوتا ہے۔ پہلے آدمی حد سے بڑھتا اور گناہ کا عادی ہوتا ہے، پھر اسے اس دن کا انکار آسان لگنے لگتا ہے جس میں اس حد شکنی کا حساب ہوگا۔ اہلِ معرفت کہتے ہیں کہ بہت سے انکار دراصل دلیل کی تلاش نہیں، سہولت کی تلاش ہوتے ہیں۔ اور "اَسَاطِيْرُ الْاَوَّلِيْنَ" اس شخص کی زبان ہے جو کلام کو ماضی میں دفن کر کے اس کے خطاب سے بچنا چاہتا ہے: یہ پرانے لوگوں کی باتیں ہیں، مجھ سے ان کا کیا تعلق؟ شیخِ اکبر کے نزدیک کلامِ الٰہی ہر تلاوت میں تازہ ہے، کیونکہ متکلم ہر آن اپنی شان میں ہے اور ہر سننے والے سے اسی لمحے مخاطب ہے؛ جس نے آیت کو پرانی کہانی بنایا، اس نے اپنے کان پر وہی تہ چڑھائی جس کا ذکر اگلی آیت میں ہے۔ "تُتْلٰى عَلَيْهِ" میں آیتیں اس کے سامنے پہنچ چکی ہیں؛ کمی پیغام میں نہیں، سننے والے کے ظرف میں ہے۔

اب وہ آیت آتی ہے جو اس سورت کا اصول ہے، اور جس پر ہدایت و گمراہی کے پورے باب کا فیصلہ ٹکا ہے: كَلَّا بَلْ ۜ رَانَ عَلٰى قُلُوْبِهِمْ مَّا كَانُوْا يَكْسِبُوْنَ۔ ہرگز نہیں، بلکہ ان کے دلوں پر اس چیز کا زنگ چڑھ گیا ہے جو وہ کماتے رہے تھے۔ یہ اصول ہے، اور اسے اصول ہی کی طرح پڑھنا چاہیے۔ دل پر جو زنگ ہے وہ نہ پیدائشی ہے، نہ اوپر سے زبردستی ڈالا گیا ہے؛ وہ "مَّا كَانُوْا يَكْسِبُوْنَ" ہے، یعنی ان کی اپنی کمائی۔ اور "كَانُوْا" استمرار کا صیغہ ہے: ایک لغزش نہیں، بلکہ مسلسل کمائی، عادت بن جانے والا عمل۔ امام ترمذی رحمۃ اللہ علیہ نے حضورِ اکرم صلی اللہ علیہ وسلم کی وہ حدیث روایت کی ہے جس کا مفہوم یہ ہے کہ بندہ جب گناہ کرتا ہے تو اس کے دل پر ایک سیاہ نکتہ پڑ جاتا ہے؛ پھر اگر وہ باز آ جائے، استغفار کرے اور توبہ کرے تو اس کا دل صاف کر دیا جاتا ہے، اور اگر گناہ میں بڑھتا جائے تو وہ نکتہ بھی بڑھتا جاتا ہے یہاں تک کہ پورے دل پر چھا جاتا ہے؛ اور یہی وہ زنگ ہے جس کا ذکر اللہ تعالیٰ نے اس آیت میں فرمایا۔ اس حدیث نے ساری بات کھول دی: زنگ قطرہ قطرہ بندے کے اپنے ہاتھ سے جمتا ہے، اور ہر مرحلے پر صفائی کا دروازہ کھلا رہتا ہے۔ حق تعالیٰ نے یہ زنگ کسی پر مسلط نہیں کیا، بندے نے اسے کمایا؛ حق نے صرف اس کمائی کو وجود دیا جس کی طرف بندے نے بار بار ہاتھ بڑھایا۔

شیخِ اکبر کے مکتب میں دل آئینہ ہے، اور اس کی ساری قدر اسی میں ہے کہ اس میں حق کی تجلی منعکس ہو۔ تجلی ہر آن جاری ہے؛ سورج کسی آئینے سے منہ نہیں موڑتا۔ فرق آئینوں میں ہے: صاف آئینہ روشنی کو قبول کرتا ہے، اور زنگ آلود آئینہ اسی روشنی کے سامنے رکھا ہوا بھی دھندلا رہتا ہے۔ اس لیے اہلِ معرفت کے نزدیک محرومی کا سبب کبھی فیض کی کمی نہیں ہوتا، ہمیشہ ظرف کا زنگ ہوتا ہے۔ اور یہاں ایک بڑی بشارت چھپی ہے: زنگ آئینے کی ذات نہیں، اس کے اوپر کی تہ ہے۔ آیت کے الفاظ "عَلٰى قُلُوْبِهِمْ" ہیں، یعنی زنگ دلوں پر ہے؛ اس کے نیچے وہی فطرت محفوظ ہے جس پر اللہ نے ہر انسان کو پیدا کیا۔ شیخِ اکبر کے نزدیک قلب کا نام ہی تقلب سے ہے، یعنی پلٹنا؛ دل ٹھہری ہوئی چیز نہیں، ہر لمحہ کسی رخ پر مڑ رہا ہے۔ جب وہ بار بار ایک ہی رخ پر مڑے، دنیا کی طرف، کم تولنے کی طرف، اپنے نفس کی طرف، تو وہ رخ اس پر جم جاتا ہے، اور یہی جماؤ زنگ ہے۔ اور جب وہ حق کی طرف پلٹ آئے تو وہی تقلب صفائی بن جاتا ہے۔ توبہ کا لغوی معنی بھی لوٹنا ہے؛ دل اپنی بناوٹ ہی میں لوٹنے کی صلاحیت رکھتا ہے، اور یہ صلاحیت آخری سانس تک اس سے چھینی نہیں جاتی۔ زنگ جتنا بھی گہرا ہو، آئینہ آئینہ ہی رہتا ہے۔

یہیں سے وہ اصول پوری طرح واضح ہوتا ہے جو اس سلسلے میں بار بار طے ہوا: حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت طلب کرے اسے کبھی گمراہ نہیں کرتا؛ جو منہ پھیرے، اس کا پھیرنا ہی اس کی گمراہی بنتا ہے۔ سورۂ الصف میں یہی ترتیب یوں بیان ہوئی تھی: فَلَمَّا زَاغُوْۤا اَزَاغَ اللّٰهُ قُلُوْبَهُمْ ۭ۔ پھر جب وہ ٹیڑھے ہوئے تو اللہ نے ان کے دل ٹیڑھے کر دیے۔ پہلے ان کا اپنا ٹیڑھا ہونا، پھر اسی رخ کو وجود ملنا۔ یہاں بھی پہلے "يَكْسِبُوْنَ" ہے، پھر "رَانَ"۔ شیخِ اکبر کے نزدیک یہ سرِّ قدر ہے: حق ہر شے کو وہی دیتا ہے جو اس کی اپنی حقیقت اور اس کا اپنا رخ مانگتا ہے۔ وہ چاہتا تو سب کو زبردستی ایک راہ پر چلا دیتا، مگر اس نے انسان کو اختیار کی عزت دی، اور اختیار کی عزت یہی ہے کہ انتخاب کا نتیجہ بھی اسی کا ہو۔ ایک لطیف اشارہ قراءت میں بھی ہے: روایتِ حفص میں "بَلْ" پر سکتہ ہے، یعنی سانس توڑے بغیر ذرا رک کر "رَانَ" پڑھا جاتا ہے تاکہ دونوں لفظ مل کر ایک نہ ہو جائیں۔ اہلِ ذوق کہتے ہیں کہ زنگ کو پہچاننے کے لیے بھی ایک ٹھہراؤ چاہیے؛ جو کبھی رک کر اپنے دل کو نہیں دیکھتا، اسے اپنا زنگ دکھائی نہیں دیتا۔ اہلِ تصوف کے نزدیک اس کی صفائی کے تین اوزار ہیں: توبہ جو نئے زنگ کو روکتی ہے، استغفار جو پرانی تہ کو اتارتا ہے، اور ذکر جو آئینے کو چمکاتا رہتا ہے۔ جس ہاتھ نے زنگ کمایا تھا، وہی ہاتھ صفائی بھی کما سکتا ہے۔

زنگ کا انجام پردہ ہے: كَلَّاۤ اِنَّهُمْ عَنْ رَّبِّهِمْ يَوْمَئِذٍ لَّمَحْجُوْبُوْنَ ۭ۔ ہرگز نہیں، بے شک وہ اس دن اپنے رب کے دیدار سے روک دیے جائیں گے۔ ترتیب پر غور کیجیے: آیت ۱۴ میں دل پر زنگ تھا، آیت ۱۵ میں رب سے حجاب ہے، اور آیت ۱۶ میں جہنم۔ گویا جو زنگ یہاں دل پر جمتا ہے، وہی وہاں حجاب کی صورت اختیار کر لیتا ہے؛ اور حجاب ہی اصل محرومی ہے، آگ اس کے بعد آتی ہے۔ امام شافعی رحمۃ اللہ علیہ نے اس آیت سے استدلال کیا، جس کا مفہوم یہ ہے کہ جب اللہ نے ایک گروہ کو ناراضی کے سبب اپنے دیدار سے محجوب رکھا، تو اس میں دلیل ہے کہ ایک گروہ رضا کے سبب اس کا دیدار کرے گا۔ اہلِ سنت کا یہی عقیدہ ہے کہ اہلِ ایمان آخرت میں اپنے رب کا دیدار کریں گے، بلا کیف، بغیر کسی سمت، حد یا تشبیہ کے۔ سورۂ القیامہ میں یہی خبر صاف الفاظ میں آئی تھی: وُجُوْهٌ يَّوْمَئِذٍ نَّاضِرَةٌ ۙ اِلٰى رَبِّهَا نَاظِرَةٌ ۚ۔ کچھ چہرے اس دن تر و تازہ ہوں گے، اپنے رب کی طرف دیکھ رہے ہوں گے۔ ایک طرف "لَّمَحْجُوْبُوْنَ"، دوسری طرف "نَاظِرَةٌ"؛ دونوں کے بیچ وہی فرق ہے جو زنگ اور صفائی کے بیچ ہے۔ اور دونوں جگہ "يَوْمَئِذٍ" ہے: ایک ہی دن، ایک ہی تجلی، اور ہر آنکھ کا حال اس کے اپنے آئینے کے مطابق۔

شیخِ اکبر کے نزدیک حجاب حق کی طرف سے نہیں، بندے کی طرف سے ہے۔ حق پر کوئی شے پردہ نہیں ڈال سکتی؛ وہ ہر شے پر ظاہر ہے اور ہر شے میں اپنی شان کے مطابق متجلی ہے۔ پردہ دیکھنے والی آنکھ پر ہوتا ہے، دیکھے جانے والے پر نہیں۔ اسی لیے آیت نے یوں نہیں فرمایا کہ ان کا رب ان سے چھپ جائے گا، بلکہ یوں کہ وہ اپنے رب سے محجوب ہوں گے؛ حجاب ان کی صفت بنا۔ اور "يَوْمَئِذٍ" کی قید میں اہلِ معرفت ایک باریک بات دیکھتے ہیں: یہ پردہ اس دن پیدا نہیں ہوگا، اس دن ظاہر ہوگا۔ دنیا میں بھی وہ محجوب تھے، مگر دنیا کی مشغولیت نے انہیں اپنی محرومی کا احساس نہ ہونے دیا؛ وہاں جب ہر مشغولیت چھن جائے گی تو صرف پردہ باقی رہے گا اور پردے کا درد۔ "رَبِّهِمْ" کا لفظ اس درد کو اور گہرا کرتا ہے: وہ کسی اجنبی سے نہیں، اپنے رب سے محروم ہوں گے، اسی سے جس نے انہیں پالا تھا اور جس کی طرف ان کی فطرت کا رخ تھا۔ اس مکتب میں مومن کے لیے یہاں سبق یہ ہے کہ دیدارِ آخرت کی تیاری دنیا کی حضوری ہے: جو یہاں مراقبہ اور ذکر کے ذریعے اپنے دل کو اس کی طرف کھلا رکھتا ہے، اس کے لیے وہاں کا دیدار اسی حضوری کا کمال ہوگا، بلا کیف اور بلا تشبیہ۔ اور جو یہاں اس کی یاد سے پردے میں رہنا پسند کرے، اسے وہاں وہی ملے گا جو اس نے خود پسند کیا تھا۔

پردے کے بعد آگ کا ذکر ہے: ثُمَّ اِنَّهُمْ لَصَالُوا الْجَحِيْمِ ۭ۔ پھر بے شک وہ جہنم میں داخل ہونے والے ہیں۔ ثُمَّ يُقَالُ هٰذَا الَّذِيْ كُنْتُمْ بِهٖ تُكَذِّبُوْنَ ۭ۔ پھر کہا جائے گا: یہی ہے وہ جسے تم جھٹلایا کرتے تھے۔ دونوں آیتیں "ثُمَّ" سے شروع ہوتی ہیں: پہلے حجاب، پھر آگ، پھر یہ ملامت؛ ہر قدم پچھلے قدم کا نتیجہ ہے۔ صلی کا مادہ آگ میں داخل ہونے اور اس کی تپش سہنے کا ہے۔ اور "هٰذَا" کا اشارہ قابلِ غور ہے: جو چیز دنیا میں غیب تھی اور جس کا وہ مذاق اڑاتے تھے، وہ اب سامنے حاضر ہے، اتنی قریب کہ اس کی طرف "یہ" کہہ کر اشارہ کیا جا سکے۔ شیخِ اکبر کے مکتب میں آخرت غیب کا شہادت بن جانا ہے؛ وہاں کوئی نئی حقیقت پیدا نہیں ہوتی، پرانی حقیقت بے پردہ ہو جاتی ہے۔ اہلِ تاویل کہتے ہیں کہ اس جملے میں سب سے سخت چیز اس کا عدل ہے: انہیں کوئی نیا الزام نہیں سنایا جاتا، صرف ان کا اپنا انکار ان کے سامنے رکھ دیا جاتا ہے۔ جو جھٹلایا تھا وہی سامنے ہے؛ بدلہ عمل کے عین مطابق، کوئی اضافہ نہیں۔ اور اس میں اہلِ ایمان کے لیے بھی ایک تنبیہ ہے: ہر وہ حقیقت جسے آج دل ٹالتا ہے، کل "هٰذَا" بن کر سامنے آئے گی؛ عقل مند وہ ہے جو اسے آج ہی حاضر جان لے، اور جو آج حاضر جان لے، اس کے لیے کل کا سامنا رسوائی نہیں، ملاقات ہوگا۔

اب ترازو کا دوسرا پلڑا: كَلَّاۤ اِنَّ كِتٰبَ الْاَبْرَارِ لَفِيْ عِلِّيِّيْنَ ۭ۔ ہرگز نہیں، بے شک نیکوکاروں کا نامۂ اعمال علیین میں ہے۔ وَمَاۤ اَدْرٰىكَ مَا عِلِّيُّوْنَ ۭ كِتٰبٌ مَّرْقُوْمٌ ۙ يَّشْهَدُهُ الْمُقَرَّبُوْنَ ۭ۔ اور تمہیں کیا معلوم کہ علیون کیا ہے؟ ایک لکھا ہوا دفتر ہے، جس کے پاس مقرب حاضر رہتے ہیں۔ ساخت بالکل وہی ہے جو سجین کی آیات کی تھی: "كَلَّاۤ"، کتاب، "وَمَاۤ اَدْرٰىكَ"، اور "كِتٰبٌ مَّرْقُوْمٌ"۔ صرف دو چیزیں بدلیں: جگہ بلندی ہو گئی، اور ایک آیت کا اضافہ ہوا کہ مقرب اس کے پاس حاضر ہیں۔ مفسرین کے نزدیک علیین علو سے ہے، یعنی بلندی در بلندی، اور یہاں مقربون سے مراد مقرب فرشتے ہیں جو اس دفتر کے پاس حاضر رہتے اور اس کی گواہی دیتے ہیں۔ اہلِ معرفت کے نزدیک یہ مماثلت خود ایک تعلیم ہے: دونوں دفتر ایک ہی قانون پر لکھے گئے؛ نہ کسی کے ساتھ بے جا رعایت، نہ کسی پر زیادتی۔ فرق لکھنے والے میں نہیں، لکھے جانے والے عمل میں ہے۔ شیخِ اکبر کے نزدیک ہر کتاب اپنے صاحب کی ہمت کے ساتھ اٹھتی یا گرتی ہے؛ جس کا دل بلندی کی طرف مائل رہا، اس کا دفتر بھی بلندی میں رکھا گیا، اور جس کا دل پستی کی طرف جھکا، اس کا دفتر پستی میں۔ ہر شے اپنے اصل کی طرف لوٹتی ہے، اور عمل کا اصل وہ رخ ہے جس سے وہ پیدا ہوا تھا۔

آیت ۲۱ کا لفظ "يَّشْهَدُهُ" اس سورت کے لطیف ترین الفاظ میں سے ہے۔ شہود کا مطلب صرف دیکھنا نہیں، حاضر ہو کر گواہ بننا ہے۔ بدکاروں کے دفتر کے ساتھ کسی گواہ کا ذکر نہیں، وہ تنگی میں تنہا پڑا ہے؛ اور نیکوکاروں کے دفتر کے گرد مقرب جمع ہیں، گویا ان کا عمل ایک محفل بن گیا ہے۔ اہلِ ذوق کہتے ہیں کہ نیکی کی ایک صفت یہ ہے کہ وہ تنہا نہیں رہتی؛ وہ اہلِ قرب کو اپنی طرف کھینچ لیتی ہے، جیسے خوشبو اپنے گرد لوگوں کو جمع کر لیتی ہے۔ شیخِ اکبر کے مکتب میں مقرب وہ ہیں جو اہلِ شہود ہیں، جن کے اور حق کے بیچ حجاب باقی نہیں رہا؛ ابرار کا دفتر ان کے سامنے ہے، یعنی نیک عمل اہلِ قرب کی نگاہ میں رہتا ہے۔ اس میں ابرار کے لیے عزت بھی ہے اور ترقی کی دعوت بھی: تمہارا عمل قرب والوں کے سامنے ہے، تو تم بھی قرب کی طرف بڑھو۔ اور سالک کے لیے یہ حدیثِ احسان کی شرح ہے، جس کا مفہوم یہ ہے کہ اللہ کی عبادت یوں کرو گویا تم اسے دیکھ رہے ہو، اور اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے۔ جو جانتا ہے کہ اس کا ہر عمل لکھا جا رہا ہے اور اہلِ قرب اس پر حاضر ہیں، وہ نہ ترازو میں کمی کرتا ہے نہ نماز میں۔ یوں سورت کا آغاز اور یہ مقام ایک دوسرے سے جڑ جاتے ہیں: تطفیف غفلت کی پیداوار ہے، اور شہود کا احساس اس کا علاج۔

اب ابرار کے حال کی تصویر کھلتی ہے: اِنَّ الْاَبْرَارَ لَفِيْ نَعِيْمٍ ۙ عَلَى الْاَرَآئِكِ يَنْظُرُوْنَ ۙ۔ بے شک نیکوکار نعمتوں میں ہوں گے، مسندوں پر بیٹھے نظارہ کر رہے ہوں گے۔ تَعْرِفُ فِيْ وُجُوْهِهِمْ نَضْرَةَ النَّعِيْمِ ۚ۔ تم ان کے چہروں پر نعمت کی تازگی پہچان لو گے۔ "يَنْظُرُوْنَ" کے بعد یہ نہیں بتایا گیا کہ وہ کس چیز کو دیکھ رہے ہوں گے۔ مفسرین نے کئی پہلو بیان کیے: اپنی نعمتوں کو، اپنے ملک کی وسعت کو، اور ایک قول میں اپنے رب کی طرف، جیسا کہ آیت ۱۵ کے ذیل میں گزرا۔ اہلِ ذوق کہتے ہیں کہ مفعول کا حذف خود نگاہ کی وسعت ہے: وہاں نگاہ کسی ایک چیز میں قید نہیں، جدھر اٹھے نعمت ہی نعمت ہے۔ اور یہاں محجوبوں سے مقابلہ پھر واضح ہے: ان کی نگاہ پر پردہ تھا، اِن کی نگاہ آزاد ہے۔ پھر "تَعْرِفُ" کا خطاب ہے، یعنی دیکھنے والا ان کے چہروں سے پہچان لے گا؛ نعمت باطن میں ہوگی اور اس کا نور چہرے پر۔ شیخِ اکبر کے مکتب میں ظاہر باطن کا آئینہ ہے؛ جو اندر ہے وہ آخرکار باہر آتا ہے۔ دنیا میں بھی دل کا زنگ چہرے کی بے رونقی بن جاتا ہے اور دل کی صفائی چہرے کا نور؛ وہاں یہ قانون پوری طرح ظاہر ہوگا، اور "نَضْرَةَ النَّعِيْمِ" ہر چہرے پر اس کے دل کی تحریر ہوگی۔

پھر ان کے پینے کا ذکر ہے: يُسْقَوْنَ مِنْ رَّحِيْقٍ مَّخْتُوْمٍ ۙ۔ انہیں سربمہر خالص مشروب پلایا جائے گا۔ خِتٰمُهٗ مِسْكٌ ۭ۔ جس کی مہر مشک کی ہوگی۔ "يُسْقَوْنَ" مجہول ہے: وہ خود نہیں انڈیلتے، انہیں پلایا جاتا ہے؛ ساقی کا نام نہیں لیا گیا، اور اہلِ ذوق کے نزدیک اس خاموشی میں ساقی کی عظمت ہے۔ رحیق وہ خالص مشروب ہے جس میں کوئی کھوٹ نہ ہو، اور "مَّخْتُوْمٍ" یعنی مہر لگا ہوا، جسے کسی ہاتھ نے نہ چھوا ہو۔ "خِتٰمُهٗ" کی تفسیر میں مفسرین کے دو معروف قول ہیں: ایک یہ کہ برتن کی مہر مشک کی ہوگی، دوسرا یہ کہ اس کا خاتمہ، یعنی آخری گھونٹ اور بعد کا ذائقہ، مشک کی خوشبو ہوگا۔ دنیا کی لذتوں کا انجام اکثر تلخی ہوتا ہے، اور وہاں انجام سب سے خوشبودار۔ اہلِ معرفت کے نزدیک یہ اسرار کی تمثیل بھی ہے: حقیقت کی شراب سربمہر ہے، ہر ہاتھ اس تک نہیں پہنچتا، اور اس کی مہر اسی کے لیے کھلتی ہے جس کے لیے کھولی جائے۔ اسی لیے ادبِ اسرار یہ ہے کہ جو مہر تمہارے لیے نہیں کھلی اسے زبردستی نہ توڑو، اور جو تمہارے لیے کھلی ہے اسے نااہلوں کے سامنے نہ انڈیلو۔ اور ختم کا دوسرا معنی سالک کو یاد دلاتا ہے کہ اصل اعتبار خاتمے کا ہے؛ جس کا انجام خوشبودار ہو، اس کا سفر کامیاب ہے۔

اسی آیت کے دوسرے حصے میں وہ جملہ آتا ہے جس نے مقابلے کا رخ ہی بدل دیا: وَفِيْ ذٰلِكَ فَلْيَتَنَافَسِ الْمُتَنَافِسُوْنَ ۭ۔ اور یہی وہ چیز ہے جس میں بازی لے جانے والوں کو ایک دوسرے سے بازی لے جانی چاہیے۔ تنافس نفیس سے ہے، یعنی قیمتی چیز؛ تنافس اس دوڑ کو کہتے ہیں جو کسی نفیس چیز کے لیے ہو، جس میں ہر شخص چاہے کہ وہ اسی کے ہاتھ آئے۔ اب سورت کی ساخت پر نظر ڈالیے: اس کا آغاز ان لوگوں سے ہوا جو طفیف کے لیے، یعنی حقیر اور معمولی مقدار کے لیے، دوڑ رہے تھے؛ اور یہاں وہ نفیس چیز دکھائی گئی جس کے لیے دوڑنا چاہیے۔ گویا قرآن نے انسان کی مسابقت کی خواہش کو مٹایا نہیں، اس کا رخ موڑ دیا۔ اہلِ معرفت کے نزدیک یہی نفس کی تربیت کا قانون ہے: نفس کی کوئی قوت جڑ سے نہیں کاٹی جاتی، اسے اس کے صحیح موضوع کی طرف پھیرا جاتا ہے۔ حرص کو آخرت کی حرص بنا دو، رشک کو نیکی کا رشک، اور مقابلے کو قرب کا مقابلہ۔ "وَفِيْ ذٰلِكَ" کو پہلے لانے میں تخصیص کا رنگ ہے، یعنی اسی میں، کسی اور میں نہیں؛ دنیا کی ہر وہ دوڑ جس میں ایک کی جیت دوسرے کی ہار ہو، اس کے مقابلے میں چھوٹی ہے۔ مطففین بھی ایک دوڑ میں تھے، مگر ان کا انعام چند دانے تھے؛ اور یہاں انعام وہ مشروب ہے جس کا خاتمہ بھی مشک ہے۔

شیخِ اکبر کے مکتب میں اس مقابلے کی ایک انوکھی خوبی ہے: یہ وہ واحد دوڑ ہے جس میں کسی کی جیت سے دوسرے کا حصہ کم نہیں ہوتا۔ دنیا کی نعمتیں محدود ہیں، اس لیے ان پر مقابلہ حسد پیدا کرتا ہے؛ حق کی عطا لامحدود ہے، اس لیے اس پر مقابلہ محبت پیدا کرتا ہے۔ یہاں ہر شخص اپنی استعداد کے مطابق بھرا جاتا ہے، اور کسی کا جام بھرنے سے دوسرے کا جام خالی نہیں ہوتا۔ اسی لیے اہلِ ذوق کہتے ہیں کہ سالک کا اصل حریف اس کا اپنا نفس ہے، اس کا ہم سفر نہیں؛ وہ کل کے اپنے آپ سے آگے بڑھنا چاہتا ہے۔ ایک اشارہ لفظ کی جڑ میں بھی ہے: نفیس، نفس اور نفَس ایک ہی خاندان کے الفاظ ہیں۔ اہلِ معرفت کہتے ہیں کہ سب سے نفیس شے وہ سانس ہے جو ذکر میں گزرے، کیونکہ ہر سانس نفَسِ رحمانی کی ایک لہر ہے جو لوٹ کر نہیں آتی؛ جو اپنی سانسوں کو اس دوڑ میں لگا دے، وہی سچا متنافس ہے۔ اس کے برعکس مطففین کی بدنصیبی یہی تھی کہ انہوں نے اپنی نفیس سانسیں طفیف مقداروں پر خرچ کر دیں، اور چند دانوں کے بدلے دل کا آئینہ بیچ دیا۔ یوں یہ سورت ہر پڑھنے والے سے ایک سوال کرتی ہے: تمہاری دوڑ کس چیز کے لیے ہے، طفیف کے لیے یا نفیس کے لیے؟

پھر اس مشروب کی آمیزش کا ذکر ہے، اور یہیں جنت کے درجات کھلتے ہیں: وَمِزَاجُهٗ مِنْ تَسْنِيْمٍ ۙ۔ اور اس میں تسنیم کی آمیزش ہوگی۔ عَيْنًا يَّشْرَبُ بِهَا الْمُقَرَّبُوْنَ ۭ۔ وہ ایک چشمہ ہے جس سے مقرب پیتے ہیں۔ تسنیم کا مادہ بلندی کا ہے، جیسے اونٹ کا کوہان سنام کہلاتا ہے؛ مفسرین نے کہا کہ یہ جنت کا سب سے اعلیٰ مشروب ہے جو بلندی سے اترتا ہے۔ اور حضرت عبداللہ بن مسعود اور حضرت عبداللہ بن عباس رضی اللہ عنہما سے بالمعنی منقول ہے کہ مقرب اسے خالص پیتے ہیں، اور ابرار کو اس کی آمیزش کے ساتھ دیا جاتا ہے۔ یوں دو گروہ سامنے آئے: ابرار، یعنی نیکوکار، جنہیں رحیق میں تسنیم ملا کر ملتا ہے، اور مقربون، جنہیں سرچشمہ خود ملتا ہے۔ نعمت دونوں کے لیے ہے، مگر ذائقے کا مرتبہ الگ ہے۔ آیت میں "اس میں سے" کا حرف نہیں، "بِهَا" فرمایا ہے۔ اہلِ تفسیر نے اس حرف کی کئی توجیہیں کی ہیں، اور اہلِ ذوق کہتے ہیں کہ مقرب کا معاملہ یہ ہے کہ چشمہ اس کے لیے محض برتن نہیں رہتا، رفیق بن جاتا ہے؛ وہ اسی کے ساتھ پیتا ہے اور اسی سے سیراب ہوتا ہے، گویا پینے والے اور پانی کے بیچ کا فاصلہ مٹ گیا۔ اور یہ بات کہ چشمہ "عَيْنًا" ہے، یعنی جاری سرچشمہ، اس طرف اشارہ ہے کہ قرب کی عطا کبھی رکتی نہیں، ہر لمحہ نئی ہے۔

شیخِ اکبر کے مکتب میں یہ آمیزش اور یہ خلوص دو طرح کے شہود کی تمثیل ہے۔ ابرار حق کو اس کے مظاہر میں دیکھتے ہیں: نعمت میں منعم کو، اسباب کے پردے میں مسبب کو، اور اسماء کے آئینوں میں ذات کی جھلک۔ ان کا مشروب ملا ہوا ہے، کیونکہ ان کی معرفت میں واسطے شامل ہیں؛ اور یہ کوئی کمی نہیں، ایک بلند مقام ہے۔ مقربون وہ ہیں جن کی نگاہ واسطوں سے گزر کر سرچشمے تک پہنچ گئی؛ ان کے لیے مظہر کا پردہ باریک ہو گیا، اس لیے وہ خالص پیتے ہیں۔ اہلِ معرفت کے نزدیک یہ وحدتِ وجود کا ذوقی پہلو ہے: وجود ایک ہے، مظاہر بے شمار؛ کوئی مظاہر کے ذریعے اس ایک تک پہنچتا ہے، اور کوئی اس ایک کے نور میں مظاہر کو دیکھتا ہے۔ اور ایک لطیف بات یہ ہے کہ ابرار کے جام میں بھی تسنیم موجود ہے؛ یعنی مقربین کے سرچشمے کا ذائقہ ابرار تک بھی پہنچتا ہے۔ جنت کے درجات دیواریں نہیں، ایک ہی چشمے کی مختلف منزلیں ہیں، اور ہر منزل اپنی جگہ لبالب ہے۔ اسی لیے وہاں کوئی کسی سے حسد نہیں کرتا۔ سالک کے لیے یہاں دعوت ہے کہ وہ ابرار کے مقام کی قدر کرتے ہوئے بھی تسنیم کی طلب نہ چھوڑے؛ یہی وہ مقابلہ ہے جس کی ابھی دعوت دی گئی تھی۔

سورت کا آخری حصہ دنیا کی طرف لوٹتا ہے، جہاں ترازو الٹا دکھائی دیتا تھا: اِنَّ الَّذِيْنَ اَجْرَمُوْا كَانُوْا مِنَ الَّذِيْنَ اٰمَنُوْا يَضْحَكُوْنَ ۖ۔ بے شک جو مجرم تھے وہ ایمان والوں پر ہنسا کرتے تھے۔ وَاِذَا مَرُّوْا بِهِمْ يَتَغَامَزُوْنَ ۖ۔ اور جب ان کے پاس سے گزرتے تو آپس میں آنکھوں سے اشارے کرتے۔ وَاِذَا انْقَلَبُوْۤا اِلٰۤى اَهْلِهِمُ انْقَلَبُوْا فَكِهِيْنَ ۖ۔ اور جب اپنے گھر والوں کی طرف لوٹتے تو مزے لیتے ہوئے لوٹتے۔ تین منظر ہیں: مجلس میں ہنسی، راستے میں آنکھوں کے اشارے، اور گھر میں اپنی باتوں کے مزے۔ تمسخر کا یہ پورا دائرہ ہے: کھلا مذاق، چھپا مذاق، اور پیٹھ پیچھے کا مذاق۔ اہلِ معرفت کہتے ہیں کہ ہنسی خود دل کا آئینہ ہے؛ جس چیز پر آدمی ہنستا ہے، اسی سے اس کے دل کا حال معلوم ہوتا ہے۔ مجرم اہلِ ایمان کی سادگی، ان کی نماز اور ان کے پورا تولنے پر ہنستے تھے، کیونکہ زنگ آلود آئینے میں حقیقت الٹی دکھائی دیتی ہے؛ جو سب سے قیمتی تھا، وہی انہیں سب سے مضحکہ خیز لگا۔ "فَكِهِيْنَ" بتاتا ہے کہ وہ اپنے حال پر مطمئن تھے، اور یہ اطمینان ہی ان کی سب سے گہری غفلت تھی۔ آیت ۳۱ میں دو بار انقلاب کا لفظ آیا، جو قلب کی جڑ سے ہے؛ اہلِ ذوق کہتے ہیں کہ ان کے قدم گھروں کی طرف پلٹتے رہے، مگر ان کا دل ایک بار بھی حق کی طرف نہ پلٹا۔ شیخِ اکبر کے نزدیک یہ اسمِ ظاہر پر رک جانے کا انجام ہے: انہوں نے مومن کی ظاہری کمزوری دیکھی، اس کے باطن کی دولت نہ دیکھی۔

تمسخر کی انتہا یہ تھی: وَاِذَا رَاَوْهُمْ قَالُوْۤا اِنَّ هٰۤؤُلَآءِ لَضَآلُّوْنَ ۙ۔ اور جب انہیں دیکھتے تو کہتے کہ یقیناً یہ لوگ گمراہ ہیں۔ وَمَاۤ اُرْسِلُوْا عَلَيْهِمْ حٰفِظِيْنَ ۭ۔ حالانکہ وہ ان پر نگران بنا کر نہیں بھیجے گئے تھے۔ جو خود راہ سے ہٹے ہوئے تھے، وہی راہ پر چلنے والوں کو گمراہ کہہ رہے تھے؛ زنگ آلود آئینے کا یہ آخری کرشمہ ہے کہ وہ صاف آئینے کو داغدار دکھاتا ہے۔ آیت ۳۳ نے ان کی حیثیت متعین کر دی: تمہیں کس نے دوسروں کے دلوں کا داروغہ بنایا؟ ظاہری معنی یہ ہے کہ ان مجرموں کو اہلِ ایمان کے اعمال کا نگران مقرر نہیں کیا گیا تھا، کہ ان پر فتوے لگاتے پھریں۔ مگر اہلِ معرفت یہاں اہلِ ایمان کے لیے بھی ایک آئینہ دیکھتے ہیں: دینی غرور میں دوسروں کے حال پر حکم لگانا اور ان کی تحقیر کرنا بھی اسی مرض کی ایک صورت ہے۔ یہ فیصلے کی تطفیف ہے: اپنے لیے عذر کا پورا پیمانہ، اور دوسروں کے لیے حسنِ ظن کا ادھورا پیمانہ۔ نیکی کی دعوت اور برائی سے روکنا اپنی جگہ فرض ہے، مگر دلوں کا نگران صرف حق ہے۔ شیخِ اکبر کے مکتب میں ہر بندے کا معاملہ اس کے ربِّ خاص کے ساتھ ہے، اور کسی کو معلوم نہیں کہ کس کا خاتمہ کس رخ پر ہوگا؛ اس لیے عارف دوسروں کو گمراہ کہنے سے پہلے اپنے آئینے کا زنگ دیکھتا ہے۔

اور پھر منظر پلٹ جاتا ہے: فَالْيَوْمَ الَّذِيْنَ اٰمَنُوْا مِنَ الْكُفَّارِ يَضْحَكُوْنَ ۙ عَلَى الْاَرَآئِكِ يَنْظُرُوْنَ ۭ۔ تو آج ایمان والے کافروں پر ہنس رہے ہیں، مسندوں پر بیٹھے نظارہ کر رہے ہیں۔ هَلْ ثُوِّبَ الْكُفَّارُ مَا كَانُوْا يَفْعَلُوْنَ۔ کیا کافروں کو ان کے کیے کا بدلہ مل گیا؟ "فَالْيَوْمَ" دنیا کے "كَانُوْا" کے مقابل ہے: وہ کل ہنستے تھے، یہ آج۔ یہ ہنسی انتقام کی ہنسی نہیں، حقیقت کے کھل جانے کی خوشی ہے؛ جو صبر انہوں نے دنیا میں کیا تھا، وہ آج مسکراہٹ بن کر لوٹا ہے۔ اور "عَلَى الْاَرَآئِكِ يَنْظُرُوْنَ" کی تکرار بتاتی ہے کہ ان کی نگاہ وہی آزاد نگاہ ہے جس کا ذکر آیت ۲۳ میں ہوا۔ آخری سوال "هَلْ ثُوِّبَ" تقریر کا سوال ہے، یعنی جواب خود ظاہر ہے: ہاں، پورا پورا۔ مفسرین کہتے ہیں کہ ثواب کا لفظ اصل میں لوٹانے کا ہے، اور یہاں اسے عمل کے لوٹ آنے کے معنی میں رکھا گیا ہے۔ شیخِ اکبر کے نزدیک جزا دراصل خود عمل ہے جو اپنی اصل صورت میں لوٹ آتا ہے؛ اسی لیے آیت نے یوں نہیں کہا کہ انہیں ان کے کیے کے بدلے میں کچھ ملا، بلکہ یہ کہ انہیں "مَا كَانُوْا يَفْعَلُوْنَ" لوٹا دیا گیا۔ یوں سورت اپنے دائرے کو مکمل کرتی ہے: جو سورت ناپ میں کمی کرنے والوں سے شروع ہوئی تھی، وہ اس اعلان پر ختم ہوتی ہے کہ حق کے ترازو میں کوئی کمی نہیں؛ ہر ایک کو ٹھیک اتنا ملا جتنا اس نے بھیجا تھا۔ اور اس سے پہلے کہ وہ دن آئے، دل کا آئینہ صاف کرنے کا دروازہ آج بھی کھلا ہے۔
