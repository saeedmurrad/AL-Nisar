پچھلا رکوع حضرت عیسیٰ علیہ السلام کے بارے میں غلو کی تردید پر ختم ہوا تھا؛ یہ رکوع اسی تاریخ کا دوسرا رخ دکھاتا ہے، اور پہلی ہی آیت میں بتا دیتا ہے کہ لعنت کسی نسل کا نام نہیں، ایک روش کا انجام ہے: لُعِنَ الَّذِيْنَ كَفَرُوْا مِنْۢ بَنِيْۤ اِسْرَآءِيْلَ عَلٰى لِسَانِ دَاوٗدَ وَعِيْسَى ابْنِ مَرْيَمَ ؕ ذٰلِكَ بِمَا عَصَوْا وَّكَانُوْا يَعْتَدُوْنَ۔ “بنی اسرائیل میں سے جنہوں نے کفر کیا ان پر حضرت داؤد اور حضرت عیسیٰ ابنِ مریم علیہما السلام کی زبان سے لعنت کی گئی؛ یہ اس لیے کہ انہوں نے نافرمانی کی اور وہ حد سے گزر جایا کرتے تھے۔” یہاں مِنْ تبعیض کے لیے ہے: سب نہیں، ان میں سے وہ جنہوں نے انکار کی راہ چنی۔ بعض مفسرین نے نقل کیا ہے کہ حضرت داؤد علیہ السلام کی زبان سے یہ کلمہ ان لوگوں کے حق میں نکلا جنہوں نے سبت کی حرمت توڑی، اور حضرت عیسیٰ علیہ السلام کی زبان سے ان کے حق میں جنہوں نے کھلی نشانیاں دیکھ کر بھی انکار کیا؛ یہ تفصیل روایت ہے، اور قرآن کا اصل زور سبب پر ہے: ذٰلِكَ بِمَا عَصَوْا، یعنی یہ ان کے اپنے عمل کا نتیجہ تھا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں لعنت کا معنی دوری ہے، اور یہ دوری حق کی طرف سے کوئی نیا جبر نہیں، اس فاصلے کا اعلان ہے جو بندہ پہلے خود اختیار کر چکا تھا۔ اہلِ ذوق کہتے ہیں کہ یہاں دو ایسے انبیاء علیہم السلام کا نام لیا گیا جن کی پہچان زبور کی حمد اور مسیحائی کی رحمت ہے، تاکہ معلوم ہو کہ رحمت کی زبان بھی جب حقیقت بیان کرتی ہے تو دوری کو دوری ہی کہتی ہے۔ یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ اور یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر وہ بات آئی جو اس نافرمانی اور زیادتی کی جڑ تھی: كَانُوْا لَا يَتَنَاهَوْنَ عَنْ مُّنْكَرٍ فَعَلُوْهُ ؕ لَبِئْسَ مَا كَانُوْا يَفْعَلُوْنَ۔ “وہ ایک دوسرے کو اس برائی سے روکتے نہ تھے جو وہ کرتے تھے؛ بہت ہی برا تھا جو وہ کر رہے تھے۔” گناہ ہر بستی میں ہوتا ہے؛ ہلاکت اس دن شروع ہوتی ہے جب گناہ پر خاموشی رواج بن جائے۔ ابو داؤد اور ترمذی میں حضرت عبد اللہ بن مسعود رضی اللہ عنہ سے روایت ہے جس کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: بنی اسرائیل میں پہلی خرابی یوں آئی کہ ایک شخص دوسرے کو برائی کرتے دیکھتا تو کہتا کہ اللہ سے ڈرو اور اسے چھوڑ دو؛ پھر اگلے دن اسے اسی حال میں پاتا تو یہ بات اسے اس کے ساتھ کھانے پینے اور بیٹھنے سے نہ روکتی؛ تب ان کے دل ایک دوسرے کے رنگ میں رنگ دیے گئے، اور یہ رنگ ان کی اپنی مجلس داری اور سمجھوتے کا نتیجہ تھا۔ پھر حضورِ اکرم صلی اللہ علیہ وسلم نے یہی آیات تلاوت فرمائیں اور تاکید فرمائی کہ تم ضرور بھلائی کا حکم دو گے اور برائی سے روکو گے۔ ترمذی میں حضرت حذیفہ رضی اللہ عنہ کی روایت کا مفہوم ہے کہ اگر تم نے یہ فریضہ چھوڑ دیا تو قریب ہے کہ اللہ تم پر اپنی طرف سے کوئی عذاب بھیجے، پھر تم دعا کرو اور قبول نہ ہو۔ اور صحیح مسلم میں حضرت ابو سعید خدری رضی اللہ عنہ کی روایت کا مفہوم ہے کہ تم میں سے جو برائی دیکھے وہ اسے ہاتھ سے بدلے، یہ نہ ہو سکے تو زبان سے، یہ بھی نہ ہو سکے تو دل سے برا جانے، اور یہ ایمان کا سب سے کمزور درجہ ہے۔ غور کیجیے کہ آیت نے فَعَلُوْهُ کہا: برائی کہیں باہر نہ تھی، وہ خود اس کے کرنے والے تھے، اس لیے روکنے والی زبان بند ہو گئی۔

اہلِ معرفت اس آیت کے صیغے پر ٹھہرتے ہیں: يَتَنَاهَوْنَ باہمی عمل ہے، یعنی ایک دوسرے کو روکنا۔ گویا جماعت ایک جسم ہے جس کا ہر عضو دوسرے کا نگہبان ہے، اور جب یہ باہمی نگہبانی ٹوٹتی ہے تو جسم میں بیماری بے روک پھیلتی ہے۔ اہلِ تاویل یہی بات انسان کے باطن میں دیکھتے ہیں: انسان اپنے اندر ایک پوری بستی ہے، عقل، دل، خواہش، غضب اور خیال اس کے باشندے ہیں۔ جب تک دل نفس کو ٹوکتا ہے، وہ نفسِ لوامہ زندہ ہے جس کی قسم سورۂ القیامہ میں کھائی گئی؛ مگر جب ٹوکنے والی آواز چپ ہو جائے اور ہر قوت دوسری کی غلطی سے سمجھوتا کر لے تو باطن کی بستی وہی منظر پیش کرتی ہے جو اس آیت میں ہے۔ شیخِ اکبر کے نزدیک ذکر کی ایک صورت یہی بیداری ہے کہ قلب اپنے احوال پر حاضر رہے اور اپنی کسی حالت سے غافل نہ ہو۔ اور ظاہر کا ادب بھی یہاں واضح رہے: ہاتھ سے روکنا ان کا حق ہے جنہیں اختیار حاصل ہو، جیسے حکومت اور عدالت اپنے دائرے میں اور گھر کا سربراہ اپنے گھر میں؛ عام آدمی کا راستہ زبان کی نرم نصیحت اور دل کی ناپسندیدگی ہے۔ قرآن نے اس کا طریقہ خود سکھایا: اُدْعُ اِلٰى سَبِيْلِ رَبِّكَ بِالْحِكْمَةِ وَالْمَوْعِظَةِ الْحَسَنَةِ۔ نہی عن المنکر سختی کا نام نہیں، خیر خواہی کا نام ہے؛ جو روکنے والا اپنے بھائی کی رسوائی کا خواہاں ہو وہ خود ایک منکر میں جا پڑا۔

اس خاموشی کا اگلا قدم یہ تھا کہ دوستی کا رخ ہی بدل گیا: تَرٰى كَثِيْرًا مِّنْهُمْ يَتَوَلَّوْنَ الَّذِيْنَ كَفَرُوْا ؕ لَبِئْسَ مَا قَدَّمَتْ لَهُمْ اَنْفُسُهُمْ اَنْ سَخِطَ اللّٰهُ عَلَيْهِمْ وَفِي الْعَذَابِ هُمْ خٰلِدُوْنَ۔ “تم ان میں سے بہت سوں کو دیکھتے ہو کہ کافروں سے دوستی رکھتے ہیں؛ بہت ہی برا ہے وہ جو ان کے نفسوں نے ان کے لیے آگے بھیجا، کہ اللہ ان پر ناراض ہوا اور وہ عذاب میں ہمیشہ رہیں گے۔” مفسرین نے اسے مدینہ منورہ کے اس دور سے جوڑا ہے جب اہلِ کتاب کے بعض سردار، باوجود اس کے کہ وہ توحید اور انبیاء علیہم السلام کے ماننے والے تھے، مکہ کے بت پرستوں کے ساتھ مسلمانوں کے خلاف عہد و پیمان کرنے لگے؛ اور بعض نے اس کا مصداق منافقین کو بتایا۔ دونوں صورتوں میں یہ ایک خاص زمانے کا خاص رویہ ہے، اور كَثِيْرًا مِّنْهُمْ کا لفظ خود بتاتا ہے کہ حکم سب پر نہیں۔ اب اس آیت کا وہ کلمہ دیکھیے جو اس پورے قصے کا اصول ہے: مَا قَدَّمَتْ لَهُمْ اَنْفُسُهُمْ، وہ جو ان کے اپنے نفسوں نے آگے بھیجا۔ ناراضی کوئی باہر سے لایا ہوا فیصلہ نہیں؛ بندہ جو کچھ آگے بھیجتا ہے، آگے جا کر اسی سے ملاقات کرتا ہے۔ اہلِ معرفت کہتے ہیں کہ حق کسی کو گمراہ ہونے پر مجبور نہیں کرتا؛ جو رخ بندے نے چنا، حق اس رخ کو وجود دے دیتا ہے، اور یہی سرِّ قدر ہے۔ شیخِ اکبر کے مکتب میں آخرت کی ہر صورت دنیا کے عمل کی باطنی شکل ہے جو وہاں ظاہر ہو جاتی ہے؛ یہاں جو چھپا تھا، وہاں وہی سامنے آ کھڑا ہوتا ہے۔

اور اسی کی جڑ اگلی آیت میں کھولی گئی: وَلَوْ كَانُوْا يُؤْمِنُوْنَ بِاللّٰهِ وَالنَّبِيِّ وَمَاۤ اُنْزِلَ اِلَيْهِ مَا اتَّخَذُوْهُمْ اَوْلِيَآءَ وَلٰكِنَّ كَثِيْرًا مِّنْهُمْ فٰسِقُوْنَ۔ “اور اگر وہ اللہ پر اور نبی صلی اللہ علیہ وسلم پر اور اس پر ایمان رکھتے جو آپ صلی اللہ علیہ وسلم کی طرف اتارا گیا تو انہیں دوست نہ بناتے؛ لیکن ان میں سے بہت سے نافرمان ہیں۔” یہ آیت بتاتی ہے کہ دوستی عقیدے کا عکس ہے۔ یہاں اولیاء سے مراد وہ سیاسی و جنگی رفاقت ہے جو ایمان والوں کی جماعت کے خلاف قائم کی جائے؛ عام انسانی بھلائی، انصاف اور لین دین اس میں داخل نہیں، جیسا کہ اسی سورت میں اہلِ کتاب کے کھانے اور ان کی پاک دامن عورتوں سے نکاح کی اجازت آئی ہے۔ اہلِ ذوق کہتے ہیں کہ دل ہمیشہ اسی کی طرف مڑتا ہے جس پر وہ حقیقت میں ایمان رکھتا ہے؛ زبان کچھ بھی کہے، وابستگی کا رخ دل کے اصل قبلے کی خبر دے دیتا ہے۔ اسی لیے اس مکتب میں ولایت کو قرب اور رخ دونوں معنی میں سمجھا جاتا ہے: ولی وہ ہے جو کسی کے قریب ہو اور اسی کی طرف رخ کیے رہے۔ جس کا دل حق کی طرف ہے، اس کی دوستیاں بھی اسی نور سے روشن ہوتی ہیں، اور جس کا دل غیر کی طرف مڑ گیا، اس کی محبتیں بھی اسی غیر کی گواہی دینے لگتی ہیں۔ اور فٰسِقُوْنَ کا لفظ بھی سبب بتاتا ہے: فسق کا اصل معنی ہے دائرے سے باہر نکل آنا، جیسے کھجور اپنے چھلکے سے نکل آئے؛ یعنی یہ نکلنا ان کا اپنا تھا، کسی نے انہیں دھکیلا نہ تھا۔

اب وہ آیت آتی ہے جس کا ادب سے پڑھنا اس رکوع کی امانت ہے۔ اس کا پہلا حصہ اس تجربے کی خبر دیتا ہے جو حضورِ اکرم صلی اللہ علیہ وسلم اور صحابۂ کرام رضی اللہ عنہم کو اس دور کے سخت ترین مخالفین سے پیش آیا، یعنی مدینہ کے بعض یہودی قبائل اور مکہ کے مشرکین کی دشمنی؛ اور دوسرا حصہ ایک گروہ کی ستائش ہے: وَلَتَجِدَنَّ اَقْرَبَهُمْ مَّوَدَّةً لِّلَّذِيْنَ اٰمَنُوا الَّذِيْنَ قَالُوْۤا اِنَّا نَصٰرٰى ؕ ذٰلِكَ بِاَنَّ مِنْهُمْ قِسِّيْسِيْنَ وَرُهْبَانًا وَّاَنَّهُمْ لَا يَسْتَكْبِرُوْنَ۔ “اور تم ایمان والوں کے ساتھ محبت میں سب سے قریب ان لوگوں کو پاؤ گے جنہوں نے کہا کہ ہم نصاریٰ ہیں؛ یہ اس لیے کہ ان میں عالم پادری اور تارکِ دنیا عبادت گزار ہیں، اور وہ تکبر نہیں کرتے۔” قرآن نے ستائش کے ساتھ اس کی وجہ بھی بیان کر دی ہے، اس لیے حکم وصف کے ساتھ ہے، نام کے ساتھ نہیں۔ مفسرین کی ایک بڑی جماعت نے اس ستائش کا مصداق حبشہ کے بادشاہ نجاشی رحمۃ اللہ علیہ اور ان کے ساتھیوں کو بتایا، اور یہ بھی لکھا کہ جہاں یہ اوصاف ہوں وہاں یہ قرب ہے۔ پس یہ آیت نہ آج کے کسی یہودی پر دشمنی کا فتویٰ ہے، نہ کسی مسیحی کے لیے بے شرط سند؛ یہ ایک تاریخی گواہی ہے اور ایک اخلاقی میزان۔ اور قرآن کا عام اصول سورۂ الممتحنہ نے بتا دیا کہ جو تم سے دین کی بنا پر نہ لڑیں ان کے ساتھ نیکی اور انصاف سے اللہ منع نہیں کرتا: اَنْ تَبَرُّوْهُمْ وَتُقْسِطُوْۤا اِلَيْهِمْ۔ اس مکتب کا ادب یہ ہے کہ قرآن جہاں کسی گروہ کی کمی بتائے وہاں قاری اپنا آئینہ دیکھے، اور جہاں کسی کی خوبی بتائے وہاں اس کا دل کھلے اور اس کا انصاف بڑھے۔

اہلِ معرفت اس آیت کی تین صفتوں میں سلوک کے تین ستون دیکھتے ہیں۔ قِسِّيْسِيْنَ علم والے ہیں، وہ جو کتاب کو جانتے اور اس کی حفاظت کرتے ہیں؛ رُهْبَانًا عبادت اور خشیت والے ہیں، جنہوں نے دنیا کی دوڑ سے ہاتھ کھینچ لیا؛ اور لَا يَسْتَكْبِرُوْنَ وہ جوہر ہے جس کے بغیر پہلی دونوں صفتیں بھی حجاب بن جاتی ہیں۔ علم ہو اور تکبر ہو تو علم دیوار بن جاتا ہے، عبادت ہو اور تکبر ہو تو عبادت خود ایک بت بن جاتی ہے۔ شیخِ اکبر کے مکتب میں استعداد وہ ظرف ہے جس میں فیض اترتا ہے، اور ظرف کا پہلا وصف یہ ہے کہ وہ خالی اور جھکا ہوا ہو؛ جو برتن اوندھا رکھا ہو یا پہلے سے بھرا ہو، اس میں بارش کا قطرہ بھی نہیں ٹھہرتا۔ تکبر وہی انا ہے جس نے سب سے پہلے ابلیس کو حق کا حکم ماننے سے روکا، جس کا ذکر سورۂ صٓ میں اس کے کلمے اَنَا خَيْرٌ مِّنْهُ کے ساتھ گزر چکا؛ اور تواضع وہ کیفیت ہے جو دل کو حق کی پہچان کے لیے نرم کر دیتی ہے۔ اسی لیے آیت نے قرب کی وجہ نسب یا نام کو نہیں، دل کی اس نرمی کو قرار دیا۔ اہلِ ذوق کہتے ہیں کہ بندہ کسی بھی راستے سے آئے، جب وہ اپنی انا کو حق کے سامنے جھکا دیتا ہے تو حق کی بات اس کے دل پر ایسے اترتی ہے جیسے پانی نشیب کی طرف بہتا ہے۔ اور یہی اگلی آیت کا منظر ہے۔

اب وہ آیت جس پر اس رکوع کا دل دھڑکتا ہے: وَاِذَا سَمِعُوْا مَاۤ اُنْزِلَ اِلَى الرَّسُوْلِ تَرٰۤى اَعْيُنَهُمْ تَفِيْضُ مِنَ الدَّمْعِ مِمَّا عَرَفُوْا مِنَ الْحَقِّ ۚ يَقُوْلُوْنَ رَبَّنَاۤ اٰمَنَّا فَاكْتُبْنَا مَعَ الشّٰهِدِيْنَ۔ “اور جب وہ سنتے ہیں جو رسول صلی اللہ علیہ وسلم کی طرف اتارا گیا تو تم دیکھتے ہو کہ ان کی آنکھیں آنسوؤں سے بہہ رہی ہیں، اس لیے کہ انہوں نے حق کو پہچان لیا؛ وہ کہتے ہیں: اے ہمارے رب! ہم ایمان لائے، تو ہمیں گواہی دینے والوں کے ساتھ لکھ لے۔” سیرت کی کتابوں میں حضرت ام سلمہ رضی اللہ عنہا کی روایت آئی ہے، جسے امام احمد نے بھی نقل کیا، کہ جب مسلمان ہجرت کر کے حبشہ پہنچے اور قریش کے سفیر نجاشی رحمۃ اللہ علیہ کے دربار میں ان کی واپسی مانگنے آئے تو بادشاہ نے مسلمانوں کو بلایا۔ حضرت جعفر بن ابی طالب رضی اللہ عنہ نے ان کے سامنے سورۂ مریم کا ابتدائی حصہ تلاوت کیا، تو نجاشی رحمۃ اللہ علیہ اتنا روئے کہ ان کی داڑھی تر ہو گئی، اور ان کے علما بھی روئے یہاں تک کہ ان کے صحیفے بھیگ گئے؛ اور روایت کے مفہوم کے مطابق بادشاہ نے کہا کہ یہ کلام اور جو حضرت عیسیٰ علیہ السلام لائے، ایک ہی چراغ کی روشنی ہیں۔ مفسرین نے یہ بھی روایت کیا ہے کہ نجاشی رحمۃ اللہ علیہ نے اپنے کچھ لوگوں کا وفد حضورِ اکرم صلی اللہ علیہ وسلم کی خدمت میں بھیجا؛ آپ صلی اللہ علیہ وسلم نے ان کے سامنے قرآن پڑھا تو وہ روتے رہے اور ایمان لائے، اور یہ آیات ان کے بارے میں اتریں۔ اور صحیح بخاری و مسلم میں ہے کہ جب نجاشی رحمۃ اللہ علیہ کا انتقال ہوا تو حضورِ اکرم صلی اللہ علیہ وسلم نے صحابۂ کرام رضی اللہ عنہم کو ان کی وفات کی خبر دی اور ان کی غائبانہ نمازِ جنازہ پڑھائی۔ یہ اس آیت کی زندہ تفسیر ہے: ایک بادشاہ جس کی آنکھ کلام سن کر بہہ نکلی، اور ایک رسول صلی اللہ علیہ وسلم جنہوں نے دور بیٹھے اس کے لیے دعا کی۔

اب اس آیت کی ترتیب پر غور کیجیے: پہلے سننا، پھر آنسو، اور آنسو کا سبب مِمَّا عَرَفُوْا مِنَ الْحَقِّ، یعنی پہچان۔ آیت نے یہ نہیں کہا کہ انہوں نے کوئی نئی بات سیکھی؛ یہ کہا کہ انہوں نے پہچان لی۔ پہچان اسی چیز کی ہوتی ہے جو پہلے سے کہیں جانی ہوئی ہو۔ اہلِ معرفت کہتے ہیں کہ ہر روح کے اندر عہدِ الست کی ایک یاد امانت رکھی ہے، اور جب سچا کلام کان سے گزر کر دل تک پہنچتا ہے تو وہ یاد جاگ اٹھتی ہے؛ تب آنسو اس ملاقات کی گواہی بنتے ہیں جو بھولی ہوئی تھی۔ شیخِ اکبر کے نزدیک معرفت علم سے اسی لحاظ سے الگ ہے کہ اس میں ذوق اور پہچان شامل ہے؛ اسی لیے صاحبِ معرفت کو عارف کہا گیا، محض عالم نہیں۔ اور تَفِيْضُ کا لفظ دیکھیے: فیض اس پانی کو کہتے ہیں جو برتن کے بھر جانے سے چھلک پڑے۔ آنکھ اس لیے نہیں بہی کہ آنکھ نے کچھ دیکھا، بلکہ اس لیے کہ دل بھر گیا اور اس کی لبریزی آنکھ سے باہر آئی۔ پھر فوراً زبان پر آیا: فَاكْتُبْنَا مَعَ الشّٰهِدِيْنَ۔ حضرت عبد اللہ بن عباس رضی اللہ عنہما سے منقول ہے کہ گواہوں سے مراد امتِ محمدیہ ہے، جس کے بارے میں فرمایا گیا: لِّتَكُوْنُوْا شُهَدَآءَ عَلَى النَّاسِ۔ اہلِ ذوق کہتے ہیں کہ شاہد وہ ہے جو حاضر ہو کر دیکھے؛ گویا آنسوؤں نے ان کی غیبت کو حضوری میں بدل دیا، اور وہ دعا کرنے لگے کہ یہ حضوری لکھ دی جائے اور ہمیشہ کے لیے ثبت ہو جائے۔

آنسوؤں کے بعد ان کی زبان پر ایک سوال آیا جو دراصل اقرار ہے: وَمَا لَنَا لَا نُؤْمِنُ بِاللّٰهِ وَمَا جَآءَنَا مِنَ الْحَقِّ ۙ وَنَطْمَعُ اَنْ يُّدْخِلَنَا رَبُّنَا مَعَ الْقَوْمِ الصّٰلِحِيْنَ۔ “اور ہمیں کیا ہوا کہ ہم اللہ پر اور اس حق پر ایمان نہ لائیں جو ہمارے پاس آیا، اور ہم امید رکھتے ہیں کہ ہمارا رب ہمیں نیک لوگوں کے ساتھ داخل کرے گا۔” یہ وہ حال ہے جب حق سامنے آ جائے اور انکار کی کوئی وجہ باقی نہ رہے؛ سچا دل اپنے آپ سے پوچھتا ہے کہ اب رکنے کا کیا عذر ہے۔ اور نَطْمَعُ کا لفظ کتنا مؤدب ہے: انہوں نے دعویٰ نہیں کیا، امید کی۔ ایمان لانے کے بعد بھی وہ اپنے آپ کو صالحین میں شمار نہیں کرتے، ان کے ساتھ شامل کیے جانے کی آرزو کرتے ہیں۔ پھر جواب آیا: فَاَثَابَهُمُ اللّٰهُ بِمَا قَالُوْا جَنّٰتٍ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِيْنَ فِيْهَا ؕ وَذٰلِكَ جَزَآءُ الْمُحْسِنِيْنَ۔ “تو اللہ نے ان کی اس بات کے بدلے انہیں ایسے باغ دیے جن کے نیچے نہریں بہتی ہیں، ہمیشہ ان میں رہیں گے؛ اور یہی نیکی کرنے والوں کا بدلہ ہے۔” بِمَا قَالُوْا، ان کے کہنے کے بدلے؛ مگر یہ وہ کہنا تھا جس کے پیچھے آنسو تھے، یعنی دل کی سچائی زبان سے بولی تھی۔ آیت نے انہیں مُحْسِنِيْنَ کہا، اور حدیثِ جبریل میں احسان کا مفہوم یہ ہے کہ تم اللہ کی عبادت یوں کرو گویا اسے دیکھ رہے ہو۔ اہلِ معرفت کہتے ہیں کہ جن کی آنکھ حق کی پہچان پر بہہ نکلی، وہ اسی دیکھنے کی دہلیز پر کھڑے تھے؛ اور شیخِ اکبر کے مکتب میں جنت کی نہریں ان علوم و احوال کی صورتیں ہیں جو یہاں دل میں جاری ہوئے۔ اشاری معنی ظاہر کی نفی نہیں: جنت حقیقی ہے، اس کی نہریں حقیقی ہیں، اور یہ معنی اس کی گہرائی ہے۔

رکوع اسی میزان کے دوسرے پلڑے پر ختم ہوتا ہے: وَالَّذِيْنَ كَفَرُوْا وَكَذَّبُوْا بِاٰيٰتِنَاۤ اُولٰۤئِكَ اَصْحٰبُ الْجَحِيْمِ۔ “اور جنہوں نے کفر کیا اور ہماری آیتوں کو جھٹلایا، وہی دوزخ والے ہیں۔” یہاں دو فعل ہیں: کفر یعنی چھپانا، اور تکذیب یعنی جھٹلانا۔ یہ پچھلی آیات کا الٹ منظر ہے: وہاں سننے پر آنکھ بہہ نکلی تھی، یہاں سننے پر دل نے پردہ ڈالا اور زبان نے انکار کیا۔ کلام ایک ہی تھا، سننے والے ایک ہی رب کے سامنے تھے، فرق صرف رخ کا تھا۔ اہلِ معرفت کہتے ہیں کہ یہی اس سورت کا وہ اصول ہے جو اسی کے تیسرے رکوع میں بیان ہوا: يَهْدِيْ بِهِ اللّٰهُ مَنِ اتَّبَعَ رِضْوَانَهٗ سُبُلَ السَّلٰمِ؛ ہدایت اسے ملتی ہے جو اس کی رضا کی پیروی کرے۔ کفر لغت میں ڈھانپنے کو کہتے ہیں، اسی لیے کسان کو بھی کافر کہا گیا کہ وہ بیج کو مٹی میں چھپاتا ہے؛ پس انکار اس روشنی پر ڈالا ہوا پردہ ہے جو دل میں پہلے سے موجود تھی، اور یہ پردہ بندے کا اپنا ہاتھ ڈالتا ہے۔ اور اَصْحٰبُ کا لفظ رفاقت اور ساتھ لگے رہنے کا ہے: جس نے دنیا میں جس چیز کی صحبت چنی، آخرت میں اسی کا ساتھی ٹھہرا۔ شیخِ اکبر کے نزدیک ہر شخص کا انجام اس کے اپنے حال کی صورت ہے؛ حکم محکوم کے مطابق۔ اور اس کے ساتھ دروازہ کھلا ہے: جب تک سانس باقی ہے، رخ بدلا جا سکتا ہے، اور حبشہ کے دربار کی طرح ایک آنسو بھی پوری سمت پلٹ سکتا ہے۔
