پچھلے رکوع میں فیصلے کی بات تھی، کہ حکم کس کا مانا جائے؛ اب وفاداری کی بات ہے، کہ مشکل وقت میں دل اور ہاتھ کس کے ساتھ ہوں: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تَتَّخِذُوا الْيَهُوْدَ وَالنَّصٰرٰۤى اَوْلِيَآءَ بَعْضُهُمْ اَوْلِيَآءُ بَعْضٍ۔ “اے ایمان والو! یہود و نصاریٰ کو اپنا حمایتی اور سرپرست نہ بناؤ؛ وہ آپس میں ایک دوسرے کے حمایتی ہیں۔” اس آیت کو اس کے تاریخی پس منظر میں پڑھنا ضروری ہے، کیونکہ اس کے بغیر اس کا مطلب بگڑ جاتا ہے۔ مدینہ کی نوخیز مسلمان جماعت اس وقت جنگ کی حالت میں تھی؛ اردگرد کے کچھ قبیلوں سے معاہدے تھے، اور ان میں سے بعض قبیلے عہد توڑ کر دشمن کے ساتھ مل رہے تھے۔ مفسرین، جیسے امام ابنِ جریر طبری رحمۃ اللہ علیہ، نقل کرتے ہیں کہ جب بنو قینقاع نے عہد توڑا تو حضرت عبادہ بن صامت رضی اللہ عنہ، جن کا ان سے پرانا حلف تھا، حضورِ اکرم صلی اللہ علیہ وسلم کے پاس آئے اور اس حلف سے براءت کا اعلان کیا، جبکہ عبداللہ بن ابی نے یہ کہہ کر اسے تھامے رکھا کہ مجھے زمانے کی گردش کا ڈر ہے۔ تو یہاں اَوْلِيَآء سے مراد وہ سیاسی اور جنگی حلیف ہیں جن کے ساتھ مل کر اپنی ہی جماعت کے خلاف کھڑا ہوا جائے، اور جن پر بھروسا اپنے دین اور اپنے لوگوں سے بڑھ کر کیا جائے۔ بَعْضُهُمْ اَوْلِيَآءُ بَعْضٍ اسی زمانے کی صورتِ حال کا بیان ہے کہ مسلمانوں کے مقابلے میں ان کے حلقے آپس میں ایک تھے۔ یہ آیت کسی قوم سے نفرت کی تعلیم نہیں، جنگ کے زمانے میں وفاداری کی حد بندی ہے، اور آج کے کسی یہودی یا عیسائی کے بارے میں عمومی حکم نہیں۔

اور یہ بات خود قرآن نے کھول دی ہے کہ یہ ممانعت انصاف، بھلائی اور معاملات کی ممانعت نہیں۔ سورۂ الممتحنہ کے صفحے پر یہ آیت گزری: لَا يَنْهٰىكُمُ اللّٰهُ عَنِ الَّذِيْنَ لَمْ يُقَاتِلُوْكُمْ فِي الدِّيْنِ وَلَمْ يُخْرِجُوْكُمْ مِّنْ دِيَارِكُمْ اَنْ تَبَرُّوْهُمْ وَتُقْسِطُوْۤا اِلَيْهِمْ، “اللہ تمہیں ان لوگوں سے منع نہیں کرتا جنہوں نے دین کے معاملے میں تم سے جنگ نہیں کی اور تمہیں تمہارے گھروں سے نہیں نکالا، کہ تم ان سے بھلائی کرو اور ان کے ساتھ انصاف کرو۔” اور اسی سورۂ المائدہ کے شروع میں اہلِ کتاب کا کھانا حلال کیا گیا: وَطَعَامُ الَّذِيْنَ اُوْتُوا الْكِتٰبَ حِلٌّ لَّكُمْ، “اور جن کو کتاب دی گئی ان کا کھانا تمہارے لیے حلال ہے”، اور اسی آیت میں ان کی پاک دامن عورتوں سے نکاح جائز رکھا گیا۔ جس قوم کی عورت سے نکاح جائز ہو، اس سے محبت اور رشتہ داری کا دروازہ خود شریعت نے کھولا ہے؛ پھر اس آیت کو عام دشمنی کا حکم بنانا قرآن کو قرآن سے ٹکرانا ہے۔ صحیح بخاری میں ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم کی زرہ وفات کے وقت ایک یہودی کے پاس رہن رکھی ہوئی تھی، اور یہ بھی کہ ایک یہودی کا جنازہ گزرا تو حضورِ اکرم صلی اللہ علیہ وسلم کھڑے ہو گئے اور جو فرمایا اس کا مفہوم ہے: کیا یہ ایک انسانی جان نہیں؟ اہلِ علم نے اسی لیے فرق کیا: ایک ہے وہ وفاداری جس میں دین کے مقابلے میں کسی کا ساتھ دیا جائے یا مسلمانوں کے خلاف اس کی مدد کی جائے، یہ ممنوع ہے؛ اور ایک ہے پڑوس، تجارت، انصاف، احسان اور اچھا برتاؤ، یہ نہ صرف جائز ہے بلکہ اسلام کے اخلاق کا حصہ ہے۔

پھر آیت کا دوسرا حصہ: وَمَنْ يَّتَوَلَّهُمْ مِّنْكُمْ فَاِنَّهٗ مِنْهُمْ ۭ اِنَّ اللّٰهَ لَا يَهْدِي الْقَوْمَ الظّٰلِمِيْنَ۔ “اور تم میں سے جو انہیں اپنا سرپرست بنائے گا وہ انہی میں سے ہے۔ بے شک اللہ ظالم لوگوں کو ہدایت نہیں دیتا۔” مفسرین کہتے ہیں کہ یہاں تولّی سے مراد وہ وفاداری ہے جس میں آدمی اپنی جماعت کو چھوڑ کر مخالف صف میں جا کھڑا ہو، اس کے دین پر راضی ہو، یا مسلمانوں کے خلاف اس کی مدد کرے؛ ایسا شخص اسی کے حکم میں ہے جس کا اس نے ساتھ دیا۔ اور آخری جملہ اس سلسلے کے بنیادی اصول کی روشنی میں پڑھا جائے: اللہ ظالموں کو ہدایت نہیں دیتا، اس لیے نہیں کہ اس نے انہیں زبردستی روک رکھا ہے، بلکہ اس لیے کہ انہوں نے خود ظلم کا راستہ چنا اور ہدایت سے منہ موڑ لیا۔ جو ہدایت مانگے اسے ہدایت ملتی ہے؛ جو منہ پھیرے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے چنا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ اور اہلِ معرفت یہاں ایک باطنی سبق بھی دیکھتے ہیں جو کسی قوم کے بارے میں نہیں، ہر دل کے بارے میں ہے: ولایت کا مطلب ہے قرب اور سپردگی؛ جس کے ہاتھ میں آدمی اپنے دل کی باگ دے دیتا ہے، رفتہ رفتہ اسی جیسا ہو جاتا ہے۔ فَاِنَّهٗ مِنْهُمْ کا یہ قانون ہر تعلق پر لاگو ہے؛ حضورِ اکرم صلی اللہ علیہ وسلم کی حدیث کا مفہوم ہے کہ آدمی اپنے گہرے دوست کے طریقے پر ہوتا ہے، تو ہر ایک دیکھ لے کہ کس سے دوستی کرتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

اب ان کا حال جن کے دل میں کمزوری تھی: فَتَرَى الَّذِيْنَ فِيْ قُلُوْبِهِمْ مَّرَضٌ يُّسَارِعُوْنَ فِيْهِمْ يَقُوْلُوْنَ نَخْشٰۤى اَنْ تُصِيْبَنَا دَآئِرَةٌ ۭ فَعَسَى اللّٰهُ اَنْ يَّاْتِيَ بِالْفَتْحِ اَوْ اَمْرٍ مِّنْ عِنْدِهٖ فَيُصْبِحُوْا عَلٰى مَآ اَسَرُّوْا فِيْۤ اَنْفُسِهِمْ نٰدِمِيْنَ۔ “تو آپ دیکھتے ہیں کہ جن کے دلوں میں بیماری ہے وہ ان کی طرف لپکتے ہیں، کہتے ہیں: ہمیں ڈر ہے کہ ہم پر کوئی گردش نہ آ پڑے۔ تو قریب ہے کہ اللہ فتح لے آئے یا اپنی طرف سے کوئی اور بات، پھر یہ اس پر جو اپنے دلوں میں چھپائے بیٹھے تھے، پشیمان ہو کر رہ جائیں۔” مفسرین کے نزدیک یہ وہی منافقین ہیں جن کی زبان پر یہ جملہ تھا؛ دَآئِرَة، یعنی گردش، اس ڈر کا نام ہے کہ کہیں پانسہ پلٹ نہ جائے اور ہم ہارنے والوں کے ساتھ رہ جائیں۔ اہلِ معرفت کہتے ہیں کہ دل کی بیماری کی جڑ یہی ہے: بھروسا بٹا ہوا ہو، ایک ہاتھ اللہ کی رسی پر اور دوسرا کسی اور سہارے پر، کہ وقت پر کام آئے۔ ایسا دل دونوں طرف دوڑتا ہے اور کہیں نہیں پہنچتا۔ اور يُسَارِعُوْنَ فِيْهِمْ پر غور کیجیے: ان کی طرف نہیں، ان میں دوڑتے ہیں؛ گویا وہ ان کے اندر گھل مل جانا چاہتے ہیں۔ مفسرین کہتے ہیں کہ اللہ کی طرف سے عسیٰ، یعنی امید کا لفظ، یقینی وعدہ ہوتا ہے؛ اور یہ وعدہ پورا ہوا۔ فتح سے مراد مفسرین کے نزدیک مسلمانوں کی کامیابی ہے، اور اہلِ معرفت فتح کو دل کے کھلنے کے معنی میں بھی لیتے ہیں: جب اللہ کسی دل پر فتح کا دروازہ کھولتا ہے تو سارے چھپے ہوئے سہارے بے معنی ہو جاتے ہیں، اور جس نے انہیں تھاما تھا وہ اپنے چھپائے ہوئے پر پشیمان ہوتا ہے۔ ندامت کی جگہ اَسَرُّوْا بتائی گئی، یعنی جو انہوں نے دل میں چھپایا؛ چھپی ہوئی بات ہی آخر کار کھلتی ہے۔

اور جب پردہ اٹھے گا تو اہلِ ایمان حیرت سے کہیں گے: وَيَقُوْلُ الَّذِيْنَ اٰمَنُوْۤا اَهٰۤؤُلَآءِ الَّذِيْنَ اَقْسَمُوْا بِاللّٰهِ جَهْدَ اَيْمَانِهِمْ ۙ اِنَّهُمْ لَمَعَكُمْ ۭ حَبِطَتْ اَعْمَالُهُمْ فَاَصْبَحُوْا خٰسِرِيْنَ۔ “اور ایمان والے کہیں گے: کیا یہی وہ لوگ ہیں جو اللہ کی پکی پکی قسمیں کھا کر کہتے تھے کہ ہم تمہارے ساتھ ہیں؟ ان کے سب اعمال برباد ہو گئے، اور وہ خسارے میں رہ گئے۔” جَهْدَ اَيْمَانِهِمْ، یعنی قسم کھانے میں پورا زور لگا دینا؛ جتنی قسمیں سخت تھیں، اتنا ہی اندر کا کھوکھلا پن تھا، کیونکہ سچ کو اتنی قسموں کی ضرورت نہیں ہوتی۔ حَبِطَتْ کا لفظ اس جانور کے لیے بولا جاتا ہے جس کا پیٹ کوئی مضر گھاس کھا کر پھول جائے اور دیکھنے میں فربہ لگے مگر اندر سے بیمار ہو؛ ان کے اعمال بھی باہر سے بھرے پرے تھے اور اندر سے خالی۔ اہلِ معرفت کہتے ہیں کہ عمل کی جان نیت ہے؛ جو عمل اللہ کے لیے نہیں تھا وہ ایک صورت تھی بغیر روح کے، اور صورت بغیر روح کے باقی نہیں رہتی۔ شیخِ اکبر کے مکتب میں ہر عمل کی ایک صورت اور ایک روح مانی جاتی ہے، اور جزا اسی روح کے مطابق ملتی ہے؛ جس عمل کی روح اخلاص نہ ہو اس کی صورت حساب کے دن بکھر جاتی ہے۔ اور مَعَكُمْ، یعنی تمہارے ساتھ، کا دعویٰ دیکھیے: وہ ساتھ ہونے کی قسم کھاتے تھے، اور ساتھ وہ ہوتا ہے جس کا دل ساتھ ہو، صرف جسم نہیں۔ اس ایک آیت میں ہر دل کے لیے ایک آئینہ ہے: کہیں میری زبان کا ساتھ میرے دل کے ساتھ سے بڑا تو نہیں۔

اب وہ آیت جو اس رکوع کا دل ہے، اور جو ایک تنبیہ سے شروع ہو کر محبت کی سب سے بڑی بشارت تک پہنچتی ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا مَنْ يَّرْتَدَّ مِنْكُمْ عَنْ دِيْنِهٖ فَسَوْفَ يَاْتِي اللّٰهُ بِقَوْمٍ يُّحِبُّهُمْ وَيُحِبُّوْنَهٗۤ۔ “اے ایمان والو! تم میں سے جو کوئی اپنے دین سے پھر جائے تو اللہ عنقریب ایسے لوگ لے آئے گا جن سے وہ محبت کرتا ہوگا اور وہ اس سے محبت کرتے ہوں گے۔” پہلا سبق یہ ہے کہ اللہ کا دین کسی کا محتاج نہیں؛ جو پھرے وہ اپنا نقصان کرتا ہے، دین کا نہیں۔ اور آیت یہ نہیں کہتی کہ میں تمہیں ہلاک کر دوں گا؛ کہتی ہے کہ میں ان کو لاؤں گا جن سے میری محبت کا رشتہ ہوگا۔ یعنی دین سے پھرنے کا اصل خسارہ یہ ہے کہ آدمی اس محبت سے محروم رہ جائے۔ یہ لوگ کون ہیں؟ مفسرین نے کئی اقوال نقل کیے۔ امام حاکم رحمۃ اللہ علیہ وغیرہ کی روایت کا مفہوم ہے کہ جب یہ آیت اتری تو حضورِ اکرم صلی اللہ علیہ وسلم نے حضرت ابو موسیٰ اشعری رضی اللہ عنہ کی طرف اشارہ کر کے فرمایا کہ وہ اس کی قوم ہیں، یعنی یمن کے اشعری؛ اور صحیح بخاری کی حدیث کا مفہوم ہے کہ اہلِ یمن آئے ہیں، ان کے دل سب سے نرم ہیں، ایمان یمنی ہے اور حکمت یمنی ہے۔ اور حضرت حسن بصری رحمۃ اللہ علیہ وغیرہ سے منقول ہے کہ اس سے حضرت ابوبکر صدیق رضی اللہ عنہ اور ان کے ساتھی مراد ہیں، جنہوں نے حضورِ اکرم صلی اللہ علیہ وسلم کے وصال کے بعد ارتداد کے فتنے کا مقابلہ کیا۔ اہلِ علم کہتے ہیں کہ ان اقوال میں ٹکراؤ نہیں؛ آیت کی صفات جس میں پائی جائیں وہ اس بشارت کا حصہ دار ہے، اور یہ دروازہ قیامت تک کھلا ہے۔

اب يُحِبُّهُمْ وَيُحِبُّوْنَهٗ پر ٹھہریے، کیونکہ یہ قرآن کے ان مقامات میں سے ہے جہاں اللہ اور بندے کی محبت ایک ہی سانس میں آئی ہے۔ اہلِ معرفت پہلا نکتہ یہ نکالتے ہیں کہ اللہ کی محبت پہلے آئی اور بندوں کی بعد میں؛ بندہ محبت شروع نہیں کرتا، محبت کا جواب دیتا ہے۔ جس دل میں اللہ کی محبت جاگی، وہ اس بات کی علامت ہے کہ پہلے اللہ نے اس سے محبت کی تھی؛ بندے کی محبت اللہ کی محبت کا عکس ہے، جیسے آئینے کی روشنی سورج کی روشنی کا عکس ہے۔ شیخِ اکبر نے فتوحاتِ مکیہ میں محبت پر جو طویل کلام کیا ہے اس کا مفہوم یہ ہے کہ محبت کی حقیقت کو تعریف میں قید نہیں کیا جا سکتا، اسے چکھ کر ہی جانا جاتا ہے؛ اور سچا محب وہ ہے جس کی اپنی مرضی محبوب کی مرضی میں گم ہو جائے۔ حدیثِ قربِ نوافل کا مفہوم یہ ہے کہ بندہ فرائض کے بعد نوافل سے قریب ہوتا رہتا ہے یہاں تک کہ اللہ اس سے محبت کرنے لگتا ہے، اور جب وہ محبت کرتا ہے تو اس کا کان، اس کی آنکھ اور اس کا ہاتھ بن جاتا ہے جس سے وہ سنتا، دیکھتا اور پکڑتا ہے۔ اس مکتب میں یہ حلول یا اتحاد نہیں؛ اس کا مطلب یہ ہے کہ بندے کے اعضاء حق کے حکم کے سوا کسی اور چیز کے لیے حرکت نہیں کرتے، اور بندہ دیکھ لیتا ہے کہ اس کی ہر قوت اسی کی عطا ہے۔ اور صحیح بخاری کی حدیث کا مفہوم ہے کہ جب اللہ کسی بندے سے محبت کرتا ہے تو حضرت جبریل علیہ السلام کو پکارتا ہے کہ میں فلاں سے محبت کرتا ہوں تم بھی کرو، پھر آسمان والے اس سے محبت کرتے ہیں، پھر زمین میں اس کے لیے قبولیت رکھ دی جاتی ہے۔ محبت اوپر سے اترتی ہے اور نیچے پھیلتی ہے؛ اس آیت کے لوگ اسی اترتی ہوئی محبت کے برتن ہیں۔

اس محبت کا پہلا اثر کہاں ظاہر ہوتا ہے؟ آیت بتاتی ہے: اَذِلَّةٍ عَلَى الْمُؤْمِنِيْنَ اَعِزَّةٍ عَلَى الْكٰفِرِيْنَ۔ “ایمان والوں کے لیے نرم اور جھکے ہوئے، کافروں کے مقابلے میں مضبوط اور باوقار۔” عربی میں جھکنے کا یہ لفظ عام طور پر لام کے ساتھ آتا ہے، یہاں عَلٰى کے ساتھ آیا؛ مفسرین کہتے ہیں کہ اس میں شفقت کا معنی شامل ہے، جیسے کوئی اوپر سے جھک کر کسی کو اٹھاتا ہے۔ یعنی یہ ذلت کمزوری کی نہیں، مہربانی کی ہے۔ اور اَعِزَّة کا مطلب تکبر یا لوگوں پر سختی نہیں؛ اس کا مطلب ہے دشمنی پر اترے ہوئے مخالف کے سامنے نہ جھکنا، اپنے دین کو کسی کے قدموں میں نہ ڈالنا۔ یہ وہی بات ہے جو سورۂ الفتح میں صحابہ رضی اللہ عنہم کے بارے میں آئی: اَشِدَّآءُ عَلَى الْكُفَّارِ رُحَمَآءُ بَيْنَهُمْ۔ اور اس آیت کا پس منظر وہی جنگ کا زمانہ ہے جس کا ذکر رکوع کے شروع میں ہوا؛ عام انسانوں کے ساتھ برتاؤ کا قانون وہی ہے جو اوپر سورۂ الممتحنہ کی آیت میں گزرا۔ اہلِ معرفت کہتے ہیں کہ محبت کرنے والا اپنے محبوب کے اخلاق اختیار کرتا ہے: اللہ کے نام الرؤف اور الرحیم بھی ہیں اور العزیز بھی؛ جس سے وہ محبت کرے اس میں دونوں رنگ آتے ہیں، اور ہر رنگ اپنی جگہ ظاہر ہوتا ہے۔ شیخِ اکبر کے مکتب میں یہی جلال و جمال کا جمع ہے: جمال اپنوں کے لیے، جلال حق کی حفاظت کے لیے، اور دونوں ایک ہی محبت سے پھوٹتے ہیں۔ جو اپنوں پر سخت اور حق کے دشمن کے آگے نرم ہو، اس نے یہ ترتیب الٹ دی؛ اور یہ محبت کی نہیں، نفس کی علامت ہے۔

پھر ان کی دو اور صفتیں اور ایک خاتمہ: يُجَاهِدُوْنَ فِيْ سَبِيْلِ اللّٰهِ وَلَا يَخَافُوْنَ لَوْمَةَ لَآئِمٍ ۭ ذٰلِكَ فَضْلُ اللّٰهِ يُؤْتِيْهِ مَنْ يَّشَآءُ ۭ وَاللّٰهُ وَاسِعٌ عَلِيْمٌ۔ “وہ اللہ کی راہ میں جہاد کرتے ہیں اور کسی ملامت کرنے والے کی ملامت سے نہیں ڈرتے۔ یہ اللہ کا فضل ہے، جسے چاہتا ہے دیتا ہے، اور اللہ وسعت والا، سب کچھ جاننے والا ہے۔” اس آیت کے ظاہر میں جہاد سے مراد دین کی حفاظت کے لیے وہ جدوجہد ہے جو جائز قیادت کے تحت ہو، اور مفسرین نے اسے ارتداد کے مقابلے میں صحابہ رضی اللہ عنہم کی جدوجہد پر چسپاں کیا؛ اہلِ معرفت اس کے ساتھ نفس کے خلاف جہاد کو بھی اسی محبت کا پھل کہتے ہیں۔ ترتیب پر غور کیجیے: محبت پہلے، جہاد بعد میں؛ جو جہاد محبت سے نہ پھوٹے وہ غصہ ہے، اور جو محبت قربانی تک نہ پہنچے وہ صرف دعویٰ ہے۔ اور لَوْمَةَ لَآئِمٍ، یعنی ملامت کرنے والے کی ملامت؛ محب کو صرف محبوب کی رضا کی پروا ہوتی ہے۔ شیخِ اکبر کے مکتب میں ملامتیہ کا مقام بہت بلند سمجھا گیا ہے: وہ اہلِ اللہ جو اپنے احوال کو لوگوں سے چھپاتے ہیں اور لوگوں کی تعریف یا ملامت کو کوئی وزن نہیں دیتے، کیونکہ ان کی نگاہ اللہ پر جمی ہے۔ پھر فرمایا کہ یہ سب فضل ہے: محبت کمائی نہیں جاتی، عطا ہوتی ہے۔ مگر مَنْ يَّشَآءُ کو اس سلسلے کے اصول کے ساتھ پڑھیے: اللہ کی مشیت اندھی نہیں، وہ عَلِيْم ہے؛ وہ جانتا ہے کہ کون سا دل اس محبت کا طالب ہے، اور جو طالب ہو وہ محروم نہیں رہتا۔ اور وَاسِعٌ کہہ کر بتا دیا کہ اس کے فضل کا خزانہ تنگ نہیں؛ کوئی یہ نہ سمجھے کہ محبت کے سب حصے بٹ چکے۔

جب بتا دیا گیا کہ کس کو سرپرست نہیں بنانا، تو اب بتایا کہ اصل سرپرست کون ہے: اِنَّمَا وَلِيُّكُمُ اللّٰهُ وَرَسُوْلُهٗ وَالَّذِيْنَ اٰمَنُوا الَّذِيْنَ يُقِيْمُوْنَ الصَّلٰوةَ وَيُؤْتُوْنَ الزَّكٰوةَ وَهُمْ رٰكِعُوْنَ۔ “تمہارا دوست اور سرپرست تو بس اللہ ہے، اور اس کا رسول صلی اللہ علیہ وسلم، اور وہ ایمان والے جو نماز قائم کرتے ہیں اور زکوٰۃ دیتے ہیں اور وہ جھکنے والے ہیں۔” اس آیت کے شانِ نزول میں ایک روایت مشہور ہے: ابنِ ابی حاتم اور ابنِ مردویہ رحمہما اللہ وغیرہ کی روایات میں منقول ہے کہ مسجد میں ایک سائل نے سوال کیا اور حضرت علی رضی اللہ عنہ نے رکوع کی حالت میں اپنی انگوٹھی اسے دے دی، اور یہ آیت اتری۔ یہ روایت تفسیر کی کتابوں میں کئی طرق سے آئی ہے، اور حضرت علی رضی اللہ عنہ کی سخاوت اور نماز میں ان کا خشوع اپنی جگہ مسلم ہیں۔

اہلِ علم نے اس روایت پر کئی پہلوؤں سے گفتگو کی ہے۔ حافظ ابنِ کثیر رحمۃ اللہ علیہ وغیرہ نے لکھا ہے کہ اس روایت کی سندوں میں سے کوئی صحیح نہیں، ان میں کمزور اور نامعلوم راوی ہیں۔ دوسرا قول، جو امام ابنِ جریر طبری رحمۃ اللہ علیہ نے نقل کیا، یہ ہے کہ یہ آیت حضرت عبادہ بن صامت رضی اللہ عنہ کے بارے میں اتری جب انہوں نے پرانے حلیفوں سے براءت کر کے اللہ، اس کے رسول صلی اللہ علیہ وسلم اور مومنوں کو اپنا ولی بنایا؛ اور یوں یہ آیت رکوع کے پہلے حصے سے جڑ جاتی ہے۔ اور امام ابنِ جریر رحمۃ اللہ علیہ ہی نے امام ابو جعفر محمد باقر رحمۃ اللہ علیہ سے نقل کیا، جس کا مفہوم یہ ہے کہ ان سے پوچھا گیا کہ یہاں ایمان والوں سے کون مراد ہیں، تو فرمایا: ایمان والے؛ کہا گیا کہ لوگ کہتے ہیں یہ حضرت علی رضی اللہ عنہ ہیں، تو فرمایا: حضرت علی رضی اللہ عنہ بھی ایمان والوں میں سے ہیں۔ جمہور اہلِ سنت کا خلاصہ یہی ہے کہ آیت کا حکم عام ہے، کیونکہ اس میں صیغہ جمع کا ہے، اور اعتبار لفظ کے عموم کا ہوتا ہے نہ کہ خاص سبب کا؛ اور حضرت علی رضی اللہ عنہ بلاشبہ اس کے بلند ترین مصداقوں میں سے ہیں۔ وَهُمْ رٰكِعُوْنَ کے بارے میں بہت سے مفسرین، جیسے حافظ ابنِ کثیر رحمۃ اللہ علیہ، کہتے ہیں کہ یہ ان کی صفت ہے: وہ اللہ کے آگے جھکنے والے اور عاجزی کرنے والے ہیں، نہ یہ کہ زکوٰۃ رکوع ہی کی حالت میں دینا شرط ہو۔

اب اِنَّمَا پر غور کیجیے، جو حصر کا لفظ ہے: ولایت اصل میں اللہ ہی کی ہے، اور رسول صلی اللہ علیہ وسلم اور مومنوں کی ولایت اسی کی ولایت کا پرتو ہے۔ آیت میں وَلِيُّكُمُ واحد کا صیغہ ہے، جمع کا نہیں؛ اہلِ معرفت کہتے ہیں کہ اصل ولی ایک ہے، باقی سب اس کی ولایت کے مظاہر ہیں۔ الولی اللہ تعالیٰ کے اسماءِ حسنیٰ میں سے ہے؛ شیخِ اکبر کے نزدیک یہی سبب ہے کہ ولایت کا دائرہ کبھی بند نہیں ہوتا: نبوت اور رسالت اپنی ہر صورت میں حضورِ اکرم صلی اللہ علیہ وسلم پر ختم ہو گئی، مگر ولایت، جو آپ صلی اللہ علیہ وسلم کی پیروی کا نام ہے، جاری ہے، کیونکہ ولی اللہ کا نام ہے اور اس کا مظہر ہر زمانے میں رہتا ہے۔ اور اس ولایت کی پہچان کیا رکھی؟ نماز قائم کرنا، زکوٰۃ دینا اور جھکنا۔ نماز اللہ سے رشتہ ہے، زکوٰۃ بندوں سے رشتہ، اور رکوع دونوں کی روح، یعنی عاجزی۔ اہلِ ذوق کہتے ہیں کہ ولی وہ نہیں جو ہوا میں اڑے یا پانی پر چلے؛ ولی وہ ہے جو نماز میں حاضر ہو، مال میں سخی ہو اور اپنے آپ میں جھکا ہوا ہو۔ اور رکوع وہ حالت ہے جس میں سر جھک کر دل کے برابر آ جاتا ہے؛ اہلِ معرفت اس میں یہ اشارہ دیکھتے ہیں کہ عقل اپنا غرور چھوڑ کر دل کی سطح پر آ جائے، تب ولایت کا دروازہ کھلتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر میں رکوع نماز کا رکن ہے، اور باطن میں وہ پورے وجود کی عاجزی۔

اور رکوع کا خاتمہ ایک بشارت پر ہے: وَمَنْ يَّتَوَلَّ اللّٰهَ وَرَسُوْلَهٗ وَالَّذِيْنَ اٰمَنُوْا فَاِنَّ حِزْبَ اللّٰهِ هُمُ الْغٰلِبُوْنَ۔ “اور جو اللہ کو، اس کے رسول صلی اللہ علیہ وسلم کو اور ایمان والوں کو اپنا ولی بنائے، تو یقیناً اللہ کی جماعت ہی غالب رہنے والی ہے۔” رکوع کے شروع میں منافقین کو گردش کا ڈر تھا، اور وہ غالب آنے والے کے ساتھ رہنا چاہتے تھے؛ آخر میں بتا دیا کہ غالب کون ہے۔ جو غلبے کی تلاش میں غیر کے پاس گیا، اس نے غلبے کو غلط جگہ ڈھونڈا۔ سورۂ المجادلہ کے صفحے پر حزب اللہ کا ذکر اسی شان سے گزرا: اَلَآ اِنَّ حِزْبَ اللّٰهِ هُمُ الْمُفْلِحُوْنَ، “سن لو! اللہ کی جماعت ہی کامیاب ہونے والی ہے۔” یہاں غالب کہا، وہاں کامیاب؛ اہلِ معرفت کہتے ہیں کہ ایک ظاہر کا غلبہ ہے اور ایک باطن کی فلاح، اور اللہ کی جماعت کو دونوں کا وعدہ ہے، اپنے اپنے وقت پر۔ غلبہ کبھی میدان میں ظاہر ہوتا ہے اور کبھی صرف دل میں، جہاں بندہ اپنے نفس پر غالب آ جاتا ہے؛ اور یہ دوسرا غلبہ پہلے کی بنیاد ہے۔ شیخِ اکبر کے مکتب میں یہ نکتہ بھی کہا جاتا ہے کہ حزب اللہ کا غلبہ دراصل اللہ ہی کا غلبہ ہے، کیونکہ عزت اور غلبہ اسی کا ہے؛ بندہ جب اس کی جماعت میں داخل ہوتا ہے تو اپنی قوت چھوڑ کر اس کی قوت میں آ جاتا ہے، اور یہ وہ قوت ہے جسے کوئی گردش نہیں پلٹ سکتی۔ یوں رکوع کا دائرہ مکمل ہوا: جس نے ولایت کے لیے اللہ کو چنا، اس کے لیے کسی گردش کا خوف باقی نہیں رہتا، اور محبت کا وہ دروازہ جو يُحِبُّهُمْ وَيُحِبُّوْنَهٗ میں کھلا تھا، اسی جماعت کے لیے کھلا ہے۔
