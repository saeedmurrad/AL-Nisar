یہ سورت قرآنِ مجید کے آخر میں نازل ہونے والے حصوں میں سے ہے، اور اس کا پہلا ہی جملہ پوری سورت کا عنوان بن گیا: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْۤا اَوْفُوْا بِالْعُقُوْدِ۔ “اے ایمان والو! اپنے عہدوں کو پورا کرو۔” عُقود عقد کی جمع ہے، اور عقد کا اصل معنی گرہ باندھنا ہے، یعنی رسی کے دو سروں کو اس طرح جوڑ دینا کہ کھینچنے سے کھل نہ سکیں۔ حضرت عبد اللہ بن عباس رضی اللہ عنہما سے منقول ہے کہ عقود سے مراد وہ سب کچھ ہے جو اللہ نے حلال کیا، حرام کیا، فرض کیا اور اپنی کتاب میں جس کی حدیں مقرر کیں۔ مفسرین اس میں بندوں کے آپس کے سب بندھن بھی داخل کرتے ہیں: نکاح، خرید و فروخت، قسم، وعدہ، امانت، یہاں تک کہ کسی دوسری قوم سے کیا ہوا معاہدہ بھی۔ خطاب “الَّذِيْنَ اٰمَنُوْۤا” سے ہے، گویا ایمان خود وہ پہلی گرہ ہے جس پر باقی سب گرہیں ٹکی ہیں؛ عقیدہ کا لفظ بھی اسی مادے سے ہے، یعنی وہ بات جو دل پر باندھ دی گئی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں سب سے پہلی گرہ وہ ہے جو روحوں نے اس وقت باندھی جب ان سے پوچھا گیا: اَلَسْتُ بِرَبِّكُمْ ؕ قَالُوْا بَلٰى۔ اہلِ معرفت کہتے ہیں کہ دنیا کے سارے سچے عہد اسی پہلے اقرار کی بازگشت ہیں؛ جو بندہ کسی انسان سے کیا ہوا وعدہ نبھاتا ہے وہ دراصل اسی ازلی اقرار کی حفاظت کرتا ہے، اور جو وعدہ توڑتا ہے وہ اپنی فطرت کی پہلی گرہ ڈھیلی کرتا ہے۔

پہلی گرہ کے بعد پہلی عطا کا ذکر آیا، تاکہ معلوم ہو کہ عہد بوجھ نہیں، نعمت کا دروازہ ہے: اُحِلَّتْ لَكُمْ بَهِيْمَةُ الْاَنْعَامِ اِلَّا مَا يُتْلٰى عَلَيْكُمْ غَيْرَ مُحِلِّي الصَّيْدِ وَاَنْتُمْ حُرُمٌ ؕ اِنَّ اللّٰهَ يَحْكُمُ مَا يُرِيْدُ۔ “تمہارے لیے مویشی چوپائے حلال کیے گئے، سوائے ان کے جن کا حکم تمہیں پڑھ کر سنایا جائے گا، مگر احرام کی حالت میں شکار کو حلال نہ سمجھنا؛ بے شک اللہ جو چاہتا ہے حکم فرماتا ہے۔” ظاہر معنی یہ ہے کہ اونٹ، گائے، بکری اور ان جیسے چوپائے حلال ہیں، جن کا استثنا ہے وہ تیسری آیت میں آ رہا ہے، اور جو حج یا عمرے کے احرام میں ہو وہ خشکی کا شکار نہ کرے۔ اہلِ ذوق کہتے ہیں کہ احرام صرف دو چادروں کا نام نہیں، حضوری کی ایک حالت ہے: جو رب کے گھر کی طرف چل پڑا اس کا کام اب پیچھا کرنا نہیں، حاضر ہونا ہے۔ شکار طلب کی علامت ہے، اور احرام میں طلب کا رخ بدل جاتا ہے؛ جو کل ہرن کے پیچھے دوڑتا تھا آج خود بلایا ہوا مہمان ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ پھر “اِنَّ اللّٰهَ يَحْكُمُ مَا يُرِيْدُ” آیا۔ شیخِ اکبر کے نزدیک ارادہ علم کے تابع ہے اور علم معلوم کے تابع؛ یعنی حق کا حکم اندھا جبر نہیں، ہر شے کی اپنی حقیقت کے عین مطابق ہے۔ یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ اسی لیے حلال و حرام کی فہرست انسان کی اپنی حقیقت کا آئینہ ہے؛ جو اس کی طہارت کے موافق تھا وہ کھول دیا گیا اور جو اس کے حال کے منافی تھا وہ روک لیا گیا۔

دوسری آیت عہد کے دائرے کو زمانے، جگہ اور جانور تک پھیلا دیتی ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تُحِلُّوْا شَعَآئِرَ اللّٰهِ وَلَا الشَّهْرَ الْحَرَامَ وَلَا الْهَدْيَ وَلَا الْقَلَآئِدَ وَلَاۤ اٰمِّيْنَ الْبَيْتَ الْحَرَامَ يَبْتَغُوْنَ فَضْلًا مِّنْ رَّبِّهِمْ وَرِضْوَانًا ؕ وَاِذَا حَلَلْتُمْ فَاصْطَادُوْا۔ “اے ایمان والو! اللہ کی نشانیوں کی بے حرمتی نہ کرو، نہ حرمت والے مہینے کی، نہ قربانی کے جانوروں کی، نہ ان جانوروں کی جن کے گلے میں پٹے ڈالے گئے ہوں، اور نہ ان لوگوں کی جو اپنے رب کا فضل اور اس کی خوشنودی ڈھونڈتے ہوئے بیت الحرام کا قصد کیے جا رہے ہوں؛ اور جب احرام کھول دو تو شکار کر سکتے ہو۔” شعائر شعیرہ کی جمع ہے، یعنی وہ علامت جس سے کسی حقیقت کا شعور ہو۔ قلائد وہ پٹے تھے جو قربانی کے جانوروں کے گلے میں ڈالے جاتے تاکہ کوئی ان سے تعرض نہ کرے۔ سورۂ الحج کے صفحے پر گزر چکا: وَمَنْ يُّعَظِّمْ شَعَآئِرَ اللّٰهِ فَاِنَّهَا مِنْ تَقْوَى الْقُلُوْبِ۔ شیخِ اکبر کے مکتب میں کائنات کی ہر شے کسی اسمِ الٰہی کا مظہر ہے، اور عارف مظاہر کا ادب اس لیے کرتا ہے کہ وہ ظاہر کرنے والے کی طرف اشارہ کرتے ہیں۔ اور غور طلب بات یہ ہے کہ قرآن نے ان مسافروں کی حرمت کو بھی شعائر کے ساتھ جوڑا جو “يَبْتَغُوْنَ فَضْلًا مِّنْ رَّبِّهِمْ وَرِضْوَانًا” کی کیفیت میں تھے؛ مفسرین کے مطابق اس وقت ان میں وہ مشرک بھی تھے جو ابھی ایمان نہیں لائے تھے، اور حرم سے متعلق آخری احکام بعد میں نازل ہوئے۔ اہلِ معرفت یہاں وہ اصل دیکھتے ہیں جس پر یہ سلسلہ قائم ہے: طلب کرنے والے کی طلب کو حق نے حرمت دی، کیونکہ وہ کسی طالب کو خود دھکیل کر محروم نہیں کرتا۔

پھر وہ حکم آیا جو انسانی نفس کے سب سے نازک زخم پر ہاتھ رکھتا ہے: وَلَا يَجْرِمَنَّكُمْ شَنَاٰنُ قَوْمٍ اَنْ صَدُّوْكُمْ عَنِ الْمَسْجِدِ الْحَرَامِ اَنْ تَعْتَدُوْا۔ “اور کسی قوم کی دشمنی، اس وجہ سے کہ انہوں نے تمہیں مسجدِ حرام سے روکا تھا، تمہیں اس بات پر نہ ابھارے کہ تم زیادتی کرو۔” یہ اشارہ حدیبیہ کے سال کی طرف ہے، جب قریش نے حضورِ اکرم صلی اللہ علیہ وسلم اور آپ صلی اللہ علیہ وسلم کے صحابہ کو عمرے سے روک دیا تھا۔ زخم تازہ تھا، اور زخمی دل بدلے کو دین کا نام دینے میں جلدی کرتا ہے؛ قرآن نے یہیں روک لگا دی۔ جنہیں حرم سے روکا گیا تھا انہیں یہ حکم ملا کہ وہ کسی اور کے لیے روک نہ بنیں۔ اہلِ معرفت کہتے ہیں کہ نفرت ایک حجاب ہے جو سامنے والے کی پوری حقیقت کو ایک زخم میں سمیٹ دیتا ہے؛ پھر آدمی اس شخص کو نہیں دیکھتا، صرف اپنا درد دیکھتا ہے۔ شیخِ اکبر کے نزدیک ہر انسان کی ایک عینِ ثابتہ ہے جو حق کے علم میں ہے، اور اس کا ایک ربِّ خاص ہے جو اس کی پرورش کر رہا ہے؛ جو اس حقیقت کو یاد رکھے وہ دشمن کے ساتھ بھی حد سے نہیں بڑھتا، کیونکہ وہ جانتا ہے کہ وہ کسی اور کے پروردہ پر ہاتھ اٹھا رہا ہے۔ یہ بات اسی سورت کی آٹھویں آیت میں عدل کے حکم کے ساتھ پھر دہرائی جائے گی، اور اس تکرار میں خود ایک تاکید ہے۔

اب وہ جملہ آتا ہے جو امت کے اجتماعی وجود کا دستور ہے: وَتَعَاوَنُوْا عَلَى الْبِرِّ وَالتَّقْوٰى ۖ وَلَا تَعَاوَنُوْا عَلَى الْاِثْمِ وَالْعُدْوَانِ ۖ وَاتَّقُوا اللّٰهَ ؕ اِنَّ اللّٰهَ شَدِيْدُ الْعِقَابِ۔ “اور نیکی اور تقویٰ میں ایک دوسرے کی مدد کرو، اور گناہ اور زیادتی میں ایک دوسرے کی مدد نہ کرو، اور اللہ سے ڈرو؛ بے شک اللہ سخت سزا دینے والا ہے۔” علما کہتے ہیں کہ بِرّ بھلائی کرنے کا نام ہے اور تقویٰ برائی سے بچنے کا؛ یوں دونوں مل کر پورا دین بن جاتے ہیں۔ تعاون کا معیار رشتہ یا گروہ نہیں، عمل ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ اپنے بھائی کی مدد کرو خواہ وہ ظالم ہو یا مظلوم؛ صحابہ نے پوچھا کہ ظالم کی مدد کیسے؟ تو آپ صلی اللہ علیہ وسلم نے فرمایا کہ اسے ظلم سے روک دو (بخاری)۔ یعنی ظالم کی حقیقی مدد اس کا ہاتھ پکڑنا ہے۔ شیخِ اکبر کے مکتب میں تقویٰ کا اصل وقایہ ہے، یعنی ڈھال؛ اور اس مکتب میں اس کی ایک لطیف صورت یہ بیان کی جاتی ہے کہ بندہ اپنی ذات کو حق کے لیے ڈھال بنائے کہ عیب اپنی طرف منسوب کرے، اور حق کو اپنے لیے ڈھال بنائے کہ ہر خیر اسی کی طرف لوٹائے۔ اہلِ معرفت کہتے ہیں کہ اسماءِ الٰہیہ ایک دوسرے کے مخالف نہیں، ایک دوسرے کی تکمیل کرتے ہیں؛ امت کے افراد انہی مختلف اسماء کے مظاہر ہیں، اس لیے ان کا تعاون اسی ہم آہنگی کا عکس ہے۔ اور “شَدِيْدُ الْعِقَابِ” یاد دلاتا ہے کہ گناہ میں تعاون کا انجام خود اس عمل کے اندر چھپا ہوا ہے۔

اب وہ استثنا کھلتا ہے جس کا وعدہ “اِلَّا مَا يُتْلٰى عَلَيْكُمْ” میں ہوا تھا۔ تیسری آیت اس سورت کی طویل آیتوں میں سے ہے اور اس میں تین جہان سمٹے ہوئے ہیں: کھانے کے احکام، دین کا کمال، اور مجبور پر رحمت۔ پہلا حصہ یہ ہے: حُرِّمَتْ عَلَيْكُمُ الْمَيْتَةُ وَالدَّمُ وَلَحْمُ الْخِنْزِيْرِ وَمَآ اُهِلَّ لِغَيْرِ اللّٰهِ بِهٖ وَالْمُنْخَنِقَةُ وَالْمَوْقُوْذَةُ وَالْمُتَرَدِّيَةُ وَالنَّطِيْحَةُ وَمَآ اَكَلَ السَّبُعُ اِلَّا مَا ذَكَّيْتُمْ۔ “تم پر حرام کیا گیا مردار، اور خون، اور سور کا گوشت، اور وہ جانور جس پر اللہ کے سوا کسی اور کا نام پکارا گیا ہو، اور جو گلا گھٹنے سے مرا ہو، اور جو چوٹ کھا کر مرا ہو، اور جو اونچائی سے گر کر مرا ہو، اور جو سینگ لگنے سے مرا ہو، اور جسے درندے نے کھایا ہو، سوائے اس کے جسے تم مرنے سے پہلے ذبح کر لو۔” جمہور مفسرین کے نزدیک “اِلَّا مَا ذَكَّيْتُمْ” کا تعلق گلا گھٹنے والے سے لے کر درندے کے زخمی کیے ہوئے تک سب سے ہے؛ اگر جان باقی تھی اور شرعی ذبح ہو گیا تو جانور حلال ہے۔ اہلِ معرفت اس فہرست میں ایک باطنی ترتیب دیکھتے ہیں: حرام وہ ہے جس سے زندگی اللہ کے نام کے بغیر رخصت ہوئی، یا کسی اور کے نام پر نکالی گئی۔ شیخِ اکبر کے مکتب میں ہر وجود نفَسِ رحمانی کی ایک سانس ہے؛ جو جان اسی کا نام لے کر لوٹائی گئی وہ انسان کی غذا بنتی ہے۔ “اُهِلَّ” میں اہلال کا لفظ ہے، یعنی آواز بلند کرنا؛ ایسے جانور کی حرمت کا سبب اس کا گوشت نہیں، وہ آواز ہے جو اس پر بلند کی گئی۔ اہلِ ذوق اسی سے سیکھتے ہیں کہ دل کی غذا بھی دیکھ کر لی جائے: جو خیال اور جو صحبت حق کے نام سے خالی ہو وہ مردار کی طرح ہے، چاہے دیکھنے میں تازہ لگے۔

آیت آگے بڑھتی ہے: وَمَا ذُبِحَ عَلَى النُّصُبِ وَاَنْ تَسْتَقْسِمُوْا بِالْاَزْلَامِ ؕ ذٰلِكُمْ فِسْقٌ۔ “اور جو آستانوں پر ذبح کیا گیا ہو، اور یہ کہ تم تیروں کے ذریعے قسمت معلوم کرو؛ یہ سب نافرمانی ہے۔” نُصُب وہ پتھر تھے جن پر جاہلیت میں قربانیاں چڑھائی جاتی تھیں، اور اَزلام وہ تیر جن پر کرنے اور نہ کرنے کا نشان ہوتا، اور عرب سفر، نکاح یا کسی بڑے کام سے پہلے ان سے فال نکالتے تھے۔ دونوں کا ایک ساتھ ذکر اس لیے ہے کہ دونوں میں انسان اپنی باگ غیر کے ہاتھ میں دے دیتا ہے: ایک میں جان پتھر کے حوالے، دوسرے میں ارادہ تیر کے حوالے۔ اس کے بدلے حضورِ اکرم صلی اللہ علیہ وسلم اپنے صحابہ کو استخارہ اس اہتمام سے سکھاتے تھے جیسے قرآن کی سورت سکھاتے تھے (بخاری)؛ یعنی فال کی جگہ دعا، اور اتفاق کی جگہ علیم و قدیر کی طرف رجوع۔ پھر یہ اعلان ہوا: اَلْيَوْمَ يَئِسَ الَّذِيْنَ كَفَرُوْا مِنْ دِيْنِكُمْ فَلَا تَخْشَوْهُمْ وَاخْشَوْنِ۔ “آج کافر تمہارے دین کی طرف سے مایوس ہو گئے، پس ان سے نہ ڈرو اور مجھ سے ڈرو۔” اہلِ معرفت کہتے ہیں کہ “فَلَا تَخْشَوْهُمْ وَاخْشَوْنِ” میں توحید کا ایک پورا باب ہے: خوف کا بکھر جانا دل کی تقسیم ہے، اور جب خوف ایک طرف جمع ہو جائے تو باقی سب خوف خود جھڑ جاتے ہیں۔ شیخِ اکبر کے نزدیک خشیت جلال کی وہ ہیبت ہے جو معرفت کے ساتھ آتی ہے؛ اسی لیے یہ خوف دور نہیں کرتا، قریب کرتا ہے۔

اور اب وہ جملہ جس کی گونج تاریخ میں کبھی مدھم نہیں ہوئی: اَلْيَوْمَ اَكْمَلْتُ لَكُمْ دِيْنَكُمْ وَاَتْمَمْتُ عَلَيْكُمْ نِعْمَتِيْ وَرَضِيْتُ لَكُمُ الْاِسْلَامَ دِيْنًا۔ “آج میں نے تمہارے لیے تمہارا دین کامل کر دیا، اور تم پر اپنی نعمت پوری کر دی، اور تمہارے لیے اسلام کو دین کے طور پر پسند کر لیا۔” صحیح بخاری میں ہے کہ ایک یہودی نے حضرت عمر رضی اللہ عنہ سے کہا: اے امیر المؤمنین! تمہاری کتاب میں ایک آیت ہے، اگر وہ ہم پر نازل ہوتی تو ہم اس دن کو عید بنا لیتے۔ آپ رضی اللہ عنہ نے پوچھا کہ کون سی؟ اس نے یہی آیت پڑھی۔ حضرت عمر رضی اللہ عنہ نے فرمایا کہ ہم وہ دن بھی جانتے ہیں اور وہ جگہ بھی جہاں یہ حضورِ اکرم صلی اللہ علیہ وسلم پر نازل ہوئی: آپ صلی اللہ علیہ وسلم جمعہ کے دن عرفات میں کھڑے تھے۔ گویا وہ دن پہلے ہی دو عیدوں کا سنگم تھا، جمعہ کا دن اور عرفہ کا دن۔ حجۃ الوداع کا یہ لمحہ دعوت کے تئیس برسوں کا خلاصہ تھا؛ بہت سے مفسرین کے مطابق اس کے بعد حلال و حرام کا کوئی بڑا حکم نازل نہیں ہوا، اور چند ماہ بعد آپ صلی اللہ علیہ وسلم رفیقِ اعلیٰ سے جا ملے۔ سورۂ النصر کے صفحے پر گزر چکا کہ حضرت عبد اللہ بن عباس رضی اللہ عنہما اور حضرت عمر رضی اللہ عنہ نے فتح کی بشارت میں رخصت کا اشارہ بھی سن لیا تھا؛ یہاں بھی کمال کا اعلان وداع کی تمہید تھا، کیونکہ جب عمارت مکمل ہو جائے تو معمار کا کام پورا ہو جاتا ہے۔ اس لمحے میں خوشی اور جدائی ایک ساتھ اتری تھیں۔

شیخِ اکبر کے مکتب میں اس آیت کے نزول کی جگہ خود ایک اشارہ ہے۔ عرفہ اور معرفت ایک ہی مادے سے ہیں؛ دین کا کمال معرفت کے دن، معرفت کے میدان میں اترا، گویا بتا دیا گیا کہ دین کی تکمیل احکام کی گنتی پوری ہونے کا نام نہیں، اس پہچان کا نام ہے جس پر احکام کھڑے ہیں۔ حکمتِ فردیہ کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم اس نوعِ انسانی میں کامل ترین مظہر ہیں، اسی لیے امر آپ صلی اللہ علیہ وسلم سے شروع ہوا اور آپ صلی اللہ علیہ وسلم پر ختم ہوا؛ پس جس دین کے لانے والے کامل ہوں اس کا کمال اسی نسبت کا ظہور ہے۔ اہلِ معرفت ایک نازک فرق بھی بتاتے ہیں: “اَكْمَلْتُ” کا تعلق دین سے ہے اور “اَتْمَمْتُ” کا نعمت سے۔ دین کی صورت مکمل ہو گئی، اب اس میں نہ کچھ بڑھایا جا سکتا ہے نہ گھٹایا جا سکتا ہے؛ لیکن نعمت کا اتمام بندے پر ہوتا ہے، اور بندے کے اندر اس کا ظہور ہر لمحہ نیا ہے۔ تجددِ امثال کے اصول پر متن ٹھہرا ہوا ہے مگر قلب پر اس کی تجلی ہر سانس تازہ؛ اسی لیے کمالِ دین کے بعد نئے دین کا دروازہ بند ہے مگر فہم کا دروازہ کھلا ہے۔ پھر “وَرَضِيْتُ لَكُمُ الْاِسْلَامَ دِيْنًا” میں سب سے بڑی بشارت ہے: حق نے اپنی رضا کو اس دین کے ساتھ باندھ دیا۔ جو بندہ اسے دل سے قبول کرتا ہے وہ رضا کے دائرے میں داخل ہو جاتا ہے، اور اسلام کا معنی ہی سپردگی ہے؛ جو سپرد ہوا وہ اسی سپردگی میں ہدایت پا گیا۔

کمال کے اعلان کے فوراً بعد آیت کا آخری حصہ آتا ہے: فَمَنِ اضْطُرَّ فِيْ مَخْمَصَةٍ غَيْرَ مُتَجَانِفٍ لِّاِثْمٍ ۙ فَاِنَّ اللّٰهَ غَفُوْرٌ رَّحِيْمٌ۔ “پھر جو شخص بھوک کی شدت میں مجبور ہو جائے، اس حال میں کہ گناہ کی طرف مائل نہ ہو، تو بے شک اللہ بخشنے والا، رحم فرمانے والا ہے۔” یہ ترتیب اتفاقی نہیں؛ کامل دین وہی ہے جس میں ٹوٹے ہوئے انسان کی جگہ ہو۔ مخمصہ پیٹ کے اندر دھنس جانے کو کہتے ہیں، یعنی ایسی بھوک جو جان کے لیے خطرہ بن جائے؛ اس حال میں حرام بھی بقدرِ ضرورت روا ہو جاتا ہے، بشرطیکہ دل گناہ کی طرف جھکا ہوا نہ ہو۔ “غَيْرَ مُتَجَانِفٍ لِّاِثْمٍ” میں سارا فیصلہ نیت کے سپرد ہے: ایک ہی لقمہ ایک دل کے لیے رخصت ہے اور دوسرے کے لیے بہانہ۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا ایک روشن ظہور ہے: حرمت کا حکم ایک حد پر رک جاتا ہے، رحمت کا حکم اس حد کے پار بھی پہنچتا ہے۔ اہلِ ذوق کہتے ہیں کہ جو رب مجبور کے لیے اپنے حرام کو کھول دیتا ہے وہ اس طالب کے لیے اپنا دروازہ کیسے بند کرے گا جو مجبور ہو کر اس کی طرف آئے؛ اضطرار وہ کنجی ہے جس سے بہت سے بند دروازے کھلتے ہیں۔

پھر صحابہ نے پوچھا کہ ان کے لیے کیا حلال ہے، اور جواب آیا: يَسْـَٔلُوْنَكَ مَاذَاۤ اُحِلَّ لَهُمْ ؕ قُلْ اُحِلَّ لَكُمُ الطَّيِّبٰتُ ۙ وَمَا عَلَّمْتُمْ مِّنَ الْجَوَارِحِ مُكَلِّبِيْنَ تُعَلِّمُوْنَهُنَّ مِمَّا عَلَّمَكُمُ اللّٰهُ فَكُلُوْا مِمَّاۤ اَمْسَكْنَ عَلَيْكُمْ وَاذْكُرُوا اسْمَ اللّٰهِ عَلَيْهِ ۖ وَاتَّقُوا اللّٰهَ ؕ اِنَّ اللّٰهَ سَرِيْعُ الْحِسَابِ۔ “وہ آپ سے پوچھتے ہیں کہ ان کے لیے کیا حلال کیا گیا؟ فرما دیجیے: تمہارے لیے سب پاکیزہ چیزیں حلال کی گئیں، اور ان شکاری جانوروں کا شکار بھی جنہیں تم نے سدھایا ہو، تم انہیں اس علم میں سے سکھاتے ہو جو اللہ نے تمہیں سکھایا؛ پس جو وہ تمہارے لیے روک رکھیں اسے کھاؤ اور اس پر اللہ کا نام لو، اور اللہ سے ڈرو؛ بے شک اللہ جلد حساب لینے والا ہے۔” اصل قاعدہ “اُحِلَّ لَكُمُ الطَّيِّبٰتُ” ہے: حلال کی وسعت اصل ہے اور حرام استثنا۔ حضرت عدی بن حاتم رضی اللہ عنہ نے شکاری کتے کے بارے میں پوچھا تو حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ جب تم سدھائے ہوئے کتے کو اللہ کا نام لے کر چھوڑو تو جو وہ تمہارے لیے روک رکھے اسے کھاؤ، اور اگر اس نے خود کھا لیا تو نہ کھاؤ، کیونکہ اندیشہ ہے کہ اس نے اپنے لیے روکا (بخاری، مسلم)۔ اہلِ ذوق اس میں نفس کی تربیت کی ایک تمثیل دیکھتے ہیں: نفس ایک شکاری قوت ہے؛ شریعت سے سدھایا ہوا ہو تو جو پکڑتا ہے اپنے مالک کے لیے روکتا ہے، اور اپنے لیے کھانے لگے تو وہی کمائی ناپاک ہو جاتی ہے۔ اور “مِمَّا عَلَّمَكُمُ اللّٰهُ” بتاتا ہے کہ شیخِ اکبر کے نزدیک ہر علم اسمِ علیم کا فیض ہے؛ سکھانے والا بھی شاگرد ہی رہتا ہے۔

پانچویں آیت میں “اَلْيَوْمَ” تیسری بار آیا: اَلْيَوْمَ اُحِلَّ لَكُمُ الطَّيِّبٰتُ ؕ وَطَعَامُ الَّذِيْنَ اُوْتُوا الْكِتٰبَ حِلٌّ لَّكُمْ ۖ وَطَعَامُكُمْ حِلٌّ لَّهُمْ۔ “آج تمہارے لیے سب پاکیزہ چیزیں حلال کر دی گئیں، اور اہلِ کتاب کا کھانا تمہارے لیے حلال ہے اور تمہارا کھانا ان کے لیے حلال ہے۔” پہلے کافروں کی مایوسی کا دن، پھر دین کے کمال کا دن، اور اب حلال کی وسعت کا دن۔ اہلِ معرفت کہتے ہیں کہ کمال کا پہلا پھل وسعت ہے؛ جو دین کامل ہو وہ تنگ دل نہیں ہوتا۔ ظاہر حکم یہ ہے کہ یہود و نصاریٰ کا ذبیحہ، فقہا کی بیان کردہ شرائط کے ساتھ، مسلمان کے لیے حلال ہے، اور مسلمان انہیں اپنا کھانا کھلا سکتا ہے۔ عرب میں ایک دسترخوان پر بیٹھنا امن اور اعتماد کی سب سے بڑی علامت تھا، اور قرآن نے یہ دروازہ کھلا رکھا۔ یہی آیت بتاتی ہے کہ آگے اس سورت میں دوستی کے بارے میں جو تنبیہ آنے والی ہے اس کا تعلق جنگ کے زمانے کی سیاسی وفاداری سے ہے، انصاف، حسنِ سلوک اور لین دین سے نہیں؛ اور سورۂ الممتحنہ کے صفحے پر گزر چکا کہ جو لڑائی پر نہیں اترے ان کے بارے میں فرمایا: اَنْ تَبَرُّوْهُمْ وَتُقْسِطُوْۤا اِلَيْهِمْ۔ شیخِ اکبر کے مکتب میں رحمت ہر شے پر وسیع ہے، اور شریعت کی یہ وسعت اسی رحمت کا عکس ہے؛ جو ہاتھ ایک دسترخوان پر بڑھتے ہیں ان کے درمیان عداوت کی دیوار خود بخود کمزور پڑ جاتی ہے۔

آیت نکاح کی طرف آتی ہے: وَالْمُحْصَنٰتُ مِنَ الْمُؤْمِنٰتِ وَالْمُحْصَنٰتُ مِنَ الَّذِيْنَ اُوْتُوا الْكِتٰبَ مِنْ قَبْلِكُمْ اِذَاۤ اٰتَيْتُمُوْهُنَّ اُجُوْرَهُنَّ مُحْصِنِيْنَ غَيْرَ مُسٰفِحِيْنَ وَلَا مُتَّخِذِيْۤ اَخْدَانٍ ؕ وَمَنْ يَّكْفُرْ بِالْاِيْمَانِ فَقَدْ حَبِطَ عَمَلُهٗ وَهُوَ فِي الْاٰخِرَةِ مِنَ الْخٰسِرِيْنَ۔ “اور پاک دامن مومن عورتیں، اور تم سے پہلے کتاب دیے جانے والوں کی پاک دامن عورتیں بھی تمہارے لیے حلال ہیں، جب تم ان کے مہر انہیں ادا کر دو، نکاح کی حفاظت میں لاتے ہوئے، نہ بدکاری کرتے ہوئے اور نہ چھپی آشنائیاں بناتے ہوئے؛ اور جو ایمان سے انکار کرے اس کا عمل ضائع ہو گیا، اور وہ آخرت میں نقصان اٹھانے والوں میں سے ہوگا۔” دونوں گروہوں کی خواتین کو ایک ہی وصف “الْمُحْصَنٰتُ” سے یاد کیا گیا، اور یہی ان کی عزت کا عنوان ہے؛ مہر کو “اُجُوْرَهُنَّ” کہا گیا، یعنی عورت کا اپنا حق، اس کی قیمت نہیں۔ فقہا نے اس نکاح کو جائز کہا ہے اور ساتھ یہ بھی بتایا کہ گھر کے ماحول اور اولاد کے دین کی ذمہ داری اس اجازت کے ساتھ جڑی ہوئی ہے۔ آخری جملہ اصول بیان کرتا ہے: عمل کا ضائع ہونا کوئی اوپر سے ڈالا ہوا جبر نہیں؛ اعمال ایمان کی جڑ پر کھڑے درخت ہیں، اور جو خود جڑ کاٹ دے اس کے درخت کا سوکھنا اسی کے انکار کا نتیجہ ہے۔ سرِّ قدر کا یہی خلاصہ ہے: حق بندے کے اپنے اختیار کیے ہوئے رخ کو وجود دیتا ہے، اور جو ایمان کی طرف رخ کرے اسے کبھی محروم نہیں کرتا۔
