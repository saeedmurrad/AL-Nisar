پچھلے رکوع میں بنی اسرائیل کا وہ واقعہ گزرا جب ایک گروہ نے ارضِ مقدس میں داخلے سے انکار کیا اور چالیس برس بیابان میں بھٹکتا رہا؛ اب قرآن انسانی تاریخ کے پہلے ناحق خون کی طرف پلٹتا ہے، گویا یہ دکھانا مقصود ہے کہ حسد اور سرکشی کی جڑ کتنی پرانی ہے: وَاتْلُ عَلَيْهِمْ نَبَاَ ابْنَيْ اٰدَمَ بِالْحَقِّ ۘ اِذْ قَرَّبَا قُرْبَانًا فَتُقُبِّلَ مِنْ اَحَدِهِمَا وَلَمْ يُتَقَبَّلْ مِنَ الْاٰخَرِ۔ “اور انہیں آدم علیہ السلام کے دو بیٹوں کی خبر ٹھیک ٹھیک سنا دیجیے، جب دونوں نے ایک ایک قربانی پیش کی تو ان میں سے ایک کی قبول ہوئی اور دوسرے کی قبول نہ ہوئی۔” لفظ بِالْحَقِّ یہاں خود ایک ادب سکھاتا ہے: یہ قصہ سچائی کے ساتھ سنایا جائے، اس میں قصہ گو کی رنگ آمیزی نہ ہو۔ قرآن نے نہ دونوں کے نام لیے، نہ قربانی کا سبب کھولا، نہ مقام بتایا۔ مفسرین نے نقل کیا ہے کہ ان کے نام ہابیل اور قابیل تھے، اور امت میں یہی نام معروف ہیں؛ اس سے آگے جو داستانیں اسرائیلیات میں ملتی ہیں ہم ان پر بنا نہیں رکھتے، کیونکہ قرآن نے جتنا بیان کیا وہی عبرت کے لیے کافی ہے۔ صحیح بخاری اور صحیح مسلم میں حضرت عبداللہ بن مسعود رضی اللہ عنہ کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: جو جان بھی ظلم سے قتل کی جاتی ہے اس کے خون کا ایک حصہ حضرت آدم علیہ السلام کے پہلے بیٹے پر جاتا ہے، کیونکہ اسی نے سب سے پہلے قتل کی راہ ڈالی۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں قرآن کا ہر قصہ دو سطحوں پر پڑھا جاتا ہے: وہ تاریخ میں ایک بار ہوا، اور انسان کے باطن میں بار بار ہوتا ہے۔ اہلِ معرفت کہتے ہیں کہ ہر سینے میں ایک وہ رخ ہے جو اپنی بہترین چیز حق کے حضور پیش کر کے راضی رہتا ہے، اور ایک وہ رخ جو دوسرے کی قبولیت دیکھ کر جل اٹھتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ واقعہ اپنی جگہ سچا ہے، اور اس کا آئینہ ہر دل میں رکھا ہے۔

پھر حسد نے زبان کھولی اور تقویٰ نے جواب دیا: قَالَ لَاَقْتُلَنَّكَ ۭ قَالَ اِنَّمَا يَتَقَبَّلُ اللّٰهُ مِنَ الْمُتَّقِيْنَ۔ “اس نے کہا: میں تجھے ضرور قتل کر دوں گا۔ اس نے کہا: اللہ تو صرف پرہیزگاروں سے قبول فرماتا ہے۔” غور کیجیے کہ قربانی رد ہونے والے کا جھگڑا دراصل اپنے رب سے تھا، مگر اس نے غصہ بھائی پر نکالا؛ بھائی کا کوئی قصور نہ تھا سوائے اس کے کہ اس کی قبول ہو گئی۔ اور جواب دینے والے نے یہ نہیں کہا کہ میں تجھ سے بہتر ہوں؛ اس نے اپنی ذات کے بجائے ایک اصول کی طرف اشارہ کیا: قبولیت کا مدار ہدیے کی مقدار پر نہیں، دینے والے کے دل پر ہے۔ گویا وہ نرمی سے علاج بتا رہا تھا: میرا خون تیری مشکل کا حل نہیں، تیرا تقویٰ اس کا حل ہے۔ بعض صحابہ رضی اللہ عنہم سے منقول ہے کہ وہ اس آیت سے لرزتے تھے اور کہتے تھے کہ اگر یقین ہو جائے کہ اللہ نے ہماری ایک نماز بھی قبول کر لی تو یہ دنیا و مافیہا سے بڑھ کر ہے۔ شیخِ اکبر کے مکتب میں تقویٰ کا اصل معنی وقایہ ہے، یعنی بچاؤ کی ڈھال: بندہ حق کو اپنی ڈھال بنائے کہ ہر برائی سے اسی کی پناہ لے، اور اپنی ذات کو حق کی ڈھال بنائے کہ ہر کمی اپنی طرف منسوب کرے اور ہر کمال اس کی طرف۔ جو یوں اپنی نسبت مٹا دے اسی کی پیشکش قبول ہوتی ہے، کیونکہ اس نے اپنا کچھ پیش ہی نہیں کیا، حق کی امانت حق کو لوٹائی۔

اہلِ معرفت کہتے ہیں کہ حسد کی حقیقت تقسیمِ الٰہی پر اعتراض ہے۔ حاسد دراصل یہ کہتا ہے کہ یہ عطا وہاں کیوں گئی، یہاں کیوں نہ آئی۔ شیخِ اکبر کے نزدیک سرِّ قدر اسی گرہ کو کھولتا ہے: حق ہر شے کو وہی عطا کرتا ہے جس کا وہ اپنے عینِ ثابتہ اور اپنی استعداد کے لحاظ سے طالب ہے؛ حکم محکوم کے مطابق ہوتا ہے، اور عطا قابل کے ظرف کے مطابق۔ قربانی ایک جیسی دکھائی دیتی تھی، مگر دو دل ایک جیسے نہ تھے؛ ایک کا رخ حق کی طرف تھا اور دوسرے کا اپنی برتری کی طرف۔ اس لیے قبولیت بھی ایک رخ کے مطابق ظاہر ہوئی اور رد بھی دوسرے رخ کے مطابق۔ حاسد کے سامنے دو راستے تھے: اپنے دل کو درست کرے تاکہ اگلی بار اس کا ہدیہ قبول ہو، یا اس آئینے کو توڑ دے جس میں اسے اپنی کمی دکھائی دیتی تھی۔ اس نے دوسرا راستہ چنا، اور یہی ہر حسد کا انجام ہے کہ وہ اپنی اصلاح کے بجائے دوسرے کی بربادی چاہتا ہے۔ اہلِ ذوق اس سے یہ سبق لیتے ہیں کہ جب کسی کی قبولیت دیکھ کر دل میں چبھن اٹھے تو وہ لمحہ اپنے تقویٰ کی خبر لینے کا ہے، دوسرے کی نعمت گننے کا نہیں؛ اور جو کسی پر حق کی عطا دیکھ کر خوش ہو جائے وہ گویا اسی عطا میں شریک ہو گیا۔

اب مظلوم بھائی کا وہ جواب آتا ہے جو صبر اور خشیت کا نمونہ ہے: لَئِنْۢ بَسَطْتَّ اِلَيَّ يَدَكَ لِتَقْتُلَنِيْ مَآ اَنَا بِبَاسِطٍ يَّدِيَ اِلَيْكَ لِاَقْتُلَكَ ۚ اِنِّيْۤ اَخَافُ اللّٰهَ رَبَّ الْعٰلَمِيْنَ۔ “اگر تو مجھے قتل کرنے کے لیے میری طرف ہاتھ بڑھائے گا تو میں تجھے قتل کرنے کے لیے تیری طرف ہاتھ بڑھانے والا نہیں؛ میں اللہ سے ڈرتا ہوں جو سارے جہانوں کا رب ہے۔” اِنِّيْۤ اُرِيْدُ اَنْ تَبُوْۤاَ بِاِثْمِيْ وَاِثْمِكَ فَتَكُوْنَ مِنْ اَصْحٰبِ النَّارِ ۚ وَذٰلِكَ جَزٰٓؤُا الظّٰلِمِيْنَ۔ “میں چاہتا ہوں کہ تو میرا گناہ اور اپنا گناہ دونوں سمیٹ لے اور دوزخ والوں میں سے ہو جائے، اور ظالموں کی یہی سزا ہے۔” مفسرین کہتے ہیں کہ میرے گناہ سے مراد میرے قتل کا گناہ ہے؛ وہ بھائی کے گناہ کا خواہاں نہ تھا، بلکہ یہ کہہ رہا تھا کہ اگر ہم میں سے کسی ایک کو ظالم بننا ہی ہے تو میں مقتول ہونا پسند کروں گا، قاتل ہونا نہیں۔ سنن ابی داود اور جامع ترمذی کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فتنے کے زمانے کے بارے میں فرمایا کہ حضرت آدم علیہ السلام کے دو بیٹوں میں سے بہتر کی طرح ہو جانا۔ فقہا واضح کرتے ہیں کہ اپنی جان، مال اور آبرو کا دفاع شریعت میں جائز ہے؛ یہ آیت ظلم میں پہل نہ کرنے اور فتنے میں ہاتھ روک لینے کا مقام دکھاتی ہے۔ اور شیخِ اکبر کے مکتب میں لفظ رَبَّ الْعٰلَمِيْنَ پر نظر ٹھہرتی ہے: اس نے یہ نہیں کہا کہ میں اپنے رب سے ڈرتا ہوں، بلکہ سارے جہانوں کے رب سے؛ یعنی اسے نظر آ رہا تھا کہ سامنے کھڑا بھائی بھی اسی ربوبیت کے سائے میں ہے۔ جو ہر چہرے میں ایک ہی رب کی پرورش دیکھ لے اس کا ہاتھ کسی پر ناحق نہیں اٹھتا۔

مگر نصیحت حسد کی آگ نہ بجھا سکی: فَطَوَّعَتْ لَهٗ نَفْسُهٗ قَتْلَ اَخِيْهِ فَقَتَلَهٗ فَاَصْبَحَ مِنَ الْخٰسِرِيْنَ۔ “آخر اس کے نفس نے اسے اپنے بھائی کے قتل پر آمادہ کر لیا، سو اس نے اسے قتل کر دیا اور خسارہ اٹھانے والوں میں سے ہو گیا۔” فعل فَطَوَّعَتْ میں تدریج کا مفہوم ہے: نفس نے بات کو رفتہ رفتہ آسان بنایا، پہلے خیال، پھر جواز، پھر عزم، یہاں تک کہ جو کام فطرت کو سب سے زیادہ ناگوار تھا وہ آسان دکھائی دینے لگا۔ اہلِ معرفت کہتے ہیں کہ ہر بڑا گناہ اسی طرح چھوٹی چھوٹی رضامندیوں سے بنتا ہے، اس لیے سالک کو پہلی سرگوشی ہی پر چوکنا رہنا چاہیے؛ سورۂ یوسف میں یہی حقیقت کھولی گئی: اِنَّ النَّفْسَ لَاَمَّارَةٌۢ بِالسُّوْۤءِ اِلَّا مَا رَحِمَ رَبِّيْ۔ اور قرآن کا اسلوب دیکھیے کہ قتل کے لمحے میں بھی مقتول کو اَخِيْهِ کہا، اس کا بھائی؛ گویا ہر بار یاد دلایا جا رہا ہے کہ جسے مٹایا جا رہا ہے وہ کوئی اجنبی نہیں، اپنا ہی خون ہے۔ شیخِ اکبر کے نزدیک تمام انسان ایک ہی حقیقتِ انسانیہ کے مظاہر ہیں، اس لیے ہر ناحق قتل کسی نہ کسی معنی میں اپنے ہی وجود پر وار ہے۔ اور خسارہ تین طرح کا تھا: بھائی کھو دیا، اپنا سکون کھو دیا، اور آخرت کی امان کھو دی۔ اس نے سمجھا تھا کہ مقابل کو ہٹا کر وہ جیت جائے گا، مگر اس نے خود کو ہی ہار دیا۔

پھر عجز اور ندامت کا منظر آتا ہے: فَبَعَثَ اللّٰهُ غُرَابًا يَّبْحَثُ فِي الْاَرْضِ لِيُرِيَهٗ كَيْفَ يُوَارِيْ سَوْءَةَ اَخِيْهِ ۭ قَالَ يٰوَيْلَتٰۤى اَعَجَزْتُ اَنْ اَكُوْنَ مِثْلَ هٰذَا الْغُرَابِ فَاُوَارِيَ سَوْءَةَ اَخِيْ ۚ فَاَصْبَحَ مِنَ النّٰدِمِيْنَ۔ “پھر اللہ نے ایک کوا بھیجا جو زمین کریدنے لگا تاکہ اسے دکھائے کہ اپنے بھائی کی لاش کیسے چھپائے۔ وہ بولا: ہائے افسوس، کیا میں اس کوے جیسا بھی نہ ہو سکا کہ اپنے بھائی کی لاش چھپا دیتا؟ پھر وہ پچھتانے والوں میں سے ہو گیا۔” جس نے برتری کے زعم میں بھائی کو مارا تھا اسے ایک پرندے کی شاگردی میں بٹھا دیا گیا؛ تکبر کا علاج اسی طرح عاجزی سے ہوتا ہے۔ اور لفظ بَعَثَ پر غور کیجیے، یہی لفظ رسولوں کے بھیجنے کے لیے آتا ہے؛ کوا بھی حق کا بھیجا ہوا معلم بن کر آیا۔ شیخِ اکبر کے مکتب میں ہر مخلوق کسی اسمِ الٰہی کا مظہر ہے، اس لیے جس کی آنکھ کھلی ہو وہ کائنات کی ہر شے سے سبق لیتا ہے، چاہے وہ کوا ہی کیوں نہ ہو۔ مفسرین کہتے ہیں کہ اس کی ندامت اپنی بے بسی اور اپنے نقصان پر تھی، اس میں رب کی طرف رجوع شامل نہ تھا، اس لیے وہ توبہ نہ بنی۔ اہلِ ذوق یہ بھی کہتے ہیں کہ پہلے قتل کے بعد انسان کو پہلا سبق چھپانے کا دیا گیا۔ صحیح مسلم کی روایت کا مفہوم ہے کہ جو کسی مسلمان کا عیب چھپائے، اللہ دنیا و آخرت میں اس کا عیب چھپائے گا۔ حق ستار ہے، اور جو بندہ اس اسم سے رنگ لیتا ہے وہ بھائی کی کمزوری کو مٹی دیتا ہے، تماشا نہیں بناتا۔

اس قصے کے بعد وہ اصول لکھا گیا جو انسانی جان کی حرمت کا منشور ہے: مِنْ اَجْلِ ذٰلِكَ ۛ كَتَبْنَا عَلٰى بَنِيْۤ اِسْرَآءِيْلَ اَنَّهٗ مَنْ قَتَلَ نَفْسًۢا بِغَيْرِ نَفْسٍ اَوْ فَسَادٍ فِي الْاَرْضِ فَكَاَنَّمَا قَتَلَ النَّاسَ جَمِيْعًا ۭ وَمَنْ اَحْيَاهَا فَكَاَنَّمَآ اَحْيَا النَّاسَ جَمِيْعًا۔ “اسی سبب سے ہم نے بنی اسرائیل پر لکھ دیا کہ جس نے کسی جان کو، بغیر اس کے کہ اس نے کسی جان کو قتل کیا ہو یا زمین میں فساد پھیلایا ہو، قتل کر ڈالا تو گویا اس نے تمام انسانوں کو قتل کر دیا؛ اور جس نے کسی ایک جان کو بچا لیا تو گویا اس نے تمام انسانوں کو زندگی بخش دی۔” مفسرین کہتے ہیں کہ یہ حکم بنی اسرائیل پر صراحت سے لکھا گیا کیونکہ تورات انہی کو ملی، مگر اس کی حرمت ساری انسانیت کے لیے ہے اور ہماری شریعت میں بھی باقی ہے۔ لفظ نَفْسًا عام ہے؛ اس میں کسی قوم، مذہب یا رنگ کی قید نہیں۔ اور جو استثنا رکھا گیا، یعنی قصاص یا زمین میں فساد، اس کا فیصلہ صرف جائز حکومت اور عدالت کرتی ہے، ثبوت اور شرعی کارروائی کے بعد؛ کسی فرد یا گروہ کو یہ حق کبھی حاصل نہیں۔ “گویا سب کو قتل کیا” کی تفسیر میں مفسرین کے کئی اقوال ہیں: یہ کہ ایک بے گناہ کا قاتل اس حرمت کو توڑتا ہے جو سب کی مشترک ہے؛ یہ کہ اس کا گناہ اللہ کے نزدیک بہت بڑا ہے؛ اور یہ کہ جو ایک کو مارنا روا رکھے اس کے لیے ہر جان کا خون آسان ہو جاتا ہے۔ اور زندگی بخشنے میں ڈوبتے کو بچانا، جلتے کو نکالنا، بھوکے کو کھلانا، اور قصاص کا حق رکھتے ہوئے معاف کر دینا، سب شامل ہیں۔

شیخِ اکبر کے مکتب میں یہ آیت انسان کی حقیقت کا راز کھولتی ہے۔ حضرت آدم علیہ السلام کے بارے میں ان کی تعلیم کا مفہوم، جو حکمتِ الٰہیہ کے عنوان سے آئی، یہ ہے کہ حضرت آدم علیہ السلام وہ جامع نسخہ ہیں جس میں اسماءِ الٰہیہ کے سب آثار جمع ہوئے، اور وہی عالم کے آئینے کی جلا ہیں۔ اور یہ جامعیت صرف پہلے انسان کی نہیں، اس کی پوری نوع کی میراث ہے؛ صحیح بخاری اور صحیح مسلم کی روایت کا مفہوم ہے کہ اللہ نے حضرت آدم علیہ السلام کو اپنی صورت پر پیدا کیا، اور اہلِ علم نے اس کی کئی توجیہات کی ہیں جن میں صفات کا عکس ہونا ایک ہے۔ جب ہر انسان پورے عالم کا خلاصہ اور حقیقتِ انسانیہ کا مکمل مظہر ہے تو ایک جان دراصل پوری انسانیت کی نمائندہ ہے؛ حقیقتِ انسانیہ افراد میں تقسیم ہو کر ٹکڑے نہیں ہوتی، ہر فرد میں پوری کی پوری جلوہ گر ہے۔ اسی لیے اہلِ معرفت کہتے ہیں کہ “فَكَاَنَّمَا قَتَلَ النَّاسَ جَمِيْعًا” محض مبالغہ نہیں، ایک حقیقت کا بیان ہے: جس نے ایک انسان کو مٹایا اس نے اس صورت پر وار کیا جس میں سب انسان شریک ہیں۔ سورۂ التین میں اسی صورت کی خبر دی گئی: لَقَدْ خَلَقْنَا الْاِنْسَانَ فِيْۤ اَحْسَنِ تَقْوِيْمٍ۔ اور زندگی بخشنے کا ایک باطنی درجہ بھی ہے: بعض مفسرین نے کہا کہ جس نے کسی کو گمراہی سے نکال کر ہدایت کی راہ دکھا دی اس نے بھی اسے زندہ کیا؛ اہلِ ذوق کہتے ہیں کہ مردہ دل کو ذکر سے زندہ کرنا بھی اسی احیا کی ایک صورت ہے۔ یہ سب ظاہر کے ساتھ ہے، اس کی جگہ نہیں۔

پھر آیت کا باقی حصہ بنی اسرائیل کے اس دور کے رویے کی گواہی دیتا ہے: وَلَقَدْ جَآءَتْهُمْ رُسُلُنَا بِالْبَيِّنٰتِ ثُمَّ اِنَّ كَثِيْرًا مِّنْهُمْ بَعْدَ ذٰلِكَ فِي الْاَرْضِ لَمُسْرِفُوْنَ۔ “اور ان کے پاس ہمارے رسول کھلی نشانیاں لے کر آئے، پھر بھی ان میں سے بہت سے لوگ اس کے بعد زمین میں حد سے گزر جانے والے ہی رہے۔” یہ ایک تاریخی گواہی ہے اور قرآن نے خود كَثِيْرًا مِّنْهُمْ کہہ کر سب کو ایک حکم میں نہیں باندھا؛ اسے آج کی کسی قوم پر چسپاں کرنا قرآن کے انصاف کے خلاف ہے۔ اصل سبق ہمارے لیے ہے: حرمت کا علم ہونا کافی نہیں، علم کے بعد اسراف بھی ممکن ہے۔ صحیح بخاری اور صحیح مسلم کی روایت کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: قیامت کے دن لوگوں کے درمیان سب سے پہلے خون کے معاملات کا فیصلہ ہوگا۔ اہلِ معرفت کہتے ہیں کہ اسراف کا مطلب صرف مال میں حد سے بڑھنا نہیں، ہر اس چیز میں حد پھلانگنا ہے جس کی حد حق نے مقرر کی؛ اور سب سے بڑی حد جان کی حرمت ہے۔ شیخِ اکبر کے نزدیک بینات، یعنی کھلی نشانیاں، دل کے سامنے حق کی تجلی ہیں؛ اور جو تجلی کے بعد بھی منہ پھیرے اس کا اعراض اس کا اپنا انتخاب ہے، جس کا بوجھ وہ خود اٹھاتا ہے۔

اب اس فساد کا قانونی حکم آتا ہے جو منظم صورت میں زمین کا امن چھینتا ہے: اِنَّمَا جَزٰٓؤُا الَّذِيْنَ يُحَارِبُوْنَ اللّٰهَ وَرَسُوْلَهٗ وَيَسْعَوْنَ فِي الْاَرْضِ فَسَادًا اَنْ يُّقَتَّلُوْۤا اَوْ يُصَلَّبُوْۤا اَوْ تُقَطَّعَ اَيْدِيْهِمْ وَاَرْجُلُهُمْ مِّنْ خِلَافٍ اَوْ يُنْفَوْا مِنَ الْاَرْضِ ۭ ذٰلِكَ لَهُمْ خِزْيٌ فِي الدُّنْيَا وَلَهُمْ فِي الْاٰخِرَةِ عَذَابٌ عَظِيْمٌ۔ “جو لوگ اللہ اور اس کے رسول سے جنگ کرتے ہیں اور زمین میں فساد پھیلانے کے لیے دوڑتے پھرتے ہیں، ان کی سزا یہی ہے کہ قتل کیے جائیں، یا سولی دیے جائیں، یا ان کے ہاتھ اور پاؤں مخالف سمتوں سے کاٹے جائیں، یا انہیں زمین سے دور کر دیا جائے؛ یہ ان کے لیے دنیا میں رسوائی ہے اور آخرت میں ان کے لیے بڑا عذاب ہے۔” فقہا اسے حرابہ کہتے ہیں: ہتھیار کے زور پر راستوں اور آبادیوں میں ڈاکہ، قتل اور دہشت، جس سے لوگوں کی جان، مال اور آبرو کا امن اٹھ جائے۔ صحیح بخاری میں حضرت انس رضی اللہ عنہ کی روایت میں ان لوگوں کا ذکر ہے جنہوں نے چرواہے کو قتل کیا اور اونٹ ہانک لے گئے، اور مفسرین نے اسے اس آیت کے پس منظر میں ذکر کیا۔ حضرت عبداللہ بن عباس رضی اللہ عنہما اور جمہور کے نزدیک یہ سزائیں جرم کے درجے کے مطابق ہیں: قتل اور لوٹ دونوں ہوں تو سخت ترین، صرف قتل ہو تو قتل، صرف مال لوٹا ہو تو ہاتھ پاؤں مخالف سمت سے، اور صرف دہشت پھیلائی ہو تو جلاوطنی؛ امام ابو حنیفہ رحمۃ اللہ علیہ کے نزدیک نفی سے مراد قید ہے، اور امام مالک رحمۃ اللہ علیہ نے امام کو ایک حد تک اختیار دیا۔ یہ سب شریعت کی حدود ہیں جنہیں صرف جائز حکومت اور عدالت مکمل ثبوت کے بعد نافذ کرتی ہے؛ کسی فرد یا جماعت کو انہیں اپنے ہاتھ میں لینے کا حق نہیں، اور جو ایسا کرے وہ خود اسی فساد میں جا پڑتا ہے جس کی یہ آیت مذمت کرتی ہے۔ اہلِ معرفت کہتے ہیں کہ ڈاکو کو اللہ اور رسول سے لڑنے والا اس لیے کہا گیا کہ مخلوق کا امن حق کے اسم المؤمن کا مظہر ہے، جو امن عطا کرنے والا ہے؛ جو اس امن کو چھینتا ہے وہ گویا اس اسم کے مظہر سے جنگ کرتا ہے۔

اور سخت ترین سزا کے فوراً بعد توبہ کا دروازہ کھول دیا گیا: اِلَّا الَّذِيْنَ تَابُوْا مِنْ قَبْلِ اَنْ تَقْدِرُوْا عَلَيْهِمْ ۚ فَاعْلَمُوْۤا اَنَّ اللّٰهَ غَفُوْرٌ رَّحِيْمٌ۔ “مگر جو لوگ اس سے پہلے توبہ کر لیں کہ تم ان پر قابو پاؤ، تو جان لو کہ اللہ بہت بخشنے والا، نہایت رحم والا ہے۔” فقہا کہتے ہیں کہ جو محارب گرفتاری سے پہلے خود توبہ کر کے لوٹ آئے اس سے حرابہ کی حد ساقط ہو جاتی ہے، کیونکہ وہ اللہ کا حق ہے؛ البتہ بندوں کے حقوق، یعنی قتل کا قصاص اور لوٹا ہوا مال، جمہور کے نزدیک باقی رہتے ہیں، الا یہ کہ صاحبِ حق معاف کر دے۔ یہاں ایک نازک نکتہ ہے: گرفتاری سے پہلے کی توبہ اختیار کی توبہ ہے، اور گرفتاری کے بعد کی توبہ میں مجبوری کا شائبہ ہوتا ہے۔ شریعت نے اسی آزاد رجوع کو قیمت دی؛ یہ اختیار کی وہ عزت ہے جو حق نے بندے کو بخشی۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا اظہار ہے: حق غضب کی بات کرتے ہوئے بھی رحمت پر کلام ختم کرتا ہے۔ لفظ فَاعْلَمُوْۤا پر غور کیجیے، یعنی جان لو؛ گویا حکم دینے والوں سے بھی کہا جا رہا ہے کہ جو لوٹ آیا اسے اس کے پرانے حال سے نہ پکارو، کیونکہ حق اسے نئے سرے سے قبول کر چکا۔ اس مکتب میں تجددِ امثال کا اصول یہی سکھاتا ہے کہ ہر لمحہ خلقِ جدید ہے، اور سچی توبہ کرنے والا کل والا آدمی نہیں رہا۔
