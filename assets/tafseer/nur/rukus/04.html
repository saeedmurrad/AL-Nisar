يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تَدْخُلُوْا بُيُوْتًا غَيْرَ بُيُوْتِكُمْ حَتّٰى تَسْتَاْنِسُوْا وَتُسَلِّمُوْا عَلٰۤى اَهْلِهَا ۚ ذٰلِكُمْ خَيْرٌ لَّكُمْ لَعَلَّكُمْ تَذَكَّرُوْنَ۔ اے ایمان والو! اپنے گھروں کے سوا دوسرے گھروں میں داخل نہ ہو جب تک اجازت نہ لے لو اور گھر والوں کو سلام نہ کر لو؛ یہ تمہارے لیے بہتر ہے تاکہ تم نصیحت حاصل کرو۔ تہمت کے بیان کے فوراً بعد گھر کی حرمت کا بیان آیا، کیونکہ عزتوں کی حفاظت دروازے سے شروع ہوتی ہے۔ ظاہری حکم واضح ہے: کسی کے گھر میں بلا اجازت داخل ہونا جائز نہیں، اور اجازت سلام کے ساتھ مانگی جائے۔ مگر قرآن نے یہاں اجازت کے لیے انس کے مادے سے لفظ "تَسْتَاْنِسُوْا" چنا: ایسی اجازت جس سے گھر والے مانوس ہو جائیں اور ان کے دل میں وحشت نہ اترے؛ گویا آنے والا پہلے اپنی آمد کو دل پسند بنائے، پھر قدم رکھے۔ اہلِ معرفت کہتے ہیں کہ دل بھی گھر ہیں، اور کسی کے دل میں داخل ہونے کا ادب بھی یہی ہے: پہلے انس، پھر سلام، پھر قدم۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ دروازے کا حکم اپنی جگہ واجب ہے۔ اور سب سے بڑی بارگاہ حق کی بارگاہ ہے، جہاں ادب کے بغیر کوئی داخل نہیں ہوتا۔ سورۂ طٰہٰ میں جب حضرت موسیٰ علیہ السلام مقدس وادی میں پہنچے تو کلام سے پہلے ادب کا حکم آیا: فَاخْلَعْ نَعْلَيْكَ ۚ اِنَّكَ بِالْوَادِ الْمُقَدَّسِ طُوًى۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں حضوری کا دروازہ ادب ہے؛ بے ادب قدم خود ایک حجاب بن جاتا ہے۔ اور "لَعَلَّكُمْ تَذَكَّرُوْنَ" بتاتا ہے کہ یہ ادب خود ایک ذکر ہے: ہر دروازے پر رکنا یاد دلاتا ہے کہ ہر ملکیت کا کوئی مالک ہے، اور حقیقی مالک ایک ہی ہے۔

فَاِنْ لَّمْ تَجِدُوْا فِيْهَاۤ اَحَدًا فَلَا تَدْخُلُوْهَا حَتّٰى يُؤْذَنَ لَكُمْ ۚ وَاِنْ قِيْلَ لَكُمُ ارْجِعُوْا فَارْجِعُوْا هُوَ اَزْكٰى لَكُمْ ۚ وَاللّٰهُ بِمَا تَعْمَلُوْنَ عَلِيْمٌ۔ پھر اگر تم وہاں کسی کو نہ پاؤ تو اندر نہ جاؤ جب تک تمہیں اجازت نہ دی جائے؛ اور اگر تم سے کہا جائے کہ واپس چلے جاؤ تو واپس چلے جاؤ، یہی تمہارے لیے زیادہ پاکیزہ ہے؛ اور اللہ تمہارے اعمال کو خوب جانتا ہے۔ لَيْسَ عَلَيْكُمْ جُنَاحٌ اَنْ تَدْخُلُوْا بُيُوْتًا غَيْرَ مَسْكُوْنَةٍ فِيْهَا مَتَاعٌ لَّكُمْ ۚ وَاللّٰهُ يَعْلَمُ مَا تُبْدُوْنَ وَمَا تَكْتُمُوْنَ۔ تم پر کوئی گناہ نہیں کہ ایسے گھروں میں داخل ہو جو کسی کی رہائش نہیں اور جن میں تمہارا کوئی سامان یا فائدہ ہے؛ اور اللہ جانتا ہے جو تم ظاہر کرتے ہو اور جو چھپاتے ہو۔ پچھلے رکوع میں فرمایا تھا کہ پاکیزگی اسی کے فضل سے ہے؛ یہاں اس فضل کا ایک دروازہ دکھا دیا: "اَزْكٰى لَكُمْ"۔ انکار کو خوش دلی سے قبول کر لینا نفس پر سب سے بھاری ہے؛ نفس دروازے پر بحث کرنا چاہتا ہے، اور بغیر رنجش کے لوٹ جانا اسے پاک کرتا ہے۔ اہلِ ذوق کہتے ہیں کہ سالک پر بھی کبھی کسی حال کا دروازہ بند کر دیا جاتا ہے؛ یہ قبض ہے۔ ادب یہ ہے کہ وہ دروازہ نہ توڑے، ادب سے لوٹ آئے؛ اس مکتب میں قبض و بسط دونوں اسی کی طرف سے ہیں، اور قبض میں ادب ہی بسط کی کنجی ہے۔ "وَاللّٰهُ بِمَا تَعْمَلُوْنَ عَلِيْمٌ" دروازے پر مراقبہ ہے: کوئی دیکھے یا نہ دیکھے، وہ دیکھ رہا ہے۔ پھر ان عمارتوں کی رخصت دی گئی جن میں کوئی بستا نہیں، جیسے سرائیں اور تجارتی ٹھکانے۔ اہلِ ذوق اس میں دنیا کی تصویر دیکھتے ہیں: یہ بھی ایک ایسا گھر ہے جس میں تمہارا کچھ متاع ہے؛ ضرورت کے لیے اس میں داخل ہونے میں حرج نہیں، مگر یہ بسنے کی جگہ نہیں۔ اور آخر میں یاد دلایا کہ ہر داخلے کے پیچھے چھپی نیت بھی اس کے علم میں ہے۔

قُلْ لِّلْمُؤْمِنِيْنَ يَغُضُّوْا مِنْ اَبْصَارِهِمْ وَيَحْفَظُوْا فُرُوْجَهُمْ ۚ ذٰلِكَ اَزْكٰى لَهُمْ ۚ اِنَّ اللّٰهَ خَبِيْرٌۢ بِمَا يَصْنَعُوْنَ۔ ایمان والے مردوں سے کہہ دیجیے کہ اپنی نگاہیں نیچی رکھیں اور اپنی شرم گاہوں کی حفاظت کریں؛ یہ ان کے لیے زیادہ پاکیزہ ہے؛ بے شک اللہ ان کاموں سے پوری طرح باخبر ہے جو وہ کرتے ہیں۔ ترتیب پر غور کیجیے: حیا کا پہلا خطاب مردوں سے ہوا۔ نگاہ کی ذمہ داری دیکھنے والے پر ہے، اس پر نہیں جسے دیکھا جائے؛ کوئی مرد اپنی بے احتیاط نظر کا بوجھ کسی دوسرے پر نہیں ڈال سکتا۔ "مِنْ اَبْصَارِهِمْ" میں مِنْ بعض کے معنی میں ہے: آنکھیں بند کرنے کا حکم نہیں، نگاہ کو اس سے روکنے کا حکم ہے جو حلال نہیں؛ آنکھ اندھی نہیں کی جاتی، مہذب کی جاتی ہے۔ حدیث کا مفہوم ہے کہ اچانک پڑ جانے والی پہلی نظر معاف ہے، مگر دوسری نظر تمہاری نہیں۔ اہلِ معرفت کہتے ہیں کہ نگاہ دل کا دروازہ ہے؛ جو آنکھ سے داخل ہوتا ہے وہ دل میں جا بستا ہے، اس لیے نگاہ کی حفاظت دراصل دل کی حفاظت ہے۔ سورۂ الحج میں فرمایا تھا: فَاِنَّهَا لَا تَعْمَى الْاَبْصَارُ وَلٰكِنْ تَعْمَى الْقُلُوْبُ الَّتِيْ فِي الصُّدُوْرِ۔ شیخِ اکبر کے مکتب میں دل تجلیاتِ حق کا آئینہ ہے؛ آنکھ جو ناروا صورت اندر بھیجتی ہے وہ اس آئینے پر غبار بن کر بیٹھ جاتی ہے، اور جھکی ہوئی نگاہ آئینے کو اس تجلی کے لیے صاف رکھتی ہے جس کے لیے دل بنا ہے۔ تزکیہ کا لفظ اس رکوع میں دوسری بار آیا: "ذٰلِكَ اَزْكٰى لَهُمْ"۔ اور "خَبِيْرٌۢ بِمَا يَصْنَعُوْنَ" میں خبیر وہ ہے جو باریک ترین بات بھی جانتا ہے؛ چوری کی وہ نظر جسے کوئی نہیں دیکھتا، اس کی خبر میں ہے۔

پھر ایمان والی خواتین سے براہِ راست خطاب ہوا: وَقُلْ لِّلْمُؤْمِنٰتِ يَغْضُضْنَ مِنْ اَبْصَارِهِنَّ وَيَحْفَظْنَ فُرُوْجَهُنَّ وَلَا يُبْدِيْنَ زِيْنَتَهُنَّ اِلَّا مَا ظَهَرَ مِنْهَا وَلْيَضْرِبْنَ بِخُمُرِهِنَّ عَلٰى جُيُوْبِهِنَّ۔ اور ایمان والی عورتوں سے کہہ دیجیے کہ اپنی نگاہیں نیچی رکھیں اور اپنی شرم گاہوں کی حفاظت کریں، اور اپنی زینت ظاہر نہ کریں سوائے اس کے جو خود ظاہر ہو جائے، اور اپنی اوڑھنیاں اپنے سینوں پر ڈالے رکھیں۔ پہلے دو حکم بعینہٖ وہی ہیں جو مردوں کو دیے گئے: وہی نگاہ کی حفاظت، وہی عصمت کی حفاظت؛ حیا کے فرض میں دونوں برابر ہیں۔ پھر لباس کے خاص احکام آئے: زینت ظاہر نہ کی جائے سوائے اس کے جو خود ظاہر ہو، جس کی حدود فقہاء نے تفصیل سے بیان کی ہیں؛ اوڑھنی سینے پر ڈالی جائے؛ اور آیت ان قریبی رشتوں کو گنواتی ہے جن کے سامنے زینت کے اظہار کی اجازت ہے، جیسے شوہر، باپ، بیٹے، بھائی اور ان کے بیٹے، اور اپنی میل جول والی مسلمان عورتیں۔ یہ احکام عورت کی قدر کو گھٹانے کے لیے نہیں، اس کی عزت کی حفاظت کے لیے ہیں: جو چیز قیمتی ہو اسے ڈھانپ کر رکھا جاتا ہے، جیسے کعبہ پر غلاف ہے اور مصحف پر جزدان۔ اور یہ بھی دیکھیے کہ قرآن نے خواتین سے خود خطاب کیا، کسی واسطے سے نہیں: وہ اپنی حیا کی مالک بھی خود ہیں اور جواب دہ بھی خود؛ کسی کو ان کی نگاہ کا داروغہ نہیں بنایا گیا۔ حدیث کا مفہوم ہے کہ حیا ایمان کا ایک شعبہ ہے؛ اور اہلِ معرفت کہتے ہیں کہ حیا مرد اور عورت دونوں کا مشترک زیور ہے، کیونکہ یہ حق کے سامنے ہونے کا احساس ہے، اور حق کے سامنے سب برابر کھڑے ہیں۔

اسی آیت کا اختتام یوں ہوا: وَلَا يَضْرِبْنَ بِاَرْجُلِهِنَّ لِيُعْلَمَ مَا يُخْفِيْنَ مِنْ زِيْنَتِهِنَّ ۚ وَتُوْبُوْۤا اِلَى اللّٰهِ جَمِيْعًا اَيُّهَ الْمُؤْمِنُوْنَ لَعَلَّكُمْ تُفْلِحُوْنَ۔ اور وہ اپنے پاؤں زمین پر اس طرح نہ ماریں کہ ان کی چھپی ہوئی زینت معلوم ہو جائے؛ اور اے ایمان والو! تم سب مل کر اللہ کی طرف توبہ کرو تاکہ تم فلاح پاؤ۔ ظاہری حکم یہ ہے کہ چال میں بھی ایسا انداز نہ ہو جو چھپی زینت کی طرف توجہ کھینچے۔ اس کے ساتھ اہلِ ذوق ایک اشاری پہلو بیان کرتے ہیں جو ہر سالک کے لیے ہے، مرد ہو یا عورت، اور جو ظاہری حکم کی جگہ نہیں لیتا بلکہ اس کی گہرائی ہے: ہر روح کی بھی ایک باطنی زینت ہوتی ہے، یعنی احوال، واردات اور عطائیں۔ ادبِ اسرار یہ ہے کہ سالک اپنی چال ڈھال اور گفتگو سے ان چھپی عطاؤں کا اشتہار نہ کرے، کیونکہ جو حال دکھانے کے لیے ظاہر کیا جائے وہ حال نہیں رہتا۔ پھر آیت کا آخری جملہ: الگ الگ خطاب کے بعد اب ایک ہی خطاب ہے، "جَمِيْعًا"۔ کوئی توبہ کی ضرورت سے بالاتر نہیں؛ مرد اور عورت، عالم اور عامی، سب ایک صف میں کھڑے کیے گئے۔ توبہ کے معنی لوٹنا ہیں، اور شیخِ اکبر کے نزدیک بندے کا لوٹنا حق ہی کی طرف ہے جس سے وہ حقیقت میں کبھی جدا نہ تھا، صرف غفلت نے فاصلہ دکھایا تھا۔ اور یہی اصول یہاں بھی کارفرما ہے کہ دروازہ ہر لوٹنے والے کے لیے کھلا ہے؛ جو رخ کرے وہ فلاح پاتا ہے۔

وَاَنْكِحُوا الْاَيَامٰى مِنْكُمْ وَالصّٰلِحِيْنَ مِنْ عِبَادِكُمْ وَاِمَآئِكُمْ ۚ اِنْ يَّكُوْنُوْا فُقَرَآءَ يُغْنِهِمُ اللّٰهُ مِنْ فَضْلِهٖ ۚ وَاللّٰهُ وَاسِعٌ عَلِيْمٌ۔ اور تم میں سے جو بے نکاح ہوں ان کے نکاح کرا دو، اور اپنے غلاموں اور لونڈیوں میں سے جو نیک ہوں ان کے بھی؛ اگر وہ محتاج ہوں گے تو اللہ اپنے فضل سے انہیں غنی کر دے گا؛ اور اللہ بڑی وسعت والا، سب کچھ جاننے والا ہے۔ عصمت کی حفاظت کا حکم دینے کے بعد شریعت نے اس کا راستہ بھی کھولا: نکاح کو آسان کرو۔ خطاب پوری جماعت سے ہے؛ بے نکاح لوگوں کا نکاح معاشرے کی ذمہ داری قرار پایا، اور معاشرے کے کمزور ترین طبقے کو بھی اس میں شامل کیا گیا۔ غربت کا خوف رکاوٹ نہ بنے، کیونکہ وعدہ "يُغْنِهِمُ اللّٰهُ مِنْ فَضْلِهٖ" کا ہے۔ شیخِ اکبر کے مکتب میں نکاح ایک ہمہ گیر حقیقت ہے: کائنات میں جو کچھ پیدا ہوتا ہے وہ کسی فاعل اور قابل کے ملاپ سے پیدا ہوتا ہے، اور انسانی نکاح اس ہمہ گیر حقیقت کا ایک مظہر ہے۔ حکمتِ فردیہ میں شیخِ اکبر نے اس حدیث پر کلام کیا ہے کہ تمہاری دنیا میں سے تین چیزیں حضورِ اکرم صلی اللہ علیہ وسلم کو محبوب کرائی گئیں، جن میں خواتین کا ذکر ہے؛ جس کا مفہوم یہ ہے کہ یہ محبت حق کی طرف سے عطا ہوئی، اس لیے یہ کمال کی محبت ہے، نفس کی پستی نہیں، اور عورت اس مکتب میں حق کی رحمت اور تخلیق کا شریف مظہر ہے۔ پھر "وَاللّٰهُ وَاسِعٌ عَلِيْمٌ": جو اس پر بھروسا کر کے یہ قدم اٹھائے، اس کی وسعت اس کی تنگی کو ڈھانپ لیتی ہے، اور اس کا علم جانتا ہے کہ کس کو کب اور کتنا دینا ہے۔

وَلْيَسْتَعْفِفِ الَّذِيْنَ لَا يَجِدُوْنَ نِكَاحًا حَتّٰى يُغْنِيَهُمُ اللّٰهُ مِنْ فَضْلِهٖ ۚ وَالَّذِيْنَ يَبْتَغُوْنَ الْكِتٰبَ مِمَّا مَلَكَتْ اَيْمَانُكُمْ فَكَاتِبُوْهُمْ اِنْ عَلِمْتُمْ فِيْهِمْ خَيْرًا ۖ وَّاٰتُوْهُمْ مِّنْ مَّالِ اللّٰهِ الَّذِيْۤ اٰتٰىكُمْ۔ اور جو نکاح کا مقدور نہیں رکھتے وہ پاک دامنی اختیار کیے رہیں یہاں تک کہ اللہ انہیں اپنے فضل سے غنی کر دے؛ اور تمہارے غلاموں میں سے جو آزادی کا معاہدہ چاہیں، اگر تم ان میں بھلائی دیکھو تو ان سے معاہدہ کر لو، اور اللہ کے اس مال میں سے انہیں دو جو اس نے تمہیں دیا ہے۔ "وَلْيَسْتَعْفِفِ" میں طلب کا مفہوم ہے: عفت محض گناہ سے رک جانا نہیں، اسے کوشش سے حاصل کرنا اور اس پر جمے رہنا ہے۔ اور انتظار بھی بے مدت نہیں رکھا، "حَتّٰى يُغْنِيَهُمُ اللّٰهُ"؛ صبر کے ساتھ امید بھی دی گئی۔ پھر اس غلام کا ذکر جو اپنی آزادی خریدنا چاہے: شریعت نے مالک کو ترغیب دی کہ معاہدہ کرے اور اپنے مال سے اس کی مدد بھی کرے۔ سب سے لطیف نکتہ "مَّالِ اللّٰهِ الَّذِيْۤ اٰتٰىكُمْ" میں ہے: مال تمہارا نہیں، اللہ کا ہے، اور تم صرف امین ہو۔ اہلِ معرفت کہتے ہیں کہ یہی فقر کی حقیقت ہے: بندے کی ملکیت عاریتی ہے، حقیقی مالک ایک ہی ہے؛ اور جو یہ پہچان لے اس کے لیے دینا آسان ہو جاتا ہے، کیونکہ وہ اپنا نہیں، مالک کا مال اس کے بندے تک پہنچا رہا ہے۔ اہلِ ذوق کہتے ہیں کہ جو کسی کی گردن آزاد کرانے میں مدد دیتا ہے وہ دراصل اپنی گردن کو مال کی غلامی سے آزاد کرتا ہے۔

پھر اسی آیت میں کمزور کی عزت کا اعلان: وَلَا تُكْرِهُوْا فَتَيٰتِكُمْ عَلَى الْبِغَآءِ اِنْ اَرَدْنَ تَحَصُّنًا لِّتَبْتَغُوْا عَرَضَ الْحَيٰوةِ الدُّنْيَا۔ اور اپنی کنیزوں کو بدکاری پر مجبور نہ کرو جبکہ وہ پاک دامن رہنا چاہتی ہوں، محض اس لیے کہ تم دنیا کی زندگی کا سامان حاصل کرو۔ مفسرین کہتے ہیں کہ "اِنْ اَرَدْنَ تَحَصُّنًا" کوئی ایسی شرط نہیں جس کے نہ ہونے پر اجازت ہو جائے؛ یہ اصل صورتِ حال کا بیان ہے، اور ظلم کی شناعت کو کھول دیتا ہے: وہ پاکیزگی چاہتی ہے اور تم اسے پستی کی طرف دھکیلتے ہو۔ پھر فرمایا: وَمَنْ يُّكْرِهْهُّنَّ فَاِنَّ اللّٰهَ مِنْۢ بَعْدِ اِكْرَاهِهِنَّ غَفُوْرٌ رَّحِيْمٌ۔ اکثر مفسرین کے نزدیک یہ مغفرت اور رحمت ان مجبور کی گئی خواتین کے لیے ہے: جبر کا گناہ جبر کرنے والے پر ہے، مجبور پر نہیں۔ یوں قرآن نے سماج کے سب سے بے بس طبقے کی عزت کو اپنی پناہ میں لے لیا، کیونکہ وَلَقَدْ كَرَّمْنَا بَنِيْۤ اٰدَمَ۔ اس کے بعد رکوع کا اختتام: وَلَقَدْ اَنْزَلْنَاۤ اِلَيْكُمْ اٰيٰتٍ مُّبَيِّنٰتٍ وَّمَثَلًا مِّنَ الَّذِيْنَ خَلَوْا مِنْ قَبْلِكُمْ وَمَوْعِظَةً لِّلْمُتَّقِيْنَ۔ اور ہم نے تمہاری طرف کھول کر بیان کرنے والی آیتیں اتاری ہیں، اور تم سے پہلے گزرے ہوئے لوگوں کی مثالیں، اور پرہیزگاروں کے لیے نصیحت۔ شیخِ اکبر کے مکتب میں شریعت کا ہر حکم بھی ایک آیت ہے، یعنی کسی اسمِ الٰہی کی نشانی: دروازے پر رکنا اسمِ حفیظ کی، معاف کرنا اسمِ غفور کی، اور کمزور کی حفاظت اسمِ رحیم کی۔ اور ترتیب پر غور کیجیے: گھر کی حرمت، نگاہ کی حفاظت اور دل کی صفائی کے بعد ہی اگلی آیت میں نور کا بیان آتا ہے؛ گویا پہلے طاق صاف کیا گیا، پھر چراغ کا ذکر ہوا۔
