يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تَتَّبِعُوْا خُطُوٰتِ الشَّيْطٰنِ ۚ وَمَنْ يَّتَّبِعْ خُطُوٰتِ الشَّيْطٰنِ فَاِنَّهٗ يَاْمُرُ بِالْفَحْشَآءِ وَالْمُنْكَرِ۔ اے ایمان والو! شیطان کے قدموں کے پیچھے نہ چلو؛ اور جو شیطان کے قدموں کے پیچھے چلتا ہے تو وہ تو بے حیائی اور برائی ہی کا حکم دیتا ہے۔ تہمت کا بیان مکمل ہوا تو حق تعالیٰ نے اس کی جڑ کی نشاندہی فرمائی۔ بات ایک سرگوشی سے چلی، پھر سننے، پھر دہرانے اور پھر پھیلانے تک پہنچی؛ یہی "خُطُوٰتِ الشَّيْطٰنِ" ہیں۔ قرآن نے قدم کہا، پورا راستہ نہیں کہا، کیونکہ شیطان کبھی پوری منزل ایک ساتھ نہیں دکھاتا؛ وہ صرف اگلا قدم دکھاتا ہے، اور ہر قدم پچھلے کے مقابلے میں چھوٹا لگتا ہے۔ اہلِ معرفت کہتے ہیں کہ کوئی بڑا گناہ اچانک سرزد نہیں ہوتا؛ وہ چھوٹی چھوٹی غفلتوں کا آخری قدم ہوتا ہے، اس لیے حفاظت پہلے قدم پر ہے۔ اور یہ بھی دیکھیے کہ ممانعت پیروی کی ہے: شیطان دعوت دیتا ہے، چلنا بندے کا اپنا فعل ہے۔ سورۂ ابراہیم میں شیطان نے خود یہ اقرار کیا تھا: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں خواطر کی بحث میں شیطانی خاطر کی ایک پہچان یہ بیان ہوتی ہے کہ وہ بندے کو اس کے موجودہ مقام سے ذرا نیچے بلاتا ہے اور اس پستی کو ہلکا کر کے دکھاتا ہے۔ سالک کا کام یہ ہے کہ ہر خیال کو اس کے انجام تک دیکھ لے: جو قدم بے حیائی کی طرف کھلتا ہو، وہ کتنا ہی چھوٹا لگے، شیطان ہی کا قدم ہے۔ اور یاد رہے کہ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہری حکم اپنی جگہ قائم ہے کہ بدگوئی اور بے حیائی کی ہر راہ حرام ہے۔

پھر اسی آیت کا دوسرا حصہ: وَلَوْلَا فَضْلُ اللّٰهِ عَلَيْكُمْ وَرَحْمَتُهٗ مَا زَكٰى مِنْكُمْ مِّنْ اَحَدٍ اَبَدًا ۙ وَّلٰكِنَّ اللّٰهَ يُزَكِّيْ مَنْ يَّشَآءُ ۚ وَاللّٰهُ سَمِيْعٌ عَلِيْمٌ۔ اور اگر تم پر اللہ کا فضل اور اس کی رحمت نہ ہوتی تو تم میں سے کوئی ایک بھی کبھی پاک نہ ہو پاتا؛ لیکن اللہ جسے چاہتا ہے پاک کر دیتا ہے، اور اللہ سب کچھ سننے والا، سب کچھ جاننے والا ہے۔ یہ فقر کا اعلان ہے۔ کوئی اپنے زور سے پاک نہیں ہوتا؛ عبادت، مجاہدہ اور ریاضت سب دروازے پر دستک ہیں، دروازہ کھولنا اسی کا کام ہے۔ قرآن ہی میں دوسری جگہ فرمایا: قَدْ اَفْلَحَ مَنْ زَكّٰىهَا۔ دونوں باتیں سچی ہیں: رخ کرنا اور کوشش بندے کی ہے، پاکیزگی کا فیض حق کا ہے۔ یہیں وہ اصول آتا ہے جس کے بغیر "يُزَكِّيْ مَنْ يَّشَآءُ" کو سمجھا ہی نہیں جا سکتا۔ حق کسی طالب کو پاکیزگی سے محروم نہیں رکھتا، اور کسی کو زبردستی ناپاک نہیں چھوڑتا۔ جو اس کی طرف رخ کرتا ہے، اس کی مشیت اسی کو پاک کرنے کی ہے؛ اور جو منہ پھیر لیتا ہے، اس کا منہ پھیرنا ہی اس کی محرومی ہے، حق تو بس اس سمت کو وجود دیتا ہے جو بندے نے خود چنی۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے، اور مشیت بندے کی عینِ ثابتہ کے تقاضے پر جاری ہوتی ہے، اس پر جبر بن کر نہیں۔ اور اگر وہ چاہتا تو سب کو مجبوراً پاک کر دیتا، مگر اس نے اختیار کی عزت رکھی۔ اسی لیے آیت "سَمِيْعٌ عَلِيْمٌ" پر ختم ہوئی: وہ طالب کی پکار سنتا ہے اور اس کے رخ کی سچائی جانتا ہے؛ کوئی سچی پکار اس کی سماعت سے باہر نہیں، اور کوئی سچا رخ اس کے علم سے پوشیدہ نہیں۔

وَلَا يَاْتَلِ اُولُوا الْفَضْلِ مِنْكُمْ وَالسَّعَةِ اَنْ يُّؤْتُوْۤا اُولِي الْقُرْبٰى وَالْمَسٰكِيْنَ وَالْمُهٰجِرِيْنَ فِيْ سَبِيْلِ اللّٰهِ ۖ وَلْيَعْفُوْا وَلْيَصْفَحُوْا ۚ اَلَا تُحِبُّوْنَ اَنْ يَّغْفِرَ اللّٰهُ لَكُمْ ۚ وَاللّٰهُ غَفُوْرٌ رَّحِيْمٌ۔ اور تم میں سے جو فضل اور وسعت والے ہیں وہ قرابت داروں، مسکینوں اور اللہ کی راہ میں ہجرت کرنے والوں کو دینے سے قسم نہ کھا بیٹھیں؛ انہیں چاہیے کہ معاف کریں اور درگزر کریں۔ کیا تم یہ نہیں چاہتے کہ اللہ تمہیں بخش دے؟ اور اللہ بخشنے والا، مہربان ہے۔ مفسرین بیان کرتے ہیں کہ حضرت ابوبکر صدیق رضی اللہ عنہ اپنے ایک نادار قرابت دار حضرت مسطح رضی اللہ عنہ کی کفالت کرتے تھے، جو مہاجر بھی تھے اور بدری صحابی بھی؛ جب وہ بھی اس بات میں شریک ہو گئے تو آپ نے آئندہ مدد نہ کرنے کی قسم کھا لی۔ یہ آیت اتری تو آپ نے، روایت کا مفہوم یہ ہے، فرمایا کہ کیوں نہیں، میں ضرور چاہتا ہوں کہ اللہ مجھے بخش دے؛ اور مدد پھر سے جاری کر دی۔ زخم سب سے گہرا تھا، کیونکہ بات ام المؤمنین کی عزت کی تھی، پھر بھی حکم معافی کا آیا۔ عفو یہ ہے کہ خطا مٹا دی جائے، اور صفح یہ کہ اس کا صفحہ ہی پلٹ دیا جائے، یہاں تک کہ دل میں اس کا ذکر بھی باقی نہ رہے۔ اہلِ معرفت کہتے ہیں کہ "اَلَا تُحِبُّوْنَ اَنْ يَّغْفِرَ اللّٰهُ لَكُمْ" تخلق باسماء اللہ کی دعوت ہے: تم بندوں کے حق میں اسمِ غفور کے مظہر بنو، تو وہی اسم تم پر لوٹ کر برسے گا۔ شیخِ اکبر کے مکتب میں جزا عمل کی جنس سے ہے؛ جو بندوں کی خطاؤں پر پردہ ڈالتا ہے وہ اپنے لیے حق کی پردہ پوشی کا راستہ کھولتا ہے۔ اور "اُولُوا الْفَضْلِ" کا اصل فضل مال کی بہتات نہیں، دل کی وسعت ہے؛ مال والا وہ ہے جو دے سکے، اور فضل والا وہ ہے جو زخم کھا کر بھی دے سکے۔

اِنَّ الَّذِيْنَ يَرْمُوْنَ الْمُحْصَنٰتِ الْغٰفِلٰتِ الْمُؤْمِنٰتِ لُعِنُوْا فِي الدُّنْيَا وَالْاٰخِرَةِ ۖ وَلَهُمْ عَذَابٌ عَظِيْمٌ يَّوْمَ تَشْهَدُ عَلَيْهِمْ اَلْسِنَتُهُمْ وَاَيْدِيْهِمْ وَاَرْجُلُهُمْ بِمَا كَانُوْا يَعْمَلُوْنَ۔ بے شک جو لوگ پاک دامن، بے خبر، ایمان والی عورتوں پر تہمت لگاتے ہیں ان پر دنیا اور آخرت میں لعنت کی گئی، اور ان کے لیے بڑا عذاب ہے، اس دن جب ان کی زبانیں، ان کے ہاتھ اور ان کے پاؤں ان کے خلاف ان کاموں کی گواہی دیں گے جو وہ کرتے تھے۔ ان خواتین کی تین صفات بیان ہوئیں اور تینوں سراسر عزت کی ہیں: "الْمُحْصَنٰتِ" یعنی وہ جن کی عصمت محفوظ ہے؛ "الْغٰفِلٰتِ" یعنی جن کے دل برائی کے خیال سے ناآشنا ہیں اور جنہیں خبر تک نہیں کہ ان کے بارے میں کیا کہا جا رہا ہے؛ اور "الْمُؤْمِنٰتِ"۔ یہاں غفلت عیب نہیں، معصومیت ہے: دل اتنا صاف کہ بدگمانی کا راستہ ہی نہیں جانتا۔ ظاہری حکم وہی ہے جو اس سورت کی چوتھی آیت میں گزرا، اور اس کا نفاذ صرف شرعی عدالت کا کام ہے، کسی فرد کا نہیں؛ یہاں اس گناہ کی سنگینی آخرت کے پیمانے پر بیان ہوئی۔ پھر گواہی کا عجیب نقشہ: جس زبان نے جھوٹ کہا وہی سچ بولے گی۔ ایک اور مقام پر اعضاء کا جواب یوں نقل ہوا: اَنْطَقَنَا اللّٰهُ الَّذِيْۤ اَنْطَقَ كُلَّ شَيْءٍ۔ شیخِ اکبر کے نزدیک کائنات کی ہر شے اپنی جگہ زندہ اور اپنے رب کو پہچاننے والی ہے؛ اعضاء بندے کی ملکیت نہیں، اس کے پاس امانت ہیں، اور ان کا اپنا علم ہے۔ آج وہ بندے کے حکم میں ہیں، اس دن حق کے حکم میں ہوں گے۔ اہلِ ذوق کہتے ہیں کہ سالک آج ہی اپنے اعضاء کو گواہ سمجھ کر برتے، تو کل کی گواہی اس کے حق میں ہو گی، خلاف نہیں۔

يَوْمَئِذٍ يُّوَفِّيْهِمُ اللّٰهُ دِيْنَهُمُ الْحَقَّ وَيَعْلَمُوْنَ اَنَّ اللّٰهَ هُوَ الْحَقُّ الْمُبِيْنُ۔ اس دن اللہ انہیں ان کا ٹھیک ٹھیک بدلہ پورا پورا دے گا، اور وہ جان لیں گے کہ اللہ ہی حق ہے، جو سب کچھ کھول کر ظاہر کر دینے والا ہے۔ "دِيْنَهُمُ" یہاں جزا کے معنی میں ہے اور "الْحَقَّ" اس کی صفت: بدلہ ذرہ بھر کم نہ ذرہ بھر زیادہ۔ اہلِ معرفت کہتے ہیں کہ یہ جزا باہر سے لائی ہوئی کوئی اجنبی چیز نہیں، بندے کا اپنا عمل ہے جو اپنی اصل صورت میں اسے لوٹا دیا جاتا ہے؛ یہاں بھی حکم محکوم کے مطابق ہے۔ پھر فرمایا: "هُوَ الْحَقُّ الْمُبِيْنُ"۔ اس مکتب میں وحدتِ وجود کا مفہوم یہی ہے کہ وجودِ حقیقی ایک ہی ہے، اور ہر ظاہر ہونے والی شے اسی کے نور سے ظاہر ہے۔ وہ آج بھی ظاہر ہے، بلکہ ہر ظہور اسی کا ظہور ہے، مگر مخلوق کی صورتیں نظر کے سامنے پردہ بن جاتی ہیں، اور آنکھ صورت پر رک کر صورت دینے والے کو بھول جاتی ہے۔ عارف یہ حقیقت یہیں کشف اور ذوق سے جان لیتا ہے؛ منکر اسے اس دن جانے گا جب پردے اٹھا دیے جائیں گے۔ تب یہ جاننا محض علم الیقین نہیں، عین الیقین ہو گا، مگر عمل کا وقت گزر چکا ہو گا۔ اسی لیے اہلِ ذوق کہتے ہیں کہ آج کی معرفت رحمت ہے اور کل کی معرفت صرف گواہی۔ جو زبان آج "هُوَ الْحَقُّ" کہنا سیکھ لے، وہ کل اپنے خلاف گواہ نہیں بنتی۔

اَلْخَبِيْثٰتُ لِلْخَبِيْثِيْنَ وَالْخَبِيْثُوْنَ لِلْخَبِيْثٰتِ ۚ وَالطَّيِّبٰتُ لِلطَّيِّبِيْنَ وَالطَّيِّبُوْنَ لِلطَّيِّبٰتِ ۚ اُولٰۤئِكَ مُبَرَّءُوْنَ مِمَّا يَقُوْلُوْنَ ۚ لَهُمْ مَّغْفِرَةٌ وَّرِزْقٌ كَرِيْمٌ۔ ناپاک عورتیں ناپاک مردوں کے لیے ہیں اور ناپاک مرد ناپاک عورتوں کے لیے، اور پاکیزہ عورتیں پاکیزہ مردوں کے لیے ہیں اور پاکیزہ مرد پاکیزہ عورتوں کے لیے؛ یہ لوگ ان باتوں سے بری ہیں جو یہ کہتے ہیں؛ ان کے لیے مغفرت اور عزت کی روزی ہے۔ مفسرین نے اس کے دو معنی بیان کیے ہیں: ناپاک باتیں ناپاک لوگوں کے لائق ہیں اور پاکیزہ باتیں پاکیزہ لوگوں کے؛ یا ناپاک لوگ ناپاک لوگوں کے لیے اور پاکیزہ لوگ پاکیزہ لوگوں کے لیے۔ دونوں معنی ایک ہی نتیجے تک پہنچاتے ہیں: مخلوق میں سب سے پاکیزہ ذات حضورِ اکرم صلی اللہ علیہ وسلم کی ہے، تو ان کا گھرانہ پاکیزگی ہی کا گھرانہ ہو سکتا ہے۔ "اُولٰۤئِكَ مُبَرَّءُوْنَ مِمَّا يَقُوْلُوْنَ" ام المؤمنین حضرت عائشہ صدیقہ رضی اللہ عنہا کی براءت کی وہ مہر ہے جو سات آسمانوں کے اوپر سے لگی، اور قیامت تک تلاوت ہوتی رہے گی؛ پھر ان کے لیے مغفرت اور رزقِ کریم کا وعدہ بھی ساتھ آیا۔ شیخِ اکبر کے مکتب میں مناسبت کو کشش کا قانون کہا گیا ہے: ہر شے اسی کی طرف کھنچتی ہے جو اس کی عین سے مناسبت رکھتی ہو، اور دو چیزوں میں جوڑ کسی باطنی ہم رنگی ہی سے پیدا ہوتا ہے۔ اہلِ ذوق اس سے یہ سبق لیتے ہیں کہ جو پاکیزہ صحبت چاہے وہ پہلے خود پاکیزہ ہو؛ جس صحبت کی طرف دل کھنچتا ہے وہ دراصل اپنے ہی حال کا آئینہ ہے۔ اور "رِزْقٌ كَرِيْمٌ" وہ روزی ہے جس میں ذلت کا شائبہ نہ ہو: جن کی عزت پر زبانیں دراز ہوئیں، ان کی عزت کا ضامن خود حق ہو گیا۔
