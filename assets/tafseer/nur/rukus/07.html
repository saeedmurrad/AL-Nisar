پچھلا رکوع ان لوگوں پر ختم ہوا تھا جو فیصلے کے وقت کترا جاتے تھے؛ یہ رکوع ایمان والوں کی بولی سے شروع ہوتا ہے: اِنَّمَا كَانَ قَوْلَ الْمُؤْمِنِيْنَ اِذَا دُعُوْۤا اِلَى اللّٰهِ وَرَسُوْلِهٖ لِيَحْكُمَ بَيْنَهُمْ اَنْ يَّقُوْلُوْا سَمِعْنَا وَاَطَعْنَا ۭ وَاُولٰۤئِكَ هُمُ الْمُفْلِحُوْنَ۔ ایمان والوں کی بات تو، جب انہیں اللہ اور اس کے رسول کی طرف بلایا جائے تاکہ رسول ان کے درمیان فیصلہ کرے، بس یہی ہوتی ہے کہ وہ کہیں: ہم نے سنا اور مان لیا؛ اور یہی لوگ فلاح پانے والے ہیں۔ تقابل لفظوں میں چھپا ہے: منافقین نے بھی کہا تھا کہ ہم نے اطاعت کی، پھر منہ موڑ لیا؛ مومن بھی وہی لفظ کہتا ہے مگر اس سے پہلے سننے کو رکھتا ہے۔ بے سنے ماننا دعویٰ ہے اور سن کر ماننا حال۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر موجود نے پہلے کلمۂ کن سنا اور فوراً وجود میں آ کر اطاعت کی؛ یہ وجودی سمع و طاعت ہر شے میں جاری ہے۔ مومن کا اَنْ يَّقُوْلُوْا سَمِعْنَا وَاَطَعْنَا کہنا اسی وجودی اطاعت کو اپنے اختیار سے دہرانا ہے: جو اطاعت اس کے وجود نے بے اختیار کی، وہی اس کا ارادہ بھی جان بوجھ کر کرے۔ پھر فرمایا: وَمَنْ يُّطِعِ اللّٰهَ وَرَسُوْلَهٗ وَيَخْشَ اللّٰهَ وَيَتَّقْهِ فَاُولٰۤئِكَ هُمُ الْفَآئِزُوْنَ۔ اور جو اللہ اور اس کے رسول کی اطاعت کرے، اللہ سے ڈرے اور اس کی نافرمانی سے بچے، تو وہی لوگ کامیاب ہیں۔ اہلِ معرفت خشیت اور تقویٰ میں فرق کرتے ہیں: خشیت معرفت سے پیدا ہونے والی وہ ہیبت ہے جو دل میں رہتی ہے، اور تقویٰ اس ہیبت کا وہ بچاؤ جو عمل میں ظاہر ہو۔ پہلے اطاعت، پھر خشیت، پھر تقویٰ: ظاہر سے باطن کی طرف اور باطن سے پھر ظاہر کی طرف ایک مکمل دائرہ۔ اور فلاح کے بعد فوز کا لفظ ایک قدم آگے ہے؛ فلاح مراد کو پہنچنے کی کامیابی ہے اور فوز مقصود کو پا کر ہر خطرے سے نکل جانا۔

اب زبانی قسموں اور سچی اطاعت کا فرق دکھایا: وَاَقْسَمُوْا بِاللّٰهِ جَهْدَ اَيْمَانِهِمْ لَئِنْ اَمَرْتَهُمْ لَيَخْرُجُنَّ ۭ قُلْ لَّا تُقْسِمُوْا ۚ طَاعَةٌ مَّعْرُوْفَةٌ ۭ اِنَّ اللّٰهَ خَبِيْرٌۢ بِمَا تَعْمَلُوْنَ۔ اور انہوں نے اللہ کی پکی سے پکی قسمیں کھائیں کہ اگر آپ انہیں حکم دیں تو وہ ضرور نکل کھڑے ہوں گے؛ کہہ دیجیے کہ قسمیں نہ کھاؤ، دستور کے مطابق اطاعت مطلوب ہے؛ بے شک اللہ تمہارے کاموں سے پوری طرح باخبر ہے۔ ظاہری معنی یہ ہے کہ منافقین نکلنے کی بھاری قسمیں کھاتے اور عمل کا وقت آتا تو پیچھے رہ جاتے؛ جواب ملا کہ قسم کی ضرورت نہیں، جانی پہچانی اطاعت خود گواہ ہے۔ مفسرین نے لَا تُقْسِمُوْا ۚ طَاعَةٌ مَّعْرُوْفَةٌ کے دو معنی بیان کیے ہیں: مطلوب وہ اطاعت ہے جو معروف ہو، یا تمہاری اطاعت کی حقیقت سب کو معلوم ہے۔ اہلِ ذوق کہتے ہیں کہ نفس جتنا سچائی میں کمزور ہو اتنی ہی زیادہ قسمیں کھاتا ہے، اور سچا عمل قسم کا محتاج نہیں ہوتا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: معروف کا مادہ وہی ہے جو معرفت کا؛ اس مکتب میں اصل اطاعت وہ ہے جو معرفت سے پھوٹے، یعنی عارف اس لیے مانتا ہے کہ اس نے پہچان لیا، اس لیے نہیں کہ اس نے قسم کھائی۔ اور آخر کی خبر، اِنَّ اللّٰهَ خَبِيْرٌۢ بِمَا تَعْمَلُوْنَ، مراقبہ کی بنیاد ہے: جو جانتا ہے کہ وہ باخبر ہے اسے اپنی سچائی ثابت کرنے کے لیے کسی قسم یا گواہ کی ضرورت نہیں رہتی۔

اس کے بعد وہ آیت آئی جو اس سورت میں ہدایت کے اصول کو خود اپنی زبان سے ادا کرتی ہے: قُلْ اَطِيْعُوا اللّٰهَ وَاَطِيْعُوا الرَّسُوْلَ ۚ فَاِنْ تَوَلَّوْا فَاِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُمْ مَّا حُمِّلْتُمْ ۭ وَاِنْ تُطِيْعُوْهُ تَهْتَدُوْا ۭ وَمَا عَلَى الرَّسُوْلِ اِلَّا الْبَلٰغُ الْمُبِيْنُ۔ کہہ دیجیے: اللہ کی اطاعت کرو اور رسول کی اطاعت کرو؛ پھر اگر تم منہ موڑو تو رسول کے ذمے وہی ہے جس کا بوجھ اس پر ڈالا گیا اور تمہارے ذمے وہ جس کا بوجھ تم پر ڈالا گیا؛ اور اگر تم اس کی اطاعت کرو گے تو ہدایت پاؤ گے؛ اور رسول کے ذمے تو بس صاف صاف پہنچا دینا ہے۔ یہی وہ اصول ہے جس سے پچھلے رکوع میں آیت ۴۶ پڑھی گئی، اور اسے اصول ہی کے طور پر پڑھنا چاہیے: وَاِنْ تُطِيْعُوْهُ تَهْتَدُوْا۔ ہدایت شرط کے جواب میں آئی ہے؛ رخ بندے کا ہے اور عطا حق کی۔ جو طلب اور اطاعت کے ساتھ آئے اسے ہدایت ملتی ہے، اور جو فَاِنْ تَوَلَّوْا فَاِنَّمَا عَلَيْهِ کے مطابق منہ موڑے، اس کا منہ موڑنا اسی کا اپنا ہے اور وہی اس کی گمراہی ہے۔ نہ رسول کسی پر زور ڈالتا ہے، نہ حق کسی کو جبراً گمراہ کرتا ہے؛ وہ چاہتا تو سب کو مجبور کر دیتا، مگر اس نے اختیار کی کرامت باقی رکھی، اور بندہ جس سمت کو چنتا ہے حق اسی کو وجود دیتا ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔

اسی آیت کے دو اور جملے غور چاہتے ہیں۔ عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُمْ مَّا حُمِّلْتُمْ میں ہر ایک کی ذمہ داری الگ کر دی گئی؛ سورۂ بنی اسرائیل میں گزرا تھا کہ ہر انسان کا نامۂ عمل اس کی گردن سے لازم کر دیا گیا ہے، اور یہاں وہی حقیقت بوجھ اور امانت کے لفظ میں ہے۔ اہلِ معرفت کہتے ہیں کہ تحمیل کا صیغہ مجہول ہے: بوجھ اٹھانے والا خود بوجھ کا انتخاب نہیں کرتا، وہ اس کی استعداد کے مطابق اس پر رکھا جاتا ہے؛ رسول پر رسالت کا بار اور امت پر قبول کا۔ اور اِلَّا الْبَلٰغُ الْمُبِيْنُ: حضورِ اکرم صلی اللہ علیہ وسلم کا کام پیغام کو پوری وضاحت سے پہنچانا تھا، اور آپ نے اسے کامل طور پر ادا فرمایا۔ شیخِ اکبر کے مکتب میں حضورِ اکرم صلی اللہ علیہ وسلم خطابِ الٰہی کے کامل مظہر ہیں، اس لیے آپ کی اطاعت اللہ کی اطاعت ہے، جیسا قرآن نے دوسری جگہ کہا: مَنْ يُّطِعِ الرَّسُوْلَ فَقَدْ اَطَاعَ اللّٰهَ۔ بلاغ کامل ہو چکا؛ اب قبول کا دروازہ بندے کے سامنے کھلا ہے، اور جو اس میں داخل ہو، اس کے لیے وَاِنْ تُطِيْعُوْهُ تَهْتَدُوْا کا وعدہ پورا ہے۔

اطاعت والوں سے پھر ایک عظیم وعدہ فرمایا: وَعَدَ اللّٰهُ الَّذِيْنَ اٰمَنُوْا مِنْكُمْ وَعَمِلُوا الصّٰلِحٰتِ لَيَسْتَخْلِفَنَّهُمْ فِي الْاَرْضِ كَمَا اسْتَخْلَفَ الَّذِيْنَ مِنْ قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِيْنَهُمُ الَّذِي ارْتَضٰى لَهُمْ وَلَيُبَدِّلَنَّهُمْ مِّنْۢ بَعْدِ خَوْفِهِمْ اَمْنًا ۭ يَعْبُدُوْنَنِيْ لَا يُشْرِكُوْنَ بِيْ شَيْـًٔا ۭ وَمَنْ كَفَرَ بَعْدَ ذٰلِكَ فَاُولٰۤئِكَ هُمُ الْفٰسِقُوْنَ۔ اللہ نے تم میں سے ان لوگوں سے جو ایمان لائے اور نیک عمل کیے وعدہ فرمایا ہے کہ وہ انہیں زمین میں ضرور خلافت دے گا جیسے ان سے پہلے لوگوں کو خلافت دی تھی، اور ان کے لیے ان کے اس دین کو ضرور جما دے گا جسے اس نے ان کے لیے پسند فرمایا، اور ان کے خوف کے بعد اسے ضرور امن سے بدل دے گا؛ وہ میری عبادت کریں گے اور میرے ساتھ کسی چیز کو شریک نہ ٹھہرائیں گے؛ اور جو اس کے بعد ناشکری کرے تو وہی لوگ نافرمان ہیں۔ ظاہر میں یہ وعدہ صحابۂ کرام رضی اللہ عنہم کے حق میں پورا ہوا، جیسا مفسرین نے بیان کیا: خوف کے دن امن میں بدلے اور دین زمین میں جم گیا۔ وعدہ مشروط ہے: ایمان، عملِ صالح اور خالص عبادت۔ اہلِ معرفت کے نزدیک يَعْبُدُوْنَنِيْ لَا يُشْرِكُوْنَ بِيْ شَيْـًٔا کے آخری لفظ کی تنکیر ہر شرک کو سمیٹ لیتی ہے، اس باریک شرک کو بھی کہ بندہ اپنے آپ کو فاعل دیکھے۔ شیخِ اکبر کے مکتب میں خلافت ملکیت کا نام نہیں؛ خلیفہ اپنے آپ کو نہیں، اسے دکھاتا ہے جس کا وہ نائب ہے، اور جس دن وہ اپنے آپ کو دکھانے لگے، نیابت کی حقیقت رخصت ہو جاتی ہے۔ اسی لیے وعدے کی شرط عبادت رکھی گئی، کیونکہ عبدیت ہی خلافت کی بنیاد ہے۔

وَلَيُمَكِّنَنَّ لَهُمْ دِيْنَهُمُ الَّذِي ارْتَضٰى لَهُمْ: تمکین کا لفظ اہلِ تصوف کی ایک مشہور اصطلاح کی جڑ ہے۔ تلوین وہ حال ہے جس میں سالک کے احوال بدلتے رہتے ہیں، اور تمکین وہ مقام جس میں وہ ٹھہر جاتا ہے؛ اشاری طور پر وعدہ یہ ہے کہ اطاعت والے کا دین اس کے دل میں تلوین سے تمکین تک پہنچا دیا جائے گا، اور وَلَيُبَدِّلَنَّهُمْ مِّنْۢ بَعْدِ خَوْفِهِمْ اَمْنًا اسی ٹھہراؤ کا پھل ہے۔ دل کی خلافت بھی یہی ہے: انسانِ کامل اسمِ جامع کا مظہر ہو کر زمین میں نائب ہے، اور ہر مومن اپنے دل کی سلطنت میں، جہاں نفس اور قویٰ اس کی رعیت ہیں۔ تمکین کے اسی لفظ نے پچھلی سورتوں کو جوڑ دیا: سورۂ الکہف میں ذوالقرنین کے حق میں اِنَّا مَكَّنَّا لَهٗ فِي الْاَرْضِ آیا، جس کے بعد اسباب کا سفر شروع ہوا؛ اور سورۂ الحج میں تمکین پانے والوں کی پہچان بتائی گئی: اَلَّذِيْنَ اِنْ مَّكَّنّٰهُمْ فِي الْاَرْضِ اَقَامُوا الصَّلٰوةَ وَاٰتَوُا الزَّكٰوةَ۔ تینوں جگہ تمکین امانت ہے، انعام کے ساتھ آزمائش۔ اور آخر میں وَمَنْ كَفَرَ بَعْدَ ذٰلِكَ فَاُولٰۤئِكَ هُمُ الْفٰسِقُوْنَ: نعمت کی تکمیل کے بعد ناشکری کا وزن بھی بڑھ جاتا ہے؛ کفر یہاں نعمت کو ڈھانپنا ہے، اور یہ ڈھانپنا بندے کا اپنا فعل ہے، اس پر ڈالا نہیں گیا۔

اسی لیے اگلی آیت نے وہی کام سونپے جو سورۂ الحج نے تمکین والوں کی پہچان بتائے تھے: وَاَقِيْمُوا الصَّلٰوةَ وَاٰتُوا الزَّكٰوةَ وَاَطِيْعُوا الرَّسُوْلَ لَعَلَّكُمْ تُرْحَمُوْنَ۔ اور نماز قائم کرو اور زکوٰۃ ادا کرو اور رسول کی اطاعت کرو تاکہ تم پر رحم کیا جائے۔ اہلِ معرفت کے نزدیک نماز وصل کا رشتہ ہے، بندے اور رب کے درمیان مناجات؛ اور زکوٰۃ پاکیزگی بھی ہے اور بڑھوتری بھی: بندہ وہ مال دیتا ہے جو حقیقت میں اسی کا تھا جس نے اسے عطا کیا۔ پہلی عبادت حق کی طرف رخ ہے، دوسری خلق کی طرف، اور تیسری، یعنی رسول کی اطاعت، دونوں رخوں کو درست رکھنے کی ضامن۔ اور ثمرہ رحمت ہے؛ سبقتِ رحمت کا اصول یہاں بھی ہے: اطاعت رحمت کی قیمت نہیں، رحمت کے دروازے پر کھڑا ہونا ہے۔ پھر رکوع اس تنبیہ پر ختم ہوا: لَا تَحْسَبَنَّ الَّذِيْنَ كَفَرُوْا مُعْجِزِيْنَ فِي الْاَرْضِ ۚ وَمَاْوٰىهُمُ النَّارُ ۭ وَلَبِئْسَ الْمَصِيْرُ۔ ان کافروں کے بارے میں ہرگز یہ گمان نہ کرنا کہ وہ زمین میں اللہ کو عاجز کر دیں گے؛ ان کا ٹھکانا آگ ہے، اور وہ بہت برا ٹھکانا ہے۔ شیخِ اکبر کے مکتب میں کوئی حق سے بھاگ کر کہاں جائے؟ اس کے وجود سے باہر کوئی جگہ نہیں جہاں پناہ ملے؛ بھاگنے والا جس طرف بھاگے اسی کی بادشاہی میں ہے، جیسا اسی سورت کی آیت ۴۲ نے کہا تھا کہ ملک اسی کا ہے اور مصیر اسی کی طرف۔ دونوں جگہ مصیر کا لفظ آیا: وہاں اللہ کی طرف لوٹنا، یہاں برا انجام؛ اور فرق اس راستے کا ہے جو بندے نے خود چنا۔ اہلِ معرفت کے نزدیک ٹھکانا عمل ہی کی وہ صورت ہے جو آخرت میں ظاہر ہوتی ہے؛ حکم محکوم کے مطابق، اور یہی عدل ہے۔ اور اس رکوع کا آخری درس یہ ہے کہ ہدایت کی راہ ہر لمحہ کھلی ہے، اور جو آج سچے دل سے کہہ دے کہ ہم نے سنا اور مانا، اس کے لیے یہ دروازہ کبھی بند نہیں ہوتا۔
