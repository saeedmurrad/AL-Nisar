گھر کی حرمت کا جو سبق چوتھے رکوع میں اجازت اور سلام سے شروع ہوا تھا، سورت کے آخر میں وہ گھر کے اندر تک پہنچ جاتا ہے؛ اب اجازت باہر والوں سے نہیں، گھر ہی کے لوگوں سے مانگی جاتی ہے۔ يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لِيَسْتَاْذِنْكُمُ الَّذِيْنَ مَلَكَتْ اَيْمَانُكُمْ وَالَّذِيْنَ لَمْ يَبْلُغُوا الْحُلُمَ مِنْكُمْ ثَلٰثَ مَرّٰتٍ۔ اے ایمان والو! تمہارے غلام اور تمہارے وہ بچے جو ابھی بلوغ کو نہیں پہنچے، تین وقتوں میں تم سے اجازت لے کر آیا کریں۔ پھر وہ تین وقت گنوائے گئے: مِنْ قَبْلِ صَلٰوةِ الْفَجْرِ وَحِيْنَ تَضَعُوْنَ ثِيَابَكُمْ مِّنَ الظَّهِيْرَةِ وَمِنْۢ بَعْدِ صَلٰوةِ الْعِشَآءِ ثَلٰثُ عَوْرٰتٍ لَّكُمْ۔ فجر کی نماز سے پہلے، دوپہر کو جب تم اپنے کپڑے اتار رکھتے ہو، اور عشاء کی نماز کے بعد؛ یہ تین وقت تمہارے پردے کے ہیں۔ ظاہر میں یہ گھر کا ادب ہے: آدمی کے کچھ لمحے آرام اور بے تکلفی کے ہوتے ہیں، اور شریعت نے انہیں چھوٹوں اور خادموں کی بے خبر آمد سے بھی محفوظ کر دیا۔ اس زمانے میں غلامی ایک رائج نظام تھا جسے شریعت نے آزادی کی طرف موڑنا شروع کیا، جیسا اسی سورت میں مکاتبت کا حکم گزرا؛ اور اہلِ علم نے اجازت کا یہ ادب گھر کے خادموں پر بھی جاری سمجھا ہے۔ مفسرین نقل کرتے ہیں کہ حضرت ابنِ عباس رضی اللہ عنہما اس پر افسوس کرتے تھے کہ لوگ اس آیت کے ادب میں سستی کرنے لگے ہیں۔ عَوْرَة کا اصل معنی وہ کمزور اور کھلی جگہ ہے جسے ڈھانپ کر رکھا جائے؛ یہاں یہ لفظ وقت پر بولا گیا، گویا جسم کی طرح زمانے کا بھی ستر ہوتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر وقت کا اپنا حکم ہے اور عارف ابن الوقت ہوتا ہے؛ اور غور ہو کہ تینوں اوقات برزخ کے لمحے ہیں: رات سے دن، مصروفیت سے آرام، اور دن سے رات کی طرف گزرنا۔ انہی دہلیزوں پر آدمی ایک حال کا لباس اتار کر دوسرے کا پہنتا ہے، اور اسی لیے یہ لمحے پردے کے مستحق ٹھہرے۔

پردے کے وقت مقرر کر کے فوراً آسانی بھی بیان کر دی: لَيْسَ عَلَيْكُمْ وَلَا عَلَيْهِمْ جُنَاحٌۢ بَعْدَهُنَّ طَوّٰفُوْنَ عَلَيْكُمْ بَعْضُكُمْ عَلٰى بَعْضٍ۔ ان اوقات کے سوا نہ تم پر کوئی گناہ ہے نہ ان پر؛ وہ تمہارے پاس بار بار آنے جانے والے ہیں، تم ایک دوسرے کے پاس آتے جاتے رہتے ہو۔ شریعت نے گھر کو قید خانہ نہیں بنایا؛ جو لوگ دن بھر ایک دوسرے کی خدمت میں پھرتے ہوں، ان پر ہر بار اجازت کی پابندی تنگی ہوتی۔ طواف کا لفظ یہاں بڑی لطافت رکھتا ہے: گھر والے ایک دوسرے کے گرد خدمت اور محبت کے دائرے میں گھومتے ہیں، اور سورۂ الحج کے قارئین جانتے ہیں کہ طواف قربت کی وہ صورت ہے جس میں مرکز ایک اور گردش کرنے والے بہت ہوتے ہیں۔ پھر فرمایا: كَذٰلِكَ يُبَيِّنُ اللّٰهُ لَكُمُ الْاٰيٰتِ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ اسی طرح اللہ تمہارے لیے آیتیں کھول کر بیان کرتا ہے، اور اللہ سب کچھ جاننے والا، حکمت والا ہے۔ شیخِ اکبر کے نزدیک حکیم وہ ہے جو ہر شے کو اس کے مقام پر رکھے؛ علم نے جانا کہ کون سا لمحہ کیا مانگتا ہے، اور حکمت نے ہر لمحے کو اس کا حق دیا، پردے کو پردہ اور آسانی کو آسانی۔ اہلِ تاویل یہاں دل کے گھر کی طرف بھی اشارہ کرتے ہیں: دل کے بھی کچھ اوقات ہیں، سحر کی مناجات، دن کی بھیڑ کے بیچ ایک لمحۂ حضوری، اور رات کی خلوت، جن میں دن بھر گھومنے والے خیالات اور حواس کو بھی بے اجازت اندر نہیں آنا چاہیے۔ یہ محض تمثیل ہے، اس میں بچوں یا خادموں کی کسی تحقیر کا شائبہ نہیں؛ ظاہر حکم اپنی جگہ قائم ہے، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر بچپن کی رخصت کی حد مقرر ہوئی: وَاِذَا بَلَغَ الْاَطْفَالُ مِنْكُمُ الْحُلُمَ فَلْيَسْتَاْذِنُوْا كَمَا اسْتَاْذَنَ الَّذِيْنَ مِنْ قَبْلِهِمْ۔ اور جب تمہارے بچے بلوغ کو پہنچ جائیں تو وہ بھی اسی طرح اجازت لیا کریں جیسے ان سے پہلے کے بڑے اجازت لیتے رہے ہیں۔ یعنی اب تین وقتوں کی قید نہیں رہی، ہر وقت اجازت لازم ہے۔ وہی بچہ، وہی گھر، وہی ماں باپ؛ صرف حال بدلا اور حکم بدل گیا۔ اہلِ معرفت کے نزدیک یہ سرِّ قدر کی ایک چھوٹی سی تصویر ہے: حکم محکوم کے مطابق ہوتا ہے، اور جیسا کسی کا حال ہو ویسا ہی اس پر حکم اترتا ہے۔ سلوک میں بھی یہی قاعدہ ہے: راہ کے آغاز میں جو بے تکلفی معاف تھی، قرب بڑھنے کے ساتھ وہی بے ادبی بن جاتی ہے؛ اہلِ تصوف میں مشہور اثر ہے کہ نیکوں کی نیکیاں مقربین کے حق میں کوتاہیاں شمار ہوتی ہیں۔ بلوغ صرف جسم کا نہیں، باطن کا بھی ہے، اور باطن جتنا بالغ ہو ادب اتنا گہرا ہوتا ہے۔ پھر وہی اختتام لوٹا، مگر ایک لفظ کے فرق کے ساتھ: كَذٰلِكَ يُبَيِّنُ اللّٰهُ لَكُمْ اٰيٰتِهٖ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ اسی طرح اللہ تمہارے لیے اپنی آیتیں کھول کر بیان کرتا ہے، اور اللہ جاننے والا، حکمت والا ہے۔ پچھلی آیت میں آیات کا ذکر عام تھا، یہاں ضمیر کے ساتھ اپنی آیتیں فرمایا۔ اہلِ ذوق کہتے ہیں کہ جیسے جیسے بندہ بالغ ہوتا ہے، اسے احکام محض قاعدے نظر نہیں آتے بلکہ اپنے رب کی نشانیاں دکھائی دینے لگتی ہیں، جن میں حکم دینے والے کا چہرہ جھلکتا ہے۔

گھر کی اس ترتیب میں عمر کے آخری حصے کو بھی عزت کے ساتھ جگہ ملی: وَالْقَوَاعِدُ مِنَ النِّسَآءِ الّٰتِيْ لَا يَرْجُوْنَ نِكَاحًا فَلَيْسَ عَلَيْهِنَّ جُنَاحٌ اَنْ يَّضَعْنَ ثِيَابَهُنَّ غَيْرَ مُتَبَرِّجٰتٍۭ بِزِيْنَةٍ۔ اور وہ بڑی عمر کی خواتین جو نکاح کی امید نہیں رکھتیں، ان پر کوئی گناہ نہیں کہ اپنی اوپر کی چادریں اتار رکھیں، بشرطیکہ زینت کی نمائش کرنے والی نہ ہوں۔ مفسرین واضح کرتے ہیں کہ یہاں کپڑوں سے مراد اوپر کی بڑی چادر ہے، اصل لباس اور ستر کا حکم اپنی جگہ باقی ہے۔ قَوَاعِد کا مادہ قعود ہے، یعنی وہ جو زندگی کا طویل سفر طے کر کے بیٹھ چکیں؛ یہ امت کی مائیں اور نانیاں دادیاں ہیں، اور یہ رخصت ان کے احترام اور ان پر شفقت کی علامت ہے، کسی کمی کا بیان نہیں۔ پھر فرمایا: وَاَنْ يَّسْتَعْفِفْنَ خَيْرٌ لَّهُنَّ وَاللّٰهُ سَمِيْعٌ عَلِيْمٌ۔ اور اگر وہ اس سے بھی بچیں تو یہ ان کے لیے بہتر ہے، اور اللہ سب کچھ سننے والا، جاننے والا ہے۔ یہاں رخصت اور عزیمت دونوں ایک ساتھ رکھ دی گئیں۔ ایک حدیث کا مفہوم ہے کہ اللہ کو یہ پسند ہے کہ اس کی رخصتوں پر عمل کیا جائے جیسے اسے یہ پسند ہے کہ اس کے عزائم پر عمل ہو۔ اس مکتب میں رخصت بھی حق کا عطیہ ہے، اسے قبول کرنا ادب ہے؛ اور عزیمت کا دروازہ کھلا رکھ کر اسے لازم نہ کرنا انتخاب کی عزت ہے۔ اور آیت سمیع و علیم پر ختم ہوئی، کیونکہ نمائش اور حیا کا اصل فیصلہ نیت میں ہوتا ہے، جسے وہی سنتا اور جانتا ہے جو دل کی ان کہی بات بھی سن لیتا ہے۔

اب دسترخوان کی آسانیاں، اور سب سے پہلے ان کا ذکر جو کمزور سمجھے جاتے ہیں: لَيْسَ عَلَى الْاَعْمٰى حَرَجٌ وَّلَا عَلَى الْاَعْرَجِ حَرَجٌ وَّلَا عَلَى الْمَرِيْضِ حَرَجٌ۔ نہ اندھے پر کوئی تنگی ہے، نہ لنگڑے پر، نہ بیمار پر۔ مفسرین کے کئی اقوال ہیں: بعض کے نزدیک بعض صحابہ ان معذور بھائیوں کے ساتھ کھانے میں اس خیال سے جھجکتے تھے کہ کہیں ان کا حق نہ مار لیں، اور بعض کے نزدیک خود وہ دوسروں کے گھروں میں کھانے سے جھجکتے تھے؛ اور یہی الفاظ ایک دوسری آیت میں جہاد کی رخصت کے لیے بھی آئے ہیں۔ شریعت ہر ایک کو اس کی وسعت کے مطابق ناپتی ہے، جیسا فرمایا: لَا يُكَلِّفُ اللّٰهُ نَفْسًا اِلَّا وُسْعَهَا۔ شیخِ اکبر کے نزدیک ہر عین کو فیض اس کی استعداد کے مطابق ملتا ہے، اور تکلیف بھی اسی پیمانے پر اترتی ہے؛ سو معذور کم درجے کا نہیں، اس کا پیمانہ مختلف ہے، اور دسترخوان پر وہ معزز مہمان ہے۔ پھر رشتوں کے دائرے: وَّلَا عَلٰۤى اَنْفُسِكُمْ اَنْ تَاْكُلُوْا مِنْۢ بُيُوْتِكُمْ اَوْ بُيُوْتِ اٰبَآئِكُمْ اَوْ بُيُوْتِ اُمَّهٰتِكُمْ۔ اور نہ خود تم پر کہ اپنے گھروں سے کھاؤ، یا اپنے باپوں کے گھروں سے، یا اپنی ماؤں کے گھروں سے؛ پھر بھائیوں، بہنوں، چچاؤں، پھوپھیوں، ماموؤں اور خالاؤں کے گھر گنوائے گئے۔ مفسرین کہتے ہیں کہ اپنے گھروں میں اولاد کے گھر بھی شامل ہیں، کیونکہ بیٹے کا گھر باپ ہی کا گھر ہے، اسی لیے ان کا الگ ذکر نہ ہوا۔ دائرہ اپنے آپ سے شروع ہوا اور رشتوں میں پھیلتا گیا، جیسے ایک ہی پانی کی لہریں کنارے کی طرف بڑھتی ہیں۔

پھر دائرہ خون کے رشتوں سے بھی آگے بڑھا: اَوْ مَا مَلَكْتُمْ مَّفَاتِحَهٗۤ اَوْ صَدِيْقِكُمْ۔ یا وہ گھر جن کی کنجیاں تمہارے اختیار میں ہیں، یا اپنے دوست کے گھر سے۔ کنجیوں والے سے مراد وہ نگران اور امین ہے جسے کسی کا گھر یا باغ سونپا گیا ہو، کہ وہ دستور کے مطابق اس میں سے کھا لے۔ اور دوست کو رشتہ داروں کی صف میں کھڑا کر دیا گیا؛ اہلِ ذوق کہتے ہیں کہ سچی دوستی خون کا رشتہ نہ ہوتے ہوئے بھی رشتہ بن جاتی ہے، اور غور ہو کہ سب رشتے جمع میں آئے مگر دوست واحد میں، گویا سچا دوست کم ہی ملتا ہے۔ صدیق کا مادہ صدق ہے، اور دوستی کی بنیاد سچائی ہے۔ پھر فرمایا: لَيْسَ عَلَيْكُمْ جُنَاحٌ اَنْ تَاْكُلُوْا جَمِيْعًا اَوْ اَشْتَاتًا۔ تم پر کوئی گناہ نہیں کہ مل کر کھاؤ یا الگ الگ۔ مفسرین نقل کرتے ہیں کہ عرب کے بعض لوگ اکیلے کھانے کو عیب سمجھتے تھے اور مہمان کے انتظار میں بھوکے رہتے تھے، اور بعض مل کر کھانے سے جھجکتے تھے؛ دونوں تنگیاں ہٹا دی گئیں۔ اہلِ معرفت یہاں جمع و فرق کا اشارہ دیکھتے ہیں: عارف کبھی جمع میں ہوتا ہے کہ سب کو ایک دیکھے، اور کبھی فرق میں کہ ہر ایک کو اس کا الگ حق دے۔ شیخِ اکبر کے مکتب میں کمال یہ ہے کہ نہ جمع فرق کو مٹائے نہ فرق جمع کو چھپائے؛ مل کر کھاؤ یا الگ، اگر رزق دینے والا ایک ہی دکھائی دے تو دونوں حال درست ہیں، کیونکہ ہر میزبان اسی رزاق کے نام کا ایک مظہر ہے۔

اور اب رکوع کا تاج، جو اس سورت کے ایک پچھلے سبق کو مکمل کرتا ہے: فَاِذَا دَخَلْتُمْ بُيُوْتًا فَسَلِّمُوْا عَلٰۤى اَنْفُسِكُمْ تَحِيَّةً مِّنْ عِنْدِ اللّٰهِ مُبٰرَكَةً طَيِّبَةً۔ پھر جب تم گھروں میں داخل ہو تو اپنے آپ پر سلام کیا کرو، اللہ کی طرف سے مقرر کی ہوئی دعا، بڑی برکت والی، پاکیزہ۔ ظاہر میں مراد یہ ہے کہ گھر والوں کو سلام کرو، کیونکہ وہ تمہاری اپنی جانوں کی طرح ہیں؛ اور اگر گھر خالی ہو تو بعض صحابہ سے منقول ہے کہ یوں کہے: ہم پر اور اللہ کے نیک بندوں پر سلام ہو۔ دوسرے رکوع میں تہمت کے موقع پر یہی تعبیر آئی تھی کہ مومن مرد اور مومن عورتیں اپنے آپ کے بارے میں نیک گمان کیوں نہ رکھیں، یعنی ایک دوسرے کے بارے میں؛ وہاں گمان میں امت ایک جان تھی، یہاں سلام میں ایک جان ہے۔ شیخِ اکبر کے مکتب میں السلام اسماءِ حسنیٰ میں سے ایک نام ہے، اور یہ دعا اسی کی طرف سے ہے؛ سلام کرنے والا اپنے بھائی کو اسی نام کا تحفہ دیتا ہے۔ اور جب حقیقت ایک ہو تو دوسرے پر سلام کرنا درحقیقت اپنے آپ پر سلام کرنا ہے؛ یہی وحدت کا ذوق ہے جو کثرت کو مٹائے بغیر اس کے اندر جھلکتا ہے۔ مبارکہ یعنی بڑھنے اور باقی رہنے والی، اور طیبہ یعنی پاکیزہ، جو تیسرے رکوع کے اس اصول سے ملتی ہے کہ پاکیزہ پاکیزہ کے لیے ہیں۔

رکوع کا اختتام ایک نئے لفظ پر ہوا: كَذٰلِكَ يُبَيِّنُ اللّٰهُ لَكُمُ الْاٰيٰتِ لَعَلَّكُمْ تَعْقِلُوْنَ۔ اسی طرح اللہ تمہارے لیے آیتیں کھول کر بیان کرتا ہے تاکہ تم سمجھو۔ پہلی دو آیتیں علم و حکمت کے ناموں پر ختم ہوئی تھیں، جو بیان کرنے والے کی صفات ہیں؛ یہ آیت سننے والے کے فرض پر ختم ہوئی۔ بیان اس کی طرف سے مکمل ہو چکا، اب سمجھنا تمہارا کام ہے۔ عقل کا مادہ عِقال ہے، یعنی وہ رسی جس سے اونٹ باندھا جاتا ہے؛ عقل وہ ہے جو سنی ہوئی بات کو باندھ کر رکھے کہ بھاگ نہ جائے۔ شیخِ اکبر کے نزدیک عقل کی فطرت باندھنا اور مقید کرنا ہے، اور دل کی فطرت پلٹنا، کہ ہر تجلی کے ساتھ اپنی صورت بدل لے۔ ادب کا سفر یہی ہے: پہلے عقل حکم کو باندھتی ہے، پھر دل اسے ذوق بنا لیتا ہے، یہاں تک کہ اجازت لینا، سلام کرنا اور دوسرے کا لحاظ رکھنا قاعدہ نہیں رہتا، فطرت بن جاتا ہے۔ یوں یہ رکوع گھر کے اندر کی چھوٹی چھوٹی باتوں سے شروع ہو کر ایک بڑی حقیقت تک پہنچا: جس گھر میں وقت کا ادب، بڑوں کا احترام، کمزور کی عزت اور اپنوں پر سلام ہو، وہ انہی گھروں میں سے ہے جن کے بارے میں اسی سورت نے بتایا تھا کہ ان میں اللہ کا نام بلند کیا جاتا ہے، اور جن میں اس کا نور اترتا ہے۔
