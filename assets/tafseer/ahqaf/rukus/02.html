منکرین کا ایک پرانا پیمانہ سامنے آتا ہے: وَقَالَ الَّذِيْنَ كَفَرُوْا لِلَّذِيْنَ اٰمَنُوْا لَوْ كَانَ خَيْرًا مَّا سَبَقُوْنَاۤ اِلَيْهِ ۚ وَاِذْ لَمْ يَهْتَدُوْا بِهٖ فَسَيَقُوْلُوْنَ هٰذَاۤ اِفْكٌ قَدِيْمٌ۔ “اور کافروں نے ایمان والوں کے بارے میں کہا: اگر یہ کوئی اچھی چیز ہوتی تو یہ لوگ اس کی طرف ہم سے آگے نہ نکل جاتے۔ اور چونکہ انہوں نے اس سے ہدایت نہ پائی، اس لیے اب کہیں گے کہ یہ تو پرانا جھوٹ ہے۔” انہوں نے اپنے آپ کو خیر کا پیمانہ بنا لیا: جو بھلائی ہوتی وہ پہلے ہمیں ملتی۔ اہلِ معرفت کے نزدیک یہی انانیت کا حجاب ہے، جو بندے کو اس کے اپنے قد سے آگے دیکھنے نہیں دیتا۔ اور آیت کا دوسرا جملہ اس سورت کا اصول ہے: پہلے یہ کہ انہوں نے ہدایت نہ لی، پھر یہ کہ اسے جھوٹ کا نام دیا۔ ہدایت سے محرومی ان کے اپنے رخ کا پھل تھی، اور نام رکھنا اس محرومی کا بہانہ۔ مشہور قول ہے کہ آدمی جس چیز سے ناواقف ہو اس کا دشمن بن جاتا ہے؛ اہلِ ذوق کہتے ہیں کہ نفس جس چیز کا ذائقہ نہ چکھے اسے پرانا کہہ کر ٹال دیتا ہے۔ اور ایک لطیف بات یہ ہے کہ ان کی زبان سے بے خبری میں سچ نکلا: یہ پیغام واقعی پرانا ہے، حضرت آدم علیہ السلام سے چلی آتی توحید، مگر جھوٹ نہیں۔ اسی لیے اگلی آیت فرماتی ہے: وَمِنْ قَبْلِهٖ كِتٰبُ مُوْسٰۤى اِمَامًا وَّرَحْمَةً ۚ وَهٰذَا كِتٰبٌ مُّصَدِّقٌ لِّسَانًا عَرَبِيًّا لِّيُنْذِرَ الَّذِيْنَ ظَلَمُوْا ۖ وَبُشْرٰى لِلْمُحْسِنِيْنَ۔ “اور اس سے پہلے موسیٰ کی کتاب رہنما اور رحمت تھی؛ اور یہ کتاب اس کی تصدیق کرنے والی ہے، عربی زبان میں، تاکہ ظالموں کو ڈرائے، اور نیکوکاروں کے لیے خوشخبری ہو۔” حضرت موسیٰ علیہ السلام کی کتاب امام تھی، یعنی آگے چلنے والی جس کے پیچھے چلا جائے۔ شیخِ اکبر کے مکتب میں حقیقت ایک ہے اور زبانیں مختلف؛ ہر کتاب ایک اسم کا مظہر ہے، اور قرآن پچھلی سچائی کو مٹاتا نہیں، اس پر مہر لگاتا ہے۔ ظالم کے لیے انذار جلال ہے اور محسن کے لیے بشارت جمال؛ اور محسن وہ ہے جو حدیثِ احسان کے مطابق یوں عبادت کرے گویا حق کو دیکھ رہا ہو۔

اب وہ آیت جو سالک کی پوری راہ کو دو لفظوں میں سمیٹ دیتی ہے: اِنَّ الَّذِيْنَ قَالُوْا رَبُّنَا اللّٰهُ ثُمَّ اسْتَقَامُوْا فَلَا خَوْفٌ عَلَيْهِمْ وَلَا هُمْ يَحْزَنُوْنَ۔ “بے شک جنہوں نے کہا کہ ہمارا رب اللہ ہے، پھر اسی پر جمے رہے، تو ان پر نہ کوئی خوف ہے اور نہ وہ غمگین ہوں گے۔” سورۂ حٰمٓ السجدہ میں یہی اعلان اس اضافے کے ساتھ آیا تھا: اِنَّ الَّذِيْنَ قَالُوْا رَبُّنَا اللّٰهُ ثُمَّ اسْتَقَامُوْا تَتَنَزَّلُ عَلَيْهِمُ الْمَلٰۤئِكَةُ۔ وہاں فرشتوں کا اترنا تھا، یہاں خوف و غم کا اٹھ جانا؛ گویا ایک ہی حقیقت کے دو رخ۔ لفظِ ثُمَّ پر غور کیجیے: کہنا ایک لمحے کا کام ہے، اس پر جمنا عمر بھر کا۔ صحیح مسلم کی روایت کا مفہوم ہے کہ ایک صحابی نے ایسی بات پوچھی جس کے بعد کسی سے پوچھنے کی حاجت نہ رہے، تو حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: کہو میں اللہ پر ایمان لایا، پھر اس پر قائم رہو۔ اور اہلِ تصوف میں مشہور قول ہے کہ استقامت ہزار کرامتوں سے بہتر ہے۔ شیخِ اکبر کے نزدیک استقامت جمود کا نام نہیں؛ قلب تو تقلب سے بنا ہے اور ہر آن نئی تجلی پر پلٹتا ہے۔ استقامت یہ ہے کہ تجلیات بدلتی رہیں اور رخ ایک ہی رہے: ہر صورت میں ربنا اللہ کی گواہی۔ یہی تلوین کے بعد تمکین ہے۔ اور خوف آنے والے کل کا ہوتا ہے، غم گزرے ہوئے کل کا؛ جو اپنے رب کے ساتھ اس آن میں قائم ہو گیا، وہ ابن الوقت ہے، نہ آگے کا ڈر اسے کھینچتا ہے نہ پیچھے کا غم۔ پھر ان کا انجام: اُولٰۤئِكَ اَصْحٰبُ الْجَنَّةِ خٰلِدِيْنَ فِيْهَا ۚ جَزَآءًۢ بِمَا كَانُوْا يَعْمَلُوْنَ۔ “یہی جنت والے ہیں، اس میں ہمیشہ رہیں گے، ان کاموں کے بدلے جو وہ کرتے تھے۔” اہلِ ذوق کہتے ہیں کہ جنت کا ایک ذائقہ یہیں ملتا ہے، دل کے اس ٹھہراؤ میں؛ اور بدلہ عمل کی اپنی صورت ہے جو وہاں کھل کر سامنے آتی ہے۔

اب وہ آیت آتی ہے جس میں ماں کی مشقت کا ایسا ذکر ہے کہ دل پگھل جائے: وَوَصَّيْنَا الْاِنْسَانَ بِوَالِدَيْهِ اِحْسٰنًا ۚ حَمَلَتْهُ اُمُّهٗ كُرْهًا وَّوَضَعَتْهُ كُرْهًا ۚ وَحَمْلُهٗ وَفِصٰلُهٗ ثَلٰثُوْنَ شَهْرًا ۚ۔ “اور ہم نے انسان کو اپنے ماں باپ کے ساتھ بھلائی کرنے کی تاکید کی۔ اس کی ماں نے اسے تکلیف سے پیٹ میں رکھا اور تکلیف ہی سے جنا، اور اس کا پیٹ میں رہنا اور دودھ چھوٹنا تیس مہینوں میں ہے۔” یہ تاکید مومن کے لیے خاص نہیں، انسان کے لیے ہے؛ والدین کا حق ہر حال میں قائم ہے۔ اور مطلوب صرف بھلائی نہیں، احسان ہے، وہی لفظ جو حدیثِ احسان میں عبادت کا اعلیٰ درجہ ہے؛ گویا ماں باپ کی خدمت یوں کرو جیسے حق تمہیں دیکھ رہا ہو۔ ذکر دونوں کا ہوا، مگر تفصیل ماں کی آئی، اور مشقت کا لفظ دو بار: ایک بار اٹھانے میں، ایک بار جننے میں۔ حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم سے پوچھا گیا کہ حسنِ سلوک کا سب سے زیادہ حق دار کون ہے، تو تین بار فرمایا: تمہاری ماں، پھر فرمایا: تمہارا باپ۔ حضرت علی رضی اللہ عنہ سے منقول ہے کہ انہوں نے اس آیت کو دودھ پلانے کی دو کامل برسوں والی آیت کے ساتھ ملا کر یہ نکالا کہ حمل کی کم سے کم مدت چھ مہینے ہے۔ شیخِ اکبر کے مکتب میں ماں کا رحم رحمت کا خاص مظہر ہے؛ حدیثِ قدسی کا مفہوم ہے کہ اللہ نے فرمایا: میں رحمٰن ہوں، میں نے رحم کو پیدا کیا اور اس کا نام اپنے نام سے نکالا۔ بچے کی زندگی میں رحمٰن کی رحمت کا پہلا اور سب سے قریبی مظہر ماں ہے۔ اہلِ ذوق اس میں ایک تمثیل دیکھتے ہیں: جس طرح نفَسِ رحمانی صورتوں کو اپنے اندر اٹھائے رکھتا ہے یہاں تک کہ وہ ظاہر ہوں، اسی طرح ماں اپنی جان میں ایک جان کو اٹھائے رکھتی ہے۔ یہ تمثیل عورت کے مقام کی بلندی کا بیان ہے، کسی کمی یا برتری کا دعویٰ نہیں۔

پھر عمر کی دو منزلیں بیان ہوئیں: حَتّٰۤى اِذَا بَلَغَ اَشُدَّهٗ وَبَلَغَ اَرْبَعِيْنَ سَنَةً ۙ۔ “یہاں تک کہ جب وہ اپنی پوری قوت کو پہنچا اور چالیس برس کا ہو گیا۔” اَشُدّ جسم اور قوت کی پختگی ہے، اور چالیس برس عقل اور نفس کی۔ مفسرین لکھتے ہیں کہ اکثر انبیاء علیہم السلام کو چالیس برس کی عمر میں نبوت ملی، اور حضورِ اکرم صلی اللہ علیہ وسلم پر بھی پہلی وحی اسی عمر میں اتری۔ حضرت موسیٰ علیہ السلام کی میقات بھی چالیس راتوں پر پوری ہوئی، اور اہلِ طریقت نے اسی سے چلّہ اور اربعین کی خلوت اخذ کی۔ اہلِ تصوف میں مشہور اثر ہے کہ حضرت آدم علیہ السلام کی مٹی چالیس صبحیں گوندھی گئی۔ شیخِ اکبر کے مکتب میں عدد بھی ایک زبان ہے، اور چالیس کمال کی تیاری کا عدد ہے: اس مدت میں استعداد پک کر تیار ہوتی ہے۔ اہلِ معرفت کہتے ہیں کہ چالیس سے پہلے آدمی زیادہ تر لیتا ہے: ماں کا دودھ، باپ کی کمائی، استاد کا علم، زمانے کا تجربہ۔ چالیس پر پہنچ کر وہ پلٹ کر دیکھتا ہے کہ اب تک کتنا کچھ ملا، اور یہیں سے لینے والا شکر کرنے والا بنتا ہے۔ اسی لیے آیت نے اس عمر کا ذکر کرتے ہی دعا رکھ دی؛ نفس کی پختگی کی پہلی علامت شکر ہے، دعویٰ نہیں۔ جو چالیس کو پہنچ کر بھی اپنی ہی تعریف میں رہے، اس کی عمر بڑھی، نفس نہیں پکا۔

اور وہ دعا یہ ہے: قَالَ رَبِّ اَوْزِعْنِيْۤ اَنْ اَشْكُرَ نِعْمَتَكَ الَّتِيْۤ اَنْعَمْتَ عَلَيَّ وَعَلٰى وَالِدَيَّ وَاَنْ اَعْمَلَ صَالِحًا تَرْضٰىهُ۔ “اس نے کہا: اے میرے رب، مجھے توفیق دے کہ میں تیری اس نعمت کا شکر ادا کروں جو تو نے مجھ پر اور میرے ماں باپ پر کی، اور یہ کہ میں ایسا نیک عمل کروں جسے تو پسند کرے۔” یہی کلمات سورۂ النمل میں حضرت سلیمان علیہ السلام کی زبان پر آئے تھے، جب انہوں نے چیونٹی کی بات سنی۔ ایک نبی اور ایک عام انسان کی دعا ایک ہو گئی؛ شکر کی بارگاہ میں سب ایک صف میں ہیں۔ اَوْزِعْنی میں سمیٹنے اور قابو میں رکھنے کا معنی ہے: میری بکھری ہوئی قوتوں کو سمیٹ کر شکر میں لگا دے۔ یعنی شکر بھی توفیق مانگتا ہے؛ شکر خود ایک نعمت ہے۔ سورۂ سبا میں آلِ داؤد کو شکر کے عمل کا حکم ملا تھا، اور اہلِ تصوف میں مشہور اثر ہے کہ حضرت داؤد علیہ السلام نے عرض کیا: میں تیرا شکر کیسے ادا کروں جب کہ میرا شکر بھی تیری نعمت ہے؟ جواب ملا: اب تم نے شکر ادا کر دیا۔ اور دیکھیے، یہ بیٹا ماں باپ پر ہونے والی نعمت کا شکر بھی اپنے ذمے لیتا ہے؛ وہ خود کو ان کا تسلسل سمجھتا ہے، ان کی نعمت کو اپنی نعمت۔ شیخِ اکبر کے مکتب میں، جیسا کہ حکمتِ رحمانیہ کا مفہوم ہے، رحمت دو طرح کی ہے: رحمتِ امتنان جو بغیر استحقاق کے ملتی ہے، اور رحمتِ وجوب جو عمل کے بعد وعدے کے طور پر ملتی ہے۔ یہ دعا دونوں کو جمع کرتی ہے: توفیق احسان سے مانگی، اور وہ عمل مانگا جو رضا کا مستحق ہو۔ اور ہر نیک عمل پسندیدہ نہیں ہوتا؛ پسندیدہ وہ ہے جس میں اخلاص ہو اور جو اپنی ملکیت سمجھ کر نہ کیا جائے۔

دعا اگلی نسل تک پھیلتی ہے: وَاَصْلِحْ لِيْ فِيْ ذُرِّيَّتِيْ ۚ اِنِّيْ تُبْتُ اِلَيْكَ وَاِنِّيْ مِنَ الْمُسْلِمِيْنَ۔ “اور میرے لیے میری اولاد میں بھی نیکی رکھ دے؛ میں نے تیری طرف رجوع کیا اور میں فرماں برداروں میں سے ہوں۔” زنجیر مکمل ہو گئی: پیچھے ماں باپ، درمیان میں اپنی ذات، آگے اولاد؛ شکر تینوں کو ایک لڑی میں پرو دیتا ہے۔ اور لِیْ کا لفظ بتاتا ہے کہ اولاد کی اصلاح بھی اپنے ہی حق میں نعمت ہے۔ پھر توبہ اور اسلام: شیخِ اکبر کے نزدیک توبہ کی حقیقت رجوع ہے، اور ہر مقام سے اگلے مقام کی طرف بڑھنا بھی ایک رجوع ہے؛ اور اسلام سپردگی ہے، یعنی اپنی تدبیر کو اس کی تدبیر کے سپرد کر دینا۔ چالیس برس کا سالک گناہ سے بھی لوٹتا ہے اور اپنے آپ کو کچھ سمجھنے سے بھی۔ ایسوں کے بارے میں فرمایا: اُولٰۤئِكَ الَّذِيْنَ نَتَقَبَّلُ عَنْهُمْ اَحْسَنَ مَا عَمِلُوْا وَنَتَجَاوَزُ عَنْ سَيِّاٰتِهِمْ فِيْۤ اَصْحٰبِ الْجَنَّةِ ۚ وَعْدَ الصِّدْقِ الَّذِيْ كَانُوْا يُوْعَدُوْنَ۔ “یہی وہ لوگ ہیں جن کے بہترین اعمال ہم قبول کرتے ہیں اور ان کی برائیوں سے درگزر کرتے ہیں، جنت والوں میں؛ یہ سچا وعدہ ہے جو ان سے کیا جاتا تھا۔” اہلِ معرفت کہتے ہیں کہ یہ سبقتِ رحمت کی تصویر ہے: حق کی نظر بندے کے سب سے اچھے عمل پر ٹھہرتی ہے، اور برائیوں سے یوں گزر جاتی ہے جیسے راہ چلتا آدمی کسی چیز سے آگے نکل جائے۔ جس نے ماں باپ کا شکر کیا، اسے رب نے اسی شکر کی صورت میں بدلہ دیا۔

اب اس کے برعکس ایک اور کردار: وَالَّذِيْ قَالَ لِوَالِدَيْهِ اُفٍّ لَّكُمَاۤ اَتَعِدٰنِنِيْۤ اَنْ اُخْرَجَ وَقَدْ خَلَتِ الْقُرُوْنُ مِنْ قَبْلِيْ ۚ وَهُمَا يَسْتَغِيْثٰنِ اللّٰهَ وَيْلَكَ اٰمِنْ ۖ اِنَّ وَعْدَ اللّٰهِ حَقٌّ ۚ فَيَقُوْلُ مَا هٰذَاۤ اِلَّاۤ اَسَاطِيْرُ الْاَوَّلِيْنَ۔ “اور جس نے اپنے ماں باپ سے کہا: تم پر اُف! کیا تم مجھے یہ وعدہ دیتے ہو کہ میں قبر سے نکالا جاؤں گا، حالانکہ مجھ سے پہلے کتنی نسلیں گزر چکیں؛ اور وہ دونوں اللہ سے فریاد کرتے ہیں: ہائے تیری خرابی، ایمان لے آ، اللہ کا وعدہ سچا ہے؛ تو وہ کہتا ہے: یہ تو بس پہلوں کی کہانیاں ہیں۔” اُولٰۤئِكَ الَّذِيْنَ حَقَّ عَلَيْهِمُ الْقَوْلُ فِيْۤ اُمَمٍ قَدْ خَلَتْ مِنْ قَبْلِهِمْ مِّنَ الْجِنِّ وَالْاِنْسِ ۚ اِنَّهُمْ كَانُوْا خٰسِرِيْنَ۔ “یہی وہ لوگ ہیں جن پر بات پوری ہو گئی، ان امتوں کے ساتھ جو ان سے پہلے جنوں اور انسانوں میں گزر چکیں؛ بے شک وہ گھاٹے میں رہے۔” صحیح یہ ہے کہ یہ کسی معین صحابی کا ذکر نہیں، ایک کردار کا نقشہ ہے؛ ام المومنین حضرت عائشہ رضی اللہ عنہا نے اسے اپنے گھر والوں پر چسپاں کرنے کی تردید فرمائی۔ اب اصول کو دیکھیے۔ ہدایت اس کے گھر میں موجود تھی: ماں باپ کی پکار، اللہ سے ان کی فریاد، وعدے کی سچائی کا اعلان۔ اس نے خود انکار چنا اور سب کو کہانیاں کہہ کر رد کیا۔ تب کہا گیا کہ بات ان پر پوری ہو گئی؛ یعنی حکم اس کے چنے ہوئے رخ کے مطابق ثابت ہوا، اس پر مسلط نہیں کیا گیا۔ حق چاہتا تو سب کو زبردستی ہدایت پر جمع کر دیتا، مگر اس نے انسان کو اختیار کی عزت دی۔ اور ماں باپ کے لیے اس میں تسلی ہے: ان کی دعا ضائع نہیں، ان کا کام پکارنا تھا، اور اولاد کا قبول یا انکار اس کا اپنا انتخاب ہے۔

پھر عدل کا قانون: وَلِكُلٍّ دَرَجٰتٌ مِّمَّا عَمِلُوْا ۚ وَلِيُوَفِّيَهُمْ اَعْمَالَهُمْ وَهُمْ لَا يُظْلَمُوْنَ۔ “اور ہر ایک کے لیے ان کے اعمال کے مطابق درجے ہیں، اور تاکہ اللہ انہیں ان کے اعمال کا پورا بدلہ دے، اور ان پر ظلم نہ ہوگا۔” مفسرین کہتے ہیں کہ جنت والوں کے لیے درجے اوپر کو جاتے ہیں اور آگ والوں کے درکات نیچے کو، اور یہاں دونوں کو ایک لفظ میں جمع کیا گیا۔ شیخِ اکبر کے نزدیک ہر ایک کا مقام اس کے اپنے عمل اور اپنی استعداد سے پھوٹتا ہے؛ اسی لیے ظلم کی کوئی گنجائش نہیں، کیونکہ بدلہ وہی ہے جو بندہ خود تھا اور جو اس نے خود کیا۔ یہی سرِّ قدر ہے۔ اب منکرین کی پیشی کا منظر: وَيَوْمَ يُعْرَضُ الَّذِيْنَ كَفَرُوْا عَلَى النَّارِ ۚ اَذْهَبْتُمْ طَيِّبٰتِكُمْ فِيْ حَيَاتِكُمُ الدُّنْيَا وَاسْتَمْتَعْتُمْ بِهَا ۚ۔ “اور جس دن کافر آگ کے سامنے پیش کیے جائیں گے: تم اپنی اچھی چیزیں دنیا کی زندگی ہی میں لے چکے اور ان سے مزے اٹھا چکے۔” روایات میں آتا ہے کہ حضرت عمر رضی اللہ عنہ کے سامنے جب عمدہ کھانا رکھا جاتا تو وہ یہ آیت یاد کرتے اور فرماتے کہ میں ڈرتا ہوں کہ کہیں میری اچھی چیزیں اسی دنیا میں پوری نہ کر دی جائیں۔ ظاہر میں آیت کا خطاب منکرین سے ہے؛ مگر اہلِ دل نے اسے اپنے لیے آئینہ بنایا، اور یہیں سے صوفیہ کا زہد پھوٹا۔

آیت کا باقی حصہ سبب بیان کرتا ہے: فَالْيَوْمَ تُجْزَوْنَ عَذَابَ الْهُوْنِ بِمَا كُنْتُمْ تَسْتَكْبِرُوْنَ فِي الْاَرْضِ بِغَيْرِ الْحَقِّ وَبِمَا كُنْتُمْ تَفْسُقُوْنَ۔ “تو آج تمہیں ذلت کا عذاب دیا جائے گا، اس لیے کہ تم زمین میں ناحق تکبر کرتے تھے اور اس لیے کہ تم نافرمانی کرتے تھے۔” غور کیجیے، جرم یہ نہیں بتایا گیا کہ انہوں نے پاکیزہ چیزیں کھائیں؛ قرآن خود حلال زینت کو حرام ٹھہرانے والوں پر سوال اٹھاتا ہے۔ جرم یہ تھا کہ نعمت دینے والے سے کٹ کر نعمت میں ڈوب گئے، اکڑے اور حد سے نکلے۔ اسی لیے سزا ذلت کی صورت میں آئی: تکبر کا بدلہ ہون، یعنی رسوائی؛ عمل نے اپنی ہی صورت میں بدلہ پایا۔ صوفیہ کے نزدیک زہد حلال سے نفرت نہیں، یہ ہے کہ دنیا ہاتھ میں رہے اور دل میں نہ اترے، اور آدمی اپنا حصہ یہیں پورا وصول کرنے کی جلدی نہ کرے۔ شیخِ اکبر کے مکتب میں عارف کا زہد اس سے بھی باریک ہے: دنیا حق کا سایہ ہے، اپنی ذات میں کوئی مستقل شے نہیں، تو جس چیز کی اپنی کوئی ہستی نہ ہو اسے چھوڑنے پر فخر کیسا؟ اسی کو اہلِ ذوق زہد فی الزہد کہتے ہیں: اپنے زہد کو بھی کچھ نہ سمجھنا۔ ظاہری معنی اپنی جگہ مقدم ہے، اور یہ اشارہ اس کی نفی نہیں، اس کی گہرائی ہے: جو نعمت کو دینے والے کی یاد کے ساتھ لے، اس کی اچھی چیزیں ختم نہیں ہوتیں، آخرت کے لیے بچی رہتی ہیں۔
