سورت اپنے نام کی طرف لوٹ آئی ہے۔ زمر گروہوں کو کہتے ہیں، اور اس آخری رکوع میں انسانیت دو قافلوں میں بٹی دکھائی گئی ہے۔ پہلا قافلہ: وَسِيْقَ الَّذِيْنَ كَفَرُوْٓا اِلٰى جَهَنَّمَ زُمَرًا ۭ حَتّٰۤى اِذَا جَآءُوْهَا فُتِحَتْ اَبْوَابُهَا وَقَالَ لَهُمْ خَزَنَتُهَآ اَلَمْ يَاْتِكُمْ رُسُلٌ مِّنْكُمْ يَتْلُوْنَ عَلَيْكُمْ اٰيٰتِ رَبِّكُمْ وَيُنْذِرُوْنَكُمْ لِقَآءَ يَوْمِكُمْ هٰذَا ۭ قَالُوْا بَلٰى وَلٰكِنْ حَقَّتْ كَلِمَةُ الْعَذَابِ عَلَى الْكٰفِرِيْنَ۔ اور کفر کرنے والے گروہ در گروہ جہنم کی طرف ہانکے جائیں گے، یہاں تک کہ جب وہ اس کے پاس پہنچیں گے تو اس کے دروازے کھول دیے جائیں گے، اور اس کے نگہبان ان سے کہیں گے: کیا تمہارے پاس تم ہی میں سے رسول نہیں آئے تھے جو تمہیں تمہارے رب کی آیتیں سناتے اور تمہیں تمہارے اس دن کی ملاقات سے ڈراتے؟ وہ کہیں گے: کیوں نہیں! لیکن عذاب کا کلمہ کافروں پر ثابت ہو کر رہا۔ یہ آیت اس سورت میں ہدایت و گمراہی کے اصول کا مقام ہے، اور اسے اسی طرح پڑھنا چاہیے۔ نگہبانوں کا سوال معلومات لینے کے لیے نہیں، اقرار لینے کے لیے ہے؛ اور جواب "بَلٰى" ہے۔ دوزخ کے دروازے پر کوئی یہ نہیں کہے گا کہ ہمیں خبر نہ تھی، یا ہم ہدایت چاہتے تھے اور ہمیں روک دیا گیا۔ رسول آئے، آیتیں پڑھی گئیں، اس دن سے ڈرایا گیا؛ پیغام پہنچا اور انہوں نے خود منہ پھیرا۔ اور "مِّنْكُمْ" میں حجت کی تکمیل ہے: رسول انہی کی زبان اور انہی کی قوم سے تھے، کوئی اجنبی آواز نہ تھی جو سمجھ میں نہ آ سکتی۔

"حَقَّتْ كَلِمَةُ الْعَذَابِ" کو اسی اقرار کی روشنی میں پڑھنا لازم ہے۔ آیت نے حکم کو ان کے وصف سے جوڑا: "عَلَى الْكٰفِرِيْنَ"؛ کلمہ ان پر اس لیے ثابت ہوا کہ وہ اپنے انکار پر قائم رہے، اس لیے نہیں کہ انکار ان پر زبردستی ڈالا گیا تھا۔ یہی اس سلسلے کا طے شدہ اصول ہے: حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت طلب کرے اسے کبھی محروم نہیں رکھتا؛ جو منہ پھیرے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک سرِّ قدر کا خلاصہ یہی ہے کہ حکم محکوم کے مطابق ہوتا ہے: حق کا علم بندے کی حقیقت کے تابع ہے، اور فیصلہ اسی حقیقت کا ظہور ہے۔ چھٹے رکوع (آیات ۵۷ تا ۵۹) میں اسی سورت نے یہ عذر پہلے ہی رد کر دیا تھا کہ اگر اللہ مجھے ہدایت دیتا؛ جواب آیا تھا کہ میری آیتیں تیرے پاس آئیں اور تو نے انہیں جھٹلایا۔ وہاں عذر کا رد تھا، یہاں خود ان کی زبان سے اقرار۔ سورۂ ابراہیم میں شیطان کا اقرار بھی اسی اصول کی گواہی تھا: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّآ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ؛ دعوت تھی، زور نہ تھا، اور قبول کرنا ان کا اپنا فعل تھا۔ اور "لِقَآءَ يَوْمِكُمْ هٰذَا" میں اہلِ معرفت ایک درد دیکھتے ہیں: ملاقات کی خبر دی گئی تھی، مگر جس نے ملاقات کو بھلا دیا اس کے لیے وہی ملاقات ڈر بن گئی۔

پھر ان کا ٹھکانا بتایا جاتا ہے: قِيْلَ ادْخُلُوْٓا اَبْوَابَ جَهَنَّمَ خٰلِدِيْنَ فِيْهَا ۚ فَبِئْسَ مَثْوَى الْمُتَكَبِّرِيْنَ۔ کہا جائے گا: جہنم کے دروازوں میں داخل ہو جاؤ، اس میں ہمیشہ رہنے کے لیے؛ سو تکبر کرنے والوں کا ٹھکانا کیا ہی برا ہے۔ آیت نے آخر میں ان کا نام "الْمُتَكَبِّرِيْنَ" رکھا: کفر کی جڑ تکبر میں دکھائی، یعنی حق کے سامنے جھکنے سے انکار۔ پہلی آیت میں وصف کفر تھا، دوسری میں تکبر؛ گویا بتایا گیا کہ انکار کا اصل سبب دلیل کی کمی نہ تھی، دل کی اکڑ تھی جو رسولوں کے سامنے نہ جھکی۔ حدیثِ قدسی کا مفہوم ہے کہ کبریائی میری چادر ہے، جو اس میں مجھ سے کھینچا تانی کرے میں اسے آگ میں ڈالوں گا۔ اہلِ معرفت کہتے ہیں کہ متکبر نے وہ لباس پہننا چاہا جو اس کا نہ تھا، اور عبدیت کا لباس اتار دیا جو اس کا اپنا تھا؛ اس نے اپنی حقیقت سے منہ پھیرا اور اپنی حقیقت سے کٹ گیا۔ شیخِ اکبر کے مکتب میں بندے کی عزت اس کی عبدیت میں ہے، اور ربوبیت کا دعویٰ، خواہ زبان سے ہو یا حال سے، ذلت کی طرف لے جاتا ہے۔ "مَثْوَى" اس جگہ کو کہتے ہیں جہاں آدمی ٹھہر جائے اور جم کر رہ جائے: جو بلندی کا طالب تھا وہ وہاں ٹھہرا جو سب سے پست ہے، اور یہ اسی کے اپنے رخ کا انجام ہے، کسی ناانصافی کا نہیں۔

اب دوسرا قافلہ: وَسِيْقَ الَّذِيْنَ اتَّقَوْا رَبَّهُمْ اِلَى الْجَنَّةِ زُمَرًا ۭ حَتّٰۤى اِذَا جَآءُوْهَا وَفُتِحَتْ اَبْوَابُهَا وَقَالَ لَهُمْ خَزَنَتُهَا سَلٰمٌ عَلَيْكُمْ طِبْتُمْ فَادْخُلُوْهَا خٰلِدِيْنَ۔ اور جو لوگ اپنے رب سے ڈرتے رہے وہ گروہ در گروہ جنت کی طرف لے جائے جائیں گے، یہاں تک کہ جب وہ اس کے پاس پہنچیں گے اور اس کے دروازے کھلے ہوں گے، اور اس کے نگہبان ان سے کہیں گے: تم پر سلام ہو، تم پاکیزہ رہے، سو اس میں ہمیشہ کے لیے داخل ہو جاؤ۔ دونوں قافلوں کے لیے ایک ہی فعل "وَسِيْقَ" آیا، مگر مفسرین کہتے ہیں کہ ایک کا ہانکا جانا قیدیوں کی طرح ہے اور دوسرے کا لے جایا جانا معزز مہمانوں کی طرح، جن کی سواریاں آگے بڑھائی جاتی ہیں۔ اور یہاں وہ لطیف نکتہ ہے جس پر قدیم مفسرین ٹھہرے: پہلی آیت میں "فُتِحَتْ" بغیر واو کے تھا، یعنی دروازے پہنچنے پر کھولے گئے، جیسے قید خانے کا دروازہ قیدی کے آنے پر کھلتا ہے؛ یہاں "وَفُتِحَتْ" واو کے ساتھ ہے، یعنی جب وہ پہنچے تو دروازے پہلے سے کھلے تھے، جیسے مہمان کے انتظار میں گھر کا دروازہ کھلا رکھا جاتا ہے۔ سورۂ صٓ میں بھی یہی فرمایا تھا: جَنّٰتِ عَدْنٍ مُّفَتَّحَةً لَّهُمُ الْاَبْوَابُ۔ اہلِ زبان یہ بھی بتاتے ہیں کہ "حَتّٰۤى اِذَا" کا جواب یہاں ذکر نہیں ہوا، اور یہ حذف خود ایک بیان ہے: جو کچھ انہیں ملا وہ لفظوں میں نہیں سماتا، اس لیے سننے والے کا تصور جہاں تک جائے، حقیقت اس سے آگے ہے۔ اہلِ معرفت کہتے ہیں کہ رحمت بندے کے آنے سے پہلے اس کے انتظار میں تھی؛ یہی سبقتِ رحمت ہے۔

دروازے پر پہلا کلمہ سلام ہے: "سَلٰمٌ عَلَيْكُمْ"۔ اہلِ معرفت کہتے ہیں کہ السلام اللہ کا نام ہے، اور جنت کی فضا اسی نام کی تجلی ہے؛ وہاں نہ خوف ہے نہ خلش، اور سلامتی کا آغاز دروازے ہی سے ہو جاتا ہے۔ پھر "طِبْتُمْ": تم پاکیزہ ہو گئے، تم خوب رہے۔ یہ صیغہ ماضی کا ہے، اور اہلِ ذوق اس میں یہ اشارہ دیکھتے ہیں کہ پاکیزگی کا کام دنیا میں ہو چکا تھا؛ دروازہ اس کا اعلان کرتا ہے، اسے پیدا نہیں کرتا۔ جنت پاک کا گھر ہے اور پاک ہی اس میں داخل ہوتا ہے۔ سورۂ النور میں فرمایا تھا: وَالطَّيِّبٰتُ لِلطَّيِّبِيْنَ؛ جو پاکیزہ ہوا وہ اپنے ہم جنس گھر میں پہنچا۔ اور "الَّذِيْنَ اتَّقَوْا رَبَّهُمْ" میں تقویٰ کی نسبت "رَبَّهُمْ" کی طرف ہے، یعنی اس رب کی طرف جو ان کی خاص پرورش کرتا رہا۔ شیخِ اکبر کے مکتب میں ہر بندے کا ایک ربِّ خاص ہے، اور اسی مناسبت سے "زُمَرًا" کو سمجھا جاتا ہے: جنت کے قافلے بھی گروہ گروہ ہیں، ہر گروہ اپنے مقام، اپنی معرفت اور اپنے امام کے ساتھ۔ شیخِ اکبر فتوحاتِ مکیہ میں اہلِ جنت کی رؤیت کا ذکر کرتے ہیں، جس کا مفہوم یہ ہے کہ وہاں دیدار میں ہر ایک کا حصہ اس معرفت کے مطابق ہو گا جو اس نے دنیا میں حاصل کی۔ یوں تقویٰ صرف ڈر نہیں، پہچان کا سفر بھی ہے۔

اور اندر پہنچ کر جنتیوں کا پہلا کلمہ حمد ہے: وَقَالُوا الْحَمْدُ لِلّٰهِ الَّذِيْ صَدَقَنَا وَعْدَهٗ وَاَوْرَثَنَا الْاَرْضَ نَتَبَوَّاُ مِنَ الْجَنَّةِ حَيْثُ نَشَآءُ ۚ فَنِعْمَ اَجْرُ الْعٰمِلِيْنَ۔ اور وہ کہیں گے: سب تعریف اللہ کے لیے ہے جس نے ہم سے اپنا وعدہ سچ کر دکھایا اور ہمیں اس زمین کا وارث بنایا، ہم جنت میں جہاں چاہیں ٹھکانا بنائیں؛ سو عمل کرنے والوں کا کیا ہی اچھا اجر ہے۔ "صَدَقَنَا وَعْدَهٗ": وعدہ دنیا میں سنا تھا، یہ علم الیقین تھا؛ آج آنکھوں سے دیکھ لیا، یہ عین الیقین ہے؛ اور اس میں رہنا بسنا حق الیقین ہے۔ "اَوْرَثَنَا" میں ایک لطیف بات ہے: میراث خریدی نہیں جاتی، عطا ہوتی ہے۔ حدیث کا مفہوم ہے کہ کوئی شخص محض اپنے عمل سے جنت میں داخل نہیں ہو گا، مگر یہ کہ اللہ اسے اپنی رحمت سے ڈھانپ لے؛ اسی لیے داخلہ وراثت کے لفظ سے بیان ہوا۔ اور پھر بھی آیت "فَنِعْمَ اَجْرُ الْعٰمِلِيْنَ" پر ختم ہوتی ہے: فضل اور عمل دونوں کو جمع کر دیا گیا۔ اہلِ معرفت کہتے ہیں کہ عمل کی توفیق بھی اسی کا فضل تھی، پھر اسی عمل کو اجر کہہ کر بندے کو عزت دی؛ یہ کریم کی شان ہے کہ اپنا دیا ہوا لوٹا کر اسے بندے کی کمائی کہتا ہے۔ اور "حَيْثُ نَشَآءُ" میں وہ آزادی ہے جو دنیا میں نہ تھی؛ اہلِ تصوف میں ایک مشہور اثر کا ذکر ملتا ہے، جسے شیخِ اکبر کے مکتب میں بھی بیان کیا جاتا ہے، جس کا مفہوم یہ ہے کہ اہلِ جنت کو ایسی عطا ہو گی کہ وہ جس چیز کا ارادہ کریں وہ ہو جائے؛ یہ کلمۂ کن کا ایک پرتو ہے جو عبدیت کے صلے میں بندے کو دیا جاتا ہے۔

اب منظر عرش کی طرف اٹھتا ہے: وَتَرَى الْمَلٰۤئِكَةَ حَآفِّيْنَ مِنْ حَوْلِ الْعَرْشِ يُسَبِّحُوْنَ بِحَمْدِ رَبِّهِمْ ۚ وَقُضِيَ بَيْنَهُمْ بِالْحَقِّ وَقِيْلَ الْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِيْنَ۔ اور تم فرشتوں کو دیکھو گے کہ عرش کے گرد حلقہ باندھے اپنے رب کی حمد کے ساتھ تسبیح کر رہے ہیں، اور لوگوں کے درمیان حق کے ساتھ فیصلہ کر دیا گیا، اور کہا گیا: سب تعریف اللہ کے لیے ہے جو سارے جہانوں کا رب ہے۔ "حَآفِّيْنَ" حلقہ بنانے کو کہتے ہیں، جیسے کوئی مرکز کے گرد دائرہ باندھ لے۔ شیخِ اکبر کے مکتب میں عرش رحمانیت کے استوا کا مقام ہے، اور اسی سے رحمت تمام موجودات پر پھیلتی ہے؛ فرشتوں کا اس کے گرد حلقہ گویا رحمت کے مرکز کے گرد ساری کائنات کا طواف ہے۔ پھر "يُسَبِّحُوْنَ بِحَمْدِ رَبِّهِمْ": تسبیح تنزیہ ہے اور حمد ثنا؛ پچھلے رکوع میں آیت ۶۷ کی مٹھی اور داہنے ہاتھ کے بیان پر یہ سورت تنزیہ اور تشبیہ کے جمع تک پہنچی تھی، اور اب فرشتوں کی زبان سے وہی جمع ادا ہو رہا ہے۔ وہ اسے ہر کمی سے پاک بھی کہتے ہیں اور ہر کمال کے ساتھ اس کی ثنا بھی کرتے ہیں۔ اور "وَقُضِيَ بَيْنَهُمْ بِالْحَقِّ" کا لفظ دوسری بار آیا ہے: پہلی بار پچھلے رکوع میں فیصلے کی ابتدا تھی، یہاں اس کی تکمیل؛ جب ہر ایک اپنی منزل کو پہنچ چکا، تب کائنات کا آخری کلمہ کہا گیا۔

"وَقِيْلَ" مجہول ہے، کہنے والے کا نام نہیں لیا گیا۔ مفسرین کہتے ہیں کہ کہنے والے فرشتے ہیں، یا اہلِ جنت، یا ساری مخلوق۔ اہلِ معرفت کہتے ہیں کہ نام اس لیے نہیں لیا گیا کہ ہر زبان اس کی کہنے والی ہے: ابتدا میں ہر شے کلمۂ کن سے وجود میں آئی تھی، اور انتہا میں ہر شے "الْحَمْدُ لِلّٰهِ" پر ٹھہرتی ہے۔ قرآن کا آغاز بھی انہی لفظوں سے ہوا تھا: اَلْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِيْنَ؛ یوں اس سورت کا انجام کلامِ الٰہی کے آغاز سے جا ملتا ہے، جیسے دائرے کا آخری نقطہ اس کے پہلے نقطے سے مل جائے۔ شیخِ اکبر کے مکتب میں ہر حمد آخرکار اسی کی طرف لوٹتی ہے، کیونکہ ہر کمال اسی کا ہے جو کسی مظہر میں ظاہر ہوا؛ اس لیے حمد کرنے والی ہر زبان دراصل اسی کے کمال کی خبر دیتی ہے۔ یہ اخلاص کی سورت تھی: اَلَا لِلّٰهِ الدِّيْنُ الْخَالِصُ کے اعلان سے شروع ہوئی اور خالص حمد پر ختم ہوئی؛ خالص دین کا پھل خالص حمد ہے۔ اور اس حمد میں عدل کا شکر بھی شامل ہے: دوزخ کے دروازے پر بھی انکار کرنے والوں نے "بَلٰى" کہا تھا، یعنی فیصلہ حق پر تھا اور اس میں کسی کے لیے شکایت کی گنجائش نہ تھی۔ اہلِ ذوق کہتے ہیں کہ سالک اس آیت کو اپنے حال کی دعا بنا لے: یا اللہ، ہمیں اس زمرے میں رکھ جس کے لیے دروازے پہلے سے کھلے ہیں، اور ہمارا آخری کلمہ بھی تیری حمد ہو۔
