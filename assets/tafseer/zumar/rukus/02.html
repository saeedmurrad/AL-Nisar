رکوع ایک محبت بھری پکار سے کھلتا ہے: قُلْ يٰعِبَادِ الَّذِيْنَ اٰمَنُوا اتَّقُوْا رَبَّكُمْ ۭ لِلَّذِيْنَ اَحْسَنُوْا فِيْ هٰذِهِ الدُّنْيَا حَسَنَةٌ۔ “کہہ دیجیے: اے میرے بندو جو ایمان لائے ہو، اپنے رب سے ڈرو؛ جنہوں نے اس دنیا میں بھلائی کی ان کے لیے بھلائی ہے۔” پیغام حضورِ اکرم صلی اللہ علیہ وسلم کی زبان سے ہے مگر پکار “میرے بندو” کی ہے؛ اہلِ ذوق کہتے ہیں کہ رسول کی زبان پر حق خود بول رہا ہے، اور نسبت اپنی طرف رکھ کر بندوں کو قریب کر لیا۔ احسان کی شرح حدیثِ احسان میں ہے: اللہ کی عبادت یوں کرو گویا تم اسے دیکھ رہے ہو، اور اگر یہ نہ ہو تو یہ جانو کہ وہ تمہیں دیکھ رہا ہے۔ جس نے دنیا ہی میں یہ حضوری پا لی، اس کا بدلہ بھی یہیں سے شروع ہو جاتا ہے۔ پھر فرمایا: وَاَرْضُ اللّٰهِ وَاسِعَةٌ ۭ اِنَّمَا يُوَفَّى الصّٰبِرُوْنَ اَجْرَهُمْ بِغَيْرِ حِسَابٍ۔ “اور اللہ کی زمین کشادہ ہے؛ صبر کرنے والوں کو ان کا اجر بے حساب پورا دیا جائے گا۔” ظاہر میں یہ ہجرت کی اجازت ہے: جہاں بندگی ممکن نہ رہے وہاں سے نکل جاؤ؛ سورۂ عنکبوت میں یہی بات یوں تھی: يٰعِبَادِيَ الَّذِيْنَ اٰمَنُوْۤا اِنَّ اَرْضِيْ وَاسِعَةٌ فَاِيَّايَ فَاعْبُدُوْنِ۔ اہلِ معرفت اس کے باطن میں قلب کی وسعت دیکھتے ہیں: نفس کی تنگ بستی میں بندگی گھٹتی ہو تو دل کی کشادہ زمین کی طرف ہجرت کرو، جسے اس مکتب میں حق کی سب سے وسیع جلوہ گاہ کہا گیا ہے۔ اور صبر کا اجر بے حساب اس لیے کہ حساب ناپی ہوئی چیزوں کا ہوتا ہے؛ صبر بندے کا اپنے آپ کو روک کر حق کے حوالے کر دینا ہے، اور جو اپنے آپ کو دے دے اسے ناپ کر نہیں دیا جاتا۔

اب حضورِ اکرم صلی اللہ علیہ وسلم کو اپنا حال سنانے کا حکم ہوا: قُلْ اِنِّيْۤ اُمِرْتُ اَنْ اَعْبُدَ اللّٰهَ مُخْلِصًا لَّهُ الدِّيْنَ ۝ وَاُمِرْتُ لِاَنْ اَكُوْنَ اَوَّلَ الْمُسْلِمِيْنَ۔ “کہہ دیجیے: مجھے حکم دیا گیا ہے کہ اللہ کی عبادت کروں، دین کو اسی کے لیے خالص کرتے ہوئے؛ اور مجھے حکم دیا گیا ہے کہ میں سب سے پہلا فرمانبردار بنوں۔” دعوت دینے والا پہلے خود حکم کے نیچے کھڑا ہے؛ امام وہی ہے جو سب سے پہلے جھکے۔ پہلے رکوع کی دوسری آیت میں یہی بات حکم کی صورت میں آئی تھی، اب وہی حکم اقرار بن کر لوٹا۔ اوّل المسلمین کا ظاہری معنی یہ ہے کہ اس امت میں سب سے پہلے آپ صلی اللہ علیہ وسلم نے سرِ تسلیم خم کیا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس کی ایک اور پرت ہے: حقیقتِ محمدیہ تعینِ اوّل ہے، یعنی وجود کا پہلا ظہور، اور مشہور حدیث کا مفہوم ہے کہ آپ صلی اللہ علیہ وسلم اس وقت بھی نبی تھے جب حضرت آدم علیہ السلام ابھی روح اور جسم کے درمیان تھے؛ سو اسلام، یعنی سپردگی، کا اوّلین مظہر وہی ہیں۔ پھر فرمایا: قُلْ اِنِّيْۤ اَخَافُ اِنْ عَصَيْتُ رَبِّيْ عَذَابَ يَوْمٍ عَظِيْمٍ۔ “کہہ دیجیے: اگر میں اپنے رب کی نافرمانی کروں تو مجھے ایک بڑے دن کے عذاب کا ڈر ہے۔” ادب یہاں لازم ہے: حضورِ اکرم صلی اللہ علیہ وسلم معصوم ہیں اور نافرمانی آپ سے واقع ہی نہیں ہو سکتی؛ یہ جملہ شرط کی صورت میں امت کی تعلیم ہے کہ جب سب سے محبوب بندہ یوں کہے تو کوئی دوسرا اپنے آپ کو بے خوف نہ سمجھے۔ اہلِ معرفت کے نزدیک عارف کا خوف شک کا خوف نہیں، ہیبت ہے: جتنا قرب بڑھتا ہے، جلال کا احساس بھی اتنا ہی بڑھتا ہے۔

اقرار اب اور گہرا ہو جاتا ہے: قُلِ اللّٰهَ اَعْبُدُ مُخْلِصًا لَّهٗ دِيْنِيْ۔ “کہہ دیجیے: میں تو اللہ ہی کی عبادت کرتا ہوں، اپنے دین کو اسی کے لیے خالص کرتے ہوئے۔” دو تبدیلیاں قابلِ غور ہیں۔ مفعول پہلے آ گیا، اللّٰهَ اَعْبُدُ، جو عربی میں حصر کا انداز ہے: صرف اسی کی۔ اور گیارہویں آیت کا الدّین یہاں دِيْنِيْ بن گیا؛ اہلِ ذوق کہتے ہیں کہ حکم جب دل میں اتر کر حال بن جائے تو “دین” بندے کا اپنا “میرا دین” ہو جاتا ہے، باہر کا قانون اندر کی پہچان بن جاتا ہے۔ پھر مخالفوں سے کہا گیا: فَاعْبُدُوْا مَا شِئْتُمْ مِّنْ دُوْنِهٖ ۭ قُلْ اِنَّ الْخٰسِرِيْنَ الَّذِيْنَ خَسِرُوْۤا اَنْفُسَهُمْ وَاَهْلِيْهِمْ يَوْمَ الْقِيٰمَةِ ۭ اَلَا ذٰلِكَ هُوَ الْخُسْرَانُ الْمُبِيْنُ۔ “تو تم اس کے سوا جس کی چاہو عبادت کرو؛ کہہ دیجیے: اصل خسارے والے وہ ہیں جنہوں نے قیامت کے دن اپنے آپ کو اور اپنے گھر والوں کو گنوا دیا؛ سن رکھو، یہی کھلا خسارہ ہے۔” فَاعْبُدُوْا مَا شِئْتُمْ تنبیہ ہے، اجازت نہیں، مگر اس میں اختیار کی عزت بھی ہے: زبردستی نہیں کی جاتی، انجام دکھا دیا جاتا ہے۔ اپنے آپ کو گنوانا سب سے بڑا خسارہ ہے، کیونکہ مال گیا تو آدمی باقی رہا، آدمی خود گیا تو کچھ باقی نہ رہا۔ اہلِ تصوف میں مشہور اثر ہے کہ جس نے اپنے نفس کو پہچانا اس نے اپنے رب کو پہچانا؛ شیخِ اکبر کے مکتب میں اپنے آپ کو پانا اپنی حقیقت کو حق کے آئینے میں پہچاننا ہے، اور جس نے عمر بھر یہ ملاقات نہ کی اس نے گویا اپنے آپ کو کبھی پایا ہی نہیں۔ پھر اس خسارے کی صورت: لَهُمْ مِّنْ فَوْقِهِمْ ظُلَلٌ مِّنَ النَّارِ وَمِنْ تَحْتِهِمْ ظُلَلٌ ۭ ذٰلِكَ يُخَوِّفُ اللّٰهُ بِهٖ عِبَادَهٗ ۭ يٰعِبَادِ فَاتَّقُوْنِ۔ “ان کے اوپر سے بھی آگ کے سائبان ہوں گے اور ان کے نیچے سے بھی سائبان؛ یہی وہ چیز ہے جس سے اللہ اپنے بندوں کو ڈراتا ہے؛ اے میرے بندو، مجھ سے ڈرو۔” ظُلَل سایہ دار چھتریاں ہیں، اور سایہ عموماً راحت ہے؛ اہلِ تاویل کہتے ہیں کہ جس نے دنیا کے سایوں کو، جو اس مکتب میں حق کا ظل ہیں، اپنا مقصود بنا لیا، اسے آخر میں سایہ بھی آگ کا ملا۔ اور ڈرانا خود رحمت ہے: وہ اس لیے خبر دیتا ہے کہ بندہ وہاں نہ پہنچے؛ اسی لیے تنبیہ کے بیچ میں بھی پکار “اے میرے بندو” ہی رہی۔

اب اس رکوع کی روح سامنے آتی ہے: وَالَّذِيْنَ اجْتَنَبُوا الطَّاغُوْتَ اَنْ يَّعْبُدُوْهَا وَاَنَابُوْۤا اِلَى اللّٰهِ لَهُمُ الْبُشْرٰى ۚ فَبَشِّرْ عِبَادِ۔ “اور جو لوگ طاغوت کی عبادت سے بچے رہے اور اللہ کی طرف رجوع کیا، ان کے لیے خوش خبری ہے؛ تو میرے ان بندوں کو خوش خبری سنا دیجیے۔” طاغوت طغیان سے ہے، یعنی حد سے بڑھ جانا؛ ہر وہ چیز جو اللہ کی جگہ اطاعت مانگے۔ اہلِ تاویل کہتے ہیں کہ سب سے قریب کا طاغوت نفس ہے، جو اپنی خواہش کو حکم بنا کر پیش کرتا ہے، اور اجتناب کا معنی ہے اس سے پہلو بچا کر ایک طرف ہو جانا۔ دو قدم ذکر ہوئے: ایک بچنا، دوسرا رجوع؛ تخلیہ پھر تحلیہ، یعنی پہلے دل کو خالی کرنا، پھر اسے حق کی طرف موڑ دینا۔ فَبَشِّرْ عِبَادِ میں اضافت کی “ی” لفظ میں نہیں لکھی گئی اور صرف زیر اس کی نشانی بن کر رہ گیا؛ اہلِ ذوق اس میں یہ اشارہ دیکھتے ہیں کہ جو بندہ سچ مچ اس کا ہو جائے، اس کی نسبت کا اعلان بلند آواز سے نہیں ہوتا، ایک لطیف نشان رہ جاتا ہے، جیسے فنا کے بعد بندے کا صرف ایک اثر باقی رہے اور ساری نسبت حق کی ہو۔ یہ لطیفہ ہے، حکم نہیں؛ ظاہری معنی وہی ہے جو ترجمے میں ہے۔ اور یہ بشارت کن کے لیے ہے؟ اگلی آیت ان کی پہچان بتاتی ہے، اور وہ پہچان سننے سے شروع ہوتی ہے۔

وہ بندے یہ ہیں: الَّذِيْنَ يَسْتَمِعُوْنَ الْقَوْلَ فَيَتَّبِعُوْنَ اَحْسَنَهٗ ۭ اُولٰۤئِكَ الَّذِيْنَ هَدٰىهُمُ اللّٰهُ وَاُولٰۤئِكَ هُمْ اُولُوا الْاَلْبَابِ۔ “جو بات کو غور سے سنتے ہیں، پھر اس میں جو سب سے بہتر ہو اس کی پیروی کرتے ہیں؛ یہی وہ ہیں جنہیں اللہ نے ہدایت دی، اور یہی عقل والے ہیں۔” يَسْتَمِعُوْنَ سماع نہیں، استماع ہے، یعنی کان لگا کر، توجہ اور ارادے کے ساتھ سننا۔ مفسرین کے دو قول ہیں: ایک یہ کہ وہ ہر بات سنتے ہیں اور اس میں سے بہترین کو لیتے ہیں، دوسرا یہ کہ قرآن سنتے ہیں اور اس کے بہترین پہلو کو اپناتے ہیں، جیسے بدلے کی اجازت کے مقابل معاف کر دینا، اور رخصت کے مقابل عزیمت۔ یہی سماع کا ادب ہے: سننے والا پہلے اپنے آپ کو خالی کرتا ہے، اپنی رائے کا شور تھما دیتا ہے، پھر سنتا ہے۔ شیخِ اکبر کے مکتب میں سماع کی اصل سماعِ الٰہی ہے: اعیانِ ثابتہ نے سب سے پہلے کلمۂ کُن سنا اور وجود میں آ گئے؛ وجود خود ایک سننے کا جواب ہے۔ اہلِ معرفت کہتے ہیں کہ کامل سننے والا ہر کہنے والے کے پیچھے اصل متکلم کو سنتا ہے، اس لیے کسی بات کو حقیر نہیں سمجھتا اور ہر بات میں سے احسن کو چن لیتا ہے۔ اور اب ترتیب دیکھیے: پہلے سننا، پھر پیروی، پھر فرمایا هَدٰىهُمُ اللّٰهُ۔ یہ اس سورت کے اصول کی گواہی ہے کہ جو طلب لے کر آئے اسے ہدایت ملتی ہے؛ ہدایت ان پر اوپر سے ٹھونسی نہیں گئی، ان کے کان کھولنے اور قدم اٹھانے کا جواب بن کر آئی۔ اور انہیں پھر اُولُوا الْاَلْبَاب کہا گیا، وہی لقب جو نویں آیت میں شب بیدار عالموں کو ملا تھا؛ مغز والا وہی ہے جو سن کر بہترین کو پہچان لے۔

اب دوسری طرف کا حال: اَفَمَنْ حَقَّ عَلَيْهِ كَلِمَةُ الْعَذَابِ ۭ اَفَاَنْتَ تُنْقِذُ مَنْ فِي النَّارِ۔ “بھلا جس پر عذاب کی بات ثابت ہو چکی، تو کیا آپ اسے بچا لیں گے جو آگ میں ہے؟” یہ حضورِ اکرم صلی اللہ علیہ وسلم کی تسلی ہے، جن کا دل ہر انکار کرنے والے کے لیے بھی کڑھتا تھا۔ حَقَّ کا معنی ہے ثابت اور لازم ہو جانا، اور اسے سرِّ قدر کے ساتھ پڑھنا ضروری ہے: بات ابتدا میں کسی پر نہیں لکھ دی گئی؛ جب بندے نے سنی ہوئی بات سے منہ پھیرا اور اعراض پر جم گیا تو حکم اس کے اپنے رخ کے مطابق ثابت ہو گیا۔ شیخِ اکبر کے نزدیک حکم محکوم کے تابع ہے؛ کلمہ اسی پر ثابت ہوتا ہے جس کا حال اس کا تقاضا کرے۔ اور رسول کو یہ بتایا گیا کہ آپ کا کام پیغام ہے، کسی کو زبردستی کھینچ لانا نہیں؛ حق نے خود بھی کسی کو مجبور نہیں کیا، اختیار کی عزت رکھی۔ پھر بشارت کی طرف واپسی: لٰكِنِ الَّذِيْنَ اتَّقَوْا رَبَّهُمْ لَهُمْ غُرَفٌ مِّنْ فَوْقِهَا غُرَفٌ مَّبْنِيَّةٌ ۙ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ ۚ وَعْدَ اللّٰهِ ۭ لَا يُخْلِفُ اللّٰهُ الْمِيْعَادَ۔ “لیکن جو لوگ اپنے رب سے ڈرتے رہے ان کے لیے بالاخانے ہیں جن کے اوپر اور بالاخانے بنے ہوئے ہیں، ان کے نیچے نہریں بہتی ہیں؛ یہ اللہ کا وعدہ ہے، اللہ وعدے کے خلاف نہیں کرتا۔” سولہویں آیت میں آگ کے سائبان اوپر اور نیچے تھے، یہاں بالاخانوں پر بالاخانے؛ ایک طرف گھیر لینے والا اندھیرا، دوسری طرف بلندی پر بلندی۔ اہلِ معرفت کے نزدیک یہ منزلوں کی تہیں ہیں، اور شیخِ اکبر کے مکتب میں جنت والوں کی ترقی رکتی نہیں: ہر آن نئی تجلی، اور ہر تجلی کے اوپر ایک اور۔ نہریں نیچے بہتی ہیں، یعنی علم اور رحمت کا بہاؤ ان کے قدموں میں ہے۔

رکوع ایک تصویر پر ختم ہوتا ہے جو ساری دنیا کا خلاصہ ہے: اَلَمْ تَرَ اَنَّ اللّٰهَ اَنْزَلَ مِنَ السَّمَآءِ مَآءً فَسَلَكَهٗ يَنَابِيْعَ فِي الْاَرْضِ ثُمَّ يُخْرِجُ بِهٖ زَرْعًا مُّخْتَلِفًا اَلْوَانُهٗ ثُمَّ يَهِيْجُ فَتَرٰىهُ مُصْفَرًّا ثُمَّ يَجْعَلُهٗ حُطَامًا ۭ اِنَّ فِيْ ذٰلِكَ لَذِكْرٰى لِاُولِي الْاَلْبَابِ۔ “کیا آپ نے نہیں دیکھا کہ اللہ نے آسمان سے پانی اتارا، پھر اسے زمین میں چشموں کی صورت میں چلا دیا، پھر اس سے کھیتی نکالتا ہے جس کے رنگ طرح طرح کے ہیں، پھر وہ پک کر خشک ہونے لگتی ہے تو تم اسے زرد دیکھتے ہو، پھر وہ اسے چورا چورا کر دیتا ہے؛ بے شک اس میں عقل والوں کے لیے بڑی نصیحت ہے۔” پانی ایک، رنگ بہت؛ سورۂ رعد میں یہی بات کہی گئی تھی کہ باغ يُسْقٰى بِمَآءٍ وَّاحِدٍ، ایک ہی پانی سے سیراب ہوتے ہیں۔ شیخِ اکبر کے مکتب میں یہ وجود کی مثال ہے: فیض ایک ہے، اور رنگ ہر زمین کی استعداد سے بنتے ہیں۔ پھر یہی رنگین کھیتی زرد ہو کر حطام بن جاتی ہے؛ سورۂ کہف میں دنیا کی زندگی کی مثال یوں دی گئی تھی: كَمَآءٍ اَنْزَلْنٰهُ مِنَ السَّمَآءِ فَاخْتَلَطَ بِهٖ نَبَاتُ الْاَرْضِ فَاَصْبَحَ هَشِيْمًا تَذْرُوْهُ الرِّيٰحُ۔ اہلِ تاویل اس آیت کو سالک کے اندر بھی پڑھتے ہیں: آسمانی کلام کا پانی دل کی زمین میں اتر کر چشمہ بن جاتا ہے، جو اسے محفوظ رکھے اس کے اندر سے علم پھوٹتا ہے؛ اور جو صرف ظاہری رنگوں پر خوش ہو، وہ دیکھے کہ ہر رنگ زرد ہو کر بکھر جاتا ہے، باقی صرف وہ رہتا ہے جو پانی اتارنے والا ہے۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں یہ بارش اور کھیتی کا بیان ہے، اور باطن میں فنا اور بقا کا سبق۔ اور تیسری بار اُولُو الْاَلْبَاب کا ذکر آیا: رات کے عالم، بہترین بات سننے والے، اور اب دنیا کے چکر سے نصیحت لینے والے؛ ایک ہی مغز تین صورتوں میں۔
