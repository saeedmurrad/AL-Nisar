یہ رکوع ایک اجازت سے کھلتا ہے جو اس امت میں پہلی بار اتری: اُذِنَ لِلَّذِيْنَ يُقٰتَلُوْنَ بِاَنَّهُمْ ظُلِمُوْا ۭ وَاِنَّ اللّٰهَ عَلٰى نَصْرِهِمْ لَقَدِيْرٌ۔ "اجازت دی گئی ان لوگوں کو جن سے لڑائی کی جاتی ہے، اس لیے کہ ان پر ظلم ہوا؛ اور بے شک اللہ ان کی مدد پر پوری قدرت رکھتا ہے۔" پھر بتایا کہ یہ مظلوم کون ہیں: الَّذِيْنَ اُخْرِجُوْا مِنْ دِيَارِهِمْ بِغَيْرِ حَقٍّ اِلَّآ اَنْ يَّقُوْلُوْا رَبُّنَا اللّٰهُ۔ "وہ جو اپنے گھروں سے ناحق نکالے گئے، صرف اس بات پر کہ وہ کہتے تھے: ہمارا رب اللہ ہے۔" ظاہری معنی مقدم اور صاف ہے: مکہ کے برسوں کے ظلم، تعذیب اور جلاوطنی کے بعد، ہجرت کے پس منظر میں، حضورِ اکرم صلی اللہ علیہ وسلم اور آپ کے اصحاب کو اپنے دفاع کی اجازت ملی۔ یہ ایک دفاعی اور تاریخی اجازت ہے، مظلوم کے لیے، شریعت کی حدوں کے اندر؛ اور ان حدوں میں زیادتی کی ممانعت، غیر جنگجو کی جان کی حرمت اور عہد کی پاسداری سب شامل ہیں۔ آیت کے الفاظ خود اپنی حد کھینچتے ہیں: "يُقٰتَلُوْنَ"، یعنی جن سے لڑا جاتا ہے، اور "ظُلِمُوْا"، یعنی جن پر ظلم ہوا۔ ان الفاظ سے جارحیت کا جواز نکالنا آیت کے مفہوم کے خلاف ہے۔ اور ان کا جرم کیا تھا؟ فقط "رَبُّنَا اللّٰهُ" کہنا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں "رَبُّنَا" کا اقرار ربِّ خاص کی پہچان کا دروازہ ہے، کہ ہر عین اپنے رب کو اسی اسم سے پہچانتی ہے جو اس کی تربیت کرتا ہے؛ اور "اللّٰهُ" اسمِ جامع ہے جو سب اسماء کو سمیٹتا ہے۔ اہلِ معرفت کہتے ہیں کہ یہی کلمہ ہر زمانے میں بندے کو اس کے مانوس گھر سے نکالتا ہے: جو سچے دل سے کہہ دے کہ میرا رب اللہ ہے، اس کے اندر غیر کی بستی اسے چین سے رہنے نہیں دیتی۔ اور "عَلٰى نَصْرِهِمْ لَقَدِيْرٌ" کی نصرت اسی کے لیے ہے جو اپنی طاقت پر نہیں، اسی اقرار پر کھڑا ہو۔

آیت آگے ایک عالمگیر قانون کھولتی ہے: وَلَوْلَا دَفْعُ اللّٰهِ النَّاسَ بَعْضَهُمْ بِبَعْضٍ لَّهُدِّمَتْ صَوَامِعُ وَبِيَعٌ وَّصَلَوٰتٌ وَّمَسٰجِدُ يُذْكَرُ فِيْهَا اسْمُ اللّٰهِ كَثِيْرًا ۭ وَلَيَنْصُرَنَّ اللّٰهُ مَنْ يَّنْصُرُهٗ ۭ اِنَّ اللّٰهَ لَقَوِيٌّ عَزِيْزٌ۔ "اور اگر اللہ لوگوں کو ایک دوسرے کے ذریعے ہٹاتا نہ رہتا تو راہبوں کی خانقاہیں، گرجے، یہودیوں کے عبادت خانے اور مسجدیں، جن میں اللہ کا نام کثرت سے لیا جاتا ہے، ڈھا دی جاتیں؛ اور اللہ ضرور اس کی مدد کرے گا جو اس کی مدد کرتا ہے، بے شک اللہ قوت والا، غالب ہے۔" دیکھیے کہ اس اجازت کی غرض کیا بتائی گئی: کسی ایک گروہ کا غلبہ نہیں، عبادت گاہوں کی حفاظت۔ اور فہرست میں مسجد سب سے آخر میں آئی؛ خانقاہ، گرجا اور کنیسہ پہلے۔ یعنی ہر وہ گھر جہاں اللہ کا نام لیا جاتا ہے، اس آیت کی پناہ میں ہے، اور اس کا ڈھانا اسی فساد میں شمار ہے جسے روکنے کے لیے دفاع کی اجازت ملی۔ یہ اس قانون کا ظاہر ہے جس سے زمین کا توازن قائم ہے۔ اہلِ تاویل کہتے ہیں کہ اس کا ایک باطنی چہرہ بھی ہے، اور وہ ظاہر کی نفی نہیں، اس کی گہرائی ہے: بندے کے اندر بھی خلوت کی کوٹھڑیاں ہیں جہاں نام لیا جاتا ہے، اور اگر حق نفس کی یورش کو روح کی قوت سے پیچھے نہ ہٹاتا رہے تو یہ سب ڈھے جائیں۔ یہی جہاد بالنفس ہے جو ہر سانس کے ساتھ جاری ہے۔ شیخِ اکبر کے نزدیک یہ "دَفْع" اسماءِ الٰہیہ کا باہمی تقابل بھی ہے: ایک اسم کا حکم دوسرے اسم کے حکم کو تھامتا ہے، اور اسی سے عالم میں اعتدال باقی ہے۔ "مَنْ يَّنْصُرُهٗ" پر اہلِ ذوق ٹھہرتے ہیں: حق کو کسی کی مدد کی حاجت نہیں، تو اس کی مدد کیا ہے؟ اپنے اندر اور باہر اس کے نام کی حرمت کو قائم رکھنا۔ جو یہ کرے، قوی و عزیز اس کی پشت پر ہے۔

پھر وہ نشانی جس سے سچے مددگار پہچانے جاتے ہیں: اَلَّذِيْنَ اِنْ مَّكَّنّٰهُمْ فِي الْاَرْضِ اَقَامُوا الصَّلٰوةَ وَاٰتَوُا الزَّكٰوةَ وَاَمَرُوْا بِالْمَعْرُوْفِ وَنَهَوْا عَنِ الْمُنْكَرِ ۭ وَلِلّٰهِ عَاقِبَةُ الْاُمُوْرِ۔ "یہ وہ لوگ ہیں کہ اگر ہم انہیں زمین میں اقتدار دیں تو نماز قائم کریں، زکوٰۃ ادا کریں، بھلائی کا حکم دیں اور برائی سے روکیں؛ اور سب کاموں کا انجام اللہ ہی کے ہاتھ میں ہے۔" ظاہر میں یہ اہلِ اقتدار کا منشور ہے: طاقت ملے تو وہ بندگی، سخاوت اور عدل میں ڈھلے، تکبر اور انتقام میں نہیں۔ کل جو مظلوم تھے، آج غالب ہوں تو ان کا پہلا کام سجدہ ہو، بدلہ نہیں۔ اہلِ معرفت کہتے ہیں کہ "مَكَّنّٰهُمْ" میں سالک کی وہ منزل جھلکتی ہے جسے تمکین کہتے ہیں: تلوین کے بعد، جب حال کی لہریں تھمتی ہیں اور قدم جم جاتا ہے۔ تمکین کی علامت کرامت یا دعویٰ نہیں، یہی چار باتیں ہیں: حق کی طرف نماز کا رخ، خلق کی طرف زکوٰۃ کا ہاتھ، اور معروف و منکر کی وہ پہچان جو صرف اپنے تک نہ رہے۔ سورۂ الکہف میں ذوالقرنین کے بارے میں یہی لفظ آیا تھا: اِنَّا مَكَّنَّا لَهٗ فِي الْاَرْضِ وَاٰتَيْنٰهُ مِنْ كُلِّ شَيْءٍ سَبَبًا؛ اور انہوں نے ہر کامیابی پر اسے اپنے رب کی رحمت کہا تھا۔ شیخِ اکبر کے مکتب میں صاحبِ تمکین وہ ہے جو اسباب کے بیچ رہ کر بھی اسباب کو اصل مؤثر نہ جانے، اور "وَلِلّٰهِ عَاقِبَةُ الْاُمُوْرِ" اسی کا اعلان ہے: ہر کام کا آغاز بندے کے ہاتھ میں دکھائی دیتا ہے، مگر انجام اسی کی طرف لوٹتا ہے جس کی طرف سے آغاز ہوا تھا۔

اب تسلی کا لہجہ ہے، اور اس کے ساتھ تاریخ کی ایک لمبی قطار: وَاِنْ يُّكَذِّبُوْكَ فَقَدْ كَذَّبَتْ قَبْلَهُمْ قَوْمُ نُوْحٍ وَّعَادٌ وَّثَمُوْدُ ۙ وَقَوْمُ اِبْرٰهِيْمَ وَقَوْمُ لُوْطٍ ۙ وَّاَصْحٰبُ مَدْيَنَ ۚ وَكُذِّبَ مُوْسٰى فَاَمْلَيْتُ لِلْكٰفِرِيْنَ ثُمَّ اَخَذْتُهُمْ ۚ فَكَيْفَ كَانَ نَكِيْرِ۔ "اور اگر یہ آپ کو جھٹلاتے ہیں تو ان سے پہلے نوح کی قوم، عاد اور ثمود جھٹلا چکے، اور ابراہیم کی قوم اور لوط کی قوم، اور مدین والے؛ اور موسیٰ کو بھی جھٹلایا گیا؛ تو میں نے کافروں کو ڈھیل دی، پھر انہیں پکڑ لیا، تو کیسا تھا میرا انکار!" حضورِ اکرم صلی اللہ علیہ وسلم سے کہا جا رہا ہے کہ یہ جھٹلانا نیا نہیں؛ حضرت نوح، حضرت ہود، حضرت صالح، حضرت ابراہیم، حضرت لوط، حضرت شعیب اور حضرت موسیٰ علیہم السلام سب اسی راہ سے گزرے۔ مفسرین نے ایک باریکی نوٹ کی ہے: باقی سب میں "قوم" کا ذکر ہے، مگر "وَكُذِّبَ مُوْسٰى" مجہول صیغے میں آیا، کہ حضرت موسیٰ علیہ السلام کو ان کی اپنی قوم بنی اسرائیل نے نہیں بلکہ فرعون اور اس کے درباریوں نے جھٹلایا تھا۔ پھر دو لفظ جو اس رکوع کی کنجی ہیں: "فَاَمْلَيْتُ" اور "ثُمَّ اَخَذْتُهُمْ"۔ اہلِ معرفت کہتے ہیں کہ ڈھیل پکڑ سے پہلے آئی، اور یہی سبقتِ رحمت ہے؛ اسمِ حلیم کا حکم اسمِ منتقم کے حکم پر مقدم رہا۔ ڈھیل میں ہر لمحہ لوٹ آنے کا دروازہ کھلا تھا۔ پکڑ تب آئی جب انہوں نے اپنے اعراض کو آخری فیصلہ بنا لیا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق؛ جو رخ بندے نے خود چنا، حق نے اسی کو وجود دے دیا۔

پھر ان بستیوں کی تصویر جو اس پکڑ کے بعد باقی رہ گئیں: فَكَاَيِّنْ مِّنْ قَرْيَةٍ اَهْلَكْنٰهَا وَهِيَ ظَالِمَةٌ فَهِيَ خَاوِيَةٌ عَلٰى عُرُوْشِهَا وَبِئْرٍ مُّعَطَّلَةٍ وَّقَصْرٍ مَّشِيْدٍ۔ "تو کتنی ہی بستیاں ہیں جنہیں ہم نے ہلاک کیا جبکہ وہ ظلم کر رہی تھیں، سو وہ اپنی چھتوں کے بل گری پڑی ہیں؛ اور کتنے ہی کنویں بیکار پڑے ہیں اور کتنے ہی پختہ اونچے محل ویران۔" "وَهِيَ ظَالِمَةٌ" کا حال پہلے آیا اور ہلاکت بعد میں: ہلاکت ان کے ظلم کا نتیجہ ہے، کوئی بے سبب فیصلہ نہیں۔ "خَاوِيَةٌ عَلٰى عُرُوْشِهَا" یعنی پہلے چھتیں گریں، پھر دیواریں ان پر آ رہیں؛ اوپر کا سہارا ٹوٹا تو نیچے کا سب ڈھ گیا۔ اب دو چیزیں ساتھ ساتھ: کنواں جس کا پانی ہے مگر کوئی کھینچنے والا نہیں، اور محل جو اونچا ہے مگر کوئی بسنے والا نہیں۔ اہلِ تاویل کہتے ہیں کہ یہ ہر غافل انسان کا نقشہ ہے: باطن کا کنواں جس میں معرفت کا پانی اب بھی موجود ہے، مگر ڈول ڈالنے والا کوئی نہیں؛ اور ظاہر کا محل جسے عمر بھر چونے اور پتھر سے اونچا کیا گیا۔ کنواں اندر کی طرف کھدتا ہے اور محل اوپر کی طرف اٹھتا ہے؛ ایک عاجزی کی گہرائی ہے، دوسرا دعوے کی بلندی۔ شیخِ اکبر کے مکتب میں دل کا پانی وہی فیض ہے جو اس کی استعداد کے مطابق اترتا ہے؛ وہ کبھی خشک نہیں ہوتا، صرف "مُعَطَّلَة" ہو جاتا ہے، یعنی چھوڑ دیا جاتا ہے۔ اور یہ ترک بندے کا اپنا ہے؛ اسی لیے جو لوٹ آئے، اس کا کنواں پھر سے جاری ہو جاتا ہے۔

اور اب وہ آیت جو اس سورت کا مرکز ہے، اور شیخِ اکبر کے مکتب میں دل کی پوری حکمت کا خلاصہ: اَفَلَمْ يَسِيْرُوْا فِي الْاَرْضِ فَتَكُوْنَ لَهُمْ قُلُوْبٌ يَّعْقِلُوْنَ بِهَاۤ اَوْ اٰذَانٌ يَّسْمَعُوْنَ بِهَا ۚ فَاِنَّهَا لَا تَعْمَى الْاَبْصَارُ وَلٰكِنْ تَعْمَى الْقُلُوْبُ الَّتِيْ فِي الصُّدُوْرِ۔ "تو کیا یہ لوگ زمین میں چلے پھرے نہیں کہ ان کے ایسے دل ہوتے جن سے یہ سمجھتے، یا ایسے کان جن سے یہ سنتے؟ حقیقت یہ ہے کہ آنکھیں اندھی نہیں ہوتیں، بلکہ وہ دل اندھے ہو جاتے ہیں جو سینوں میں ہیں۔" ظاہر میں حکم ہے کہ اجڑی بستیوں کو دیکھو اور عبرت لو؛ اور عبرت کا لفظ ہی عبور سے ہے، یعنی دیکھی ہوئی صورت سے گزر کر اس کے معنی تک پہنچنا۔ آنکھ کھنڈر دیکھ لیتی ہے، مگر اس کھنڈر سے گزر کر معنی تک صرف دل جاتا ہے۔ غور کیجیے کہ قرآن نے سمجھنے کو دل سے جوڑا: "قُلُوْبٌ يَّعْقِلُوْنَ بِهَا"۔ اہلِ معرفت کہتے ہیں کہ زمین میں چلنے کی ایک سیر ظاہر کی ہے اور ایک سیر نفس کے اندر، جسے اہلِ طریق سیر و سلوک کہتے ہیں؛ پہلی دوسری کا دروازہ ہے، اس کا بدل نہیں۔ شیخِ اکبر کے مکتب میں عقل کا لفظ "عِقال" سے ہے، یعنی وہ رسی جو اونٹ کو باندھتی ہے؛ عقل ایک معنی کو باندھ کر رکھتی ہے۔ اور قلب "تقلّب" سے ہے، یعنی پلٹنا؛ دل ہر تجلی کے ساتھ پلٹتا ہے اور اپنی استعداد کے مطابق اس کی صورت قبول کرتا ہے۔ حکمتِ قلبیہ میں، جو حضرت شعیب علیہ السلام سے منسوب ہے، شیخِ اکبر کی تعلیم کا مفہوم یہ ہے کہ عارف کا دل حق کی وسعت کو سما لیتا ہے اور ہر تجلی میں اسے پہچانتا ہے۔ تو قلبِ عاقل وہ دل ہے جس میں یہ دونوں جمع ہوں: اتنا ٹھہراؤ کہ معنی کو تھام لے، اور اتنی وسعت کہ ہر نئی تجلی کے ساتھ پلٹ سکے۔

اب آیت کا دوسرا حصہ، جو اندھے پن کی جگہ بدل دیتا ہے: "لَا تَعْمَى الْاَبْصَارُ وَلٰكِنْ تَعْمَى الْقُلُوْبُ"۔ صرف و نحو پر ٹھہریے: "تَعْمَى" فعل لازم ہے اور اس کا فاعل خود "الْقُلُوْبُ" ہیں؛ یعنی دل اندھے ہوتے ہیں، ان پر اندھا پن زبردستی نہیں ڈالا جاتا۔ یہی وہ اصول ہے جس پر اس سلسلے کا ہر صفحہ کھڑا ہے: حق کسی کو جبر سے گمراہ نہیں کرتا؛ دل جب تجلی کی طرف سے رخ پھیرتا ہے تو وہ رخ پھیرنا ہی اس کا اندھا پن ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ پھر یہ قید کہ "الَّتِيْ فِي الصُّدُوْرِ"، جب کہ دل تو سینے ہی میں ہوتا ہے۔ اہلِ معرفت کہتے ہیں کہ اس تاکید میں ایک اشارہ ہے: صدر دل کا صحن ہے، جہاں وسوسے اور خیالات آتے جاتے ہیں، اور قلب اس صحن کے اندر کا گھر۔ جب صحن بھیڑ سے بھر جائے تو گھر کی کھڑکی پر گرد جم جاتی ہے، اور دل اپنے اندر ہوتے ہوئے بھی صحن کے شور میں کھو جاتا ہے۔ سورۂ بنی اسرائیل نے اسی اندھے پن کا انجام بتایا تھا: وَمَنْ كَانَ فِيْ هٰذِهٖۤ اَعْمٰى فَهُوَ فِي الْاٰخِرَةِ اَعْمٰى وَاَضَلُّ سَبِيْلًا؛ اور یہاں اس کی جگہ بتا دی گئی کہ وہ اندھا پن آنکھ کا نہیں، دل کا ہے۔ اہلِ تصوف میں یہ اثر مشہور ہے کہ حق فرماتا ہے: نہ میری زمین مجھے سما سکی نہ میرا آسمان، مگر میرے مومن بندے کے دل نے مجھے سما لیا۔ جس دل میں یہ وسعت رکھی گئی ہو، اس کا اندھا ہونا سب سے بڑا خسارہ ہے؛ اور اس کا علاج مراقبہ اور حضوری ہے: صحن کو خالی کرنا تاکہ گھر کی کھڑکی پھر نور کی طرف کھل جائے۔

اندھے دل کی ایک علامت جلد بازی ہے: وَيَسْتَعْجِلُوْنَكَ بِالْعَذَابِ وَلَنْ يُّخْلِفَ اللّٰهُ وَعْدَهٗ ۭ وَاِنَّ يَوْمًا عِنْدَ رَبِّكَ كَاَلْفِ سَنَةٍ مِّمَّا تَعُدُّوْنَ۔ "اور یہ آپ سے عذاب جلدی مانگتے ہیں، اور اللہ ہرگز اپنا وعدہ خلاف نہ کرے گا؛ اور بے شک آپ کے رب کے ہاں ایک دن تمہاری گنتی کے ہزار برس کے برابر ہے۔" وہ ڈھیل کو انکار کی دلیل سمجھتے تھے، حالانکہ ڈھیل مہلت تھی۔ "مِمَّا تَعُدُّوْنَ" پر ٹھہریے: پیمانہ گننے والے کا ہے؛ وقت ہر مرتبے میں اپنے ناپنے والے کے مطابق ہے۔ شیخِ اکبر کے نزدیک ہر اسم کا اپنا دن ہے، اور ایامِ رب ہمارے دنوں جیسے نہیں۔ ایک طرف قرآن کہتا ہے: كُلَّ يَوْمٍ هُوَ فِيْ شَاْنٍ؛ اور اس مکتب میں یہ "یوم" وہ ناقابلِ تقسیم آن ہے جس میں تجددِ امثال سے ہر شے نئی پیدا ہوتی ہے۔ دوسری طرف یہاں ایک دن ہزار برس کا ہے۔ تو اہلِ ذوق کہتے ہیں کہ حق کے ہاں سب سے چھوٹا دن اور سب سے لمبا دن ایک ہی حضور میں ہیں؛ جلدی اور دیر صرف گننے والے کی نسبت سے ہیں۔ پھر رکوع اسی قانون پر ختم ہوتا ہے جس سے پچھلی آیات میں بات چلی تھی: وَكَاَيِّنْ مِّنْ قَرْيَةٍ اَمْلَيْتُ لَهَا وَهِيَ ظَالِمَةٌ ثُمَّ اَخَذْتُهَا ۚ وَاِلَيَّ الْمَصِيْرُ۔ "اور کتنی ہی بستیاں ہیں جنہیں میں نے ڈھیل دی جبکہ وہ ظلم کر رہی تھیں، پھر میں نے انہیں پکڑ لیا، اور میری ہی طرف لوٹنا ہے۔" پینتالیسویں آیت میں "اَهْلَكْنٰهَا" جمع کے صیغے میں تھا، یہاں "اَمْلَيْتُ" اور "اَخَذْتُهَا" واحد متکلم میں؛ گویا ڈھیل اور پکڑ دونوں ایک ہی ذات کی براہِ راست نسبت ہیں۔ اور "وَاِلَيَّ الْمَصِيْرُ" سب کا انجام ہے۔ اہلِ معرفت کہتے ہیں کہ لوٹنا تو سب کو ہے؛ عارف وہ ہے جو اختیار سے آج لوٹ آئے، اس سے پہلے کہ لوٹایا جائے۔
