وَاِذْ بَوَّاْنَا لِاِبْرٰهِيْمَ مَكَانَ الْبَيْتِ اَنْ لَّا تُشْرِكْ بِيْ شَيْـًٔا وَّطَهِّرْ بَيْتِيَ لِلطَّاۤئِفِيْنَ وَالْقَاۤئِمِيْنَ وَالرُّكَّعِ السُّجُوْدِ۔ اور جب ہم نے ابراہیم کے لیے اس گھر کی جگہ مقرر کر کے ٹھکانا دیا (اور فرمایا) کہ میرے ساتھ کسی چیز کو شریک نہ ٹھہرانا، اور میرے گھر کو طواف کرنے والوں، قیام کرنے والوں اور رکوع و سجدہ کرنے والوں کے لیے پاک رکھنا۔ ظاہری معنی میں "بَوَّاْنَا" کا مطلب ہے جگہ مقرر کر کے ٹھکانا دینا۔ مفسرین بیان کرتے ہیں کہ بیت اللہ کے نشان مٹ چکے تھے، تو حق تعالیٰ نے حضرت ابراہیم علیہ السلام کو اس کی جگہ دکھائی اور انہوں نے حضرت اسماعیل علیہ السلام کے ساتھ اس کی بنیادیں اٹھائیں۔ تطہیر کا ظاہری حکم یہ ہے کہ بیت اللہ کو بتوں، نجاستوں اور ہر ناپاکی سے پاک رکھا جائے، اور یہ حکم آج بھی امت پر قائم ہے۔ اب ترتیب دیکھیے: گھر کی صفائی سے پہلے فرمایا کہ میرے ساتھ کسی کو شریک نہ کرنا؛ یعنی کعبہ کی پہلی بنیاد پتھر نہیں، توحید ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس کا اشارہ یہ ہے کہ مومن کا دل بھی حق کا گھر ہے۔ اہلِ تصوف میں مشہور اثر ہے کہ نہ میری زمین مجھے سما سکی نہ میرا آسمان، مگر میرے مومن بندے کے دل نے مجھے سما لیا؛ اور وسعتِ قلب کی بحث اسی مضمون پر قائم ہے۔ دل کی تطہیر بھی اسی ترتیب سے ہے: پہلے غیر کی نفی، پھر گھر کی صفائی۔ اہلِ تاویل کہتے ہیں کہ طواف کرنے والے وہ خیالات ہیں جو حق کے گرد گھومتے ہیں، قیام کرنے والے وہ عزائم جو اس کے حکم پر کھڑے ہیں، اور رکوع و سجود کرنے والے وہ قوتیں جو اس کے سامنے جھک گئیں؛ گھر انہی کے لیے خالی کیا جاتا ہے۔ اور "بَيْتِيَ" کی اضافت عزت کی اضافت ہے، ورنہ حق کسی مکان میں محدود نہیں۔ سورۂ ابراہیم میں خلت کے راز میں گزرا تھا کہ خلیل وہ ہے جس کے وجود میں صفاتِ حق رچ بس جائیں؛ جس کے دل میں غیر کی جگہ نہ رہی، اسی کو حق کے گھر کی جگہ دکھائی گئی۔

وَاَذِّنْ فِي النَّاسِ بِالْحَجِّ يَاْتُوْكَ رِجَالًا وَّعَلٰى كُلِّ ضَامِرٍ يَّاْتِيْنَ مِنْ كُلِّ فَجٍّ عَمِيْقٍ۔ اور لوگوں میں حج کا اعلان کر دو، وہ تمہارے پاس پیدل آئیں گے اور ہر دبلی سواری پر بھی، جو ہر دور دراز گہری راہ سے چلی آئیں گی۔ حج اسی اعلان کا جواب ہے، اور شریعتِ محمدی میں یہ ہر عاقل بالغ مسلمان پر عمر میں ایک بار فرض ہے جو استطاعت رکھتا ہو: وَلِلّٰهِ عَلَى النَّاسِ حِجُّ الْبَيْتِ مَنِ اسْتَطَاعَ اِلَيْهِ سَبِيْلًا۔ مفسرین نے روایت کیا ہے، جس کا مفہوم یہ ہے کہ حضرت ابراہیم علیہ السلام نے عرض کیا کہ میری آواز کہاں تک پہنچے گی، تو فرمایا گیا: پکارنا تمہارا کام ہے، پہنچانا ہمارا؛ اور جن کے مقدر میں حج تھا انہوں نے اسی وقت لبیک کہا۔ اہلِ معرفت کہتے ہیں کہ ہر حاجی کی تلبیہ اسی پرانی پکار کا جواب ہے، اور شیخِ اکبر کے مکتب میں یہ پکار اس سے بھی پرانی سماعت کی بازگشت ہے جب ارواح سے پوچھا گیا تھا: اَلَسْتُ بِرَبِّكُمْ ۭ قَالُوْا بَلٰى۔ روح نے ازل میں جو سنا تھا، حرم کی طرف سفر اسی سماعت کا یاد آ جانا ہے؛ جس نے وہاں "بَلٰى" کہا تھا وہ یہاں "لبیک" کہتا ہے۔ پھر "رِجَالًا" اور "ضَامِرٍ" پر غور کیجیے: کچھ پیدل، کچھ ایسی سواریوں پر جو طویل سفر سے دبلی ہو چکی ہوں۔ اہلِ ذوق کہتے ہیں کہ نفس بھی ریاضت سے یوں ہی ہلکا ہو کر منزل تک پہنچتا ہے؛ بوجھل نفس "فَجٍّ عَمِيْقٍ" کی گھاٹیاں پار نہیں کرتا۔ اور احرام کی دو سادہ چادریں اسی ادب کی صورت ہیں جو وادیِ طویٰ میں حضرت موسیٰ علیہ السلام کو سکھایا گیا: فَاخْلَعْ نَعْلَيْكَ ۚ اِنَّكَ بِالْوَادِ الْمُقَدَّسِ طُوًى؛ مقدس وادی میں داخل ہونے والا اپنی پہچان اور امتیاز کے لباس اتار دیتا ہے، اور بادشاہ اور فقیر ایک ہی کفن نما لباس میں حاضر ہوتے ہیں۔

لِّيَشْهَدُوْا مَنَافِعَ لَهُمْ وَيَذْكُرُوا اسْمَ اللّٰهِ فِيْۤ اَيَّامٍ مَّعْلُوْمٰتٍ عَلٰى مَا رَزَقَهُمْ مِّنْۢ بَهِيْمَةِ الْاَنْعَامِ ۚ فَكُلُوْا مِنْهَا وَاَطْعِمُوا الْبَآئِسَ الْفَقِيْرَ۔ تاکہ وہ اپنے فائدوں کے لیے حاضر ہوں، اور مقررہ دنوں میں ان چوپایوں پر اللہ کا نام لیں جو اس نے انہیں عطا کیے؛ پس ان میں سے خود بھی کھاؤ اور تنگ حال محتاج کو بھی کھلاؤ۔ ثُمَّ لْيَقْضُوْا تَفَثَهُمْ وَلْيُوْفُوْا نُذُوْرَهُمْ وَلْيَطَّوَّفُوْا بِالْبَيْتِ الْعَتِيْقِ۔ پھر وہ اپنا میل کچیل دور کریں، اپنی نذریں پوری کریں، اور اس قدیم گھر کا طواف کریں۔ پہلے ظاہری احکام: "مَنَافِعَ" میں دینی فائدے بھی ہیں اور دنیوی بھی، چنانچہ ایامِ حج میں جائز تجارت کی اجازت دی گئی۔ "اَيَّامٍ مَّعْلُوْمٰتٍ" کے بارے میں اہلِ علم کے دو مشہور اقوال ہیں: ذی الحجہ کے پہلے دس دن، یا قربانی کے دن۔ ان دنوں ہدی اور قربانی کے جانور اللہ کا نام لے کر ذبح کیے جاتے ہیں، اور ان کا گوشت خود بھی کھایا جاتا ہے اور فقیر کو بھی کھلایا جاتا ہے۔ "تَفَث" سے مراد وہ پراگندگی ہے جو احرام کی پابندیوں میں جمع ہوتی ہے؛ سر منڈوانا یا بال کتروانا، ناخن تراشنا اور احرام سے نکلنا اسی کا پورا کرنا ہے۔ پھر نذریں پوری کی جائیں اور بیت اللہ کا طواف ہو؛ جمہور مفسرین کے نزدیک یہاں طوافِ زیارت مراد ہے جو حج کا رکن ہے۔ اہلِ تاویل کہتے ہیں کہ اس ترتیب میں سلوک کی ترتیب چھپی ہے: نفس کی خواہش کی قربانی، پھر اس کا میل اترنا، پھر ازلی عہد کی وفا، پھر دل کا حق کے گرد گھومنا۔ "الْعَتِيْق" کے معنی قدیم بھی ہیں اور آزاد بھی؛ مفسرین سے منقول ہے کہ یہ گھر جابروں کے تسلط سے آزاد رکھا گیا۔ شیخِ اکبر کے نزدیک اس کا اشاری پہلو یہ ہے کہ عارف کا دل بھی عتیق ہے: غیر کی غلامی سے آزاد، اور اپنی اصل میں قدیم کہ اس کی عینِ ثابتہ علمِ حق میں ہمیشہ سے تھی۔ اور "مَنَافِعَ لَهُمْ" میں "لَهُمْ" قابلِ غور ہے: فائدہ بندوں کا ہے؛ حق غنی ہے، اسے حج کی حاجت نہیں، بندے کو حق کی حاجت ہے۔

ذٰلِكَ ۗ وَمَنْ يُّعَظِّمْ حُرُمٰتِ اللّٰهِ فَهُوَ خَيْرٌ لَّهٗ عِنْدَ رَبِّهٖ۔ یہ (حکم) ہے؛ اور جو اللہ کی حرمتوں کی تعظیم کرے تو یہ اس کے رب کے ہاں اس کے لیے بہتر ہے۔ فَاجْتَنِبُوا الرِّجْسَ مِنَ الْاَوْثَانِ وَاجْتَنِبُوْا قَوْلَ الزُّوْرِ۔ پس بتوں کی گندگی سے بچو اور جھوٹی بات سے بچو۔ "حُرُمٰتِ اللّٰهِ" وہ سب ہیں جنہیں حق نے محترم ٹھہرایا: احرام کی پابندیاں، حرم کی سرزمین، حرمت والے مہینے، اور شریعت کی حدود۔ اسی آیت کے درمیان یہ بھی فرمایا کہ چوپائے تمہارے لیے حلال کیے گئے سوائے ان کے جن کا حکم تمہیں پڑھ کر سنایا جاتا ہے؛ یعنی حلال و حرام کا فیصلہ حق کا ہے، جاہلیت کی خود ساختہ رسموں کا نہیں۔ پھر دو چیزوں سے بچنے کا حکم ایک ہی سانس میں آیا: بتوں کی گندگی اور جھوٹی بات۔ حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے جھوٹی گواہی کو بڑے گناہوں میں شرک کے ساتھ شمار فرمایا۔ اہلِ معرفت کہتے ہیں کہ بت اور جھوٹ ایک ہی جنس کے ہیں: بت وہ صورت ہے جسے حقیقت کا درجہ دے دیا جائے، اور جھوٹ وہ بات ہے جو حقیقت کے بغیر حقیقت کا دعویٰ کرے۔ شیخِ اکبر کے مکتب میں ہر صورت حق کا مظہر ہے، مگر مظہر کو ظاہر سمجھ لینا ہی بت پرستی کی جڑ ہے؛ عارف صورت میں جلوہ دیکھتا ہے اور سجدہ صرف صاحبِ جلوہ کو کرتا ہے۔ اور بت صرف پتھر کے نہیں، قرآن نے خواہش کو معبود بنانے والوں کا بھی ذکر کیا ہے: اَفَرَءَيْتَ مَنِ اتَّخَذَ اِلٰهَهٗ هَوٰىهُ۔ پس اجتناب کا ظاہر بتوں اور جھوٹ سے دوری ہے، اور اس کا باطن دل کو ان خواہشوں سے پاک کرنا جو چپکے سے معبود بن بیٹھتی ہیں۔

حُنَفَآءَ لِلّٰهِ غَيْرَ مُشْرِكِيْنَ بِهٖ ۭ وَمَنْ يُّشْرِكْ بِاللّٰهِ فَكَاَنَّمَا خَرَّ مِنَ السَّمَآءِ فَتَخْطَفُهُ الطَّيْرُ اَوْ تَهْوِيْ بِهِ الرِّيْحُ فِيْ مَكَانٍ سَحِيْقٍ۔ اللہ کے لیے یکسو ہو کر، اس کے ساتھ کسی کو شریک نہ ٹھہراتے ہوئے؛ اور جو اللہ کے ساتھ شریک ٹھہراتا ہے تو گویا وہ آسمان سے گر پڑا، پھر پرندے اسے اچک لیتے ہیں یا ہوا اسے کسی دور دراز جگہ لے جا پھینکتی ہے۔ "حُنَفَآء" حنیف کی جمع ہے، یعنی ہر باطل سے کٹ کر ایک حق کی طرف جھکنے والے؛ اور یہی ملتِ ابراہیمی کا عنوان ہے۔ پھر شرک کی تمثیل آئی: آسمان سے گرنا۔ اہلِ معرفت کہتے ہیں کہ انسان کی فطرت بلند ہے اور توحید اس کا آسمان؛ مشرک اس آسمان سے گرتا ہے، اور فعل "خَرَّ" بتاتا ہے کہ وہ خود گرا، اسے دھکیلا نہیں گیا۔ یہی اصول ہے: گمراہی بندے کا اپنا اعراض ہے، حق اسے زبردستی نہیں گراتا۔ گرنے والے کو پرندے ٹکڑے ٹکڑے اچک لیتے ہیں: یہ خواہشیں ہیں جن میں سے ہر ایک اس کا ایک حصہ لے اڑتی ہے؛ یا ہوا اسے دور پھینک دیتی ہے: یہ زمانے کی بدلتی رائیں ہیں جو اسے کسی ٹھکانے پر ٹکنے نہیں دیتیں۔ دونوں صورتوں میں انجام بکھراؤ ہے۔ یہی بکھراؤ حضرت یوسف علیہ السلام نے قید خانے کے ساتھیوں کے سامنے یوں بیان کیا تھا: ءَاَرْبَابٌ مُّتَفَرِّقُوْنَ خَيْرٌ اَمِ اللّٰهُ الْوَاحِدُ الْقَهَّارُ۔ شیخِ اکبر کے نزدیک بہت سے رب بندے کو بہت سے ٹکڑوں میں بانٹ دیتے ہیں، کیونکہ ہر رب اپنے بندے میں سے اپنا حصہ مانگتا ہے؛ اور حنیف وہ ہے جو ایک کی طرف جھک کر خود بھی جمع ہو گیا۔ توحید صرف عقیدہ نہیں، وجود کی یکجائی بھی ہے۔

ذٰلِكَ ۗ وَمَنْ يُّعَظِّمْ شَعَآئِرَ اللّٰهِ فَاِنَّهَا مِنْ تَقْوَى الْقُلُوْبِ۔ یہ (حکم) ہے؛ اور جو اللہ کے شعائر کی تعظیم کرے تو یہ دلوں کے تقویٰ سے ہے۔ لَكُمْ فِيْهَا مَنَافِعُ اِلٰۤى اَجَلٍ مُّسَمًّى ثُمَّ مَحِلُّهَاۤ اِلَى الْبَيْتِ الْعَتِيْقِ۔ تمہارے لیے ان میں ایک مقررہ مدت تک فائدے ہیں، پھر ان کے ذبح ہونے کی جگہ اسی قدیم گھر کے پاس ہے۔ شعائر وہ نشانیاں ہیں جو اللہ کی طرف توجہ دلاتی ہیں: صفا و مروہ، عرفات، مزدلفہ، جمرات، اور خاص طور پر قربانی کے وہ جانور جو حرم کے لیے نامزد کیے جائیں۔ مفسرین کہتے ہیں کہ ان کی تعظیم یہ ہے کہ عمدہ، تندرست اور فربہ جانور چنا جائے اور مناسک سنت کے مطابق ٹھیک ٹھیک ادا ہوں۔ اگلی آیت نے بتایا کہ ان جانوروں سے ایک مدت تک سواری اور دودھ کا فائدہ لیا جا سکتا ہے، پھر ان کا انجام حرم میں قربانی ہے۔ اب اس جملے کی باریکی دیکھیے: تعظیم ہاتھ پاؤں کرتے ہیں، مگر اسے "تَقْوَى الْقُلُوْبِ" کہا گیا۔ یعنی ظاہری عمل باطنی حال کی صورت ہے، اور باطنی حال ظاہری عمل کی روح۔ شیخِ اکبر کے نزدیک ظاہر اور باطن ایک ہی حقیقت کے دو رخ ہیں اور دونوں حق کے اسماء ہیں؛ جو ظاہر کو چھوڑ کر باطن کا دعویٰ کرے وہ آدھی حقیقت کا منکر ہے، اور جو باطن کے بغیر ظاہر پر قانع ہو وہ صورت کے پاس رک گیا۔ اسی لیے اہلِ معرفت کہتے ہیں کہ جو دل کے تقویٰ کا دعویٰ کرے اور مناسک یا قربانی کو ہلکا جانے، اس کا دعویٰ سچا نہیں؛ آیت نے تقویٰ کی پہچان ہی شعائر کی تعظیم بتائی ہے۔ لفظ "شعیرہ" شعور سے ہے: جو نشانی کی تعظیم کرتا ہے وہ اس کے ذریعے صاحبِ نشانی کو محسوس کرنے لگتا ہے۔ اور "مَحِلُّهَاۤ اِلَى الْبَيْتِ الْعَتِيْقِ" میں ہر سفر کا خلاصہ ہے: ہر نفع کی ایک مقررہ مدت ہے، اور ہر چیز کا آخری پڑاؤ وہی گھر ہے؛ سب کچھ اسی کی طرف لوٹتا ہے جہاں سے چلا تھا۔
