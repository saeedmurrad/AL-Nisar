وَلِكُلِّ اُمَّةٍ جَعَلْنَا مَنْسَكًا لِّيَذْكُرُوا اسْمَ اللّٰهِ عَلٰى مَا رَزَقَهُمْ مِّنْۢ بَهِيْمَةِ الْاَنْعَامِ ۭ فَاِلٰهُكُمْ اِلٰهٌ وَّاحِدٌ فَلَهٗۤ اَسْلِمُوْا ۭ وَبَشِّرِ الْمُخْبِتِيْنَ۔ اور ہم نے ہر امت کے لیے قربانی کا ایک طریقہ مقرر کیا تاکہ وہ ان چوپایوں پر اللہ کا نام لیں جو اس نے انہیں عطا کیے؛ پس تمہارا معبود ایک ہی معبود ہے، سو اسی کے فرمانبردار بن جاؤ، اور عاجزی کرنے والوں کو خوشخبری سنا دو۔ ظاہری معنی یہ ہے کہ قربانی کوئی نئی رسم نہیں؛ ہر شریعت میں اللہ کے نام پر جانور ذبح کرنے کا کوئی نہ کوئی طریقہ رہا ہے۔ "مَنْسَك" قربانی کو بھی کہتے ہیں اور عبادت کی جگہ اور طریقے کو بھی۔ مقصد ایک ہی بتایا گیا: "لِّيَذْكُرُوا اسْمَ اللّٰهِ"؛ ذبح کی روح اللہ کا نام ہے، اسی لیے جس جانور پر غیر اللہ کا نام لیا جائے وہ حرام ہے۔ اور یہ بھی واضح رہے کہ امتوں کے مناسک کا یہ تنوع تاریخ کا بیان ہے؛ ہمارے لیے شریعتِ محمدی ہی واجب العمل ہے۔ پھر صورتوں کی کثرت کے فوراً بعد فرمایا: "فَاِلٰهُكُمْ اِلٰهٌ وَّاحِدٌ"۔ یہی بات سورۂ انبیاء میں آئی: اِنَّ هٰذِهٖۤ اُمَّتُكُمْ اُمَّةً وَّاحِدَةً ۖ وَّاَنَا رَبُّكُمْ فَاعْبُدُوْنِ۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں شریعتوں کی کثرت اسماءِ الٰہیہ کے مظاہر کی کثرت ہے: ہر امت کو اس کی استعداد کے مطابق طریقہ ملا، مگر سب طریقوں کا قبلہ ایک ذات ہے۔ "مُخْبِت" کی اصل "خَبْت" ہے، یعنی نشیبی ہموار زمین۔ اہلِ ذوق کہتے ہیں کہ پانی بلندی پر نہیں ٹھہرتا، نشیب میں جمع ہوتا ہے؛ اسی طرح رحمت عاجز دل میں ٹھہرتی ہے۔ خوشخبری ان کے لیے ہے جنہوں نے اپنے آپ کو نیچا کیا، کیونکہ "اَسْلِمُوْا" کا عملی روپ یہی پستی ہے۔

الَّذِيْنَ اِذَا ذُكِرَ اللّٰهُ وَجِلَتْ قُلُوْبُهُمْ وَالصّٰبِرِيْنَ عَلٰى مَاۤ اَصَابَهُمْ وَالْمُقِيْمِي الصَّلٰوةِ ۙ وَمِمَّا رَزَقْنٰهُمْ يُنْفِقُوْنَ۔ وہ کہ جب اللہ کا ذکر کیا جائے تو ان کے دل کانپ اٹھتے ہیں، اور جو مصیبت ان پر آئے اس پر صبر کرتے ہیں، اور نماز قائم کرتے ہیں، اور جو کچھ ہم نے انہیں دیا ہے اس میں سے خرچ کرتے ہیں۔ مخبتین کی چار نشانیاں بیان ہوئیں، اور ترتیب پر غور کیجیے: پہلی دل کی، دوسری نفس کی، تیسری بدن کی، چوتھی مال کی۔ دل ذکر سے لرزتا ہے، نفس مصیبت پر صبر کرتا ہے، بدن نماز میں کھڑا ہوتا ہے، اور ہاتھ رزق میں سے دیتا ہے؛ گویا باطن سے ظاہر تک پورا وجود ایک ہی سمت میں آ گیا۔ "وَجِلَ" محض ڈر نہیں، یہ وہ ہیبت ہے جو عظمت کے سامنے طاری ہوتی ہے۔ اہلِ معرفت کہتے ہیں کہ یہ جلال کی تجلی پر دل کا جواب ہے۔ سورۂ رعد میں ذکر ہی کا دوسرا اثر بیان ہوا تھا: اَلَا بِذِكْرِ اللّٰهِ تَطْمَئِنُّ الْقُلُوْبُ۔ ایک ہی ذکر سے یہاں دل کانپتا ہے اور وہاں ٹھہر جاتا ہے۔ شیخِ اکبر کے نزدیک یہ تضاد نہیں، قلب کا نام ہی تقلب سے ہے: وہ کبھی جلال میں قبض پاتا ہے اور کبھی جمال میں بسط، اور عارف دونوں حالتوں میں ایک ہی حق کو دیکھتا ہے۔ صبر کا ذکر "عَلٰى مَاۤ اَصَابَهُمْ" کے ساتھ ہے، یعنی جو کچھ بھی پہنچے؛ یہ تمکین ہے کہ حال بدلے مگر رخ نہ بدلے، اور یہ اس شخص کا الٹ ہے جس کا ذکر دوسرے رکوع میں کنارے پر کھڑے ہو کر عبادت کرنے والے کے طور پر آیا تھا۔ نماز کا ذکر "اَلْمُقِيْمِي الصَّلٰوةِ" سے ہوا، یعنی قائم کرنے والے، صرف پڑھنے والے نہیں؛ اور اقامت حضوری کے بغیر نہیں ہوتی۔ آخر میں "مِمَّا رَزَقْنٰهُمْ": دینے والا اپنا مال نہیں دیتا، اسی کی امانت آگے بڑھاتا ہے جس نے دی تھی؛ اسی لیے انفاق میں احسان جتانا ادب کے خلاف ہے۔

وَالْبُدْنَ جَعَلْنٰهَا لَكُمْ مِّنْ شَعَآئِرِ اللّٰهِ لَكُمْ فِيْهَا خَيْرٌ ۖ فَاذْكُرُوا اسْمَ اللّٰهِ عَلَيْهَا صَوَآفَّ۔ اور قربانی کے اونٹوں کو ہم نے تمہارے لیے اللہ کے شعائر میں سے بنایا ہے، ان میں تمہارے لیے بھلائی ہے؛ پس انہیں قطار میں کھڑا کر کے ان پر اللہ کا نام لو۔ فَاِذَا وَجَبَتْ جُنُوْبُهَا فَكُلُوْا مِنْهَا وَاَطْعِمُوا الْقَانِعَ وَالْمُعْتَرَّ ۭ كَذٰلِكَ سَخَّرْنٰهَا لَكُمْ لَعَلَّكُمْ تَشْكُرُوْنَ۔ پھر جب وہ پہلو کے بل گر پڑیں تو ان میں سے کھاؤ، اور قناعت سے بیٹھے رہنے والے کو بھی کھلاؤ اور سامنے آ کر مانگنے والے کو بھی؛ اسی طرح ہم نے انہیں تمہارے قابو میں کر دیا تاکہ تم شکر کرو۔ "بُدْن" بڑے جانور ہیں جو حرم میں قربانی کے لیے لائے جائیں؛ لفظ کی اصل اونٹ کے لیے ہے، اور احناف کے نزدیک گائے بھی اس حکم میں داخل ہے۔ "صَوَآفَّ" کا مطلب ہے قطار میں کھڑے ہوئے؛ سنت یہ ہے کہ اونٹ کو کھڑا کر کے، اس کا ایک اگلا پاؤں باندھ کر، اللہ کا نام لے کر نحر کیا جائے۔ "وَجَبَتْ جُنُوْبُهَا" یعنی جب وہ پہلو کے بل گر جائیں اور جان نکل جائے، تب گوشت کاٹا اور کھایا جائے۔ "اَلْقَانِع" وہ محتاج ہے جو قناعت سے بیٹھا رہے اور سوال نہ کرے، اور "اَلْمُعْتَرّ" وہ جو سامنے آ کر اپنی حاجت ظاہر کرے؛ دونوں کا حق رکھا گیا، اور فقہا نے گوشت کے تین حصے کرنا مستحب بتایا ہے: اپنے لیے، عزیزوں کے لیے، اور فقیروں کے لیے۔ پھر "سَخَّرْنٰهَا": اتنا بڑا جانور جسے ایک بچہ نکیل سے کھینچ لے جاتا ہے، یہ انسان کی قوت نہیں، حق کی تسخیر ہے۔ شیخِ اکبر کے مکتب میں تسخیر متبادل ہے: جانور انسان کے لیے مسخر ہے، اور انسان بھی اس کے چارے اور نگہبانی کے لیے مسخر ہے؛ ہر شے دوسری کی کسی حاجت کے لیے مسخر ہے، اور حقیقی بے نیاز صرف حق ہے۔ اہلِ تاویل کہتے ہیں کہ نفس کا اونٹ بھی ریاضت سے مسخر ہوتا ہے، اور جب اس کی سرکشی پہلو کے بل گر جائے تب اس سے دوسروں کو فیض پہنچتا ہے۔ یہ تاویل ظاہری قربانی کی جگہ نہیں لیتی؛ اونٹ کی قربانی اپنی جگہ قائم ہے، اور نفس کی قربانی اس کا باطن ہے۔

لَنْ يَّنَالَ اللّٰهَ لُحُوْمُهَا وَلَا دِمَآؤُهَا وَلٰكِنْ يَّنَالُهُ التَّقْوٰى مِنْكُمْ۔ اللہ کو نہ ان کا گوشت پہنچتا ہے نہ ان کا خون، بلکہ اسے تمہاری طرف سے تقویٰ پہنچتا ہے۔ مفسرین بیان کرتے ہیں کہ جاہلیت کے لوگ قربانی کا خون کعبہ کی دیواروں پر ملتے اور گوشت اس کے پاس رکھتے تھے، گویا یہ چیزیں معبود تک پہنچائی جا رہی ہوں۔ آیت نے اس تصور کو جڑ سے کاٹ دیا: حق تعالیٰ گوشت اور خون سے بے نیاز ہے۔ مگر یہ آیت قربانی کو ساقط نہیں کرتی، اس کی غرض بیان کرتی ہے۔ قربانی کا حکم باقی ہے: احناف کے نزدیک صاحبِ نصاب پر واجب ہے اور دوسرے ائمہ کے نزدیک سنتِ مؤکدہ؛ اور حج کی ہدی اپنے احکام کے ساتھ لازم ہے۔ جو اس آیت کو بہانہ بنا کر کہے کہ اصل تو نیت ہے، جانور کی کیا ضرورت، اس نے آیت کا آدھا حصہ لیا اور آدھا چھوڑ دیا؛ کیونکہ تقویٰ اسی ذبح میں سے ہو کر پہنچتا ہے، ذبح کو چھوڑ کر نہیں۔ اب لفظ "يَنَال" پر غور کیجیے، یعنی پہنچنا اور قبول ہونا۔ گوشت زمین کا تھا، زمین پر رہا؛ خون مٹی میں جذب ہو گیا؛ جو اوپر گیا وہ دل کی کیفیت تھی۔ اہلِ معرفت کہتے ہیں کہ ہر عمل کا ایک جسم ہے اور ایک روح؛ جسم یہیں رہ جاتا ہے، روح بارگاہ تک جاتی ہے۔ شیخِ اکبر کے نزدیک حق غنی ہے اور اس کی بارگاہ میں کوئی چیز اس کی حاجت کے طور پر نہیں پہنچتی؛ پہنچتا وہ ہے جو بندے کی اپنی حقیقت سے اٹھا ہو، اور وہ عبدیت ہے۔ اور "مِنْكُمْ" نے بتایا کہ تقویٰ بندے کی اپنی طرف سے ہے؛ حق اسے قبول کرتا ہے، کسی پر تھوپتا نہیں۔ پچھلا رکوع (آیت ۳۲) "تَقْوَى الْقُلُوْبِ" کہہ چکا تھا، اور یہاں اسی تقویٰ کو قربانی کا اصل حاصل بتایا۔

كَذٰلِكَ سَخَّرَهَا لَكُمْ لِتُكَبِّرُوا اللّٰهَ عَلٰى مَا هَدٰىكُمْ ۭ وَبَشِّرِ الْمُحْسِنِيْنَ۔ اسی طرح اس نے انہیں تمہارے لیے مسخر کیا تاکہ تم اس ہدایت پر اللہ کی بڑائی بیان کرو جو اس نے تمہیں دی، اور نیکی کرنے والوں کو خوشخبری سنا دو۔ ظاہری حکم یہ ہے کہ ذبح کے وقت اللہ کا نام اور تکبیر کہی جائے، اور ایامِ تشریق میں نمازوں کے بعد تکبیر کا معمول اسی روح سے ہے۔ اب اس قربانی کا باطن دیکھیے۔ حضرت ابراہیم علیہ السلام سے سب سے عزیز چیز مانگی گئی، انہوں نے پیش کر دی، تو بیٹے کی جگہ مینڈھا آیا: وَفَدَيْنٰهُ بِذِبْحٍ عَظِيْمٍ۔ شیخِ اکبر کے مکتب میں اس فدیے کا اشارہ یہ ہے کہ حق کو بندے کا خون نہیں چاہیے، بندے کی وہ خودی چاہیے جو عزیز چیز کو اپنی ملکیت سمجھتی ہے؛ جب وہ دعویٰ قربان ہو گیا تو بیٹا بھی سلامت رہا اور خلت بھی کامل ہوئی۔ ہر سال کی قربانی اسی سنتِ ابراہیمی کی یاد ہے، اور اس کی روح نفس کی قربانی ہے: چھری جانور کی گردن پر ہے، مگر نیت نفس کی گردن پر۔ پھر تکبیر "عَلٰى مَا هَدٰىكُمْ" کے ساتھ آئی: بڑائی ہدایت پر بیان کی جاتی ہے، کیونکہ ہدایت سب سے بڑی عطا ہے؛ اور یہ عطا ان کو ملی جو اس کی طرف بڑھے، نہ کسی پر زبردستی ڈالی گئی نہ کسی طالب سے روکی گئی۔ آخر میں محسنین کو خوشخبری ہے۔ حدیثِ احسان کا مفہوم ہے کہ اللہ کی عبادت یوں کرو گویا تم اسے دیکھ رہے ہو، اور اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے۔ سورۂ نحل کے آخر میں یہی دونوں صفتیں ساتھ آئیں: اِنَّ اللّٰهَ مَعَ الَّذِيْنَ اتَّقَوْا وَّالَّذِيْنَ هُمْ مُّحْسِنُوْنَ۔ تقویٰ قربانی کی روح ہے اور احسان اس روح کا کمال: قربانی کرنے والا یہ دیکھے کہ وہ کس کے سامنے کھڑا ہے۔

اِنَّ اللّٰهَ يُدٰفِعُ عَنِ الَّذِيْنَ اٰمَنُوْا ۭ اِنَّ اللّٰهَ لَا يُحِبُّ كُلَّ خَوَّانٍ كَفُوْرٍ۔ بے شک اللہ ایمان والوں کا دفاع کرتا ہے؛ بے شک اللہ کسی دغاباز ناشکرے کو پسند نہیں کرتا۔ ظاہری پس منظر یہ ہے کہ مسلمان مکہ میں ستائے جا رہے تھے اور حرم سے روکے گئے تھے؛ اس آیت نے تسلی دی کہ ان کا دفاع حق خود کرے گا، اور اسی کے بعد اگلے رکوع میں مظلوموں کو دفاع کی اجازت کا ذکر آتا ہے۔ "يُدٰفِعُ" مفاعلہ کے وزن پر ہے، جس میں تکرار اور مسلسل مقابلے کا مفہوم ہے: یہ ایک بار کی مدد نہیں، ہر لمحے کی نگہبانی ہے۔ شیخِ اکبر کے نزدیک یہ تجددِ امثال کا ایک رخ ہے: جس طرح ہر لحظہ خلقِ جدید ہے، اسی طرح ہر لحظہ دفاعِ جدید ہے۔ اہلِ ذوق کہتے ہیں کہ جو اپنا دفاع خود کرنے پر تُلا رہے وہ اپنے نفس کے حوالے کر دیا جاتا ہے، اور جو اپنا معاملہ حق کو سونپ دے اس کا کارساز حق بن جاتا ہے۔ اس کا مطلب اسباب کا ترک نہیں؛ شریعت کی حدود میں حفاظت کی تدبیر خود حکم ہے، مگر دل کا بھروسا تدبیر پر نہیں، دفاع کرنے والے پر ہو۔ پھر آیت کا دوسرا حصہ اصول بیان کرتا ہے: حق "خَوَّان" اور "كَفُوْر" کو پسند نہیں کرتا۔ دونوں مبالغے کے صیغے ہیں، یعنی وہ جو بار بار خیانت کرے اور بار بار ناشکری۔ یہ ناپسندیدگی کسی کی پیشانی پر پہلے سے لگائی ہوئی مہر نہیں، بندے کی اپنی صفت کا نتیجہ ہے۔ امانتوں میں سب سے بڑی امانت توحید اور شعائر کی حرمت ہے جن کا ذکر ان رکوعوں میں ہوا؛ جو اس امانت میں خیانت کرے اور نعمتِ ہدایت کی ناشکری کرے، اس نے خود اپنے آپ کو محبت کے دائرے سے نکالا۔ حکم محکوم کے مطابق ہے، اور یہی سرِّ قدر ہے۔
