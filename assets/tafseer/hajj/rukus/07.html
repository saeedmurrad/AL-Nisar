پچھلا رکوع اندھے دلوں اور ڈھیل پانے والی بستیوں پر ختم ہوا تھا؛ اب حضورِ اکرم صلی اللہ علیہ وسلم کو حکم ہوتا ہے کہ اپنا منصب صاف لفظوں میں بیان فرما دیں: قُلْ يٰۤاَيُّهَا النَّاسُ اِنَّمَاۤ اَنَا لَكُمْ نَذِيْرٌ مُّبِيْنٌ۔ "فرما دیجیے: اے لوگو! میں تو بس تمہیں کھول کر خبردار کرنے والا ہوں۔" پھر دونوں راہوں کا انجام: فَالَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ لَهُمْ مَّغْفِرَةٌ وَّرِزْقٌ كَرِيْمٌ۔ "تو جو ایمان لائے اور نیک عمل کیے، ان کے لیے بخشش ہے اور عزت کی روزی۔" وَالَّذِيْنَ سَعَوْا فِيْۤ اٰيٰتِنَا مُعٰجِزِيْنَ اُولٰۤئِكَ اَصْحٰبُ الْجَحِيْمِ۔ "اور جو ہماری آیتوں کو نیچا دکھانے کی دوڑ دھوپ میں لگے رہے، وہی دوزخ والے ہیں۔" وہ عذاب جلدی مانگتے تھے، تو جواب ملا کہ عذاب لانا رسول کا کام نہیں؛ ان کا کام خبردار کرنا ہے، اور خبردار کرنا خود رحمت ہے۔ اہلِ معرفت کہتے ہیں کہ "نَذِيْرٌ مُّبِيْنٌ" میں انذار اور بیان ایک ہو گئے: جو خطرہ کھول کر دکھا دے، وہ دراصل راستہ دکھا رہا ہے۔ "رِزْقٌ كَرِيْمٌ" پر ٹھہریے: یہ روزی اسمِ کریم کی ہے، اور شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں کریم وہ ہے جو استحقاق کے پیمانے سے نہیں، اپنے جود سے دیتا ہے۔ مغفرت پہلے آئی اور روزی بعد میں: پہلے پردہ ڈالا، پھر نوازا۔ اور تیسری آیت کا فعل دیکھیے: "سَعَوْا"، انہوں نے خود دوڑ دھوپ کی؛ "مُعٰجِزِيْنَ"، اس گمان میں کہ حق کو عاجز کر دیں گے۔ یہی اصول ہے: دوزخ کی راہ پر بندہ دھکیلا نہیں جاتا، خود دوڑتا ہے؛ اور اس کا انجام اسی دوڑ کی صورت ہے جو اس نے چنی۔

اب وہ آیت جس کی تفسیر میں سب سے زیادہ احتیاط درکار ہے: وَمَاۤ اَرْسَلْنَا مِنْ قَبْلِكَ مِنْ رَّسُوْلٍ وَّلَا نَبِيٍّ اِلَّاۤ اِذَا تَمَنّٰۤى اَلْقَى الشَّيْطٰنُ فِيْۤ اُمْنِيَّتِهٖ ۚ فَيَنْسَخُ اللّٰهُ مَا يُلْقِي الشَّيْطٰنُ ثُمَّ يُحْكِمُ اللّٰهُ اٰيٰتِهٖ ۭ وَاللّٰهُ عَلِيْمٌ حَكِيْمٌ۔ "اور ہم نے آپ سے پہلے کوئی رسول اور کوئی نبی ایسا نہیں بھیجا کہ جب اس نے کوئی آرزو کی (یا تلاوت کی) تو شیطان نے اس کی آرزو (یا تلاوت) میں وسوسے نہ ڈالے ہوں؛ پھر اللہ شیطان کے ڈالے ہوئے کو مٹا دیتا ہے، پھر اپنی آیتوں کو پختہ کر دیتا ہے، اور اللہ جاننے والا، حکمت والا ہے۔" پہلے ایک بات صاف کہہ دی جائے: اس آیت کے ساتھ جو قصہ جوڑا گیا ہے، جسے "غرانیق کا قصہ" کہا جاتا ہے، امت کے بڑے محققین، جیسے قاضی عیاض اور امام فخرالدین رازی رحمہما اللہ اور ان کے بعد کے بہت سے اہلِ تحقیق نے اسے بے اصل اور حضورِ اکرم صلی اللہ علیہ وسلم کی عصمت کے منافی قرار دے کر رد کیا ہے؛ اس لیے یہاں اسے بیان نہیں کیا جاتا۔ وحی کی ادائیگی میں نبی معصوم ہے، اور قرآن خود گواہ ہے: وَمَا يَنْطِقُ عَنِ الْهَوٰى ۭ اِنْ هُوَ اِلَّا وَحْيٌ يُّوْحٰى۔ آیت کا صحیح مفہوم یہ ہے کہ "تَمَنّٰى" کے دو معنی ہیں: آرزو کرنا اور پڑھنا۔ پہلے معنی میں: ہر نبی کی آرزو اپنی قوم کی ہدایت تھی، اور شیطان لوگوں کے دلوں میں شبہے ڈال کر اس آرزو کی راہ میں رکاوٹیں کھڑی کرتا رہا۔ دوسرے معنی میں: نبی نے وحی تلاوت کی تو شیطان نے سننے والوں کے ذہنوں میں اس کے بارے میں غلط معنی اور شکوک ڈالے۔ دونوں صورتوں میں وسوسہ سننے والوں کے ذہن میں ہے، وحی میں نہیں؛ اور حق ان وسوسوں کو مٹا کر اپنی آیات کو محکم کر دیتا ہے۔ یہی وہ حفاظت ہے جس کا وعدہ سورۂ الحجر میں ہوا تھا: اِنَّا نَحْنُ نَزَّلْنَا الذِّكْرَ وَاِنَّا لَهٗ لَحٰفِظُوْنَ۔

ظاہری مفہوم طے ہو جانے کے بعد اہلِ معرفت اس آیت کی ترتیب پر غور کرتے ہیں، اور یہ غور ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ ترتیب تین قدموں کی ہے: "اَلْقَى"، پھر "فَيَنْسَخُ"، پھر "يُحْكِمُ"۔ شیطان کا کام صرف "اِلقاء" ہے، یعنی باہر سے کچھ پھینک دینا؛ اس کے پاس نہ جڑ ہے نہ بقا۔ "فَيَنْسَخُ" میں "فَ" فوراً کا مفہوم رکھتی ہے: مٹانا دیر سے نہیں آتا۔ اور "ثُمَّ يُحْكِمُ" بتاتا ہے کہ مٹانے کے بعد آیت پہلے سے زیادہ مضبوط ہو کر کھڑی ہوتی ہے؛ گویا وسوسے کا آنا بھی آیات کے احکام کا ایک ذریعہ بن گیا۔ شیخِ اکبر کے مکتب میں حق ہی باقی ہے اور باطل وہ ہے جس کا اپنا کوئی وجود نہیں؛ اسی لیے شیطان کا ڈالا ہوا ٹک نہیں سکتا، اور محکم وہی رہتا ہے جو اسمِ حکیم سے صادر ہو۔ اب ایک اشارہ سالک کے حق میں، اور یہ صاف رہے کہ انبیا علیہم السلام کی عصمت اس سے بلند ہے: طالب کا دل جب حق کی آرزو میں اٹھتا ہے تو اس کی تمنا کے گرد وسوسوں کا شور اٹھتا ہے۔ اہلِ ذوق کہتے ہیں کہ یہ شور آرزو کی سچائی کی علامت ہے، رد کی نہیں؛ جو سچ کی طرف اٹھے، اسی کے راستے میں رکاوٹ ڈالی جاتی ہے۔ سالک کا کام ذکر اور حضوری پر قائم رہنا ہے؛ وسوسہ مٹانا اور اپنی آیت کو دل میں جمانا حق کا کام ہے، اور "عَلِيْمٌ حَكِيْمٌ" کی ضمانت اس کے ساتھ ہے۔

تو پھر یہ وسوسے آنے کیوں دیے جاتے ہیں؟ اگلی آیت جواب دیتی ہے: لِّيَجْعَلَ مَا يُلْقِي الشَّيْطٰنُ فِتْنَةً لِّلَّذِيْنَ فِيْ قُلُوْبِهِمْ مَّرَضٌ وَّالْقَاسِيَةِ قُلُوْبُهُمْ ۭ وَاِنَّ الظّٰلِمِيْنَ لَفِيْ شِقَاقٍۢ بَعِيْدٍ۔ "تاکہ شیطان کے ڈالے ہوئے کو ان لوگوں کے لیے آزمائش بنا دے جن کے دلوں میں روگ ہے اور جن کے دل سخت ہیں؛ اور بے شک یہ ظالم دور کی مخالفت میں پڑے ہیں۔" یہ آیت اس رکوع کا اصول ہے، اور اسے اسی طرح پڑھنا چاہیے: آزمائش وہی ہے جو دل میں پہلے سے موجود حال کے مطابق ہو۔ وسوسہ ایک ہی تھا؛ جس کے دل میں روگ پہلے سے تھا، اس نے اسی روگ کو غذا دی، اور جس کا دل پہلے سے پتھر تھا، اس پر بات پھسل گئی۔ "مَرَضٌ" اور "الْقَاسِيَة" دونوں دل کی صفتیں ہیں جو آزمائش سے پہلے موجود تھیں؛ آزمائش نے انہیں پیدا نہیں کیا، صرف ظاہر کیا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کی روح ہے: حکم محکوم کے مطابق، اور ہر شے پر وہی ظاہر ہوتا ہے جو اس کی عین میں ہے۔ حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت کا طالب ہو اسے ہرگز گمراہ نہیں کرتا؛ آزمائش ایک آئینہ ہے، اور آئینے میں وہی چہرہ دکھائی دیتا ہے جو سامنے آئے۔ "شِقَاقٍۢ بَعِيْدٍ" میں فاصلہ ان کا اپنا طے کیا ہوا ہے: شقاق یعنی ایک طرف ہو جانا، اور جو ایک طرف ہٹتا رہے، وہ دور ہی ہوتا جاتا ہے۔

اور اسی ایک واقعے کا دوسرا رخ: وَّلِيَعْلَمَ الَّذِيْنَ اُوْتُوا الْعِلْمَ اَنَّهُ الْحَقُّ مِنْ رَّبِّكَ فَيُؤْمِنُوْا بِهٖ فَتُخْبِتَ لَهٗ قُلُوْبُهُمْ ۭ وَاِنَّ اللّٰهَ لَهَادِ الَّذِيْنَ اٰمَنُوْۤا اِلٰى صِرَاطٍ مُّسْتَقِيْمٍ۔ "اور تاکہ جنہیں علم دیا گیا ہے وہ جان لیں کہ یہ آپ کے رب کی طرف سے حق ہے، پھر وہ اس پر ایمان لائیں اور ان کے دل اس کے آگے جھک جائیں؛ اور بے شک اللہ ایمان والوں کو سیدھی راہ کی ہدایت دینے والا ہے۔" وہی وسوسہ جو بیمار دل کے لیے فتنہ بنا، اہلِ علم کے لیے یقین کا سبب بن گیا؛ انہوں نے دیکھا کہ وسوسہ مٹا اور آیت قائم رہی، تو جان لیا کہ یہ حق ہے۔ ایک بارش، دو زمینیں؛ پانی ایک ہے، پیداوار زمین کی اپنی ہے۔ سورۂ بنی اسرائیل میں یہی بات قرآن کے بارے میں کہی گئی تھی: وَنُنَزِّلُ مِنَ الْقُرْاٰنِ مَا هُوَ شِفَآءٌ وَّرَحْمَةٌ لِّلْمُؤْمِنِيْنَ ۙ وَلَا يَزِيْدُ الظّٰلِمِيْنَ اِلَّا خَسَارًا۔ ترتیب دیکھیے: علم، پھر ایمان، پھر اِخبات۔ اہلِ معرفت کہتے ہیں کہ یہ علم الیقین سے عین الیقین اور پھر حق الیقین کی طرف سفر ہے: پہلے جاننا، پھر ماننا، پھر دل کا اس کے آگے بچھ جانا۔ "فَتُخْبِتَ" اسی "مُخْبِتِيْن" کا فعل ہے جنہیں پانچویں رکوع (آیت ۳۴) میں خوشخبری ملی تھی؛ خبت نرم اور پست زمین کو کہتے ہیں، جو پانی کو جذب کرتی ہے۔ اور "لَهَادِ الَّذِيْنَ اٰمَنُوْا" اسی اصول کی مثبت شکل ہے: ہدایت اس کو ملتی ہے جو ایمان کی طرف رخ کرے، اور حق اس کا ہاتھ تھام کر سیدھی راہ تک لے جاتا ہے۔

رکوع کا آخری حصہ انجام کی طرف لے جاتا ہے: وَلَا يَزَالُ الَّذِيْنَ كَفَرُوْا فِيْ مِرْيَةٍ مِّنْهُ حَتّٰى تَاْتِيَهُمُ السَّاعَةُ بَغْتَةً اَوْ يَاْتِيَهُمْ عَذَابُ يَوْمٍ عَقِيْمٍ۔ "اور کافر اس کی طرف سے برابر شک میں رہیں گے یہاں تک کہ ان پر اچانک قیامت آ جائے، یا ایک بانجھ دن کا عذاب آ پہنچے۔" "مِرْيَة" وہ شک ہے جس میں بحث اور کھینچ تان ہو؛ وہ تحقیق کا سوال نہیں جو ہدایت کی طرف لے جائے، بلکہ وہ اٹکاؤ ہے جو ماننا ہی نہ چاہے۔ "يَوْمٍ عَقِيْمٍ" کو مفسرین نے قیامت کا دن یا کسی فیصلہ کن پکڑ کا دن کہا ہے؛ بانجھ اس لیے کہ اس کے بعد کوئی کل نہیں جنما جاتا، نہ کوئی اور مہلت۔ اہلِ ذوق کہتے ہیں کہ ڈھیل کے دن حاملہ تھے، ان سے توبہ کی صبح پیدا ہو سکتی تھی؛ وہ دن بانجھ ہے۔ پھر اعلان: اَلْمُلْكُ يَوْمَئِذٍ لِّلّٰهِ ۭ يَحْكُمُ بَيْنَهُمْ ۭ فَالَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ فِيْ جَنّٰتِ النَّعِيْمِ۔ "اس دن بادشاہی اللہ ہی کی ہوگی، وہ ان کے درمیان فیصلہ کرے گا؛ تو جو ایمان لائے اور نیک عمل کیے، وہ نعمت کے باغوں میں ہوں گے۔" وَالَّذِيْنَ كَفَرُوْا وَكَذَّبُوْا بِاٰيٰتِنَا فَاُولٰۤئِكَ لَهُمْ عَذَابٌ مُّهِيْنٌ۔ "اور جنہوں نے کفر کیا اور ہماری آیتوں کو جھٹلایا، ان کے لیے ذلت کا عذاب ہے۔" بادشاہی تو آج بھی اسی کی ہے؛ "يَوْمَئِذٍ" کی قید اس لیے ہے کہ اس دن مجازی ملکیت کا پردہ اٹھ جائے گا۔ شیخِ اکبر کے نزدیک بندے کے ہاتھ میں جو کچھ ہے، وہ عاریتاً ہے؛ عارف آج ہی اسے لوٹا دیتا ہے، یہی فقر ہے، اور اس کے لیے وہ دن کوئی نیا انکشاف نہیں۔ اور انجام کے لفظ بھی دیکھیے: "عَذَابٌ مُّهِيْنٌ"، یعنی ذلت کا عذاب، ان کے لیے جنہوں نے آیات کو اپنے تکبر کے آگے ہیچ جانا؛ تکبر کی صورت ذلت میں ظاہر ہوئی، اور ہر عمل نے اپنی ہی شکل پائی۔
