یہ رکوع ایک سوال سے کھلتا ہے جس نے ہر زمانے کے ذہن کو بے چین رکھا: وَيَسْئَلُوْنَكَ عَنِ الرُّوْحِ قُلِ الرُّوْحُ مِنْ اَمْرِ رَبِّيْ وَمَآ اُوْتِيْتُمْ مِّنَ الْعِلْمِ اِلَّا قَلِيْلًا۔ "اور وہ تم سے روح کے بارے میں پوچھتے ہیں۔ کہہ دو: روح میرے رب کے امر سے ہے، اور تمہیں علم میں سے تھوڑا ہی دیا گیا ہے۔" روایات کے مطابق یہ سوال آزمائش کے طور پر اٹھایا گیا تھا۔ مفسرین نے روح کے معنی میں کئی اقوال نقل کیے: وہ روح جس سے بدن زندہ ہے، یا حضرت جبرائیل علیہ السلام، یا خود قرآن؛ اور اکثر نے پہلے قول کو ترجیح دی۔ جواب پر غور کیجیے: روح کی حقیقت نہیں کھولی گئی، اس کا سرچشمہ بتایا گیا۔ قرآن نے دوسری جگہ دو دائرے الگ الگ نام لے کر ذکر کیے ہیں: اَلَا لَهُ الْخَلْقُ وَالْاَمْرُ۔ اس مکتب میں عالمِ خلق وہ ہے جو مقدار، مادے اور تدریج سے بنتا ہے، اور عالمِ امر وہ ہے جو مقدار کے بغیر "کن" سے وجود پاتا ہے۔ روح کو "امرِ رب" سے بتا کر گویا یہ کہا گیا کہ یہ ناپ تول کی دنیا کی چیز نہیں، اس لیے ناپ تول کے آلات اس تک نہیں پہنچتے۔ سورۂ الحجر میں حضرت آدم علیہ السلام کے بارے میں فرمایا گیا تھا: وَنَفَخْتُ فِيْهِ مِنْ رُّوْحِيْ؛ مٹی کا پیکر خلق سے تھا اور جو پھونکا گیا وہ امر سے۔ انسان انہی دونوں کا ملاپ ہے، اسی لیے وہ برزخ ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب نے اس باب میں اتنا ہی کہا جتنا نص نے کھولا: روح کی نسبت بتائی، اس کی کنہ کا دعویٰ نہیں کیا۔ اور یہی اس آیت کا پہلا سبق ہے۔

اس جواب میں ادب کا وہ درس ہے جو ہر سالک کو چاہیے۔ جس ذاتِ گرامی کو سب سے بڑھ کر علم عطا ہوا، یعنی حضورِ اکرم صلی اللہ علیہ وسلم، انہیں حکم ہوا کہ کہیں: یہ میرے رب کے امر سے ہے۔ جب سب سے بڑے عالم کی زبان پر یہ ادب ہے تو کسی اور کے لیے روح کی کنہ کا دعویٰ کیسے روا ہو؟ اہلِ معرفت کہتے ہیں کہ جس نے سمجھا کہ اس نے روح کو گھیر لیا، اس نے دراصل روح کا ایک تصور گھیرا ہے، روح کو نہیں۔ "تھوڑا علم" کا مطلب علم کی تحقیر نہیں، اس کی حد کی پہچان ہے۔ شیخِ اکبر کے نزدیک علم کی انتہا حیرت ہے: عارف جتنا بڑھتا ہے اتنا ہی جانتا ہے کہ وہ نہیں جانتا، اور یہ حیرت جہالت نہیں بلکہ معرفت کا اونچا زینہ ہے۔ اہلِ تصوف میں ایک مشہور اثر ہے: "جس نے اپنے آپ کو پہچانا اس نے اپنے رب کو پہچانا"۔ اس آیت کی روشنی میں اہلِ ذوق اس کا ایک رخ یہ بتاتے ہیں کہ جب انسان اپنی ہی روح کی کنہ تک نہیں پہنچتا تو اسے اپنے عجز کی پہچان ہوتی ہے، اور یہی عجز رب کی تنزیہ کی پہچان ہے: جو اپنے قریب ترین جوہر کو نہیں گھیر سکتا، وہ رب کو کیسے گھیرے گا۔ ہمیں روح کا جو حصہ ملا ہے وہ اس کی کنہ کا علم نہیں، اس کا ذوق ہے: ذکر میں اس کی بیداری، سجدے میں اس کا سکون، قرب میں اس کی خوشبو۔ یہی ادبِ اسرار ہے، اور اسی لیے یہاں ہم بھی وہیں رکتے ہیں جہاں نص رکی ہے؛ روح کے بارے میں وہی کہتے ہیں جو کہا گیا، اور اس کی حقیقت اس کے رب کے سپرد کرتے ہیں۔

روح کے فوراً بعد وحی کا ذکر آیا، اور دونوں کا سرچشمہ ایک ہے: وَلَئِنْ شِئْنَا لَنَذْهَبَنَّ بِالَّذِيْۤ اَوْحَيْنَآ اِلَيْكَ ثُمَّ لَا تَجِدُ لَكَ بِهٖ عَلَيْنَا وَكِيْلًا۔ "اور اگر ہم چاہیں تو جو کچھ ہم نے تمہاری طرف وحی کیا ہے اسے لے جائیں، پھر تم اس کے لیے ہمارے مقابل کوئی کارساز نہ پاؤ۔" اِلَّا رَحْمَةً مِّنْ رَّبِّكَ اِنَّ فَضْلَهٗ كَانَ عَلَيْكَ كَبِيْرًا۔ "مگر یہ تمہارے رب کی رحمت ہے؛ بے شک تم پر اس کا فضل بہت بڑا ہے۔" خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے مگر درس امت کے لیے ہے: جو علم ملا ہے وہ ملکیت نہیں، امانت ہے۔ مفسرین نے لکھا کہ یہاں یہ بتانا مقصود ہے کہ وحی سراسر عطا ہے اور اس کی بقا بھی عطا ہے۔ اہلِ معرفت اس میں تجددِ امثال کا اشارہ دیکھتے ہیں: شیخِ اکبر کے نزدیک ہر شے ہر آن نئے سرے سے وجود پاتی ہے، تو دل میں قرآن کی موجودگی بھی ہر لمحے ایک نئی عطا ہے، ایک بار کی دی ہوئی چیز نہیں جو بندے کے قبضے میں آ گئی ہو۔ "لئن شئنا" میں امکان باقی رکھا گیا مگر فیصلہ رحمت کا کیا گیا؛ یہی سبقتِ رحمت ہے۔ فقر کی حقیقت بھی یہیں کھلتی ہے: بندہ اپنے علم، اپنے حال اور اپنی یاد تک کا مالک نہیں؛ جو کچھ ہے "من ربک" ہے۔ اور جو آیت امکان کے ذکر سے شروع ہوئی تھی، وہ حضورِ اکرم صلی اللہ علیہ وسلم کے حق میں فضلِ کبیر کے اعلان پر ختم ہوئی۔ اسی میں ادب کا راز ہے: عطا کو یاد رکھا جائے اور اپنا دعویٰ بھلا دیا جائے۔

پھر وہ چیلنج جو قیامت تک کھلا ہے: قُلْ لَّئِنِ اجْتَمَعَتِ الْاِنْسُ وَالْجِنُّ عَلٰۤى اَنْ يَّاْتُوْا بِمِثْلِ هٰذَا الْقُرْاٰنِ لَا يَاْتُوْنَ بِمِثْلِهٖ وَلَوْ كَانَ بَعْضُهُمْ لِبَعْضٍ ظَهِيْرًا۔ "کہہ دو: اگر انسان اور جن سب اس پر جمع ہو جائیں کہ اس قرآن جیسا لے آئیں تو اس جیسا نہ لا سکیں گے، چاہے وہ ایک دوسرے کے مددگار بن جائیں۔" ظاہر میں یہ قرآن کے اعجاز کی دلیل ہے: اس کی فصاحت، اس کی خبریں اور اس کی تاثیر۔ اس مکتب میں ایک اور پہلو بھی ہے: انسان اور جن مل کر مخلوق کے سارے کلام کی نمائندگی کرتے ہیں، اور قرآن کلامِ حق ہے۔ مخلوق کا کلام خالق کے کلام کا مثل کیسے لائے، جبکہ اس کی شان یہ ہے: لَيْسَ كَمِثْلِهٖ شَيْءٌ۔ جس کا متکلم بے مثل ہو، اس کے کلام کا مثل بھی ممکن نہیں۔ پھر فرمایا: وَلَقَدْ صَرَّفْنَا لِلنَّاسِ فِيْ هٰذَا الْقُرْاٰنِ مِنْ كُلِّ مَثَلٍ فَاَبٰۤى اَكْثَرُ النَّاسِ اِلَّا كُفُوْرًا۔ "اور ہم نے اس قرآن میں لوگوں کے لیے ہر طرح کی مثال پھیر پھیر کر بیان کی، پھر بھی اکثر لوگ انکار ہی پر اڑے رہے۔" تصریف کا مطلب ایک بات کو مختلف رخوں سے پیش کرنا ہے۔ شیخِ اکبر کے نزدیک قرآن ہر پڑھنے والے کی استعداد کے مطابق نیا رخ دکھاتا ہے، اس لیے ہر مثال کسی نہ کسی دل کے دروازے کی کنجی ہے۔ اور جب سب کنجیاں دے دی گئیں تو "ابیٰ" کا لفظ آیا: انکار ان کا اپنا فعل ہے، ان کی اپنی ضد۔ یہاں وہی اصول سادہ صورت میں سامنے ہے جس پر یہ پورا سلسلہ قائم ہے: حق نے راہ کھولنے میں کوئی کمی نہ کی، اور جس نے منہ موڑا اس کا منہ موڑنا ہی اس کی گمراہی ٹھہرا۔

اب ان مطالبات کی فہرست آتی ہے جو سردارانِ قریش نے رکھے: وَقَالُوْا لَنْ نُّؤْمِنَ لَكَ حَتّٰى تَفْجُرَ لَنَا مِنَ الْاَرْضِ يَنْۢبُوْعًا۔ "اور انہوں نے کہا: ہم تمہاری بات نہیں مانیں گے جب تک تم ہمارے لیے زمین سے ایک چشمہ جاری نہ کر دو۔" اَوْ تَكُوْنَ لَكَ جَنَّةٌ مِّنْ نَّخِيْلٍ وَّعِنَبٍ فَتُفَجِّرَ الْاَنْهٰرَ خِلٰلَهَا تَفْجِيْرًا۔ "یا تمہارے لیے کھجوروں اور انگوروں کا باغ ہو اور تم اس کے بیچ نہریں بہا دو۔" اَوْ تُسْقِطَ السَّمَآءَ كَمَا زَعَمْتَ عَلَيْنَا كِسَفًا اَوْ تَاْتِيَ بِاللّٰهِ وَالْمَلٰۤئِكَةِ قَبِيْلًا۔ "یا جیسا تم کہتے ہو، آسمان ٹکڑے ٹکڑے کر کے ہم پر گرا دو، یا اللہ اور فرشتوں کو ہمارے سامنے لا کھڑا کرو۔" ابھی کہا گیا تھا کہ ہر مثال پھیر پھیر کر دی گئی، اور جواب میں انہوں نے مکہ کی خشک زمین سے چشمہ مانگا، جبکہ وحی کا چشمہ ان کے درمیان بہہ رہا تھا۔ اہلِ ذوق کہتے ہیں کہ نفس کا مطالبہ ہمیشہ حسی ہوتا ہے: چشمہ جو آنکھ دیکھے، باغ جس کا پھل زبان چکھے؛ دل کا چشمہ آنکھ کو دکھائی نہیں دیتا۔ آسمان گرانے کا مطالبہ عذاب کو دلیل بنانا ہے، اور یہ وہی جلد بازی ہے جس کا ذکر اس سورت کے شروع میں آ چکا۔ "قبیلا" کا مطالبہ سب سے گہرا پردہ ہے: وہ اللہ کو یوں سامنے دیکھنا چاہتے تھے جیسے کسی قبیلے کو دیکھا جاتا ہے، یعنی تنزیہ کے بغیر خالص تشبیہ۔ شیخِ اکبر کے نزدیک معرفت تنزیہ اور تشبیہ کے جمع میں ہے، نہ صرف تنزیہ جو اسے عالم سے کاٹ دے، نہ صرف تشبیہ جو اسے عالم کی ایک چیز بنا دے۔ سالک کے لیے بھی یہاں تنبیہ ہے: نفس کبھی ایمان کی شرط کرامت کو بنا لیتا ہے، حالانکہ اہلِ تصوف میں مشہور قول ہے کہ استقامت ہزار کرامتوں سے بڑھ کر ہے۔

مطالبات کی آخری کڑی اور اس کا جواب ایک ہی آیت میں ہیں: اَوْ يَكُوْنَ لَكَ بَيْتٌ مِّنْ زُخْرُفٍ اَوْ تَرْقٰى فِي السَّمَآءِ وَلَنْ نُّؤْمِنَ لِرُقِيِّكَ حَتّٰى تُنَزِّلَ عَلَيْنَا كِتٰبًا نَّقْرَؤُهٗ قُلْ سُبْحٰنَ رَبِّيْ هَلْ كُنْتُ اِلَّا بَشَرًا رَّسُوْلًا۔ "یا تمہارے لیے سونے کا گھر ہو، یا تم آسمان پر چڑھ جاؤ، اور ہم تمہارے چڑھنے کو بھی نہ مانیں گے جب تک تم ہم پر ایک کتاب نہ اتار لاؤ جسے ہم پڑھیں۔ کہہ دو: پاک ہے میرا رب، میں تو بس ایک بشر ہوں جو رسول بنا کر بھیجا گیا۔" غور کیجیے: یہ سورت اسی عروج سے کھلی تھی جس کا یہ لوگ مطالبہ کر رہے تھے۔ حضورِ اکرم صلی اللہ علیہ وسلم آسمانوں پر گئے، مگر ان کے مطالبے پر نہیں، رب کے بلاوے پر، اور اس لقب کے ساتھ: اَسْرٰى بِعَبْدِهٖ لَيْلًا۔ اور یہ بھی دیکھیے کہ چڑھنے کے بعد بھی وہ کتاب کی شرط رکھ رہے تھے؛ جو مطالبہ طلب سے نہ ہو اس کی کوئی حد نہیں ہوتی۔ جواب "سبحان" سے شروع ہوا، جیسے سورت "سبحان" سے شروع ہوئی تھی: یہ تنزیہ ہے، یعنی میرا رب تمہاری شرطوں کا پابند نہیں؛ نشانیاں اس کے اختیار میں ہیں، میرے نہیں۔ پھر دو لفظ: "بشر" اور "رسول"۔ اس مکتب میں انسانِ کامل وہی ہے جو خلق اور امر کے درمیان برزخ ہو؛ بشریت اسے مخلوق سے جوڑتی ہے اور رسالت اسے امرِ رب کا امین بناتی ہے۔ یہ بشریت عام بشریت نہیں، حضورِ اکرم صلی اللہ علیہ وسلم نے خود فرمایا کہ تم میں کون میری مانند ہے؛ مگر آپ صلی اللہ علیہ وسلم نے اپنے لیے یہی عاجزانہ لفظ چنا، کیونکہ عبودیت ہر مقام سے اونچی ہے۔ اور سونے کے گھر کے مقابلے میں آپ صلی اللہ علیہ وسلم نے فقر کو چنا، کہ زخرف ظاہر کی زینت ہے اور رسالت باطن کا نور۔
