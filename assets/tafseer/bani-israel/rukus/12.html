یہ سورت بنی اسرائیل کے نام سے بھی پہچانی جاتی ہے، اور جس طرح اس کا آغاز حضرت موسیٰ علیہ السلام کی کتاب کے ذکر سے ہوا تھا، انجام پر بھی وہی داستان لوٹتی ہے: وَلَقَدْ اٰتَيْنَا مُوْسٰى تِسْعَ اٰيٰتٍۭ بَيِّنٰتٍ فَسْئَلْ بَنِيْۤ اِسْرَآءِيْلَ اِذْ جَآءَهُمْ فَقَالَ لَهٗ فِرْعَوْنُ اِنِّيْ لَاَظُنُّكَ يٰمُوْسٰى مَسْحُوْرًا۔ "اور ہم نے حضرت موسیٰ علیہ السلام کو نو کھلی نشانیاں دیں، تو بنی اسرائیل سے پوچھ لیجیے جب وہ ان کے پاس آئے تو فرعون نے ان سے کہا: اے موسیٰ! میں تو تمہیں جادو زدہ سمجھتا ہوں۔" نشانیاں جتنی بھی روشن ہوں، سرکش نفس انہیں اپنی ہی بیماری کا نام دے دیتا ہے؛ جو خود اقتدار کے سحر میں گرفتار تھا اس نے صاحبِ آیات کو مسحور کہا۔ حضرت موسیٰ علیہ السلام کا جواب آیا: قَالَ لَقَدْ عَلِمْتَ مَاۤ اَنْزَلَ هٰۤؤُلَآءِ اِلَّا رَبُّ السَّمٰوٰتِ وَالْاَرْضِ بَصَآئِرَ ۚ وَاِنِّيْ لَاَظُنُّكَ يٰفِرْعَوْنُ مَثْبُوْرًا۔ "انہوں نے کہا: تو خوب جان چکا ہے کہ ان نشانیوں کو آسمانوں اور زمین کے رب ہی نے بصیرت کا سامان بنا کر اتارا ہے، اور اے فرعون! میں تو تجھے ہلاک ہونے والا سمجھتا ہوں۔" "لقد علمت" اس رکوع کا نازک نکتہ ہے: فرعون جانتا تھا۔ اس کی گمراہی روشنی کی کمی سے نہیں، روشنی سے منہ پھیرنے سے تھی؛ یہی اس سورت کا اصول ہے کہ ضلالت بندے کا اپنا اعراض ہے اور حق اسی رخ کو وجود دیتا ہے جو اس نے چنا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ایک علم وہ ہے جو ذہن میں ٹھہرتا ہے اور ایک وہ جو پورے وجود کو جھکا دیتا ہے؛ فرعون کے پاس پہلا تھا۔ اور "بصائر" کا لفظ بتاتا ہے کہ نشانیاں صرف دلیلیں نہیں، آنکھیں ہیں؛ جو انہیں دل کی آنکھ بنا لے وہ دیکھتا ہے، جو انہیں محض بحث کا موضوع بنائے وہ اندھا رہتا ہے۔

جاننے کے باوجود انکار کا اگلا قدم جبر تھا: فَاَرَادَ اَنْ يَّسْتَفِزَّهُمْ مِّنَ الْاَرْضِ فَاَغْرَقْنٰهُ وَمَنْ مَّعَهٗ جَمِيْعًا۔ "پھر اس نے چاہا کہ انہیں اس سرزمین سے اکھاڑ پھینکے، تو ہم نے اسے اور اس کے سب ساتھیوں کو غرق کر دیا۔" "یستفزھم" وہی لفظ ہے جو اسی سورت میں ابلیس سے کہا گیا تھا: وَاسْتَفْزِزْ مَنِ اسْتَطَعْتَ مِنْهُمْ بِصَوْتِكَ؛ فرعونی جبر ابلیسی طریقے کی زمینی صورت ہے: ہلا دینا اور قدم اکھاڑ دینا۔ مگر جو دوسروں کو زمین سے اکھاڑنا چاہتا تھا خود پانی میں اکھڑ گیا، اور وہی دریا جو حضرت موسیٰ علیہ السلام کے لیے راستہ بنا فرعون کے لیے قبر بنا: ایک پانی، دو انجام، جیسے سورۂ رعد میں ایک ہی پانی سے مختلف پھل۔ پھر فرمایا: وَقُلْنَا مِنْۢ بَعْدِهٖ لِبَنِيْۤ اِسْرَآءِيْلَ اسْكُنُوا الْاَرْضَ فَاِذَا جَآءَ وَعْدُ الْاٰخِرَةِ جِئْنَا بِكُمْ لَفِيْفًا۔ "اور اس کے بعد ہم نے بنی اسرائیل سے کہا: اس سرزمین میں بسو، پھر جب آخرت کا وعدہ آ پہنچے گا تو ہم تم سب کو سمیٹ کر لے آئیں گے۔" اہلِ تاویل یہاں ایک تمثیل بیان کرتے ہیں، اور یہ صرف تمثیل ہے جس میں کسی قوم کی برتری یا تحقیر کا کوئی دعویٰ نہیں: فرعون نفسِ امارہ ہے جو بدن کی سلطنت میں ربوبیت کا دعویٰ کرتا ہے اور دل کی قوتوں کو اس سرزمین سے نکالنا چاہتا ہے؛ جب وہ اپنی ہی خواہشوں کے سمندر میں ڈوب جاتا ہے تو دل کی قوتیں بدن کی زمین میں امن سے بستی ہیں۔ "لفیفا" میں اشارہ ہے کہ جو قوتیں دنیا میں بکھری رہیں، وعدے کے دن سب ایک جگہ جمع ہوں گی: تفرقہ کے بعد جمع۔ ظاہر معنی اپنی جگہ قائم ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

قصے سے کتاب کی طرف رخ ہوا: وَبِالْحَقِّ اَنْزَلْنٰهُ وَبِالْحَقِّ نَزَلَ ۭ وَمَاۤ اَرْسَلْنٰكَ اِلَّا مُبَشِّرًا وَّنَذِيْرًا۔ "اور ہم نے اسے حق کے ساتھ اتارا اور وہ حق ہی کے ساتھ اترا، اور ہم نے آپ کو صرف خوش خبری دینے والا اور ڈرانے والا بنا کر بھیجا ہے۔" دو بار "بالحق": اتارتے وقت بھی حق اور پہنچتے وقت بھی حق؛ راستے میں کوئی آمیزش نہیں ہوئی۔ شیخِ اکبر کے نزدیک ایک اصطلاح "الحق المخلوق بہ" ہے: وہ حقیقت جس کے ذریعے سب کچھ پیدا ہوا، جسے نفَسِ رحمانی سے بھی تعبیر کیا جاتا ہے۔ اہلِ معرفت اس آیت میں اشارہ پاتے ہیں کہ قرآن اسی حق سے اترا جس سے کائنات بنی؛ اسی لیے کلامِ منزل اور کتابِ کائنات ایک دوسرے کی تصدیق کرتے ہیں۔ "مبشرا و نذیرا" میں جمال و جلال کا جمع ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم دونوں کے کامل مظہر ہیں۔ پھر نزول کی ترتیب بیان ہوئی: وَقُرْاٰنًا فَرَقْنٰهُ لِتَقْرَاَهٗ عَلَى النَّاسِ عَلٰى مُكْثٍ وَّنَزَّلْنٰهُ تَنْزِيْلًا۔ "اور قرآن کو ہم نے جدا جدا کر کے اتارا تاکہ آپ اسے لوگوں کو ٹھہر ٹھہر کر پڑھ کر سنائیں، اور ہم نے اسے بتدریج نازل کیا۔" اس مکتب میں یہ تدریجِ تجلی کا قانون ہے: حق ہر دل کو اس کی استعداد کے مطابق دیتا ہے، اور دل ایک ساتھ سارا بوجھ نہیں اٹھا سکتا۔ "مکث" قاری کا ادب ہے: آیت پر ٹھہرنا اور اسے دل میں اترنے دینا۔ اور "فرقناہ" میں فرقان کا اشارہ ہے: جو حق کے ہاں جمع ہے وہ بندے کے لیے تفصیل بن کر آتا ہے۔

پھر ایمان کا معاملہ سننے والے کے اختیار پر چھوڑ دیا گیا: قُلْ اٰمِنُوْا بِهٖۤ اَوْ لَا تُؤْمِنُوْا ۭ اِنَّ الَّذِيْنَ اُوْتُوا الْعِلْمَ مِنْ قَبْلِهٖۤ اِذَا يُتْلٰى عَلَيْهِمْ يَخِرُّوْنَ لِلْاَذْقَانِ سُجَّدًا۔ "کہہ دیجیے: تم اس پر ایمان لاؤ یا نہ لاؤ، بے شک جنہیں اس سے پہلے علم دیا گیا جب ان کے سامنے اس کی تلاوت ہوتی ہے تو وہ ٹھوڑیوں کے بل سجدے میں گر پڑتے ہیں۔" "آمنوا بہ او لا تؤمنوا" میں انتخاب کی عزت ہے: حق کو کسی کے ایمان کی حاجت نہیں، وہ چاہتا تو جبر کر سکتا تھا، مگر اس نے دعوت اور اختیار کا راستہ رکھا۔ اب دو علموں کا مقابلہ دیکھیے: فرعون جانتا تھا اور اکڑ گیا، یہ لوگ جانتے ہیں اور سجدے میں گر جاتے ہیں۔ وَيَقُوْلُوْنَ سُبْحٰنَ رَبِّنَاۤ اِنْ كَانَ وَعْدُ رَبِّنَا لَمَفْعُوْلًا۔ "اور کہتے ہیں: پاک ہے ہمارا رب، بے شک ہمارے رب کا وعدہ پورا ہو کر ہی رہنا تھا۔" یہ "سبحان" سورت کے پہلے لفظ کی بازگشت ہے؛ سچے عالم کا پہلا قول تنزیہ ہے۔ وَيَخِرُّوْنَ لِلْاَذْقَانِ يَبْكُوْنَ وَيَزِيْدُهُمْ خُشُوْعًا۔ "اور وہ روتے ہوئے ٹھوڑیوں کے بل گرتے ہیں، اور یہ قرآن ان کی عاجزی اور بڑھا دیتا ہے۔" اہلِ ذوق کہتے ہیں کہ "للاذقان" میں چہرہ، جو انسان کی عزت کا مقام ہے، پورا زمین پر رکھ دیا جاتا ہے: یہ فنا کی صورت ہے۔ سجدہ دو بار آیا: پہلا علم کا سجدہ ہے جس کے ساتھ تسبیح ہے، دوسرا گریہ کا سجدہ ہے جس میں شہود ہے؛ گویا علم الیقین سے عین الیقین کی طرف سفر۔ اور "یزیدھم" بتاتا ہے کہ یہ حال رکتا نہیں، ہر تلاوت میں خشوع بڑھتا ہے۔ ایک ہی کلام کسی کو اکڑ میں بڑھاتا ہے اور کسی کو خشوع میں؛ فرق سننے والے کے رخ کا ہے۔

اب وہ آیت جسے اس مکتب میں اس سورت کا دل کہنا چاہیے: قُلِ ادْعُوا اللّٰهَ اَوِ ادْعُوا الرَّحْمٰنَ ۭ اَيًّا مَّا تَدْعُوْا فَلَهُ الْاَسْمَآءُ الْحُسْنٰى۔ "کہہ دیجیے: اللہ کہہ کر پکارو یا رحمٰن کہہ کر پکارو، جس نام سے بھی پکارو، سب اچھے نام اسی کے ہیں۔" روایات کا مفہوم یہ ہے کہ مشرکین نے حضورِ اکرم صلی اللہ علیہ وسلم کو "یا اللہ" اور "یا رحمٰن" پکارتے سنا تو کہا کہ یہ ہمیں دو معبودوں سے روکتے ہیں اور خود دو کو پکارتے ہیں؛ جواب آیا کہ نام بہت ہیں اور مسمّیٰ ایک۔ شیخِ اکبر کے نزدیک "اللہ" اسمِ جامع ہے جو تمام اسماء کو اپنے اندر رکھتا ہے، اور "الرحمٰن" وجودی رحمت کا نام ہے جس سے ہر چیز کو ہستی ملی؛ نفَسِ رحمانی اسی رحمت کا سانس ہے جس میں کائنات کے سب کلمات ظاہر ہوئے۔ ان دو ناموں کا ذکر اس لیے کہ ایک سب ناموں کا جامع ہے اور دوسرا سب مخلوقات تک پہنچنے والی رحمت۔ اس مکتب میں اسماء ذات اور عالم کے درمیان نسبتیں ہیں؛ ہر بندے کا ایک ربِّ خاص ہے، کوئی نام جو اس پر غالب ہے، مگر سب نام اسی ایک کے ہیں۔ "ایاما تدعوا" میں بشارت ہے: بیمار الشافی کو، گنہگار الغفور کو، محتاج الرزاق کو پکارے، ہر نام ایک دروازہ ہے اور ہر دروازہ اسی ایک بارگاہ میں کھلتا ہے۔ جو کسی بھی نام سے سچے دل سے پکارتا ہے لوٹایا نہیں جاتا؛ یہ اصولِ ہدایت کی سب سے روشن صورت ہے کہ طالب کے لیے کوئی دروازہ بند نہیں۔ اور "الحسنیٰ" بتاتا ہے کہ جلال کے نام بھی حسین ہیں، کیونکہ جلال بھی رحمت کی خدمت میں ہے اور رحمت ہر چیز پر سبقت رکھتی ہے۔

اسی آیت کا دوسرا حصہ پکارنے کا ادب سکھاتا ہے: وَلَا تَجْهَرْ بِصَلَاتِكَ وَلَا تُخَافِتْ بِهَا وَابْتَغِ بَيْنَ ذٰلِكَ سَبِيْلًا۔ "اور اپنی نماز نہ بہت بلند آواز سے پڑھیے اور نہ بالکل پست آواز سے، اور ان دونوں کے درمیان راستہ اختیار کیجیے۔" ظاہر میں یہ مکی دور کا ادب ہے: قرأت اتنی بلند نہ ہو کہ منکرین سن کر قرآن کو برا کہیں، اور اتنی دھیمی نہ ہو کہ اصحاب نہ سن سکیں۔ یہ حکم اپنی جگہ ہے، اور اہلِ معرفت اس میں ایک گہری بات بھی دیکھتے ہیں۔ جہر ظہور کی طرف اشارہ ہے اور اخفا بطون کی طرف؛ جہر تشبیہ کی زبان ہے اور اخفا تنزیہ کی۔ صرف تنزیہ حق کو دور کر دیتی ہے اور صرف تشبیہ اسے محدود کر دیتی ہے؛ شیخِ اکبر کے نزدیک کامل معرفت دونوں کا جمع ہے، اور یہی "بین ذلک سبیلا" ہے۔ یہ درمیان کا راستہ برزخ ہے، اور انسانِ کامل خود یہی برزخ ہے جس میں ظاہر و باطن دونوں کا حق ادا ہوتا ہے۔ ذکر کے مراتب میں بھی یہی سبق ہے: سالک نہ دکھاوے کی بلندی اختیار کرے، نہ ایسی خاموشی جس میں دل غافل ہو جائے۔ اور لفظ "سبیلا" ہے، "مقاما" نہیں: درمیان ٹھہرنے کی جگہ نہیں، چلنے کا راستہ ہے، ہر آن نئے توازن کی تلاش۔ اسماءِ حسنیٰ کے فوراً بعد آواز کا ادب اس لیے آیا کہ جس نے ناموں کی وسعت پہچان لی، اس کا ذکر بھی اعتدال کا رنگ لے لیتا ہے۔

اور سورت اس آیت پر ختم ہوتی ہے: وَقُلِ الْحَمْدُ لِلّٰهِ الَّذِيْ لَمْ يَتَّخِذْ وَلَدًا وَّلَمْ يَكُنْ لَّهٗ شَرِيْكٌ فِي الْمُلْكِ وَلَمْ يَكُنْ لَّهٗ وَلِيٌّ مِّنَ الذُّلِّ وَكَبِّرْهُ تَكْبِيْرًا۔ "اور کہیے: سب تعریف اللہ کے لیے ہے جس نے نہ کوئی اولاد بنائی، نہ بادشاہی میں کوئی اس کا شریک ہے، اور نہ کمزوری کی وجہ سے کوئی اس کا حمایتی ہے، اور اس کی بڑائی بیان کیجیے، خوب بڑائی۔" تین نفی ہیں اور ہر ایک ایک وہم کو کاٹتی ہے۔ اولاد نہیں: حق کی کوئی جنس نہیں جس میں کوئی اس کا ہم مثل ہو۔ بادشاہی میں شریک نہیں: افعال کا سرچشمہ ایک ہے، اور جو اسباب نظر آتے ہیں وہ اسی کے افعال کے مظاہر ہیں۔ کمزوری کی وجہ سے کوئی ولی نہیں: اہلِ معرفت یہاں نکتہ بیان کرتے ہیں کہ حق کے اولیاء ہیں، مگر وہ اس کی محبت سے ہیں، اس کی حاجت سے نہیں؛ "من الذل" کی قید نے دوستی کو باقی رکھا اور محتاجی کی نفی کی۔ یوں آیت تنزیہ بھی ہے اور قرب کا دروازہ بھی کھلا رکھتی ہے۔ "الحمد" وہ ثنا ہے جو کمالات کے ظہور پر ہوتی ہے؛ تسبیح کہتی ہے کہ وہ ہر نقص سے پاک ہے، حمد کہتی ہے کہ ہر کمال اسی کا ہے۔ اور "وقل" کا خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے: جو حمد ہر ذرہ اپنی زبانِ حال سے کر رہا ہے، جیسا اسی سورت میں آیا کہ ہر چیز اس کی حمد کے ساتھ تسبیح کرتی ہے، انسانِ کامل اسے زبانِ قال سے ادا کرتا ہے۔

اب سورت کا پورا دائرہ سامنے ہے۔ آغاز ہوا تھا: سُبْحٰنَ الَّذِيْۤ اَسْرٰى بِعَبْدِهٖ لَيْلًا، اور انجام ہے: وَلَمْ يَكُنْ لَّهٗ وَلِيٌّ مِّنَ الذُّلِّ وَكَبِّرْهُ تَكْبِيْرًا۔ پہلے تسبیح، یعنی تنزیہ؛ درمیان میں حمد، یعنی کمالات کا اثبات؛ اور آخر میں تکبیر۔ شیخِ اکبر کے نزدیک تکبیر ان دونوں سے آگے کا مقام ہے: "اللہ اکبر" کا مطلب یہ ہے کہ وہ ہماری ہر تنزیہ اور ہر تشبیہ سے، ہمارے ہر علم سے بڑا ہے؛ یہاں عقل حیرت میں ٹھہرتی ہے، اور یہ حیرت جہل نہیں، معرفت کی انتہا ہے۔ آغاز میں حضورِ اکرم صلی اللہ علیہ وسلم کو "عبد" کہا گیا اور آخر میں "قل" کا خطاب بھی اسی عبد سے ہے: سورت عبودیت سے شروع ہوئی اور تکبیر پر ختم ہوئی، یعنی بلندی کا سفر بندگی ہی کے راستے طے ہوا۔ حضورِ اکرم صلی اللہ علیہ وسلم کا اسراء و معراج حقیقی واقعہ اور آپ صلی اللہ علیہ وسلم کا خاص اعزاز ہے، جو جمہور اہلِ سنت کے نزدیک جسم و روح دونوں کے ساتھ ہوا اور جس میں کوئی شریک نہیں؛ اہلِ معرفت کے نزدیک اولیاء کو اس کی وراثت سے جو روحانی حصہ ملتا ہے وہ کبھی اس کے برابر نہیں، اور اس کا خلاصہ یہی ہے کہ دل تسبیح سے چل کر حمد میں ٹھہرے اور تکبیر میں فنا ہو۔ یوں رات کے سفر کی سورت روشنی پر ختم ہوتی ہے: سبحان سے الحمد اور الحمد سے اللہ اکبر تک، توحید کا مکمل دائرہ۔
