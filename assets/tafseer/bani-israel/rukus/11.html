پچھلے رکوع میں مطالبوں کی پوری فہرست کا جواب سُبْحٰنَ رَبِّيْ هَلْ كُنْتُ اِلَّا بَشَرًا رَّسُوْلًا پر ختم ہوا تھا، اور یہ رکوع اسی گرہ کو کھولتا ہے: وَمَا مَنَعَ النَّاسَ اَنْ يُّؤْمِنُوْۤا اِذْ جَآءَهُمُ الْهُدٰۤى اِلَّاۤ اَنْ قَالُوْۤا اَبَعَثَ اللّٰهُ بَشَرًا رَّسُوْلًا۔ "اور جب لوگوں کے پاس ہدایت آ گئی تو انہیں ایمان لانے سے کسی چیز نے نہیں روکا سوائے اس بات کے کہ وہ کہنے لگے: کیا اللہ نے ایک انسان کو رسول بنا کر بھیجا ہے؟" ظاہر میں یہ منکرین کا اعتراض ہے، اور اہلِ معرفت کے نزدیک یہ ہر زمانے کا حجاب ہے: آدمی نور کو اس کے برتن کی وجہ سے ٹھکرا دیتا ہے۔ اسی لیے جواب میں قانونِ مناسبت بیان ہوا: قُلْ لَّوْ كَانَ فِي الْاَرْضِ مَلٰۤئِكَةٌ يَّمْشُوْنَ مُطْمَئِنِّيْنَ لَنَزَّلْنَا عَلَيْهِمْ مِّنَ السَّمَآءِ مَلَكًا رَّسُوْلًا۔ "کہہ دیجیے: اگر زمین میں فرشتے اطمینان سے چلتے پھرتے ہوتے تو ہم ان پر آسمان سے کسی فرشتے ہی کو رسول بنا کر اتارتے۔" پیغام لانے والا سننے والوں کی جنس سے ہوتا ہے، جیسے سورۂ ابراہیم میں ہر رسول اپنی قوم کی زبان میں بھیجا گیا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں انسان اسمِ جامع کا مظہر ہے؛ اس میں فرشتے کی لطافت بھی ہے اور مٹی کی کثافت بھی، اس لیے بشر رسول ہی وہ برزخ ہے جس سے آسمان کی بات زمین کی سمجھ میں اترتی ہے۔ فرشتہ فرشتوں کا امام بن سکتا ہے، بشر رسول دونوں عالموں کے بیچ پل ہے۔ یوں جسے انہوں نے عیب سمجھا وہی رحمت کی صورت تھی، اور رسول کی بشریت کو حجاب بنانا دراصل اپنی ہی آنکھ کا حجاب تھا۔

جب دلیل ٹھکرا دی جائے تو داعی بحث کو طول نہیں دیتا، معاملہ حق کے سپرد کر دیتا ہے: قُلْ كَفٰى بِاللّٰهِ شَهِيْدًۢا بَيْنِيْ وَبَيْنَكُمْ ۭ اِنَّهٗ كَانَ بِعِبَادِهٖ خَبِيْرًۢا بَصِيْرًا۔ "کہہ دیجیے: میرے اور تمہارے درمیان اللہ گواہ کافی ہے، بے شک وہ اپنے بندوں کی پوری خبر رکھنے والا، سب کچھ دیکھنے والا ہے۔" حضورِ اکرم صلی اللہ علیہ وسلم کا یہ انداز ادبِ دعوت ہے: داعی کا کام پہنچانا ہے، منوانا نہیں۔ اہلِ ذوق کہتے ہیں کہ "شہید" حاضر کو بھی کہتے ہیں؛ جو ہر حال میں حاضر ہو اس کی گواہی کسی واسطے کی محتاج نہیں۔ "بینی و بینکم" کا "بین" وہ فاصلہ ہے جو منکر نے اپنے اور رسول کے درمیان خود کھڑا کیا، اور حق اسی فاصلے میں بھی شاہد ہے۔ پھر دو نام آئے: خبیر اور بصیر۔ شیخِ اکبر کے نزدیک خبیر باطن کی حقیقت جانتا ہے اور بصیر ظاہر کی صورت دیکھتا ہے؛ حق بندے کے عینِ ثابت کو بھی جانتا ہے اور اس کے عمل کو بھی دیکھتا ہے۔ یہی دونوں نام اگلی آیت کی کنجی ہیں: جو جانتا ہے کہ کس دل نے طلب کی اور کس نے منہ پھیرا، وہی ہر ایک کو اس کے رخ کے مطابق دیتا ہے۔ اور "بعبادہ" میں منکرین کو بھی بندہ کہا گیا: رشتۂ عبودیت انکار سے ٹوٹتا نہیں، صرف بندے کی اپنی آنکھ سے اوجھل ہو جاتا ہے۔

اب وہ آیت آتی ہے جو اس رکوع کا مرکز ہے: وَمَنْ يَّهْدِ اللّٰهُ فَهُوَ الْمُهْتَدِ ۚ وَمَنْ يُّضْلِلْ فَلَنْ تَجِدَ لَهُمْ اَوْلِيَآءَ مِنْ دُوْنِهٖ۔ "اور جسے اللہ ہدایت دے وہی راہ پانے والا ہے، اور جسے وہ اس کی اپنی روگردانی کے سبب گمراہی میں رہنے دے تو تم اس کے سوا ان کے لیے کوئی کارساز نہ پاؤ گے۔" اس آیت کو اسی سورت کی پندرھویں آیت کے ساتھ پڑھنا لازم ہے، کیونکہ وہاں اصول صاف لفظوں میں بیان ہو چکا: مَنِ اهْتَدٰى فَاِنَّمَا يَهْتَدِيْ لِنَفْسِهٖ ۚ وَمَنْ ضَلَّ فَاِنَّمَا يَضِلُّ عَلَيْهَا۔ یہی اس باب کا محکم اصول ہے: حق تعالیٰ کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت کا طالب ہو اسے کبھی گمراہ نہیں کرتا۔ جو طلب کرتا ہے پاتا ہے؛ جو منہ پھیرتا ہے، اس کا منہ پھیرنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق ہوتا ہے، فیصلہ اس چیز کی اپنی حقیقت کی پیروی کرتا ہے۔ "فھو المھتد" میں ادب کا سبق بھی ہے: راہ پانے والا ہدایت کو اپنا کمال نہ سمجھے، طلب اس کی تھی اور عطا حق کی۔ "من دونہ" کا راز یہ ہے کہ جس نے اس ایک دروازے سے اعراض کیا، اس کے لیے باقی سب دروازے خالی ہیں۔ اور یاد رہے کہ حق چاہتا تو سب کو ایک ہی راہ پر ڈال دیتا، جیسا سورۂ النحل میں فرمایا: وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ؛ اس نے جبر نہیں چاہا، انتخاب کی عزت چاہی۔

پھر اسی آیت کا دوسرا حصہ اس انتخاب کا انجام دکھاتا ہے: وَنَحْشُرُهُمْ يَوْمَ الْقِيٰمَةِ عَلٰى وُجُوْهِهِمْ عُمْيًا وَّبُكْمًا وَّصُمًّا ۭ مَاْوٰىهُمْ جَهَنَّمُ ۭ كُلَّمَا خَبَتْ زِدْنٰهُمْ سَعِيْرًا۔ "اور ہم قیامت کے دن انہیں ان کے چہروں کے بل اندھے، گونگے اور بہرے اٹھائیں گے؛ ان کا ٹھکانا جہنم ہے، جب کبھی اس کی لپٹ دھیمی ہو گی ہم اسے ان پر اور بھڑکا دیں گے۔" ظاہر اپنی جگہ حق ہے اور اس پر ایمان لازم؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ معرفت کے نزدیک یہ حشر اس صورت کا اظہار ہے جو بندے نے دنیا میں خود اختیار کی۔ جس چہرے کو اس نے ہمیشہ زمین کی طرف جھکائے رکھا وہ اسی رخ پر اٹھا؛ جس آنکھ نے نشانیاں دیکھنے سے، جس زبان نے گواہی سے اور جس کان نے پکار سننے سے انکار کیا، وہ وہاں اسی انکار کی شکل میں ظاہر ہوئے۔ یہ وہی بات ہے جو اسی سورت میں پہلے آ چکی: وَمَنْ كَانَ فِيْ هٰذِهٖۤ اَعْمٰى فَهُوَ فِي الْاٰخِرَةِ اَعْمٰى۔ اس مکتب میں آخرت وہ عالم ہے جہاں معانی صورت اختیار کرتے ہیں، اسے تجسدِ اعمال کہتے ہیں؛ وہاں کوئی اجنبی چیز نہیں دی جاتی، جو چھپا تھا وہ کھل جاتا ہے۔ "کلما خبت" کے بارے میں اہلِ تاویل کہتے ہیں کہ یہ آگ باہر کی سزا بھی ہے اور اندر کی جلن کا ظہور بھی؛ اس کا ایندھن وہ وابستگیاں ہیں جو بندہ ساتھ لے کر آیا۔ یوں آیت ڈراتی بھی ہے اور آج کے دن کی قدر بھی سکھاتی ہے: دل کی آنکھ یہیں بنتی ہے۔

انجام کا سبب بھی ساتھ ہی بتا دیا گیا: ذٰلِكَ جَزَآؤُهُمْ بِاَنَّهُمْ كَفَرُوْا بِاٰيٰتِنَا وَقَالُوْۤا ءَاِذَا كُنَّا عِظَامًا وَّرُفَاتًا ءَاِنَّا لَمَبْعُوْثُوْنَ خَلْقًا جَدِيْدًا۔ "یہ ان کا بدلہ ہے، اس لیے کہ انہوں نے ہماری آیتوں کا انکار کیا اور کہا: کیا جب ہم ہڈیاں اور ریزہ ریزہ ہو جائیں گے تو ہم نئی پیدائش میں اٹھائے جائیں گے؟" "بانھم" کی باء بتاتی ہے کہ بدلہ ان کے اپنے عمل کا نتیجہ ہے، باہر سے لادی ہوئی چیز نہیں۔ پھر جواب: اَوَلَمْ يَرَوْا اَنَّ اللّٰهَ الَّذِيْ خَلَقَ السَّمٰوٰتِ وَالْاَرْضَ قَادِرٌ عَلٰۤى اَنْ يَّخْلُقَ مِثْلَهُمْ وَجَعَلَ لَهُمْ اَجَلًا لَّا رَيْبَ فِيْهِ ۭ فَاَبَى الظّٰلِمُوْنَ اِلَّا كُفُوْرًا۔ "کیا انہوں نے نہیں دیکھا کہ جس اللہ نے آسمان اور زمین پیدا کیے وہ ان جیسوں کو پیدا کرنے پر قادر ہے، اور اس نے ان کے لیے ایک مدت مقرر کر رکھی ہے جس میں کوئی شک نہیں؛ پھر بھی ظالموں نے ناشکری کے سوا ہر بات سے انکار کیا۔" شیخِ اکبر کے نزدیک خلقِ جدید کوئی دور کی بات نہیں؛ عالم ہر آن نیا پیدا ہوتا ہے اور ہر لمحہ اپنی مثل سے بدلتا رہتا ہے، اسے تجددِ امثال کہتے ہیں۔ آیت کا لفظ "مثلھم" اسی حقیقت کی طرف اشارہ ہے: جو ہر لمحہ تمہیں تمہاری مثل عطا کرتا ہے اس کے لیے موت کے بعد مثل لوٹانا کیا مشکل؟ منکر اسی خلقِ جدید کا انکار کرتا ہے جس میں وہ خود ہر سانس جی رہا ہے؛ اسے یہ تسلسل اس لیے نظر نہیں آتا کہ ہر نئی صورت پچھلی جیسی ہے۔ اور "فابی" کا لفظ پھر وہی اصول دہراتا ہے: انکار ان کا اپنا تھا، دلیل کی کمی نہ تھی؛ جس نے نہ ماننے کا رخ چنا اس کے لیے دلیل بھی حجاب بن گئی۔

رکوع اس انکار کی جڑ پر ختم ہوتا ہے: قُلْ لَّوْ اَنْتُمْ تَمْلِكُوْنَ خَزَآئِنَ رَحْمَةِ رَبِّيْۤ اِذًا لَّاَمْسَكْتُمْ خَشْيَةَ الْاِنْفَاقِ ۭ وَكَانَ الْاِنْسَانُ قَتُوْرًا۔ "کہہ دیجیے: اگر تم میرے رب کی رحمت کے خزانوں کے مالک ہوتے تو خرچ ہو جانے کے ڈر سے ضرور روک رکھتے، اور انسان بڑا تنگ دل ہے۔" جو لوگ اس پر معترض تھے کہ رسالت کی رحمت ایک بشر کو کیوں ملی، ان کی اصل بیماری یہی تنگ دلی تھی؛ وہ رحمت کو اپنے پیمانے سے ناپنا چاہتے تھے۔ سورۂ الحجر میں فرمایا تھا: وَاِنْ مِّنْ شَيْءٍ اِلَّا عِنْدَنَا خَزَآئِنُهٗ؛ خزانے اسی کے پاس ہیں اور وہ انہیں قدرِ معلوم سے اتارتا ہے۔ اہلِ معرفت کہتے ہیں کہ بندہ خرچ سے اس لیے ڈرتا ہے کہ وہ خود کو مالک اور اپنے پاس کی چیز کو محدود دیکھتا ہے؛ حق کے خزانے اس لیے کم نہیں ہوتے کہ اس کی عطا بے نہایت ہے اور اس کی رحمت غضب پر سبقت رکھتی ہے۔ "قتورا" انسان کی جبلت کا بیان ہے، اس کا لازمی مقدر نہیں۔ شیخِ اکبر کے نزدیک اس کا علاج فقر کا شعور ہے: جب بندہ جان لیتا ہے کہ وہ کسی چیز کا مالک نہیں، صرف گزرگاہ ہے، تو اس کی مٹھی کھل جاتی ہے، کیونکہ ختم ہونے کا ڈر اسی کو ہوتا ہے جو گزرتی چیز کو اپنا سمجھے۔ یہی تخلق باسماء اللہ ہے: الکریم اور الوہاب کا عکس بندے کے ہاتھ میں۔ رکوع بشر رسول کے انکار سے شروع ہوا اور اس انکار کی جڑ پر ختم ہوا: رحمت کو محدود سمجھنا۔
