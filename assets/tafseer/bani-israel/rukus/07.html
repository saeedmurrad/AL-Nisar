یہ رکوع وہی داستان پھر سامنے لاتا ہے جو پڑھنے والا سورۂ الحجر میں دیکھ چکا ہے، مگر یہاں زاویہ بدل گیا ہے۔ وَاِذْ قُلْنَا لِلْمَلٰۤئِكَةِ اسْجُدُوْا لِاٰدَمَ فَسَجَدُوْۤا اِلَّاۤ اِبْلِيْسَ ۭ قَالَ ءَاَسْجُدُ لِمَنْ خَلَقْتَ طِيْنًا۔ اور جب ہم نے فرشتوں سے کہا کہ آدم کو سجدہ کرو تو سب نے سجدہ کیا سوائے ابلیس کے؛ اس نے کہا: کیا میں اسے سجدہ کروں جسے تو نے مٹی سے بنایا؟ ظاہر یہ ہے کہ سجدہ حق تعالیٰ کے حکم کی تعمیل تھا اور حضرت آدم علیہ السلام اس حکم کا قبلہ تھے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک ابلیس کی ٹھوکر یہ تھی کہ وہ صورت پر رک گیا: اس نے مٹی دیکھی اور وہ نفخِ روح نہ دیکھا جس نے اس مٹی کو اسمِ جامع کا آئینہ بنا دیا۔ پھر وہ بولا: قَالَ اَرَءَيْتَكَ هٰذَا الَّذِيْ كَرَّمْتَ عَلَيَّ لَئِنْ اَخَّرْتَنِ اِلٰى يَوْمِ الْقِيٰمَةِ لَاَحْتَنِكَنَّ ذُرِّيَّتَهٗۤ اِلَّا قَلِيْلًا۔ بھلا دیکھ تو، یہی ہے وہ جسے تو نے مجھ پر عزت دی! اگر تو مجھے قیامت کے دن تک مہلت دے تو میں اس کی اولاد کو، سوائے تھوڑوں کے، لگام ڈال کر اپنے پیچھے لگا لوں گا۔ اہلِ ذوق کہتے ہیں کہ دشمن کی زبان سے بھی حق کی گواہی نکلتی ہے: ابلیس نے حسد میں جو لفظ کہا وہ تکریم ہی کا اعتراف تھا، اور یہی رکوع آیتِ ستر پر اس تکریم پر الٰہی مہر لگاتا ہے۔ رکوع کا آغاز ابلیس کے اعتراف سے اور اختتام حق کے اعلان سے ہوتا ہے۔ احتناک کی اصل گھوڑے کے جبڑے میں رسی ڈالنا ہے، اور اس مکتب میں اشارہ یہ ہے کہ لگام اسی کے منہ میں پڑتی ہے جو خود منہ کھول دے۔ ابلیس کا دعویٰ بھی اپنی حد جانتا ہے، اسی لیے "تھوڑوں" کو مستثنیٰ کرتا ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

جواب میں نہ بحث ہے نہ مناظرہ، صرف رخصت کا حکم: قَالَ اذْهَبْ فَمَنْ تَبِعَكَ مِنْهُمْ فَاِنَّ جَهَنَّمَ جَزَآؤُكُمْ جَزَآءً مَّوْفُوْرًا۔ فرمایا: جا، پھر ان میں سے جو تیری پیروی کرے گا تو تم سب کی سزا جہنم ہے، پوری پوری سزا۔ غور طلب لفظ "تَبِعَكَ" ہے: فیصلہ پیروی سے جڑا ہے، اور پیروی پیروی کرنے والے کا اپنا فعل ہے۔ پھر فرمایا: وَاسْتَفْزِزْ مَنِ اسْتَطَعْتَ مِنْهُمْ بِصَوْتِكَ وَاَجْلِبْ عَلَيْهِمْ بِخَيْلِكَ وَرَجِلِكَ وَشَارِكْهُمْ فِي الْاَمْوَالِ وَالْاَوْلَادِ وَعِدْهُمْ ۭ وَمَا يَعِدُهُمُ الشَّيْطٰنُ اِلَّا غُرُوْرًا۔ اور ان میں سے جسے تو اپنی آواز سے بہکا سکے بہکا لے، ان پر اپنے سوار اور پیادے چڑھا لا، ان کے مال اور اولاد میں شریک ہو جا اور ان سے وعدے کر؛ اور شیطان کے وعدے فریب کے سوا کچھ نہیں۔ مفسرین کا اتفاق ہے کہ یہ امر اجازت نہیں، امرِ تہدید ہے، یعنی "جو کر سکتا ہے کر دیکھ"۔ اور ذرا اس کے سارے ہتھیار گن لیجیے: آواز، لشکر، شرکت، وعدہ۔ ان میں ایک بھی جبر نہیں؛ سب دعوت کی صورتیں ہیں۔ اہلِ تاویل کہتے ہیں کہ آواز دل پر اترنے والا شیطانی خاطر ہے، سوار تیز رو خواہشیں اور پیادے سست رو عادتیں؛ مال میں شرکت یہ ہے کہ بندہ عطا کو "میرا" کہنے لگے، اور اولاد میں شرکت یہ کہ اعمال کے ثمرات کو اپنی طرف منسوب کرے۔ یہ محض تمثیل ہے، اس میں اولاد کی تحقیر کا کوئی شائبہ نہیں، اور ظاہر معنی، یعنی حرام کمائی اور اولاد کو اللہ سے دور پالنا، اپنی جگہ قائم ہے۔ شیخِ اکبر کے نزدیک شیطان کا وعدہ غرور اس لیے ہے کہ باطل کا اپنا کوئی وجود نہیں؛ وہ ایک ایسی صورت کا وعدہ کرتا ہے جس کے پیچھے کوئی حقیقت کھڑی نہیں۔

اب وہ آیت آتی ہے جو اس رکوع کا اصلِ حاکم ہے، اور ہدایت و گمراہی کے باب میں اس سارے سلسلے کا بنیادی ضابطہ: اِنَّ عِبَادِيْ لَيْسَ لَكَ عَلَيْهِمْ سُلْطٰنٌ ۭ وَكَفٰى بِرَبِّكَ وَكِيْلًا۔ بے شک میرے بندوں پر تیرا کوئی زور نہیں، اور تیرا رب کارسازی کے لیے کافی ہے۔ یہی اصول ہے، اور اسے اصول ہی کی طرح پڑھنا چاہیے: حق تعالیٰ کسی کو زبردستی گمراہ نہیں کرتا، اور شیطان کو کسی پر جبر کا اختیار نہیں دیا گیا۔ سورۂ الحجر میں یہی بات استثنا کے ساتھ آئی تھی: اِنَّ عِبَادِيْ لَيْسَ لَكَ عَلَيْهِمْ سُلْطٰنٌ اِلَّا مَنِ اتَّبَعَكَ مِنَ الْغٰوِيْنَ، اور سورۂ ابراہیم میں خود شیطان نے اقرار کیا: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ۔ تینوں مقامات کو ملا کر دیکھیے تو تصویر مکمل ہو جاتی ہے: شیطان دعوت دیتا ہے، بندہ قبول یا رد کرتا ہے، اور حق تعالیٰ اس رخ کو وجود عطا فرماتا ہے جو بندے نے خود چنا۔ جو ہدایت کا طالب ہو اسے ہدایت ملتی ہے، اور جو منہ موڑے اس کا منہ موڑنا ہی اس کی گمراہی ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ "عِبَادِيْ" کی اضافت بھی اشارہ رکھتی ہے: اس سورت کے آغاز میں حضورِ اکرم صلی اللہ علیہ وسلم کو معراج کی رات عبد کے نام سے یاد کیا گیا، اور یہاں وہی نسبت حفاظت کا حصار بن گئی۔ جو عبودیت کو اپنا گھر بنا لے، اسی کا وکیل اس کا رب ہوتا ہے، اور جس کا وکیل حق ہو اس پر کسی غیر کا سلطان کیسے چلے؟

ابلیس کی دعوت کے بعد اب رب کی رحمت کا منظر: رَبُّكُمُ الَّذِيْ يُزْجِيْ لَكُمُ الْفُلْكَ فِي الْبَحْرِ لِتَبْتَغُوْا مِنْ فَضْلِهٖ ۭ اِنَّهٗ كَانَ بِكُمْ رَحِيْمًا۔ تمہارا رب وہی ہے جو تمہارے لیے سمندر میں کشتیاں آہستگی سے چلاتا ہے تاکہ تم اس کا فضل تلاش کرو؛ بے شک وہ تم پر بہت مہربان ہے۔ "یُزجی" نرمی سے ہانکنے کو کہتے ہیں، گویا کشتی اپنی لکڑی سے نہیں، اس کے لطف سے تیرتی ہے۔ آیت کی ترتیب میں سبقتِ رحمت ہے: پہلے فضل کی راہ اور رحیمیت کا ذکر، پھر مصیبت کا۔ وَاِذَا مَسَّكُمُ الضُّرُّ فِي الْبَحْرِ ضَلَّ مَنْ تَدْعُوْنَ اِلَّاۤ اِيَّاهُ ۚ فَلَمَّا نَجّٰىكُمْ اِلَى الْبَرِّ اَعْرَضْتُمْ ۭ وَكَانَ الْاِنْسَانُ كَفُوْرًا۔ اور جب سمندر میں تم پر مصیبت آتی ہے تو اس کے سوا جنہیں تم پکارتے ہو سب گم ہو جاتے ہیں؛ پھر جب وہ تمہیں بچا کر خشکی پر لے آتا ہے تو تم منہ پھیر لیتے ہو، اور انسان بڑا ناشکرا ہے۔ اہلِ معرفت کے نزدیک اضطرار فطرت کا پردہ اٹھا دیتا ہے: طوفان میں سارے بت خود بخود غائب ہو جاتے ہیں، کیونکہ غیر کا اپنا وجود کبھی تھا ہی نہیں؛ وہ آسودگی کے پردے میں قائم ایک وہم تھا۔ یہاں "ضَلَّ" گم ہو جانے کے معنی میں ہے، اور یہ لطیف اشارہ ہے کہ جھوٹے معبود خود راہ گم کر دیتے ہیں۔ مگر ساحل پر پہنچتے ہی "اَعْرَضْتُمْ": اعراض بندے کا اپنا فعل ہے، اور اسی کو کفور کہا گیا، یعنی دیکھی ہوئی حقیقت کو ڈھانپ دینا۔

پھر ایک سوال جو ہر اس شخص سے ہے جو خشکی کو امان سمجھ بیٹھا: اَفَاَمِنْتُمْ اَنْ يَّخْسِفَ بِكُمْ جَانِبَ الْبَرِّ اَوْ يُرْسِلَ عَلَيْكُمْ حَاصِبًا ثُمَّ لَا تَجِدُوْا لَكُمْ وَكِيْلًا۔ کیا تم اس سے بے خوف ہو گئے کہ وہ خشکی ہی کے کنارے تمہیں زمین میں دھنسا دے یا تم پر پتھر برسانے والی آندھی بھیج دے، پھر تم اپنا کوئی کارساز نہ پاؤ؟ اَمْ اَمِنْتُمْ اَنْ يُّعِيْدَكُمْ فِيْهِ تَارَةً اُخْرٰى فَيُرْسِلَ عَلَيْكُمْ قَاصِفًا مِّنَ الرِّيْحِ فَيُغْرِقَكُمْ بِمَا كَفَرْتُمْ ثُمَّ لَا تَجِدُوْا لَكُمْ عَلَيْنَا بِهٖ تَبِيْعًا۔ یا اس سے بے خوف ہو گئے کہ وہ تمہیں دوبارہ سمندر میں لے جائے، پھر تم پر توڑ دینے والی ہوا بھیجے اور تمہاری ناشکری کے سبب تمہیں غرق کر دے، پھر تم ہمارے خلاف اس کا کوئی پیچھا کرنے والا نہ پاؤ؟ خشکی اور تری دونوں اسی کے ہیں؛ امان کسی جگہ میں نہیں، حضوری میں ہے۔ اہلِ تاویل کہتے ہیں کہ سالک جب کسی حال کے طوفان سے نکل کر کسی مقام کے ساحل پر پہنچتا ہے تو اسے امن کا دھوکا ہوتا ہے؛ حالانکہ شیخِ اکبر کے نزدیک ہر آن خلقِ جدید ہے، اور تجددِ امثال کا مطلب یہ ہے کہ کل کی سلامتی آج کی ضمانت نہیں۔ "تَارَةً اُخْرٰى" بتاتا ہے کہ احوال لوٹ آتے ہیں، اور جو ایک بار نجات پا کر بھول گیا وہ پھر اسی دریا میں اتارا جا سکتا ہے۔ اور سبب کی تصریح بھی کر دی: "بِمَا كَفَرْتُمْ"، یعنی ہلاکت ان کے اپنے فعل کی صورت ہے، کوئی بے سبب قہر نہیں۔ دونوں آیتوں کا خاتمہ ایک ہی سبق پر ہے: اس کے مقابل نہ کوئی وکیل ہے نہ کوئی دعوے دار۔

اور اب رکوع اپنے عروج پر پہنچتا ہے، وہیں جہاں سے ابلیس کا اعتراض شروع ہوا تھا: وَلَقَدْ كَرَّمْنَا بَنِيْۤ اٰدَمَ وَحَمَلْنٰهُمْ فِي الْبَرِّ وَالْبَحْرِ وَرَزَقْنٰهُمْ مِّنَ الطَّيِّبٰتِ وَفَضَّلْنٰهُمْ عَلٰى كَثِيْرٍ مِّمَّنْ خَلَقْنَا تَفْضِيْلًا۔ اور بے شک ہم نے آدم کی اولاد کو عزت دی، انہیں خشکی اور تری میں سواری دی، پاکیزہ چیزوں سے رزق دیا اور اپنی بہت سی مخلوق پر انہیں نمایاں فضیلت دی۔ ابلیس نے کہا تھا "جسے تو نے عزت دی"، اور حق نے فرمایا "ہم نے عزت دی"؛ حسد کا جملہ اعلان کے جملے میں ڈھل گیا۔ ظاہر معنی یہ ہے کہ یہ تکریم ہر انسان کی ہے، بلا امتیازِ نسل و قوم؛ انسان ہونا بجائے خود حرمت رکھتا ہے۔ شیخِ اکبر کے نزدیک یہ تکریم اس لیے ہے کہ انسان کونِ جامع ہے: عالم کی ہر حقیقت اس میں سمٹی ہوئی ہے اور وہ اسمِ جامع کا مظہر ہے۔ خشکی اور تری میں اٹھائے جانے کا اشارہ یہ ہے کہ انسان برزخ ہے، عالمِ خلق کی کثافت اور عالمِ امر کی لطافت کے بیچ؛ دونوں اسے اٹھائے پھرتے ہیں۔ اور ادب دیکھیے کہ "سب پر" نہیں، "بہت سوں پر" فرمایا؛ فضیلت کا دعویٰ بھی حد میں رکھا گیا۔ اہلِ معرفت کہتے ہیں کہ یہ کرامت بالقوہ سب کو ملی ہے، مگر بالفعل اس کا کمال انسانِ کامل میں ظاہر ہوتا ہے، اور اس کا دروازہ وہی عبودیت ہے جس کا ذکر آیتِ پینسٹھ میں ہوا۔ رزقِ طیبات میں روح کا رزق بھی شامل ہے، یعنی معرفت، جو ہر پاک رزق سے بڑھ کر پاک ہے۔
