بدر کی فتح کا بیان مکمل ہوا تو خطاب کا رخ میدان سے دل کی طرف مڑ گیا۔ جس جماعت نے ابھی ابھی دیکھا تھا کہ کنکریاں کس کے حکم سے چلیں اور فرشتے کس کے امر سے اترے، اسے اب یہ سبق دیا جاتا ہے کہ اصل معرکہ سننے اور ماننے کا ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْۤا اَطِيْعُوا اللّٰهَ وَرَسُوْلَهٗ وَلَا تَوَلَّوْا عَنْهُ وَاَنْتُمْ تَسْمَعُوْنَ۔ “اے ایمان والو! اللہ اور اس کے رسول صلی اللہ علیہ وسلم کی اطاعت کرو، اور اس سے منہ نہ پھیرو جبکہ تم سن رہے ہو۔” مفسرین نے یہ نکتہ لکھا ہے کہ اللہ اور رسول صلی اللہ علیہ وسلم دونوں کا ذکر کر کے ضمیر واحد لائی گئی، عَنْهُ، یعنی اس سے۔ گویا دونوں اطاعتیں ایک ہی اطاعت ہیں، کیونکہ رسول صلی اللہ علیہ وسلم کا حکم اللہ ہی کا حکم ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس واحد ضمیر سے یہ بات کھلتی ہے کہ رسالت کوئی دوسرا مرکز نہیں جس کی طرف دل بٹ جائے؛ حضورِ اکرم صلی اللہ علیہ وسلم وہ کامل آئینہ ہیں جس میں حکمِ حق بغیر کسی کمی بیشی کے دکھائی دیتا ہے۔ اور وَاَنْتُمْ تَسْمَعُوْنَ کی قید بڑی نازک ہے: منہ پھیرنا اس وقت جرم بنتا ہے جب بات کان تک پہنچ چکی ہو۔ اہلِ ذوق کہتے ہیں کہ سن لینے کے بعد کی غفلت نہ سننے کی غفلت سے بھاری ہے، کیونکہ اب حجت دل کے دروازے پر کھڑی ہے۔ ظاہر معنی اپنی جگہ قائم ہے کہ میدانِ جنگ اور تقسیمِ غنیمت میں حضورِ اکرم صلی اللہ علیہ وسلم کے حکم کی پوری پابندی کی جائے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

وَلَا تَكُوْنُوْا كَالَّذِيْنَ قَالُوْا سَمِعْنَا وَهُمْ لَا يَسْمَعُوْنَ۔ “اور ان لوگوں جیسے نہ ہو جاؤ جنہوں نے کہا ہم نے سن لیا، حالانکہ وہ نہیں سنتے۔” ایک ہی آیت میں سننے کا اثبات بھی ہے اور نفی بھی: زبان نے کہا سَمِعْنَا اور قرآن نے فرمایا لَا يَسْمَعُوْنَ۔ مفسرین کہتے ہیں کہ کان سے سننا اور قبول کے ساتھ سننا دو الگ چیزیں ہیں؛ جس سماع کے بعد عمل نہ ہو قرآن اسے سماع ہی شمار نہیں کرتا۔ اہلِ معرفت کے نزدیک سماع کے مراتب ہیں۔ پہلا کان کا سماع ہے جس میں انسان اور حیوان شریک ہیں۔ دوسرا دل کا سماع ہے جس میں بات اندر اتر کر حال بدل دیتی ہے۔ تیسرا وہ سماع ہے جس کی طرف حدیثِ قربِ نوافل اشارہ کرتی ہے، جس کا مفہوم صحیح بخاری میں یہ ہے کہ بندہ نوافل کے ذریعے قریب ہوتا جاتا ہے یہاں تک کہ حق تعالیٰ فرماتا ہے کہ میں اس کا کان بن جاتا ہوں جس سے وہ سنتا ہے۔ شیخِ اکبر کے نزدیک یہ حلول نہیں، بلکہ یہ کہ بندے کی سماعت اپنی خود رائی سے خالی ہو کر حق کی بات کو جیسی وہ ہے ویسی قبول کرنے لگتی ہے۔ جو لوگ سَمِعْنَا کہہ کر رک گئے وہ پہلے مرتبے پر ٹھہر گئے اور دعویٰ آخری مرتبے کا کیا۔ اس آیت میں مومنوں کو ان جیسا بننے سے روکا گیا، یعنی خطرہ صرف منکر کے لیے نہیں، ماننے والے کے لیے بھی ہے کہ اس کا سننا ایک رسم بن کر رہ جائے۔ اگلی آیت میں جس بہرے پن کا ذکر آ رہا ہے، اس کی جڑ یہی خالی سَمِعْنَا ہے۔

اِنَّ شَرَّ الدَّوَآبِّ عِنْدَ اللّٰهِ الصُّمُّ الْبُكْمُ الَّذِيْنَ لَا يَعْقِلُوْنَ۔ “بیشک اللہ کے نزدیک بدترین چلنے پھرنے والے وہ بہرے گونگے ہیں جو عقل سے کام نہیں لیتے۔” سب سے پہلے یہ بات صاف رہے کہ یہاں جسمانی معذوری مراد نہیں۔ جو شخص کان یا زبان سے محروم پیدا ہوا وہ اللہ کے نزدیک ذرہ برابر کم نہیں، اور ایسے بہت سے بندے دل کی سماعت میں سننے والوں سے آگے ہیں۔ یہاں وہ بہرا پن مراد ہے جو حق کی بات سن کر اسے اندر نہ جانے دے، اور وہ گونگا پن جو حق کا اقرار نہ کرے؛ اور آخری صفت، الَّذِيْنَ لَا يَعْقِلُوْنَ، ساری بات کی کنجی ہے۔ عقل کا مادہ عِقال ہے، یعنی وہ رسی جس سے اونٹ باندھا جاتا ہے۔ اہلِ معرفت کہتے ہیں کہ عقل وہ ہے جو سنی ہوئی سچائی کو دل سے باندھ دے؛ جس نے سنا اور باندھا نہیں، اس نے عقل سے کام نہیں لیا۔ الدَّوَآبِّ کا لفظ زمین پر چلنے والی ہر جان دار شے کو شامل ہے، اور انسان کا شرف یہ ہے کہ وہ چلنے کے ساتھ سمجھنے والا بھی ہے؛ جب سمجھنا معطل ہو جائے تو صرف چلنا باقی رہ جاتا ہے۔ یہ کسی قوم کی تحقیر نہیں، ایک حال کا بیان ہے جس سے ہر دل کو پناہ مانگنی چاہیے۔ سورۂ الملک میں اہلِ دوزخ کا یہی اعتراف گزرا: لَوْ كُنَّا نَسْمَعُ اَوْ نَعْقِلُ مَا كُنَّا فِيْۤ اَصْحٰبِ السَّعِيْرِ، “اگر ہم سنتے یا سمجھتے تو دہکتی آگ والوں میں نہ ہوتے۔” وہاں بھی سننا اور سمجھنا ساتھ آئے، اور یہاں بھی بہرا پن اور بے عقلی ایک ہی حال کے دو نام ہیں۔ اور عِنْدَ اللّٰهِ کی قید بتاتی ہے کہ یہ لوگوں کی رائے نہیں، حقیقت کا ترازو ہے، جس میں آدمی کی قدر اس کے قبولِ حق سے ہے۔

وَلَوْ عَلِمَ اللّٰهُ فِيْهِمْ خَيْرًا لَّاَسْمَعَهُمْ ۗ وَلَوْ اَسْمَعَهُمْ لَتَوَلَّوْا وَّهُمْ مُّعْرِضُوْنَ۔ “اور اگر اللہ ان میں کوئی بھلائی جانتا تو انہیں ضرور سنوا دیتا؛ اور اگر وہ انہیں سنوا دیتا تب بھی وہ منہ پھیر کر چل دیتے، اس حال میں کہ وہ اعراض کرنے والے ہیں۔” یہ آیت ہدایت و گمراہی کے باب میں ایک بنیادی اصول بیان کرتی ہے، اور اسے اسی ترتیب سے پڑھنا چاہیے جس ترتیب سے یہ اتری ہے۔ پہلا جملہ یہ نہیں کہتا کہ اللہ نے ان سے بھلائی روک لی؛ وہ کہتا ہے کہ اللہ نے ان میں کوئی بھلائی پائی ہی نہیں۔ مفسرین کے نزدیک یہاں خیر سے مراد طلبِ حق، صدقِ نیت اور قبول کی آمادگی ہے۔ یعنی کمی ان کی طرف سے تھی: ان میں طلب نہ تھی۔ اور اس جملے کا لازمی مفہوم یہ ہے کہ جس میں ذرہ بھر خیر ہو اللہ اسے ضرور سنواتا ہے؛ کوئی طالب ایسا نہیں جسے سماع سے محروم رکھا گیا ہو۔ یہی وہ اصول ہے جس پر پوری بات قائم ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت چاہے اسے کبھی گمراہ نہیں کرتا۔ یہاں سنوانے سے مراد قبول کا سماع ہے، ورنہ کان سے تو یہ لوگ قرآن روز سنتے تھے۔ اسی لیے آیت نے سماعِ قبول کو حق کی عطا کہا اور اس کی شرط بندے کے اندر کی بھلائی کو ٹھہرایا؛ عطا حق کی ہے، اور طلب بندے کا اپنا رخ ہے جس کا وہ جواب دہ ہے۔

شیخِ اکبر کے نزدیک اس آیت کی گرہ سرِّ قدر سے کھلتی ہے۔ اس مکتب کا معروف اصول ہے کہ علم معلوم کے تابع ہے: حق تعالیٰ ہر شے کو ویسا ہی جانتا ہے جیسی وہ اپنی حقیقت میں ہے، اور اس کا حکم اسی علم کے مطابق جاری ہوتا ہے۔ وَلَوْ عَلِمَ اللّٰهُ فِيْهِمْ خَيْرًا کا مطلب یہ ہوا کہ ان کی حقیقت میں طلب کا وہ جوہر نہ تھا جس پر سماع کا فیض اترتا۔ علم نے کچھ پیدا نہیں کیا، صرف جو تھا اسے جانا؛ اور حکم نے کچھ مسلط نہیں کیا، صرف وہی دیا جو حال کی زبان سے مانگا گیا۔ اسی کو اہلِ معرفت حکم کا محکوم کے مطابق ہونا کہتے ہیں۔ بندے نے جس طرف رخ کیا، حق نے اس رخ کو وجود دیا، کیونکہ وجود کا فیض اسی سے ہے؛ مگر رخ کا انتخاب بندے کا اپنا تھا۔ اس میں ایک بڑی تسلی بھی ہے: جو شخص اپنے اندر ذرا سی بھی طلب پاتا ہے، چاہے وہ ایک بے چینی ہو یا سچ جاننے کی ایک خواہش، وہ اسی خیر کا نشان ہے جسے حق جانتا ہے؛ اور جس خیر کو حق جان لے اسے وہ سنوائے بغیر نہیں چھوڑتا۔ اہلِ ذوق کہتے ہیں کہ طالب کا اپنی طلب پر نظر رکھنا شکر ہے اور اسے اپنا کمال سمجھنا غفلت، کیونکہ طلب بھی اسی کی توفیق سے جاگتی ہے، اگرچہ اس کی ذمہ داری بندے پر ہے۔

دوسرا جملہ اس اصول پر مہر لگا دیتا ہے: وَلَوْ اَسْمَعَهُمْ لَتَوَلَّوْا وَّهُمْ مُّعْرِضُوْنَ۔ فرض کیجیے انہیں بات سنوا بھی دی جاتی، تو کیا ہوتا؟ وہ منہ پھیر لیتے۔ یعنی رکاوٹ سنانے والے کی طرف سے نہیں، سننے والے کے رخ میں تھی۔ مُعْرِضُوْنَ اسمِ فاعل ہے جو جمی ہوئی حالت بتاتا ہے: اعراض ان کی عارضی کیفیت نہیں، ان کا چنا ہوا مستقل رخ تھا۔ اور یہی منہ پھیرنا بعینہٖ ان کی گمراہی ہے؛ گمراہی کوئی الگ شے نہیں جو اوپر سے ان پر ڈالی گئی ہو۔ اہلِ تاویل کہتے ہیں کہ اگر اللہ چاہتا تو سب کو ماننے پر مجبور کر سکتا تھا، جیسا کہ فرمایا: وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ، “اور اگر وہ چاہتا تو تم سب کو ہدایت دے دیتا۔” مگر اس نے جبر نہیں چاہا، کیونکہ جبری ایمان ایمان نہیں، اور انسان کا شرف اس کے اختیار میں ہے۔ یہ بھی ملحوظ رہے کہ آیت ان لوگوں کا حال بتاتی ہے جو اعراض پر جمے رہے؛ دروازہ کسی پر بند نہیں کیا گیا۔ تاریخ گواہ ہے کہ قریش کے بہت سے لوگ جو کبھی سننے سے انکاری تھے، بعد میں سن کر مانے اور اسلام کے ستون بنے۔ جس نے رخ بدلا اس کے لیے حکم بھی بدل گیا۔ یہی سرِّ قدر کی رحمت ہے کہ حکم بندے کے حال کے ساتھ چلتا ہے، اور حال بدلنے کا اختیار آخری سانس تک باقی رہتا ہے۔

اب وہ آیت آتی ہے جو اس رکوع کا دل ہے: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوا اسْتَجِيْبُوْا لِلّٰهِ وَلِلرَّسُوْلِ اِذَا دَعَاكُمْ لِمَا يُحْيِيْكُمْ۔ “اے ایمان والو! اللہ اور رسول صلی اللہ علیہ وسلم کی پکار پر لبیک کہو جب وہ تمہیں اس چیز کی طرف بلائے جو تمہیں زندگی بخشتی ہے۔” پچھلی آیات میں ان لوگوں کا ذکر تھا جنہوں نے سنا اور نہ سنا؛ اب ایمان والوں کو وہ راستہ دکھایا جاتا ہے جو بہرے پن کا علاج ہے، یعنی استجابت۔ اسْتَجِيْبُوْا میں باب استفعال کا وزن ہے جو یہاں مبالغہ اور تاکید کا مفہوم دیتا ہے: صرف جواب دینا نہیں، دل کی پوری آمادگی کے ساتھ لپک کر جواب دینا۔ یہاں بھی وہی لطیف نکتہ دہرایا گیا جو بیسویں آیت میں تھا: اللہ اور رسول صلی اللہ علیہ وسلم دونوں کا ذکر ہوا مگر فعل واحد آیا، اِذَا دَعَاكُمْ، یعنی جب وہ تمہیں بلائے۔ پکار ایک ہے؛ حضورِ اکرم صلی اللہ علیہ وسلم کی زبانِ مبارک سے جو دعوت آتی ہے وہی اللہ کی دعوت ہے۔ شیخِ اکبر کے نزدیک یہ انسانِ کامل کا مقام ہے کہ اس کی آواز میں حق کی آواز کے سوا کچھ نہیں ہوتا؛ وہ ایسا برزخ ہے جس سے پکار کسی آمیزش کے بغیر بندوں تک پہنچتی ہے۔ اسی لیے اس پکار کا جواب ٹالا نہیں جا سکتا، جیسا کہ صحیح بخاری کا ایک واقعہ بتاتا ہے۔

صحیح بخاری میں حضرت ابو سعید بن معلیٰ رضی اللہ عنہ کا واقعہ آیا ہے، جس کا مفہوم یہ ہے کہ وہ مسجد میں نماز پڑھ رہے تھے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے انہیں پکارا۔ انہوں نے اس وقت جواب نہ دیا، پھر حاضر ہو کر عرض کیا کہ یا رسول اللہ صلی اللہ علیہ وسلم، میں نماز پڑھ رہا تھا۔ آپ صلی اللہ علیہ وسلم نے فرمایا: کیا اللہ نے یہ نہیں فرمایا کہ اللہ اور رسول صلی اللہ علیہ وسلم کی پکار پر لبیک کہو جب وہ تمہیں بلائے؟ پھر آپ صلی اللہ علیہ وسلم نے فرمایا کہ مسجد سے نکلنے سے پہلے میں تمہیں قرآن کی سب سے عظیم سورت سکھاؤں گا۔ جب آپ صلی اللہ علیہ وسلم نکلنے لگے تو حضرت ابو سعید رضی اللہ عنہ نے یاد دلایا، اور آپ صلی اللہ علیہ وسلم نے فرمایا کہ وہ اَلْحَمْدُ لِلّٰهِ رَبِّ الْعٰلَمِيْنَ ہے، یہی سبعِ مثانی ہے اور وہ قرآنِ عظیم جو مجھے عطا ہوا۔ اس واقعے میں ایک صحابی ہیں جو نماز جیسی عبادت میں مشغول ہیں، ان کا عذر بظاہر بہترین عذر ہے، اور پھر بھی حضورِ اکرم صلی اللہ علیہ وسلم نے آیت پڑھ کر یاد دلایا کہ اس پکار کا مقام سب سے پہلے ہے۔ بعض اہلِ علم نے اس سے یہ مسئلہ نکالا کہ حضورِ اکرم صلی اللہ علیہ وسلم کی پکار کا جواب نماز کی حالت میں بھی دینا تھا اور یہ آپ صلی اللہ علیہ وسلم کی خصوصیت تھی؛ اس کی فقہی تفصیل اہلِ فقہ کے ہاں ہے۔ یہاں ہمیں ادب کا وہ نکتہ لینا ہے جو اس واقعے کی روح ہے: جس پکار میں زندگی ہو، اس کے آگے کوئی مشغولیت، چاہے وہ کتنی ہی مقدس ہو، دیوار نہیں بننی چاہیے۔

اہلِ معرفت اس واقعے کے انجام پر ٹھہرتے ہیں۔ پکار کا جواب ذرا تاخیر سے ہوا تو تنبیہ ہوئی، مگر تنبیہ کے ساتھ ہی ایک عطا آئی: سورۂ فاتحہ کی یہ معرفت کہ وہ قرآن کی سب سے عظیم سورت ہے۔ گویا پکار کے پیچھے ہمیشہ ایک تحفہ ہوتا ہے، اور جسے بلایا جاتا ہے اسے کچھ دینے کے لیے بلایا جاتا ہے۔ شیخِ اکبر کے مکتب میں فاتحہ کو اُمّ الکتاب اس لیے کہا جاتا ہے کہ سارا قرآن اس میں جمع ہے، جیسے بیج میں پورا درخت؛ اور اس دن پکار کا جواب دینے والے کو وہی جامع سورت ملی۔ اہلِ ذوق کہتے ہیں کہ یہی لِمَا يُحْيِيْكُمْ کی ایک عملی تفسیر ہے: جو پکار پر لپکتا ہے اسے علم کا وہ دروازہ ملتا ہے جس سے دل زندہ ہوتا ہے۔ آج حضورِ اکرم صلی اللہ علیہ وسلم کی آوازِ مبارک کان سے نہیں سنی جاتی، مگر آپ صلی اللہ علیہ وسلم کی پکار باقی ہے؛ وہ قرآن کی ہر آیت میں، سنت کے ہر حکم میں اور اذان کی ہر صدا میں موجود ہے۔ اذان سن کر کام سمیٹ لینا، حکم سامنے آئے تو تاویل کے بجائے تعمیل کرنا، اور دل میں نیکی کا خیال اٹھے تو اسے کل پر نہ ٹالنا، یہ سب اسی استجابت کی صورتیں ہیں۔ اس مکتب میں طالب کو یہ ادب سکھایا جاتا ہے کہ وہ پکار کو پہچانے، کیونکہ حق کی پکار اکثر ایک ہلکی سی آہٹ کی صورت آتی ہے، اور جو اسے پہلی بار نہ سنے اسے دوسری بار زیادہ کڑے انداز میں سننی پڑتی ہے۔

اب لِمَا يُحْيِيْكُمْ پر نظر کیجیے۔ مفسرین نے اس کی کئی تفسیریں کی ہیں: قرآن، ایمان، حق اور اسلام؛ اور بعض نے بدر کے اس معرکے کو بھی مراد لیا جس کے ذریعے اللہ نے کمزور جماعت کو عزت دی۔ یہ سب اپنی جگہ درست ہیں اور ایک ہی حقیقت کے پہلو ہیں۔ آیت یہ نہیں کہتی کہ زندوں کو بلایا جا رہا ہے؛ وہ کہتی ہے کہ بلایا اس چیز کی طرف جا رہا ہے جو زندگی دیتی ہے۔ گویا بلائے جانے سے پہلے کی حالت ایک طرح کی موت ہے، اور پچھلی آیات کے بہرے گونگے اسی موت کی تصویر تھے۔ شیخِ اکبر کے نزدیک ہر شے کی زندگی اسمِ الٰہی الحی کا فیض ہے، اور دل کی زندگی علم اور معرفت سے ہے۔ جسم سانس سے جیتا ہے اور دل حق کی بات قبول کرنے سے۔ اس مکتب میں کہا جاتا ہے کہ استجابت اور حیات ایک دوسرے کو بڑھاتی ہیں: جو جتنا جواب دیتا ہے اتنا زندہ ہوتا ہے، اور جتنا زندہ ہوتا ہے اتنا ہی سننے کے قابل ہوتا ہے۔ یہاں بھی وہی اصول نظر آتا ہے جو تیئیسویں آیت میں تھا: زندگی پکار میں رکھی گئی ہے اور پکار سب کے لیے ہے؛ جو اسے قبول کرے وہ زندگی پا لیتا ہے، اور جو منہ پھیرے وہ اپنی موت خود چنتا ہے۔ کسی طالب کے لیے یہ دروازہ بند نہیں۔

پھر آیت کا دوسرا حصہ آتا ہے: وَاعْلَمُوْۤا اَنَّ اللّٰهَ يَحُوْلُ بَيْنَ الْمَرْءِ وَقَلْبِهٖ وَاَنَّهٗۤ اِلَيْهِ تُحْشَرُوْنَ۔ “اور جان لو کہ اللہ آدمی اور اس کے دل کے درمیان حائل ہو جاتا ہے، اور یہ کہ تم اسی کی طرف اکٹھے کیے جاؤ گے۔” یہ جملہ ڈرانے سے پہلے قربت کی خبر ہے۔ امام قتادہ رحمۃ اللہ علیہ سے مروی ہے کہ اس کا مفہوم یہ ہے کہ اللہ بندے سے اس کے دل سے بھی زیادہ قریب ہے؛ آدمی اور اس کے دل کے درمیان بھی اگر کوئی فاصلہ ہو تو حق اس فاصلے سے زیادہ قریب ہے۔ سورۂ قٓ میں یہی بات اس طرح آئی: وَنَحْنُ اَقْرَبُ اِلَيْهِ مِنْ حَبْلِ الْوَرِيْدِ، “اور ہم اس کی شہ رگ سے بھی زیادہ اس کے قریب ہیں۔” دوسرا مفہوم یہ ہے کہ دلوں کا مالک وہی ہے؛ انسان اپنے دل کو اپنی جاگیر سمجھتا ہے، مگر اس کے احوال اس کے اپنے قبضے میں نہیں۔ اسی لیے حضورِ اکرم صلی اللہ علیہ وسلم کثرت سے یہ دعا فرماتے تھے، جس کا مفہوم امام ترمذی رحمۃ اللہ علیہ نے روایت کیا ہے: اے دلوں کو پلٹنے والے، میرے دل کو اپنے دین پر ثابت رکھ۔ یہ ملکیت جبر کی خبر نہیں۔ جو دل حق کی طرف رخ کرے، اس کا مالک اسے جمانے والا ہے؛ اور جس کا مالک اتنا قریب ہو، طالب کے لیے اس سے بڑی پناہ کوئی نہیں۔ اس جملے کو اس طرح ہرگز نہ پڑھا جائے کہ اللہ کسی کے دل کو زبردستی حق سے پھیر دیتا ہے؛ تیئیسویں آیت بتا چکی کہ رکاوٹ اعراض کی ہے، اور اعراض بندے کا اپنا رخ ہے۔

شیخِ اکبر کے نزدیک قلب کا نام تقلب سے ہے، یعنی پلٹتے رہنا۔ دل ایک حال پر نہیں ٹھہرتا، ہر آن نئی تجلی اس پر آتی ہے اور وہ اس کے رنگ میں رنگ جاتا ہے؛ اس مکتب میں اسے تجددِ امثال کہا جاتا ہے، کہ ہر لمحہ خلق نئی ہے۔ آدمی گمان کرتا ہے کہ اس کا دل وہی ہے جو کل تھا، حالانکہ ہر لمحہ وہ ایک نئی عطا ہے، اور حق ہر عطا اور اس کے قبول کرنے والے کے بیچ موجود ہے۔ یہ يَحُوْلُ بَيْنَ الْمَرْءِ وَقَلْبِهٖ کی ایک ذوقی تعبیر ہے، اور ظاہر کی تفسیر اپنی جگہ قائم ہے۔ اسی سے وہ فوری پن پیدا ہوتا ہے جو آیت کے دونوں حصوں کو جوڑتا ہے: پکار پر ابھی لبیک کہو، کیونکہ جو حال اس وقت نصیب ہے وہ اگلی گھڑی ویسا نہیں رہے گا۔ بعض مفسرین نے اسی کو یوں بیان کیا کہ موت آدمی اور اس کے دل کی آرزوؤں کے درمیان آ جاتی ہے؛ جو نیکی آج دل میں ہے اس کا موقع کل ملے یا نہ ملے۔ آخر میں فرمایا: وَاَنَّهٗۤ اِلَيْهِ تُحْشَرُوْنَ۔ جس نے پکار پر لبیک کہا وہ آخرکار اسی کے حضور پہنچے گا جس نے پکارا تھا، اور جس نے منہ پھیرا وہ بھی اسی کے سامنے کھڑا ہوگا؛ فرق صرف اتنا ہے کہ ایک دعوت قبول کر کے آیا اور دوسرا دعوت ٹھکرا کر۔

وَاتَّقُوْا فِتْنَةً لَّا تُصِيْبَنَّ الَّذِيْنَ ظَلَمُوْا مِنْكُمْ خَآصَّةً ۚ وَاعْلَمُوْۤا اَنَّ اللّٰهَ شَدِيْدُ الْعِقَابِ۔ “اور اس فتنے سے بچو جو تم میں سے صرف ظلم کرنے والوں ہی کو نہیں پہنچے گا، اور جان لو کہ اللہ سخت سزا دینے والا ہے۔” یہ آیت فرد سے آگے بڑھ کر جماعت سے بات کرتی ہے۔ مفسرین کہتے ہیں کہ جب کسی معاشرے میں برائی کھل کر ہونے لگے اور اچھے لوگ خاموش رہیں تو اس کا وبال صرف کرنے والوں تک نہیں رہتا۔ صحیح بخاری میں حضرت نعمان بن بشیر رضی اللہ عنہ کی روایت ہے، جس کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے ایک کشتی کی مثال دی جس کے نچلے حصے والے پانی کے لیے کشتی میں سوراخ کرنا چاہیں؛ اگر اوپر والے ان کا ہاتھ پکڑ لیں تو سب بچ جائیں، اور اگر چھوڑ دیں تو سب ڈوب جائیں۔ اور ام المومنین حضرت زینب بنت جحش رضی اللہ عنہا نے پوچھا کہ کیا ہم ہلاک ہو جائیں گے جبکہ ہم میں نیک لوگ موجود ہوں؟ تو آپ صلی اللہ علیہ وسلم نے فرمایا: ہاں، جب خباثت بڑھ جائے۔ یہ بات اپنی اصلاح اور نیکی کی تلقین کی دعوت ہے، کسی قوم پر فتویٰ نہیں۔ اہلِ معرفت کے نزدیک امت ایک جسم ہے، اور نیکی کا حکم دینا اور برائی سے روکنا اس جسم کی حفاظت کا نظام ہے۔ باطن میں بھی یہی قانون چلتا ہے: جب انسان اپنے اندر کسی ایک برائی کو جگہ دیتا ہے تو اس کا اثر اسی گوشے تک نہیں رہتا، ساری عبادتوں میں سرایت کر جاتا ہے۔ اسی لیے فتنے سے بچنے کا حکم ہوا، صرف ظلم سے نہیں: یعنی وہ دروازے بند کرو جن سے فتنہ عام ہوتا ہے۔ اور شَدِيْدُ الْعِقَابِ کے ساتھ یہ یاد رہے کہ اسی ذات کی رحمت اس کے غضب پر سبقت رکھتی ہے؛ سزا کی خبر رحمت کی طرف لوٹنے کی دعوت ہے۔

وَاذْكُرُوْۤا اِذْ اَنْتُمْ قَلِيْلٌ مُّسْتَضْعَفُوْنَ فِي الْاَرْضِ تَخَافُوْنَ اَنْ يَّتَخَطَّفَكُمُ النَّاسُ فَاٰوٰىكُمْ وَاَيَّدَكُمْ بِنَصْرِهٖ وَرَزَقَكُمْ مِّنَ الطَّيِّبٰتِ لَعَلَّكُمْ تَشْكُرُوْنَ۔ “اور یاد کرو جب تم تھوڑے تھے، زمین میں کمزور سمجھے جاتے تھے، ڈرتے تھے کہ لوگ تمہیں اچک لے جائیں گے؛ پھر اس نے تمہیں ٹھکانا دیا، اپنی مدد سے تمہیں قوت دی، اور پاکیزہ چیزوں سے رزق دیا، تاکہ تم شکر کرو۔” یہ مکہ کے ان دنوں کی یاد ہے جب مسلمان مٹھی بھر تھے، ستائے جاتے تھے اور آخر ہجرت پر مجبور ہوئے؛ پھر مدینہ میں انہیں پناہ ملی، انصار رضی اللہ عنہم کی صورت میں مددگار ملے، اور بدر میں نصرت آئی۔ تین نعمتیں تین حاجتوں کے مقابل ہیں: بے ٹھکانا ہونے کے مقابل اٰوٰى، کمزوری کے مقابل اَيَّدَ، اور تنگی کے مقابل رَزَقَ۔ اہلِ معرفت کہتے ہیں کہ اپنی پچھلی کمزوری کو یاد رکھنا نفس کے غرور کی دوا ہے۔ فتح کے بعد سب سے بڑا خطرہ یہ ہے کہ آدمی اپنی قوت کو اپنی کمائی سمجھنے لگے، اسی لیے بدر کی فتح کے فوراً بعد وَاذْكُرُوْۤا کہہ کر وہ دن یاد دلائے گئے۔ شیخِ اکبر کے نزدیک بندے کی اصل حقیقت فقر ہے؛ اس کی ہر قوت اور ہر رزق ایک امانت ہے جو اسمائے الٰہیہ، القوی اور الرزاق، کا پرتو ہے۔ اور لَعَلَّكُمْ تَشْكُرُوْنَ اس یاد کا مقصد ہے: شکر یہ ہے کہ نعمت کو منعم کی طرف لوٹا دیا جائے اور اسی کے کام میں لگایا جائے۔ جو شکر کرتا ہے وہ نعمت کو محفوظ کر لیتا ہے، اور جو بھول جاتا ہے وہ اپنی اس کمزوری کو پھر سے دعوت دیتا ہے جس سے اسے نکالا گیا تھا۔

يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تَخُوْنُوا اللّٰهَ وَالرَّسُوْلَ وَتَخُوْنُوْۤا اَمٰنٰتِكُمْ وَاَنْتُمْ تَعْلَمُوْنَ۔ “اے ایمان والو! اللہ اور رسول صلی اللہ علیہ وسلم سے خیانت نہ کرو، اور نہ جانتے بوجھتے اپنی امانتوں میں خیانت کرو۔” بعض مفسرین نے اس آیت کے ساتھ حضرت ابو لبابہ رضی اللہ عنہ کا واقعہ ذکر کیا ہے۔ روایت کا مفہوم یہ ہے کہ بنو قریظہ کے محاصرے کے دنوں میں انہوں نے ان سے مشورہ کیا کہ کیا وہ حضورِ اکرم صلی اللہ علیہ وسلم کے فیصلے پر اتر آئیں؛ حضرت ابو لبابہ رضی اللہ عنہ نے ہاں کہا مگر ہاتھ سے گلے کی طرف اشارہ کر دیا، گویا انجام بتا دیا۔ روایت میں ہے کہ ابھی ان کے قدم اپنی جگہ سے ہٹے نہ تھے کہ انہیں احساس ہو گیا کہ انہوں نے اللہ اور اس کے رسول صلی اللہ علیہ وسلم سے خیانت کی ہے۔ وہ سیدھے مسجدِ نبوی گئے اور خود کو ایک ستون سے باندھ لیا، اور عہد کیا کہ جب تک اللہ ان کی توبہ قبول نہ فرمائے وہ نہ کھلیں گے؛ پھر توبہ قبول ہوئی اور حضورِ اکرم صلی اللہ علیہ وسلم نے اپنے دستِ مبارک سے انہیں کھولا۔ مفسرین یہ بھی لکھتے ہیں کہ آیت کا حکم عام ہے اور شانِ نزول کی تعیین میں روایات مختلف ہیں۔ اس واقعے کی اصل خوبصورتی اس صحابی کی بیداری ہے: لغزش ایک لمحے کی تھی، مگر احساس فوراً جاگا اور توبہ پوری جان سے ہوئی۔ اہلِ ذوق کہتے ہیں کہ جس دل میں خیانت کا ایک اشارہ بھی کانٹا بن کر چبھے وہ زندہ دل ہے؛ اور یہی اس زندگی کی ایک جھلک ہے جس کی طرف چوبیسویں آیت نے بلایا تھا۔

شیخِ اکبر کے مکتب میں امانت کا دائرہ بہت وسیع ہے۔ مال اور راز تو امانت ہیں ہی، مگر انسان کا وجود، اس کی سماعت، بصارت اور دل سب امانتیں ہیں جو اسے اسمائے الٰہیہ کا مظہر بننے کے لیے دی گئیں۔ ان امانتوں میں خیانت یہ ہے کہ انہیں اس کام کے سوا کسی کام میں لگایا جائے جس کے لیے وہ دی گئیں: کان کو حق سننے کے بجائے لغو میں، اور دل کو یادِ حق کے بجائے غیر کی محبت میں۔ اس طرح یہ آیت رکوع کی ابتدا سے جڑ جاتی ہے: جو سن کر منہ پھیرتا ہے وہ سماعت کی امانت میں خیانت کرتا ہے۔ وَاَنْتُمْ تَعْلَمُوْنَ کی قید وہی ہے جو وَاَنْتُمْ تَسْمَعُوْنَ کی تھی: جاننے کے بعد خیانت زیادہ سنگین ہے۔ پھر فرمایا: وَاعْلَمُوْۤا اَنَّمَاۤ اَمْوَالُكُمْ وَاَوْلَادُكُمْ فِتْنَةٌ ۙ وَّاَنَّ اللّٰهَ عِنْدَهٗۤ اَجْرٌ عَظِيْمٌ۔ “اور جان لو کہ تمہارے مال اور تمہاری اولاد بس ایک آزمائش ہیں، اور یہ کہ اللہ کے پاس بڑا اجر ہے۔” اس آیت کو خیانت کی ممانعت کے فوراً بعد رکھنا بتاتا ہے کہ خیانت کی جڑ اکثر یہی دو محبتیں ہوتی ہیں۔ مفسرین نے لکھا ہے کہ حضرت ابو لبابہ رضی اللہ عنہ کے اہل و مال بنو قریظہ کے درمیان تھے، اور اسی تعلق نے اس لمحے کی لغزش کا راستہ کھولا تھا؛ اسی لیے ان دونوں آیتوں کو ساتھ پڑھا جاتا ہے۔

فتنہ کا اصل معنی سونے کو آگ میں ڈالنا ہے تاکہ کھرا اور کھوٹا الگ ہو جائے۔ مال اور اولاد کو فتنہ کہنا ان کی برائی بیان کرنا نہیں؛ وہ اللہ کی نعمتیں ہیں، اور اولاد کی محبت فطرت کا حصہ ہے جسے حضورِ اکرم صلی اللہ علیہ وسلم نے رحمت فرمایا۔ آزمائش یہ ہے کہ یہ نعمتیں دل میں کس جگہ بیٹھتی ہیں: منعم کے راستے کا زادِ راہ بنتی ہیں یا اس راستے کی دیوار۔ سورۂ المنافقون میں یہی تنبیہ یوں آئی: لَا تُلْهِكُمْ اَمْوَالُكُمْ وَلَاۤ اَوْلَادُكُمْ عَنْ ذِكْرِ اللّٰهِ، “تمہارے مال اور تمہاری اولاد تمہیں اللہ کی یاد سے غافل نہ کر دیں۔” شیخِ اکبر کے نزدیک دل حق کی تجلی کا گھر ہے، اور گھر کا ادب یہ ہے کہ عزیزوں کو صحن میں جگہ ملے اور صدر مقام صاحبِ خانہ کے لیے خالی رہے۔ مال اور اولاد سے محبت رکھی جائے، ان کی خدمت عبادت سمجھ کر کی جائے، مگر انہیں وہ مقام نہ دیا جائے جہاں سے حق کی پکار سنی جاتی ہے۔ آیت کا آخری جملہ ترازو کا دوسرا پلڑا دکھاتا ہے: وَّاَنَّ اللّٰهَ عِنْدَهٗۤ اَجْرٌ عَظِيْمٌ۔ جو کچھ ہاتھ میں ہے وہ گنا ہوا اور فانی ہے، اور جو اس کے پاس ہے وہ عظیم ہے۔ اہلِ معرفت کہتے ہیں کہ جس نے اس عظیم کو دیکھ لیا اس کے لیے فانی کو امانت سمجھ کر برتنا آسان ہو جاتا ہے، اور یہی اس آزمائش میں کامیابی ہے۔
