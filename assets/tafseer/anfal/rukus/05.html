بدر کا معرکہ ختم ہو چکا ہے، قریش کا لشکر اپنے مقتولوں کو چھوڑ کر مکہ لوٹ گیا ہے، اور پچھلی آیات نے ان کے مال خرچ کرنے اور اللہ کی راہ سے روکنے کا انجام بیان کیا ہے۔ ایسے وقت میں، جب فتح بھی تازہ ہے اور زخم بھی تازہ، قرآن ایک ایسی آیت اتارتا ہے جو تلوار کی نہیں، دروازے کی بات کرتی ہے: قُلْ لِّلَّذِيْنَ كَفَرُوْۤا اِنْ يَّنْتَهُوْا يُغْفَرْ لَهُمْ مَّا قَدْ سَلَفَ ۚ وَاِنْ يَّعُوْدُوْا فَقَدْ مَضَتْ سُنَّتُ الْاَوَّلِيْنَ۔ “آپ ان کافروں سے فرما دیجیے کہ اگر وہ باز آ جائیں تو جو کچھ گزر چکا وہ انہیں معاف کر دیا جائے گا، اور اگر وہ پھر وہی کریں تو پہلے لوگوں کا دستور گزر ہی چکا ہے۔” مفسرین فرماتے ہیں کہ یہ خطاب انہی لوگوں سے ہے جو کل تک حضورِ اکرم صلی اللہ علیہ وسلم اور آپ صلی اللہ علیہ وسلم کے ساتھیوں کو ستاتے، گھروں سے نکالتے اور میدان میں مقابل آتے رہے۔ انہی کو بتایا گیا کہ ماضی کا سارا بوجھ ایک لمحے کے فیصلے سے گر سکتا ہے۔ غور کیجیے کہ آیت نے مغفرت کی شرط میں صرف باز آنا رکھا، اور مغفرت کو مجہول صیغے يُغْفَرْ میں بیان کیا، گویا معاف کرنے والا اتنا قریب اور اتنا یقینی ہے کہ اس کا نام لینے کی بھی حاجت نہیں۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اسے سبقتِ رحمت کہا جاتا ہے: رحمت غضب سے آگے ہے، اور جہاں بندہ پلٹنے کا ایک قدم اٹھاتا ہے، رحمت اس کے پیچھے کی پوری مسافت کو ڈھانپ لیتی ہے۔

تاریخ نے اس آیت کو اپنی آنکھوں سے پورا ہوتے دیکھا۔ جو لوگ بدر، احد اور خندق کے معرکوں میں مسلمانوں کے مقابل رہے تھے، ان میں سے بہت سے بعد میں اسی دروازے سے داخل ہوئے: حضرت خالد بن ولید، حضرت عکرمہ بن ابی جہل، حضرت ابوسفیان اور حضرت عمرو بن العاص رضی اللہ عنہم؛ اور حضرت وحشی رضی اللہ عنہ بھی، جن کے ہاتھ سے احد میں سید الشہدا حضرت حمزہ رضی اللہ عنہ شہید ہوئے تھے۔ صحیح مسلم میں حضرت عمرو بن العاص رضی اللہ عنہ کا واقعہ ہے، جس کا مفہوم یہ ہے کہ جب وہ اسلام لانے کے لیے حاضر ہوئے تو بیعت کے وقت ہاتھ کھینچ لیا اور یہ شرط رکھنی چاہی کہ ان کی مغفرت ہو جائے؛ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ کیا تم نہیں جانتے کہ اسلام اپنے سے پہلے کی سب چیزوں کو ڈھا دیتا ہے۔ یہ اسی آیت کی نبوی شرح ہے۔ فقہا نے بھی اسی سے یہ اصول لیا کہ جو کافر اسلام لے آئے، اس سے کفر کے زمانے کی عبادات کی قضا طلب نہیں کی جاتی اور اس زمانے کے گناہوں پر مواخذہ نہیں ہوتا۔ اہلِ معرفت کہتے ہیں کہ یہاں ایک لطیف بات ہے: آیت نے ان کے پچھلے اعمال کو مَا قَدْ سَلَفَ کہا، یعنی ماضی کو ماضی ہی رہنے دیا۔ جو بندہ لوٹ آیا، اس کی پہچان اب اس کے کل سے نہیں، اس کے آج سے ہے۔ حق اسے اسی نام سے پکارتا ہے جو اس نے ابھی اپنے لیے چنا ہے۔

شیخِ اکبر کے نزدیک یہ آیت تجددِ امثال کی ایک روشن مثال بھی ہے۔ اس مکتب میں کائنات ہر آن نئی پیدا ہوتی ہے؛ ہر لمحہ حق کی ایک نئی تجلی ہے، اور کوئی تجلی بعینہٖ دہرائی نہیں جاتی۔ جو شخص اس لمحے پلٹ آیا، وہ اس لمحے کی نئی تخلیق ہے، اور کل کا گناہ اس نئے وجود پر کندہ نقش نہیں رہتا۔ اسی لیے دوسری جگہ تائبین کے بارے میں فرمایا گیا: فَاُولٰۤئِكَ يُبَدِّلُ اللّٰهُ سَيِّاٰتِهِمْ حَسَنٰتٍ، اور سورۂ فرقان کے صفحے پر ہم اس تبدیل کا ذکر پڑھ چکے ہیں۔ بدر کے بعد کا یہ اعلان اسی قانون کا اجتماعی رخ ہے: اس پوری قوم کے لیے، جس نے مکہ میں تیرہ برس ایمان والوں کو ستایا، دروازہ بند نہیں کیا گیا۔ اہلِ ذوق کہتے ہیں کہ دروازہ کھلا رکھنا اسمِ الغفور کا تقاضا ہے، اور اس دروازے کا ذکر تلوار کے ذکر سے پہلے لانا قرآن کا ادب ہے: پہلے پکار، پھر تنبیہ۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ ظاہر میں یہ کفار کو مغفرت کی دعوت ہے، اور باطن میں ہر اس دل کے لیے بشارت ہے جو کسی پرانی عادت یا غفلت کے کسی پرانے دور سے نکل کر لوٹنا چاہتا ہے۔ اس کے لیے بھی یہی کلمہ ہے کہ رک جاؤ، تو جو گزر گیا وہ گزر گیا۔

آیت کا دوسرا حصہ اسی دروازے کا دوسرا پہلو ہے: وَاِنْ يَّعُوْدُوْا، اگر وہ پھر لوٹ کر وہی کریں۔ یہاں بھی قرآن نے یہ نہیں کہا کہ اللہ انہیں لوٹا دے گا؛ لوٹنا ان کا اپنا فعل بتایا گیا۔ پھر سُنَّتُ الْاَوَّلِيْنَ کا ذکر ہے، یعنی وہ دستور جو پچھلی قوموں کے ساتھ جاری ہوا: حضرت نوح علیہ السلام کی قوم، عاد، ثمود اور فرعون کے لوگ۔ مفسرین فرماتے ہیں کہ اس سے مراد یہ ہے کہ جو قوم انبیاء علیہم السلام کی دعوت کے مقابل ڈٹی رہی، اس پر حق کا فیصلہ آیا، اور بدر کا دن بھی اسی دستور کی ایک کڑی تھا۔ اہلِ معرفت یہاں سرِّ قدر کی جھلک دیکھتے ہیں: حکم محکوم کے مطابق ہوتا ہے۔ جو باز آیا، اس کے لیے مغفرت کا حکم؛ جو لوٹ کر پھر وہی راہ لے، اس کے لیے وہی سنت جو اس راہ کا انجام ہے۔ حق کسی پر زبردستی کوئی راہ نہیں ڈالتا؛ وہ بندے کے اختیار کیے ہوئے رخ کو وجود دیتا ہے۔ اور دونوں راستوں کا بیان ایک ہی آیت میں اس لیے آیا کہ سننے والا سمجھ لے کہ فیصلہ اب بھی اس کے اپنے ہاتھ میں ہے، اور دروازے کی چوکھٹ پر کھڑا آدمی جب تک سانس لے رہا ہے، اندر قدم رکھ سکتا ہے۔

اب وہ آیت آتی ہے جسے اس کے تاریخی سیاق ہی میں پڑھنا لازم ہے: وَقٰتِلُوْهُمْ حَتّٰى لَا تَكُوْنَ فِتْنَةٌ وَّيَكُوْنَ الدِّيْنُ كُلُّهٗ لِلّٰهِ ۚ فَاِنِ انْتَهَوْا فَاِنَّ اللّٰهَ بِمَا يَعْمَلُوْنَ بَصِيْرٌ۔ “اور ان سے لڑو یہاں تک کہ فتنہ باقی نہ رہے اور دین سارا کا سارا اللہ کا ہو جائے، پھر اگر وہ باز آ جائیں تو اللہ ان کے کاموں کو خوب دیکھنے والا ہے۔” مفسرین کے نزدیک یہاں فتنہ سے مراد وہ ظلم و جبر ہے جو مکہ کے مشرکین نے ایمان والوں پر روا رکھا: ایمان لانے والوں کو ستانا، انہیں دین سے پھیرنے کے لیے عذاب دینا اور انہیں گھروں سے نکالنا۔ حضرت عبداللہ بن عباس رضی اللہ عنہما اور دیگر سلف سے منقول ہے کہ فتنہ سے مراد شرک ہے، اور بہت سے مفسرین نے یہ وضاحت کی کہ مقصود یہ تھا کہ کوئی مومن اپنے دین کی وجہ سے ستایا نہ جائے۔ صحیح بخاری میں حضرت عبداللہ بن عمر رضی اللہ عنہما کا قول منقول ہے، جس کا مفہوم یہ ہے کہ ہم نے یہ کام حضورِ اکرم صلی اللہ علیہ وسلم کے زمانے میں کر دیا تھا، جب اسلام کم تھا اور آدمی کو اس کے دین کی وجہ سے آزمائش میں ڈالا جاتا تھا؛ اور انہوں نے یہ بات اس وقت فرمائی جب کچھ لوگ انہیں اپنے زمانے کی باہمی لڑائی میں اسی آیت کے حوالے سے شریک کرنا چاہتے تھے، اور انہوں نے انکار فرمایا۔ یہ قول خود بتاتا ہے کہ صحابۂ کرام رضی اللہ عنہم اس آیت کو ایک خاص تاریخی مرحلے کا حکم سمجھتے تھے، جو حضورِ اکرم صلی اللہ علیہ وسلم کی قیادت اور آپ صلی اللہ علیہ وسلم کے حکم کے تحت پورا ہوا۔

اسی لیے اس صفحے کا ادب یہ ہے کہ اس آیت کا حکم تاریخی ہے، اسی دور اور اسی سرزمین سے متعلق جب حضورِ اکرم صلی اللہ علیہ وسلم بنفسِ نفیس امامت فرما رہے تھے اور قتال آپ صلی اللہ علیہ وسلم ہی کے حکم سے ہوتا تھا۔ نہ کوئی فرد اپنی رائے سے اسے اپنے اوپر نافذ کر سکتا ہے، نہ اسے آج کی کسی قوم یا کسی گروہ پر چسپاں کیا جا سکتا ہے۔ آیت خود اپنی حد بتاتی ہے: فَاِنِ انْتَهَوْا، اگر وہ رک جائیں تو بات ختم؛ اور آیت کا اختتام یہ ہے کہ اللہ ان کے عمل کو دیکھ رہا ہے، یعنی دلوں کا حساب بندوں کے سپرد نہیں۔ اہلِ معرفت اس آیت کا ایک باطنی رخ بھی دیکھتے ہیں جو ظاہر کی جگہ نہیں لیتا بلکہ اس کے ساتھ چلتا ہے: ہر سالک کے اندر ایک فتنہ ہے، نفس کی وہ کشمکش جو دل کو حق سے پھیرتی ہے۔ اس اندرونی مجاہدے کا مقصد بھی یہی ہے کہ دل میں کوئی فتنہ باقی نہ رہے اور وہ پورا کا پورا حق کا ہو جائے۔ شیخِ اکبر کے نزدیک دین کے معنی میں انقیاد اور جزا دونوں شامل ہیں؛ جب دل کا ہر میلان حق کے آگے جھک جائے تو دین سارا کا سارا اللہ کا ہو جاتا ہے، اور بندے کے اندر کوئی گوشہ ایسا نہیں بچتا جہاں کسی اور کی حکومت ہو۔ اہلِ تصوف میں مشہور اثر ہے کہ نفس سے مجاہدہ بڑا جہاد ہے، اگرچہ محدثین نے اس کی سند کو کمزور کہا ہے۔

پھر وہ آیت جو تنہا رہ جانے والے کو سہارا دیتی ہے: وَاِنْ تَوَلَّوْا فَاعْلَمُوْۤا اَنَّ اللّٰهَ مَوْلٰىكُمْ ؕ نِعْمَ الْمَوْلٰى وَنِعْمَ النَّصِيْرُ۔ “اور اگر وہ منہ پھیریں تو جان رکھو کہ اللہ تمہارا کارساز ہے، کیا خوب کارساز اور کیا خوب مددگار۔” ظاہر میں یہ مسلمانوں کی تسلی ہے کہ اگر مخالفین پلٹنے سے انکار کریں تو تمہاری پشت پر وہ ہے جس نے بدر میں تمہیں تھاما۔ لفظوں کی ترتیب دیکھیے: تَوَلَّوْا اور مَوْلٰى ایک ہی مادے سے ہیں۔ جو منہ پھیرتا ہے وہ پیٹھ دے کر ولایت کا رشتہ توڑتا ہے، اور جو قریب ہو کر کام سنبھالتا ہے وہ مولیٰ ہے۔ گویا آیت یہ کہتی ہے کہ مخلوق کا رخ پھیر لینا تمہارا نقصان نہیں، کیونکہ ولایت کا اصل سرچشمہ اپنا رخ کبھی نہیں پھیرتا۔ شیخِ اکبر کے نزدیک اسمِ الولی کی تجلی بندے پر پوری طرح اسی وقت کھلتی ہے جب اسباب کے سہارے ایک ایک کر کے ہٹ جائیں؛ تب بندہ جانتا ہے کہ اسباب بھی اسی کے ہاتھ تھے، اور جو سہارا ٹوٹا وہ دراصل ایک پردہ تھا جو اٹھ گیا۔ اہلِ ذوق کہتے ہیں کہ نِعْمَ کا دو بار آنا مدح کا اعادہ ہے، جیسے دل کسی نعمت کو دیکھ کر پکار اٹھے اور پھر شکر سے دوبارہ پکارے۔

اب غنیمت کا حکم ہے، جسے علما کے بیان کے مطابق اختصار سے سمجھنا کافی ہے: وَاعْلَمُوْۤا اَنَّمَا غَنِمْتُمْ مِّنْ شَيْءٍ فَاَنَّ لِلّٰهِ خُمُسَهٗ وَلِلرَّسُوْلِ وَلِذِي الْقُرْبٰى وَالْيَتٰمٰى وَالْمَسٰكِيْنِ وَابْنِ السَّبِيْلِ ۙ۔ “اور جان رکھو کہ جو کچھ تمہیں غنیمت میں ملے، اس کا پانچواں حصہ اللہ کا ہے، اور رسول صلی اللہ علیہ وسلم کا، اور قرابت داروں، یتیموں، مسکینوں اور مسافروں کا۔” علما فرماتے ہیں کہ سورت کی پہلی آیت میں غنیمت کو اللہ اور رسول صلی اللہ علیہ وسلم کے اختیار میں بتایا گیا تھا، اور یہاں اس کی تقسیم بیان ہوئی: چار حصے لڑنے والوں کے، اور پانچواں حصہ ان مدوں کے لیے جو آیت میں مذکور ہیں۔ اکثر مفسرین کے نزدیک اللہ کے نام کا ذکر برکت اور تعظیم کے لیے ہے، اور قرابت داروں سے مراد حضورِ اکرم صلی اللہ علیہ وسلم کے رشتہ دار بنو ہاشم اور بنو مطلب ہیں۔ حضورِ اکرم صلی اللہ علیہ وسلم کے وصال کے بعد اس حصے کی تقسیم کی تفصیل میں فقہا کی آرا مختلف ہیں، جو کتبِ فقہ میں مذکور ہیں۔ یہ حکم قتال کی طرح اجتماعی نظم اور جائز حکومت سے متعلق ہے، افراد کے ہاتھ میں نہیں۔ اہلِ معرفت اس میں یہ اشارہ پاتے ہیں کہ جو کچھ ہاتھ آئے، اس میں کمزوروں کا حصہ پہلے سے لکھا ہوا ہے؛ یتیم اور مسافر کا حق ادا ہونے کے بعد ہی بندے کا اپنا حصہ پاک ہوتا ہے، اور جو دیتا ہے وہ دراصل امانت لوٹاتا ہے۔

آیت کا دوسرا حصہ اس حکم کو ایمان سے جوڑتا ہے: اِنْ كُنْتُمْ اٰمَنْتُمْ بِاللّٰهِ وَمَآ اَنْزَلْنَا عَلٰى عَبْدِنَا يَوْمَ الْفُرْقَانِ يَوْمَ الْتَقَى الْجَمْعٰنِ ؕ وَاللّٰهُ عَلٰى كُلِّ شَيْءٍ قَدِيْرٌ۔ “اگر تم اللہ پر ایمان رکھتے ہو اور اس پر جو ہم نے اپنے بندے، حضورِ اکرم صلی اللہ علیہ وسلم، پر فیصلے کے دن اتارا، جس دن دونوں جماعتیں آمنے سامنے ہوئیں، اور اللہ ہر چیز پر قادر ہے۔” مفسرین کے نزدیک یومِ فرقان بدر کا دن ہے، سترہ رمضان دو ہجری، جب تین سو تیرہ کے قریب بے سروسامان مسلمان ایک ہزار کے قریب مسلح لشکر کے سامنے کھڑے ہوئے، اور حق و باطل کے درمیان وہ فرق ظاہر ہوا جس نے تاریخ کا رخ بدل دیا۔ پچھلے رکوع میں ہم نے پڑھا کہ تقویٰ والوں کو فرقان دیا جاتا ہے؛ یہاں وہی فرقان ایک دن کی صورت میں دکھایا گیا۔ اہلِ معرفت کہتے ہیں کہ جو فرقان دل میں اترتا ہے وہی کبھی تاریخ کے میدان میں بھی ظاہر ہوتا ہے؛ باطن کی تمیز اور ظاہر کا فیصلہ ایک ہی نور کے دو رخ ہیں۔ اور سب سے لطیف بات یہ ہے کہ اس عظیم دن کے ذکر میں حضورِ اکرم صلی اللہ علیہ وسلم کو عَبْدِنَا کہا گیا۔ شیخِ اکبر کے نزدیک عبدیت انسانِ کامل کا سب سے بلند مقام ہے، اور حق نے اپنے حبیب صلی اللہ علیہ وسلم کو سب سے بڑے مواقع پر اسی نسبت سے یاد فرمایا: معراج کی رات سُبْحٰنَ الَّذِيْۤ اَسْرٰى بِعَبْدِهٖ، جیسا کہ سورۂ بنی اسرائیل کے صفحے پر گزرا، اور فتحِ بدر کے دن بھی عَبْدِنَا۔ فتح کے دن بندگی کا ذکر یہ سکھاتا ہے کہ جیت بندے کی نہیں، مولیٰ کی ہے۔

پھر میدان کا نقشہ کھینچا گیا: اِذْ اَنْتُمْ بِالْعُدْوَةِ الدُّنْيَا وَهُمْ بِالْعُدْوَةِ الْقُصْوٰى وَالرَّكْبُ اَسْفَلَ مِنْكُمْ ؕ وَلَوْ تَوَاعَدْتُّمْ لَاخْتَلَفْتُمْ فِي الْمِيْعٰدِ ۙ۔ “جب تم وادی کے قریب والے کنارے پر تھے اور وہ دور والے کنارے پر، اور قافلہ تم سے نیچے کی طرف تھا؛ اور اگر تم آپس میں وقت طے کرتے تو اس وعدے میں ضرور اختلاف کرتے۔” مفسرین بیان کرتے ہیں کہ مسلمان وادیِ بدر کے اس کنارے پر تھے جو مدینہ منورہ کی طرف تھا، اور قریش کا لشکر اس کنارے پر جو مکہ کی طرف تھا، اور قریش کا تجارتی قافلہ حضرت ابوسفیان رضی اللہ عنہ کی سرکردگی میں، جو بعد میں فتحِ مکہ کے موقع پر اسلام لائے، ساحلِ سمندر کی طرف نشیب میں نکل رہا تھا۔ مسلمان قافلے کے ارادے سے نکلے تھے، لشکر سے مقابلے کا انہیں گمان نہ تھا؛ اور قریش قافلہ بچانے نکلے تھے۔ اگر دونوں فریق آپس میں مقابلے کا دن طے کرتے تو مسلمانوں کی قلتِ تعداد اور دوسری طرف کی ہچکچاہٹ کے سبب کوئی نہ کوئی پیچھے ہٹ جاتا۔ مگر دونوں کو بغیر کسی وعدے کے ایک ہی وادی میں لا کھڑا کیا گیا۔ اہلِ معرفت کہتے ہیں کہ بندے کی تدبیر اکثر اختلاف پر جا کر رک جاتی ہے، اور حق کی تدبیر وہاں سے شروع ہوتی ہے جہاں بندے کا وعدہ نہیں پہنچتا۔ جو ملاقاتیں زندگی کا رخ بدل دیتی ہیں، وہ کم ہی کسی جنتری میں لکھی ہوتی ہیں۔

پھر وہ حکمت بیان ہوئی جس کے لیے یہ سب ہوا: وَلٰكِنْ لِّيَقْضِيَ اللّٰهُ اَمْرًا كَانَ مَفْعُوْلًا ۙ لِّيَهْلِكَ مَنْ هَلَكَ عَنْۢ بَيِّنَةٍ وَّيَحْيٰى مَنْ حَيَّ عَنْۢ بَيِّنَةٍ ؕ وَاِنَّ اللّٰهَ لَسَمِيْعٌ عَلِيْمٌ۔ “لیکن تاکہ اللہ اس کام کو پورا کر دے جو ہو کر رہنا تھا، تاکہ جسے ہلاک ہونا ہے وہ روشن دلیل دیکھ کر ہلاک ہو اور جسے زندہ رہنا ہے وہ روشن دلیل دیکھ کر زندہ رہے، اور بے شک اللہ سب کچھ سننے والا، سب کچھ جاننے والا ہے۔” مفسرین، جیسے امام محمد بن اسحاق اور امام ابنِ جریر طبری رحمہما اللہ، کے بیان کا مفہوم یہ ہے کہ بدر کا دن ایک کھلی نشانی بن گیا: ایک چھوٹی بے سروسامان جماعت کا ایک بڑے لشکر پر غالب آنا ایسی دلیل تھی جس کے بعد کسی کے لیے عذر باقی نہ رہا۔ اب جو کفر پر رہا وہ دلیل دیکھ کر رہا، اور جو ایمان لایا وہ دلیل دیکھ کر لایا۔ یہاں ہلاکت اور زندگی سے مراد ظاہر میں موت و حیات بھی ہے، اور بہت سے مفسرین کے نزدیک کفر و ایمان بھی، جیسا کہ قرآن ایمان کو زندگی اور کفر کو موت سے تعبیر کرتا ہے۔ آیت کا آغاز بھی قابلِ غور ہے: اَمْرًا كَانَ مَفْعُوْلًا، ایسا کام جو گویا پہلے ہی ہو چکا تھا۔ حق کے علم میں جو طے ہے، وہ وقت پر ظاہر ہو کر رہتا ہے، اور بندوں کے وعدے اور ارادے اس کے ظہور کے راستے بن جاتے ہیں۔

یہ آیت اس سورت میں اصولِ ہدایت کی ایک بنیادی گواہی ہے، اور اسے اسی طرح پڑھنا چاہیے۔ حق تعالیٰ کسی کو اندھیرے میں ہلاک نہیں کرتا؛ ہلاکت سے پہلے بَيِّنَة، یعنی روشن دلیل، سامنے رکھی جاتی ہے۔ جو ہلاک ہوا وہ دلیل دیکھنے کے بعد منہ پھیر کر ہلاک ہوا، اور یہی منہ پھیرنا اس کی گمراہی تھی۔ کوئی ڈھونڈنے والا بدر کے دن کے بعد ہدایت سے محروم نہیں رہا؛ جس نے دیکھنا چاہا اس نے دیکھا، اور اسی دن کی نشانی نے بعد میں بہت سوں کو اسلام کی طرف کھینچا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم علم کے تابع ہے اور علم معلوم کے تابع؛ حق ہر عین کو وہی دیتا ہے جو اس کی اپنی حقیقت اور اس کا اپنا رخ مانگتا ہے، اور بَيِّنَة وہ آئینہ ہے جس میں ہر ایک کا رخ ظاہر ہو جاتا ہے۔ بدر کا دن کسی کے رخ کو زبردستی بدلنے والا جبر نہ تھا، بلکہ ایک روشنی تھی جس میں ہر رخ اپنی اصل صورت میں نظر آ گیا۔ اسی لیے آیت کا اختتام سَمِيْعٌ عَلِيْمٌ پر ہوا: وہ ہر دل کی پکار سنتا ہے اور ہر دل کی طلب جانتا ہے، اور کسی طالب کی پکار ان سنی نہیں رہتی۔ ہدایت اسی کو ملتی ہے جو مانگے، اور مانگنے والا کبھی خالی نہیں لوٹایا جاتا۔

اہلِ تاویل نے دونوں کناروں میں بھی ایک اشارہ پایا ہے، جو تاریخ کی جگہ نہیں لیتا بلکہ اس پر روشنی ڈالتا ہے۔ الْعُدْوَةِ الدُّنْيَا قریب والا کنارہ ہے اور الْعُدْوَةِ الْقُصْوٰى دور والا۔ مومن قریب والے کنارے پر تھے، اس شہر کی سمت جہاں حضورِ اکرم صلی اللہ علیہ وسلم کا مسکن تھا؛ گویا قرب ہی ان کا مقام تھا۔ اہلِ ذوق کہتے ہیں کہ سالک کی وادی کے بھی دو کنارے ہیں: ایک وہ جو حق کی طرف ہے، اور ایک وہ جو نفس کی دوری کی طرف۔ اور بیچ میں نشیب کی طرف جاتا ہوا قافلہ ہے، یعنی دنیا کا وہ سامان جس کے پیچھے آدمی نکلتا ہے۔ مسلمان قافلے کی طلب میں نکلے تھے اور حق نے انہیں قافلے سے کہیں بڑی چیز دی: فرقان کا دن۔ یہ بندے کی چھوٹی طلب پر حق کا بڑا عطیہ ہے؛ اور اس میں صحابۂ کرام رضی اللہ عنہم یا ان کے ارادے کی کوئی تنقیص نہیں، کیونکہ قافلے کا قصد بھی ان کے لیے جائز تھا اور ان کے چھینے ہوئے مال کی بازیابی تھا۔ سالک کے لیے سبق یہ ہے کہ وہ اپنی طلب کو حق کے حوالے کر دے؛ کبھی وہ جو مانگتا ہے وہ نہیں ملتا، اس لیے کہ اسے اس سے بہتر ملنا ہوتا ہے۔

اب ان دو مناظر کا ذکر ہے جن میں حق نے آنکھوں کو اپنی تدبیر کا ذریعہ بنایا: اِذْ يُرِيْكَهُمُ اللّٰهُ فِيْ مَنَامِكَ قَلِيْلًا ؕ وَّلَوْ اَرٰىكَهُمْ كَثِيْرًا لَّفَشِلْتُمْ وَلَتَنَازَعْتُمْ فِي الْاَمْرِ وَلٰكِنَّ اللّٰهَ سَلَّمَ ؕ اِنَّهٗ عَلِيْمٌۢ بِذَاتِ الصُّدُوْرِ۔ “جب اللہ آپ کو آپ کے خواب میں انہیں تھوڑا دکھا رہا تھا، اور اگر انہیں زیادہ دکھاتا تو تم ہمت ہار جاتے اور معاملے میں جھگڑنے لگتے، لیکن اللہ نے بچا لیا؛ بے شک وہ سینوں کی باتوں کو خوب جانتا ہے۔” مفسرین، جن میں امام مجاہد رحمۃ اللہ علیہ بھی ہیں، بیان کرتے ہیں کہ حضورِ اکرم صلی اللہ علیہ وسلم نے خواب میں ان کی تعداد تھوڑی دیکھی اور صحابۂ کرام رضی اللہ عنہم کو بتایا تو ان کے دل مضبوط ہو گئے۔ انبیاء علیہم السلام کا خواب وحی ہے، اور اس میں کوئی خلافِ واقعہ بات نہیں ہوتی۔ علما نے وضاحت کی ہے کہ خواب میں ان کی کمی ان کی حقیقی حیثیت کی تعبیر تھی: گنتی میں وہ زیادہ تھے، مگر حق کے مقابل ان کی قوت تھوڑی تھی اور ان کا انجام شکست تھا۔ سورۂ یوسف کے صفحے پر ہم خواب اور تعبیر کا یہ اصول پڑھ چکے ہیں کہ خواب معنی کو صورت کا لباس پہنا کر دکھاتا ہے۔ شیخِ اکبر کے نزدیک خواب عالمِ خیال کا دروازہ ہے، ایک برزخ جہاں معانی جسم پہنتے ہیں؛ وہاں ان کی کمی دراصل ان کی بے وزنی کی صورت تھی، اور جو آنکھ نبوت کی ہو وہ صورت میں معنی ہی دیکھتی ہے۔

آیت میں دو لفظ قابلِ غور ہیں جو اگلے رکوع کی تمہید بھی ہیں: لَفَشِلْتُمْ اور لَتَنَازَعْتُمْ، یعنی ہمت ہارنا اور آپس میں جھگڑنا۔ اگلے رکوع میں یہی دونوں لفظ حکم کی صورت میں لوٹ کر آئیں گے۔ گویا حق نے پہلے خواب کے ذریعے ان دونوں آفتوں سے بچایا، اور پھر انہیں سکھایا کہ وہ خود بھی ان سے بچیں؛ پہلے عطا، پھر تعلیم۔ پھر فرمایا کہ اللہ نے سلامت رکھا؛ اسمِ السلام کی تجلی یہاں دلوں کی حفاظت کی صورت میں ہوئی۔ اہلِ معرفت کہتے ہیں کہ کمزوری اور اختلاف دونوں دل کے اندر سے پیدا ہوتے ہیں، اسی لیے اختتام بِذَاتِ الصُّدُوْرِ پر ہوا: وہ جانتا ہے کہ کس سینے میں کتنا خوف چھپا ہے، اور اسی کے مطابق اپنے بندوں کو اتنا ہی دکھاتا ہے جتنا وہ اٹھا سکیں۔ یہ اس کی رحمت کا ادب ہے کہ وہ ہر ایک پر اس کی استعداد کے مطابق کھولتا ہے، اور کسی دل پر وہ بوجھ نہیں ڈالتا جو اسے توڑ دے۔

پھر خواب کا منظر بیداری میں بھی دہرایا گیا: وَاِذْ يُرِيْكُمُوْهُمْ اِذِ الْتَقَيْتُمْ فِيْۤ اَعْيُنِكُمْ قَلِيْلًا وَّيُقَلِّلُكُمْ فِيْۤ اَعْيُنِهِمْ لِيَقْضِيَ اللّٰهُ اَمْرًا كَانَ مَفْعُوْلًا ؕ وَاِلَى اللّٰهِ تُرْجَعُ الْاُمُوْرُ۔ “اور جب تمہارے آمنے سامنے ہونے کے وقت وہ انہیں تمہاری آنکھوں میں تھوڑا دکھا رہا تھا اور تمہیں ان کی آنکھوں میں تھوڑا کر رہا تھا، تاکہ اللہ اس کام کو پورا کر دے جو ہو کر رہنا تھا، اور سارے معاملات اللہ ہی کی طرف لوٹائے جاتے ہیں۔” مفسرین نے حضرت عبداللہ بن مسعود رضی اللہ عنہ سے روایت کیا ہے، جس کا مفہوم یہ ہے کہ انہوں نے اپنے پہلو کے ایک ساتھی سے پوچھا: کیا تم انہیں ستر سمجھتے ہو؟ اس نے کہا: میرا خیال ہے سو ہوں گے؛ حالانکہ وہ ایک ہزار کے قریب تھے۔ دوسری طرف قریش کی آنکھوں میں مسلمان کم دکھائے گئے تاکہ وہ لوٹ نہ جائیں اور مقابلہ ہو کر رہے۔ یوں دونوں کی آنکھیں ایک ہی تقدیر کے پورا ہونے کا ذریعہ بن گئیں۔ شیخِ اکبر کے نزدیک دیکھنا بھی حق کی ایک تجلی ہے؛ آنکھ اپنے بل پر کچھ نہیں دیکھتی، دیکھنے کی قوت اسی کی عطا ہے، اور وہ چاہے تو ایک ہی منظر کو دو آنکھوں میں دو طرح دکھائے۔ آیت کا اختتام وَاِلَى اللّٰهِ تُرْجَعُ الْاُمُوْرُ اس سارے بیان کی جڑ ہے: خواب اور بیداری، قلت اور کثرت، سب اسی کی طرف لوٹتے ہیں۔ سالک کے لیے سبق یہ ہے کہ وہ اپنی آنکھ پر نہیں، آنکھ دینے والے پر بھروسا کرے۔
