پچھلا رکوع بدر کے میدان میں شیطان کے الٹے پاؤں پھر جانے پر ختم ہوا تھا؛ یہ رکوع اسی دن کی ایک اور آواز سناتا ہے جو میدان کے کنارے سے اٹھی تھی: اِذْ يَقُوْلُ الْمُنٰفِقُوْنَ وَالَّذِيْنَ فِيْ قُلُوْبِهِمْ مَّرَضٌ غَرَّ هٰۤؤُلَآءِ دِيْنُهُمْ ۗ وَمَنْ يَّتَوَكَّلْ عَلَى اللّٰهِ فَاِنَّ اللّٰهَ عَزِيْزٌ حَكِيْمٌ۔ “جب منافق اور وہ لوگ جن کے دلوں میں بیماری تھی کہہ رہے تھے کہ ان لوگوں کو ان کے دین نے دھوکے میں ڈال رکھا ہے؛ اور جو اللہ پر بھروسا کرے تو بےشک اللہ غالب ہے، حکمت والا ہے۔” مفسرین بیان کرتے ہیں کہ جب مٹھی بھر مسلمان بے سروسامانی کے عالم میں ایک بڑے اور مسلح لشکر کے مقابل نکلے تو کچھ کمزور دلوں نے سمجھا کہ دین کا جوش انہیں ہلاکت کی طرف لیے جا رہا ہے۔ یہ ایک تاریخی واقعے کا بیان ہے، اور اس کا دروازہ کسی کی طرف انگلی اٹھانے کے لیے نہیں، اپنے اندر جھانکنے کے لیے کھلتا ہے۔ اہلِ معرفت کہتے ہیں کہ دل کا مرض وہ کیفیت ہے جس میں آنکھ صرف اسباب کو دیکھتی ہے؛ ایسی آنکھ کو توکل دھوکا اور یقین دیوانگی دکھائی دیتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہ بات بار بار آتی ہے کہ ہر دیکھنے والا وہی دیکھتا ہے جو اس کی استعداد اسے دکھاتی ہے؛ جس دل میں اسباب کے سوا کچھ نہیں، وہ مسبب الاسباب پر بھروسے کو فریب ہی کہے گا۔ سالک کے لیے سوال یہ نہیں کہ اس دن یہ بات کس نے کہی تھی؛ سوال یہ ہے کہ کیا میرے اندر بھی کوئی آواز ہے جو ہر بھروسے کو نادانی اور ہر قربانی کو خسارہ کہتی ہے۔

آیت کا دوسرا حصہ جواب ہے، اور یہ جواب کوئی بحث نہیں بلکہ اسمائے الٰہیہ کی ایک حقیقت ہے: وَمَنْ يَّتَوَكَّلْ عَلَى اللّٰهِ فَاِنَّ اللّٰهَ عَزِيْزٌ حَكِيْمٌ۔ طعنہ دینے والوں نے کہا تھا کہ دین نے دھوکا دیا؛ قرآن نے ان سے مناظرہ نہیں کیا، صرف یہ بتایا کہ بھروسا کس پر ہے۔ جس کا بھروسا اللہ پر ہے وہ عزیز کے سہارے ہے، یعنی اس ذات کے سہارے جس پر کوئی غالب نہیں آ سکتا؛ اور حکیم کے سہارے ہے، یعنی اس ذات کے جس کا ہر فیصلہ اپنی جگہ پر ہے، چاہے اس کی حکمت بندے کو اس وقت دکھائی نہ دے۔ شیخِ اکبر کے نزدیک توکل اسباب کا انکار نہیں، اسباب کو ان کی صحیح جگہ پر رکھنا ہے: سبب ایک پردہ ہے اور حقیقی فاعل پردے کے پیچھے۔ اسی لیے صحابۂ کرام رضی اللہ عنہم نے تیاری بھی کی اور بھروسا بھی کیا؛ ہاتھ سبب پر تھا اور دل مسبب پر۔ اہلِ ذوق کہتے ہیں کہ ان دو اسموں کا جوڑ متوکل کے لیے دو تسلیاں ہیں: عزیز کہتا ہے کہ میرے چاہے بغیر تمہیں کوئی زیر نہیں کر سکتا، اور حکیم کہتا ہے کہ اگر بظاہر کوئی آزمائش آئے بھی تو وہ بے مقصد نہیں۔ یوں متوکل ہر حال میں محفوظ ہے: کامیابی میں عزت کے سائے میں اور آزمائش میں حکمت کے سائے میں۔ جو دل یہ دونوں اسم پہچان لے، اسے کوئی طعنہ اپنی جگہ سے نہیں ہلا سکتا۔

اب منظر دنیا سے اٹھ کر موت کی گھڑی میں داخل ہوتا ہے: وَلَوْ تَرٰۤى اِذْ يَتَوَفَّى الَّذِيْنَ كَفَرُوا ۙ الْمَلٰٓئِكَةُ يَضْرِبُوْنَ وُجُوْهَهُمْ وَاَدْبَارَهُمْ ۚ وَذُوْقُوْا عَذَابَ الْحَرِيْقِ۔ “اور کاش آپ دیکھیں جب فرشتے کافروں کی جانیں قبض کرتے ہیں، ان کے چہروں اور پیٹھوں پر مارتے ہیں، اور کہتے ہیں کہ جلنے کا عذاب چکھو۔” ذٰلِكَ بِمَا قَدَّمَتْ اَيْدِيْكُمْ وَاَنَّ اللّٰهَ لَيْسَ بِظَلَّامٍ لِّلْعَبِيْدِ۔ “یہ اس کا بدلہ ہے جو تمہارے ہاتھوں نے آگے بھیجا، اور یہ کہ اللہ اپنے بندوں پر ہرگز ظلم کرنے والا نہیں۔” مفسرین میں سے بعض نے اسے بدر کے دن مارے جانے والوں سے متعلق کہا اور بعض نے ہر منکر کی موت کا عمومی بیان؛ دونوں صورتوں میں اصل بات ایک ہے: موت پردہ اٹھا دیتی ہے۔ چہرے اور پیٹھ کا ذکر قابلِ غور ہے؛ اہلِ تاویل کہتے ہیں کہ جس نے زندگی بھر حق کی طرف چہرہ نہ کیا اور اس سے پیٹھ پھیرے رکھی، اس کی وہی دونوں سمتیں اس پر گواہ بن گئیں۔ آیت ۵۱ کا اصل وزن بِمَا قَدَّمَتْ اَيْدِيْكُمْ پر ہے: عذاب باہر سے لایا ہوا کوئی اجنبی بوجھ نہیں، ہاتھوں کا آگے بھیجا ہوا سامان ہے۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کا ایک رخ ہے: حکم محکوم کے مطابق؛ جزا عمل ہی کی صورت ہے جو وقت آنے پر ظاہر ہو جاتی ہے۔ اسی لیے آیت عدل کے اعلان پر ختم ہوئی۔ لفظ ظلام مبالغے کا صیغہ ہے، اور علما کہتے ہیں کہ یہاں نفی میں تاکید ہے: ظلم کا ادنیٰ ذرہ بھی حق کی شان سے دور ہے۔

پھر تاریخ کا آئینہ سامنے رکھا گیا: كَدَاْبِ اٰلِ فِرْعَوْنَ ۙ وَالَّذِيْنَ مِنْ قَبْلِهِمْ ۚ كَفَرُوْا بِاٰيٰتِ اللّٰهِ فَاَخَذَهُمُ اللّٰهُ بِذُنُوْبِهِمْ ۗ اِنَّ اللّٰهَ قَوِيٌّ شَدِيْدُ الْعِقَابِ۔ “ان کا حال فرعون والوں اور ان سے پہلے لوگوں جیسا ہے؛ انہوں نے اللہ کی آیات کا انکار کیا تو اللہ نے انہیں ان کے گناہوں کے سبب پکڑ لیا؛ بےشک اللہ قوت والا، سخت سزا دینے والا ہے۔” لفظ دأب کا اصل معنی کسی کام میں مسلسل لگے رہنا ہے، پھر یہ عادت اور روش کے لیے بولا جانے لگا۔ گویا انکار ان لوگوں کے لیے ایک لمحے کی لغزش نہیں تھا، ایک مستقل روش تھی، اور روش ہی پکڑ کا سبب بنی۔ یہاں بھی آیت نے سبب صاف بتا دیا: فَاَخَذَهُمُ اللّٰهُ بِذُنُوْبِهِمْ؛ پکڑ گناہوں کے سبب ہے، بلا سبب نہیں۔ حضرت موسیٰ علیہ السلام کی دعوت برسوں فرعون کے دروازے پر دستک دیتی رہی؛ پکڑ اس طویل مہلت کے بعد آئی۔ اہلِ معرفت کہتے ہیں کہ فرعونیت ایک مزاج ہے جو آیات دیکھ کر بھی اپنی بڑائی سے نہیں اترتا؛ اور سالک کے لیے کارآمد بات یہ ہے کہ وہ اس مزاج کو تاریخ کے صفحوں میں نہیں، اپنے نفس کے کونوں میں تلاش کرے۔ شیخِ اکبر کے نزدیک اللہ کی ہر آیت ایک تجلی ہے، اور تجلی کا انکار اپنی آنکھ پر پردہ ڈالنا ہے؛ تجلی اپنی جگہ قائم رہتی ہے، محروم وہی ہوتا ہے جس نے منہ موڑا۔ اور اِنَّ اللّٰهَ قَوِيٌّ شَدِيْدُ الْعِقَابِ جلال کا ظہور ہے، مگر یہ جلال اسی پر کھلتا ہے جو جمال کے سارے دروازے خود اپنے اوپر بند کر چکا ہو۔

اب وہ آیت آتی ہے جو اس سورت کا بنیادی اصول ہے اور جس پر یہ پورا رکوع گھومتا ہے: ذٰلِكَ بِاَنَّ اللّٰهَ لَمْ يَكُ مُغَيِّرًا نِّعْمَةً اَنْعَمَهَا عَلٰى قَوْمٍ حَتّٰى يُغَيِّرُوْا مَا بِاَنْفُسِهِمْ ۙ وَاَنَّ اللّٰهَ سَمِيْعٌ عَلِيْمٌ۔ “یہ اس لیے کہ اللہ کسی نعمت کو، جو اس نے کسی قوم کو عطا کی ہو، بدلنے والا نہیں جب تک وہ لوگ خود اپنی اس حالت کو نہ بدل دیں جو ان کے اندر ہے؛ اور بےشک اللہ سننے والا، جاننے والا ہے۔” یہی اصل اصول ہے، اور اسے اسی حیثیت سے پڑھنا چاہیے۔ پچھلی دو آیتوں میں پکڑ کا ذکر تھا؛ یہ آیت بتاتی ہے کہ پکڑ کی جڑ کہاں ہے۔ ترتیب دیکھیے: پہلے نعمت ہے، اور نعمت حق کی طرف سے پہل ہے، کسی استحقاق کے بغیر؛ پھر بندے کی طرف سے تبدیلی ہے؛ پھر اسی تبدیلی کے مطابق حق کا حکم۔ یعنی آغاز ہمیشہ رحمت سے ہوتا ہے، اور زوال کا آغاز ہمیشہ بندے کے اندر سے۔ شیخِ اکبر کے نزدیک یہ سبقتِ رحمت کا اعلان ہے: وجود خود پہلی نعمت ہے، اور حق کسی سے اپنی نعمت واپس لینے میں پہل نہیں کرتا۔ حق تعالیٰ کی طرف سے فیض کبھی منقطع نہیں ہوتا؛ جو بدلتا ہے وہ قبول کرنے والے کا ظرف اور اس کا رخ ہے۔ اسی اصول پر اس پورے سلسلے کی بنیاد ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا، اور کسی طالب سے ہدایت کی نعمت نہیں چھینتا؛ نعمت اسی وقت اٹھتی ہے جب بندہ خود منہ موڑ لے، اور وہ منہ موڑنا ہی اس کی محرومی ہے۔

یہی اصول سورۂ رعد میں ذرا وسیع لفظوں میں آیا ہے: اِنَّ اللّٰهَ لَا يُغَيِّرُ مَا بِقَوْمٍ حَتّٰى يُغَيِّرُوْا مَا بِاَنْفُسِهِمْ۔ دونوں آیتوں کو ساتھ رکھ کر پڑھیے تو ایک ہی قانون کے دو رخ سامنے آتے ہیں۔ سورۂ رعد میں اس حالت کا ذکر ہے جو کسی قوم کے ساتھ ہے، اور یہ ہر حالت کو شامل ہے، اچھی ہو یا کٹھن؛ یہاں خاص اس نعمت کا ذکر ہے جو ناشکری سے چھن جاتی ہے۔ مفسرین نے لکھا ہے کہ رعد کی آیت دونوں سمتوں میں کھلتی ہے: جو اپنی کٹھن حالت کو اندر سے بدل دے، حق اس کی حالت بھی بدل دیتا ہے۔ یوں انفال کی آیت تنبیہ ہے اور رعد کی آیت امید، اور دونوں مل کر ایک کامل قانون بناتی ہیں۔ اہلِ معرفت کہتے ہیں کہ دونوں جگہ ایک ٹکڑا مشترک ہے، حَتّٰى يُغَيِّرُوْا مَا بِاَنْفُسِهِمْ، اور یہی ساری کنجی ہے: تبدیلی کا دروازہ نفس کے اندر کھلتا ہے، باہر نہیں۔ شیخِ اکبر کے مکتب میں کہا جاتا ہے کہ ظاہر باطن کا عکس ہے؛ جو کچھ آفاق میں بدلتا ہے، اس کی پہلی لکیر انفس میں کھنچتی ہے۔ اس لیے سالک کا کام حالات کی شکایت سے پہلے اپنے اندر کی خبر لینا ہے۔ اور یہ اصول اپنے اوپر لاگو کرنے کے لیے ہے؛ اسے کسی دوسرے فرد یا کسی قوم پر فیصلہ صادر کرنے کا ہتھیار بنانا اس کی روح کے خلاف ہے۔

اس آیت میں سرِّ قدر کا خلاصہ بھی چھپا ہے۔ شیخِ اکبر کے نزدیک حق تعالیٰ کا حکم ہر شے پر اسی کے مطابق ہوتا ہے جو وہ شے اپنی حقیقت میں ہے؛ علم معلوم کے تابع ہے، اور حکم محکوم کے مطابق۔ حق نے نعمت دی؛ بندے نے اسے شکر کے بجائے ناشکری سے، حضوری کے بجائے غفلت سے، اور طلب کے بجائے اعراض سے بدل دیا؛ تب حکم بھی بدل گیا۔ اس میں نہ کوئی جبر ہے نہ کوئی ظلم؛ یہ اس رخ کو وجود دینا ہے جو بندے نے خود اختیار کیا۔ اسی لیے یہ آیت ہدایت اور گمراہی کے پورے مسئلے کی کلید ہے۔ قرآن میں جہاں کہیں آتا ہے کہ اللہ نے کسی کے دل پھیر دیے، اسے اس آیت کی روشنی میں پڑھا جائے: پہلے انہوں نے اپنے اندر کی حالت بدلی، پھر حق نے اسی تبدیلی کو ثبات دیا، جیسا کہ سورۂ صف میں آیا کہ جب وہ خود ٹیڑھے ہوئے تو اللہ نے ان کے دل ٹیڑھے کر دیے۔ ترتیب ہمیشہ یہی ہے: پہلے بندے کا اعراض، پھر اسی کے مطابق حکم۔ حق چاہتا تو سب کو مجبور کر کے ہدایت پر رکھ دیتا، مگر اس نے اختیار کی عزت رکھی؛ جبری شکر شکر نہیں، اور جبری ایمان ایمان نہیں۔ اور جو طالب بن کر آئے، اس سے نعمت کبھی نہیں چھینی جاتی، کیونکہ طلب خود اندر کی وہ تبدیلی ہے جو نعمت کو بڑھاتی ہے۔

اہلِ ذوق کہتے ہیں کہ نعمتیں دو طرح کی ہیں: ظاہر کی نعمتیں، جیسے امن، رزق اور عزت؛ اور باطن کی نعمتیں، جیسے ایمان کی حلاوت، نماز کا حضور، ذکر کی لذت اور دل کی نرمی۔ سالک اکثر شکایت کرتا ہے کہ پہلے جو ذوق تھا وہ نہیں رہا، عبادت میں جو آنسو تھے وہ خشک ہو گئے۔ یہ آیت اسے سیدھا اپنے اندر بھیجتی ہے: دینے والے کا ہاتھ نہیں بدلا، دیکھو تمہارے اندر کیا بدلا ہے؛ کوئی نگاہ جو آوارہ ہو گئی، کوئی زبان جو بے احتیاط ہو گئی، کوئی حق جو ادا نہ ہوا، کوئی دل جو کسی مخلوق سے اٹک گیا۔ شیخِ اکبر کے نزدیک تجددِ امثال کا قانون ہر لمحہ جاری ہے: حق ہر آن عالم کو نئے سرے سے وجود دیتا ہے، اور ہر آن نعمت بھی نئے سرے سے اترتی ہے۔ ہر نیا لمحہ اسی ظرف میں اترتا ہے جو بندے نے پچھلے لمحے میں تیار کیا تھا؛ اس لیے نعمت کا تسلسل بندے کے رخ کے تسلسل پر موقوف ہے۔ اسی بات کو سورۂ ابراہیم نے شکر اور زیادتی کے قانون کے طور پر بیان کیا: لَئِنْ شَكَرْتُمْ لَاَزِيْدَنَّكُمْ۔ شکر نعمت کو باندھ لیتا ہے، اور ناشکری وہ پہلی تبدیلی ہے جو اندر سے شروع ہوتی ہے اور باہر تک پہنچتی ہے۔

آیت کا خاتمہ وَاَنَّ اللّٰهَ سَمِيْعٌ عَلِيْمٌ پر ہوا، اور یہ خاتمہ اپنے اندر گہری تسلی رکھتا ہے۔ وہ سننے والا ہے، یعنی نفس کے اندر اٹھنے والی ہر آہ اور ہر ارادہ اس تک پہنچتا ہے؛ وہ جاننے والا ہے، یعنی تبدیلی کی پہلی جنبش، جو ابھی عمل بھی نہیں بنی، اس کے علم میں ہے۔ اس کا مطلب یہ ہے کہ جس طرح زوال اندر سے شروع ہوتا ہے، اسی طرح واپسی بھی اندر سے شروع ہو سکتی ہے؛ اور جو اندر سے پلٹے، اس کی پہلی آہ ہی سن لی جاتی ہے۔ دروازہ دونوں طرف کھلتا ہے۔ اہلِ معرفت یہ بھی کہتے ہیں کہ اس آیت کو دوسروں کے احوال پر فتوے لگانے کا آلہ نہ بنایا جائے؛ کسی شخص کی مصیبت دیکھ کر یہ کہنا کہ اس نے ضرور اپنے اندر کچھ بگاڑا ہوگا، اس آیت کے ادب کے خلاف ہے، کیونکہ آزمائش درجات کی بلندی کے لیے بھی آتی ہے۔ یہ آیت آئینہ ہے، ترازو نہیں: اسے اپنے سامنے رکھا جائے، دوسروں کو اس پر نہ تولا جائے۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ظاہر میں یہ قریش کی اس نعمت کا ذکر ہے جو حرم کی امان اور حضورِ اکرم صلی اللہ علیہ وسلم کی بعثت کی صورت میں ان کے پاس آئی، اور باطن میں ہر دل کی وہ نعمت جو اسے حضوری کی صورت میں ملی۔

پھر وہی آئینہ دوبارہ سامنے لایا گیا: كَدَاْبِ اٰلِ فِرْعَوْنَ ۙ وَالَّذِيْنَ مِنْ قَبْلِهِمْ ۚ كَذَّبُوْا بِاٰيٰتِ رَبِّهِمْ فَاَهْلَكْنٰهُمْ بِذُنُوْبِهِمْ وَاَغْرَقْنَاۤ اٰلَ فِرْعَوْنَ ۚ وَكُلٌّ كَانُوْا ظٰلِمِيْنَ۔ “ان کا حال فرعون والوں اور ان سے پہلے لوگوں جیسا ہے؛ انہوں نے اپنے رب کی آیات کو جھٹلایا تو ہم نے انہیں ان کے گناہوں کے سبب ہلاک کر دیا اور فرعون والوں کو غرق کر دیا؛ اور وہ سب ظالم تھے۔” پہلی نظر میں یہ آیت ۵۲ کی تکرار لگتی ہے، مگر اہلِ تفسیر نے فرق بتایا ہے: وہاں اللہ کی آیات کے انکار کا ذکر تھا، یہاں رب کی آیات کو جھٹلانے کا؛ وہاں پکڑ کا اجمال تھا، یہاں ہلاکت اور غرق کی تفصیل۔ بعض مفسرین کہتے ہیں کہ پہلی آیت موت کے وقت کے انجام سے جڑی ہے اور دوسری دنیا کی ہلاکت سے۔ اہلِ معرفت لفظِ رب پر ٹھہرتے ہیں: رب وہ ہے جو پالتا اور درجہ بدرجہ پروان چڑھاتا ہے؛ تو جھٹلانے والوں نے صرف ایک دعوے کا انکار نہیں کیا، اس ہاتھ کو جھٹلایا جو انہیں پال رہا تھا۔ شیخِ اکبر کے نزدیک ہر شخص کا ایک ربِّ خاص ہے، یعنی وہ اسمِ الٰہی جس کے سائے میں اس کی پرورش ہوتی ہے؛ اپنے ربِّ خاص کی آیات کو جھٹلانا اپنی ہی جڑ کاٹنا ہے۔ اور آخر میں وَكُلٌّ كَانُوْا ظٰلِمِيْنَ نے وہی اصول دہرایا: ظلم انہوں نے خود کیا، اور قرآن کے محاورے میں سب سے بڑا ظلم اپنی جان پر ظلم ہے۔ نعمت کی تبدیلی پچھلی آیت کا اصول تھی، اور یہ آیت اس اصول کی تاریخی گواہی ہے۔

اب اسی اصول کا ایک اور مقام آتا ہے، اور اسے بھی اصول ہی کی روشنی میں پڑھنا ہے: اِنَّ شَرَّ الدَّوَآبِّ عِنْدَ اللّٰهِ الَّذِيْنَ كَفَرُوْا فَهُمْ لَا يُؤْمِنُوْنَ۔ “بےشک اللہ کے نزدیک تمام جانداروں میں بدترین وہ ہیں جنہوں نے کفر کیا، پس وہ ایمان نہیں لاتے۔” آیت نے یہ نہیں کہا کہ وہ ایمان لا ہی نہیں سکتے؛ کہا کہ وہ ایمان نہیں لاتے، یعنی ان کا انکار ایک ٹھہری ہوئی روش بن چکا ہے۔ “پس” کا حرف اس روش کو ان کے کفر پر مرتب کرتا ہے: انہوں نے انکار کیا، پھر اسی پر اڑ گئے۔ بدترین ہونا کسی کی پیدائشی تقدیر نہیں، اپنی اختیار کی ہوئی ضد کا نتیجہ ہے۔ اسی سورت میں پہلے ان لوگوں کا ذکر گزر چکا جو جان بوجھ کر بہرے اور گونگے بن گئے، اور وہاں یہ بھی فرمایا گیا تھا کہ اگر اللہ ان میں کوئی بھلائی جانتا تو انہیں سنا دیتا؛ یعنی طلب کی ایک چنگاری بھی ہوتی تو جواب ضرور آتا۔ لفظ دواب، یعنی زمین پر چلنے پھرنے والے، اس لیے آیا کہ جس نے عقل اور دل کو، جو انسان کا امتیاز تھے، کام میں نہ لایا، وہ صرف چلنے پھرنے والا ایک وجود رہ گیا۔ شیخِ اکبر کے نزدیک انسان اس لیے سب سے بلند ساخت پر ہے کہ وہ تمام اسماء کا جامع ہو سکتا ہے؛ جو اس جامعیت کو ضد سے ڈھانپ دے، وہ سب سے نیچے آ گرتا ہے، کیونکہ جتنا اونچا امکان تھا، اتنی ہی گہری گراوٹ ہے۔

پھر ان کی پہچان بتائی گئی: اَلَّذِيْنَ عٰهَدْتَّ مِنْهُمْ ثُمَّ يَنْقُضُوْنَ عَهْدَهُمْ فِيْ كُلِّ مَرَّةٍ وَّهُمْ لَا يَتَّقُوْنَ۔ “وہ لوگ جن سے آپ نے عہد لیا، پھر وہ ہر بار اپنا عہد توڑ دیتے ہیں اور وہ ڈرتے نہیں۔” یہ ایک تاریخی صورتِ حال کا بیان ہے: مفسرین کے مطابق یہ ان معاہد گروہوں کا ذکر ہے جنہوں نے حضورِ اکرم صلی اللہ علیہ وسلم سے معاہدہ کیا اور پھر بار بار اسے توڑا۔ آیت کا حکم ان مخصوص لوگوں اور اس مخصوص عمل سے متعلق ہے؛ یہ کسی نسل یا کسی قوم پر، نہ اس زمانے میں اور نہ آج، کوئی عمومی فیصلہ نہیں۔ برائی کا محل عہد شکنی کا عمل ہے، اور فِيْ كُلِّ مَرَّةٍ نے اسی تکرار کو ان کی پہچان بتایا۔ اہلِ معرفت اس آیت کا رخ اندر کی طرف کرتے ہیں: سب سے پہلا عہد وہ ہے جو روحوں سے لیا گیا، اور ہر توبہ اسی عہد کی تجدید ہے۔ نفس کی عادت یہ ہے کہ ہر بار عہد کرتا ہے اور ہر بار توڑ دیتا ہے؛ سالک کو اپنے اندر کے اس عہد شکن سے ہوشیار رہنا ہے۔ اس کا علاج تقویٰ ہے، کیونکہ آیت نے عہد شکنی کی جڑ وَّهُمْ لَا يَتَّقُوْنَ بتائی۔ شیخِ اکبر کے نزدیک تقویٰ کا ایک لطیف معنی یہ ہے کہ بندہ حق کو اپنی ڈھال بنا لے؛ اور جس کی ڈھال حق ہو، اس کا ہر عہد حق کی امانت بن جاتا ہے، جسے وہ توڑ نہیں سکتا۔

اس کے بعد عہد شکنوں کے بارے میں ایک تاریخی حکم آیا: فَاِمَّا تَثْقَفَنَّهُمْ فِي الْحَرْبِ فَشَرِّدْ بِهِمْ مَّنْ خَلْفَهُمْ لَعَلَّهُمْ يَذَّكَّرُوْنَ۔ “پس اگر آپ جنگ میں انہیں پائیں تو ان کے ذریعے ان کے پیچھے والوں کو بھی منتشر کر دیں، تاکہ وہ نصیحت حاصل کریں۔” یہ حکم اسی تاریخی پس منظر سے متعلق ہے جس میں ایک معاہدہ بار بار توڑا گیا تھا، اور اس کا تعلق حضورِ اکرم صلی اللہ علیہ وسلم کی قیادت میں ایک قائم ریاست کے معاملات سے ہے؛ اس میں کسی فرد کے لیے کوئی اختیار نہیں، اور اسے کسی دوسرے زمانے کے کسی گروہ پر چسپاں کرنا جائز نہیں۔ آیت نے خود مقصد بتا دیا: لَعَلَّهُمْ يَذَّكَّرُوْنَ؛ سختی کا مقصد انتقام نہیں، آئندہ عہد شکنی کا دروازہ بند کرنا اور سوچنے کا موقع دینا ہے۔ اہلِ تاویل، ظاہر کو اس کی جگہ قائم رکھتے ہوئے، اس میں نفس کے مجاہدے کی ایک تمثیل دیکھتے ہیں: جب سالک کسی بڑی اور سرکش خواہش پر قابو پا لیتا ہے تو اس کے پیچھے کھڑی چھوٹی خواہشیں خود بکھر جاتی ہیں۔ اور لفظ تذکر میں ایک لطیف بات ہے؛ شیخِ اکبر کے مکتب میں نصیحت باہر سے ڈالی ہوئی چیز نہیں، اس عہد کی یاد دہانی ہے جو روح پہلے سے جانتی ہے؛ اسی لیے قرآن نصیحت کو ذکر کہتا ہے۔

اور اب وہ حکم جو اسلام کے اخلاقِ معاہدہ کی بنیاد ہے: وَاِمَّا تَخَافَنَّ مِنْ قَوْمٍ خِيَانَةً فَانْبِذْ اِلَيْهِمْ عَلٰى سَوَآءٍ ۗ اِنَّ اللّٰهَ لَا يُحِبُّ الْخَآئِنِيْنَ۔ “اور اگر آپ کو کسی قوم سے خیانت کا اندیشہ ہو تو ان کا عہد برابری کی بنیاد پر ان کی طرف پھینک دیں؛ بےشک اللہ خیانت کرنے والوں کو پسند نہیں کرتا۔” علما نے اس کی تفسیر یہ کی کہ معاہدے کے خاتمے کا اعلان کھلے طور پر کیا جائے تاکہ دونوں فریق برابر علم پر ہوں؛ کبھی دھوکے سے، اچانک اور بے خبری میں نہیں۔ یعنی دوسرے کی خیانت کا اندیشہ بھی جواب میں خیانت کا جواز نہیں بنتا۔ مروی ہے کہ حضرت معاویہ رضی اللہ عنہ ایک معاہدے کی مدت کے آخر میں لشکر کو سرحد کے قریب لے گئے تو حضرت عمرو بن عبسہ رضی اللہ عنہ نے حضورِ اکرم صلی اللہ علیہ وسلم کا ارشاد بمعنی سنایا کہ جس کا کسی قوم سے معاہدہ ہو وہ اس کی گرہ نہ کھولے جب تک مدت پوری نہ ہو جائے یا وہ برابری کے ساتھ عہد واپس نہ کر دے؛ اور حضرت معاویہ رضی اللہ عنہ لشکر سمیت لوٹ آئے۔ اہلِ معرفت کہتے ہیں کہ اِنَّ اللّٰهَ لَا يُحِبُّ الْخَآئِنِيْنَ نے دشمن کے ساتھ معاملے کو بھی محبتِ الٰہی کے ترازو پر رکھ دیا۔ شیخِ اکبر کے نزدیک بندہ جس صفتِ حمیدہ سے آراستہ ہو اسی اسمِ الٰہی کا مظہر بنتا ہے؛ وفا اسمِ وفی کا عکس ہے، اور خیانت کسی اسمِ الٰہی کا عکس نہیں، اس لیے خائن محبت کی تجلی سے خود کو محروم کر لیتا ہے۔
