پچھلی قوموں کے انجام کے بعد اب رخ پھر مکہ کے منکرین کی طرف ہوتا ہے: وَمَا يَنْظُرُ هٰۤؤُلَآءِ اِلَّا صَيْحَةً وَّاحِدَةً مَّا لَهَا مِنْ فَوَاقٍ۔ “اور یہ لوگ بس ایک ہی چیخ کی راہ دیکھ رہے ہیں جس میں ذرا سا وقفہ بھی نہ ہو گا۔” فواق اس مختصر وقفے کو کہتے ہیں جو اونٹنی کو دو بار دوہنے کے درمیان آتا ہے؛ یعنی اتنی مہلت بھی نہ ملے گی۔ اہلِ معرفت ایک کے لفظ پر ٹھہرتے ہیں: حق کا امر ایک ہے، کثرت قبول کرنے والوں کی طرف سے ہے۔ جس کلمۂ کن سے سب کچھ وجود میں آیا، اسی امر کی ایک صدا پر سب کچھ لپیٹ دیا جائے گا؛ جو چیز کلمے سے بنی ہو وہ کلمے کو فوراً جواب دیتی ہے۔ پھر ان کا تمسخر: وَقَالُوْا رَبَّنَا عَجِّلْ لَّنَا قِطَّنَا قَبْلَ يَوْمِ الْحِسَابِ۔ “اور وہ کہتے ہیں: اے ہمارے رب! ہمارا حصہ ہمیں حساب کے دن سے پہلے ہی جلدی دے دے۔” قِطّ اس پرچے کو کہتے تھے جس پر کسی کا حصہ لکھا ہو؛ وہ مذاق میں اپنا نامۂ اعمال اور اپنے حصے کا عذاب مانگ رہے تھے۔ غور کیجیے کہ استہزا میں بھی زبان پر ہمارے رب کا لفظ آیا؛ ربوبیت کا اقرار انسان کی جڑ میں ایسا گڑا ہے کہ انکار بھی اسی لفظ میں ہوتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ہر شے کا حصہ اس کے عینِ ثابتہ کے مطابق ہے، اور حکم محکوم کے تابع ہے؛ یہی سرِّ قدر ہے۔ انہوں نے جلدی کا حصہ خود مانگا اور اپنی ضد پر قائم رہے، تو وہی ان کا حصہ بنا؛ کسی پر زبردستی نہیں ہوئی۔ اہلِ ذوق کہتے ہیں کہ سالک کا اصل حصہ تو خود حق ہے، مگر جو جلدی میں ہو وہ ہمیشہ کم تر حصہ مانگتا ہے۔

یہاں سے سورت کا دل شروع ہوتا ہے: اِصْبِرْ عَلٰى مَا يَقُوْلُوْنَ وَاذْكُرْ عَبْدَنَا دَاوٗدَ ذَا الْاَيْدِ ۚ اِنَّهٗۤ اَوَّابٌ۔ “جو کچھ یہ کہتے ہیں اس پر صبر کیجیے، اور ہمارے بندے داؤد علیہ السلام کو یاد کیجیے جو بڑی قوت والے تھے؛ بے شک وہ بہت رجوع کرنے والے تھے۔” حضورِ اکرم صلی اللہ علیہ وسلم کی تسلی دلیل بازی سے نہیں، انبیاء علیہم السلام کے ذکر سے کی گئی؛ زبانوں کے زخم کا مرہم ذکر ہے۔ چھٹی آیت میں سرداروں نے بتوں پر صبر کی تلقین کی تھی، یہاں حق کی خاطر صبر کا حکم ہے۔ نام سے پہلے عبدنا آیا: حضرت داؤد علیہ السلام بادشاہ بھی تھے اور نبی بھی، مگر ان کا پہلا تعارف بندگی ہے۔ شیخِ اکبر کے مکتب میں عبدیت سب سے بلند مقام ہے، اور آگے آنے والی خلافت اسی بنیاد پر کھڑی ہے۔ الاید کے معنی قوت کے ہیں، ہاتھوں کے نہیں؛ یہ فرق یاد رہے، کیونکہ سورت کے آخر میں دونوں ہاتھوں کا ذکر الگ معنی میں آئے گا۔ یہ قوت عبادت کی قوت تھی: حدیث کے مفہوم میں اللہ کو سب سے محبوب روزہ حضرت داؤد علیہ السلام کا روزہ ہے، ایک دن روزہ اور ایک دن افطار، اور سب سے محبوب نماز ان کی نماز، آدھی رات سونا، تہائی رات قیام، پھر چھٹا حصہ سونا۔ اوّاب مبالغے کا صیغہ ہے: بار بار لوٹنے والا۔ شیخِ اکبر کے نزدیک ہر آن خلقِ جدید ہے، بندہ ہر لمحہ نئے سرے سے وجود پاتا ہے؛ تو رجوع بھی ایک بار کا واقعہ نہیں، ہر آن کی تجدید ہے۔ اوّاب وہ ہے جس کا لوٹنا اس کی سانسوں کے ساتھ چلتا ہے۔ یہی لفظ آگے حضرت سلیمان علیہ السلام اور حضرت ایوب علیہ السلام کے لیے بھی آئے گا، اور یہ اس سورت کا دھاگا ہے۔

پھر اس قوت کا ایک عجیب پہلو: اِنَّا سَخَّرْنَا الْجِبَالَ مَعَهٗ يُسَبِّحْنَ بِالْعَشِيِّ وَالْاِشْرَاقِ۔ “ہم نے پہاڑوں کو ان کے ساتھ مسخر کر دیا تھا کہ شام کو اور اشراق کے وقت ان کے ساتھ تسبیح کرتے تھے۔” وَالطَّيْرَ مَحْشُوْرَةً ۚ كُلٌّ لَّهٗۤ اَوَّابٌ۔ “اور پرندوں کو بھی، جو جمع کر لیے جاتے تھے؛ سب اس کی طرف بار بار رجوع کرنے والے تھے۔” سورۂ سبا میں یہی بات یوں آئی: يٰجِبَالُ اَوِّبِيْ مَعَهٗ وَالطَّيْرَ، اے پہاڑو! ان کے ساتھ تسبیح دہراؤ، اور اے پرندو تم بھی۔ حضرت ابنِ عباس رضی اللہ عنہما سے منقول ہے، جس کا مفہوم یہ ہے کہ انہوں نے چاشت کی نماز کا اشارہ اسی آیت سے پایا۔ ظاہر میں یہ معجزہ ہے، اور اہلِ معرفت کے نزدیک یہ ایک حقیقت کا پردہ اٹھنا ہے: شیخِ اکبر کے مکتب میں کائنات کی ہر چیز زندہ ہے اور اپنی زبان سے تسبیح کرتی ہے، مگر عام کان اسے نہیں سنتے۔ حضرت داؤد علیہ السلام کی آواز نے اس پوشیدہ تسبیح کو ظاہر کر دیا؛ پہاڑ ان کی آواز کی بازگشت نہیں دے رہے تھے، اپنی تسبیح سنا رہے تھے جو ان کی ہم آہنگی سے سنائی دینے لگی۔ اور پھر وہی لفظ: سب اوّاب ہیں۔ مفسرین نے لہٗ کی ضمیر کو اللہ کی طرف بھی لوٹایا ہے اور حضرت داؤد علیہ السلام کی طرف بھی۔ اہلِ ذوق کہتے ہیں کہ دونوں معنی ایک جگہ ملتے ہیں: پوری کائنات اوّاب ہے، کیونکہ جس کا وجود عطا کردہ ہو وہ ہر آن اپنے دینے والے کی طرف لوٹتا ہے۔ جب ایک دل اوّاب بنتا ہے تو کائنات کے ساتھ ہم آواز ہو جاتا ہے۔

پھر عطاؤں کا بیان: وَشَدَدْنَا مُلْكَهٗ وَاٰتَيْنٰهُ الْحِكْمَةَ وَفَصْلَ الْخِطَابِ۔ “اور ہم نے ان کی سلطنت مضبوط کر دی، اور انہیں حکمت اور فیصلہ کن بات کہنے کی صلاحیت عطا کی۔” سلطنت کی مضبوطی مفسرین کے نزدیک ہیبت، لشکر اور عدل سے تھی۔ فصل الخطاب کے کئی معنی بیان ہوئے: ایسا فیصلہ جو جھگڑا چکا دے، ایسی گفتگو جو صاف اور جدا جدا ہو، اور بعض نے اس سے فیصلے کا یہ اصول مراد لیا کہ دلیل دعویٰ کرنے والے پر ہے اور قسم انکار کرنے والے پر۔ حکمت کی تعریف اہلِ علم نے یوں کی ہے کہ ہر چیز کو اس کی جگہ رکھنا۔ شیخِ اکبر کی کتاب فصوص الحکم میں حضرت داؤد علیہ السلام کا باب حکمتِ وجودیہ ہے، جس کا مفہوم یہ ہے کہ نبوت اور رسالت خالص اختصاصِ الٰہی ہیں، ان میں کسب کا دخل نہیں، اس لیے حضرت داؤد علیہ السلام کو جو کچھ ملا وہ عطا اور انعام تھا، کسی عمل کا بدلہ نہیں۔ ان آیات کے صیغے خود یہی کہتے ہیں: سخرنا، شددنا، آتیناہ؛ ہر فعل ہمارا ہے۔ اہلِ معرفت کے نزدیک فصل الخطاب کی گہری صورت یہ ہے کہ بندہ حق اور خلق کے درمیان فرق کو پہچانے، اور ہر ایک کو اس کا حق دے؛ جو یہ فرق پہچان لے، اس کی بات میں الجھاؤ نہیں رہتا۔ یہ عطا خلافت کی تمہید تھی، کیونکہ فیصلہ وہی کر سکتا ہے جسے جدا کرنے والی بصیرت ملی ہو۔

اب وہ واقعہ آتا ہے جس میں ادب کی سب سے زیادہ ضرورت ہے: وَهَلْ اَتٰىكَ نَبَؤُا الْخَصْمِ اِذْ تَسَوَّرُوا الْمِحْرَابَ۔ “اور کیا آپ تک جھگڑنے والوں کی خبر پہنچی، جب وہ دیوار پھاند کر عبادت خانے میں آ گئے؟” اِذْ دَخَلُوْا عَلٰى دَاوٗدَ فَفَزِعَ مِنْهُمْ قَالُوْا لَا تَخَفْ خَصْمٰنِ بَغٰى بَعْضُنَا عَلٰى بَعْضٍ فَاحْكُمْ بَيْنَنَا بِالْحَقِّ وَلَا تُشْطِطْ وَاهْدِنَاۤ اِلٰى سَوَآءِ الصِّرَاطِ۔ “جب وہ داؤد علیہ السلام کے پاس اندر آ گئے تو وہ ان سے گھبرا گئے؛ انہوں نے کہا: ڈریے نہیں، ہم دو فریق ہیں، ہم میں سے ایک نے دوسرے پر زیادتی کی ہے، تو ہمارے درمیان حق کے ساتھ فیصلہ کیجیے، بے انصافی نہ کیجیے، اور ہمیں سیدھی راہ بتا دیجیے۔” یہاں ہم صرف وہی کہیں گے جو آیات کہتی ہیں۔ وہ عام دروازے سے نہیں، دیوار پھاند کر آئے، اور ایسے وقت آئے جو عبادت کے لیے خاص تھا؛ گھبراہٹ ایک فطری انسانی کیفیت ہے، کوئی عیب نہیں۔ آیات یہ نہیں بتاتیں کہ وہ کون تھے؛ مفسرین میں سے بعض نے انہیں انسانی صورت میں آئے ہوئے فرشتے کہا اور بعض نے انسان، اور ہم وہیں ٹھہرتے ہیں جہاں قرآن ٹھہرا۔ اہلِ ذوق اس منظر کو سالک کے اپنے حال پر بھی پڑھتے ہیں، اور یہ تمثیل صرف سالک کے لیے ہے، کسی نبی کے حال کی تشریح نہیں: سالک کا دل اس کا محراب ہے، اور عبادت کے وقت بھی خیالات دیوار پھاند کر اندر آ جاتے ہیں؛ اس وقت اس کا کام گھبرانا نہیں، حق کے ساتھ فیصلہ کرنا ہے۔ اور فریقین کی درخواست میں عدالت کا پورا آداب نامہ ہے: حق کے ساتھ فیصلہ، حد سے نہ بڑھنا، اور سیدھی راہ کی رہنمائی۔

پھر مقدمہ پیش ہوا: اِنَّ هٰذَاۤ اَخِيْ لَهٗ تِسْعٌ وَّتِسْعُوْنَ نَعْجَةً وَّلِيَ نَعْجَةٌ وَّاحِدَةٌ فَقَالَ اَكْفِلْنِيْهَا وَعَزَّنِيْ فِي الْخِطَابِ۔ “یہ میرا بھائی ہے، اس کے پاس ننانوے دنبیاں ہیں اور میرے پاس ایک ہی دنبی ہے، اور یہ کہتا ہے کہ وہ بھی میرے حوالے کر دو، اور بات چیت میں اس نے مجھے دبا لیا۔” پھر فیصلہ آیا: قَالَ لَقَدْ ظَلَمَكَ بِسُؤَالِ نَعْجَتِكَ اِلٰى نِعَاجِهٖ وَاِنَّ كَثِيْرًا مِّنَ الْخُلَطَآءِ لَيَبْغِيْ بَعْضُهُمْ عَلٰى بَعْضٍ اِلَّا الَّذِيْنَ اٰمَنُوْا وَعَمِلُوا الصّٰلِحٰتِ وَقَلِيْلٌ مَّا هُمْ۔ “داؤد علیہ السلام نے فرمایا: اس نے تیری دنبی کو اپنی دنبیوں میں ملانے کا مطالبہ کر کے یقیناً تجھ پر زیادتی کی، اور اکثر شریک ایک دوسرے پر زیادتی کرتے ہیں، سوائے ان کے جو ایمان لائے اور نیک عمل کیے، اور ایسے لوگ تھوڑے ہی ہیں۔” بھائی کا لفظ بتاتا ہے کہ زیادتی اکثر قریب ترین رشتوں میں ہوتی ہے، اور ننانوے کا ایک پر ہاتھ ڈالنا بتاتا ہے کہ لالچ ضرورت سے نہیں، نفس کی بھوک سے ناپتا ہے۔ مفسرین کی ایک جماعت کہتی ہے کہ آزمائش یہیں تھی: فیصلہ دوسرے فریق کی بات سننے سے پہلے زبان پر آ گیا، اور عدالت کا کمال یہ ہے کہ دونوں کو سنا جائے۔ اہلِ ذوق اس تمثیل کو سالک کے باطن پر پڑھتے ہیں: جس کے پاس ننانوے نعمتیں ہوں اور اس کی نظر دوسرے کی ایک نعمت پر ہو، وہ اپنی کثرت کے باوجود فقیر ہے، اور جس کے پاس ایک ہو اور دل راضی ہو، وہ غنی ہے۔ اور قلیل ما ہم کا جملہ اس مکتب میں اس حقیقت کی طرف اشارہ ہے کہ شراکت میں عدل کرنے والے ہمیشہ تھوڑے ہوتے ہیں، کیونکہ اس کے لیے نفس کو اپنے حق سے کم پر راضی کرنا پڑتا ہے۔

اور پھر وہ لمحہ جس پر یہ رکوع جھکتا ہے: وَظَنَّ دَاوٗدُ اَنَّمَا فَتَنّٰهُ فَاسْتَغْفَرَ رَبَّهٗ وَخَرَّ رَاكِعًا وَّاَنَابَ۔ “اور داؤد علیہ السلام سمجھ گئے کہ ہم نے انہیں آزمایا ہے، تو انہوں نے اپنے رب سے مغفرت مانگی، اور جھک کر گر پڑے، اور رجوع کیا۔” ظن یہاں یقین کے معنی میں ہے۔ آزمائش کیا تھی؟ آیات نام نہیں لیتیں، اور مفسرین کے اقوال اسی دائرے میں ہیں: یا فیصلے میں جلدی کہ ایک فریق کو سن کر حکم دے دیا، یا یہ کہ وہ اچانک آنے والوں سے گھبرا گئے اور ان کے بارے میں اندیشہ دل میں آیا۔ رہی وہ اسرائیلی داستان جو ایک سپاہی کی بیوی کے حوالے سے بعض کتابوں میں در آئی، تو وہ بے بنیاد ہے اور انبیاء علیہم السلام کی عصمت کے صریح منافی، اس لیے نہ اسے نقل کیا جائے گا نہ اس پر کوئی تاویل بنائی جائے گی۔ انبیاء علیہم السلام کا استغفار ان کے بلند مقام کی دلیل ہے، کمی کی نہیں؛ اہلِ تصوف میں مشہور قول ہے کہ نیکوں کی نیکیاں مقربین کے نزدیک کوتاہیاں شمار ہوتی ہیں۔ خرّ راکعا میں رکوع سے مراد اکثر مفسرین کے نزدیک سجدہ ہے، اور یہ آیت سجدے کی آیت ہے؛ پڑھنے اور سننے والا یہاں سجدۂ تلاوت کرے۔ صحیح بخاری کی روایت کا مفہوم ہے کہ حضرت ابنِ عباس رضی اللہ عنہما نے حضورِ اکرم صلی اللہ علیہ وسلم کو اس مقام پر سجدہ کرتے دیکھا، اور ایک روایت کا مفہوم یہ ہے کہ حضرت داؤد علیہ السلام نے یہ سجدہ توبہ کے طور پر کیا اور ہم شکر کے طور پر کرتے ہیں۔ شیخِ اکبر کے مکتب میں انابت اوّاب کا عملی ظہور ہے: لغزش کا خیال آتے ہی پیشانی زمین پر، اور دل رب کی طرف۔

اور جواب میں عطا ہی عطا: فَغَفَرْنَا لَهٗ ذٰلِكَ وَاِنَّ لَهٗ عِنْدَنَا لَزُلْفٰى وَحُسْنَ مَاٰبٍ۔ “تو ہم نے ان کی وہ بات معاف کر دی، اور یقیناً ان کے لیے ہمارے ہاں قرب ہے اور اچھا ٹھکانا۔” یہاں لفظوں کا ایک خوبصورت رشتہ ہے: اوّاب اور مآب ایک ہی مادے سے ہیں، اَوب یعنی لوٹنا۔ جو دنیا میں بار بار لوٹتا رہا، اسے لوٹنے کی بہترین جگہ ملی؛ رجوع کا بدلہ اچھا مرجع ہے۔ اور زلفیٰ، یعنی قرب، سجدے کے فوراً بعد آیا؛ حدیث کا مفہوم ہے کہ بندہ اپنے رب سے سب سے زیادہ قریب سجدے کی حالت میں ہوتا ہے۔ جو جھک کر گرا تھا، وہ قرب میں اٹھایا گیا۔ عندنا کا لفظ اس قرب کی نوعیت بتاتا ہے: یہ مکان کا قرب نہیں، عنایت اور مرتبے کا قرب ہے۔ اہلِ معرفت کہتے ہیں کہ مقربین کی آزمائش انہیں پیچھے نہیں دھکیلتی، ان کی عبدیت کو تازہ کر کے آگے بڑھا دیتی ہے، کیونکہ عاجزی کا ہر نیا شعور قرب کا نیا زینہ ہے۔ اور ترتیب پر غور کیجیے: آزمائش، پھر رجوع، پھر مغفرت، پھر قرب، اور اس کے فوراً بعد خلافت کا اعلان۔ گویا خلافت کا تاج اسی سر پر رکھا گیا جو ابھی سجدے سے اٹھا تھا؛ خلیفہ وہی ہے جو اوّاب ہو۔

اب اس رکوع کی سب سے بڑی آیت: يٰدَاوٗدُ اِنَّا جَعَلْنٰكَ خَلِيْفَةً فِي الْاَرْضِ فَاحْكُمْ بَيْنَ النَّاسِ بِالْحَقِّ وَلَا تَتَّبِعِ الْهَوٰى فَيُضِلَّكَ عَنْ سَبِيْلِ اللّٰهِ۔ “اے داؤد! ہم نے تمہیں زمین میں خلیفہ بنایا ہے، تو لوگوں کے درمیان حق کے ساتھ فیصلہ کرو، اور خواہش کی پیروی نہ کرو کہ وہ تمہیں اللہ کی راہ سے بھٹکا دے گی۔” ظاہری معنی صاف ہے: حضرت داؤد علیہ السلام کو زمین میں حکمرانی سونپی گئی اور اس کا دستور بتا دیا گیا۔ قرآنِ مجید میں حضرت آدم علیہ السلام کے بارے میں فرشتوں سے فرمایا گیا تھا کہ میں زمین میں ایک خلیفہ بنانے والا ہوں، مگر نام لے کر براہِ راست خطاب میں خلیفہ بنائے جانے کا اعلان یہاں حضرت داؤد علیہ السلام سے ہوا۔ شیخِ اکبر کے مکتب میں خلیفہ وہ ہے جس میں اس کے صفات ظاہر ہوں جس کا وہ نائب ہے؛ نائب اپنی مرضی سے نہیں، اصل کی مرضی سے حکم دیتا ہے۔ اس لیے خلیفہ کی حکمرانی اسماءِ الٰہیہ کا ظہور ہے: اس کے عدل میں العدل کا سایہ ہو، اس کی شفقت میں الرحیم کا، اس کی گرفت میں الحکیم کی حکمت، اور اس کی بخشش میں الوہاب کی عطا۔ اور حکم بالحق کا مطلب صرف درست فیصلہ نہیں؛ الحق اللہ کا نام بھی ہے، تو حق کے ساتھ فیصلہ یہ ہے کہ فیصلہ کرنے والا درمیان سے ہٹ جائے اور اس کے ذریعے حق کا حکم جاری ہو۔ حدیثِ قربِ نوافل کا مفہوم یہی ہے کہ محبوب بندے کی سماعت اور بصارت حق کی عنایت سے سنورتی ہیں؛ یہ حلول یا اتحاد نہیں، بندہ بندہ ہی رہتا ہے، مگر اس کا ارادہ حق کے ارادے کے تابع ہو جاتا ہے۔

اور یہیں وہ اصول ہے جس پر ہدایت اور گمراہی کا سارا باب کھڑا ہے: وَلَا تَتَّبِعِ الْهَوٰى فَيُضِلَّكَ۔ بھٹکانے کا فعل یہاں خواہش کی طرف منسوب ہے۔ گمراہی کہیں باہر سے مسلط نہیں کی جاتی، وہ خواہش کی پیروی سے پیدا ہوتی ہے۔ یہ اس سورت کا اعلانِ اصول ہے، اور اسی کو ہم پوری تفسیر کا حاکم اصول مانتے ہیں: حق کسی کو زبردستی گمراہ نہیں کرتا، اور جو ہدایت طلب کرے اسے کبھی گمراہ نہیں کرتا؛ جو منہ پھیر کر خواہش کے پیچھے چل پڑے، اس کا یہی چلنا اس کی گمراہی ہے، اور حق اس رخ کو وجود دے دیتا ہے جو بندے نے خود چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ سورۂ فرقان میں اسی کی آخری صورت دکھائی گئی تھی: اَرَءَيْتَ مَنِ اتَّخَذَ اِلٰهَهٗ هَوٰىهُ، کیا آپ نے اسے دیکھا جس نے اپنی خواہش کو اپنا معبود بنا لیا؟ اور سورۂ قصص میں فیصلہ سنا دیا گیا تھا: وَمَنْ اَضَلُّ مِمَّنِ اتَّبَعَ هَوٰىهُ بِغَيْرِ هُدًى مِّنَ اللّٰهِ، اور اس سے بڑھ کر گمراہ کون ہو گا جو اللہ کی ہدایت کے بغیر اپنی خواہش کے پیچھے چلے؟ تینوں آیتیں ایک ہی بات کہتی ہیں: گمراہی کا دروازہ ہویٰ ہے، اور ہدایت کا دروازہ حق۔ یہ حکم ایک معصوم نبی کو دیا گیا، تاکہ ہر حاکم جان لے کہ جب نبی کو یہ تنبیہ ہے تو کوئی بھی اپنی خواہش کو حکم بنانے کا حق نہیں رکھتا؛ اور نبی کے حق میں یہ حکم دوام اور ثبات کے معنی میں ہے۔

ہویٰ کا لفظ خود اپنا حال بتاتا ہے: عربی میں هویٰ کا ایک معنی گرنا ہے، اور خواہش وہ جھکاؤ ہے جو آدمی کو نیچے گرا دیتا ہے۔ حق اوپر اٹھاتا ہے، ہویٰ نیچے گراتا ہے؛ ایک کا رخ عروج ہے اور دوسرے کا ہبوط۔ اہلِ ذوق کہتے ہیں کہ یہ خطاب صرف بادشاہوں کے لیے نہیں۔ حدیث کا مفہوم ہے کہ تم میں سے ہر ایک نگہبان ہے اور اس سے اس کی نگہبانی کے بارے میں پوچھا جائے گا؛ تو ہر انسان اپنی چھوٹی سی سلطنت کا خلیفہ ہے: گھر، کام، رشتے، اور سب سے بڑھ کر اپنا وجود۔ اس مکتب میں انسان کا باطن ایک مملکت ہے: دل تخت ہے، عقل وزیر ہے، اعضا رعایا ہیں، اور خواہشیں وہ فریق ہیں جو ہر وقت دیوار پھاند کر محراب میں آ جاتی ہیں۔ جو ان کے درمیان حق کے ساتھ فیصلہ کرے وہ اپنی مملکت کا سچا خلیفہ ہے، اور جو ہر آنے والی خواہش کے حق میں فیصلہ دیتا جائے وہ تخت پر بیٹھا ہوا قیدی ہے۔ اور اس پورے قصے کی ترتیب یہ سکھاتی ہے کہ خلافت کی شرط معصوم ہونے کا دعویٰ نہیں، بلکہ اوّاب ہونا ہے: لغزش کا ڈر ہو تو فوراً لوٹ آنا، فیصلے میں جلدی ہو تو سجدے میں گر جانا۔ جو خلیفہ رجوع کرنا جانتا ہو، اس کی حکمرانی میں ہویٰ کو جڑ پکڑنے کا موقع نہیں ملتا۔

آیت کا آخری حصہ اس گمراہی کی جڑ بتاتا ہے: اِنَّ الَّذِيْنَ يَضِلُّوْنَ عَنْ سَبِيْلِ اللّٰهِ لَهُمْ عَذَابٌ شَدِيْدٌۢ بِمَا نَسُوْا يَوْمَ الْحِسَابِ۔ “بے شک جو لوگ اللہ کی راہ سے بھٹکتے ہیں، ان کے لیے سخت عذاب ہے، اس لیے کہ انہوں نے حساب کے دن کو بھلا دیا۔” فعل پر غور کیجیے: یضلون، وہ خود بھٹکتے ہیں؛ اور سبب بتایا گیا: نسوا، انہوں نے بھلا دیا۔ پہلے بھولنا، پھر خواہش کی پیروی، پھر بھٹکنا، پھر عذاب؛ ہر قدم بندے کا اپنا ہے اور ہر نتیجہ اسی قدم کا پھل۔ اب اس سورت کے آغاز کی طرف دیکھیے: قرآن کو ذی الذکر کہا گیا، حضورِ اکرم صلی اللہ علیہ وسلم سے فرمایا گیا کہ ہمارے بندے داؤد علیہ السلام کو یاد کیجیے، اور یہاں گمراہی کی جڑ بھول بتائی گئی۔ سورت کا پورا نظام ذکر اور نسیان کے درمیان کھڑا ہے۔ اہلِ معرفت کہتے ہیں کہ ہویٰ نسیان کے اندھیرے میں پلتا ہے؛ جہاں حساب کی یاد زندہ ہو، وہاں خواہش کو تخت نہیں ملتا۔ اور سولہویں آیت میں منکرین نے حساب کے دن کا مذاق اڑا کر اپنا حصہ جلدی مانگا تھا، جبکہ یہاں حساب کے دن کو بھلانا ہی گمراہی کا سبب ٹھہرا؛ یوں رکوع جہاں سے چلا تھا وہیں لوٹ آیا۔ حضرت داؤد علیہ السلام کا استغفار اسی نسیان کا الٹ تھا: انہوں نے ایک لمحے کی آزمائش کو بھی یاد رکھا اور فوراً لوٹ آئے۔ اوّاب ہونا دراصل ہمیشہ یاد رکھنا ہے، اور یاد رکھنے والا ہی زمین میں حق کا سچا خلیفہ ہے۔
