پچھلے رکوع کی آخری آیت نے بتایا تھا کہ آخرت کے بارے میں ان کا علم تھک کر رہ گیا اور وہ اس سے اندھے ہیں؛ اب اسی اندھے پن کی زبان سنائی گئی: وَقَالَ الَّذِيْنَ كَفَرُوْۤا ءَاِذَا كُنَّا تُرٰبًا وَّاٰبَآؤُنَآ اَئِنَّا لَمُخْرَجُوْنَ۔ اور کافروں نے کہا: کیا جب ہم اور ہمارے باپ دادا مٹی ہو جائیں گے تو ہمیں سچ مچ نکالا جائے گا؟ لَقَدْ وُعِدْنَا هٰذَا نَحْنُ وَاٰبَآؤُنَا مِنْ قَبْلُ ۙ اِنْ هٰذَاۤ اِلَّاۤ اَسَاطِيْرُ الْاَوَّلِيْنَ۔ اس کا وعدہ ہم سے بھی کیا گیا اور اس سے پہلے ہمارے باپ دادا سے بھی؛ یہ تو بس اگلوں کی کہانیاں ہیں۔ ظاہر میں یہ بعث کا انکار ہے، اور یہی معنی اصل ہے؛ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اس دلیل کی کمزوری یہ بتائی جاتی ہے کہ منکر نے صورت کے بکھر جانے کو شے کا مٹ جانا سمجھ لیا۔ مٹی صورت ہوتی ہے، عینِ ثابتہ مٹی نہیں ہوتا؛ وہ حق کے علم میں اسی طرح قائم ہے جیسے وجود سے پہلے تھا۔ جس نے پہلی بار اس عین کو صورت دی وہ دوبارہ بھی دے گا، بلکہ اہلِ معرفت کے نزدیک وہ ہر آن نئے سرے سے دے رہا ہے، اس لیے دوسری پیدائش اس موجودہ لمحے سے زیادہ عجیب نہیں۔ پھر انہوں نے وعدے کو پرانا کہہ کر رد کیا: چونکہ بار بار سنا تھا، اسی لیے بےقیمت ٹھہرا۔ اہلِ ذوق کہتے ہیں کہ سب سے موٹا پردہ عادت کا ہے؛ جو بات کان میں دہرائی جاتی رہے اس کا اعجاز آنکھ سے اوجھل ہو جاتا ہے۔ عارف ہر آیت کو یوں سنتا ہے گویا ابھی اتری ہو، کیونکہ کلامِ حق اس کے دل پر ہر بار نئی تجلی کے ساتھ اترتا ہے۔ اور باپ دادا کا حوالہ دے کر انہوں نے اسلاف کو حق کی کسوٹی بنا لیا، جو تقلید کا پرانا روگ ہے۔

جواب میں مناظرہ نہیں، سفر کا حکم آیا: قُلْ سِيْرُوْا فِي الْاَرْضِ فَانْظُرُوْا كَيْفَ كَانَ عَاقِبَةُ الْمُجْرِمِيْنَ۔ کہہ دیجیے: زمین میں چلو پھرو، پھر دیکھو کہ مجرموں کا انجام کیسا ہوا۔ جو بات دلیل سے دل میں نہ اترے وہ کبھی آنکھ کے راستے اتر جاتی ہے؛ اجڑی بستیاں خاموش خطیب ہیں۔ اہلِ معرفت لفظ عاقبت پر ٹھہرتے ہیں: عاقبت وہ ہے جو عمل کی ایڑی کے ساتھ لگی چلی آئے۔ انجام باہر سے لا کر کسی پر رکھا نہیں جاتا، وہ عمل ہی کا پچھلا سرا ہے؛ اسی لیے مجرموں کا انجام فرمایا، یعنی جرم نے خود اپنا انجام اگایا۔ اہلِ تاویل کہتے ہیں کہ ایک سیر آفاق میں ہے اور ایک نفس میں؛ سالک اپنے ہی دل کی ان ویران بستیوں سے گزرتا ہے جہاں کبھی غفلت آباد تھی، اور وہاں بھی یہی قانون پڑھتا ہے۔ پھر خطاب حضورِ اکرم صلی اللہ علیہ وسلم کے قلبِ اطہر کی طرف پلٹا: وَلَا تَحْزَنْ عَلَيْهِمْ وَلَا تَكُنْ فِيْ ضَيْقٍ مِّمَّا يَمْكُرُوْنَ۔ اور آپ ان پر غم نہ کیجیے، اور نہ ان کی چالوں سے دل تنگ ہوں۔ آپ کا غم آپ کی رحمت سے تھا کہ منکروں کے انجام پر بھی دل دکھتا تھا، اور حق سبحانہ نے غم کو بھی تھاما اور تنگی کو بھی۔ تنگی شرحِ صدر کی ضد ہے۔ شیخِ اکبر کے نزدیک انسانِ کامل کا قلب سب سے وسیع ظرف ہے، اور اہلِ تصوف میں مشہور اثر ہے کہ زمین و آسمان حق کو نہ سما سکے اور مومن بندے کا دل سما گیا۔ جس دل میں حق کی گنجائش ہو، چالیں اسے تنگ نہیں کر سکتیں؛ ان کا مکر حق کے علم کے دائرے کے اندر ہے، اور اسی سورت میں قومِ ثمود کے قصے میں گزر چکا کہ مکر کرنے والوں کی تدبیر انہی پر پلٹتی ہے۔

اب سوال کا رخ وقت کی طرف ہوا: وَيَقُوْلُوْنَ مَتٰى هٰذَا الْوَعْدُ اِنْ كُنْتُمْ صٰدِقِيْنَ۔ اور وہ کہتے ہیں: یہ وعدہ کب پورا ہوگا، اگر تم سچے ہو؟ قُلْ عَسٰۤى اَنْ يَّكُوْنَ رَدِفَ لَكُمْ بَعْضُ الَّذِيْ تَسْتَعْجِلُوْنَ۔ کہہ دیجیے: کیا عجب کہ جس چیز کی تم جلدی مچا رہے ہو اس کا کچھ حصہ تمہارے پیچھے ہی آ لگا ہو۔ کب کا سوال وہی کرتا ہے جو وعدے کو دور سمجھتا ہے، اور جواب میں ایک لفظ نے فاصلہ مٹا دیا۔ عربی میں ردیف وہ سوار ہے جو اسی سواری پر پیچھے بیٹھا ہو؛ یعنی جس انجام کو تم کسی دور کے دن میں ڈھونڈ رہے ہو وہ تمہاری ہی سواری پر، تمہاری پشت سے لگا بیٹھا ہے۔ اہلِ معرفت کے نزدیک یہ عمل اور جزا کی ہم سواری ہے۔ شیخِ اکبر کے مکتب میں جزا عمل ہی کی وہ صورت ہے جو دوسری نشأت میں ظاہر ہوتی ہے، اس لیے وہ عمل سے جدا کر کے کہیں دور نہیں رکھی گئی۔ یہ لوگ اسے دیکھتے اس لیے نہیں کہ ان کا منہ دوسری طرف ہے؛ پشت پر بیٹھی چیز وہی نہیں دیکھتا جس نے رخ پھیر رکھا ہو۔ یہی اشارہ آگے چل کر اس رکوع کے اصول میں کھلے گا۔ مفسرین نے لکھا ہے کہ اللہ کی طرف سے امید کا کلمہ یقین کا معنی دیتا ہے، اور ان کے ہاں یہ بھی منقول ہے کہ اس کا کچھ حصہ بدر کے دن سامنے آ گیا۔ اور بعض کہہ کر بتا دیا کہ جو دکھا وہ پورا نہیں؛ رحمت نے سب کچھ ایک ساتھ نہیں اتارا، کہ لوٹنے کا دروازہ کھلا رہے۔

اسی مہلت کا راز اگلی آیت نے کھولا: وَاِنَّ رَبَّكَ لَذُوْ فَضْلٍ عَلَى النَّاسِ وَلٰكِنَّ اَكْثَرَهُمْ لَا يَشْكُرُوْنَ۔ اور بےشک آپ کا رب لوگوں پر بڑا فضل فرمانے والا ہے، لیکن ان میں سے اکثر شکر نہیں کرتے۔ تاخیر بھی فضل ہے؛ جو انجام پیچھے بیٹھا ہے وہ ابھی اترا نہیں، اور یہ وقفہ لوٹ آنے کی دعوت ہے۔ فضل سب لوگوں پر ہے، منکروں سمیت؛ یہی سبقتِ رحمت ہے کہ دینا پہلے ہے اور حساب بعد میں۔ اور شکر نہ کرنا اس کے سوا کیا ہے کہ نعمت کو نعمت نہ پہچانا جائے۔ وَاِنَّ رَبَّكَ لَيَعْلَمُ مَا تُكِنُّ صُدُوْرُهُمْ وَمَا يُعْلِنُوْنَ۔ اور آپ کا رب خوب جانتا ہے جو کچھ ان کے سینے چھپائے ہوئے ہیں اور جو وہ ظاہر کرتے ہیں۔ یہاں سینوں کا ذکر ہے، دلوں کا نہیں۔ اہلِ معرفت کے ہاں صدر دل کا بیرونی صحن ہے جہاں منصوبے، کینے اور وسوسے رکھے جاتے ہیں، اور قلب وہ اندرونی گھر ہے جو حق کی تجلی کا ظرف بن سکتا ہے؛ مکر صحن ہی میں پلتا ہے۔ وَمَا مِنْ غَآئِبَةٍ فِي السَّمَآءِ وَالْاَرْضِ اِلَّا فِيْ كِتٰبٍ مُّبِيْنٍ۔ اور آسمان و زمین میں کوئی چھپی ہوئی چیز نہیں مگر وہ ایک روشن کتاب میں درج ہے۔ شیخِ اکبر کے نزدیک اس کتابِ مبین میں اس علمِ الٰہی کی طرف اشارہ ہے جس میں ہر شے کا عینِ ثابتہ وجود سے پہلے ثبت ہے۔ غائب ہونا ہمارے اعتبار سے ہے؛ اس کے علم کے سامنے کچھ غائب نہیں۔ اور جو کچھ اس کتاب میں لکھا ہے وہ وہی ہے جو شے خود ہے؛ یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہوتا ہے، اور علم معلوم کے تابع ہے، معلوم پر جبر نہیں۔

پھر قرآن کا اپنا تعارف آیا: اِنَّ هٰذَا الْقُرْاٰنَ يَقُصُّ عَلٰى بَنِيْۤ اِسْرَآءِيْلَ اَكْثَرَ الَّذِيْ هُمْ فِيْهِ يَخْتَلِفُوْنَ۔ بےشک یہ قرآن بنی اسرائیل کے سامنے ان میں سے اکثر باتیں کھول کر بیان کرتا ہے جن میں وہ اختلاف رکھتے ہیں۔ وَاِنَّهٗ لَهُدًى وَّرَحْمَةٌ لِّلْمُؤْمِنِيْنَ۔ اور بےشک وہ ایمان والوں کے لیے ہدایت اور رحمت ہے۔ قصّ کی اصل نقشِ قدم کا پیچھا کرنا ہے۔ اہلِ ذوق کہتے ہیں کہ قرآن ہر اختلاف کے نقوشِ قدم پر پیچھے چلتا ہوا اس موڑ تک جاتا ہے جہاں راہ دو ہوئی تھی، اور وہاں اصل راستہ دکھا دیتا ہے؛ یوں وہ آراء میں سے ایک رائے نہیں، اصل کا گواہ ہے۔ یہ خطاب ملامت کا نہیں، دعوت کا ہے، اور اس میں پچھلی کتابوں اور ان کے انبیاء علیہم السلام کی تعظیم شامل ہے۔ اس مکتب میں اختلاف کا ایک بڑا سبب یہ بتایا جاتا ہے کہ ہر گروہ حق کی کسی ایک تجلی کو تھام کر اسی کو کل سمجھ لیتا ہے؛ سورۂ المؤمنون کا جملہ كُلُّ حِزْبٍۢ بِمَا لَدَيْهِمْ فَرِحُوْنَ اسی حال کی تصویر تھا۔ پھر فرمایا کہ یہی قرآن ایمان والوں کے لیے ہدایت و رحمت ہے۔ کتاب ایک ہے، اثر ظرف کے مطابق ہے؛ سورج سب پر یکساں چمکتا ہے، روشنی وہی پاتا ہے جس کی آنکھ کھلی ہو۔ اہلِ معرفت کے نزدیک یہ استعداد کا راز ہے: فیض میں بخل نہیں، قبول میں فرق ہے، اور قبول کا دروازہ ایمان کا رخ ہے۔

فیصلہ کس کے ہاتھ ہے، یہ بتا کر دل کو ٹھہرایا: اِنَّ رَبَّكَ يَقْضِيْ بَيْنَهُمْ بِحُكْمِهٖ ۚ وَهُوَ الْعَزِيْزُ الْعَلِيْمُ۔ بےشک آپ کا رب ان کے درمیان اپنے حکم سے فیصلہ فرمائے گا، اور وہی غالب، سب کچھ جاننے والا ہے۔ فَتَوَكَّلْ عَلَى اللّٰهِ ۭ اِنَّكَ عَلَى الْحَقِّ الْمُبِيْنِ۔ پس آپ اللہ پر بھروسا رکھیے؛ یقیناً آپ روشن حق پر ہیں۔ اپنے حکم سے فیصلہ کرنے کا مطلب، اس مکتب کے نزدیک، یہ ہے کہ حکم علم کے مطابق ہے اور علم ہر عین کی اپنی حقیقت کے مطابق؛ اس دن کا فیصلہ ہر ایک کو وہی دکھا دے گا جو وہ تھا۔ دو نام اس کے ساتھ آئے: العزیز، کہ اس فیصلے کو کوئی ٹال نہیں سکتا، اور العلیم، کہ یہ غلبہ بےعلم زور نہیں، کامل علم کا حکم ہے۔ پھر حرفِ فا نے نتیجہ نکالا: جب فیصلہ اسی کا ہے تو حضورِ اکرم صلی اللہ علیہ وسلم کا کام توکل ہے۔ اہلِ ذوق اس آیت میں دو بار آنے والے حرفِ علیٰ پر ٹھہرتے ہیں: توکل اللہ پر، اور قیام حق پر۔ جو اللہ پر ٹیک لگائے وہ گویا حق کی زمین پر کھڑا ہے، اور جس کے پاؤں تلے حق ہو اسے کسی کے مکر سے لغزش کا اندیشہ نہیں۔ شیخِ اکبر کے مکتب میں توکل کی حقیقت یہ بتائی جاتی ہے کہ بندہ دیکھ لے کہ فاعلِ حقیقی حق ہے، اور اپنی تدبیر کو اس کے سپرد کر کے خود کو محض اس کے کام کا محل جانے۔ اور الحق حق تعالیٰ کا نام بھی ہے؛ تو اس میں یہ لطیفہ بھی ہے کہ آپ اس نام کے کامل ترین مظہر ہیں۔

اب وہ آیت آئی جس پر یہ رکوع ٹھہرا ہے، اور جسے ہدایت و گمراہی کے باب میں اس سورت کا اصول سمجھنا چاہیے: اِنَّكَ لَا تُسْمِعُ الْمَوْتٰى وَلَا تُسْمِعُ الصُّمَّ الدُّعَآءَ اِذَا وَلَّوْا مُدْبِرِيْنَ۔ آپ مُردوں کو نہیں سنا سکتے، اور نہ بہروں کو پکار سنا سکتے ہیں جب وہ پیٹھ پھیر کر چل دیں۔ وَمَاۤ اَنْتَ بِهٰدِي الْعُمْيِ عَنْ ضَلٰلَتِهِمْ۔ اور نہ آپ اندھوں کو ان کی گمراہی سے نکال کر راہ پر لا سکتے ہیں۔ اہلِ معرفت کہتے ہیں کہ ساری بات اس شرط میں ہے: جب وہ پیٹھ پھیر لیں۔ جو بہرا سامنے کھڑا ہو وہ ہونٹوں اور اشاروں سے بات سمجھ لیتا ہے؛ بہرا پن مکمل اسی وقت ہوتا ہے جب وہ منہ موڑ کر چل دے۔ یوں آیت نے خود بتا دیا کہ یہاں بہرا پن کان کی بیماری نہیں، اعراض ہے۔ اور مُردے وہ دل ہیں جنہوں نے پکار پر جینا قبول نہ کیا۔ پھر گمراہی کی نسبت دیکھیے: ان کی گمراہی، ان کی اپنی۔ نہ حق کسی کو زبردستی گمراہ کرتا ہے، نہ کسی طالبِ ہدایت کو گمراہ کرتا ہے؛ جس نے منہ پھیرا اس کا منہ پھیرنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ سورۂ الحج میں یہی بات یوں آئی تھی: فَاِنَّهَا لَا تَعْمَى الْاَبْصَارُ وَلٰكِنْ تَعْمَى الْقُلُوْبُ الَّتِيْ فِي الصُّدُوْرِ۔ آنکھیں اندھی نہیں ہوتیں، سینوں کے دل اندھے ہوتے ہیں۔ اور اسی لیے رسول کا کام سنانا ہے، کان کھولنا نہیں؛ کان وہی کھلتا ہے جو خود رخ کرے۔

پھر اسی آیت نے دوسری جانب کا دروازہ کھولا: اِنْ تُسْمِعُ اِلَّا مَنْ يُّؤْمِنُ بِاٰيٰتِنَا فَهُمْ مُّسْلِمُوْنَ۔ آپ تو بس اسی کو سنا سکتے ہیں جو ہماری آیتوں پر ایمان لاتا ہے، سو وہی فرمانبردار ہیں۔ جو رخ کرے وہ سنتا ہے، اور کوئی طالب لوٹایا نہیں جاتا؛ طلب خود کان کھول دیتی ہے۔ یہی سرِّ قدر ہے کہ حکم محکوم کے اپنے حال کے مطابق ہوتا ہے۔ اور یہ اختیار بندے کی عزت ہے، مجبوری نہیں؛ سورۂ النحل میں فرمایا تھا: وَلَوْ شَآءَ لَهَدٰىكُمْ اَجْمَعِيْنَ۔ وہ چاہتا تو تم سب کو راہ پر لے آتا؛ یعنی وہ جبر پر قادر تھا اور اس نے جبر نہیں چاہا، تاکہ ایمان بندے کا اپنا قبول ہو۔ پھر اس رکوع کی آخری آیت: وَاِذَا وَقَعَ الْقَوْلُ عَلَيْهِمْ اَخْرَجْنَا لَهُمْ دَآبَّةً مِّنَ الْاَرْضِ تُكَلِّمُهُمْ ۙ اَنَّ النَّاسَ كَانُوْا بِاٰيٰتِنَا لَا يُوْقِنُوْنَ۔ اور جب ان پر بات پوری ہونے کا وقت آ جائے گا تو ہم ان کے لیے زمین سے ایک جانور نکالیں گے جو ان سے کلام کرے گا کہ لوگ ہماری آیتوں پر یقین نہیں رکھتے تھے۔ ظاہری معنی یہ ہے کہ یہ قیامت کے قریب کی بڑی نشانیوں میں سے ہے، جیسا کہ صحیح احادیث میں قیامت کی نشانیوں کے ضمن میں اس کا ذکر آیا ہے؛ اس کی صورت اور مقام کی کھوج میں ہم وہیں رکتے ہیں جہاں نص رکتی ہے۔ اہلِ ذوق صرف اتنا نکتہ لیتے ہیں کہ جو رسول کی بات نہ سنیں، ان سے زمین کی ایک مخلوق بات کرے گی، اور اس کی بات بھی انہی کا حال ہوگی: یقین نہ رکھنا۔ گویا فیصلہ ان کے اپنے حال کو زبان دے دیتا ہے۔
