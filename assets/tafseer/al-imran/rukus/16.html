احد کی ہزیمت کے بعد کچھ آوازیں اٹھیں جو کہتی تھیں کہ اس راہ میں کیا رکھا ہے، پرانے طریقے کی طرف لوٹ چلو۔ مفسرین کے نزدیک یہ آیات انہی آوازوں کا جواب ہیں، اور ان کا تعلق اسی تاریخی موقع سے ہے، آج کی کسی قوم پر عمومی حکم لگانے سے نہیں: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْۤا اِنْ تُطِيْعُوا الَّذِيْنَ كَفَرُوْا يَرُدُّوْكُمْ عَلٰۤى اَعْقَابِكُمْ فَتَنْقَلِبُوْا خٰسِرِيْنَ۔ “اے ایمان والو! اگر تم کافروں کا کہا مانو گے تو وہ تمہیں الٹے پاؤں پھیر دیں گے، پھر تم خسارے میں جا پڑو گے۔” اور فوراً اس کے بعد: بَلِ اللّٰهُ مَوْلٰىكُمْ ۚ وَهُوَ خَيْرُ النّٰصِرِيْنَ۔ “بلکہ اللہ تمہارا مولیٰ ہے، اور وہی سب سے بہتر مدد کرنے والا ہے۔” پچھلے رکوع میں “انْقَلَبْتُمْ عَلٰۤى اَعْقَابِكُمْ” کا خطرہ اندر سے تھا، یعنی حضورِ اکرم صلی اللہ علیہ وسلم کی وفات کی خبر سے دل کا پلٹ جانا؛ یہاں وہی خطرہ باہر سے ہے، یعنی دوسروں کی بات مان کر پلٹ جانا۔ اہلِ اشارہ کہتے ہیں کہ “بَلِ” ایک موڑ ہے: بہت سی آوازوں سے منہ پھیر کر ایک ذات کی طرف رخ کرنا۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں ولایت کی اصل یہ ہے کہ اسمِ ولی اسمائے الٰہیہ میں سے ہے؛ بندے کی ولایت اسی اسم کا عکس ہے۔ جب حق تعالیٰ خود کو “مَوْلٰىكُمْ” کہتا ہے تو بندے کو یاد دلاتا ہے کہ اس کی کارسازی کسی اور کے سپرد نہیں۔ اور اہلِ معرفت کے نزدیک باطن میں بھی ایسی آوازیں ہیں، نفس کے وسوسے اور غفلت کے مشورے، جو آدمی کو ایڑیوں کے بل لوٹانا چاہتے ہیں؛ ان کا علاج بھی یہی “بَلِ” ہے۔ یہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر مولیٰ کی مدد کا ایک انداز بیان ہوتا ہے: سَنُلْقِيْ فِيْ قُلُوْبِ الَّذِيْنَ كَفَرُوا الرُّعْبَ بِمَاۤ اَشْرَكُوْا بِاللّٰهِ مَا لَمْ يُنَزِّلْ بِهٖ سُلْطٰنًا ۚ وَمَاْوٰىهُمُ النَّارُ ۗ وَبِئْسَ مَثْوَى الظّٰلِمِيْنَ۔ “ہم عنقریب کافروں کے دلوں میں رعب ڈال دیں گے، اس لیے کہ انہوں نے اللہ کے ساتھ ان چیزوں کو شریک ٹھہرایا جن کی کوئی سند اللہ نے نہیں اتاری؛ اور ان کا ٹھکانا آگ ہے، اور ظالموں کا وہ ٹھکانا بہت برا ہے۔” بعض مفسرین کہتے ہیں کہ احد کے بعد مکہ کا لشکر غالب ہونے کے باوجود لوٹ گیا اور مدینہ پر نہ چڑھا، اور یہ اسی رعب کا اثر تھا۔ صحیح بخاری کی حدیث کا مفہوم ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا: ایک مہینے کی مسافت تک رعب کے ذریعے میری مدد کی گئی ہے۔ آیت میں رعب کا سبب صاف بتا دیا گیا: “بِمَاۤ اَشْرَكُوْا”، یعنی ان کا اپنا عمل۔ رعب ان پر کسی بلاوجہ زبردستی کے طور پر نہیں ڈالا گیا؛ جس دل نے بے سند سہاروں کو اپنا سہارا بنایا، اس کا خوف اسی انتخاب کا پھل ہے۔ شیخِ اکبر کے نزدیک ہر وہ چیز جو اللہ کے سوا معبود بنا لی جائے، ایک نام ہے جس کے پیچھے الوہیت کی کوئی حقیقت نہیں؛ “مَا لَمْ يُنَزِّلْ بِهٖ سُلْطٰنًا” اسی بے حقیقتی کا بیان ہے۔ جو دل بے سند چیز پر ٹکا ہو وہ لرزتا رہتا ہے، کیونکہ اس کی بنیاد ہی ہوا میں ہے؛ اور جو دل توحید پر ٹکا ہو اسے امن ملتا ہے۔ اسی رکوع میں آگے مومنوں پر “اَمَنَةً” کا نزول اسی فرق کا دوسرا رخ ہے۔

اب احد کے واقعے کی طرف رخ ہوتا ہے: وَلَقَدْ صَدَقَكُمُ اللّٰهُ وَعْدَهٗۤ اِذْ تَحُسُّوْنَهُمْ بِاِذْنِهٖ ۚ حَتّٰۤى اِذَا فَشِلْتُمْ وَتَنَازَعْتُمْ فِي الْاَمْرِ وَعَصَيْتُمْ مِّنْۢ بَعْدِ مَاۤ اَرٰىكُمْ مَّا تُحِبُّوْنَ۔ “اور اللہ نے تم سے اپنا وعدہ سچ کر دکھایا جب تم اس کے اذن سے انہیں کاٹ رہے تھے؛ یہاں تک کہ جب تم نے ہمت چھوڑی اور حکم کے بارے میں آپس میں اختلاف کیا، اور اس کے بعد نافرمانی ہوئی کہ اللہ نے تمہیں وہ دکھا دیا تھا جو تم چاہتے تھے۔” جنگ کے آغاز میں فتح مسلمانوں کی تھی۔ صحیح بخاری کی روایت کا مفہوم یہ ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم نے پچاس تیر اندازوں کو حضرت عبد اللہ بن جبیر رضی اللہ عنہ کی سرکردگی میں ایک ٹیلے پر مقرر فرمایا اور تاکید کی کہ اگر تم دیکھو کہ پرندے ہمیں نوچ رہے ہیں تب بھی اپنی جگہ نہ چھوڑنا جب تک میں پیغام نہ بھیجوں۔ جب دشمن بھاگا تو تیر اندازوں میں سے اکثر نے یہ سمجھا کہ جنگ ختم ہو گئی ہے اور مالِ غنیمت کی طرف بڑھے، جبکہ حضرت عبد اللہ بن جبیر رضی اللہ عنہ نے انہیں حضورِ اکرم صلی اللہ علیہ وسلم کا حکم یاد دلایا اور چند ساتھیوں کے ساتھ اپنی جگہ ڈٹے رہے یہاں تک کہ شہید ہوئے۔ یہ وہ صحابۂ کرام رضی اللہ عنہم تھے جو اپنی جانیں ہتھیلی پر رکھ کر نکلے تھے؛ ان سے سمجھنے میں چوک ہوئی، اور قرآن نے اس چوک کو بیان کیا تو اسی آیت کے اندر معافی کا اعلان بھی کر دیا۔ کسی امتی کو یہ حق نہیں کہ جس معاملے کو اللہ نے بیان کر کے معاف کر دیا، اس پر زبان دراز کرے۔

آیت آگے بڑھتی ہے: مِنْكُمْ مَّنْ يُّرِيْدُ الدُّنْيَا وَمِنْكُمْ مَّنْ يُّرِيْدُ الْاٰخِرَةَ ۚ ثُمَّ صَرَفَكُمْ عَنْهُمْ لِيَبْتَلِيَكُمْ ۚ وَلَقَدْ عَفَا عَنْكُمْ ۗ وَاللّٰهُ ذُوْ فَضْلٍ عَلَى الْمُؤْمِنِيْنَ۔ “تم میں سے کچھ دنیا چاہتے تھے اور کچھ آخرت چاہتے تھے؛ پھر اس نے تمہیں ان سے پھیر دیا تاکہ تمہیں آزمائے؛ اور یقیناً اس نے تمہیں معاف کر دیا، اور اللہ ایمان والوں پر بڑا فضل والا ہے۔” یہاں آیت ۱۴۵ کا اصول پھر سامنے آتا ہے: ارادے دو تھے، اور ہر ارادے کا اپنا نتیجہ؛ مگر فیصلہ سنانے سے پہلے ہی فرما دیا گیا: “وَلَقَدْ عَفَا عَنْكُمْ”۔ اہلِ معرفت اسے سبقتِ رحمت کہتے ہیں: عتاب کی بات ابھی پوری نہیں ہوئی کہ معافی آ پہنچی۔ “مَّا تُحِبُّوْنَ” پر اہلِ ذوق ٹھہرتے ہیں: محبوب چیز بھی، جب نظر کے سامنے آ جائے، حکم سے نظر ہٹا سکتی ہے؛ اسی لیے سالک کو پسندیدہ چیز کے سامنے سب سے زیادہ ہوشیار رہنا ہے۔ اور “تَنَازَعْتُمْ فِي الْاَمْرِ” کی اشاری جہت یہ ہے کہ جب ایک حکم کے مقابلے میں بہت سی رائیں کھڑی ہو جائیں تو وحدتِ امر ٹوٹتی ہے، اور جہاں وحدت ٹوٹے وہاں قوت بکھر جاتی ہے۔ شیخِ اکبر کے نزدیک ابتلا سزا نہیں، ظہور ہے: آزمائش وہ ظاہر کرتی ہے جو پہلے سے اندر موجود ہے، تاکہ بندہ خود کو پہچانے اور حق کی طرف لوٹے۔

پھر ہزیمت کا منظر: اِذْ تُصْعِدُوْنَ وَلَا تَلْوٗنَ عَلٰۤى اَحَدٍ وَّالرَّسُوْلُ يَدْعُوْكُمْ فِيْۤ اُخْرٰىكُمْ فَاَثَابَكُمْ غَمًّۢا بِغَمٍّ لِّكَيْلَا تَحْزَنُوْا عَلٰى مَا فَاتَكُمْ وَلَا مَاۤ اَصَابَكُمْ ۗ وَاللّٰهُ خَبِيْرٌۢ بِمَا تَعْمَلُوْنَ۔ “جب تم چڑھے جا رہے تھے اور کسی کو مڑ کر نہ دیکھتے تھے، اور رسول صلی اللہ علیہ وسلم تمہارے پیچھے سے تمہیں پکار رہے تھے؛ تو اللہ نے تمہیں غم پر غم دیا، تاکہ جو چیز تمہارے ہاتھ سے نکل گئی اور جو مصیبت تم پر آئی اس پر تم رنجیدہ نہ ہو؛ اور اللہ تمہارے کاموں سے پوری طرح باخبر ہے۔” اس منظر کا مرکز “وَّالرَّسُوْلُ يَدْعُوْكُمْ” ہے: جب صفیں بکھریں تو حضورِ اکرم صلی اللہ علیہ وسلم اپنی جگہ قائم رہے، چند جاں نثاروں کے ساتھ، زخمی چہرۂ مبارک کے ساتھ، اور لوگوں کو اپنی طرف بلاتے رہے۔ “غَمًّۢا بِغَمٍّ” کی ایک تفسیر یہ ہے کہ ایک غم کے بعد دوسرا غم آیا: پہلے فتح اور غنیمت کے ہاتھ سے نکلنے کا غم، پھر حضورِ اکرم صلی اللہ علیہ وسلم کی شہادت کی افواہ کا غم؛ اور جب دوسرا غم آیا تو پہلا چھوٹا ہو گیا، اور جب معلوم ہوا کہ حضورِ اکرم صلی اللہ علیہ وسلم سلامت ہیں تو ہر غم ہلکا ہو گیا۔ یوں ایک غم دوسرے غم کا علاج بن گیا، اور “لِّكَيْلَا تَحْزَنُوْا” کی حکمت ظاہر ہوئی۔ سورۂ الحدید میں یہی تربیت ایک اصول کی صورت میں آئی ہے: لِّكَيْلَا تَاْسَوْا عَلٰى مَا فَاتَكُمْ وَلَا تَفْرَحُوْا بِمَاۤ اٰتٰىكُمْ۔ اہلِ معرفت کہتے ہیں کہ حق تعالیٰ کبھی جلال کی تجلی سے وہ کام لیتا ہے جو جمال سے نہیں ہوتا؛ یہ غم نتیجوں سے دل کی وابستگی کو توڑنے کے لیے تھا، تاکہ بندہ نہ کھوئی ہوئی چیز پر ٹوٹے، نہ آنے والی مصیبت پر بکھرے۔

غم کے بعد عجیب عطا اتری: ثُمَّ اَنْزَلَ عَلَيْكُمْ مِّنْۢ بَعْدِ الْغَمِّ اَمَنَةً نُّعَاسًا يَّغْشٰى طَآئِفَةً مِّنْكُمْ ۙ وَطَآئِفَةٌ قَدْ اَهَمَّتْهُمْ اَنْفُسُهُمْ يَظُنُّوْنَ بِاللّٰهِ غَيْرَ الْحَقِّ ظَنَّ الْجَاهِلِيَّةِ۔ “پھر اس غم کے بعد اس نے تم پر امن اتارا، ایک اونگھ جو تم میں سے ایک گروہ پر چھا رہی تھی؛ اور ایک گروہ وہ تھا جسے اپنی جانوں کی فکر پڑی تھی، وہ اللہ کے بارے میں ناحق جاہلیت والے گمان کر رہے تھے۔” صحیح بخاری میں حضرت ابو طلحہ رضی اللہ عنہ کا بیان ہے، جس کا مفہوم یہ ہے کہ احد کے دن میں ان لوگوں میں تھا جن پر اونگھ چھا گئی، یہاں تک کہ میری تلوار بار بار میرے ہاتھ سے گرتی اور میں اسے اٹھا لیتا۔ میدانِ جنگ میں، دشمن کے سامنے، نیند کا آ جانا اس بات کی علامت تھی کہ دلوں سے خوف اٹھا لیا گیا ہے۔ اہلِ معرفت کے نزدیک نیند ایک برزخ اور چھوٹی موت ہے؛ اس اونگھ میں بندہ اپنی تدبیر کو چھوڑ دیتا ہے اور حق اسے تھام لیتا ہے۔ گویا خوف کے بیچ انہیں فنا کا ایک ذائقہ چکھایا گیا: جب “میں” کی فکر ڈھیلی پڑی تو امن اتر آیا۔ دوسرے گروہ کے بارے میں اکثر مفسرین کہتے ہیں کہ یہ منافقین تھے، صحابۂ کرام رضی اللہ عنہم کا مومن گروہ نہیں۔ ان کی پہچان “قَدْ اَهَمَّتْهُمْ اَنْفُسُهُمْ” بتائی گئی: اپنی جان کی فکر ہی ان کا حجاب بن گئی۔ اہلِ ذوق کہتے ہیں کہ جو اپنے آپ میں الجھا رہے اسے امن کی اونگھ نصیب نہیں ہوتی؛ اسے یہ ہلکی نیند اس لیے نہ ملی کہ وہ اپنی ہی ذات کے پہرے سے فارغ نہ ہوا تھا۔

اسی گروہ کی باتیں آگے نقل ہوتی ہیں: يَقُوْلُوْنَ هَلْ لَّنَا مِنَ الْاَمْرِ مِنْ شَيْءٍ ۗ قُلْ اِنَّ الْاَمْرَ كُلَّهٗ لِلّٰهِ ۗ يُخْفُوْنَ فِيْۤ اَنْفُسِهِمْ مَّا لَا يُبْدُوْنَ لَكَ ۗ يَقُوْلُوْنَ لَوْ كَانَ لَنَا مِنَ الْاَمْرِ شَيْءٌ مَّا قُتِلْنَا هٰهُنَا۔ “وہ کہتے ہیں: کیا اس معاملے میں ہمارا بھی کچھ اختیار ہے؟ کہہ دیجیے: سارا معاملہ اللہ ہی کے اختیار میں ہے۔ وہ اپنے دلوں میں وہ بات چھپاتے ہیں جو آپ صلی اللہ علیہ وسلم پر ظاہر نہیں کرتے؛ کہتے ہیں: اگر اس معاملے میں ہمارا کچھ اختیار ہوتا تو ہم یہاں نہ مارے جاتے۔” ان کے سوال کا جواب ایک جملے میں ہے: “اِنَّ الْاَمْرَ كُلَّهٗ لِلّٰهِ”۔ شیخِ اکبر کے مکتب میں یہ توحیدِ افعالی کی بنیاد ہے: ہر فعل کا حقیقی فاعل حق ہے اور مخلوق اس کے افعال کے مظاہر ہیں۔ مگر یہ توحید بندے کے اختیار کی نفی نہیں کرتی؛ اسی رکوع میں “مِنْكُمْ مَّنْ يُّرِيْدُ” بندے کے ارادے کا اقرار ہے۔ حکم حق کا ہے، اور حق بندے کے رخ کے مطابق حکم کرتا ہے؛ یہی سرِّ قدر ہے۔ اور “لَوْ” کا یہ لفظ حسرت کی زبان ہے جو ہو چکے کو بدلنا چاہتی ہے؛ اہلِ معرفت کہتے ہیں کہ جو ہو چکا اس پر “لَوْ” کہنا اپنی تدبیر کو حق کی تقدیر کے مقابلے میں کھڑا کرنا ہے۔

اور جواب مکمل ہوتا ہے: قُلْ لَّوْ كُنْتُمْ فِيْ بُيُوْتِكُمْ لَبَرَزَ الَّذِيْنَ كُتِبَ عَلَيْهِمُ الْقَتْلُ اِلٰى مَضَاجِعِهِمْ ۚ وَلِيَبْتَلِيَ اللّٰهُ مَا فِيْ صُدُوْرِكُمْ وَلِيُمَحِّصَ مَا فِيْ قُلُوْبِكُمْ ۗ وَاللّٰهُ عَلِيْمٌۢ بِذَاتِ الصُّدُوْرِ۔ “کہہ دیجیے: اگر تم اپنے گھروں میں بھی ہوتے تو جن کے لیے قتل ہونا لکھا تھا وہ خود نکل کر اپنی قتل گاہوں تک پہنچ جاتے؛ اور یہ اس لیے ہوا کہ اللہ تمہارے سینوں کی باتوں کو آزمائے اور تمہارے دلوں کے کھوٹ کو نکال کر صاف کر دے؛ اور اللہ سینوں کے بھید خوب جانتا ہے۔” آیت ۱۴۵ کا “كِتٰبًا مُّؤَجَّلًا” یہاں ایک منظر بن گیا: موت اپنی جگہ پر ہے، آدمی خود چل کر وہاں پہنچتا ہے۔ پھر ایک باریک فرق: سینے کے لیے “لِيَبْتَلِيَ” آیا اور دل کے لیے “لِيُمَحِّصَ”۔ اہلِ تصوف صدر اور قلب میں فرق کرتے ہیں: صدر دل کا صحن ہے جہاں خیالات اور وسوسے آتے جاتے ہیں، جیسا کہ سورۂ الناس میں الَّذِيْ يُوَسْوِسُ فِيْ صُدُوْرِ النَّاسِ فرمایا گیا؛ اور قلب اندر کا گھر ہے جہاں ایمان بستا ہے۔ صحن میں جو کچھ ہے اسے آزمائش ظاہر کرتی ہے، اور گھر کے اندر جو کچھ ہے اسے تمحیص صاف کرتی ہے، جیسے آگ سونے کو کھوٹ سے الگ کرتی ہے۔ یوں احد کی آزمائش ایک ہی واقعے میں دو کام کر گئی: باہر کا پردہ اٹھایا اور اندر کا جوہر نکھارا۔

رکوع کا خاتمہ اس اعلان پر ہے جو صحابۂ کرام رضی اللہ عنہم کی عزت کی دائمی سند ہے: اِنَّ الَّذِيْنَ تَوَلَّوْا مِنْكُمْ يَوْمَ الْتَقَى الْجَمْعٰنِ ۙ اِنَّمَا اسْتَزَلَّهُمُ الشَّيْطٰنُ بِبَعْضِ مَا كَسَبُوْا ۚ وَلَقَدْ عَفَا اللّٰهُ عَنْهُمْ ۗ اِنَّ اللّٰهَ غَفُوْرٌ حَلِيْمٌ۔ “تم میں سے جو لوگ اس دن پیٹھ پھیر گئے جب دونوں لشکر آمنے سامنے ہوئے، انہیں شیطان ہی نے ان کے بعض اعمال کی وجہ سے پھسلا دیا تھا؛ اور یقیناً اللہ نے انہیں معاف کر دیا؛ بے شک اللہ بخشنے والا، بردبار ہے۔” یہ آیت اس رکوع کا اصولی مقام ہے: “بِبَعْضِ مَا كَسَبُوْا” بتاتا ہے کہ شیطان کو راستہ بندے کی اپنی کمائی سے ملتا ہے، اس کا اپنا کوئی زور نہیں؛ جیسا کہ سورۂ ابراہیم میں شیطان خود اقرار کرتا ہے: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّاۤ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ۔ پھر “بِبَعْضِ” کا لفظ خود رحمت ہے: سب اعمال نہیں، بعض؛ اور “اسْتَزَلَّهُمُ” کا معنی پھسلانا ہے، گرانا نہیں۔ پھسلا ہوا قدم پھر جم جاتا ہے، اور ربانیوں کی دعا “وَثَبِّتْ اَقْدَامَنَا” یہیں قبول ہوتی دکھائی دیتی ہے۔

اور سب سے بڑی بات “وَلَقَدْ عَفَا اللّٰهُ عَنْهُمْ” ہے۔ جن کی معافی کا اعلان قرآن کرے، ان کے بارے میں کسی امتی کا زبان کھولنا بے ادبی ہے۔ صحیح بخاری میں ہے، جس کا مفہوم یہ ہے کہ ایک شخص نے حضرت عبد اللہ بن عمر رضی اللہ عنہما سے حضرت عثمان غنی رضی اللہ عنہ کے بارے میں کہا کہ وہ احد کے دن پیچھے ہٹ گئے تھے؛ حضرت عبد اللہ بن عمر رضی اللہ عنہما نے فرمایا: میں گواہی دیتا ہوں کہ اللہ نے انہیں معاف کر دیا اور بخش دیا۔ یہی صحابۂ کرام رضی اللہ عنہم کے بارے میں اہلِ سنت کا ادب ہے: ان کی لغزش کا ذکر قرآن کی پیروی میں ہو تو اس کے ساتھ اللہ کی معافی کا ذکر لازم ہے۔ آیت دو اسموں پر ختم ہوتی ہے: “غَفُوْرٌ” اور “حَلِيْمٌ”۔ شیخِ اکبر کے مکتب میں مغفرت کی اصل ستر ہے، یعنی ڈھانپ لینا؛ غفور وہ ہے جو لغزش کو ڈھانپ کر اس کے اثر سے بچا لے۔ اور حلیم وہ ہے جو قدرت کے باوجود گرفت میں جلدی نہ کرے، تاکہ بندے کو لوٹنے کی مہلت ملے۔ یوں یہ رکوع ہزیمت سے شروع ہو کر معافی پر ختم ہوتا ہے، اور اہلِ معرفت اسے سبقتِ رحمت کا نمونہ کہتے ہیں: آزمائش آئی تو صاف کرنے کے لیے، لغزش ہوئی تو سکھانے کے لیے، اور آخری لفظ معافی کا رہا۔
