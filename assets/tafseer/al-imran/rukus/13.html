یہاں سے سورت کا رخ میدانِ احد کی طرف مڑتا ہے، اور آگے کے کئی رکوع اسی غزوے کے سبق ہیں۔ تاریخی پس منظر یہ ہے کہ شوال تین ہجری میں قریش بدر کا بدلہ لینے کے لیے تقریباً تین ہزار کا لشکر لے کر مدینہ منورہ کی طرف بڑھے۔ حضورِ اکرم صلی اللہ علیہ وسلم نے صحابۂ کرام رضی اللہ عنہم سے مشورہ فرمایا، اور جب شہر سے باہر نکل کر مقابلہ کرنے کی رائے غالب آئی تو آپ صلی اللہ علیہ وسلم زرہ پہن کر نکلے۔ راستے میں منافقین کا سردار عبداللہ بن اُبی اپنے تقریباً تین سو ساتھیوں کو لے کر لوٹ گیا، اور سات سو کے قریب جاں نثار باقی رہے۔ اسی صبح کا نقشہ آیت کھینچتی ہے: وَاِذْ غَدَوْتَ مِنْ اَهْلِكَ تُبَوِّئُ الْمُؤْمِنِيْنَ مَقَاعِدَ لِلْقِتَالِ ؕ وَاللّٰهُ سَمِيْعٌ عَلِيْمٌ۔ “اور یاد کیجیے جب آپ صبح سویرے اپنے گھر والوں کے پاس سے نکلے، ایمان والوں کو جنگ کے لیے ان کے مورچوں پر بٹھا رہے تھے، اور اللہ سب کچھ سننے والا، سب کچھ جاننے والا ہے۔” آپ صلی اللہ علیہ وسلم نے صفیں خود ترتیب دیں، اور پچاس تیر اندازوں کو حضرت عبداللہ بن جبیر رضی اللہ عنہ کی قیادت میں پہاڑی پر کھڑا کر کے تاکید فرمائی کہ کسی حال میں اپنی جگہ نہ چھوڑیں۔ لفظ تُبَوِّئُ کی اصل وہ جگہ ہے جہاں آدمی ٹھہرے اور لوٹ کر آئے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں یہ لفظ ایک بڑی حقیقت کی طرف اشارہ کرتا ہے: انسانِ کامل ہر ایک کو اس کا مقام اس کی استعداد کے مطابق دیتا ہے، جیسے حق تعالیٰ ہر موجود کو اس کا مقامِ معلوم عطا کرتا ہے۔ کوئی صف کے آگے کھڑا کیا گیا، کوئی پہاڑی پر، اور ہر ایک کی نجات اسی میں تھی کہ اپنے مقام پر جما رہے۔ آیت کا اختتام سننے اور جاننے پر ہوا، کیونکہ اس نے مشورے کی ہر بات سنی اور ہر دل کا حال جانا۔ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے۔

پھر ایک نازک لمحے کا ذکر ہے: اِذْ هَمَّتْ طَّآئِفَتٰنِ مِنْكُمْ اَنْ تَفْشَلَا ۙ وَاللّٰهُ وَلِيُّهُمَا ؕ وَعَلَى اللّٰهِ فَلْيَتَوَكَّلِ الْمُؤْمِنُوْنَ۔ “جب تم میں سے دو جماعتوں نے سوچا کہ ہمت ہار دیں، حالانکہ اللہ ان کا مددگار تھا، اور ایمان والوں کو اللہ ہی پر بھروسا کرنا چاہیے۔” مفسرین کے نزدیک یہ انصار کے دو قبیلے تھے، بنو سلمہ اور بنو حارثہ، جن کے دل میں منافقین کی واپسی دیکھ کر لمحہ بھر کو کمزوری کا خیال آیا، مگر وہ ثابت قدم رہے اور میدان میں ڈٹے رہے۔ حضرت جابر بن عبداللہ رضی اللہ عنہما فرمایا کرتے تھے، جس کا مفہوم صحیح بخاری میں ہے، کہ ہمیں یہ پسند نہیں کہ یہ آیت نازل نہ ہوتی، کیونکہ اس میں اللہ فرماتا ہے کہ وہ ان دونوں کا ولی ہے۔ صحابۂ کرام رضی اللہ عنہم کا ذوق دیکھیے: جس آیت میں ان کی ایک لمحے کی کمزوری کا ذکر تھا، اسی میں انہوں نے اپنی سب سے بڑی سند پا لی۔ اہلِ معرفت کہتے ہیں کہ هَمّ ارادے کی پہلی لرزش ہے، عزم نہیں؛ ایک خیال دل سے گزرتا ہے اور ولایت اسے عمل بننے سے روک لیتی ہے۔ شیخِ اکبر کے نزدیک بندے کی ثابت قدمی اس کی اپنی کمائی سے پہلے حق کی ولایت کا اثر ہے؛ اسمِ الولی اپنے دوستوں کو گرنے سے پہلے تھام لیتا ہے۔ توکل کا حکم آخر میں اس لیے آیا کہ جس نے کمزوری کا لمحہ دیکھا ہو وہی جانتا ہے کہ سہارا اپنے اندر نہیں۔ اس مکتب میں توکل اسباب چھوڑنے کا نام نہیں: حضورِ اکرم صلی اللہ علیہ وسلم نے صفیں باندھیں، زرہ پہنی، تیر انداز بٹھائے، اور پھر بھی دل کا رخ صرف اللہ کی طرف رکھا۔ توکل یہ ہے کہ ہاتھ سبب پر ہو اور نگاہ مسبب پر۔

اب یاد دلایا کہ مدد کا تجربہ تمہارے لیے نیا نہیں: وَلَقَدْ نَصَرَكُمُ اللّٰهُ بِبَدْرٍ وَّاَنْتُمْ اَذِلَّةٌ ۚ فَاتَّقُوا اللّٰهَ لَعَلَّكُمْ تَشْكُرُوْنَ۔ “اور اللہ بدر میں تمہاری مدد کر چکا ہے جب تم بے سروسامان تھے، پس اللہ سے ڈرو تاکہ تم شکر گزار بنو۔” رمضان دو ہجری میں بدر کے مقام پر تین سو تیرہ کے قریب صحابۂ کرام رضی اللہ عنہم تھے، چند گھوڑے اور ستر کے قریب اونٹ، اور سامنے ہزار کے قریب ہتھیار بند لشکر۔ یہاں اَذِلَّةٌ کا مطلب ذلت نہیں، ظاہری سامان اور تعداد کی کمی ہے؛ عزت تو اسی دن ان کے نام لکھ دی گئی، اور اہلِ بدر رضی اللہ عنہم کا مرتبہ امت میں ہمیشہ سب سے بلند مانا گیا۔ اہلِ ذوق کہتے ہیں کہ یہاں ایک قانون بیان ہوا ہے: مدد افتقار پر اترتی ہے۔ جب بندہ اپنی کمزوری کو پہچان لیتا ہے اور اپنے سامان پر بھروسا چھوڑ دیتا ہے، تب نصرت کا دروازہ کھلتا ہے۔ شیخِ اکبر کے مکتب میں فقر بندے کی اصل حقیقت ہے، کوئی عارضی حال نہیں، اور سورۂ فاطر کی آیت يٰۤاَيُّهَا النَّاسُ اَنْتُمُ الْفُقَرَآءُ اِلَى اللّٰهِ اسی حقیقت کا اعلان ہے۔ جس دن بندہ اپنے فقر کو دیکھ لے، اسی دن وہ غنی کی مدد کا مظہر بنتا ہے۔ پھر فرمایا کہ تقویٰ اختیار کرو تاکہ شکر کرو؛ گویا شکر صرف زبان کا لفظ نہیں، تقویٰ کی صورت میں ادا ہوتا ہے۔ جو نعمت پا کر نافرمانی کرے اس نے نعمت کا شکر نہیں کیا، اور جو نعمت کو دینے والے کی رضا میں صرف کرے وہی شاکر ہے۔ اور اس یاد دہانی کا موقع دیکھیے: احد کے زخموں کے بیچ بدر کی یاد، تاکہ دل جان لے کہ جس نے کل مدد کی تھی وہ آج بھی وہی ہے۔

پھر فرشتوں کی مدد کا ذکر آیا: اِذْ تَقُوْلُ لِلْمُؤْمِنِيْنَ اَلَنْ يَّكْفِيَكُمْ اَنْ يُّمِدَّكُمْ رَبُّكُمْ بِثَلٰثَةِ اٰلٰفٍ مِّنَ الْمَلٰٓئِكَةِ مُنْزَلِيْنَ۔ “جب آپ ایمان والوں سے فرما رہے تھے: کیا تمہارے لیے یہ کافی نہیں کہ تمہارا رب تین ہزار اتارے ہوئے فرشتوں سے تمہاری مدد کرے؟” بَلٰۤى ۙ اِنْ تَصْبِرُوْا وَتَتَّقُوْا وَيَاْتُوْكُمْ مِّنْ فَوْرِهِمْ هٰذَا يُمْدِدْكُمْ رَبُّكُمْ بِخَمْسَةِ اٰلٰفٍ مِّنَ الْمَلٰٓئِكَةِ مُسَوِّمِيْنَ۔ “کیوں نہیں، اگر تم صبر کرو اور تقویٰ اختیار کرو، اور وہ اسی دم تم پر چڑھ آئیں، تو تمہارا رب پانچ ہزار نشان والے فرشتوں سے تمہاری مدد کرے گا۔” مفسرین کے یہاں اس میں دو رائیں ہیں۔ اکثر کے نزدیک یہ بدر کا ذکر ہے، جہاں پہلے ہزار فرشتوں کا وعدہ ہوا جیسا کہ سورۂ انفال میں ہے، پھر تین ہزار، پھر صبر و تقویٰ کی شرط پر پانچ ہزار۔ بعض کے نزدیک یہ وعدہ احد کے دن کا تھا، صبر و تقویٰ کی شرط کے ساتھ، اور جب حالات بدل گئے تو پانچ ہزار کے اس مشروط وعدے کی نوبت نہ آئی۔ دونوں صورتوں میں بات ایک ہے: آسمانی مدد کا پیمانہ بندوں کے صبر اور تقویٰ سے بندھا ہوا ہے۔ مُسَوِّمِيْنَ کے معنی نشان والے ہیں، اور بعض روایات میں آیا ہے کہ بدر کے دن فرشتے خاص نشانوں اور عماموں کے ساتھ اترے۔ اہلِ معرفت اس میں ایک باطنی ترتیب دیکھتے ہیں: تین سے پانچ تک کا بڑھنا بتاتا ہے کہ صبر اور تقویٰ کے ساتھ مدد بڑھتی جاتی ہے۔ دل کے میدان میں بھی یہی ہوتا ہے؛ جو سالک آزمائش میں صبر کرتا ہے، اس کے دل پر خواطرِ ملکی بڑھ جاتے ہیں، یعنی خیر کی طرف بلانے والے الہامات، یہاں تک کہ نفس کے لشکر ان کے سامنے ٹھہر نہیں سکتے۔ شیخِ اکبر کے نزدیک فرشتے اسماءِ الٰہیہ کے مظاہر ہیں جو حق کے حکم سے اترتے ہیں؛ وہ مدد کی صورت ہیں، مدد کا سرچشمہ نہیں۔ اسی بات کو اگلی آیت کھول دیتی ہے۔

اب توحید کا وہ لطیف سبق جس پر اس رکوع کی بنیاد ہے: وَمَا جَعَلَهُ اللّٰهُ اِلَّا بُشْرٰى لَكُمْ وَلِتَطْمَئِنَّ قُلُوْبُكُمْ بِهٖ ؕ وَمَا النَّصْرُ اِلَّا مِنْ عِنْدِ اللّٰهِ الْعَزِيْزِ الْحَكِيْمِ۔ “اور اللہ نے اسے نہیں بنایا مگر تمہارے لیے خوش خبری، اور تاکہ اس سے تمہارے دل مطمئن ہو جائیں، اور مدد تو صرف اللہ ہی کی طرف سے ہے جو غالب ہے، حکمت والا ہے۔” فرشتوں کا اترنا حق ہے اور اہلِ سنت اس پر ایمان رکھتے ہیں؛ مگر آیت نے ساتھ ہی بتا دیا کہ فرشتے دلوں کی تسلی ہیں، اصل فاعل نہیں۔ انسان کا دل اسباب سے مانوس ہے، اس لیے رب نے اس کی کمزوری پر رحم فرما کر اسباب بھی دکھائے، اور پردہ بھی اٹھا دیا کہ سبب کے پیچھے کون ہے۔ شیخِ اکبر کے مکتب میں اسے تجلیِ افعالی کا شہود کہتے ہیں: عارف فرشتوں میں مدد کو دیکھتا ہے، فرشتوں سے مدد کو نہیں دیکھتا۔ اسباب پردے بھی ہیں اور آئینے بھی؛ غافل کے لیے پردہ، کیونکہ وہ ان پر رک جاتا ہے، اور عارف کے لیے آئینہ، کیونکہ وہ ان میں دکھانے والے کو پہچان لیتا ہے۔ پھر دو اسماء پر آیت ختم ہوئی۔ اَلْعَزِيْزُ وہ ہے جس پر کوئی غالب نہیں آ سکتا، اس لیے نصرت اسی کی ہے؛ اور اَلْحَكِيْمُ وہ ہے جو ہر چیز کو اس کے مقام پر رکھتا ہے، اس لیے وہ اسباب کو بے کار نہیں کرتا بلکہ انہیں ان کی جگہ پر رکھتا ہے۔ عزت کہتی ہے کہ مدد صرف وہیں سے ہے، اور حکمت کہتی ہے کہ مدد اسباب کے راستے سے آئے گی۔ ان دونوں کا جمع ہی صحیح توحید ہے، جس میں نہ سبب کا انکار ہے نہ سبب کی پرستش۔ اور خوش خبری کا لفظ بتاتا ہے کہ رب اپنے بندوں کے دلوں کی خوشی کا بھی خیال رکھتا ہے؛ وہ صرف فتح نہیں دیتا، فتح سے پہلے دل کو سکون بھی دیتا ہے۔

اور اس مدد کا مقصد بیان ہوا: لِيَقْطَعَ طَرَفًا مِّنَ الَّذِيْنَ كَفَرُوْۤا اَوْ يَكْبِتَهُمْ فَيَنْقَلِبُوْا خَآئِبِيْنَ۔ “تاکہ کافروں کا ایک حصہ کاٹ دے، یا انہیں ذلیل کر کے پسپا کر دے، تو وہ نامراد ہو کر لوٹ جائیں۔” بدر کے دن یہی ہوا: قریش کے ستر سردار مارے گئے اور ستر قید ہوئے، اور جو بچے وہ ناکام لوٹے۔ طَرَفٌ کنارے اور حصے کو کہتے ہیں، اور بعض مفسرین نے اسے سرداروں اور اشراف پر محمول کیا ہے، کیونکہ کسی قوم کی قوت انہی سے ہوتی ہے۔ يَكْبِتَ کے معنی ہیں ایسا رسوا کرنا کہ آدمی غم سے دب کر رہ جائے۔ اہلِ تاویل اس آیت کو باطن کے بدر پر بھی پڑھتے ہیں، اس وضاحت کے ساتھ کہ یہ تمثیل ہے اور ظاہر کا معنی اپنی جگہ قائم ہے۔ ہر سالک کے اندر ایک معرکہ ہے: ایک طرف روح اور اس کے ساتھی، یعنی ذکر، صدق اور تقویٰ؛ دوسری طرف نفس کے لشکر، یعنی حرص، تکبر اور غفلت۔ جب حق کی مدد آتی ہے تو پہلے نفس کے سرداروں کو کاٹا جاتا ہے، کیونکہ باقی خواہشیں انہی کے سہارے کھڑی ہیں۔ جو سالک تکبر کا سر کاٹ دے، اس کی بہت سی چھوٹی برائیاں خود پسپا ہو جاتی ہیں۔ شیخِ اکبر کے نزدیک یہ مجاہدہ بھی حق کی توفیق ہی سے ہے؛ بندہ میدان میں کھڑا رہتا ہے اور فتح اوپر سے اترتی ہے۔ اور خَآئِبِيْنَ کا لفظ یاد دلاتا ہے کہ جو قوت حق کے مقابلے پر آئے، اس کی امید ہی اس کی شکست بن جاتی ہے: وہ جس آس کو لے کر آئی تھی، اسی سے خالی ہاتھ لوٹتی ہے۔

اب وہ آیت جو اس رکوع کا دل ہے: لَيْسَ لَكَ مِنَ الْاَمْرِ شَيْءٌ اَوْ يَتُوْبَ عَلَيْهِمْ اَوْ يُعَذِّبَهُمْ فَاِنَّهُمْ ظٰلِمُوْنَ۔ “اس معاملے کا کوئی اختیار آپ کے پاس نہیں؛ اللہ چاہے تو ان کی توبہ قبول کرے یا انہیں عذاب دے، کیونکہ وہ ظالم ہیں۔” صحیح مسلم میں، جس کا مفہوم یہ ہے، آیا ہے کہ احد کے دن حضورِ اکرم صلی اللہ علیہ وسلم کا مبارک چہرہ زخمی ہوا اور سامنے کا دانت مبارک شہید ہوا، اور آپ صلی اللہ علیہ وسلم خون پونچھتے ہوئے فرما رہے تھے کہ وہ قوم کیسے فلاح پائے گی جس نے اپنے نبی صلی اللہ علیہ وسلم کو زخمی کیا حالانکہ وہ انہیں ان کے رب کی طرف بلاتے ہیں؛ تب یہ آیت اتری۔ صحیح بخاری کی ایک روایت میں ہے کہ آپ صلی اللہ علیہ وسلم نے فجر کی نماز میں رکوع کے بعد قریش کے چند سرداروں کے خلاف دعا فرمائی، جن میں بعد کے تین صحابہ، حضرت صفوان بن امیہ، حضرت سہیل بن عمرو اور حضرت حارث بن ہشام رضی اللہ عنہم کے نام آتے ہیں، جو اس وقت ابھی اسلام نہیں لائے تھے؛ اور اس پر یہ آیت نازل ہوئی۔ پھر تاریخ نے اس آیت کی تفسیر خود لکھی: یہ تینوں فتحِ مکہ کے موقع پر یا اس کے کچھ بعد اسلام لائے اور صحابۂ کرام رضی اللہ عنہم کی صف میں شامل ہوئے۔ احد کے دن قریش کے لشکر میں حضرت خالد بن ولید اور حضرت ابو سفیان رضی اللہ عنہما بھی تھے، اور دونوں بعد میں اسلام کی دولت سے سرفراز ہوئے؛ حضرت خالد رضی اللہ عنہ کو حضورِ اکرم صلی اللہ علیہ وسلم نے سیف اللہ کا لقب عطا فرمایا۔ توبہ کا جو دروازہ اس دن کھلا رکھا گیا تھا، ان کے حق میں سچ ہو کر رہا۔ یہ عتاب کی آیت نہیں، رحمت کے راز کی آیت ہے: جن کا معاملہ رب نے اپنے ہاتھ میں رکھا، انہی کو اپنے حبیب صلی اللہ علیہ وسلم کے جاں نثاروں میں داخل کر دیا۔

اہلِ معرفت کہتے ہیں کہ لَيْسَ لَكَ مِنَ الْاَمْرِ شَيْءٌ حضورِ اکرم صلی اللہ علیہ وسلم کے مقام کو گھٹاتا نہیں، بلکہ اس کی بلندی کا اعلان کرتا ہے۔ شیخِ اکبر کے مکتب میں عبودیت سب سے اونچا مقام ہے، اور حضورِ اکرم صلی اللہ علیہ وسلم عبدِ کامل ہیں، جن کی اپنی کوئی مرضی حق کی مرضی کے مقابل باقی نہیں رہی۔ امر یعنی تکوین اور فیصلے کا اختیار حق کا ہے، اور اس کا سب سے کامل اقرار وہ ہے جو سب سے کامل بندے کی ذات میں ظاہر ہو۔ اسی لیے آپ صلی اللہ علیہ وسلم نے، جس کا مفہوم حدیث میں ہے، بادشاہ نبی ہونے کے بجائے بندہ رسول ہونا پسند فرمایا۔ پھر رحمتِ نبوی کا حال دیکھیے: صحیحین میں ہے، جس کا مفہوم یہ ہے، کہ آپ صلی اللہ علیہ وسلم نے ایک نبی علیہ السلام کا حال بیان فرمایا جنہیں ان کی قوم نے لہولہان کر دیا، اور وہ چہرے سے خون پونچھتے ہوئے کہتے تھے: اے اللہ میری قوم کو بخش دے کہ وہ نہیں جانتے؛ اور شارحین نے لکھا ہے کہ اس میں آپ صلی اللہ علیہ وسلم کے اپنے حال کی جھلک بھی ہے۔ جس دل سے ایسی دعا نکلے، اسے یہ آیت گویا یہ تسلی دیتی ہے کہ ان کا معاملہ ہم پر چھوڑ دیجیے، ہماری رحمت ہر رحمت سے وسیع ہے۔ پھر آیت کی ترتیب پر غور کیجیے: توبہ کا ذکر عذاب سے پہلے آیا۔ اسے سبقتِ رحمت کہتے ہیں؛ کلام کی ترتیب میں بھی رحمت غضب سے آگے رہتی ہے۔ اور عذاب کا جو ذکر آیا، اس کی وجہ ساتھ ہی بتا دی گئی کہ وہ ظالم ہیں؛ یعنی اگر عذاب ہو تو ان کے اپنے ظلم کے سبب سے ہو گا، کسی جبر سے نہیں۔ یہ وہی اصول ہے جو سرِّ قدر کہلاتا ہے: حکم بندے کے اپنے رخ کے مطابق۔

رکوع کا خاتمہ ملکیت کے اعلان پر ہوا: وَلِلّٰهِ مَا فِي السَّمٰوٰتِ وَمَا فِي الْاَرْضِ ؕ يَغْفِرُ لِمَنْ يَّشَآءُ وَيُعَذِّبُ مَنْ يَّشَآءُ ؕ وَاللّٰهُ غَفُوْرٌ رَّحِيْمٌ۔ “اور اللہ ہی کا ہے جو کچھ آسمانوں میں ہے اور جو کچھ زمین میں ہے؛ جسے چاہے بخش دے اور جسے چاہے عذاب دے، اور اللہ بخشنے والا، مہربان ہے۔” پچھلی آیت نے کہا تھا کہ امر آپ کا نہیں؛ یہ آیت بتاتی ہے کہ پھر کس کا ہے۔ ملکیت اس کی ہے، اس لیے فیصلہ بھی اسی کا ہے۔ مگر چاہنے کے اس لفظ کو سرِّ قدر کی روشنی میں پڑھنا لازم ہے، اور یہی اس سلسلے کا بنیادی اصول ہے: حق تعالیٰ کسی کو زبردستی عذاب کا مستحق نہیں بناتا، اور کسی طلب کرنے والے کو محروم نہیں لوٹاتا۔ شیخِ اکبر کے نزدیک حکم محکوم کے مطابق ہوتا ہے؛ حق کی مشیت ہر شے کو وہی دیتی ہے جو اس کی اپنی حقیقت مانگتی ہے، اور بندہ جس رخ کو اختیار کرتا ہے، حق اسی رخ کو وجود عطا کرتا ہے۔ اسی لیے پچھلی آیت میں عذاب کی وجہ ظلم بتائی گئی، اور اسی لیے یہاں بخشش کا ذکر عذاب سے پہلے ہے۔ پھر آیت کا آخری کلمہ دیکھیے: عذاب اور مغفرت دونوں کا ذکر ہوا، مگر مہر مغفرت اور رحمت کی لگی۔ حدیثِ قدسی میں، جس کا مفہوم صحیحین میں ہے، آیا ہے کہ میری رحمت میرے غضب پر سبقت لے گئی۔ احد کے زخموں کے بیچ، دشمنوں کے ذکر کے بعد، رکوع کا آخری لفظ رحمت ہے؛ گویا اس میدان میں بھی جہاں تلواریں چلیں، رحمت کا دروازہ بند نہیں ہوا۔ اور جو بندہ اس دروازے کی طرف رخ کرے، اسے کبھی خالی نہیں لوٹایا جاتا۔
