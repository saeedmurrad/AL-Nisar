احد کے بیان کے بیچ اچانک سود کی ممانعت آتی ہے، اور یہ بے ربط نہیں: يٰۤاَيُّهَا الَّذِيْنَ اٰمَنُوْا لَا تَاْكُلُوا الرِّبٰوۤا اَضْعَافًا مُّضٰعَفَةً وَّاتَّقُوا اللّٰهَ لَعَلَّكُمْ تُفْلِحُوْنَ۔ “اے ایمان والو! سود نہ کھاؤ کئی گنا بڑھا چڑھا کر، اور اللہ سے ڈرو تاکہ تم فلاح پاؤ۔” جاہلیت میں دستور تھا کہ قرض کی مدت پوری ہوتی اور مقروض ادا نہ کر پاتا تو مہلت کے بدلے رقم بڑھا دی جاتی، پھر اگلی مدت پر پھر بڑھا دی جاتی، یہاں تک کہ اصل رقم کئی گنا ہو جاتی۔ یہ الفاظ اسی ظالمانہ صورت کا بیان ہیں جو اس وقت رائج تھی؛ اس کا مطلب ہرگز یہ نہیں کہ کم سود جائز ہے۔ امت کا اتفاق ہے کہ سود تھوڑا ہو یا بہت، حرام ہے، اور سورۂ بقرہ کی آیات نے اسے پوری قطعیت سے ختم کیا۔ علما نے یہاں ربط یہ بیان کیا ہے کہ احد کے دن ایک لغزش غنیمت کے خیال سے پیش آئی تھی، جسے اللہ تعالیٰ نے صاف لفظوں میں معاف فرما دیا، جیسا کہ آگے آتا ہے: وَلَقَدْ عَفَا اللّٰهُ عَنْهُمْ؛ اس لیے یہاں مال کے بارے میں پوری امت کے دلوں کی تربیت کی گئی۔ سود مال کی محبت کی سب سے سخت صورت ہے، جس میں ضرورت مند کی مجبوری کمائی کا ذریعہ بن جاتی ہے؛ اور جو دل مال کا اسیر ہو، وہ حق کے میدان میں پورا ثابت قدم نہیں رہتا۔ پھر ممانعت کے فوراً بعد تقویٰ اور فلاح کا ذکر کر کے بتایا کہ سود میں جو بڑھوتری نظر آتی ہے وہ دھوکا ہے، اصل بڑھوتری تقویٰ میں ہے۔

شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں رِبا کو اس کے لغوی معنی سے بھی دیکھا جاتا ہے، یعنی زیادتی اور بڑھوتری۔ اہلِ معرفت کہتے ہیں کہ ساری بڑھوتری حق کی ہے؛ وہی اسمِ الرزاق اور اسمِ الکریم سے بڑھاتا ہے۔ جو شخص کسی محنت اور عوض کے بغیر، دوسرے کی مجبوری سے بڑھوتری چاہے، وہ گویا وہ زیادتی اپنے لیے چھین رہا ہے جو صرف دینے والے کے ہاتھ میں ہے۔ اسی لیے سورۂ بقرہ میں فرمایا: يَمْحَقُ اللّٰهُ الرِّبٰوا وَيُرْبِي الصَّدَقٰتِ۔ ظاہر میں سود بڑھتا اور صدقہ گھٹتا دکھائی دیتا ہے، اور حقیقت میں معاملہ الٹا ہے۔ اہلِ ذوق اس کا ایک باطنی پہلو بھی بیان کرتے ہیں، اور اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے: ایک سود دلوں کا بھی ہے۔ جو بندہ اپنے عمل کو اپنی ملکیت سمجھ کر رب سے اس کا کئی گنا بدلہ ایک قرض خواہ کی طرح مانگے، اس کی عبادت میں بھی ایک طرح کی سود خوری آ جاتی ہے۔ عارف اپنے عمل کو اپنی کمائی نہیں، رب کی توفیق دیکھتا ہے؛ پھر جو بدلہ ملتا ہے اسے فضل سمجھتا ہے، اپنا حق نہیں۔ اسی لیے رب نے صدقے کو بڑھانا اپنے ذمے لیا، اور بندے سے یہ نہیں چاہا کہ وہ حساب لگا کر دے۔ اور اس رکوع میں ایک لطیف نکتہ ہے: مٹانے کے معنی والا یہی مادہ آگے کافروں کے لیے آیا ہے؛ گویا سود کا مال اور حق کے مقابل کھڑی طاقت، دونوں کا انجام ایک ہے، آہستہ آہستہ گھلتے گھلتے مٹ جانا۔

پھر دو مختصر آیتیں آئیں جو ڈرانے اور امید دلانے کو جمع کرتی ہیں: وَاتَّقُوا النَّارَ الَّتِيْۤ اُعِدَّتْ لِلْكٰفِرِيْنَ۔ “اور اس آگ سے بچو جو کافروں کے لیے تیار کی گئی ہے۔” وَاَطِيْعُوا اللّٰهَ وَالرَّسُوْلَ لَعَلَّكُمْ تُرْحَمُوْنَ۔ “اور اللہ اور رسول صلی اللہ علیہ وسلم کی اطاعت کرو تاکہ تم پر رحم کیا جائے۔” آگ اصل میں انکار کرنے والوں کے لیے تیار کی گئی، پھر بھی ایمان والوں کو اس سے ڈرایا گیا۔ اہلِ معرفت کہتے ہیں کہ یہ ڈرانا شفقت ہے: ماں بچے کو آگ سے اسی لیے ڈراتی ہے کہ وہ اسے جلتا نہیں دیکھ سکتی۔ اور سود کی ممانعت کے ساتھ یہ تنبیہ اس لیے آئی کہ سود خوری انسان کو آہستہ آہستہ اسی راستے پر لے جاتی ہے جو انکار والوں کا ہے۔ اگلی آیت میں اللہ اور رسول صلی اللہ علیہ وسلم کی اطاعت ایک ہی فعل میں جمع کی گئی؛ شیخِ اکبر کے نزدیک رسول صلی اللہ علیہ وسلم کا حکم حق ہی کے حکم کی ترجمانی ہے، اور رسول صلی اللہ علیہ وسلم کی اطاعت عین اللہ کی اطاعت ہے، جیسا کہ فرمایا: مَنْ يُّطِعِ الرَّسُوْلَ فَقَدْ اَطَاعَ اللّٰهَ۔ احد کا سبق بھی یہی تھا کہ حضورِ اکرم صلی اللہ علیہ وسلم کا حکم ہر حال میں مقدم ہے، چاہے حالات بدلے ہوئے نظر آئیں۔ اور اطاعت کا پھل رحمت بتایا گیا، جو اسی سورت کی اس آیت کی یاد دلاتا ہے: فَاتَّبِعُوْنِيْ يُحْبِبْكُمُ اللّٰهُ۔ اس مکتب میں حضورِ اکرم صلی اللہ علیہ وسلم کی پیروی محبوب بننے کا دروازہ ہے، اور رحمت اسی محبت کی پہلی جھلک ہے۔

اب دوڑ کا حکم آتا ہے: وَسَارِعُوْۤا اِلٰى مَغْفِرَةٍ مِّنْ رَّبِّكُمْ وَجَنَّةٍ عَرْضُهَا السَّمٰوٰتُ وَالْاَرْضُ ۙ اُعِدَّتْ لِلْمُتَّقِيْنَ۔ “اور اپنے رب کی مغفرت اور اس جنت کی طرف تیزی سے بڑھو جس کی چوڑائی آسمانوں اور زمین جتنی ہے، جو پرہیزگاروں کے لیے تیار کی گئی ہے۔” چلنے کا نہیں، تیزی سے بڑھنے کا حکم ہے؛ نیکی میں سستی اس راہ کے مزاج کے خلاف ہے۔ اور ترتیب دیکھیے: پہلے مغفرت، پھر جنت۔ اہلِ تصوف اسے تخلیہ اور تحلیہ کا اصول کہتے ہیں: پہلے برتن دھویا جاتا ہے، پھر اس میں خوشبو رکھی جاتی ہے۔ جنت کی چوڑائی آسمانوں اور زمین جتنی بتائی گئی، تو اس کی لمبائی کا اندازہ کون کرے۔ شیخِ اکبر کے مکتب میں یہاں وسعتِ قلب کی طرف اشارہ دیکھا جاتا ہے۔ اہلِ تصوف میں مشہور اثر ہے، جس کا مفہوم یہ ہے کہ نہ میری زمین میری سمائی رکھتی ہے نہ میرا آسمان، مگر میرے مومن بندے کا دل مجھے سما لیتا ہے۔ جو دل اپنے رب کی معرفت کو سما لے، وہ آسمانوں اور زمین سے وسیع ہے؛ اور ایسے دل کی جنت بھی ایسی ہی وسیع ہے۔ پھر ایک اور توازن دیکھیے: آگ کے بارے میں فرمایا کہ وہ کافروں کے لیے تیار ہے، اور جنت کے بارے میں کہ وہ متقین کے لیے تیار ہے۔ دونوں گھر تیار ہیں، اور بندہ اپنے رخ سے طے کرتا ہے کہ کس کی طرف چلے؛ یہی بندے کے اختیار کا وقار ہے۔

پھر متقین کی پہچان بیان ہوئی، اور یہ اس سورت کی سب سے روشن آیتوں میں سے ہے: الَّذِيْنَ يُنْفِقُوْنَ فِي السَّرَّآءِ وَالضَّرَّآءِ وَالْكٰظِمِيْنَ الْغَيْظَ وَالْعَافِيْنَ عَنِ النَّاسِ ؕ وَاللّٰهُ يُحِبُّ الْمُحْسِنِيْنَ۔ “جو خوشحالی اور تنگی میں خرچ کرتے ہیں، اور غصہ پی جانے والے ہیں، اور لوگوں کو معاف کر دینے والے ہیں، اور اللہ احسان کرنے والوں سے محبت کرتا ہے۔” پہلی صفت خرچ کرنا ہے، اور اس کی شرط یہ نہیں کہ مال بچ جائے؛ خوشحالی میں بھی دیتے ہیں اور تنگی میں بھی۔ اہلِ معرفت کہتے ہیں کہ جو اپنے آپ کو مالک سمجھے وہ تنگی میں ہاتھ روک لیتا ہے، اور جو اپنے آپ کو اسمِ الوہاب کا مظہر اور ایک گزرگاہ سمجھے، اس کے لیے خوشحالی اور تنگی برابر ہیں، کیونکہ دینے والا وہ خود نہیں۔ دوسری صفت غصہ پی جانا ہے۔ عربی میں کظم مشک کو بھر کر اس کا منہ باندھ دینے کو کہتے ہیں؛ پانی اندر موجود ہے مگر بہنے نہیں دیا جاتا۔ یعنی متقی وہ نہیں جسے غصہ آتا ہی نہ ہو، بلکہ وہ ہے جس کے اندر غصہ بھرا ہو اور وہ اس کا منہ باندھ دے۔ اور غیظ عام غصہ نہیں، سخت جوش مارنے والا غصہ ہے۔ حضورِ اکرم صلی اللہ علیہ وسلم نے، جس کا مفہوم صحیحین میں ہے، فرمایا کہ طاقتور وہ نہیں جو کشتی میں پچھاڑ دے، طاقتور وہ ہے جو غصے کے وقت اپنے آپ پر قابو رکھے۔

شیخِ اکبر کے مکتب میں غصے کو جلال کا عکس کہا جاتا ہے: بندے کے اندر جو قوتِ غضب رکھی گئی ہے، وہ حق کے اسماءِ جلال کا ایک پرتو ہے، اور اپنی جگہ بے کار نہیں؛ اسی سے بندہ ظلم کے مقابل کھڑا ہوتا ہے۔ مگر جب یہ قوت نفس کی خاطر جوش مارے تو آدمی کو اندھا کر دیتی ہے۔ غصہ پی جانے والا اس قوت کو مٹاتا نہیں، اسے حکم کا پابند کرتا ہے، اور یوں اسمِ الحلیم سے حصہ پاتا ہے؛ اسے تخلق باسماء اللہ کہتے ہیں۔ حلیم وہ ہے جو قدرت کے باوجود جلدی سزا نہیں دیتا، اور غصہ پینے کی فضیلت بھی اسی میں ہے کہ بدلے کی قدرت ہو اور آدمی رک جائے۔ ام المومنین حضرت عائشہ صدیقہ رضی اللہ عنہا بیان فرماتی ہیں، جس کا مفہوم صحیحین میں ہے، کہ حضورِ اکرم صلی اللہ علیہ وسلم نے کبھی اپنی ذات کے لیے بدلہ نہیں لیا، ہاں جب اللہ کی حرمت پامال کی جاتی تو اللہ کے لیے بدلہ لیتے۔ یہی جلال اور جمال کا جمع ہے: اپنی ذات کے لیے جمال، اور حق کی حرمت کے لیے جلال۔ اہلِ ذوق کہتے ہیں کہ جس نے اپنے غصے کو اپنی ذات کے لیے پی لیا، اس کا دل اس برتن کی طرح ہو جاتا ہے جس کا پانی ٹھہر جائے؛ پھر اس میں حق کا عکس صاف دکھائی دیتا ہے، جیسے ٹھہرے ہوئے پانی میں چاند۔ اور جس کا دل غصے سے کھولتا رہے، اس میں کوئی عکس نہیں ٹھہرتا۔

تیسری صفت معاف کرنا ہے، اور اس کے بعد احسان کا ذکر آیا۔ گویا یہ تین سیڑھیاں ہیں: غصہ پی جانا، پھر دل سے معاف کر دینا، پھر برائی کرنے والے کے ساتھ بھلائی کرنا۔ مشہور روایت ہے کہ حضرت امام زین العابدین علی بن حسین رضی اللہ عنہما کی ایک کنیز وضو کراتے ہوئے برتن ان پر گرا بیٹھی، اور آپ کی طرف دیکھا تو اس نے آیت کا پہلا ٹکڑا پڑھا؛ فرمایا میں نے غصہ پی لیا۔ اس نے دوسرا ٹکڑا پڑھا؛ فرمایا میں نے تجھے معاف کیا۔ اس نے تیسرا ٹکڑا پڑھا؛ فرمایا جا، تو اللہ کے لیے آزاد ہے۔ یہ اہلِ بیت کا اخلاق ہے، اور اس آیت کی زندہ تفسیر۔ احسان کی تعریف حضورِ اکرم صلی اللہ علیہ وسلم نے حدیثِ جبریل علیہ السلام میں فرمائی، جس کا مفہوم یہ ہے کہ تم اللہ کی عبادت ایسے کرو گویا اسے دیکھ رہے ہو، اور اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے۔ شیخِ اکبر کے نزدیک احسان شہود کا مقام ہے، اور اسی سے معاف کرنے کا راز کھلتا ہے: محسن برائی کرنے والے کے پیچھے بھی حق کی تقدیر کا ہاتھ دیکھتا ہے، اس لیے اس کے دل میں کینہ جڑ نہیں پکڑتا، اور بندے کی ذمہ داری کا انکار بھی نہیں کرتا۔ پھر آیت کا خاتمہ محبت پر ہوا؛ احسان کا بدلہ جنت سے پہلے محبوبیت ہے۔ اور سورۂ رحمٰن کا سوال اسی کی گونج ہے کہ احسان کا بدلہ احسان کے سوا کیا ہے۔

اب اس رکوع کی سب سے امید افزا آیت آتی ہے، اور حیرت کی بات یہ ہے کہ یہ بھی متقین ہی کے بیان میں ہے: وَالَّذِيْنَ اِذَا فَعَلُوْا فَاحِشَةً اَوْ ظَلَمُوْۤا اَنْفُسَهُمْ ذَكَرُوا اللّٰهَ فَاسْتَغْفَرُوْا لِذُنُوْبِهِمْ۔ “اور وہ لوگ کہ جب کوئی بے حیائی کا کام کر بیٹھیں یا اپنی جانوں پر ظلم کر لیں تو اللہ کو یاد کرتے ہیں، پھر اپنے گناہوں کی معافی مانگتے ہیں۔” تقویٰ کا مطلب معصوم ہونا نہیں؛ متقی وہ ہے جو گر کر لوٹ آئے۔ فاحشہ بڑا اور کھلا گناہ ہے، اور اپنی جان پر ظلم اس سے کم درجے کی لغزش؛ دونوں کا علاج ایک ہی ہے۔ اور پہلا قدم ذکر ہے: گناہ کے بعد انہیں اللہ یاد آیا، اس کی عظمت بھی اور اس کی بخشش کا وعدہ بھی؛ وہ اپنے گناہ کو مایوسی کی تاریکی میں نہیں، اللہ کی یاد کی روشنی میں دیکھتے ہیں۔ سنن ابی داؤد اور جامع ترمذی میں حضرت علی رضی اللہ عنہ، حضرت ابوبکر صدیق رضی اللہ عنہ سے روایت کرتے ہیں کہ حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا، جس کا مفہوم یہ ہے کہ جو بندہ کوئی گناہ کر بیٹھے، پھر اچھی طرح وضو کرے، دو رکعت پڑھے اور اللہ سے معافی مانگے، اللہ اسے معاف فرما دیتا ہے؛ پھر آپ صلی اللہ علیہ وسلم نے یہی آیت تلاوت فرمائی۔ شیخِ اکبر کے مکتب میں اسماءِ الٰہیہ اپنے مظاہر چاہتے ہیں؛ اسمِ الغفار اور اسمِ التواب کا ظہور توبہ کرنے والے بندے میں ہوتا ہے۔ صحیح مسلم کی حدیث، جس کا مفہوم یہ ہے کہ اگر تم گناہ نہ کرتے تو اللہ ایسی قوم لے آتا جو گناہ کرتی، پھر معافی مانگتی اور وہ انہیں بخش دیتا، گناہ کی ترغیب نہیں، لوٹ آنے کے وقار کا اعلان ہے۔

پھر آیت کے بیچ ایک جملہ آیا جو گویا گفتگو روک کر کہا گیا: وَمَنْ يَّغْفِرُ الذُّنُوْبَ اِلَّا اللّٰهُ وَلَمْ يُصِرُّوْا عَلٰى مَا فَعَلُوْا وَهُمْ يَعْلَمُوْنَ۔ “اور اللہ کے سوا گناہوں کو بخشنے والا کون ہے؟ اور وہ جانتے بوجھتے اپنے کیے پر اڑے نہیں رہتے۔” اہلِ ذوق کہتے ہیں کہ یہ جملۂ معترضہ محبت کا جملہ ہے: جیسے کوئی شفیق باپ بیٹے کی غلطی کا ذکر کرتے کرتے رک کر کہے کہ آخر اسے میرے سوا کون معاف کرے گا۔ مغفرت کی اصل ڈھانپ لینا ہے، اور سر کو ڈھانپنے والے جنگی خود کو اسی لیے مِغفر کہتے ہیں؛ حق گناہ کو ڈھانپتا بھی ہے اور اس کے انجام سے بچاتا بھی ہے۔ شیخِ اکبر کے نزدیک جس نے یہ جان لیا کہ بخشنے والا صرف وہی ہے، وہ توحیدِ افعال کے ایک مقام پر پہنچ گیا؛ اب وہ نہ اپنے عمل پر بھروسا کرتا ہے نہ کسی اور در پر جاتا ہے۔ پھر شرط یہ بتائی کہ وہ اصرار نہیں کرتے۔ اصل پردہ لغزش نہیں، اصرار ہے؛ لغزش ایک لمحے کی ہوتی ہے، اصرار دل کا رخ بن جاتا ہے۔ حضرت عبداللہ بن عباس رضی اللہ عنہما سے ایک اثر منقول ہے، جس کا مفہوم یہ ہے کہ اصرار کے ساتھ کوئی گناہ چھوٹا نہیں رہتا اور استغفار کے ساتھ کوئی گناہ بڑا نہیں رہتا۔ اور جانتے بوجھتے کا لفظ دو طرف اشارہ کرتا ہے: وہ جانتے ہیں کہ یہ گناہ ہے، اور یہ بھی جانتے ہیں کہ ان کا رب بخشنے والا ہے؛ اسی علم سے وہ لوٹتے ہیں۔

اور ان سب کا بدلہ بیان ہوا: اُولٰۤئِكَ جَزَآؤُهُمْ مَّغْفِرَةٌ مِّنْ رَّبِّهِمْ وَجَنّٰتٌ تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ خٰلِدِيْنَ فِيْهَا ؕ وَنِعْمَ اَجْرُ الْعٰمِلِيْنَ۔ “یہی لوگ ہیں جن کا بدلہ ان کے رب کی طرف سے مغفرت ہے، اور ایسے باغ جن کے نیچے نہریں بہتی ہیں، وہ ان میں ہمیشہ رہیں گے، اور عمل کرنے والوں کا کیا ہی اچھا اجر ہے۔” دائرہ مکمل ہو گیا: جس مغفرت اور جنت کی طرف دوڑنے کا حکم دیا گیا تھا، وہی ان کا بدلہ ٹھہرا۔ اور غور کیجیے کہ گناہ کر کے لوٹنے والوں کو بھی وہی بدلہ ملا جو خرچ کرنے والوں اور غصہ پینے والوں کا ہے؛ کیونکہ یہ سب ایک ہی جماعت ہیں، متقین کی جماعت۔ پھر انہیں عمل کرنے والے کہا، یعنی مزدور جن کی مزدوری اچھی ہے؛ اہلِ معرفت کہتے ہیں کہ یہ اسمِ الشکور کا ظہور ہے، جو تھوڑے عمل پر بہت دیتا ہے اور اپنے فضل کو بندے کی کمائی کا نام دے کر اس کا دل خوش کرتا ہے۔ اہلِ تاویل باغوں کے نیچے بہتی نہروں کو ان علوم اور معارف کی تمثیل بھی کہتے ہیں جو استغفار سے دھلے ہوئے دل کی گہرائی سے پھوٹتے ہیں، اور یہ ظاہری جنت اور اس کی حقیقی نہروں کے انکار کے بغیر ہے۔ جو دل بار بار لوٹتا رہے، اس کی زمین نرم رہتی ہے، اور نرم زمین ہی سے چشمے پھوٹتے ہیں۔

اب گفتگو پھر احد کی طرف لوٹتی ہے، تاریخ کی روشنی میں: قَدْ خَلَتْ مِنْ قَبْلِكُمْ سُنَنٌ ۙ فَسِيْرُوْا فِي الْاَرْضِ فَانْظُرُوْا كَيْفَ كَانَ عَاقِبَةُ الْمُكَذِّبِيْنَ۔ “تم سے پہلے بہت سے طریقے گزر چکے ہیں، تو زمین میں چلو پھرو اور دیکھو کہ جھٹلانے والوں کا انجام کیسا ہوا۔” هٰذَا بَيَانٌ لِّلنَّاسِ وَهُدًى وَّمَوْعِظَةٌ لِّلْمُتَّقِيْنَ۔ “یہ لوگوں کے لیے واضح بیان ہے، اور پرہیزگاروں کے لیے ہدایت اور نصیحت۔” سنن سے مراد قوموں کے ساتھ اللہ کا دستور ہے؛ تاریخ اتفاقات کا ڈھیر نہیں، ایک قانون کے تحت چلتی ہے۔ شیخِ اکبر کے مکتب میں یہ سنن حق کی تجلی کی عادتیں ہیں، جو ہر زمانے میں نئے لباس میں ظاہر ہوتی ہیں مگر اپنی حقیقت میں ایک رہتی ہیں۔ اور زمین میں سیر کے دو رخ ہیں: ایک باہر کی زمین میں چلنا اور قوموں کے کھنڈر دیکھنا، اور دوسرا اپنے اندر کی زمین میں سفر کرنا اور نفس کے انجام دیکھنا۔ سورۂ حٰمٓ السجدہ میں اسی دوہری سیر کی طرف اشارہ ہے: سَنُرِيْهِمْ اٰيٰتِنَا فِي الْاٰفَاقِ وَفِيْۤ اَنْفُسِهِمْ۔ پھر دوسری آیت نے ایک ہی کلام کے تین درجے بتائے: بیان سب لوگوں کے لیے ہے، مگر ہدایت اور نصیحت متقین کے لیے۔ روشنی سب پر یکساں پڑتی ہے، مگر فائدہ وہی اٹھاتا ہے جس کی آنکھ کھلی ہو؛ اور آنکھ کا کھلنا طلب سے ہے۔ یہ اسی اصول کی تصویر ہے کہ جو طلب کرے اسے ہدایت ملتی ہے، اور جو منہ پھیرے اس کا منہ پھیرنا ہی اس کی محرومی ہے۔

اب زخمی دلوں سے براہِ راست خطاب ہے: وَلَا تَهِنُوْا وَلَا تَحْزَنُوْا وَاَنْتُمُ الْاَعْلَوْنَ اِنْ كُنْتُمْ مُّؤْمِنِيْنَ۔ “اور نہ ہمت ہارو اور نہ غم کرو، تم ہی سربلند رہو گے اگر تم مومن ہو۔” احد کے بعد کا منظر سامنے رکھیے: ستر کے قریب صحابۂ کرام رضی اللہ عنہم شہید ہوئے، جن میں سید الشہداء حضرت حمزہ رضی اللہ عنہ بھی تھے، حضورِ اکرم صلی اللہ علیہ وسلم زخمی تھے، اور مدینہ منورہ کے گھروں میں سوگ تھا۔ ایسے وقت میں یہ آیت اتری، اور اگلے ہی دن حضورِ اکرم صلی اللہ علیہ وسلم انہی زخمی جاں نثاروں کو لے کر دشمن کے تعاقب میں حمراء الاسد تک تشریف لے گئے؛ گویا آیت کا جواب عمل سے دیا گیا۔ وہن کمزوری ہے جو آنے والے کام سے روکے، اور حزن غم ہے جو گزرے ہوئے پر ہو؛ آیت نے دونوں سمتوں کا دروازہ بند کر دیا۔ اہلِ تصوف سالک کو ابن الوقت کہتے ہیں، یعنی جو گزرے پر غم اور آنے والے کے خوف میں نہیں جیتا، بلکہ اس لمحے میں حق کے ساتھ ہوتا ہے۔ اور سربلندی کا وعدہ شکست کے بعد دیا گیا، فتح کے بعد نہیں؛ کیونکہ مومن کی بلندی میدان کے نتیجے سے نہیں، ایمان سے ہے۔ شیخِ اکبر کے نزدیک علو اسمِ العلی کا ہے، اور بندے کو اس میں سے اتنا حصہ ملتا ہے جتنا وہ اس کے ساتھ ہے؛ اسی لیے شرط لگائی کہ اگر تم مومن ہو۔ احد کے دن جب ایک بت کی بلندی کا نعرہ لگا تو حضورِ اکرم صلی اللہ علیہ وسلم نے، جس کا مفہوم صحیح بخاری میں ہے، جواب دلوایا کہ اللہ سب سے بلند اور سب سے بزرگ ہے۔

پھر ایک ایسی حقیقت بیان ہوئی جو ہر دور کے دلوں کا مرہم ہے: اِنْ يَّمْسَسْكُمْ قَرْحٌ فَقَدْ مَسَّ الْقَوْمَ قَرْحٌ مِّثْلُهٗ ؕ وَتِلْكَ الْاَيَّامُ نُدَاوِلُهَا بَيْنَ النَّاسِ۔ “اگر تمہیں زخم پہنچا ہے تو اس قوم کو بھی ویسا ہی زخم پہنچ چکا ہے، اور یہ دن ہیں جنہیں ہم لوگوں کے درمیان پھیرتے رہتے ہیں۔” بدر میں قریش زخمی ہوئے تھے، احد میں مسلمان؛ زخم صرف ایک طرف نہیں لگتا۔ نُدَاوِلُهَا کی اصل ہاتھوں ہاتھ پھرنا ہے، جیسے کوئی چیز ایک ہاتھ سے دوسرے ہاتھ میں جاتی رہے۔ اہلِ معرفت کہتے ہیں کہ دن حق کی تجلیات ہیں، اور ہر دن کسی اسم کا دن ہے: کوئی جمال کا، کوئی جلال کا؛ کسی دن بسط ہے، کسی دن قبض۔ سورۂ رحمٰن کی آیت اسی کو کھولتی ہے: كُلَّ يَوْمٍ هُوَ فِيْ شَاْنٍ۔ شیخِ اکبر کے نزدیک یہ تجددِ امثال کی ایک صورت ہے؛ کوئی حال ٹھہرتا نہیں، ہر لمحہ نئی تخلیق ہے۔ عارف فتح کے دن غرور میں نہیں آتا اور شکست کے دن ٹوٹتا نہیں، کیونکہ وہ دنوں کو نہیں، دنوں کے پھیرنے والے کو دیکھتا ہے۔ اسی کو تلوین میں تمکین کہتے ہیں: حالات بدلتے ہیں، دل کا قبلہ نہیں بدلتا۔

اور اس گردش کی حکمت بیان ہوئی: وَلِيَعْلَمَ اللّٰهُ الَّذِيْنَ اٰمَنُوْا وَيَتَّخِذَ مِنْكُمْ شُهَدَآءَ ؕ وَاللّٰهُ لَا يُحِبُّ الظّٰلِمِيْنَ۔ “اور تاکہ اللہ ایمان والوں کو جان لے، اور تم میں سے کچھ کو شہید بنائے، اور اللہ ظالموں کو پسند نہیں کرتا۔” اللہ تو پہلے سے جانتا ہے؛ یہاں جاننے سے مراد وہ علم ہے جو ظہور کے بعد واقعے کی صورت میں سامنے آئے، تاکہ جو چھپا تھا وہ ظاہر ہو جائے۔ شیخِ اکبر کے نزدیک علم معلوم کے تابع ہے: حق ہر شے کو ویسا ہی جانتا ہے جیسی وہ اپنی ثابت حقیقت میں ہے، اور آزمائش اس حقیقت کو عالمِ شہادت میں لے آتی ہے۔ اسی لیے کسی پر ظلم نہیں ہوتا؛ کسی کا فیصلہ اس پر نہیں ہوتا جو حق جانتا تھا، بلکہ اس پر جو بندے سے ظاہر ہوا۔ پھر فرمایا کہ وہ تم میں سے شہید بناتا ہے؛ یہ لفظ چن لینے کا ہے، گویا شہادت ایک انتخاب ہے۔ احد کے شہداء میں حضرت حمزہ، حضرت مصعب بن عمیر اور حضرت عبداللہ بن جبیر رضی اللہ عنہم جیسے جلیل القدر صحابہ تھے؛ حضرت مصعب رضی اللہ عنہ کے بارے میں صحیح بخاری میں ہے، جس کا مفہوم یہ ہے، کہ ان کا کفن اتنا چھوٹا تھا کہ سر ڈھانپا جاتا تو پاؤں کھل جاتے۔ اور شہید کے معنی گواہ کے بھی ہیں؛ اہلِ ذوق کہتے ہیں کہ شہید وہ ہے جس نے حق کی گواہی اپنے خون سے دی اور شہود کے مقام تک پہنچا۔

آزمائش کا دوسرا مقصد یہ بتایا: وَلِيُمَحِّصَ اللّٰهُ الَّذِيْنَ اٰمَنُوْا وَيَمْحَقَ الْكٰفِرِيْنَ۔ “اور تاکہ اللہ ایمان والوں کو نکھار دے اور کافروں کو مٹا دے۔” تمحیص سونے کو آگ میں ڈال کر اس کا میل الگ کرنے کو کہتے ہیں، اور محق کسی چیز کا آہستہ آہستہ گھٹتے گھٹتے مٹ جانا ہے، جیسے چاند مہینے کے آخر میں۔ یہاں سرِّ قدر کی ایک صاف تصویر ہے: آگ ایک ہی ہے، مگر سونا اس میں نکھرتا ہے اور کھوٹ جل جاتا ہے۔ آزمائش کسی پر ظلم نہیں، وہ ہر شے کو اس کی اپنی حقیقت کی طرف لوٹا دیتی ہے؛ ایمان والے کا ایمان اور صاف ہو جاتا ہے، اور انکار پر جمے ہوئے کی قوت گھلتی جاتی ہے۔ حکم محکوم کے مطابق؛ اور یہی شیخِ اکبر کے نزدیک عدل کی اصل ہے۔ احد کے بعد یہی ہوا: مدینہ منورہ میں منافقین الگ پہچانے گئے، صحابۂ کرام رضی اللہ عنہم کا اخلاص اور نکھر کر سامنے آیا، اور قریش کی قوت چند برسوں میں گھلتے گھلتے فتحِ مکہ تک ختم ہو گئی۔ سالک کے لیے بھی یہی قانون ہے: جو مصیبت آئے اسے یوں دیکھے کہ یہ مجھے نکھارنے آئی ہے، مٹانے نہیں؛ اور اگر دل اس میں رب کی طرف مڑ جائے تو سمجھ لے کہ آگ نے سونے کا کام کیا۔

پھر ایک خوش فہمی کا پردہ چاک کیا گیا: اَمْ حَسِبْتُمْ اَنْ تَدْخُلُوا الْجَنَّةَ وَلَمَّا يَعْلَمِ اللّٰهُ الَّذِيْنَ جٰهَدُوْا مِنْكُمْ وَيَعْلَمَ الصّٰبِرِيْنَ۔ “کیا تم نے یہ سمجھ رکھا ہے کہ جنت میں داخل ہو جاؤ گے حالانکہ ابھی اللہ نے یہ ظاہر ہی نہیں کیا کہ تم میں سے کون جہاد کرنے والے ہیں اور کون صبر کرنے والے؟” جنت آرزو سے نہیں ملتی؛ آرزو اور منزل کے درمیان آزمائش کا میدان ہے۔ لَمَّا کا لفظ عربی میں ایسی نفی ہے جس کے بعد اس بات کے واقع ہونے کی توقع ہو؛ گویا کہا گیا کہ ابھی وہ گھڑی آئی نہیں، مگر آنے والی ہے۔ اور جہاد کے ساتھ صبر کا ذکر آیا، کیونکہ ایک لمحے کی بہادری آسان ہے، اس پر جمے رہنا مشکل۔ اہلِ تصوف میں مشہور اثر ہے، جس کا مفہوم یہ ہے کہ ہم چھوٹے جہاد سے بڑے جہاد کی طرف لوٹ آئے، یعنی نفس کے جہاد کی طرف؛ اور اس مکتب میں ظاہری جہاد اپنے مقام پر قائم رہتے ہوئے اندر کے مجاہدے کو بھی اسی آیت کا حصہ سمجھا جاتا ہے۔ شیخِ اکبر کے نزدیک بندے کی ہر استعداد ایک بیج ہے جو آزمائش کی بارش کے بغیر نہیں پھوٹتا؛ جو آزمائش سے بچ کر جنت چاہے، وہ بیج بو کر بارش سے بھاگنا چاہتا ہے۔

رکوع کا آخری کلمہ ایک یاد دہانی ہے: وَلَقَدْ كُنْتُمْ تَمَنَّوْنَ الْمَوْتَ مِنْ قَبْلِ اَنْ تَلْقَوْهُ فَقَدْ رَاَيْتُمُوْهُ وَاَنْتُمْ تَنْظُرُوْنَ۔ “اور تم تو موت کے سامنے آنے سے پہلے اس کی تمنا کیا کرتے تھے، سو اب تم نے اسے اپنی آنکھوں سے دیکھ لیا۔” جو صحابۂ کرام رضی اللہ عنہم بدر میں شریک نہ ہو سکے تھے، وہ شہادت کی تمنا رکھتے تھے، اور انہی کی رائے سے شہر سے باہر نکل کر مقابلہ ہوا۔ ان میں حضرت انس بن نضر رضی اللہ عنہ تھے، جن کے بارے میں صحیح بخاری میں ہے، جس کا مفہوم یہ ہے، کہ انہوں نے احد کے دامن میں جنت کی خوشبو پائی، اور ایسے لڑے کہ اسی سے زیادہ زخموں کے ساتھ شہید ہوئے، اور ان کی بہن نے انہیں انگلیوں کے پوروں سے پہچانا۔ یہ آیت ان کی تمنا پر طعن نہیں، اس کی سچائی کا امتحان ہے، اور جو ڈٹے رہے انہوں نے ثابت کر دیا کہ ان کی تمنا سچی تھی۔ اہلِ معرفت کہتے ہیں کہ سننا اور تمنا کرنا علم الیقین ہے، اور آنکھ سے دیکھنا عین الیقین؛ اور ان دونوں کے درمیان وہ مقام ہے جہاں بندے کا دعویٰ پرکھا جاتا ہے۔ شیخِ اکبر کے مکتب میں سالک کو یہی ادب سکھایا جاتا ہے کہ حال کا دعویٰ نہ کرے اور آزمائش نہ مانگے، بلکہ عافیت مانگے؛ اور جب آزمائش آ جائے تو ثابت قدمی مانگے۔ یہیں سے اگلا رکوع شروع ہوتا ہے، جو احد کے سب سے کڑے لمحے کا بیان ہے۔
