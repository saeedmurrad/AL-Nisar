رکوع ایک تدبیر کے بیان سے کھلتا ہے، اور قرآن پھر وہی ناپا تولا لفظ لاتا ہے: سب نہیں، ایک گروہ۔ وَقَالَتْ طَّآئِفَةٌ مِّنْ اَهْلِ الْكِتٰبِ اٰمِنُوْا بِالَّذِيْۤ اُنْزِلَ عَلَى الَّذِيْنَ اٰمَنُوْا وَجْهَ النَّهَارِ وَاكْفُرُوْۤا اٰخِرَهٗ لَعَلَّهُمْ يَرْجِعُوْنَ۔ “اور اہلِ کتاب کے ایک گروہ نے کہا: جو کچھ ایمان والوں پر اتارا گیا ہے اس پر دن کے شروع میں ایمان لے آؤ اور اس کے آخر میں انکار کر دو، شاید وہ بھی پلٹ جائیں۔” مفسرین کے نزدیک یہ مدینہ کے چند لوگوں کی سوچی سمجھی چال تھی: صبح ایمان کا اظہار، شام کو واپسی، تاکہ سادہ دل مسلمان سوچیں کہ اہلِ علم نے اندر سے دیکھ کر کچھ خرابی پائی ہوگی۔ ظاہر میں یہ ایک تاریخی واقعہ ہے اور اسی معنی میں پڑھا جائے؛ اشاری معنی اس کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ معرفت کہتے ہیں کہ “وَجْهَ النَّهَارِ” اور “اٰخِرَهٗ” ہر سالک کے دن کی بھی تصویر ہیں: صبح کی نماز کے بعد دل حق کی طرف رخ کیے ہوتا ہے، اور شام تک دنیا کے شور میں وہ رخ کہیں گم ہو جاتا ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے مکتب میں اسے تلوین کہتے ہیں، یعنی احوال کا بدلتے رہنا؛ اور منزل تمکین ہے، جہاں صبح کا رخ شام تک قائم رہے۔ فرق یہ ہے کہ اُس گروہ نے بدلنے کو تدبیر بنایا، اور سالک بدلنے پر روتا ہے اور واپس لوٹتا ہے؛ اور یہی رونا اس کے لیے لوٹنے کا دروازہ ہے۔

ان کی بات آگے چلتی ہے، اور بیچ میں حق تعالیٰ اپنے رسول صلی اللہ علیہ وسلم کی زبان سے ایک جملہ داخل کر دیتا ہے: وَلَا تُؤْمِنُوْۤا اِلَّا لِمَنْ تَبِعَ دِيْنَكُمْ ۗ قُلْ اِنَّ الْهُدٰى هُدَى اللّٰهِ ۙ۔ “اور کسی کی بات نہ مانو سوائے اس کے جو تمہارے دین پر چلے۔ کہہ دیجیے: ہدایت تو اللہ ہی کی ہدایت ہے۔” وہ لوگ اعتماد اور ہدایت کو اپنے دائرے میں بند کرنا چاہتے تھے، اور جواب ان کی گفتگو کے بیچ میں آ کر اس دائرے کو توڑ دیتا ہے۔ “الْهُدٰى” کی نسبت “اللّٰهِ” کی طرف ہے: ہدایت کسی قوم کی جاگیر نہیں، کسی خاندان کی میراث نہیں، اور کسی گروہ کا ٹھپا نہیں۔ اہلِ ذوق کہتے ہیں کہ یہ جملہ ہر دور کے ہر گروہ کے لیے ہے، اور ہمارے لیے بھی: جو یہ سمجھے کہ ہدایت اس کی جیب میں ہے، وہ اسی لمحے ہدایت سے خالی ہونے لگتا ہے۔ شیخِ اکبر کے نزدیک ہدایت دراصل حق کا وہ نور ہے جو ہر طالب کے دل پر اس کی طلب کے مطابق اترتا ہے؛ اور جو اسے مانگے اسے روکا نہیں جاتا۔ یہاں اس سلسلے کا بنیادی اصول بھی جھلکتا ہے: ہدایت اللہ کی ہے، اس لیے کوئی بندہ کسی طالب کو اس سے محروم نہیں کر سکتا؛ محرومی صرف وہاں ہے جہاں بندہ خود منہ پھیر لے۔

آیت کا باقی حصہ دو طرح پڑھا گیا ہے: اَنْ يُّؤْتٰۤى اَحَدٌ مِّثْلَ مَاۤ اُوْتِيْتُمْ اَوْ يُحَآجُّوْكُمْ عِنْدَ رَبِّكُمْ ۗ قُلْ اِنَّ الْفَضْلَ بِيَدِ اللّٰهِ ۚ يُؤْتِيْهِ مَنْ يَّشَآءُ ۗ وَاللّٰهُ وَاسِعٌ عَلِيْمٌ۔ ایک قول میں یہ انہی کی بات کا تسلسل ہے: “اور یہ نہ مانو کہ کسی اور کو بھی ویسا ہی دیا جائے جیسا تمہیں دیا گیا، یا وہ تمہارے رب کے حضور تم پر حجت قائم کریں۔” دوسرے قول میں یہ اللہ کا جواب ہے کہ یہ سب تمہاری اس جلن کی وجہ سے ہے کہ کسی اور کو بھی وہ نعمت مل گئی۔ دونوں صورتوں میں انجام ایک ہے: “کہہ دیجیے: فضل اللہ کے ہاتھ میں ہے، وہ جسے چاہتا ہے دیتا ہے، اور اللہ وسعت والا، جاننے والا ہے۔” فضل وہ ہے جو استحقاق سے بڑھ کر دیا جائے، اس لیے اس پر حسد کرنا گویا دینے والے سے جھگڑنا ہے۔ شیخِ اکبر کے مکتب میں عطا کی دو منزلیں ہیں: فیضِ اقدس، جس سے ہر حقیقت کو اس کی استعداد ملی، اور فیضِ مقدس، جس سے ہر استعداد کو اس کا مانگا ہوا ملتا ہے۔ “مَنْ يَّشَآءُ” یہاں اندھا انتخاب نہیں، حکمت کا انتخاب ہے جو “عَلِيْمٌ” کے ساتھ جڑا ہے، یعنی وہ ہر برتن کی گنجائش جانتا ہے؛ اور “وَاسِعٌ” بتاتا ہے کہ ایک کو دینے سے دوسرے کا حصہ کم نہیں ہوتا۔ حسد کرنے والا عطا کو تنگ نہیں کرتا، اپنا ہی دل تنگ کرتا ہے۔

اور پھر رحمت کی خاص عطا کا ذکر: يَّخْتَصُّ بِرَحْمَتِهٖ مَنْ يَّشَآءُ ۗ وَاللّٰهُ ذُو الْفَضْلِ الْعَظِيْمِ۔ “وہ اپنی رحمت کے لیے جسے چاہتا ہے خاص کر لیتا ہے، اور اللہ بڑے فضل والا ہے۔” مفسرین کے نزدیک یہاں رحمت سے مراد نبوت اور وحی ہے، جو اس وقت بنی اسماعیل میں حضورِ اکرم صلی اللہ علیہ وسلم کو عطا ہوئی۔ شیخِ اکبر کے مکتب میں رحمت کی دو قسمیں بیان ہوتی ہیں: رحمتِ امتنان، جو بغیر کسی عمل کے محض احسان کے طور پر ملتی ہے، اور رحمتِ وجوب، جو حق تعالیٰ نے اپنے اوپر عمل کرنے والوں کے لیے لازم کر لی۔ نبوت سراسر رحمتِ امتنان ہے؛ کوئی اسے کما کر نہیں پاتا۔ مگر “يَخْتَصُّ” کا مطلب یہ نہیں کہ باقی سب رحمت سے باہر ہیں؛ عام رحمت ہر شے کو گھیرے ہوئے ہے اور رحمت غضب پر سبقت رکھتی ہے۔ اہلِ معرفت کہتے ہیں کہ خاص رحمت کی کچھ قسمیں نبیوں کے ساتھ ختم ہوئیں، مگر اس کا ایک دروازہ ہر طالب کے لیے کھلا ہے: طلب، تقویٰ اور اتباع۔ اور “ذُو الْفَضْلِ الْعَظِيْمِ” یاد دلاتا ہے کہ جس کا فضل عظیم ہو، اس کے دروازے پر مانگنے والا کبھی اس خوف سے نہ رکے کہ دوسروں کو پہلے مل چکا۔

اب وہ آیت آتی ہے جس میں قرآن کا انصاف اپنے کمال پر ہے، اور وہ تعریف سے شروع ہوتی ہے: وَمِنْ اَهْلِ الْكِتٰبِ مَنْ اِنْ تَاْمَنْهُ بِقِنْطَارٍ يُّؤَدِّهٖۤ اِلَيْكَ ۚ۔ “اور اہلِ کتاب میں ایسے بھی ہیں کہ اگر تم ان کے پاس مال کا ڈھیر امانت رکھو تو وہ تمہیں لوٹا دیں۔” قنطار مال کا بڑا ڈھیر ہے، اور اس کا لوٹانا دیانت کی انتہائی گواہی ہے۔ غور کیجیے: جس گروہ کی کچھ تدبیروں پر ابھی تنقید ہو رہی تھی، اسی کے دیانت داروں کی تعریف پہلے کی گئی۔ بعض مفسرین نے اس کی مثال میں حضرت عبداللہ بن سلام رضی اللہ عنہ کا ذکر کیا ہے۔ یہ قرآن کا ادب ہے، اور ہمارے لیے حکم بھی: کسی قوم کو ایک رنگ میں رنگ دینا انصاف نہیں؛ ہر فرد اپنے عمل سے پہچانا جائے۔ آگے اسی سورت میں اہلِ کتاب کے ایک نیک گروہ کی کھلی تعریف آئے گی جو راتوں کو اللہ کی آیتیں پڑھتے اور سجدے کرتے ہیں۔ اہلِ معرفت امانت کو اس سے بھی گہرا دیکھتے ہیں: سورۂ الاحزاب میں جس امانت کا ذکر ہے وہ انسان کا پورا وجود ہے، جو اسے مالک کی طرف سے سونپا گیا۔ شیخِ اکبر کے مکتب میں ڈھیر لوٹا دینے والا اس بڑی امانت کا اشارہ ہے: وہ عارف جو اپنا وجود، اپنی قوتیں اور اپنی صفات اصل مالک کو لوٹا دیتا ہے، اور اپنے پاس کچھ اپنا نہیں سمجھتا؛ یہی فنا کا ایک رخ ہے۔

اور تعریف کے بعد تنبیہ: وَمِنْهُمْ مَّنْ اِنْ تَاْمَنْهُ بِدِيْنَارٍ لَّا يُؤَدِّهٖۤ اِلَيْكَ اِلَّا مَا دُمْتَ عَلَيْهِ قَآئِمًا ۗ ذٰلِكَ بِاَنَّهُمْ قَالُوْا لَيْسَ عَلَيْنَا فِي الْاُمِّيّٖنَ سَبِيْلٌ ۚ وَيَقُوْلُوْنَ عَلَى اللّٰهِ الْكَذِبَ وَهُمْ يَعْلَمُوْنَ۔ “اور ان میں ایسے بھی ہیں کہ اگر ایک دینار بھی ان کے پاس امانت رکھو تو نہ لوٹائیں، مگر جب تک تم ان کے سر پر کھڑے رہو؛ یہ اس لیے کہ وہ کہتے ہیں: ان پڑھ لوگوں کے معاملے میں ہم پر کوئی گرفت نہیں؛ اور وہ جان بوجھ کر اللہ پر جھوٹ باندھتے ہیں۔” خرابی کی جڑ ایک دعویٰ ہے: اپنے سے باہر والوں کے ساتھ خیانت جائز ہے۔ یہ اخلاق کا دوہرا پیمانہ ہے، اور قرآن اسے “اللہ پر جھوٹ” کہتا ہے۔ یہ تنبیہ ہم پر بھی اسی طرح لاگو ہے: جو مسلمان کسی غیر مسلم کی امانت میں خیانت کو جائز سمجھے، وہ اسی دعوے میں شریک ہو گیا۔ امانت ہر ایک کی لوٹائی جاتی ہے۔ اور “مَا دُمْتَ عَلَيْهِ قَآئِمًا” میں ایک گہرا اشارہ ہے: جو صرف نگرانی میں دیانت دار ہو وہ مراقبہ کی پہلی سیڑھی پر بھی نہیں پہنچا۔ اہلِ معرفت کہتے ہیں کہ حقیقی نگران ہر وقت کھڑا ہے: اَفَمَنْ هُوَ قَآئِمٌ عَلٰى كُلِّ نَفْسٍ بِمَا كَسَبَتْ؛ اور القیوم کا یہ دھیان جس دل میں اتر آئے، اسے کسی انسانی چوکیدار کی ضرورت نہیں رہتی۔

پھر قرآن پیمانہ گروہ سے ہٹا کر عمل پر رکھ دیتا ہے: بَلٰى مَنْ اَوْفٰى بِعَهْدِهٖ وَاتَّقٰى فَاِنَّ اللّٰهَ يُحِبُّ الْمُتَّقِيْنَ۔ “کیوں نہیں! جو کوئی اپنا عہد پورا کرے اور تقویٰ اختیار کرے تو اللہ تقویٰ والوں سے محبت کرتا ہے۔” “بَلٰى” ان کے دعوے کا جواب ہے: گرفت ضرور ہے، اور نجات کسی نسل سے نہیں، وفا اور تقویٰ سے ہے۔ “مَنْ” عام ہے: جو بھی، جس قوم سے بھی۔ “بِعَهْدِهٖ” میں ضمیر کے دو پہلو ہیں: اپنا عہد بھی، اور اللہ کا عہد بھی؛ اور دونوں ایک دوسرے سے جدا نہیں۔ اہلِ ذوق یہاں ایک لطیف ہم آہنگی پر ٹھہرتے ہیں: آیت “بَلٰى” سے شروع ہوتی ہے، اور یہی وہ لفظ ہے جو روحوں نے ازل میں کہا تھا: اَلَسْتُ بِرَبِّكُمْ ۗ قَالُوْا بَلٰى۔ شیخِ اکبر کے مکتب میں ہر عہد کی جڑ وہی عہدِ الست ہے؛ بندہ جب کسی انسان سے کیا ہوا وعدہ نبھاتا ہے تو دراصل اپنے اس پہلے “بَلٰى” کی لاج رکھتا ہے۔ اور انجام محبت ہے: “يُحِبُّ الْمُتَّقِيْنَ”۔ وفا کرنے والا بدلے میں صرف اجر نہیں پاتا، محبوب بن جاتا ہے۔

اور اس کے مقابل وہ لوگ جو عہد بیچ دیتے ہیں: اِنَّ الَّذِيْنَ يَشْتَرُوْنَ بِعَهْدِ اللّٰهِ وَاَيْمَانِهِمْ ثَمَنًا قَلِيْلًا اُولٰۤئِكَ لَا خَلَاقَ لَهُمْ فِي الْاٰخِرَةِ وَلَا يُكَلِّمُهُمُ اللّٰهُ وَلَا يَنْظُرُ اِلَيْهِمْ يَوْمَ الْقِيٰمَةِ وَلَا يُزَكِّيْهِمْ وَلَهُمْ عَذَابٌ اَلِيْمٌ۔ “جو لوگ اللہ کے عہد اور اپنی قسموں کو تھوڑی قیمت پر بیچ دیتے ہیں، ان کا آخرت میں کوئی حصہ نہیں، اللہ قیامت کے دن نہ ان سے بات کرے گا، نہ ان کی طرف دیکھے گا، نہ انہیں پاک کرے گا، اور ان کے لیے دردناک عذاب ہے۔” صحیح بخاری کی روایت کا مفہوم ہے کہ یہ آیت ایک جائیداد کے جھگڑے کے موقع پر اتری، جب حضورِ اکرم صلی اللہ علیہ وسلم نے فرمایا کہ جو جھوٹی قسم کھا کر کسی کا مال ہتھیائے وہ اللہ سے اس حال میں ملے گا کہ اللہ اس پر ناراض ہوگا۔ یعنی یہ آیت کسی ایک قوم پر نہیں، ہر جھوٹی قسم کھانے والے پر ہے، خواہ وہ کوئی بھی ہو۔ اہلِ معرفت کہتے ہیں کہ یہاں سب سے بڑی سزا آگ سے پہلے بیان ہوئی: کلام سے محرومی اور نظرِ رحمت سے محرومی؛ یہی حجاب ہے، جس کے بارے میں سورۂ المطففین میں فرمایا: كَلَّاۤ اِنَّهُمْ عَنْ رَّبِّهِمْ يَوْمَئِذٍ لَّمَحْجُوْبُوْنَ۔ اور یہ سزا ان کے اپنے عمل کا عکس ہے: انہوں نے اللہ کے عہد کو بیچا، یعنی اس رشتے کو جو کلام سے بندھا تھا، تو کلام ہی ان سے روک لیا گیا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔

اب ایک اور گروہ کا ذکر، اور پھر “ان میں سے” کی قید کے ساتھ: وَاِنَّ مِنْهُمْ لَفَرِيْقًا يَّلْوٗنَ اَلْسِنَتَهُمْ بِالْكِتٰبِ لِتَحْسَبُوْهُ مِنَ الْكِتٰبِ وَمَا هُوَ مِنَ الْكِتٰبِ ۚ وَيَقُوْلُوْنَ هُوَ مِنْ عِنْدِ اللّٰهِ وَمَا هُوَ مِنْ عِنْدِ اللّٰهِ ۚ وَيَقُوْلُوْنَ عَلَى اللّٰهِ الْكَذِبَ وَهُمْ يَعْلَمُوْنَ۔ “اور ان میں ایک فریق ایسا ہے جو کتاب پڑھتے ہوئے اپنی زبانیں مروڑتا ہے تاکہ تم اسے کتاب کا حصہ سمجھو، حالانکہ وہ کتاب میں سے نہیں؛ اور وہ کہتے ہیں: یہ اللہ کی طرف سے ہے، حالانکہ وہ اللہ کی طرف سے نہیں؛ اور وہ جان بوجھ کر اللہ پر جھوٹ باندھتے ہیں۔” یہ آیت ہر اس شخص کے سامنے کھڑی ہے جو کسی بھی آسمانی کتاب کی تشریح کرتا ہے، اور اس صفحے کے لکھنے والے کے سامنے سب سے پہلے۔ اسی لیے اس سلسلے کا اصول ہے کہ اشاری معنی ظاہر کی نفی نہیں، اس کی گہرائی ہے؛ اور کوئی ذوقی نکتہ “هُوَ مِنْ عِنْدِ اللّٰهِ” کا دعویٰ نہیں کر سکتا۔ شیخِ اکبر ظاہرِ شریعت کی پابندی پر بہت زور دیتے ہیں، اور ان کے مکتب میں کشف کی کسوٹی ہمیشہ کتاب و سنت ہے؛ جو کشف ظاہر سے ٹکرائے وہ کشف نہیں، وہم ہے۔ اہلِ ادب کہتے ہیں کہ اپنے ذوق کو وحی کا لباس پہنانا وہی تلبیس ہے جس سے پچھلے رکوع کے آخر میں منع کیا گیا۔

اب وہ آیت جو ہر نبی کے دامن کو پاک دکھاتی ہے: مَا كَانَ لِبَشَرٍ اَنْ يُّؤْتِيَهُ اللّٰهُ الْكِتٰبَ وَالْحُكْمَ وَالنُّبُوَّةَ ثُمَّ يَقُوْلَ لِلنَّاسِ كُوْنُوْا عِبَادًا لِّيْ مِنْ دُوْنِ اللّٰهِ۔ “کسی انسان کا یہ کام نہیں کہ اللہ اسے کتاب، حکم اور نبوت دے، پھر وہ لوگوں سے کہے کہ اللہ کو چھوڑ کر میرے بندے بن جاؤ۔” سیرت کی روایات کا مفہوم ہے کہ کچھ لوگوں نے حضورِ اکرم صلی اللہ علیہ وسلم سے پوچھا: کیا آپ چاہتے ہیں کہ ہم آپ کی عبادت کریں؟ تو یہ آیت اتری۔ مگر اس کا دائرہ ہر نبی کو گھیرتا ہے، اور حضرت عیسیٰ علیہ السلام کی پاکیزگی کی بھی گواہی ہے؛ وہ خود قیامت کے دن عرض کریں گے: مَا قُلْتُ لَهُمْ اِلَّا مَاۤ اَمَرْتَنِيْ بِهٖۤ اَنِ اعْبُدُوا اللّٰهَ رَبِّيْ وَرَبَّكُمْ۔ آیت کی ترتیب میں ایک راز ہے: جسے جتنا زیادہ دیا گیا، کتاب، پھر حکم، پھر نبوت، وہ اتنا ہی زیادہ بندہ ہوتا ہے۔ شیخِ اکبر کے مکتب میں عبدیت سب سے اونچا مقام ہے، کیونکہ اس میں بندہ اپنی حقیقت پر قائم رہتا ہے اور ربوبیت کا کوئی دعویٰ نہیں کرتا۔ سورۂ بنی اسرائیل کے صفحے پر گزرا کہ معراج کی رات کا ذکر بھی “عبد” کے لفظ سے ہوا؛ سب سے بلند سفر پر بھی حضورِ اکرم صلی اللہ علیہ وسلم کا نام بندگی تھا۔

نبی لوگوں کو اپنی طرف نہیں بلاتا، تو پھر کس طرف بلاتا ہے؟ وَلٰكِنْ كُوْنُوْا رَبّٰنِيّٖنَ بِمَا كُنْتُمْ تُعَلِّمُوْنَ الْكِتٰبَ وَبِمَا كُنْتُمْ تَدْرُسُوْنَ۔ “بلکہ وہ کہے گا: رب والے بن جاؤ، اس لیے کہ تم کتاب سکھاتے ہو اور اس لیے کہ تم اسے پڑھتے ہو۔” صحیح بخاری میں حضرت عبداللہ بن عباس رضی اللہ عنہما سے “رَبّٰنِيّٖنَ” کی تفسیر کا مفہوم یہ ہے کہ ربانی وہ ہے جو حکمت و بردباری والا اور دین کی سمجھ رکھنے والا ہو، اور یہ بھی کہا گیا کہ وہ لوگوں کی تربیت بڑے علم سے پہلے چھوٹے علم سے کرتا ہے۔ قرأت میں “تُعَلِّمُوْنَ” یعنی سکھاتے ہو بھی ہے اور بعض قراء کے ہاں “تَعْلَمُوْنَ” یعنی جانتے ہو بھی؛ دونوں میں پیغام ایک ہے کہ علم کا حق اسے حال بنانا ہے۔ ربانی کی نسبت رب کی طرف ہے، اور رب کا مطلب ہے پالنے والا، درجہ بدرجہ کمال تک پہنچانے والا۔ اہلِ معرفت کہتے ہیں کہ ربانی وہ ہے جس میں اسمِ رب کی صفتِ تربیت ظاہر ہو؛ یہ تخلق باسماء اللہ ہے، دعویٰ نہیں۔ شیخِ اکبر کے نزدیک علم کی معراج یہ ہے کہ “تَدْرُسُوْنَ” بدل کر “كُوْنُوْا” بن جائے: پڑھنا ہونا بن جائے، اور معلومات وجود میں ڈھل جائیں۔ اسی لیے آیت “سیکھو” نہیں کہتی، “بن جاؤ” کہتی ہے۔

رکوع کا خاتمہ اسی بات کو آخری حد تک پہنچاتا ہے: وَلَا يَاْمُرَكُمْ اَنْ تَتَّخِذُوا الْمَلٰٓئِكَةَ وَالنَّبِيّٖنَ اَرْبَابًا ۗ اَيَاْمُرُكُمْ بِالْكُفْرِ بَعْدَ اِذْ اَنْتُمْ مُّسْلِمُوْنَ۔ “اور نہ وہ تمہیں یہ حکم دے گا کہ فرشتوں اور نبیوں کو رب بنا لو؛ کیا وہ تمہیں کفر کا حکم دے گا، اس کے بعد کہ تم مسلمان ہو؟” فرشتے اور نبی مخلوقات میں سب سے روشن ہیں؛ اگر وہ بھی رب نہیں تو کوئی اور کیسے ہو سکتا ہے؟ اس مکتب میں یہ نکتہ بہت نازک ہے: ہر موجود حق کے کسی اسم کا مظہر ہے، اور نبی سب سے کامل مظہر ہیں؛ مگر مظہر کی تعظیم اور مظہر کی عبادت میں زمین آسمان کا فرق ہے۔ شیخِ اکبر کے نزدیک حق اپنی تجلی میں ظاہر ہے مگر کسی چیز میں حلول نہیں کرتا؛ جو آئینے کو سورج سمجھ بیٹھے اس نے آئینے کا بھی حق ادا نہیں کیا اور سورج کا بھی نہیں۔ اور غور کیجیے: رکوعوں کا یہ جوڑا “مسلمان” کے لفظ پر کھلا اور اسی پر بند ہوا؛ پچھلے رکوع میں تھا “اشْهَدُوْا بِاَنَّا مُسْلِمُوْنَ”، بیچ میں حضرت ابراہیم علیہ السلام “حَنِيْفًا مُّسْلِمًا”، اور یہاں “بَعْدَ اِذْ اَنْتُمْ مُّسْلِمُوْنَ”۔ کلمۂ سواء کی پوری بحث اسی ایک لفظ کی شرح ہے: سپرد کر دینا، اور سپرد کر دینے کے بعد کسی اور کو رب نہ بنانا۔
