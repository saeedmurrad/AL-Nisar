پہلے رکوع نے انصار کا وہ دل دکھایا جو تنگی میں بھی بھائی کو اپنے اوپر ترجیح دیتا ہے؛ یہ رکوع اسی آئینے کو دوسری طرف پھیرتا ہے اور وہ دل دکھاتا ہے جو زبان سے بھائی چارہ بانٹتا ہے اور گھڑی آنے پر خالی نکلتا ہے۔ اَلَمْ تَرَ اِلَى الَّذِيْنَ نَافَقُوْا يَقُوْلُوْنَ لِاِخْوَانِهِمُ الَّذِيْنَ كَفَرُوْا مِنْ اَهْلِ الْكِتٰبِ لَئِنْ اُخْرِجْتُمْ لَنَخْرُجَنَّ مَعَكُمْ وَلَا نُطِيْعُ فِيْكُمْ اَحَدًا اَبَدًا ۙ وَّاِنْ قُوْتِلْتُمْ لَنَنْصُرَنَّكُمْ ۭ وَاللّٰهُ يَشْهَدُ اِنَّهُمْ لَكٰذِبُوْنَ۔ کیا تم نے ان لوگوں کو نہیں دیکھا جنہوں نے نفاق کی راہ لی؟ وہ اہلِ کتاب میں سے اپنے کافر بھائیوں سے کہتے ہیں: اگر تمہیں نکالا گیا تو ہم بھی ضرور تمہارے ساتھ نکلیں گے، تمہارے معاملے میں کبھی کسی کا کہا نہ مانیں گے، اور اگر تم سے جنگ کی گئی تو ہم ضرور تمہاری مدد کریں گے؛ اور اللہ گواہی دیتا ہے کہ یہ بالکل جھوٹے ہیں۔ ظاہری تاریخ اتنی ہے کہ حضورِ اکرم صلی اللہ علیہ وسلم کے زمانے میں مدینہ کے منافقین نے بنو نضیر کو پیغام بھیجا کہ ڈٹے رہو، ہم تمہارے ساتھ ہیں؛ اور جب محاصرہ ہوا تو ان میں سے کوئی نہ آیا۔ یہ ایک خاص واقعے کی خبر ہے، آج کی کسی قوم پر چسپاں کرنے کا حکم نہیں؛ اور اس پر جو اشاری معنی کھلتا ہے وہ ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ معرفت لفظِ اخوان پر ٹھہرتے ہیں: بھائی چارہ مناسبت سے بنتا ہے، اور حدیث کا مفہوم ہے کہ روحیں صف بستہ لشکر ہیں، جن میں شناسائی ہوئی وہ مانوس ہو گئیں۔ ان دونوں گروہوں کو ایک ہی چیز نے بھائی بنایا تھا: حق سے اعراض۔ پھر لفظِ ابد کو دیکھیے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک مخلوق ہر آن نئی پیدا ہوتی ہے اور اسے اگلے سانس پر بھی اختیار نہیں؛ تو جو اپنی طاقت پر ہمیشگی کا وعدہ کرے وہ اس وجود کا دعویٰ کرتا ہے جو اس کا اپنا نہیں۔ اسی لیے آیت کا آخری کلمہ حق کی گواہی ہے: وہ الشہید ہے، باطن پر بھی حاضر، اور نفاق ظاہر و باطن کا وہی شگاف ہے جو اس گواہ سے چھپ نہیں سکتا۔

قرآن ان کے وعدوں کو ایک ایک کر کے کاٹتا ہے: لَئِنْ اُخْرِجُوْا لَا يَخْرُجُوْنَ مَعَهُمْ ۚ وَلَئِنْ قُوْتِلُوْا لَا يَنْصُرُوْنَهُمْ ۚ وَلَئِنْ نَّصَرُوْهُمْ لَيُوَلُّنَّ الْاَدْبَارَ ثُمَّ لَا يُنْصَرُوْنَ۔ اگر وہ نکالے گئے تو یہ ان کے ساتھ نہ نکلیں گے، اگر ان سے جنگ ہوئی تو یہ ان کی مدد نہ کریں گے، اور اگر مدد کو آئے بھی تو پیٹھ پھیر کر بھاگیں گے، پھر انہیں کہیں سے مدد نہ ملے گی۔ اہلِ تاویل کہتے ہیں کہ پیٹھ پھیرنا ان کے حال کی صورت ہے، کوئی باہر سے آئی ہوئی سزا نہیں: جس نے پہلے حق سے رخ پھیرا وہ کسی معرکے کا سامنا نہیں کر سکتا، اس کا اعراض ہی میدان میں پیٹھ بن کر ظاہر ہوتا ہے۔ اس مکتب میں ہر مدد کی جڑ حق کے اسم الناصر میں ہے؛ جو وعدہ اس جڑ سے کٹا ہوا ہو وہ لفظ ہے، وجود نہیں۔ پھر اس کمزوری کی جڑ بتائی: لَاَنْتُمْ اَشَدُّ رَهْبَةً فِيْ صُدُوْرِهِمْ مِّنَ اللّٰهِ ۭ ذٰلِكَ بِاَنَّهُمْ قَوْمٌ لَّا يَفْقَهُوْنَ۔ ان کے سینوں میں تمہاری ہیبت اللہ سے بڑھ کر ہے؛ یہ اس لیے کہ وہ سمجھ نہیں رکھتے۔ یہاں خوف کی جگہ صدر کہی گئی، قلب نہیں۔ اہلِ ذوق کہتے ہیں کہ صدر دل کا صحن ہے جہاں مخلوق کی آوازیں پہنچتی ہیں، اور قلب اندر کا وہ حجرہ ہے جو حق کے لیے ہے؛ ان کا خوف صحن ہی میں بسا رہا، اندر تک بات پہنچی ہی نہیں۔ شیخِ اکبر کے نزدیک ہر فعل کا فاعلِ حقیقی حق ہے اور مخلوق اس کے افعال کے مظاہر ہیں؛ جو مظہر کو دیکھے اور اس میں ظاہر ہونے والے کو نہ دیکھے وہ مظہر سے ڈرتا ہے، اور جو توحیدِ افعالی تک پہنچ جائے وہ کسی سے نہیں ڈرتا مگر اسی سے۔ یہی وہ سمجھ ہے جس کی نفی آیت کرتی ہے: فقہ یعنی ظاہر سے گزر کر اس کے مالک تک پہنچ جانا۔

اب ان کے لشکر کا نقشہ: لَا يُقَاتِلُوْنَكُمْ جَمِيْعًا اِلَّا فِيْ قُرًى مُّحَصَّنَةٍ اَوْ مِنْ وَّرَآءِ جُدُرٍ ۭ بَاْسُهُمْ بَيْنَهُمْ شَدِيْدٌ ۭ۔ وہ اکٹھے ہو کر تم سے کھلے میدان میں نہیں لڑیں گے، مگر قلعہ بند بستیوں میں یا دیواروں کی اوٹ سے؛ ان کی آپس کی لڑائی بڑی سخت ہے۔ ظاہر میں یہ اس بزدلی کا بیان ہے جو قلعوں اور فصیلوں کے پیچھے چھپتی ہے، جیسے پہلے رکوع میں بنو نضیر کو اپنے قلعوں پر بھروسا تھا۔ اہلِ معرفت کے نزدیک یہی نفس کی جنگی چال ہے: نفس کبھی کھلے میدان میں حق کا سامنا نہیں کرتا، وہ تاویلوں کی بستیوں اور عادتوں کی دیواروں کے پیچھے سے لڑتا ہے؛ ہر حجاب ایک فصیل ہے جس کی اوٹ سے وہ تیر چلاتا ہے۔ اور جو قوت حق کے سامنے نہیں نکلتی وہ اندر خرچ ہوتی ہے: ان کا زور آپس ہی میں سخت ہے۔ جس دل کا کوئی ایک مرکز نہ ہو اس میں خواہش خواہش سے، حرص خوف سے، اور غصہ شرم سے لڑتا رہتا ہے۔ اس مکتب میں انسان اسمِ جامع کا مظہر ہے، اس میں بہت سے اسماء کے آثار جمع ہیں؛ جب دل حق کی طرف رخ رکھتا ہے تو یہ آثار ایک دوسرے کو سنوارتے ہیں، جلال جمال کی حد بنتا ہے اور جمال جلال کی نرمی؛ اور جب نفس خود کو مرکز بنا لے تو یہی قوتیں آپس میں ٹکراتی ہیں، اور باہر کی شکست سے پہلے اندر کی خانہ جنگی شروع ہو جاتی ہے۔

اور پھر وہ جملہ جو اس رکوع کا دل ہے: تَحْسَبُهُمْ جَمِيْعًا وَّقُلُوْبُهُمْ شَتّٰى ۭ ذٰلِكَ بِاَنَّهُمْ قَوْمٌ لَّا يَعْقِلُوْنَ۔ تم انہیں اکٹھا سمجھتے ہو حالانکہ ان کے دل جدا جدا ہیں؛ یہ اس لیے کہ وہ عقل سے کام نہیں لیتے۔ باہر سے صفیں جڑی ہوئی، نعرہ ایک، جھنڈا ایک؛ اور اندر ہر دل کا قبلہ الگ۔ لفظِ شتیٰ بکھری ہوئی چیزوں کے لیے ہے، جیسے ٹوٹی ہوئی تسبیح کے دانے۔ شیخِ اکبر کے مکتب میں قلب کا نام تقلب سے ہے، یعنی پلٹنا؛ اور حکمتِ قلبیہ میں، جو حضرت شعیب علیہ السلام سے منسوب ہے، یہ مضمون ملتا ہے جس کا مفہوم یہ ہے کہ عارف کا دل حق کی ہر نئی تجلی کی صورت قبول کرتا ہے اور اسی وسعت میں اس کا کمال ہے۔ تو پلٹنا دو طرح کا ہوا: ایک تقلب، جو حق کے ساتھ ہے؛ صورتیں بہت، رخ ایک۔ دوسرا تشتت، جو اغیار کے درمیان ہے؛ ہر خواہش ایک آقا، ہر آقا ایک قبلہ۔ پہلا دل بہت سی صورتوں میں بھی ایک رہتا ہے، دوسرا ایک ہی صورت میں بھی ٹکڑے ٹکڑے ہے۔ سورۂ الزمر کی مثال یہیں کھلتی ہے: ضَرَبَ اللّٰهُ مَثَلًا رَّجُلًا فِيْهِ شُرَكَآءُ مُتَشٰكِسُوْنَ وَرَجُلًا سَلَمًا لِّرَجُلٍ، ایک غلام جس میں کئی جھگڑالو ساجھی ہیں اور ایک جو پورا کا پورا ایک ہی آقا کا ہے۔ اور سورۂ الاحزاب نے اصل بتا دی: مَا جَعَلَ اللّٰهُ لِرَجُلٍ مِّنْ قَلْبَيْنِ فِيْ جَوْفِهٖ۔ دل ایک ہی رکھا گیا ہے، اس لیے اس میں دو برابر کی سلطنتیں سما نہیں سکتیں؛ جو اسے بانٹتا ہے وہ اسے کھو دیتا ہے۔

آیت نے اس بکھراؤ کی وجہ ان کی اپنی محرومی بتائی: وہ عقل سے کام نہیں لیتے۔ اہلِ لغت کہتے ہیں کہ عقل کی اصل عِقال ہے، وہ رسی جس سے اونٹ کا گھٹنا باندھا جاتا ہے؛ اور اہلِ ذوق کہتے ہیں کہ حقیقی عقل وہی ہے جو دل کے اونٹ کو ایک کھونٹے سے باندھ دے، ورنہ وہ ہر چراگاہ کی طرف دوڑتا ہے۔ یہاں کسی جبر کا ذکر نہیں؛ بکھراؤ ان کا اپنا ہے، اور حق نے اسی رخ کو وجود دیا جو انہوں نے خود چنا۔ اہلِ تصوف میں مشہور قول ہے کہ تفرقہ بغیر جمع کے تعطیل ہے، جمع بغیر تفرقہ کے زندقہ، اور جمع کے ساتھ تفرقہ توحید ہے۔ منافق کا حال اس کے الٹ ہے: باہر کا جمع، اندر کا تفرقہ؛ صورت کی یکجائی اور حقیقت کا انتشار۔ عارف کا حال اس کے برعکس ہے: باہر وہ ہزار کاموں میں بٹا ہوا دکھائی دے، اندر اس کا دل ایک ہی حضوری میں جمع ہے۔ اسی لیے سالک کے لیے یہ آیت آئینہ ہے، دوسروں پر فتویٰ نہیں: نماز کی صف میں کندھا ملا ہو اور دل بازار میں ہو تو اس آیت کا سایہ اپنے اوپر بھی پڑ سکتا ہے۔ علاج ذکر اور مراقبہ ہے، جو بکھرے ہوئے خیالات کو ایک نام پر جمع کرتا ہے، یہاں تک کہ جمعیتِ خاطر نصیب ہو؛ اور امت کی حقیقی یکجائی بھی بھیڑ سے نہیں، ایک قبلے پر جمع دلوں سے بنتی ہے۔

پھر ایک قریبی مثال: كَمَثَلِ الَّذِيْنَ مِنْ قَبْلِهِمْ قَرِيْبًا ذَاقُوْا وَبَالَ اَمْرِهِمْ ۚ وَلَهُمْ عَذَابٌ اَلِيْمٌ۔ ان کا حال ان لوگوں جیسا ہے جو ان سے کچھ ہی پہلے گزرے؛ انہوں نے اپنے کیے کا وبال چکھ لیا، اور ان کے لیے دردناک عذاب ہے۔ مفسرین کے نزدیک یہ بنو قینقاع کی طرف اشارہ ہے جو اسی طرح عہد توڑ کر مدینہ سے نکلے، اور بعض کے نزدیک بدر کے مشرکین کی طرف؛ دونوں صورتوں میں سبق یہ ہے کہ عبرت دور کی کہانیوں میں نہیں، کل کے واقعے میں موجود تھی، اور یہی وہ عبور ہے جس کی دعوت سورت کے آغاز میں دی گئی: ظاہر سے گزر کر سبق تک پہنچنا۔ اہلِ معرفت لفظِ ذاقوا پر ٹھہرتے ہیں: نتیجہ سنا نہیں جاتا، چکھا جاتا ہے؛ جو خبر سے نہ مانے اسے ذوق سے ماننا پڑتا ہے۔ اور وبال کو ان کے اپنے امر کی طرف منسوب کیا گیا، کسی اجنبی چیز کی طرف نہیں۔ شیخِ اکبر کے نزدیک سرِّ قدر یہی ہے کہ حکم محکوم کے مطابق ہوتا ہے: حق کسی پر وہ نہیں ڈالتا جو اس کی اپنی حقیقت اور اس کے اپنے رخ میں نہ ہو۔ اس مکتب میں جزا عمل سے باہر کی کوئی چیز نہیں، عمل ہی کی حقیقت ہے جو اپنی صورت میں ظاہر ہو جاتی ہے؛ اس لیے بندہ جو بوتا ہے وہی چکھتا ہے، اور اس چکھنے میں حق کا عدل بھی ہے اور تنبیہ کی رحمت بھی۔

اب وہ آیت آتی ہے جو اس رکوع کو، بلکہ ہدایت و گمراہی کے سارے مسئلے کو کھول دیتی ہے؛ اسے اصول سمجھیے: كَمَثَلِ الشَّيْطٰنِ اِذْ قَالَ لِلْاِنْسَانِ اكْفُرْ ۚ۔ منافقین کی مثال شیطان جیسی ہے، جب وہ انسان سے کہتا ہے: کفر کر۔ شیطان کا سارا سرمایہ ایک لفظ ہے، ایک حکم نما دعوت؛ اس کے ہاتھ میں نہ دل کی کنجی ہے نہ ارادے کی باگ۔ یہی بات سورۂ ابراہیم میں خود اس کی زبان سے ادا ہوئی: وَمَا كَانَ لِيَ عَلَيْكُمْ مِّنْ سُلْطٰنٍ اِلَّآ اَنْ دَعَوْتُكُمْ فَاسْتَجَبْتُمْ لِيْ ۚ فَلَا تَلُوْمُوْنِيْ وَلُوْمُوْٓا اَنْفُسَكُمْ؛ مجھے تم پر کوئی زور نہ تھا، سوائے اس کے کہ میں نے تمہیں بلایا اور تم نے میری بات مان لی، تو مجھے ملامت نہ کرو، اپنے آپ کو ملامت کرو۔ اصول یہ ہے: نہ شیطان کسی کو زبردستی گمراہ کر سکتا ہے، نہ حق تعالیٰ کسی کو زبردستی گمراہ کرتا ہے، اور جو ہدایت کا طالب ہو اسے وہ کبھی گمراہ نہیں کرتا۔ جو مانگتا ہے اسے ہدایت ملتی ہے؛ جو منہ موڑتا ہے، اس کا منہ موڑنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر ہے: حکم محکوم کے مطابق۔ اور اختیار کی عزت بھی اسی اصول کا حصہ ہے کہ وہ چاہتا تو سب کو مجبور کر کے سیدھی راہ پر کر دیتا، مگر اس نے انسان کو مجبور نہیں، مخاطب بنایا۔ اسی لیے آیت میں شیطان کا فعل صرف کہنا ہے، اور کفر کرنے والا خود انسان۔

دعوت کا انجام بھی اسی آیت میں ہے: فَلَمَّا كَفَرَ قَالَ اِنِّيْ بَرِيْٓءٌ مِّنْكَ اِنِّيْٓ اَخَافُ اللّٰهَ رَبَّ الْعٰلَمِيْنَ۔ پھر جب وہ کفر کر بیٹھا تو شیطان نے کہا: میں تجھ سے بری ہوں، میں تو اللہ رب العالمین سے ڈرتا ہوں۔ یہاں رکوع کی پہلی آیت کا وعدہ اور اس آیت کی بیزاری ایک ہی تصویر کے دو رخ بن جاتے ہیں: منافقین نے کہا تھا ہم تمہارے ساتھ نکلیں گے، اور شیطان کہتا ہے میں تجھ سے بری ہوں۔ اہلِ معرفت کہتے ہیں کہ غیرِ حق کی ہر دعوت کا آخری کلمہ بیزاری ہے؛ ساتھ صرف وہی نبھاتا ہے جو بلاتا بھی ہے اور جس کی معیت کبھی ختم نہیں ہوتی۔ پھر شیطان کا اقرار دیکھیے: وہ رب العالمین کو جانتا ہے اور اس سے ڈرنے کا نام بھی لیتا ہے۔ اس مکتب میں یہ اس علم کی مثال ہے جو قدم نہ بنا: معرفت جو سپردگی تک نہ پہنچے وہ اپنے صاحب کے خلاف حجت بن جاتی ہے، اور ڈر جو ادب میں نہ ڈھلے وہ عارفوں کی خشیت نہیں، انجام کا خوف ہے۔ اور خطاب انسان سے ہے؛ اہلِ معرفت کے نزدیک انسان اسمِ جامع کا مظہر اور برزخ ہے، اس میں بلندی کی استعداد بھی ہے اور پستی کی بھی، اسی لیے دعوت اسی تک پہنچتی ہے اور فیصلہ بھی اسی کا ہوتا ہے۔ یہی اس کی آزمائش ہے اور یہی اس کی عزت۔

اور رکوع کا خاتمہ: فَكَانَ عَاقِبَتَهُمَآ اَنَّهُمَا فِي النَّارِ خَالِدَيْنِ فِيْهَا ۭ وَذٰلِكَ جَزٰٓؤُا الظّٰلِمِيْنَ۔ تو دونوں کا انجام یہ ہوا کہ دونوں آگ میں ہیں، اس میں ہمیشہ رہنے والے؛ اور یہی ظالموں کی جزا ہے۔ تثنیہ کا صیغہ قابلِ غور ہے: بلانے والا اور ماننے والا ایک ہی انجام میں ہیں۔ بیزاری کے اعلان نے بلانے والے کو نہیں بچایا، اور بلانے والے پر الزام ڈالنے سے ماننے والا بری نہیں ہوا۔ ظلم کے معنی ہیں کسی چیز کو اس کی جگہ کے سوا کہیں اور رکھ دینا؛ اہلِ ذوق کہتے ہیں کہ سب سے بڑا ظلم دل کا رخ اس کے رب کے سوا کسی اور طرف رکھ دینا ہے، کیونکہ دل اسی کے لیے بنا تھا۔ اور اس مکتب میں جزا اسی ظلم کی حقیقت کا ظہور ہے: جنہوں نے حق سے جدائی چنی، جدائی ہی ان کا ٹھکانا بن گئی۔ یوں یہ رکوع دو تصویریں چھوڑ جاتا ہے: منافق کا بکھرا ہوا دل، اور شیطان کی دعوت جس میں کوئی زور نہیں۔ دونوں کا علاج ایک ہے: دل کا ایک قبلہ۔ اور جب تک سانس باقی ہے دروازہ کھلا ہے؛ حدیث کا مفہوم ہے کہ حق کی رحمت اس کے غضب پر سبقت رکھتی ہے، اور جو پلٹ آئے وہ اسی لمحے بکھراؤ سے جمع کی طرف قدم رکھ دیتا ہے۔ اگلا رکوع اسی پلٹنے کی دعوت سے کھلتا ہے: محاسبہ، اور یہ کہ ہر جان دیکھے کہ اس نے کل کے لیے کیا آگے بھیجا۔
