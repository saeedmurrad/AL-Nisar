قَدْ مَكَرَ الَّذِيْنَ مِنْ قَبْلِهِمْ فَاَتَى اللّٰهُ بُنْيَانَهُمْ مِّنَ الْقَوَاعِدِ۔ ان سے پہلے والوں نے بھی چالیں چلیں، تو اللہ نے ان کی عمارت کو بنیادوں سے آ لیا۔ فَخَرَّ عَلَيْهِمُ السَّقْفُ مِنْ فَوْقِهِمْ وَاَتٰىهُمُ الْعَذَابُ مِنْ حَيْثُ لَا يَشْعُرُوْنَ۔ پھر چھت اوپر سے ان پر آ گری اور عذاب وہاں سے آیا جہاں کا انہیں گمان بھی نہ تھا۔ ظاہر میں یہ پچھلی قوموں کی تدبیروں کا انجام ہے، اور بعض مفسرین نے اسے اس بلند عمارت سے جوڑا ہے جو حضرت ابراہیم علیہ السلام کے زمانے کے سرکش بادشاہ نے بنوائی تھی؛ اشاری معنی اس ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ اہلِ تاویل کہتے ہیں کہ ہر متکبر کے اندر بھی ایک عمارت ہے، نفس کی عمارت، جس کی بنیاد "میں" پر ہے اور جس کی چھت اپنی تدبیر پر۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک بندے کی ہستی عاریت ہے، اور جو عاریت پر ملکیت کی عمارت کھڑی کرے اس کی بنیاد ہی کھوکھلی ہے؛ اس لیے حق اسے "قواعد" سے آ لیتا ہے، اور چھت کا گرنا بنیاد کے ہٹنے کا نتیجہ ہے۔ سورۂ ابراہیم کے شجرۂ خبیثہ کی طرح اس عمارت کو بھی قرار نہیں۔ ثُمَّ يَوْمَ الْقِيٰمَةِ يُخْزِيْهِمْ وَيَقُوْلُ اَيْنَ شُرَكَآءِيَ الَّذِيْنَ كُنْتُمْ تُشَآقُّوْنَ فِيْهِمْ۔ پھر قیامت کے دن وہ انہیں رسوا کرے گا اور فرمائے گا: کہاں ہیں میرے وہ شریک جن کی خاطر تم جھگڑا کرتے تھے؟ اس مکتب میں شریک کوئی ہستی نہ تھے، وہم کی صورتیں تھے جنہیں بندے نے حق کے مقابل کھڑا کیا۔ قَالَ الَّذِيْنَ اُوْتُوا الْعِلْمَ اِنَّ الْخِزْيَ الْيَوْمَ وَالسُّوْۤءَ عَلَى الْكٰفِرِيْنَ۔ جنہیں علم دیا گیا تھا وہ کہیں گے: آج رسوائی اور برائی کافروں پر ہے۔ اہلِ معرفت کہتے ہیں کہ یہاں بولنے والے وہ ہیں جنہیں علم "دیا گیا"، کمایا نہیں گیا؛ علمِ لدنی عطا ہی ہوتا ہے، اور اس دن وہی گواہی دیتا ہے۔

الَّذِيْنَ تَتَوَفّٰىهُمُ الْمَلٰۤئِكَةُ ظَالِمِيْۤ اَنْفُسِهِمْ۔ وہ جن کی روحیں فرشتے اس حال میں قبض کرتے ہیں کہ وہ اپنی جانوں پر ظلم کر رہے ہوتے ہیں۔ ظلم کا مفعول "انفسہم" ہے: انہوں نے حق کا کچھ نہ بگاڑا، اپنی ہی جان کا حق مارا۔ اہلِ معرفت کے نزدیک ہر جان کی ایک استعداد ہے جو اس کی عینِ ثابتہ میں رکھی گئی، اور اس استعداد کو اس کے کمال تک نہ پہنچنے دینا ہی اپنے نفس پر ظلم ہے۔ فَاَلْقَوُا السَّلَمَ مَا كُنَّا نَعْمَلُ مِنْ سُوْۤءٍ۔ تب وہ فرماں برداری کا اظہار کرتے ہیں کہ ہم تو کوئی برا کام نہیں کرتے تھے۔ جو سر زندگی بھر نہ جھکا وہ موت کی گھڑی جھک جاتا ہے، مگر اس جھکنے میں بھی انکار ملا ہوا ہے؛ کبر آخری سانس تک اپنی صفائی دیتا ہے۔ بَلٰۤى اِنَّ اللّٰهَ عَلِيْمٌۢ بِمَا كُنْتُمْ تَعْمَلُوْنَ۔ کیوں نہیں، بے شک اللہ خوب جانتا ہے جو تم کرتے تھے۔ شیخِ اکبر کے نزدیک علمِ الٰہی تابعِ معلوم ہے: حق نے بندے کو ویسا ہی جانا جیسا وہ اپنی حقیقت میں تھا، اور اسی علم کے مطابق حکم ہوا؛ یہی سرِّ قدر ہے کہ حکم محکوم کے مطابق ہے۔ فَادْخُلُوْۤا اَبْوَابَ جَهَنَّمَ خٰلِدِيْنَ فِيْهَا۔ پس جہنم کے دروازوں میں داخل ہو جاؤ، اسی میں ہمیشہ رہنے کو۔ فَلَبِئْسَ مَثْوَى الْمُتَكَبِّرِيْنَ۔ سو بہت برا ٹھکانا ہے تکبر کرنے والوں کا۔ اہلِ ذوق کہتے ہیں کہ پچھلے رکوع کا "مستکبرون" یہاں "متکبرین" کے انجام تک پہنچا: جس نے دنیا میں اپنے دل کو منکر اور تنگ رکھا، اس کا ٹھکانا بھی اسی تنگی کی صورت میں ظاہر ہوا۔

وَقِيْلَ لِلَّذِيْنَ اتَّقَوْا مَاذَآ اَنْزَلَ رَبُّكُمْ ۭ قَالُوْا خَيْرًا۔ اور پرہیزگاروں سے پوچھا گیا کہ تمہارے رب نے کیا اتارا ہے؟ انہوں نے کہا: بھلائی۔ اہلِ تفسیر نے باریک فرق بتایا ہے کہ منکروں کا جواب "اساطیر" مرفوع تھا، گویا "وہ تو کہانیاں ہیں"، اور نزول کا اقرار اس میں نہ تھا؛ متقین کا جواب "خیرا" منصوب ہے، یعنی "رب نے بھلائی اتاری"، تو انہوں نے رب اور نزول دونوں کا اقرار کیا۔ ایک ہی سوال پر ایک دل نے داستان سنی اور دوسرے نے خیر۔ لِلَّذِيْنَ اَحْسَنُوْا فِيْ هٰذِهِ الدُّنْيَا حَسَنَةٌ۔ جنہوں نے اس دنیا میں احسان کیا ان کے لیے بھلائی ہے۔ حدیثِ احسان کا مفہوم یہ ہے کہ اللہ کی عبادت ایسے کرو گویا اسے دیکھ رہے ہو، اور اگر تم اسے نہیں دیکھتے تو وہ تمہیں دیکھ رہا ہے۔ شیخِ اکبر کے نزدیک احسان کا کمال اس وقت کھلتا ہے جب بندہ اپنی ہستی سے فانی ہو جائے، کیونکہ دیکھنے میں رکاوٹ خود دیکھنے والے کی "میں" ہے۔ وَلَدَارُ الْاٰخِرَةِ خَيْرٌ ۭ وَلَنِعْمَ دَارُ الْمُتَّقِيْنَ۔ اور آخرت کا گھر کہیں بہتر ہے، اور کیا ہی خوب گھر ہے متقین کا۔ جَنّٰتُ عَدْنٍ يَّدْخُلُوْنَهَا تَجْرِيْ مِنْ تَحْتِهَا الْاَنْهٰرُ۔ ہمیشگی کے باغ جن میں وہ داخل ہوں گے، ان کے نیچے نہریں بہتی ہیں۔ "عدن" کے معنی ٹھہرنے اور قیام کے ہیں؛ اہلِ معرفت اس میں تمکین کا اشارہ دیکھتے ہیں، وہ حال جس میں دل کا ٹھہراؤ پھر نہیں ڈگمگاتا۔ لَهُمْ فِيْهَا مَا يَشَآءُوْنَ ۭ كَذٰلِكَ يَجْزِي اللّٰهُ الْمُتَّقِيْنَ۔ وہاں ان کے لیے وہ سب ہے جو وہ چاہیں؛ اسی طرح اللہ متقین کو بدلہ دیتا ہے۔ اس مکتب میں اشارہ یہ ہے کہ جس نے دنیا میں اپنی چاہت حق کی چاہت میں فنا کر دی، آخرت میں اس کی چاہت ہی عطا کی صورت بن جاتی ہے۔

الَّذِيْنَ تَتَوَفّٰىهُمُ الْمَلٰۤئِكَةُ طَيِّبِيْنَ ۙ يَقُوْلُوْنَ سَلٰمٌ عَلَيْكُمُ۔ وہ جن کی روحیں فرشتے اس حال میں قبض کرتے ہیں کہ وہ پاکیزہ ہوتے ہیں، کہتے ہیں: تم پر سلامتی ہو۔ ادْخُلُوا الْجَنَّةَ بِمَا كُنْتُمْ تَعْمَلُوْنَ۔ جنت میں داخل ہو جاؤ اپنے ان اعمال کے سبب جو تم کرتے تھے۔ دونوں گروہوں کے لیے فعل ایک ہے، "تتوفاهم الملائكة"، اور فرشتے بھی وہی؛ فرق صرف قبض کیے جانے والے کے حال میں ہے: ایک "ظالمی انفسہم" اور دوسرا "طیبین"۔ اہلِ معرفت کے نزدیک موت ایک آئینہ ہے جس میں بندہ وہی دیکھتا ہے جو وہ تھا؛ تجلی ایک ہے اور ظہور قابل کی استعداد کے مطابق ہے۔ "طیب" وہ دل ہے جو ذکر سے مطمئن ہوا، جیسے سورۂ رعد میں فرمایا کہ دلوں کا اطمینان اللہ کے ذکر میں ہے؛ اور ایسے دل کا استقبال "سلام" سے ہوتا ہے، جو حق کا ایک اسم بھی ہے۔ گویا جس نے دنیا میں اپنے آپ کو اسمِ السلام کا مظہر بنایا، موت کی گھڑی وہی اسم اس کا استقبال کرتا ہے۔ هَلْ يَنْظُرُوْنَ اِلَّاۤ اَنْ تَاْتِيَهُمُ الْمَلٰۤئِكَةُ اَوْ يَاْتِيَ اَمْرُ رَبِّكَ۔ کیا یہ لوگ اسی کے منتظر ہیں کہ ان کے پاس فرشتے آ جائیں یا تمہارے رب کا حکم آ پہنچے؟ خطاب حضورِ اکرم صلی اللہ علیہ وسلم سے ہے اور "ربک" کی اضافت میں تسلی ہے۔ سورہ کی پہلی آیت نے کہا تھا کہ اللہ کا امر آ چکا؛ اہلِ ذوق کہتے ہیں کہ امر تو ہر آن حاضر ہے، منکر اسے دیکھنے کے لیے موت کی گھڑی کا انتظار کرتا ہے۔

كَذٰلِكَ فَعَلَ الَّذِيْنَ مِنْ قَبْلِهِمْ ۭ وَمَا ظَلَمَهُمُ اللّٰهُ وَلٰكِنْ كَانُوْۤا اَنْفُسَهُمْ يَظْلِمُوْنَ۔ ان سے پہلے والوں نے بھی یہی کیا تھا؛ اور اللہ نے ان پر ظلم نہیں کیا، بلکہ وہ خود اپنی جانوں پر ظلم کرتے تھے۔ یہ آیت اس سورہ میں ہدایت و ضلالت کے باب کا اصولِ حاکم ہے، اور جو کچھ اس سے پہلے اور بعد میں آتا ہے اسی کی روشنی میں پڑھا جائے گا۔ اصول یہ ہے: اللہ کسی کو زبردستی گمراہ نہیں کرتا اور جو ہدایت طلب کرے اسے گمراہ نہیں کرتا؛ جو طلب کرتا ہے وہ پاتا ہے، اور جو منہ موڑتا ہے اس کا منہ موڑنا ہی اس کی گمراہی ہے، اور حق اسی رخ کو وجود دیتا ہے جو بندے نے خود چنا۔ عربی ترکیب میں "انفسہم" کو فعل سے پہلے لایا گیا تاکہ حصر پیدا ہو: ظلم کا نشانہ انہی کی جانیں تھیں، کوئی اور نہیں۔ شیخِ اکبر کے نزدیک یہی سرِّ قدر کی کنجی ہے کہ حکم محکوم کے مطابق ہے؛ حق نے ان کے اعیان کو وہی دیا جو ان کی حقیقت نے مانگا، اس لیے شکوے کی کوئی جگہ نہیں۔ سورۂ ابراہیم میں شیطان نے اقرار کیا تھا کہ اسے کسی پر زور حاصل نہ تھا؛ یہاں حق سبحانہ خود اعلان فرماتا ہے کہ ظلم اس کی جانب سے نہیں۔ فَاَصَابَهُمْ سَيِّاٰتُ مَا عَمِلُوْا وَحَاقَ بِهِمْ مَّا كَانُوْا بِهٖ يَسْتَهْزِءُوْنَ۔ پس ان کے اعمال کی برائیاں انہی پر آ پڑیں، اور جس چیز کا وہ مذاق اڑاتے تھے اسی نے انہیں گھیر لیا۔ یہاں سزا کو "سیئات ما عملوا" کہا گیا، یعنی ان کے اپنے اعمال ہی لوٹ کر آئے۔ اس مکتب میں جزا عمل ہی کی صورت ہے جو آخرت میں ظاہر ہوتی ہے؛ اور جس حقیقت کا مذاق اڑایا گیا تھا وہی گھیرنے والی بن گئی، کیونکہ حقیقت سے بچ کر جانے کی کوئی جگہ نہیں۔
