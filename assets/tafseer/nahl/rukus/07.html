وَقَالَ اللّٰهُ لَا تَتَّخِذُوْٓا اِلٰهَيْنِ اثْنَيْنِ ۚ اِنَّمَا هُوَ اِلٰهٌ وَّاحِدٌ ۚ فَاِيَّايَ فَارْهَبُوْنِ۔ اور اللہ نے فرمایا: دو معبود نہ بناؤ، وہ تو بس ایک ہی معبود ہے، سو صرف مجھی سے ڈرو۔ وَلَهٗ مَا فِي السَّمٰوٰتِ وَالْاَرْضِ وَلَهُ الدِّيْنُ وَاصِبًا ۭ اَفَغَيْرَ اللّٰهِ تَتَّقُوْنَ۔ اور اسی کا ہے جو کچھ آسمانوں اور زمین میں ہے، اور اطاعت ہمیشہ اسی کی واجب ہے؛ تو کیا اللہ کے سوا کسی اور سے ڈرتے ہو؟ ظاہر میں یہ ان لوگوں کا رد ہے جو نور و ظلمت یا خیر و شر کے دو الگ خدا مانتے تھے، اور اشاری معنی اس ظاہر کی نفی نہیں، اس کی گہرائی ہے۔ "اِلٰهَيْنِ" کے بعد "اثْنَيْنِ" کی تاکید پر اہلِ معرفت ٹھہرتے ہیں: گویا منع صرف دو بتوں سے نہیں، خود دوئی سے ہے۔ شیخِ اکبر محی الدین ابنِ عربی رحمۃ اللہ علیہ کے نزدیک شرکِ خفی یہ ہے کہ بندہ حق کی ہستی کے پہلو میں اپنی ایک مستقل ہستی بھی دیکھے، گویا ایک وجود وہ ہے اور ایک میں۔ اس مکتب میں بندے کا وجود ظل ہے، اصل نہیں، اور ظل کو اصل کا ہمسر ٹھہرانا ہی باریک ترین "دو الٰہ" بنانا ہے۔ پھر اسلوب دیکھیے: بات "هُوَ" یعنی غائب کے صیغے سے چلی اور یکایک "فَاِيَّايَ" یعنی متکلم پر آ ٹھہری۔ اہلِ ذوق کہتے ہیں کہ یہ غیبت سے حضوری کا سفر ہے؛ جو "وہ ایک ہے" کو دل سے مان لے، اسے "مجھ سے ڈرو" کا خطاب بے واسطہ سنایا جاتا ہے۔ اور "وَاصِبًا" بتاتا ہے کہ دین لمحاتی نہیں، دائمی ہے؛ اس مکتب میں ہر ذرہ اپنی تکوینی اطاعت میں ہر آن مصروف ہے، جیسا کہ سورۂ رعد میں سایوں کے سجدے کا ذکر گزرا۔ پس جو خوف غیر کی طرف جائے، وہ اسی دوئی کا دوسرا چہرہ ہے۔

وَمَا بِكُمْ مِّنْ نِّعْمَةٍ فَمِنَ اللّٰهِ ثُمَّ اِذَا مَسَّكُمُ الضُّرُّ فَاِلَيْهِ تَجْئَرُوْنَ۔ اور تمہارے پاس جو بھی نعمت ہے وہ اللہ ہی کی طرف سے ہے، پھر جب تمہیں تکلیف چھوتی ہے تو اسی کے آگے گڑگڑاتے ہو۔ ثُمَّ اِذَا كَشَفَ الضُّرَّ عَنْكُمْ اِذَا فَرِيْقٌ مِّنْكُمْ بِرَبِّهِمْ يُشْرِكُوْنَ۔ پھر جب وہ تکلیف تم سے ہٹا دیتا ہے تو اچانک تم میں سے ایک گروہ اپنے رب کے ساتھ شریک ٹھہرانے لگتا ہے۔ لِيَكْفُرُوْا بِمَآ اٰتَيْنٰهُمْ ۭ فَتَمَتَّعُوْا ۣ فَسَوْفَ تَعْلَمُوْنَ۔ نتیجہ یہ کہ ہمارے دیے کی ناشکری کریں؛ سو چند روز برت لو، عنقریب جان لو گے۔ "مِنْ نِّعْمَةٍ" میں نکرہ ہر نعمت کو سمیٹ لیتا ہے، اور اہلِ معرفت کے نزدیک پہلی نعمت خود وجود ہے، جو بندے کے پاس امانت ہے، ملکیت نہیں۔ "تَجْئَرُوْنَ" کا مادہ بیل کے بلند ڈکرانے کے لیے آتا ہے؛ یعنی مصیبت میں آواز تکلف سے خالی، خام اور سچی ہو جاتی ہے۔ اس مکتب میں اضطرار وہ لمحہ ہے جس میں اسباب کے پردے گر جاتے ہیں اور فطرت اپنے اصل رب کو پکار اٹھتی ہے؛ گویا تکلیف فطرت کا آئینہ ہے۔ پھر "فَرِيْقٌ مِّنْكُمْ" کی قید میں بھی رحمت ہے: سب نہیں پلٹتے، ایک گروہ پلٹتا ہے۔ اور پلٹنا اکثر یوں ہوتا ہے کہ نجات کو دوا، طبیب یا اپنی تدبیر کے نام لکھ دیا جائے؛ اہلِ تاویل اسے شرکِ اسباب کہتے ہیں۔ "لِيَكْفُرُوْا" کا لام انجام بتاتا ہے: ان کی روش کا حاصل ناشکری نکلا۔ یہاں وہی اصول ہے جس پر یہ سلسلہ قائم ہے: حق نے فریاد سنی، تکلیف ہٹائی، اور منہ پھیرنا بندے کا اپنا انتخاب رہا؛ سرِّ قدر میں حکم محکوم کے اپنے حال کے مطابق ہے۔ "فَتَمَتَّعُوْا" مہلت ہے، اور "فَسَوْفَ تَعْلَمُوْنَ" بتاتا ہے کہ جو علم آج اختیار سے نہ لیا گیا وہ کل مشاہدے سے دیا جائے گا۔

وَيَجْعَلُوْنَ لِمَا لَا يَعْلَمُوْنَ نَصِيْبًا مِّمَّا رَزَقْنٰهُمْ ۭ تَاللّٰهِ لَتُسْئَلُنَّ عَمَّا كُنْتُمْ تَفْتَرُوْنَ۔ اور جن کی حقیقت وہ جانتے تک نہیں، ان کے لیے ہمارے دیے ہوئے رزق میں سے حصہ مقرر کرتے ہیں؛ اللہ کی قسم! جو بہتان تم باندھتے ہو اس کی تم سے ضرور پوچھ ہوگی۔ وَيَجْعَلُوْنَ لِلّٰهِ الْبَنٰتِ سُبْحٰنَهٗ ۙ وَلَهُمْ مَّا يَشْتَهُوْنَ۔ اور وہ اللہ کے لیے بیٹیاں ٹھہراتے ہیں، وہ اس سے پاک ہے! اور اپنے لیے وہ جو ان کا جی چاہے۔ ظاہر میں مشرکین کھیتی اور مویشیوں کا ایک حصہ بتوں کے نام کر دیتے اور فرشتوں کو اللہ کی بیٹیاں کہتے تھے۔ یہاں دو خطائیں جمع ہیں۔ پہلی عقیدے کی: حق کے لیے اولاد ماننا وہ تشبیہ ہے جس میں تنزیہ نہ ہو، اسی لیے بیچ میں "سُبْحٰنَهٗ" آ گیا، جیسے تنزیہ نے فوراً اس گمان کا دامن تھام لیا۔ دوسری خطا اخلاق کی ہے، اور اسے غور سے سمجھنا چاہیے: قرآن یہ نہیں کہتا کہ بیٹی کمتر ہے؛ وہ یہ دکھاتا ہے کہ ان لوگوں نے بیٹی کو اپنے ترازو میں کمتر سمجھا، پھر وہی کمتر گمان کی ہوئی چیز اللہ کی طرف منسوب کر دی۔ ملامت ان کے ترازو پر ہے، بیٹی پر نہیں۔ "لِمَا لَا يَعْلَمُوْنَ" پر اہلِ تاویل کہتے ہیں کہ غیر کی کوئی ایسی حقیقت ہی نہیں جو جانی جا سکے، اس لیے اس کے نام کا ہر حصہ گویا خلا میں ڈالا گیا۔ شیخِ اکبر کے نزدیک رزق صرف روٹی نہیں، توجہ اور علم بھی رزق ہے؛ دل کی توجہ کا ایک حصہ غیرِ حق کے نام کر دینا وہ پوشیدہ نذر ہے جو آج بھی چڑھائی جاتی ہے۔ اور "لَتُسْئَلُنَّ" میں بھی ایک عزت چھپی ہے: سوال اسی سے ہوتا ہے جسے اختیار دیا گیا ہو؛ پتھر سے کوئی نہیں پوچھتا۔

وَاِذَا بُشِّرَ اَحَدُهُمْ بِالْاُنْثٰى ظَلَّ وَجْهُهٗ مُسْوَدًّا وَّهُوَ كَظِيْمٌ۔ اور جب ان میں سے کسی کو بیٹی کی خوشخبری دی جاتی ہے تو اس کا چہرہ دن بھر سیاہ پڑا رہتا ہے اور وہ اندر ہی اندر غم سے گھٹتا ہے۔ يَتَوَارٰى مِنَ الْقَوْمِ مِنْ سُوْٓءِ مَا بُشِّرَ بِهٖ ۭ اَيُمْسِكُهٗ عَلٰى هُوْنٍ اَمْ يَدُسُّهٗ فِي التُّرَابِ ۭ اَلَا سَآءَ مَا يَحْكُمُوْنَ۔ جس کی اسے بشارت دی گئی اسے (اپنے گمان میں) برا جان کر لوگوں سے چھپتا پھرتا ہے: آیا اسے ذلت سہہ کر رکھ لے یا مٹی میں دبا دے؟ سن لو! بہت ہی برا ہے جو فیصلہ وہ کرتے ہیں۔ پہلی بات لفظ میں ہے: قرآن بیٹی کی ولادت کو "بُشِّرَ" کہتا ہے، یعنی خوشخبری۔ لفظ حق کا ہے اور وہ بشارت ہے؛ اداسی ان کی اپنی ہے۔ دوسری بات یہ کہ سیاہی بیٹی کے چہرے پر نہیں، اسے حقیر جاننے والے کے چہرے پر اتری؛ قرآن نے داغ اس حقارت پر لگا دیا۔ تیسری بات نہایت لطیف ہے: آیت میں "سُوْٓءِ مَا بُشِّرَ بِهٖ" ان کے گمان کی حکایت ہے، اور آیت ختم ہوتے ہوتے وہی لفظ پلٹ کر "سَآءَ مَا يَحْكُمُوْنَ" بن جاتا ہے۔ گویا جو برائی وہ بیٹی میں دیکھتے تھے، قرآن نے اسے اٹھا کر ان کے فیصلے پر رکھ دیا۔ بیٹی کو زندہ دفن کرنا جاہلیت کا وہ ظلم تھا جسے اسلام نے جڑ سے مٹایا، اور سورۂ تکویر میں ہے کہ اس معصوم سے پوچھا جائے گا کہ وہ کس گناہ پر ماری گئی۔ حضورِ اکرم صلی اللہ علیہ وسلم نے، حدیث کے مفہوم کے مطابق، فرمایا کہ جو دو بیٹیوں کی پرورش کرے یہاں تک کہ وہ جوان ہو جائیں، وہ قیامت کے دن میرے ساتھ یوں ہوگا جیسے یہ دو انگلیاں ملی ہوئی ہیں۔ اور حضورِ اکرم صلی اللہ علیہ وسلم سیدہ فاطمہ رضی اللہ عنہا کے آنے پر کھڑے ہو جاتے اور انہیں اپنے جگر کا ٹکڑا فرماتے تھے۔

اہلِ معرفت اس مقام پر ادب سے رکتے ہیں، کیونکہ یہاں تاویل کو ظاہر کا خادم رہنا ہے، اس کا بدل نہیں بننا۔ اسی سورت میں آگے اعلان ہے کہ جو بھی نیک عمل کرے، مرد ہو یا عورت، اور وہ مومن ہو، اسے پاکیزہ زندگی عطا ہوگی؛ پس ولایت و قرب کے دروازے پر مرد و عورت کی کوئی تخصیص نہیں۔ شیخِ اکبر کے نزدیک روحانی مراتب میں کوئی مرتبہ ایسا نہیں جو عورت کو حاصل نہ ہو سکے، اور انہوں نے اپنی رہنما عارفات کا ذکر نہایت ادب سے کیا ہے، جن میں اشبیلیہ کی ایک بزرگ خاتون کی خدمت کو وہ اپنی سعادت سمجھتے تھے۔ حدیثِ "حُبِّبَ اِلَیَّ" کی شرح میں شیخِ اکبر نے عورت کو شہودِ حق کے کامل مظاہر میں شمار کیا ہے، کیونکہ اس میں حق کی تخلیقی شان اور رحمت کی نرمی دونوں جلوہ گر ہیں۔ اس مکتب میں ایک تمثیل یہ بھی ہے کہ ساری کائنات حق کے سامنے قابلیت یعنی قبول کرنے کے مقام پر ہے، اور یہ قابلیت ہی ظہور کا دروازہ ہے؛ یہ محض تمثیل ہے جس میں کسی کی برتری یا کسی کی تحقیر کا کوئی دعویٰ نہیں، بلکہ یہ بتانا مقصود ہے کہ جو نسوانیت کو حقیر جانے وہ اس مقام کو حقیر جان رہا ہے جس پر ہر بندہ، مرد ہو یا عورت، اپنے رب کے آگے کھڑا ہے۔ اہلِ ذوق مزید کہتے ہیں کہ جو حق کی دی ہوئی بشارت کو مٹی میں دباتا ہے، وہ دراصل اپنے دل کی نرمی کو دفن کرتا ہے؛ مگر یہ اشارہ اس اٹل ظاہر کے بعد ہے کہ بیٹی خود ایک مکمل، باعزت اور مکرم ہستی ہے، اللہ کی نعمت اور گھر کی رحمت۔

لِلَّذِيْنَ لَا يُؤْمِنُوْنَ بِالْاٰخِرَةِ مَثَلُ السَّوْءِ ۚ وَلِلّٰهِ الْمَثَلُ الْاَعْلٰى ۭ وَهُوَ الْعَزِيْزُ الْحَكِيْمُ۔ جو آخرت پر ایمان نہیں رکھتے بری صفت انہی کی ہے، اور اللہ کی شان سب سے بلند ہے، اور وہی زبردست، حکمت والا ہے۔ غور کیجیے کہ اس رکوع میں لفظ "سوء" تیسری بار آیا: پہلے ان کے گمان میں، پھر ان کے فیصلے پر، اور اب ان کی اپنی مثال پر۔ قرآن نے برائی کا لفظ تین بار حقارت کرنے والوں پر لوٹایا اور ایک بار بھی بیٹی پر نہیں۔ "وَلِلّٰهِ الْمَثَلُ الْاَعْلٰى" اس مکتب میں تنزیہ و تشبیہ کے جمع کا بیان ہے۔ شیخِ اکبر کے نزدیک حق ہر مظہر میں جلوہ گر ہے، یہ تشبیہ ہے؛ اور کوئی مظہر اس کی ہمسری نہیں کرتا، یہ تنزیہ ہے؛ اور عارف وہ ہے جو دونوں کو ایک نگاہ میں رکھے۔ مشرکین نے تشبیہ کو بگاڑ کر اولاد تک پہنچا دیا، اور تنزیہ بھول گئے؛ "الْمَثَلُ الْاَعْلٰى" نے بتایا کہ جو صفت بھی کمال کی ہے اس کا سب سے اونچا درجہ اسی کا ہے، اور جو نقص ہے وہ اس سے پاک ہے۔ اسمائے الٰہیہ میں "الْعَزِيْزُ" وہ ہے جس تک رسائی نہ ہو، یعنی تنزیہ کی شان؛ اور "الْحَكِيْمُ" وہ جو ہر چیز کو اس کے مقام پر رکھے، یعنی ظہور کی ترتیب۔ یوں یہ رکوع "دو الٰہ نہ بناؤ" سے شروع ہو کر "الْمَثَلُ الْاَعْلٰى" پر ختم ہوا، اور ان دونوں کے بیچ بیٹی کی حرمت رکھی گئی؛ گویا قرآن نے بیٹی کی عزت کا دفاع توحید کے دفاع کا حصہ بنا دیا۔
